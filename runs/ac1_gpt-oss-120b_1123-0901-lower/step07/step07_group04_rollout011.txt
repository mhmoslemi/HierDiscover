def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4096   # maybe adjust to 8192 later
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # initial candidates
    candidates = _generate_initial_candidates(n)
    # evaluate fast objective, keep top few
    top_k = 5
    best_seq = None
    best_val = float('inf')
    # loop through candidates
    for cand in candidates:
        if time.time() >= deadline: break
        cand = cand / cand.sum()
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    # if no candidate, set uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)

    # Local refinement with gradient descent (softmax)
    # multiple restarts
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline: break
        # random perturb of best_seq
        a0 = best_seq.copy()
        # add random gaussian perturb in log space
        perturb = np.random.randn(n) * 0.1
        a0 = np.maximum(a0 * np.exp(perturb), 0.0)
        a0 = a0 / a0.sum()
        a_opt, val_opt = _adam_softmax_opt(a0, n, conv_fft_len, corr_fft_len, deadline, best_val)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"Improved to {best_val:.6f}")

    # Optionally upscale
    if time.time() < deadline - 30:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)
        # upsample using interpolation
        a_big = np.interp(np.linspace(0,1, n_big, endpoint=False), np.linspace(0,1, n, endpoint=False), best_seq)
        a_big = a_big / a_big.sum()
        a_opt_big, val_opt_big = _adam_softmax_opt(a_big, n_big, conv_fft_len_big, corr_fft_len_big, deadline, best_val)
        if val_opt_big < best_val:
            best_seq = a_opt_big
            best_val = val_opt_big
            n = n_big

    # Clip
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value approx: {best_val:.6f} (via approximate; actual evaluate later)")
    # Actual evaluation
    try:
        actual = evaluate_sequence(best_seq.tolist())
        print(f"Actual evaluate_sequence score: {actual}")
    except Exception as e:
        print(f"Error evaluating: {e}")
    return best_seq.tolist()

def _project_simplex(v):
    """Project vector v onto probability simplex (non-negative, sum=1)."""
    n = len(v)
    # shift v to ensure sum to 1 and non-negative
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _next_pow2(x: int) -> int:

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.

    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)

    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    while time.time() < deadline and it < 40000:
        # ----- convolution & current objective (fast) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = float(2.0 * n * max_c / (a.sum() ** 2))

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {cur_val:.6f}")

        # ----- soft‑max weights (smooth max) -----
        # scale by current temperature, subtract max for numerical stability
        scaled = conv * beta_soft
        scaled_max = np.max(scaled)
        w = np.exp(scaled - scaled_max)
        w /= w.sum()

        # ----- gradient of weighted sum of convolutions -----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)          # enforce non‑negativity & Σ=1

        # ----- temperature schedule (slow annealing) -----
        if it % 500 == 0 and beta_soft < max_beta:
            beta_soft = min(max_beta, beta_soft * 1.03)

        # ----- occasional random jitter to escape plateaus -----
        if time.time() - last_improve > 7.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(jitter)
            a = _project_simplex(a)
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimizes the
    evaluate_sequence objective.  The routine respects the provided time
    budget, prints progress, and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Base grid size – start with a medium resolution (fast enough)
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start (if a previous run stored a vector in the global
    # variable ``height_sequence_1``)
    # --------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        warm_start = ws
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a broad candidate pool and keep the top few
    # --------------------------------------------------------------
    all_candidates = _generate_candidates(n, warm_start)

    # Fast evaluation (FFT‑based) – keep the best 8
    evaluated = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        val = _fast_objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        if idx % 50 == 0:
            print(f" evaluated {idx+1}/{len(all_candidates)} candidates, best so far {min(v for v, _ in evaluated):.6f}")

    if not evaluated:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _fast_objective(best_seq, n, conv_fft_len)
        print("*** No candidates evaluated – using uniform vector.")
    else:
        evaluated.sort(key=lambda x: x[0])
        top_k = 8
        top_candidates = [c for _, c in evaluated[:top_k]]
        best_seq, best_val = evaluated[0]      # initialise with the absolute best

    # --------------------------------------------------------------
    # Stage 2 – local refinement with Adam (soft‑max surrogate)
    # --------------------------------------------------------------
    for start_id, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Refinement from start #{start_id+1}/{len(top_candidates)} (init val≈{_fast_objective(start_vec, n, conv_fft_len):.6f})")
        seq, val = _adam_opt(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            init_best_val=best_val,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f">>> Improvement after refinement: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – upscale to a finer grid (if time permits) and polish
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation onto the fine grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("--- Polishing on larger grid (n=8192)")
        seq_big, val_big = _adam_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            init_best_val=best_val,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = seq_big
            n = n_big
            print(f"+++ New best after up‑sampling: {best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanitisation and evaluation with the provided function
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"*** Final true evaluate_sequence score: {true_score:.6f}")
    except Exception as exc:
        print(f"*** evaluate_sequence raised an exception: {exc}")
        true_score = None

    # If for any reason the true score is worse than the best approximate,
    # we fall back to the uniform vector (guaranteed finite).
    if true_score is not None and true_score > best_val * 1.02:
        print("*** Detected possible over‑optimism – falling back to uniform.")
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
