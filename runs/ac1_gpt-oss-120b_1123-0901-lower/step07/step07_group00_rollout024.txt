def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]
```

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000, eta0=0.6):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)   # sum ~1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # optional print
        # active set
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        it += 1
    return best_a, best_val

"""
Search for a non‑negative step‑function (a vector of heights) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The strategy is:

1. Enumerate several low‑dimensional families of candidate shapes
   (U‑shaped Beta densities, left/right exponentials, power‑law edges,
   centred Gaussians) on a coarse grid.  For each candidate we
   normalise to L1=1 and evaluate the exact objective using an FFT‑based
   convolution.  The best candidate seen so far is kept.

2. Use the best candidate as a warm start for a cheap exponentiated‑gradient
   (sub‑gradient) optimiser that directly minimises the exact max‑convolution.
   The optimiser runs until the remaining time budget is exhausted.

3. Return the best vector (as a plain Python list).  The routine never
   raises an exception and always produces a non‑empty, non‑negative
   sequence, thus satisfying the evaluator’s security checks.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _eval_fast(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast exact evaluation assuming sum(a)==1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) minimisation of the exact
    max‑convolution objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set = indices attaining the max (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # gradient of the (exact‑max) objective
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule η_t = η₀ / √t
        eta = eta0 / np.sqrt(it + 1)

        # mirror‑descent update on the simplex
        a = a * np.exp(-eta * grad)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Return a non‑negative step‑function (list of floats) with a low
    evaluation score.  The routine respects the supplied time budget
    (≤ 1000 s) and always returns the best vector found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1.   Grid search over a few parametrised families
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    rev_i = n - 1 - i
    x = (i + 0.5) / n                       # points in (0,1)
    uniform = np.full(n, 1.0 / n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ---- helper to test a candidate ----------------------------------
    def _test_candidate(a: np.ndarray, label: str) -> None:
        nonlocal best_seq, best_val
        val = _eval_fast(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[grid {label}] new best {val:.6f}")

    # ---- Beta‑U shapes (U‑shaped Beta densities) --------------------
    for alpha in np.linspace(0.05, 0.95, 19):      # α in (0,1)
        # f(x) ∝ x^{α‑1} (1‑x)^{α‑1}
        shape = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
        if np.any(np.isnan(shape)) or shape.sum() == 0:
            continue
        shape /= shape.sum()
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            a = lam * uniform + (1 - lam) * shape
            _test_candidate(a, f"beta α={alpha:.3f} λ={lam:.2f}")

    # ---- Exponential edge families -----------------------------------
    for gamma in np.linspace(0.01, 4.0, 80):
        shape = np.exp(-gamma * i) + np.exp(-gamma * rev_i)
        shape /= shape.sum()
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            a = lam * uniform + (1 - lam) * shape
            _test_candidate(a, f"exp γ={gamma:.3f} λ={lam:.2f}")

    # ---- Power‑law edge families (prod^α) ---------------------------
    for power in np.linspace(-4.0, -0.2, 39):
        prod = (i + 1.0) * (n - i)
        shape = prod ** power
        if np.any(np.isnan(shape)) or shape.sum() == 0:
            continue
        shape /= shape.sum()
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            a = lam * uniform + (1 - lam) * shape
            _test_candidate(a, f"poly p={power:.2f} λ={lam:.2f}")

    # ---- Narrow Gaussian bumps at the centre -------------------------
    for sigma in (0.04, 0.06, 0.08, 0.10, 0.15, 0.20):
        # centred Gaussian
        centre = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((i - centre) / (sigma * n)) ** 2)
        shape /= shape.sum()
        for lam in (0.0, 0.5, 1.0):
            a = lam * uniform + (1 - lam) * shape
            _test_candidate(a, f"gauss σ={sigma:.2f} λ={lam:.2f}")

    # ---- Warm start (if a previous solution exists) -----------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() > 1e-12:
            w = w / w.sum()
            _test_candidate(w, "warm")

    # If, for any reason, nothing was found, fall back to uniform.
    if best_seq is None:
        best_seq = uniform.copy()
        best_val = _eval_fast(best_seq, n, conv_fft_len)

    # ------------------------------------------------------------------
    # 2.   Polishing with a long sub‑gradient optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            eta0=0.6,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[refine] sub‑gradient best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 3.   Final safety clamp & return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
