"""
Fast search for a low‑evaluation step function.

The evaluation criterion is

    E(a) = 2·n·max(a*a) / (sum(a)^2) ,

where ``a*a`` denotes the linear convolution of the coefficient vector
with itself.  Because the expression is invariant under scaling we can
always normalise the vector to have sum = 1 and minimise simply 2·n·max
conv.  The code builds a diverse pool of parametric candidates, keeps the
most promising few and then refines them with three complementary
optimisers:

1. Adam on a log‑sum‑exp surrogate of the max‑convolution (smooth‑max).
2. Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
3. Projected‑gradient descent with an exact gradient and Euclidean simplex
   projection.

All stages run until the global deadline (≈ 1000 s) is reached.  After
the deep optimiser a cheap stochastic hill‑climbing phase is performed
to squeeze out a few more digits.  The best sequence found is returned
as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast objective (assumes Σa ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # parameters of a soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    lam = 0.6                # start temperature
    lam_factor = 1.02        # slower schedule
    lam_cap = 1e8

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate evaluation ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 5e-3
            logits += jitter
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.55
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # ---- active set ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient update ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- schedule ----
        eta *= 0.9999
        it += 1

    return best_seq, best_val

def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the exact objective with explicit simplex projection.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.18
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.68
            if lr < 5e-6:
                break

        it += 1

    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Create a rich library of cheap parametric starts."""
    cand = []

    # warm‑start and a few cyclic shifts
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm_start, shift))

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # random Dirichlet draws
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # edge‑exponential family (possibly mixed with uniform)
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=16):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # mixtures of two Beta shapes (symmetric)
    for a1 in (0.5, 1.0, 2.0):
        for a2 in (0.5, 1.0, 2.0):
            for w in (0.2, 0.5, 0.8):
                beta1 = (i + 1) ** (a1 - 1) * (n - i) ** (a1 - 1)
                beta2 = (i + 1) ** (a2 - 1) * (n - i) ** (a2 - 1)
                shape = w * beta1 + (1.0 - w) * beta2
                shape = np.maximum(shape, 0.0)
                shape /= shape.sum()
                cand.append(shape)

    # a few handcrafted smooth shapes
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand.append(quad)

    # sinusoidal half‑wave (positive part)
    sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin = np.maximum(sin, 0.0)
    sin /= sin.sum()
    cand.append(sin)

    return cand

    """Very cheap stochastic perturbations + exact objective."""
    rng = np.random.default_rng()
    step = 0.02          # maximal share moved per trial

    while time.time() < deadline:
        # pick a source with non‑negligible mass
        src = rng.choice(n, p=a)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)

        delta = min(a[src] * step, (rng.random() * step))
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        conv = _full_convolution(a_new, conv_fft_len)
        val = 2.0 * n * conv.max() / (a_new.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[HillClimb] new best {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Main entry point – search for a step‑function with minimal score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # start from the previously discovered construction if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > 4096:
            warm = warm[:4096]
        else:
            warm = np.pad(warm, (0, 4096 - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # problem size and FFT lengths
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # generate a large pool of cheap candidates
    all_cand = _generate_parametric_candidates(n, warm)

    # quick evaluation, keep the best K
    K = 15
    scored = []
    for v in all_cand:
        if time.time() >= deadline:
            break
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-8:
            v = np.random.dirichlet(np.ones(n))
            s = v.sum()
        v = v / s
        val = _objective(v, n, conv_fft_len)
        scored.append((val, v))

    scored.sort(key=lambda x: x[0])
    top = [c for _, c in scored[:K]]

    # ------------------------------------------------------------------
    # allocate the remaining budget proportionally among the top starts
    remaining = deadline - time.time()
    if remaining <= 0:
        remaining = 0.1
    per_start = remaining / max(len(top), 1)

    best_seq = None
    best_val = float("inf")
    for idx, start in enumerate(top):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_start
        print(f"\n=== Optimising start {idx + 1}/{len(top)} (budget {per_start:.1f}s) ===")
        best_seq, best_val = _deep_optimize(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # final cheap stochastic hill‑climbing phase
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- Local hill‑climbing refinement ---")
        best_seq, best_val = _local_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # If time permits try a larger grid (8192) – copy the current best and refine
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))
        print("\n--- Optimising enlarged grid (n=8192) ---")
        sub_deadline = deadline
        a_big, val_big = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            _next_pow2(3 * n_big - 2),
            sub_deadline,
            best_seq,
            best_val,
        )
        if val_big < best_val:
            best_seq, best_val = a_big, val_big
            n = n_big

    # ------------------------------------------------------------------
    # fallback to uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # final safety clipping (evaluate_sequence will also clip)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
