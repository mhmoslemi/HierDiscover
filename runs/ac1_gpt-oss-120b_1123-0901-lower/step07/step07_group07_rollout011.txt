def _generate_candidates(n, warm_start):
    candidates = []
    # Warm start if provided
    if warm_start is not None:
        ws = warm_start.copy()
        if ws.shape[0] != n:
            # resample
            ws = np.interp(np.linspace(0,1,n,endpoint=False), np.linspace(0,1,ws.shape[0],endpoint=False), ws)
        ws = np.clip(ws, 0, None)
        ws_sum = ws.sum()
        if ws_sum < 0.01:
            ws = np.random.dirichlet(np.ones(n))
        else:
            ws = ws / ws_sum
        candidates.append(ws)
    # Uniform
    candidates.append(np.full(n, 1.0/n))
    # Random Dirichlet with various conc
    for conc in [0.01,0.05,0.1,0.2,0.5,1.0,2.0]:
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)
    # Random sparse
    for sparsity in [0.02,0.05,0.1]:
        k = max(2, int(sparsity * n))
        idx = np.random.choice(n, k, replace=False)
        a = np.zeros(n)
        a[idx] = np.random.rand(k)
        a = a / a.sum()
        candidates.append(a)
    # Random permutations of symmetric shapes (beta, exponential, power)
    # produce a symmetric shape then shuffle.
    i = np.arange(n, dtype=np.float64)
    # beta shape
    for alpha in [0.4,0.6,0.8,1.2,2.0,3.0]:
        shape = (i+1)**(alpha-1) * (n-i)**(alpha-1)
        shape = shape / shape.sum()
        perm = np.random.permutation(n)
        candidates.append(shape[perm])
    # exponential shape
    for alpha in np.logspace(-2, 1.0, num=7):
        shape = np.exp(-alpha * i) + np.exp(-alpha * (n-1-i))
        shape = shape / shape.sum()
        perm = np.random.permutation(n)
        candidates.append(shape[perm])
    return candidates
```

5. **Final clip** – clip to the evaluator’s allowed range and return the
   best list of floats.

All heavy work is performed with FFT‑based convolutions; the routine never
exceeds the allotted 1000 s budget.  No external solvers are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (sum(a) ≈ 1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     init_best_seq: np.ndarray | None,
                     init_best_val: float,
                     max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp (smooth‑max) surrogate.
    Returns the best (seq, value) observed.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # start in soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.4
    lam_factor = 1.02
    lam_cap = 1e6

    best_seq = init_best_seq if init_best_seq is not None else a0.copy()
    best_val = init_best_val

    last_improve = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        # soft‑max → probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate objective (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] AdamSmooth it={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        # jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
    return best_seq, best_val


def _adam_power_max(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    deadline: float,
                    init_best_seq: np.ndarray | None,
                    init_best_val: float,
                    max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a p‑norm surrogate of the max‑convolution.
    The exponent `p` is increased slowly from 4 up to 64.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    p = 4.0
    p_factor = 1.002
    p_max = 64.0

    best_seq = init_best_seq if init_best_seq is not None else a0.copy()
    best_val = init_best_val
    last_improve = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        # probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] AdamPower it={it:06d} p={p:.1f} new best {val:.6f}")

        # p‑norm weights (w_i ∝ conv_i^{p-1})
        # avoid NaNs for zero entries
        w = np.power(np.maximum(conv, eps), p - 1.0)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = p * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase p slowly
        p = min(p * p_factor, p_max)

        # jitter if stagnating
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float,
                            max_iter: int = 60_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (guarantees non‑negativity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        it += 1
    return best_seq, best_val


def _guided_local_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         deadline: float,
                         init_best_val: float,
                         max_iter: int = 15_000) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a small amount of mass from the
    index with largest sub‑gradient to the one with smallest sub‑gradient.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = init_best_val

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # recompute active‑set weights (sub‑gradient of the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i = np.argmax(grad_a)   # moving mass *away* from i tends to decrease max
        j = np.argmin(grad_a)   # moving mass *to* j tends to decrease max

        if i == j:
            break

        # transfer a modest fraction of the mass at i
        delta = min(a[i] * 0.25, 0.01)   # never move more than 0.01 total mass
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        conv_new = _full_convolution(a_new, conv_fft_len)
        max_c_new = conv_new.max()
        val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
            print(f"[{n}] GuidedMove it={it:04d} improved to {val_new:.6f}")
        else:
            # if no improvement, shrink delta or break
            delta *= 0.5
            if delta < 1e-9:
                break
        it += 1
    return best_seq, best_val


def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Run a cascade of optimisation phases (smooth‑max Adam → p‑norm Adam →
    sub‑gradient → guided local search) and update the global best.
    """
    # Phase 1 – smooth‑max Adam
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val,
    )
    # Phase 2 – p‑norm Adam
    best_seq, best_val = _adam_power_max(
        best_seq, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    # Phase 3 – exact sub‑gradient refinement
    best_seq, best_val = _subgradient_refinement(
        best_seq, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    # Phase 4 – guided local hill‑climb
    best_seq, best_val = _guided_local_search(
        best_seq, n, conv_fft_len, corr_fft_len, deadline,
        best_val,
    )
    return best_seq, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Create a diverse pool of candidate vectors of length `n`.
    All candidates are normalised to sum ≈ 1.
    """
    candidates: list[np.ndarray] = []

    # 1) warm‑start (if any) and a few cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        if ws.shape[0] != n:
            # resample by linear interpolation
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0.0, None)
        s = ws.sum()
        ws = ws / max(s, 1e-12)
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # 2) uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # 3) Dirichlet draws with many concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # 4) random sparse vectors
    for sparsity in (0.02, 0.05, 0.10):
        k = max(2, int(sparsity * n))
        idx = np.random.choice(n, k, replace=False)
        a = np.zeros(n)
        a[idx] = np.random.rand(k) + 0.1   # avoid exact zeros
        a /= a.sum()
        candidates.append(a)

    # 5) permuted symmetric families (beta, exponential, power‑law)
    i = np.arange(n, dtype=np.float64)

    # symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.6, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        perm = np.random.permutation(n)
        candidates.append(shape[perm])

    # edge exponential families
    for alpha in np.logspace(-2, 1.0, num=12):
        shape = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        shape /= shape.sum()
        perm = np.random.permutation(n)
        candidates.append(shape[perm])

    # edge power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        shape = (i + 1.0) ** (-p) + (n - i) ** (-p)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        perm = np.random.permutation(n)
        candidates.append(shape[perm])

    return candidates


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget,
    prints progress information and finally returns the best list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary grid size – start with 4096 (fast) and enlarge later if time permits
    initial_n = 4096
    sizes = [initial_n, 8192]   # later sizes are tried only if time remains

    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)

    for n in sizes:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimisation on grid n = {n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Stage 1 – generate & evaluate candidates
        cand_pool = _generate_candidates(n, warm_start)
        evaluated: list[tuple[float, np.ndarray]] = []
        for idx, cand in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 0.01:
                cand = np.random.dirichlet(np.ones(n))
                s = cand.sum()
            cand = cand / s
            val = _objective(cand, n, conv_fft_len)
            evaluated.append((val, cand))
        evaluated.sort(key=lambda x: x[0])
        top_k = min(8, len(evaluated))
        top_candidates = [vec for _, vec in evaluated[:top_k]]
        print(f"[{n}] Stage 1 selected {top_k} top starts (best ≈ {evaluated[0][0]:.6f})")

        # ------------------------------------------------------------------
        # Stage 2 – deep optimisation on each start
        for start_id, a0 in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(f"\n--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} (n={n}) ---")
            best_seq, best_val = _deep_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )

        # ------------------------------------------------------------------
        # Optional upscale to larger grid (if next size exists and time left)
        if n != sizes[-1]:
            # prepare warm_start for the next larger grid
            warm_start = best_seq
        # allow a small pause for GC, etc.
        time.sleep(0.05)

    # ------------------------------------------------------------------
    # Final sanity: ensure we have a feasible vector
    if best_seq is None:
        best_seq = np.full(initial_n, 1.0 / initial_n, dtype=np.float64)

    # Clip to evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Search complete: best objective ≈ {best_val:.6f} (n = {best_seq.shape[0]})")
    return best_seq.tolist()
