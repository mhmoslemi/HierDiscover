def _pgd_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, init_best_val):
   a = a0.copy()
   best = a.copy()
   best_val = init_best_val
   step = 0.2
   reduce_factor = 0.5
   no_improve = 0
   while time.time()<deadline:
       conv = _full_convolution(a, conv_fft_len)
       max_c = conv.max()
       val = 2.0 * n * max_c
       if val < best_val:
           best_val = val
           best = a.copy()
           no_improve=0
           # maybe increase step?
       else:
           no_improve+=1
       # compute active mask
       active = np.isclose(conv, max_c, atol=1e-12)
       # if many active maybe just top few? but ok
       w = np.zeros_like(conv)
       if active.any():
           w[active] = 1.0 / active.sum()
       # gradient
       grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
       # take step in negative gradient direction
       a_new = a - step * grad
       # projection onto simplex
       a_new = _project_simplex(a_new)
       # maybe adjust step adaptively
       # compute new objective
       conv_new = _full_convolution(a_new, conv_fft_len)
       max_c_new = conv_new.max()
       val_new = 2.0 * n * max_c_new
       if val_new > val:
           # step not improving, reduce step
           step *= reduce_factor
           # retain previous a (maybe)
       else:
           # accept
           a = a_new
           # maybe increase step a bit
           step = min(step*1.01, 1.0) # keep step bounded
       # check for stagnation
       if step < 1e-6:
           break
   return best, best_val

def _project_simplex(v):
    """Projects v onto the probability simplex (non-negative sum=1)."""
    # based on Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Improved step‑function search.

The core idea is a large parametric pool plus a very long
projected‑gradient / exponentiated‑gradient descent that works
directly on the exact max‑convolution (no smooth‑max surrogate).
We run many more sub‑gradient iterations than the previous
implementation, add occasional random perturbations to escape
plates, and perform a few random‑restart passes when time permits.
All vectors are kept on the simplex (non‑negative, sum = 1) so the
objective reduces to 2·n·max(a*a) and scaling does not matter.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a)  (the denominator is 1 after normalisation)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of the max‑convolution.
    The schedule is a bit slower (more iterations) than the original version.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in the log‑space (soft‑max later)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 2.0               # start temperature for the smooth‑max
    lam_factor = 1.006      # slower increase → smoother transition
    lam_cap = 1e6

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ----- soft‑max → probability vector -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate convolution (exact conv, smooth max later) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (chain‑rule) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient w.r.t logits (soft‑max derivative)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # shake if we are stuck for a while
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    The step‑size starts larger and decays slowly; occasional random log‑perturbations
    are injected when progress stalls.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                              # keep on the simplex

    eta = 1.2                           # initial step‑size
    decay = 0.99995                     # very slow decay
    best_seq = a.copy() if best_seq is None else best_seq
    best_val = best_val

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:07d} new best {val:.6f}")

        # ----- active set (indices attaining the maximum) -----
        # a tiny tolerance avoids missing nearly‑maximal entries
        tol = max(1e-12, max_c * 1e-12)
        active = np.isclose(conv, max_c, atol=tol, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # ----- gradient of the exact max -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- exponentiated‑gradient update (keeps non‑negativity) -----
        a = a * np.exp(-eta * grad_a)

        # ----- renormalise onto the simplex -----
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # ----- inject random log‑perturbation if we have stalled -----
        if time.time() - last_improve > 6.0:
            log_a = np.log(a + 1e-15)
            log_a += (np.random.rand(n) - 0.5) * 1e-3
            a = np.exp(log_a)
            a /= a.sum()
            last_improve = time.time()

    return best_seq, best_val


def _deep_optimize_stage(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run a long Adam phase followed by an extended sub‑gradient phase.
    Returns the best sequence/value seen in this stage.
    """
    # Adam (more iterations than the original code)
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iter=800_000,
    )
    # Sub‑gradient refinement (more iterations)
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        max_iter=300_000,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse pool of candidate vectors.
    Families include uniform, Dirichlet draws, symmetric/asymmetric
    exponentials, power‑laws, linear/triangular, sinusoidal, and sparse spikes.
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # ---- warm start & shifts ----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ---- uniform ----
    candidates.append(np.full(n, 1.0 / n))

    # ---- random Dirichlet draws (various concentrations) ----
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(12):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # ---- symmetric edge‑exponential & edge‑power‑law ----
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    for p in np.linspace(0.2, 6.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ---- asymmetric two‑sided exponentials ----
    alphas = np.logspace(-2, 1.0, num=10)
    w_vals = np.linspace(0.0, 1.0, 6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ---- asymmetric power‑law ----
    p_vals = np.linspace(0.2, 5.0, num=8)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ---- simple linear / triangular ----
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate(
        [np.linspace(0, 1, n // 2, endpoint=False), np.linspace(1, 0, n - n // 2)]
    )
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ---- sinusoidal half‑wave ----
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ---- sparse spike vectors (few non‑zeros) ----
    for spikes in (3, 5, 8):
        for _ in range(20):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: perturb the logits of the current best,
    keep changes that improve the *exact* objective.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.  Returns a plain Python
    list (the best sequence found before the deadline).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # 0.5 s safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem size – 4096 gave the previous best; we stick with it.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously found sequence, if present.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.size != n:
            if w.size > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.size))
        w = np.clip(w, 0.0, None)
        s = w.sum()
        if s >= 0.01:
            warm_start = w / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate and evaluate a large pool of cheap candidates.
    candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best={best_val:.6f}")

    # Keep the top‑K candidates for the expensive optimisation.
    cand_vals.sort(key=lambda x: x[0])
    top_k = min(12, len(cand_vals))
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts.
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize_stage(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – random‑restart sub‑gradient passes while time permits.
    while time.time() < deadline - 8.0:
        # generate a fresh random Dirichlet (bias towards spikier shapes)
        conc = np.random.choice([0.02, 0.05, 0.1, 0.2, 0.5])
        init = np.random.dirichlet(np.full(n, conc))
        init_val = _objective(init, n, conv_fft_len)
        if init_val < best_val:
            best_val = init_val
            best_seq = init.copy()
        # sub‑gradient refinement from this point
        best_seq, best_val = _subgradient_refinement(
            init,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            max_iter=200_000,
        )
        # if the remaining budget is tiny, break early
        if deadline - time.time() < 3.0:
            break

    # ------------------------------------------------------------------
    # Stage 4 – cheap stochastic hill‑climbing from the overall best.
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq, n, conv_fft_len, deadline, best_val
        )

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range and return.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
