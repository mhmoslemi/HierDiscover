def adam_optimize(a0, n, conv_fft_len, max_iters, deadline):
    eps = 1e-12
    logits = np.log(a0 + eps) # initial.

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8

    lr = 0.05

    lam = 0.1
    lam_factor = 1.02

    best_a = None; best_val = inf

    it = 0
    while it < max_iters and time.time() < deadline:
        # compute probabilities:
        max_logit = logits.max()
        exp_log = np.exp(logits - max_logit)
        a = exp_log / exp_log.sum()
        # compute conv & max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c # sum a =1 by design
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam] iter {it} best {val:.6f}")

        # compute smooth max weights:
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)
        # gradient wrt a:
        grad_a = 2.0 * _cross_correlation(w, a, conv_fft_len*2) # but need proper size, maybe corr_fft_len.
        # compute gradient wrt logits: chain rule:
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update:
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**(it+1))
        v_hat = v/(1-beta2**(it+1))

        logits -= lr * m_hat/(np.sqrt(v_hat)+eps_adam)

        lam = min(lam * lam_factor, 1e6)  # increase

        it += 1

    if best_a is None: best_a = a0
    return best_a, best_val

def subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, max_iters, deadline):
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = compute_objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2) # but a sum =1; it's 2n*maxc
        # Actually compute using objective function to ensure proper.
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # Compute subgradient with equal weight on active max positions
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = 0.5 / np.sqrt(it+1)  # step size decaying
        a = a * np.exp(-eta * grad_a)
        # reproject onto simplex:
        total = a.sum()
        if total <= 0:
            a = np.random.dirichlet(np.ones(n))
            total = a.sum()
        a = a / total

        it += 1

    return best_a, best_val

def simulated_annealing(a0, n, conv_fft_len, deadline, best_val):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max

    best_a = a.copy()
    best_val_local = cur_val

    start = time.time()
    it = 0
    # Base temperature setting:
    while time.time() < deadline:
        # compute fraction of time left:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        T = max(1e-7, 0.1 * (1-fraction))  # cooling
        move_scale = max(0.001, 0.2 * (1-fraction))

        # Choose i weighted by a (source)
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # choose j different uniformly
        j = np.random.randint(n-1)
        if j >= i: j += 1

        delta = a[i] * move_scale * np.random.rand()
        if delta <= 0:
            continue

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # Update conv efficiently
        # Subtract contributions from index i:
        # conv_new[i : i+n] -= 2*delta * a (old a values)
        conv_new[i : i + n] -= 2.0 * delta * a
        # Add contributions from index j:
        conv_new[j : j + n] += 2.0 * delta * a
        # delta^2 terms:
        # need to ensure indices within conv length
        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j
        if idx2i < conv_new.size:
            conv_new[idx2i] += delta * delta
        if idx2j < conv_new.size:
            conv_new[idx2j] += delta * delta
        if idxij < conv_new.size:
            conv_new[idxij] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val)/T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val_local:
                best_val_local = new_val
                best_a = a.copy()
                print(f"[SA] iter {it} new best {new_val:.6f}")

        it += 1
        if it % 2000 == 0:
            print(f"[SA] iter {it} best {best_val_local:.6f}")

    return best_a, best_val_local

def upscale(a_small, n_big):
    n_small = a_small.shape[0]
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_big = np.linspace(0, 1, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0)
    a_big /= a_big.sum()
    return a_big

"""
Hybrid multi‑scale optimizer for the step‑function problem.

The algorithm works in several stages:
1.  Small problem (n≈512) – many diverse random starts are
    optimized quickly with Adam on a smooth‑max (log‑sum‑exp)
    surrogate and a short exact‑max sub‑gradient pass.
2.  The best small solution is interpolated to a larger grid
    (2048 → 4096 → 8192) and refined again.  On each scale we
    run a longer Adam phase, an exact‑max sub‑gradient descent,
    and a fast O(n) simulated‑annealing local search.
3.  Random perturbations of the current best are occasionally
    re‑optimised to escape shallow basins.

All vectors are kept on the probability simplex (non‑negative,
unit sum) via the soft‑max parametrisation (logits) or explicit
renormalisation, so the objective
    F(a)=2·n·max(a∗a) / (sum a)^2
reduces to 2·n·max(a∗a) when sum(a)=1.

The routine respects the supplied time budget (default 1000 s) and
always returns a valid non‑negative list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of length len(a).
    Result index k corresponds to sum_i w[i+k] * a[i] (zero‑padded).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective from a convolution (assumes sum(a)=1)."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Convenient wrapper – works for any non‑negative a."""
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        return float("inf")
    a_norm = a / total
    n = a.shape[0]
    conv_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a_norm, conv_len)
    return _objective_from_conv(conv, n)


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors for a grid of size n."""
    cand = []

    # warm‑start and three rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform) and spiky Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))

    # Edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # arcsine (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # double‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # a few spiky patterns (k equally spaced spikes)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # smooth random shapes – Gaussian filtered noise
    for _ in range(4):
        rnd = np.random.randn(n)
        sigma = np.random.uniform(0.01, 0.05) * n
        # frequency domain Gaussian filter
        freqs = np.fft.rfftfreq(n)
        filt = np.exp(-0.5 * (freqs / (sigma / n)) ** 2)
        smooth = np.fft.irfft(np.fft.rfft(rnd) * filt)
        smooth -= smooth.min()
        if smooth.sum() > 0:
            smooth /= smooth.sum()
            cand.append(smooth)

    # uniform (fallback)
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp) with a slowly rising
    temperature `lam`.  The optimisation works on logits → soft‑max,
    guaranteeing non‑negativity and unit‑sum.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise logits
    m = np.zeros_like(logits)          # first moment
    v = np.zeros_like(logits)          # second moment

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                          # step size for logits
    lam = 0.05                         # start temperature
    lam_factor = 1.01                  # schedule factor

    best_a = a0.copy()
    best_val = _objective_from_conv(_full_convolution(best_a, conv_fft_len), n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = _objective_from_conv(conv, n)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing (actually *increase* to sharpen max)
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter near the deadline (helps escape plateaus)
        if it % 5000 == 0 and time.time() > deadline - 2.0:
            logits += (np.random.rand(n) - 0.5) * 1e-4

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with multiplicative (exponentiated‑gradient)
    updates.  The iterate stays on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a_sum

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = _objective_from_conv(conv, n)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {cur_val:.6f}")

        # active set – indices where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max(conv) w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient style update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a_sum

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """
    Fast O(n) simulated annealing that transfers a small amount of mass
    from a source index (chosen proportionally to its current weight)
    to a target index (uniformly random).  The convolution is updated
    incrementally.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = _objective_from_conv(conv, n)

    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)

        # linear cooling schedule
        T = max(1e-7, 0.12 * (1.0 - frac))
        move_scale = max(0.001, 0.25 * (1.0 - frac))

        # pick source index weighted by its mass
        if a.sum() == 0.0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] <= 1e-14:
            continue

        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * move_scale * np.random.rand()
        if delta <= 0.0:
            continue

        # propose new state
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution (linear‑time incremental update)
        conv_new[i:i + n] -= 2.0 * delta * a          # contribution of removed mass
        conv_new[j:j + n] += 2.0 * delta * a          # contribution of added mass

        # the quadratic delta² terms
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        if idx2i < conv_new.size:
            conv_new[idx2i] += delta * delta
        if idx2j < conv_new.size:
            conv_new[idx2j] += delta * delta
        if idxij < conv_new.size:
            conv_new[idxij] -= 2.0 * delta * delta

        new_val = _objective_from_conv(conv_new, n)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:06d}] best so far {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _upscale(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from a_small to length n_big, then normalise."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    total = a_big.sum()
    if total < 1e-12:
        a_big = np.full(n_big, 1.0 / n_big)
    else:
        a_big /= total
    return a_big

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``2·n·max(a∗a)/(sum a)^2``.
    The routine performs a multi‑scale optimisation consisting of
    Adam (smooth‑max), exact‑max sub‑gradient descent and a fast
    simulated‑annealing local search.  A warm‑start (global variable
    ``height_sequence_1``) is incorporated if available.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Helper to obtain a warm start of a given size (or None)
    def _get_warm(n_target: int) -> np.ndarray | None:
        warm = globals().get("height_sequence_1")
        if warm is None:
            return None
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_target:
            warm = warm[:n_target]
        elif warm.shape[0] < n_target:
            warm = np.pad(warm, (0, n_target - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s < 1e-12:
            return None
        return warm / s

    # ------------------------------------------------------------------
    # Stage 1 – small grid (n=512) -------------------------------------------------
    n_small = 512
    conv_fft_small = _next_pow2(2 * n_small - 1)
    corr_fft_small = _next_pow2(3 * n_small - 2)

    cand_small = _initial_candidates(n_small, _get_warm(n_small))

    best_seq = None
    best_val = float("inf")

    # Run a quick Adam + sub‑gradient on each start (time‑bounded)
    for cid, cand in enumerate(cand_small):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n_small))
        else:
            a0 = a0 / a0.sum()

        # allocate a tiny slice of the remaining budget for this candidate
        local_deadline = min(deadline - 0.5,
                             time.time() + min(9.0, (deadline - time.time()) / 2.0))

        a_adam, _ = _adam_optimize(a0, n_small,
                                   conv_fft_small, corr_fft_small,
                                   local_deadline, max_iters=200_000)

        val_adam = _objective(a_adam)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Stage 1 cand {cid}] Adam improved best to {val_adam:.6f}")

        # a short exact‑max sub‑gradient refinement
        if time.time() < deadline:
            a_sub, _ = _subgradient_optimize(best_seq, n_small,
                                             conv_fft_small, corr_fft_small,
                                             deadline, max_iters=80_000)
            val_sub = _objective(a_sub)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Stage 1 cand {cid}] Sub‑grad improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # If the small stage yields nothing (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # Stage 2 – upscale to 2048 ----------------------------------------------------
    if time.time() < deadline - 10.0:
        n_mid = 2048
        a_mid = _upscale(best_seq, n_mid)
        conv_fft_mid = _next_pow2(2 * n_mid - 1)
        corr_fft_mid = _next_pow2(3 * n_mid - 2)

        # longer Adam phase
        mid_deadline = min(deadline - 2.0, time.time() + 30.0)
        a_mid, _ = _adam_optimize(a_mid, n_mid,
                                  conv_fft_mid, corr_fft_mid,
                                  mid_deadline, max_iters=400_000)

        # sub‑gradient refinement
        if time.time() < deadline:
            a_mid, _ = _subgradient_optimize(a_mid, n_mid,
                                             conv_fft_mid, corr_fft_mid,
                                             deadline, max_iters=150_000)

        # fast SA
        if time.time() < deadline:
            a_mid, _ = _simulated_annealing(a_mid, n_mid,
                                            conv_fft_mid,
                                            deadline,
                                            best_val)

        best_seq = a_mid
        best_val = _objective(best_seq)
        print(f"[Stage 2] best value after up‑scale = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – upscale to 4096 ----------------------------------------------------
    if time.time() < deadline - 10.0:
        n_big = 4096
        a_big = _upscale(best_seq, n_big)
        conv_fft_big = _next_pow2(2 * n_big - 1)
        corr_fft_big = _next_pow2(3 * n_big - 2)

        # extensive Adam optimisation
        big_deadline = min(deadline - 2.0, time.time() + 40.0)
        a_big, _ = _adam_optimize(a_big, n_big,
                                  conv_fft_big, corr_fft_big,
                                  big_deadline, max_iters=600_000)

        # sub‑gradient descent
        if time.time() < deadline:
            a_big, _ = _subgradient_optimize(a_big, n_big,
                                             conv_fft_big, corr_fft_big,
                                             deadline, max_iters=250_000)

        # simulated annealing (lighter, because cost grows with n)
        if time.time() < deadline:
            a_big, _ = _simulated_annealing(a_big, n_big,
                                            conv_fft_big,
                                            deadline,
                                            best_val)

        best_seq = a_big
        best_val = _objective(best_seq)
        print(f"[Stage 3] best value after up‑scale = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional final upscale to 8192 ------------------------------------
    if time.time() < deadline - 20.0:
        n_huge = 8192
        a_huge = _upscale(best_seq, n_huge)
        conv_fft_huge = _next_pow2(2 * n_huge - 1)
        corr_fft_huge = _next_pow2(3 * n_huge - 2)

        # a final Adam pass (very long but inexpensive per iteration)
        huge_deadline = min(deadline - 2.0, time.time() + 45.0)
        a_huge, _ = _adam_optimize(a_huge, n_huge,
                                   conv_fft_huge, corr_fft_huge,
                                   huge_deadline, max_iters=800_000)

        # short sub‑gradient to polish
        if time.time() < deadline:
            a_huge, _ = _subgradient_optimize(a_huge, n_huge,
                                              conv_fft_huge, corr_fft_huge,
                                              deadline, max_iters=150_000)

        # final SA (very cheap – a few thousand moves)
        if time.time() < deadline:
            a_huge, _ = _simulated_annealing(a_huge, n_huge,
                                             conv_fft_huge,
                                             deadline,
                                             best_val)

        best_seq = a_huge
        best_val = _objective(best_seq)
        print(f"[Stage 4] final value (n={n_huge}) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a valid output – clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report the final objective using the judge's routine (if available)
    try:
        final_report = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_report:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
