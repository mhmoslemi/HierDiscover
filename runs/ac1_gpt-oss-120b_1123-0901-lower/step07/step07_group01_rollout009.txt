def _initial_candidates(n, warm):
    cand = []
    if warm is not None:
        cand.append(warm.copy())
        cand.append(np.roll(warm, n//2))
    # Uniform
    cand.append(np.full(n, 1.0/n))
    # Dirichlet small concentration (spiky)
    cand.append(np.random.dirichlet(0.1*np.ones(n)))
    # Dirichlet moderate
    cand.append(np.random.dirichlet(np.ones(n)))
    # Power-law edges
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1)
    for alpha in (-1.5, -1.0, -0.5):
        shape = np.where(prod>0, prod**alpha, 0.0)
        shape /= shape.sum()
        cand.append(shape)
    # Beta shapes random
    for _ in range(5):
        a = 10**np.random.uniform(-2, 1) # between 0.01 and 10
        b = 10**np.random.uniform(-2, 1)
        x = np.arange(1, n+1, dtype=np.float64)
        shape = (x**(a-1)) * ((n - x + 1)**(b-1))
        shape = np.maximum(shape, 0)
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            cand.append(shape)
    # Gaussian edges mixture
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.2, 0.4, 0.6):
        shape = np.exp(- (i**2) / (2*(sigma*n)**2)) + np.exp(- ((n-1-i)**2) / (2*(sigma*n)**2))
        shape /= shape.sum()
        cand.append(shape)
    # double-spike pattern
    for k in (128,256,512):
        a = np.zeros(n)
        pos = np.round(np.linspace(0, n-1, k)).astype(int)
        a[pos] = 1.0/k
        cand.append(a)
    return cand

def adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.03
    lam = 0.1 # start temperature low
    lam_factor = 1.02
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # Compute probability vector a
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # Convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it}] new best {val:.6f}")
        # Smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # Gradient wrt a (via cross-correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        lam = min(lam * lam_factor, 1e6)
        it += 1
    return best_a, best_val
```

def subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200000, eta0=0.6):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")
        # identify active indices where conv near max
        act = np.isclose(conv, max_c, atol=1e-12)
        if act.any():
            w = np.zeros_like(conv)
            w[act] = 1.0 / act.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)   # multiplicative
        # Renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val

def simulated_annealing(a0, n, conv_fft_len, deadline, cur_best_val):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    while time.time() < deadline:
        # cooling schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)
        T = max(1e-7, 0.1 * (1 - frac))
        move_scale = max(0.01, 0.3 * (1 - frac))
        # choose source by mass-proportional selection
        total_mass = a.sum()
        if total_mass <= 0:
            break
        i = np.random.choice(n, p=a/total_mass)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        delta = a[i] * move_scale * np.random.rand()
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # update convolution efficiently (linear)
        # conv = a * a via linear conv; moving delta from i to j changes contributions:
        # For each k, conv_new[k] = conv[k] - 2*delta*a[i - offset] + 2*delta*a_new[j - offset] + ...,
        # but simpler: recompute conv from scratch occasionally (say every 500 moves) else use incremental.
        # For simplicity and correctness, recompute conv fully each move (cost O(n log n) ) may be too heavy.
        # Instead we use O(n) update: conv_new = conv.copy()
        # For all positions p where a[i] participates:
        # conv[p+i] loses 2*delta*a[p] (since term a[i]*a[p]), similarly conv[p+j] gains 2*delta*a[p].
        # Since we have linear convolution, we can use vectorized updates.
        # We'll implement loop using slices.
        # Subtract contributions of i:
        conv_new[i:i+n] -= 2.0 * delta * a
        # Add contributions of j:
        conv_new[j:j+n] += 2.0 * delta * a
        # Adjust for the moved mass overlapping with itself:
        conv_new[2*i] += delta**2
        conv_new[2*j] += delta**2
        conv_new[i+j] -= 2.0 * delta**2
        new_max = conv_new.max()
        new_val = 2.0 * n * new_max
        # acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            conv = conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it}] new best {best_val:.6f}")
        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it}] best {best_val:.6f}")
    return best_a, best_val

Hybrid multiscale / evolutionary search for a step‑function that minimises

    F(a) = 2·n·max(a∗a) / (sum a)^2

The routine follows four phases:

1.  Generate a rich set of warm‑starts (uniform, Dirichlet,
    edge‑concentrated power‑law, beta‑shapes, Gaussian edges,
    spaced spikes and the optional previous best).
2.  For each start run a short Adam optimisation on a smooth‑max
    surrogate and a follow‑up exact‑max sub‑gradient descent.
3.  Refine the current global best with a fast O(n) simulated‑annealing
    walk.
4.  If time permits, up‑sample the best candidate to a finer grid
    (8192 points) and repeat phases 2–3 on the larger problem.

All work respects the supplied time budget (default 1000 s) and
returns a non‑negative vector that is already normalised to sum = 1,
so the internal value `2·n·max_c` equals the reported objective.
"""

def _next_pow2(x: int) -> int:

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:

    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

            cand.append(shape)

    # two‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.2, 0.5, 0.9):
        shape = np.exp(- (i ** 2) / (2 * (sigma * n) ** 2))
        shape += np.exp(- ((n - 1 - i) ** 2) / (2 * (sigma * n) ** 2))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # spaced‑spike families
    for k in (64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03               # modest learning rate
    lam = 0.1               # start temperature (smooth‑max sharpness)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # probabilities from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & objective (sum = 1 → denominator = 1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weighting
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                # ⟨a,∇⟩
        grad_log = a * (grad_a - avg)          # chain rule for softmax

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)
        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set of binding constraints (where conv ≈ max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size (mirrors classic sub‑gradient schedule)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update → stays on the simplex
        a = a * np.exp(-eta * grad_a)

        # projection onto the simplex (renormalisation)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast O(n) simulated‑annealing on the exact max."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # simple linear cooling
        T = max(1e-7, 0.12 * (1.0 - fraction))
        move_scale = max(0.01, 0.35 * (1.0 - fraction))

        # source index weighted by current mass
        mass = a.sum()
        if mass <= 0:
            break
        i = np.random.choice(n, p=a / mass)
        if a[i] < 1e-12:
            continue

        # distinct target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * move_scale * np.random.rand()      # amount to transfer

        # apply the transfer on copies (cheap O(n) convolution update)
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # linear update of the convolution
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _interpolate_and_refine(a_small: np.ndarray,
                            n_small: int,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Upscale to a larger grid and perform a short Adam + sub‑gradient pass."""
    # linear interpolation onto the finer grid
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # split the remaining budget roughly 60 % Adam / 40 % sub‑gradient
    remaining = deadline - time.time()
    if remaining < 5.0:
        return a_big, float("inf")
    t_adam = remaining * 0.6
    t_sub = remaining * 0.4

    a_adam, _ = _adam_optimize(a_big, n_big,
                               conv_fft_len_big,
                               corr_fft_len_big,
                               time.time() + t_adam,
                               max_iters=100_000)

    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           time.time() + t_sub,
                                           max_iters=120_000)
    return a_ref, val_ref

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `2·n·max(a∗a)/(sum a)^2`.  The routine respects the time budget,
    combines Adam, exact‑max sub‑gradient descent, simulated annealing
    and a final upscale to 8192 points.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size for the main search
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from the previous best (global variable)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # generate a pool of diverse starts
    candidates = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam + sub‑gradient on each candidate
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        # give each candidate a modest dedicated budget (≈15 s)
        local_deadline = min(deadline - 5.0, time.time() + 15.0)

        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len,
                                          corr_fft_len, local_deadline,
                                          max_iters=120_000)

        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam improved global best to {val_adam:.6f}")

        # a brief exact‑max refinement
        if time.time() < deadline:
            sub_deadline = min(deadline - 2.0, time.time() + 8.0)
            a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                   conv_fft_len,
                                                   corr_fft_len,
                                                   sub_deadline,
                                                   max_iters=80_000)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Sub‑grad improved global best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing refinement of the current best
    if best_seq is not None and time.time() < deadline - 5.0:
        a_sa, val_sa = _simulated_annealing(best_seq, n,
                                            conv_fft_len,
                                            deadline,
                                            best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] global best improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – upscale to 8192 points and repeat a short refinement
    if best_seq is not None and time.time() + 20.0 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(best_seq, n, n_big, deadline)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big] final best after up‑scaling {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce limits imposed by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final reporting (uses the judge's routine if present)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
