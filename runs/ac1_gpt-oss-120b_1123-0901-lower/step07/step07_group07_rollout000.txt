   * A short random‑mass‑transfer local search (still respecting symmetry)
     to squeeze out any remaining improvement.

All operations respect the evaluator's constraints (non‑negative,
clipping to [0,1000]).  The routine prints occasional progress messages and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT.
    ``fft_len`` must be ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce exact symmetry a[i] = a[-i‑1]."""
    return (a + a[::-1]) / 2.0


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of max(conv).
    ``a0`` must already be symmetric and sum to 1.
    Returns the best vector/value seen.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09                         # relatively aggressive learning rate
    lam = 0.3                         # initial temperature (smooth‑max)
    lam_factor = 1.008                # slower temperature increase
    lam_cap = 5e5

    it = 0
    max_iter = 250_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        a = _symmetrize(a)                # enforce symmetry each step

        # ----- evaluate -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d}  best {val:.6f}")

        # ----- smooth‑max weights -----
        # w_i ∝ exp(lam * (conv_i - max_c)), all values ≤ 1
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- gradient w.r.t. logits (soft‑max chain rule) -----
        avg = np.dot(a, grad_a)                # ⟨a,∇⟩ term
        grad_log = a * (grad_a - avg)          # shape (n,)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # maintain symmetry in the log‑space as well
        logits = (logits + logits[::-1]) / 2.0

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- jitter if stuck for a while -----
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 2e-2
            logits += jitter
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    The iterate stays on the simplex and is symmetrised after each step.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.65
    it = 0
    max_iter = 55_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:06d}  best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (diminishing)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # restore symmetry & normalisation
        a = _symmetrize(a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if no progress for a while
        if time.time() - last_improve > 8.0:
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Updates and returns the global best sequence/value.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse pool of symmetric candidate vectors (sum≈1).
    """
    candidates: list[np.ndarray] = []
    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1. Warm‑start (if provided) and its simple shifts (still symmetric)
    if warm_start is not None:
        ws = warm_start.copy()
        ws = _symmetrize(ws)
        ws = ws / ws.sum()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand = np.roll(ws, shift)
            cand = _symmetrize(cand)
            cand = cand / cand.sum()
            candidates.append(cand)

    # ------------------------------------------------------------------
    # 2. Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # 3. Dirichlet draws with many concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        cand = _symmetrize(cand)
        candidates.append(cand)

    # ------------------------------------------------------------------
    # 4. Exponential edge families (mixture with uniform)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge = _symmetrize(edge)
        edge = edge / edge.sum()
        for w in (0.0, 0.3, 0.6, 0.9, 1.0):
            cand = w * edge + (1.0 - w) / n
            cand = _symmetrize(cand)
            cand = cand / cand.sum()
            candidates.append(cand)

    # ------------------------------------------------------------------
    # 5. Power‑law edge families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = _symmetrize(edge)
        edge = edge / edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            cand = w * edge + (1.0 - w) / n
            cand = _symmetrize(cand)
            cand = cand / cand.sum()
            candidates.append(cand)

    # ------------------------------------------------------------------
    # 6. Symmetric beta‑shapes  a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1}
    for alpha in (0.2, 0.4, 0.6, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = _symmetrize(shape)
        shape = shape / shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # 7. Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = _symmetrize(shape)
        shape = shape / shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # 8. Two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = _symmetrize(shape)
        shape = shape / shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # 9. Arcsine‑type shape   a_i ∝ 1 / sqrt((i+0.5)*(n-i-0.5))
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = _symmetrize(shape)
    shape = shape / shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # Ensure all candidates are non‑negative and normalised.
    final = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        final.append(c / s)
    return final


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Try to reuse any previous solution that may be present
    warm_start = globals().get("height_sequence_1")
    if isinstance(warm_start, (list, tuple, np.ndarray)):
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            # resize by simple trunc/pad, preserve symmetry afterwards
            if ws.shape[0] > n:
                ws = ws[:n]
            else:
                ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = _symmetrize(ws)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    print(f"=== Stage 1 (n={n}) – generating candidates ===")
    candidates = _generate_candidates(n, ws)

    # Quick evaluation of all candidates
    evals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        evals.append((val, cand))
        # Optional debug print:
        # print(f"[cand {idx}]  val≈{val:.6f}")

    evals.sort(key=lambda x: x[0])
    top_k = min(8, len(evals))
    top_candidates = [c for _, c in evals[:top_k]]
    print(f"Stage 1: selected {top_k} promising starts (best≈{evals[0][0]:.6f})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on coarse grid
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} (n={n}) ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"After start #{start_id + 1}: global best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – upscale to finer grid if time permits
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 40.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation to finer grid (preserve symmetry)
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = _symmetrize(a_big)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print(f"--- Upscaling to n={n_big} and further optimisation ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Stage 4 – final random hill‑climb (short)
    if best_seq is not None and time.time() < deadline - 3.0:
        a = best_seq.copy()
        best_local = best_val
        it = 0
        max_iter = 10_000
        print("=== Final random hill‑climb refinement ===")
        while time.time() < deadline and it < max_iter:
            # propose a tiny random mass transfer between two independent indices
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # maintain symmetry: also affect mirrored positions
            mi, mj = n - 1 - i, n - 1 - j

            # choose a transfer amount limited by the smaller source
            src = a[i] + a[mi]
            dst = a[j] + a[mj]
            max_transfer = min(src, dst)
            if max_transfer < 1e-14:
                continue
            delta = (np.random.rand() - 0.5) * 0.4 * max_transfer   # up to ±20 % of smaller pair
            # apply to both members of each symmetric pair
            a[i] += delta / 2.0
            a[mi] += delta / 2.0
            a[j] -= delta / 2.0
            a[mj] -= delta / 2.0

            # enforce bounds
            a = np.maximum(a, 0.0)
            a = _symmetrize(a)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s

            # evaluate
            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max() / (a.sum() ** 2)

            if val < best_local:
                best_local = val
                best_seq = a.copy()
                # keep change
            else:
                # revert the move
                a[i] -= delta / 2.0
                a[mi] -= delta / 2.0
                a[j] += delta / 2.0
                a[mj] += delta / 2.0
                # ensure still valid and normalised
                a = np.maximum(a, 0.0)
                a = _symmetrize(a)
                a /= a.sum()
            it += 1

        best_val = best_local
        print(f"Hill‑climb finished after {it} attempts, best≈{best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing found (should never happen)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed range and return as plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete – final objective ≈ {best_val:.6f} (n={best_seq.shape[0]}) ***")
    return best_seq.tolist()
