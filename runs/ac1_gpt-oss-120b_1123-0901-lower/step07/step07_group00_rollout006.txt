"""
Propose a non‑negative step function that (empirically) improves the
evaluation bound for

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The search combines:

1. A very diverse pool of start vectors (Dirichlet with many
   concentration parameters, edge‑exponential/polynomial families,
   centred & double Gaussians, sinusoid, uniform and sparse spikes,
   plus random convex combinations of a small basis).
2. A short “soft‑max” Adam phase (log‑sum‑exp surrogate) for each
   start vector.
3. A long exponentiated‑gradient (sub‑gradient) phase on the exact
   max‑convolution, with a decreasing step size.
4. A simulated‑annealing hill‑climber that swaps a random fraction of
   mass between two coordinates.  The temperature schedule is
   exponential and the move size scales with the temperature.
5. (If time permits) a final refinement on a finer grid (8192 points).

All phases respect the global deadline (default 1000 s).  The function
always returns a valid sequence (non‑negative, ≤1000) and prints the
best value seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value 2·n·max(a∗a) / (sum(a)²)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors (no closures)."""
    cand = []

    # warm start and its cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Random Dirichlet with many concentration levels
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Edge‑exponential families (left/right)
    for gamma in np.logspace(-2, 1, 12):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian (different widths) – already present in the old code
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.1, 0.2, 0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Centered Gaussian
    mid = (n - 1) / 2.0
    for sigma in (0.05, 0.12, 0.25):
        shape = np.exp(-0.5 * ((i - mid) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Cosine bump (positive part of a cosine)
    shape = np.cos(np.pi * (np.arange(n) - mid) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates – equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


def _basis_vectors(n: int) -> list[np.ndarray]:
    """A small hand‑crafted basis for random convex combinations."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    basis = []

    # Uniform
    basis.append(np.full(n, 1.0 / n))

    # Half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    basis.append(s)

    # Centered Gaussian (sigma=0.15)
    sigma = 0.15
    g = np.exp(-0.5 * ((i - mid) / (sigma * n)) ** 2)
    g /= g.sum()
    basis.append(g)

    # Edge exponential (gamma=0.5)
    gamma = 0.5
    left = np.exp(-gamma * i)
    right = left[::-1]
    e = left + right
    e /= e.sum()
    basis.append(e)

    # Quadratic bump
    q = (i * (n - i - 1)) ** 0.8
    q /= q.sum()
    basis.append(q)

    return basis


def _random_convex_combination(basis: list[np.ndarray]) -> np.ndarray:
    """Create a random convex combination of the given basis vectors."""
    w = np.random.rand(len(basis))
    w /= w.sum()
    comb = np.zeros_like(basis[0])
    for coeff, vec in zip(w, basis):
        comb += coeff * vec
    comb /= comb.sum()
    return comb

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.07,
    lam0: float = 0.05,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  Returns the best
    vector observed (already normalised to sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    lam = lam0

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max projection onto the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp surrogate)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake when we are close to deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # mild learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on the simplex) using the
    exact max‑convolution sub‑gradient.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise (protect against under‑/overflow)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber that transfers a random
    fraction of mass from one coordinate to another.
    """
    a = a0.copy()
    cur_val = _objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = cur_val

    # temperature proportional to the current objective
    temp = cur_val * 0.02
    min_temp = cur_val * 1e-6
    decay = 0.99995

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # magnitude of the move adapts to the temperature
        max_frac = 0.25
        frac = max_frac * (temp / (cur_val * 0.02 + 1e-12))
        if frac > max_frac:
            frac = max_frac
        delta = a[i] * (np.random.rand() * frac)
        if delta <= 0:
            continue

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_fft_len)
        diff = val2 - cur_val
        if diff < 0 or np.random.rand() < np.exp(-diff / max(temp, 1e-12)):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[Anneal it={it:06d}] improved to {val2:.6f}")

        # geometric temperature decay
        temp = max(temp * decay, min_temp)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects the time budget and always returns the
    best sequence found so far (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build a generous pool of start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # add random convex combinations of a small basis (adds extra diversity)
    basis = _basis_vectors(n)
    for _ in range(12):
        pool.append(_random_convex_combination(basis))

    # keep only a manageable number (shuffle then slice)
    np.random.shuffle(pool)
    max_pool = 80
    pool = pool[:max_pool]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – short Adam optimisation for every candidate
    phase1_deadline = deadline - 300.0  # reserve ~5 min for later phases
    for cid, cand in enumerate(pool):
        if time.time() >= phase1_deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # short Adam (30 k iterations)
        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=30_000,
            lr=0.09,
            lam0=0.02,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deep sub‑gradient refinement of the current best
    if best_seq is not None and time.time() < deadline - 20:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=350_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climbing
    if best_seq is not None and time.time() < deadline - 15:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sample to a finer grid (8192) if time permits
    if best_seq is not None and time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
            lr=0.07,
            lam0=0.02,
            lam_factor=1.06,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=140_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # --------------------------------------------------------------
    # Fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
