def solve_lp_max_sum(f, M, n):
    # Using scipy.optimize.linprog
    # Variables: g[0..n-1]
    # Constraints: A_ub @ g <= b_ub where A_ub shape (2n-1, n)
    # b_ub = M * np.ones(2*n-1)
    # Also g >=0 automatically by bounds.
    # objective: minimize -np.ones(n) @ g

    # Build A_ub.
    # Build toeplitz matrix using numpy broadcasting: create f padded.
    # Approach: using scipy.linalg.toeplitz: efficient.

    from scipy.linalg import toeplitz
    c = np.concatenate([f, np.zeros(n-1)], dtype=np.float64) # (2n-1,)
    r = np.concatenate([f[:1], np.zeros(n-1)], dtype=np.float64)
    A = toeplitz(c, r)  # shape (2n-1, n)

    # A_ub = A, b_ub = M*ones

    # bounds: (0, None) for each variable.

    # Use linprog:
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # if fails, fallback to uniform
        return np.full(n, 1.0/n)
    g = res.x
    # Ensure non-negative (set negative to 0)
    g = np.maximum(g, 0)
    # Normalise (maybe keep original sum)
    # but we want to mix, not required.
    return g

def solve_lp_max_sum(f, M):
    n = f.shape[0]
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog

    c = -np.ones(n)  # maximize sum => minimize -sum
    col = np.concatenate([f, np.zeros(n-1)], dtype=np.float64)
    row = np.concatenate([f[:1], np.zeros(n-1)], dtype=np.float64)
    A_ub = toeplitz(col, row)  # shape (2n-1, n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # Use method='highs'.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        # fallback - a uniform distribution; or random
        g = np.full(n, 1.0 / n)
    # Clip negative small values
    g = np.maximum(g, 0.0)
    # To prevent scaling issues, normalise sum to 1
    s = g.sum()
    if s < 1e-10:
        g = np.full(n, 1.0/n)
    else:
        g = g / s
    return g
```

def mix_optimize(f0, n, deadline):

        val_g = compute_objective(g, n)

        # try pure g
        if val_g < best_val - 1e-9:
            f = g
            best_f = g
            best_val = val_g
            improved = True
            print(f"[Mix] iter {iteration} adopt g val {val_g:.6f}")
        else:

            for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):

def local_random_search(a, n, deadline, best_val):
    a_best = a.copy()
    val_best = best_val
    logits_best = np.log(a_best + 1e-15)
    while time.time() < deadline:
        noise = np.random.randn(n) * 0.02
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a_new = exp_log / exp_log.sum()
        val_new = compute_objective(a_new, n)
        if val_new < val_best:
            val_best = val_new
            a_best = a_new.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] new best {val_best:.6f}")
    return a_best, val_best

def compute_objective(a, n, fft_len=None):
    if fft_len is None:
        fft_len = 1 << ((2*n-2).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum()**2))

"""
Propose a low‑value step‑function.
Algorithm
---------
1.  Initialise a modest‑size vector (n_small = 1024).  If a previous
    construction (``height_sequence_1``) is available it is down‑sampled,
    otherwise a Dirichlet draw is used.
2.  **LP‑mixing phase** – repeat:
        •  compute M = max (f * f)
        •  solve a linear programme that maximises Σ g_j subject to
           the linear constraints (f * g) ≤ M and g ≥ 0
        •  either replace f by g or mix f←(1‑t)f + t g (t decreasing)
    The LP uses ``scipy.optimize.linprog`` (HiGHS) with a dense Toeplitz
    matrix; the loop stops when no improvement is found or the deadline
    is reached.
3.  Upsample the best small‑vector to the target size (n_big = 4096) by
    linear interpolation.
4.  **Stochastic refinement** – a log‑soft‑max hill‑climber perturbs the
    logits of the current best vector with Gaussian noise, evaluates the
    exact objective using an FFT‑based convolution and keeps any
    improvement.  The step‑size decays over time.
5.  Return the best vector (clipped to the evaluator’s limits).

The whole search is time‑aware; each stage checks the global deadline
(≈ 1000 s) and stops early if required.
"""

import time
import math
import numpy as np

# optional SciPy import – the LP phase degrades to a uniform start if unavailable
try:
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective  F = 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _max_convolution(a: np.ndarray) -> float:
    """Return max(a*a) using direct convolution (small vectors)."""
    return np.convolve(a, a).max()


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_j   s.t.   (f * g) ≤ M   and   g ≥ 0.
    Returns a feasible (and optimal) g, normalised to sum 1.
    """
    n = f.shape[0]
    if not _SCIPY_AVAILABLE:
        # fallback – uniform distribution (guaranteed feasible)
        g = np.full(n, 1.0 / n)
        return g

    # Build Toeplitz matrix that maps g ↦ f * g (full linear convolution)
    col = np.concatenate([f, np.zeros(n - 1)], dtype=np.float64)
    row = np.concatenate([f[:1], np.zeros(n - 1)], dtype=np.float64)
    A = toeplitz(col, row)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        g = np.maximum(res.x, 0.0)
    else:
        # If the LP fails, revert to a uniform vector – always feasible.
        g = np.full(n, 1.0 / n)

    s = g.sum()
    if s < 1e-12:
        g = np.full(n, 1.0 / n)
    else:
        g = g / s
    return g


def mix_optimize(f0: np.ndarray, n: int, deadline: float,
                 max_outer: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve a vector by the LP‑mixing step.
    Returns the best vector found and its objective value.
    """
    f = f0.copy()
    best_f = f.copy()
    best_val = _objective(f, n, _next_pow2(2 * n - 1))

    it = 0
    while time.time() < deadline and it < max_outer:
        M = _max_convolution(f)
        g = solve_lp_max_sum(f, M)

        # evaluate g
        val_g = _objective(g, n, _next_pow2(2 * n - 1))
        improved = False

        # direct replacement if g itself is better
        if val_g < best_val - 1e-12:
            f = g
            best_f = g
            best_val = val_g
            improved = True
            print(f"[Mix] iter {it:02d} adopted g, val = {val_g:.6f}")

        else:
            # try a few mixing factors
            for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
                cand = (1 - t) * f + t * g
                val_cand = _objective(cand, n, _next_pow2(2 * n - 1))
                if val_cand < best_val - 1e-12:
                    f = cand
                    best_f = cand
                    best_val = val_cand
                    improved = True
                    print(f"[Mix] iter {it:02d} mixed t={t:.3f}, val = {val_cand:.6f}")
                    break

        if not improved:
            # no further improvement – stop early
            break

        it += 1

    return best_f, best_val


def upsample_sequence(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from length len(seq_small) to n_big."""
    n_small = seq_small.shape[0]
    x_old = np.linspace(0, 1, n_small, endpoint=False)
    x_new = np.linspace(0, 1, n_big, endpoint=False)
    up = np.interp(x_new, x_old, seq_small)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s < 1e-12:
        up = np.full(n_big, 1.0 / n_big)
    else:
        up = up / s
    return up


def local_random_search(a_best: np.ndarray, n: int, deadline: float,
                        start_val: float = None) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing in log‑space.
    Perturb the logits with Gaussian noise, keep any exact improvement.
    """
    fft_len = _next_pow2(2 * n - 1)

    if start_val is None:
        best_val = _objective(a_best, n, fft_len)
    else:
        best_val = start_val

    logits_best = np.log(a_best + 1e-15)

    # step‑size schedule
    sigma = 0.1
    it = 0
    while time.time() < deadline:
        # decay the perturbation magnitude slowly
        sigma = max(0.003, sigma * 0.9999)

        noise = np.random.randn(n) * sigma
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, fft_len)
        if val < best_val - 1e-12:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] iter {it:06d} improved to {val:.6f}")

        it += 1

    return a_best, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point for the competition.
    Returns a list of non‑negative heights that (hopefully) beats the target bound.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem sizes
    n_big = 4096
    n_small = 1024

    # ------------------------------------------------------------------
    # Warm start handling (previous best construction, if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # bring to length n_big (interpolate / truncate)
        if warm.shape[0] != n_big:
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            warm_big = np.interp(x_new, x_old, warm)
        else:
            warm_big = warm.copy()
        warm_big = np.maximum(warm_big, 0.0)
        s = warm_big.sum()
        if s < 0.01:
            warm_big = np.full(n_big, 1.0 / n_big)
        else:
            warm_big = warm_big / s
        # down‑sample to the small grid for the LP phase
        f_warm_small = upsample_sequence(warm_big, n_small)
    else:
        f_warm_small = None

    # ------------------------------------------------------------------
    # Build a handful of start vectors for the LP‑mixing phase
    start_vectors = []

    if f_warm_small is not None:
        start_vectors.append(f_warm_small)

    # a few random Dirichlet draws (moderate concentration)
    for _ in range(4):
        vec = np.random.dirichlet(np.ones(n_small))
        start_vectors.append(vec)

    best_overall_seq = None
    best_overall_val = float("inf")

    # ------------------------------------------------------------------
    # LP‑mixing stage (runs on the small grid)
    for idx, f0 in enumerate(start_vectors):
        if time.time() >= deadline:
            break
        print(f"--- Mixing start #{idx + 1}/{len(start_vectors)} ---")
        f_opt, val_opt = mix_optimize(f0, n_small, deadline, max_outer=30)
        print(f"   Mixing result: val = {val_opt:.6f}")

        # Upsample to the target size
        f_big = upsample_sequence(f_opt, n_big)

        # Quick objective evaluation on the big grid
        val_big = _objective(f_big, n_big, _next_pow2(2 * n_big - 1))
        print(f"   Upsampled objective: {val_big:.6f}")

        if val_big < best_overall_val:
            best_overall_val = val_big
            best_overall_seq = f_big.copy()
            print(f"   New global best after up‑sampling: {val_big:.6f}")

    # ------------------------------------------------------------------
    # If the mixing phase produced nothing (unlikely), fall back to uniform
    if best_overall_seq is None:
        best_overall_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)
        best_overall_val = _objective(best_overall_seq, n_big,
                                      _next_pow2(2 * n_big - 1))

    # ------------------------------------------------------------------
    # Stochastic local refinement on the large grid
    print("--- Local stochastic refinement (big grid) ---")
    best_overall_seq, best_overall_val = local_random_search(
        best_overall_seq, n_big, deadline, start_val=best_overall_val
    )
    print(f"*** Finished search: best objective ≈ {best_overall_val:.6f} (n={n_big})")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence will also clip)
    final_seq = np.clip(best_overall_seq, 0.0, 1000.0)

    return final_seq.tolist()
