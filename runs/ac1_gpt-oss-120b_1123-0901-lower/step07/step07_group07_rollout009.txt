propose_candidate – search for a low‑value step‑function.

Algorithm
---------

   probability distributions (they are normalised to unit L¹‑norm) so that the
   evaluation function is scale‑invariant.

2. **Rich seed pool** –  a collection of deterministic shapes
   (one‑sided exponentials, power‑laws, linear / quadratic ramps,
   asymmetric beta‑shapes) together with random Dirichlet draws,
   their mirrors, and a warm‑start (if ``height_sequence_1`` exists).
   Each seed is evaluated with a cheap FFT‑based objective and the best

3. **Two‑phase continuous optimisation** for every kept seed

   *Phase A – Adam on a smooth‑max surrogate*  
   The vector is encoded as a soft‑max of free logits; the surrogate for
   ``max(conv)`` is the log‑sum‑exp with an increasing temperature λ.
   Adam with bias‑correction updates the logits.

   *Phase B – exact sub‑gradient (exponentiated‑gradient)*  
   Using the true max‑convolution we perform an exponentiated‑gradient step

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)²).  The vector is assumed to sum≈1."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    The vector is represented in the log‑soft‑max space (logits).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.4          # temperature – low → smooth, high → close to true max
    lam_factor = 1.02
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # probability vector on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate objective (log‑sum‑exp of the convolution)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (smooth surrogate)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule to logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature slowly
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        corr_fft_len: int,
        deadline: float,
        init_best_seq: np.ndarray | None,
        init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *true* max‑convolution.
    Works directly on the probability simplex.
    """
    a = init_best_seq if init_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    max_iter = 30_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # indices attaining the maximum (active set)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying multiplicative step
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numerical collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if no progress for a while
        if time.time() - last_improve > 6.0:
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        corr_fft_len: int,
        deadline: float,
        global_best_seq: np.ndarray | None,
        global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient → (optional second Adam) and keep the best.
    """
    # 1) Adam (smooth‑max)
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val,
    )
    # 2) Exact sub‑gradient
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    # 3) A short second Adam run with lower temperature to polish
    if time.time() < deadline - 2.0:
        best_seq, best_val = _adam_smooth_max(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            best_seq, best_val,
        )
    return best_seq, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse pool of seed vectors (all normalised to sum≈1).
    Mixes deterministic families and random draws.
    """
    candidates = []

    # Warm‑start (if supplied) and its reverse
    if warm_start is not None:
        ws = warm_start.copy()
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
            candidates.append(ws)
            candidates.append(ws[::-1])

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # One‑sided exponential families
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):      # 0.01 … 10
        exp_shape = np.exp(-alpha * i)
        exp_shape /= exp_shape.sum()
        for w in (0.0, 0.5, 0.9):
            cand = w * exp_shape + (1.0 - w) / n
            candidates.append(cand)

    # One‑sided power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        plaw = (i + 1.0) ** (-p)
        plaw /= plaw.sum()
        for w in (0.0, 0.5, 0.9):
            cand = w * plaw + (1.0 - w) / n
            candidates.append(cand)

    # Linear / quadratic ramps (decreasing)
    for power in (1, 2):
        ramp = (n - i) ** power
        ramp = np.maximum(ramp, 0.0)
        ramp /= ramp.sum()
        candidates.append(ramp)
        candidates.append(ramp[::-1])   # the opposite direction

    # Asymmetric beta‑shapes   a_i ∝ (i+1)^{α‑1}(n‑i)^{β‑1}
    for alpha, beta in ((0.2, 2.0), (0.4, 1.0), (1.0, 0.4), (2.0, 0.2)):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (beta - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Random permutations of symmetric families (to break symmetry)
    for _ in range(4):
        perm = np.random.permutation(n)
        rand_candidate = np.random.dirichlet(np.ones(n))
        candidates.append(rand_candidate[perm])

    # Sorted random Dirichlet (monotone) – both orders
    for _ in range(3):
        r = np.random.dirichlet(np.ones(n))
        candidates.append(np.sort(r)[::-1])   # decreasing
        candidates.append(np.sort(r))        # increasing

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best vector found (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Grid size – start with 8192; if enough time remains we will try 16384 later
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start (previous best) if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a large pool and retain the most promising seeds
    pool = _generate_candidates(n, warm_start)

    evaluated = []
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # optional progress print:
        # print(f"[seed {idx}] quick val ≈ {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    seeds = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} seeds (n={n})")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation on the selected seeds
    for start_id, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(seeds)} ---")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – if time permits, upscale to n=16384 and repeat optimisation
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=16384) ---")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val,
        )
        n, conv_fft_len, corr_fft_len = n_big, conv_fft_len_big, corr_fft_len_big

    # --------------------------------------------------------------
    # Stage 4 – final local random hill‑climb
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local_iter = 10000
        while time.time() < deadline and it < max_local_iter:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # propose a tiny transfer of probability mass
            max_transfer = min(a[i], a[j])
            if max_transfer < 1e-14:
                continue
            delta = (np.random.rand() - 0.5) * 0.4 * max_transfer  # up to ±20 %
            # keep positivity
            if delta > a[j]:
                delta = a[j]
            if -delta > a[i]:
                delta = -a[i]
            a[i] += delta
            a[j] -= delta

            # quick evaluation
            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max() / (a.sum() ** 2)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
            else:
                # revert the move
                a[i] -= delta
                a[j] += delta
            it += 1

        best_val = best_local_val
        print(f"Final local refinement finished after {it} attempts, best={best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform if everything failed
    if best_seq is None:
        best_seq = np.full(8192, 1.0 / 8192, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
