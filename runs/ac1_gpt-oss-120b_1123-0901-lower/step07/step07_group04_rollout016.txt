    # Initialize best

    best_val = float('inf')

    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:

            x_old = np.linspace(0,1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0,1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0)
        warm_sum = warm.sum()
        if warm_sum < 0.01:
            warm = None
        else:
            warm = warm / warm_sum
    else:
        warm = None
    
    # define function to run optimizer given init a
    def run_opt(init_a):
        # transform to logit x
        eps = 1e-12
        x = np.log(np.maximum(init_a, eps))
        # Adam states
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8

        beta = 1.0
        beta_inc = 1.001   # multiply per iteration

            a = np.exp(x - np.max(x))  # shift for stability
            a_sum = a.sum()
            a = a / a_sum

            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a_sum**2)  # a_sum = 1 so val = 2*n*max_c
            
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a.copy()
                last_improve_time = time.time()
                print(f"[n={n} iter={t}] new best {val:.6f}")
            
            # weights w via softmax
            # compute stable softmax
            conv_shift = conv - max_c  # subtract max for stability
            w = np.exp(beta * conv_shift)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.zeros_like(w)
                w[conv.argmax()] = 1.0
            else:
                w = w / w_sum
            
            # gradient wrt a
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

            grad_x = a * (grad_a - np.dot(grad_a, a))

            x = x - step

            # optional early stop if t huge...

       temperature β that is increased slowly.
     – Adam gives adaptive step‑sizes; occasional jitter escapes plateaus.
  3. If time permits, interpolate the best solution to a larger grid (n=8192)
     and run a short refinement with the same optimiser.
  4. Clip the final vector to the evaluator’s allowed range and return it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT of size fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (assumes a already normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Candidate generation (same families as the reference implementation)
def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Generate a diverse pool of initial vectors."""
    candidates = []

    # warm‑start and its rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet samples with a range of concentrations
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential mixtures
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # edge‑power‑law mixtures
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Adam optimiser that minimises a smooth approximation of max(conv)
def _adam_optimize(
    init_a: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_global_seq: np.ndarray | None,
    best_global_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Optimise a starting probability vector `init_a` using Adam on the softmax‑
    smoothed max‑convolution.  Returns the best vector/value seen during the
    run.
    """
    # initialise logits (log‑softmax variables)
    eps = 1e-12
    x = np.log(np.maximum(init_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.12                     # base learning rate

    # softmax temperature for the convolution (continuation)
    beta = 1.0                    # start with a smooth approximation
    beta_inc = 1.001              # geometric increase per iteration
    beta_max = 200.0

    last_improve = time.time()
    t = 0

    # keep track of the best observed vector locally
    best_seq_local = None
    best_val_local = float("inf")

    while time.time() < deadline:
        # ----- build probability vector a from logits -----
        a_exp = np.exp(x - np.max(x))   # stability shift
        a = a_exp / a_exp.sum()

        # ----- convolution and objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() == 1 ⇒ val = 2·n·max_c

        # record global best
        if val < best_global_val - 1e-14:
            best_global_val = val
            best_global_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={t:07d}] NEW GLOBAL BEST {val:.6f}")

        # record local best (used for the return values)
        if val < best_val_local - 1e-14:
            best_val_local = val
            best_seq_local = a.copy()

        # ----- compute soft‑max weights over the convolution -----
        # w mimics the sub‑gradient of max by focusing on large entries
        shift = conv - max_c               # subtract max for numeric stability
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fall back to the hard max (one‑hot)
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w /= w_sum

        # ----- gradient of the weighted sum -----
        # d/da Σ_k w_k conv_k = 2 * (w ★ a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the softmax a = softmax(x)
        grad_x = a * (grad_a - np.dot(grad_a, a))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        x = x - step

        # ----- temperature schedule -----
        beta = min(beta * beta_inc, beta_max)

        # ----- occasional jitter if we have stalled -----
        if time.time() - last_improve > 15.0:
            # random log‑space perturbation
            jitter = (np.random.rand(n) - 0.5) * 0.08
            x += jitter
            last_improve = time.time()
            print(f"[n={n} iter={t:07d}] jitter applied")

        t += 1

    return best_seq_local if best_seq_local is not None else best_global_seq, best_global_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Main entry point – searches for a low‑evaluation‑score step function."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Primary grid size – a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # resample / pad to required length
        if warm.shape[0] != n:
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s >= 0.01:
            warm = warm / s
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large, diverse pool and keep the best few
    candidates = _generate_parametric_candidates(n, warm)

    # add a few extra Dirichlet draws with a broader concentration sweep
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(15):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

    # quick evaluation of every candidate (FFT‑based)
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fall back to uniform – it is always valid
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("*** No candidates evaluated – returning uniform vector.")
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return best_seq.tolist()

    # keep the top‑k seeds
    cand_vals.sort(key=lambda p: p[0])
    top_k = 5
    seeds = [cand for _, cand in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of each seed (Adam + continuation)
    best_seq = None
    best_val = float("inf")
    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"--- Optimising seed #{idx + 1}/{len(seeds)} (init val={_objective(seed_vec, n, conv_fft_len):.6f})")
        best_seq, best_val = _adam_optimize(
            seed_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        # store the best result in a global name for possible warm‑starts next run
        globals()["height_sequence_1"] = best_seq.tolist()

    # ------------------------------------------------------------------
    # Stage 3 – if there is still time, upscale to a finer grid and refine
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("*** Refinement on larger grid (n=8192) ***")
        best_seq, best_val = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        best_val = _objective(best_seq, 4096, _next_pow2(2 * 4096 - 1))

    # Clip to the evaluation routine's allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
