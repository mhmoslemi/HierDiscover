"""
Annealed soft‑max mirror descent with momentum.

We search for a non‑negative step‑function (probability vector) a that
minimises

    F(a) = 2·n·max(a∗a) / (∑a)²

which is invariant to scaling, so we keep a on the simplex (∑a = 1).

The algorithm proceeds in three stages:

1.  Generate a diverse pool of parametric seeds (uniform, Dirichlet,
    edge‑exponential, power‑law, Gaussian‑edge, sinusoid) and, if
    available, a warm‑start from the previous run (global variable
    ``height_sequence_1``).
2.  Keep the best few seeds (by a cheap FFT‑based evaluation) and run a
    long‑running annealed mirror‑descent from each.  At iteration *t* we
    compute the full convolution, replace the hard max by a soft‑max
    with temperature *Tₜ*, obtain the gradient
        ∇F ≈ 2·corr(w, a)           with  wₖ ∝ exp(convₖ/Tₜ),
    apply a momentum term and an exponentiated‑gradient update
        a ← a·exp(−ηₜ·mₜ) .
    Both the temperature *Tₜ* and the step size ηₜ decay
    exponentially, while occasional jitter escapes plateaus.
3.  If time permits we up‑sample the best vector to a finer grid
    (n = 8192), perform a brief refinement with the same optimiser,
    and finally apply a short random‑perturbation local search.

All FFTs use powers‑of‑two lengths for speed.  The code respects the
1000 s budget, prints progress and returns the best sequence found as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (fast, based on FFT)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(a∗a) (sum(a) is assumed to be 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a heterogeneous list of seed vectors."""
    candidates = []

    # warm‑start and its circular shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniformly distributed seed
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet seeds with various concentrations
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential family
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law family
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Soft‑max gradient of the smoothed max‑convolution
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, conv: np.ndarray, temp: float, corr_fft_len: int) -> np.ndarray:
    """
    Gradient of the temperature‑smoothed max‑convolution.
    wₖ ∝ exp(convₖ / temp) and ∇ ≈ 2·corr(w, a).
    """
    # Stabilise exponentials
    maxc = conv.max()
    w = np.exp((conv - maxc) / temp)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
    return grad


# ----------------------------------------------------------------------
# Annealed exponentiated‑gradient with momentum
# ----------------------------------------------------------------------
def _annealed_expgd_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent on a temperature‑smoothed max‑convolution.
    Uses momentum on the logarithmic (multiplicative) updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # hyper‑parameters
    T0 = 0.8                 # initial temperature
    T_decay = 0.99996        # exponential decay per iteration
    T_min = 1e-4

    eta0 = 0.4               # initial step size (in log‑space)
    eta_decay = 0.99995
    eta_min = 5e-3

    beta = 0.9               # momentum coefficient
    mom = np.zeros_like(a)   # momentum buffer

    last_improve = time.time()
    it = 0
    max_iter = 10_000_000    # safety cap – loop stops by deadline

    while time.time() < deadline and it < max_iter:
        # full convolution and current objective value
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum()≈1

        # keep global best
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Anneal iter={it:07d}] new best {val:.6f}")

        # annealed temperature
        T = max(T_min, T0 * (T_decay ** it))

        # gradient of the smoothed max (soft‑max weights)
        grad = _grad_softmax(a, conv, T, corr_fft_len)

        # momentum on the gradient (in log‑space)
        mom = beta * mom + (1.0 - beta) * grad

        # step size
        eta = max(eta_min, eta0 * (eta_decay ** it))

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * mom)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # stagnation handling – occasional jitter
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # log‑space
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(30):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ------------------------------------------------------------------
    # Stage 2 – quick evaluation of all seeds, keep the best few.
    cand_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 8
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 3 – deep optimisation of the selected starts.
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Optimising start #{start_id + 1}/{len(top_candidates)} "
              f"(init val={init_val:.6f})")
        best_seq, best_val = _annealed_expgd_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – optional up‑sampling to a finer grid, then brief refinement.
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 0.01:
            a_big = np.random.dirichlet(np.ones(n_big))
        a_big = a_big / a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _annealed_expgd_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Stage 5 – tiny random‑perturbation local search (few milliseconds per trial).
    if best_seq is not None:
        attempts = 0
        while time.time() < deadline and attempts < 2000:
            # multiplicative log‑normal jitter
            jitter = np.random.randn(best_seq.size) * 0.015
            a_new = best_seq * np.exp(jitter)
            a_new = a_new.clip(0.0)
            if a_new.sum() < 0.01:
                a_new = np.random.dirichlet(np.ones(best_seq.size))
            a_new = a_new / a_new.sum()
            val = _objective(a_new, n, conv_fft_len if n == best_seq.size else _next_pow2(2 * n - 1))
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a_new.copy()
                print(f"[local perturb] improved to {val:.6f}")
            attempts += 1

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing succeeded.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
