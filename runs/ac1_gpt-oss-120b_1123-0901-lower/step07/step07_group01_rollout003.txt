    # Precompute product index arrays for param shapes

    prod = idx * (n - idx + 1.0)   # i+1 * (n - i)

    # main search loop
    best_seq = None

    while time.time() < deadline - 30:  # reserve for refinement

"""
Hybrid random‑parameter search + long Adam / sub‑gradient refinement.

* Sample simple symmetric families (Beta‑distribution, power‑law,
  mixture with uniform) on a grid of length 8192.
* Keep the best candidate (objective = 2·n·max(a∗a) because the vector is
  L1‑normalised).
* Refine the winner with Adam on a smooth‑max (log‑sum‑exp) surrogate,
  then with exact‑max sub‑gradient descent (exponentiated‑gradient).
* Finish with a fast O(n) simulated‑annealing local search.
All convolutions are performed with FFTs for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _beta_shape(n: int, beta: float) -> np.ndarray:
    """Symmetric Beta(beta,beta) density on a discrete grid (unnormalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = (i ** (beta - 1)) * ((n - i + 1) ** (beta - 1))
    return shape


def _powerlaw_shape(n: int, alpha: float) -> np.ndarray:
    """Shape proportional to (i·(n‑i+1))**alpha (unnormalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1)
    shape = prod ** alpha
    return shape


def _mix_with_uniform(shape: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture of `shape` with the uniform distribution."""
    return w * shape + (1.0 - w) * np.ones_like(shape)


# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Returns F(a) = 2·n·max(a∗a) / (sum a)^2.
    The vector `a` is assumed non‑negative; no normalisation is performed
    here, the function computes the exact value used by the judge.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04               # a slightly larger learning rate
    lam = 0.3               # initial temperature for smooth‑max
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c           # sum(a)==1, so denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # tiny jitter near the end to escape tiny plateaus
        if it % 5000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast O(n) SA that directly maintains the convolution during moves."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling
        T = max(1e-6, 0.12 * (1.0 - fraction))
        move_scale = max(0.005, 0.4 * (1.0 - fraction))

        # pick source proportional to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution incrementally (see original script)
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:06d}] current best {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
    2·n·max(a∗a)/(sum a)^2.  The method consists of:

    1. Random sampling of a few low‑dimensional families
       (Beta, power‑law, mixtures) on a grid of size 8192.
    2. Long Adam optimisation on a smooth‑max surrogate.
    3. Exact max sub‑gradient descent (exponentiated‑gradient).
    4. A short simulated‑annealing local search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # margin for safety

    # ------------------------------------------------------------------
    # Problem size – the larger the grid the finer the attainable shape.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start, if the platform already stored a previous best.
    warm = globals().get("height_sequence_1")
    best_seq = None
    best_val = float("inf")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            val = _objective_value(w, n, conv_fft_len)
            best_seq = w
            best_val = val
            print(f"[Warm start] value {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Random parametric search (takes most of the budget)
    print("[Stage 1] Random parametric search")
    while time.time() < deadline - 20.0:   # reserve ~20 s for refinements
        # Choose a family
        family = np.random.choice(3)
        if family == 0:
            # Beta family (symmetric)
            beta = np.random.uniform(0.1, 2.0)
            shape = _beta_shape(n, beta)
        elif family == 1:
            # Pure power‑law
            alpha = np.random.uniform(-2.0, -0.1)
            shape = _powerlaw_shape(n, alpha)
        else:
            # Mixture of power‑law and uniform
            alpha = np.random.uniform(-2.0, -0.1)
            w_mix = np.random.uniform(0.0, 1.0)
            shape = _mix_with_uniform(_powerlaw_shape(n, alpha), w_mix)

        # Occasionally blend a tiny amount of uniform mass
        if np.random.rand() < 0.12:
            eps = np.random.uniform(0.0, 0.04)
            shape = (1.0 - eps) * shape + eps * np.ones_like(shape)

        shape = np.maximum(shape, 0.0)
        total = shape.sum()
        if total < 1e-12:
            continue
        a = shape / total

        # Small random jitter to increase diversity
        if np.random.rand() < 0.18:
            noise = np.random.randn(n) * 5e-4
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()

        val = _objective_value(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

        # Occasionally try a pure Dirichlet sample (helps escape local traps)
        if np.random.rand() < 0.04:
            a_dir = np.random.dirichlet(np.ones(n))
            val_dir = _objective_value(a_dir, n, conv_fft_len)
            if val_dir < best_val:
                best_val = val_dir
                best_seq = a_dir.copy()
                print(f"[Dirichlet] new best {val_dir:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 10.0:
        print("[Stage 2] Adam refinement")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=800_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact max sub‑gradient descent
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Stage 3] Sub‑gradient refinement")
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=400_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Short simulated‑annealing local search
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[Stage 4] Simulated annealing")
        a_sa, val_sa = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_value(best_seq, n, conv_fft_len)

    # Clip to the limits enforced by the judge
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final reporting (uses the provided evaluate_sequence if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
