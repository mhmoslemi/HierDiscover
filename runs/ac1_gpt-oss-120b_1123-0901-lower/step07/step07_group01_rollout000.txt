"""
Hybrid search for a non‑negative step function minimizing
    F(a) = 2·n·max(a∗a) / (sum a)^2

The algorithm consists of several stages:

1. **Rich candidate pool** – many deterministic shapes (Beta,
   power‑law, edge‑Gaussian, sine, spikes) and random Dirichlet draws.
   Small jittered copies are added to increase diversity.

2. **Adam on a smooth‑max surrogate** – each candidate is refined by
   Adam optimisation of a log‑sum‑exp approximation of the max.
   The temperature (smoothness) is increased slowly.

3. **Exponentiated‑gradient sub‑gradient descent** – starting from the
   best Adam result we perform many exact‑max sub‑gradient steps with
   a diminishing learning rate.

4. **Simulated annealing (SA)** – a fast O(n) local search that moves
   small mass between indices.  The SA schedule is tuned for aggressive
   early exploration and a long cooling phase.

5. **Upscaling** – if time permits the best sequence is interpolated
   to a finer grid (8192 points) and the whole pipeline (Adam+sub‑grad)
   is repeated.

All steps respect the global time budget (default 1000 s).  The final
sequence is clipped to the limits required by `evaluate_sequence` and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse set of start vectors."""
    cand = []

    # Warm‑start and rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    rng = np.random.default_rng()

    # Simple Dirichlet draws
    cand.append(rng.dirichlet(np.ones(n)))
    cand.append(rng.dirichlet(0.2 * np.ones(n)))   # spiky Dirichlet

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Half‑sine (positive half‑wave)
    half_sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    half_sine = np.maximum(half_sine, 0.0)
    half_sine /= half_sine.sum()
    cand.append(half_sine)

    # Power‑law edge‑concentrated shapes (prod = i·(n-i+1))
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Two‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # Arcsine‑type (Beta(½,½))
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # Beta(α,α) family (symmetric)
    alphas = np.linspace(0.2, 3.0, 15)
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    eps = 1e-14
    for a in alphas:
        pdf = (np.maximum(x, eps) ** (a - 1)) * (np.maximum(1 - x, eps) ** (a - 1))
        pdf_sum = pdf.sum()
        if pdf_sum > 0:
            pdf /= pdf_sum
            cand.append(pdf)

    # Skewed Beta(α,β) – a few asymmetric shapes
    skew_params = [(0.4, 1.5), (1.5, 0.4), (0.3, 0.8), (0.8, 0.3)]
    for a, b in skew_params:
        pdf = (np.maximum(x, eps) ** (a - 1)) * (np.maximum(1 - x, eps) ** (b - 1))
        pdf_sum = pdf.sum()
        if pdf_sum > 0:
            pdf /= pdf_sum
            cand.append(pdf)

    # Spike families – k equally spaced spikes
    for k in (64, 128, 256, 512):
        a_spike = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a_spike[pos] = 1.0 / k
        cand.append(a_spike)

    # Periodic piecewise‑constant patterns (repeat a short Dirichlet)
    for m in (128, 256, 512):
        sub = rng.dirichlet(np.ones(m))
        a_rep = np.tile(sub, n // m + 1)[:n]
        a_rep /= a_rep.sum()
        cand.append(a_rep)

    # Add a few jittered copies of the above (helps escape symmetry)
    jittered = []
    for base in cand:
        # small multiplicative noise, then renormalise
        noise = 1.0 + 0.02 * (rng.random(n) - 0.5)
        noisy = np.maximum(base * noise, 0.0)
        s = noisy.sum()
        if s > 0:
            jittered.append(noisy / s)
    cand.extend(jittered)

    # Ensure all candidates are proper probability vectors
    clean = []
    for v in cand:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            continue
        clean.append(v / s)
    return clean

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 150_000,
                   lr: float = 0.04,
                   lam_start: float = 0.3,
                   lam_factor: float = 1.02) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best vector seen and its objective value (2·n·max_conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to enforce non‑negativity and unit sum
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (through the surrogate)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # convert to gradient for logits (soft‑max chain rule)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny jitter (helps escape tiny plateaus)
        if it % 5000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 5e-4

        it += 1

    return best_a, best_val

def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # exact max indicator
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that works directly on the convolution (O(n) per move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # temperature schedule: start 0.2, decay to 1e‑6
        T = max(1e-6, 0.2 * (1.0 - fraction))
        # move magnitude decays also
        move_scale = max(0.005, 0.4 * (1.0 - fraction))

        # choose a source index proportionally to its current mass
        total_mass = a.sum()
        if total_mass == 0:
            break
        src = np.random.choice(n, p=a / total_mass)
        if a[src] < 1e-14:
            continue
        # choose a different target uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        delta = a[src] * (np.random.rand() * move_scale)

        # apply move on copies (linear update of convolution)
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # update convolution using the *old* a (the one before the move)
        conv_new[src : src + n] -= 2.0 * delta * a
        conv_new[tgt : tgt + n] += 2.0 * delta * a
        conv_new[2 * src] += delta * delta
        conv_new[2 * tgt] += delta * delta
        conv_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 10000 == 0:
            print(f"[SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Interpolate a solution to a larger grid and run a short Adam + sub‑gradient
    refinement.  This often yields a modest improvement when time permits.
    """
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    a_adam, _ = _adam_optimize(a_big, n_big, conv_fft_len_big,

                                           deadline,
                                           max_iters=120_000,

    return a_ref, val_ref

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

        else:
            warm = None
    else:
        warm = None

        # allocate a short budget per candidate (≈15 s)
