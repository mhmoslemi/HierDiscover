"""
Propose a step‑function that (empirically) beats the previous best bound ≈1.508.
The core ideas are

* 1️⃣  A cheap parametric sweep over simple families (geometric/exponential,
      power‑law) to locate a good starting shape.
* 2️⃣  An Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate, with a slow
      temperature increase and many iterations.
* 3️⃣  An exact‑max sub‑gradient (exponentiated‑gradient) phase for final
      polishing.
* 4️⃣  A fast simulated‑annealing (SA) local search that updates the convolution
      in O(n) per elementary mass‑transfer move.
* 5️⃣  Optional up‑scaling to a finer grid (8192 points) and a short refinement
      pass.

All phases respect the global time budget (default 1000 s).  The routine
always returns a non‑negative vector; the best sequence found is stored in the
global variable ``height_sequence_1`` for warm‑starting the next run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _generate_exponential(n: int, eps: float) -> np.ndarray:
    """Geometric decreasing shape a_i ∝ (1‑eps)^i, normalised to sum‑1."""
    r = 1.0 - eps
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _generate_powerlaw(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to enforce a≥0, Σa=1
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # small jitter near the end to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that works directly on the convolution (O(n) per elementary move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling from 0.1 to 1e‑6
        T = max(1e-6, 0.1 * (1.0 - fraction))
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # pick source index proportional to its mass
        if a.sum() == 0.0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # apply the move on copies
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # convolution update (O(n))
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,

    """Interpolate to a larger grid and run a short Adam + sub‑gradient pass."""

            break

        max_c = _full_convolution(a, conv_fft_len).max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Random] -> val={val:.6f}")

    # Include the explicit warm‑start if present
    if warm is not None:
        max_c = _full_convolution(warm, conv_fft_len).max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val, best_seq = val, warm.copy()
            print(f"[Warm] -> val={val:.6f}")

    # ------------------------------------------------------------------
    # Quick exit if we already beat the target
    target = 1.503
    if best_val <= target:
        print(f"Early success: val={best_val:.6f} ≤ target={target}")
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        global height_sequence_1
        height_sequence_1 = best_seq.tolist()
        return best_seq.tolist()

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation (smooth‑max surrogate)
    a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len,
                                      corr_fft_len, deadline,
                                      max_iters=180_000)
    if val_adam < best_val:
        best_val, best_seq = val_adam, a_adam.copy()
        print(f"[Adam] improved best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exact‑max sub‑gradient refinement
    if time.time() < deadline:
        a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                               conv_fft_len,
                                               corr_fft_len,
                                               deadline,
                                               max_iters=150_000)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Subgrad] improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Heavier simulated annealing on the global best
    if best_seq is not None and time.time() < deadline - 5:
        a_sa, val_sa = _simulated_annealing(best_seq, n,
                                            conv_fft_len,
                                            deadline,
                                            best_val)
        if val_sa < best_val:
            best_val, best_seq = val_sa, a_sa.copy()
            print(f"[SA] best improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Upscale to 8192 and a short refinement pass (if time permits)
    if time.time() + 20 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(
            best_seq if best_seq is not None else np.random.dirichlet(np.ones(n_big)),
            n_big,
            deadline)
        if val_big < best_val:
            best_val, best_seq = val_big, a_big.copy()
            print(f"[Upscale] final best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the bounds enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store for the next run
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    # Final reporting – use the judge's routine if available
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:          # pragma: no cover
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)}) – {exc}")

    return best_seq.tolist()
