def _hill_climb(a, n, conv_fft_len, deadline, current_best_val):
    best_seq = a.copy()
    best_val = current_best_val
    while time.time() < deadline:
        # propose random mass transfer
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # random delta proportion of min(a[i], a[j])
        delta = (np.random.rand() - 0.5) * 0.5 * min(a[i], a[j])
        # adjust
        a[i] -= delta
        a[j] += delta
        # enforce nonnegativity
        if a[i] < 0:
            a[j] += a[i]  # a[i] negative
            a[i] = 0.0
        if a[j] < 0:
            a[i] += a[j]
            a[j] = 0.0
        # renorm
        total = a.sum()
        if total < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= total
        # eval
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        else:
            # revert changes
            a[i] += delta
            a[j] -= delta
        # maybe break after many attempts with no improvement
    return best_seq, best_val

"""
Search for a non‑negative step function that minimizes
``evaluate_sequence`` (the objective is 2·n·max(a∗a)/(sum(a)²)).
The algorithm consists of:

1. **Diversified seeding** – generate a large pool of symmetric and
   asymmetric candidates (uniform, Dirichlet draws, beta‑shapes,
   exponentials, power‑law, sin‑powers, two‑gaussians, arcsine,
   plus a few sparse “spikes”).  All candidates are normalised to sum=1
   and quickly evaluated.

2. **Intensive continuous optimisation** – the best seed is refined
   with a long Adam run on a smooth‑max (log‑sum‑exp) surrogate of the
   autocorrelation peak, followed by an exponentiated‑gradient sub‑gradient
   descent on the true max.  The Adam schedule uses a slower temperature
   increase and a smaller learning‑rate, allowing many more iterations
   (up to the remaining time budget).

3. **Fine‑grained local search** – if time permits a hill‑climbing stage
   performs many tiny random mass‑transfer moves, keeping any improvement.
   This often squeezes the objective a few percent further.

All operations are performed via FFT for speed; the code respects the
1000 s time limit, returns a list of floats, and prints progress and the
final objective value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (sum(a) is assumed ≠ 0)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence/value seen.
    """
    eps = 1e-15
    # initialise logits (soft‑max representation) – avoid exact 0 entries
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                # smaller than before – more stable for many steps
    lam = 0.3                # initial temperature (log‑sum‑exp weight)
    lam_factor = 1.001       # slow increase, allows the surrogate to tighten slowly
    lam_cap = 5e4

    best_seq = init_best_seq
    best_val = init_best_val
    last_improve = time.time()
    it = 0
    max_iter = 800_000       # “as many as time permits”

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & surrogate objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:07d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule term for the simplex (log‑softmax) parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- jitter if stuck for a while ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # keep on simplex

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.55                       # initial step‑size
    it = 0
    max_iter = 200_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:07d} new best {val:.6f}")

        # active set – indices attaining the maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # re‑initialise if we collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 12.0:   # no improvement for a while → stop early
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the two‑phase optimiser (Adam → sub‑gradient) and keep the
    best sequence/value found so far.
    """
    # Adam phase – long run
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Sub‑gradient phase – precise refinement
    best_seq, best_val = _subgradient_refinement(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: repeatedly transfer a tiny random
    amount of mass between two positions and keep the move if the
    objective improves.  Stops when time runs out or after a large
    number of attempts without improvement.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = init_best_val
    attempts = 0
    max_no_improve = 20000
    no_improve = 0

    while time.time() < deadline and attempts < 200_000:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a random fraction of the smaller of the two weights
        delta_max = min(a[i], a[j])
        if delta_max < 1e-12:
            continue
        delta = (np.random.rand() - 0.5) * 0.5 * delta_max   # ±25 % of the smaller amount
        a[i] = max(a[i] - delta, 0.0)
        a[j] = max(a[j] + delta, 0.0)

        # renormalise to stay on the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            no_improve = 0
            # print only occasionally to avoid flooding output
            if attempts % 5000 == 0:
                print(f"[{n}] Hill‑climb it={attempts:06d} new best {val:.6f}")
        else:
            # revert the move
            a[i] = max(a[i] + delta, 0.0)
            a[j] = max(a[j] - delta, 0.0)
            # re‑normalise again (cheap because sum is unchanged)
            a /= a.sum()
            no_improve += 1

        attempts += 1
        if no_improve >= max_no_improve:
            break

    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a heterogeneous pool of seed vectors (both symmetric and random).
    All vectors are normalised to sum ≈ 1.
    """
    candidates: list[np.ndarray] = []

    # Warm‑start (if supplied) and its simple shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws with many concentration values
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # Sparse “spike” vectors – choose a few random positions
    for k in (3, 5, 8, 12):
        mask = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, size=k, replace=False)
        vals = np.random.rand(k) + 0.1
        mask[idx] = vals
        mask /= mask.sum()
        candidates.append(mask)

    # Edge‑exponential families
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.67, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Symmetric beta‑shapes (a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1})
    for alpha in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

            deadline,
            best_val,
        )
        print(f"After hill‑climbing: best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clip to the evaluator’s allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
