def _lp_optimize(f, max_iter, step, deadline):
    n = f.shape[0]
    best_f = f.copy()
    best_obj = 2*n*_full_convolution(f, _next_pow2(2*n-1)).max()
    for it in range(max_iter):

            break
        # compute M
        conv = np.convolve(f, f)
        M = conv.max()
        # build matrix
        A = _build_conv_matrix(f)
        # solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:

            break
        b = res.x
        s = b.sum()
        if s <= 1 + 1e-12:
            # can't improve bound

        # convex combination

        # compute new objective
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()
        obj_new = 2*n*M_new
        if obj_new < best_obj - 1e-12:

            best_f = f_new.copy()

        # if improvement, accept; else maybe reduce step?

            f = f_new
        else:
            # no improvement, reduce step or break
            step *= 0.5
            if step < 1e-4:
                break
    return best_f, best_obj

from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _build_conv_matrix(f):

    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel()
    cols = np.tile(j, n)
    data = np.repeat(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def _max_conv(a):
    conv = np.convolve(a, a)
    return conv.max()

def _lp_iterative_optimize(a0, max_iter, step, deadline):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = 2 * n * _max_conv(a)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = _max_conv(a)
        A = _build_conv_matrix(a)
        # Solve LP: maximize sum(b) s.t A @ b <= M
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP failed at iteration {it}: {res.message}")
            break
        b = res.x
        s = b.sum()

            # No improvement possible
            break
        g = b / s
        # convex combination
        a_new = (1 - step) * a + step * g
        # compute new objective
        obj_new = 2 * n * _max_conv(a_new)
        if obj_new + 1e-12 < best_obj:

            best_a = a_new.copy()

        if obj_new < M:

"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1.  Initialise a candidate of length n_small (default 2048) from the previous
    construction ``height_sequence_1`` (or from a random Dirichlet draw).
    The vector is normalised to sum = 1.

2.  Iterative LP improvement (the classic “Gilbert‑type’’ update):
        * Let M = max (f * f).
        * Solve   max 1ᵀb   s.t.  f ★ b ≤ M  and  b ≥ 0.
          The convolution matrix f ★ · is built as a sparse Toeplitz matrix.
        * If the optimal sum(b) > 1 we can improve the bound:
          scale  g = b / sum(b)  (now sum(g)=1) and replace
                f ← (1‑t)·f + t·g   (t≈0.5, reduced if no improvement).

        The loop stops when no increase of sum(b) is possible or the time
        budget expires.  The objective for a normalised vector is
                obj = 2·n·max(f * f).

3.  If time permits, the best small‑scale solution is linearly interpolated
    to the target length n_big = 4096 and refined by a few projected‑gradient
    steps on the exact objective (gradient obtained from the active set of
    the convolution max).

4.  The final vector is clipped to the evaluator’s allowed range and returned.

The method differs from the previous pipeline by explicitly exploiting the
linear program that maximises the total mass under the current convolution
bound, a technique that often drives the max‑convolution down further than
pure gradient descent.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_convolution(a: np.ndarray) -> float:
    """Return max of linear self‑convolution a*a."""
    return np.convolve(a, a).max()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2*n‑1)×n convolution matrix ``A`` such that
        (A @ b)[k] = (f ★ b)[k]  (linear cross‑correlation).
    The matrix is built once per LP iteration and stored as CSR.
    """
    n = f.shape[0]
    # i = row offset contributed by f_i, j = column index of b
    i = np.arange(n, dtype=np.int32)[:, None]          # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]          # shape (1,n)
    rows = (i + j).ravel()                             # length n*n
    cols = np.tile(j, (n, 1)).ravel()
    data = np.repeat(f, n)                             # each f_i repeats n times
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _cross_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of two vectors of equal length n.
    Returns a vector of length n.
    """
    n = a.shape[0]
    # convolution of reversed w with a, then take the centred part
    conv = np.convolve(w[::-1], a, mode="full")
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------
def _lp_improve(f0: np.ndarray,
                max_iter: int,
                step: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based improvement loop on a normalised vector ``f0``.
    Returns the best vector found and its objective value (2·n·max_conv).
    """
    f = f0.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_obj = 2.0 * n * _max_convolution(f)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        M = _max_convolution(f)                     # current bound
        A = _build_conv_matrix(f)                  # sparse (2n‑1)×n matrix
        # Linear programme: maximise sum(b)  ⇔ minimise -1ᵀb
        lp_res = linprog(c=-np.ones(n),
                         A_ub=A,
                         b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                         bounds=[(0.0, None)] * n,
                         method='highs',
                         options={'presolve': True})

        if not lp_res.success:
            # LP failure is rare but we stop to avoid pathological loops
            print(f"[LP] iteration {it} failed: {lp_res.message}")
            break

        b_opt = lp_res.x
        sum_b = b_opt.sum()

        # If we cannot increase the total mass, the current bound is optimal.
        if sum_b <= 1.0 + 1e-12:
            # No further improvement possible
            break

        # Scale to a feasible vector with sum = 1
        g = b_opt / sum_b

        # Convex update
        f_candidate = (1.0 - step) * f + step * g
        M_new = _max_convolution(f_candidate)
        obj_new = 2.0 * n * M_new

        # Accept if we made progress; otherwise shrink the step size.
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_candidate.copy()
            print(f"[LP it {it:03d}] new best obj {best_obj:.6f}")

        if M_new < M:
            f = f_candidate
            # slowly increase the step when we are descending
            step = min(step * 1.05, 0.9)
        else:
            # no descent – reduce aggressiveness
            step *= 0.5
            if step < 1e-4:
                break

    return best_f, best_obj


def _projected_gradient_refine(a0: np.ndarray,
                               max_iter: int,
                               lr: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple projected‑gradient descent on the exact objective.
    Only the active set (indices attaining the maximal convolution) is used.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_obj = 2.0 * n * _max_convolution(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            print(f"[ProjGrad it {it:04d}] new best obj {best_obj:.6f}")

        # active mask (positions attaining the maximal value)
        active = np.isclose(conv, M, atol=1e-12)
        if not active.any():
            break

        w = active.astype(float)
        w /= w.sum()
        grad = 2.0 * _cross_correlation(w, a)      # ∂max/∂a

        # gradient step & projection onto the simplex
        a_next = _proj_simplex(a - lr * grad)

        # accept if objective improves, otherwise shrink learning rate
        if 2.0 * n * _max_convolution(a_next) < obj:
            a = a_next
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    return best_a, best_obj
