import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2*a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _proj_simplex(v):
    # Euclidean projection onto simplex
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("Simplex projection requires 1-D vector")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(a, n, conv_fft_len):
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _generate_parametric_candidates(n, warm_start):
    cand = []
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n//4, n//2, 3*n//4):
            cand.append(np.roll(warm_start, shift))
    cand.append(np.full(n, 1.0 / n))
    # Dirichlet random
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))
    i = np.arange(n, dtype=np.float64)
    # One‑sided exponential
    for alpha in np.logspace(-3, 1.0, num=15):
        shape = np.exp(-alpha * i)
        shape /= shape.sum()
        cand.append(shape)
        # Mix with uniform
        for w in (0.2, 0.5, 0.8):
            mix = w * shape + (1.0 - w) / n
            mix /= mix.sum()
            cand.append(mix)
    # Symmetric exponential (original)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n-1-i))
        edge /= edge.sum()
        cand.append(edge)
        for w in (0.0, 0.3, 0.6, 1.0):
            mix = w * edge + (1.0 - w) / n
            mix /= mix.sum()
            cand.append(mix)
    # Power‑law decreasing
    for p in np.linspace(0.2, 5.0, num=12):
        shape = (i+1) ** (-p)
        shape /= shape.sum()
        cand.append(shape)
        for w in (0.0, 0.5, 1.0):
            mix = w * shape + (1.0 - w) / n
            mix /= mix.sum()
            cand.append(mix)
    # Beta‑type shapes (more flexible)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for beta in (0.2, 0.5, 1.0, 2.0):
            edge = (i+1) ** (alpha-1) * (n-i) ** (beta-1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand.append(edge)
    # Triangular
    tri = np.minimum(i+1, n-i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)
    # Quadratic (parabolic)
    quad = (i+1) * (n-i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand.append(quad)
    return cand

def _adam_smooth_max(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.01
    lam_cap = 1e7
    last_improve = time.time()
    it = 0
    max_iter = 300_000
    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it}] new best {val:.6f}")
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        lam = min(lam * lam_factor, lam_cap)
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        it += 1
    return best_seq, best_val

def _max_gradient_descent(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= s
    eta = 0.5
    last_improve = time.time()
    it = 0
    max_iter = 200_000
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Grad n={n} it={it}] new best {val:.6f}")
        # active set
        active = np.isclose(conv, M, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a_new = _proj_simplex(a - eta * grad)
        # evaluate candidate
        conv_cand = _full_convolution(a_new, conv_fft_len)
        M_cand = conv_cand.max()
        val_cand = 2.0 * n * M_cand / (a_new.sum() ** 2)
        if val_cand < val:
            a = a_new
            eta = min(eta * 1.02, 1.0)
        else:
            eta *= 0.7
            if eta < 1e-6:
                break
        if time.time() - last_improve > 5.0:
            # stagnation, add tiny random jitter
            a += (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = time.time()
        it += 1
    return best_seq, best_val

def _deep_optimize_one(a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val):
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val
    )
    best_seq, best_val = _max_gradient_descent(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val
    )
    return best_seq, best_val

def _local_perturbations(base_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.01
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s
        seq, val = _max_gradient_descent(
            a_pert, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            # small drift towards original base
            a0 = 0.9 * a0 + 0.1 * base_seq
    return best_seq, best_val

"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum a)²

using a rich pool of parametric seeds, Adam on a smooth‑max
surrogate (log‑sum‑exp) and a projected sub‑gradient descent on the
exact max‑convolution.  One‑sided exponential families are added to the
candidate set (they tend to reduce the reverse‑product term) and the
Adam temperature grows more slowly, giving a finer optimisation.
The routine respects the 1000 s budget, prints progress, and returns the
best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection needs a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (assumes sum(a)=1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

    cand = []

    # warm start and rolled versions
    if warm_start is not None:

        for shift in (n // 4, n // 2, 3 * n // 4):

    cand.append(np.full(n, 1.0 / n))

    # random Dirichlet draws
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):

    for alpha in np.logspace(-3, 1.0, num=14):
        shape = np.exp(-alpha * i)          # decreasing from the left
        shape /= shape.sum()
        cand.append(shape)
        # mix with a uniform component

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # work in log‑space (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # start temperature for smooth‑max
    lam_factor = 1.01        # grow slowly – more gradual refinement
    lam_cap = 1e7

    last_improve = time.time()
    it = 0
    max_iter = 250_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max (log‑sum‑exp trick) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate evaluation ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _max_gradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses the active set (indices attaining the maximum) to form a
    sub‑gradient.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full_like(a, 1.0 / n)

    eta = 0.5
    last_improve = time.time()
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Grad n={n} it={it}] new best {val:.6f}")

        # ---- active set (where the max is attained) ----
        active = np.isclose(conv, M, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # ---- sub‑gradient of the max function ----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- projected descent step ----
        a_candidate = _proj_simplex(a - eta * grad)

        # evaluate candidate
        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        M_cand = conv_cand.max()
        val_cand = 2.0 * n * M_cand / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            eta = min(eta * 1.02, 1.0)          # modest acceleration when improving
        else:
            eta *= 0.7                           # back‑off if not improving
            if eta < 1e-6:
                break

        # ---- stagnation handling ----
        if time.time() - last_improve > 5.0:
            # tiny random jitter to escape flat regions
            a += (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _deep_optimize_one(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run the two‑phase optimisation pipeline for a single start point."""
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _max_gradient_descent(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _local_perturbations(
    base_seq: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search: small Gaussian jitter + short max‑gradient
    refinements.
    """
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.02
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s

        seq, val = _max_gradient_descent(
            a_pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            a0 = 0.9 * a0 + 0.1 * base_seq      # gentle drift back

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a step‑function with minimal
    evaluation score.  Returns the best sequence (plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Grid sizes – start modest, then enlarge if time permits
    for n in (4096, 8192):
        if time.time() >= deadline:
            break
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # warm start from a previously discovered construction, if any
        warm_start = None
        if "height_sequence_1" in globals():
            ws = globals()["height_sequence_1"]
            ws = np.asarray(ws, dtype=np.float64)
            if ws.shape[0] > n:
                ws = ws[:n]
            elif ws.shape[0] < n:
                ws = np.pad(ws, (0, n - ws.shape[0]))
            ws = np.maximum(ws, 0.0)
            if ws.sum() >= 0.01:
                warm_start = ws / ws.sum()

        # ------------------------------------------------------------------
        # Stage 1 – generate a rich pool of parametric candidates
        all_candidates = _generate_parametric_candidates(n, warm_start)

        cand_vals = []
        for cand in all_candidates:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            total = cand.sum()
            if total < 1e-12:
                continue
            cand = cand / total
            val = _objective(cand, n, conv_fft_len)
            cand_vals.append((val, cand))

        # keep the most promising starts
        cand_vals.sort(key=lambda x: x[0])
        top_k = 10
        top_candidates = [c for _, c in cand_vals[:top_k]]

        # ------------------------------------------------------------------
        # Stage 2 – deep optimisation of each promising start
        for idx, start_vec in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(
                f"\n--- Optimising start {idx+1}/{len(top_candidates)} (n={n}) ---"
            )
            best_seq, best_val = _deep_optimize_one(
                start_vec,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )
            if best_seq is not None:
                print(f"*** Global best now {best_val:.6f} (n={n}) ***")

        # ------------------------------------------------------------------
        # Stage 3 – stochastic local perturbations around the current best
        if best_seq is not None and time.time() < deadline - 5.0:
            print("\n--- Local stochastic perturbations ---")
            best_seq, best_val = _local_perturbations(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )
            if best_seq is not None:
                print(f"*** Global best after perturb {best_val:.6f} ***")

        # ------------------------------------------------------------------
        # If we have just finished the smaller grid and still have ample
        # time, the loop automatically proceeds to the larger grid (n=8192).

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(
        f"\n=== Search finished: best objective ≈ {best_val:.6f}, length {best_seq.shape[0]} ==="
    )
    return best_seq.tolist()
