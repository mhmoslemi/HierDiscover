"""
Hybrid differential‑evolution + gradient refinement for the
max‑convolution minimisation problem.

The search proceeds in three stages:

1. **Population initialisation** – a modest population (≈30) is seeded
   with a variety of hand‑crafted patterns (uniform, Dirichlet,
   power‑law, double‑exponential, sinusoidal, linear‑decrease and a few
   exponential decays) together with random permutations of those
   patterns.  All vectors are normalised to sum = 1.

2. **Differential Evolution (DE)** – a classic DE/rand/1/bin scheme
   runs on the simplex (non‑negative, sum = 1).  Each candidate is mutated,
   clipped to non‑negative values and re‑normalised.  The objective value
   is `2·n·max(conv(a,a))`, computed by an FFT‑based convolution.
   The DE loop runs until the time budget is almost exhausted, constantly
   updating the current global best.

3. **Fine tuning** – the best vector found by DE is refined with a short
   Adam optimisation on the smooth “log‑sum‑exp’’ surrogate of the max‑
   convolution, followed by a cheap hill‑climbing random‑move phase.

The routine respects the supplied time budget (default 1000 s) and always
returns the best feasible sequence it has discovered.  The final sequence
is stored in the global name `height_sequence_1` for warm‑starting later
runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate of the
    convolution max.  The parameters are kept on the simplex via a
    soft‑max (logits → probabilities).  Returns the best vector and its
    exact objective (2·n·max(conv)).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.7           # start temperature for the smooth max
    lam_factor = 1.04

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

    for s in base:
        sh = np.random.permutation(s)
        shuffled.append(sh)
    base.extend(shuffled)

    # --------------------------------------------------------------
    # Initialise DE population (size ≈30)
    popsize = 30
    population = []
    # take the first popsize vectors from base (or random Dirichlet if not enough)
    for s in base[:popsize]:
        population.append(s.copy())
    while len(population) < popsize:
        population.append(np.random.dirichlet(np.ones(n)))

    # Evaluate initial population
    objectives = []
    for a in population:
        conv = _full_convolution(a, conv_fft_len)
        objectives.append(2.0 * n * conv.max())
    best_idx = int(np.argmin(objectives))
    best_a = population[best_idx].copy()
    best_val = objectives[best_idx]

    # --------------------------------------------------------------
    # Differential Evolution loop
    F = 0.7                # mutation factor
    CR = 0.9               # crossover probability
    it = 0

    while time.time() < deadline - 5:   # keep a small safety margin
        for target in range(popsize):
            # pick three distinct indices different from target
            idxs = list(range(popsize))
            idxs.remove(target)
            a_idx, b_idx, c_idx = np.random.choice(idxs, 3, replace=False)
            a = population[a_idx]
            b = population[b_idx]
            c = population[c_idx]

            # mutation and projection onto the simplex
            mutant = a + F * (b - c)
            mutant = np.maximum(mutant, 0.0)
            s_mut = mutant.sum()
            if s_mut < 1e-12:
                mutant = np.random.dirichlet(np.ones(n))
                s_mut = 1.0
            mutant = mutant / s_mut

            # binomial crossover
            cross_mask = np.random.rand(n) < CR
            if not np.any(cross_mask):               # guarantee at least one gene
                cross_mask[np.random.randint(n)] = True
            trial = np.where(cross_mask, mutant, population[target])

            # ensure feasibility
            trial = np.maximum(trial, 0.0)
            s_tr = trial.sum()
            if s_tr < 1e-12:
                trial = np.random.dirichlet(np.ones(n))
                s_tr = 1.0
            trial = trial / s_tr

            # evaluate trial
            val = 2.0 * n * _full_convolution(trial, conv_fft_len).max()
            if val < objectives[target]:
                population[target] = trial
                objectives[target] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    print(f"[DE iter={it:04d} tgt={target}] new best {best_val:.6f}")

        it += 1
        # early exit if time is almost up
        if time.time() > deadline - 5:
            break

    # --------------------------------------------------------------
    # Fine‑tune the best vector with a short Adam run
    if time.time() < deadline - 5:
        adam_deadline = min(deadline - 1.0, time.time() + 12.0)
        a_ref, val_ref = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline, max_iters=150_000
        )
        if val_ref < best_val:
            best_val = val_ref
            best_a = a_ref.copy()
            print(f"[Adam refine] improved best {best_val:.6f}")

    # --------------------------------------------------------------
    # Final cheap local random move phase
    if time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_a, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_a = a_loc.copy()
            print(f"[Local search] improved best {best_val:.6f}")

    # --------------------------------------------------------------
    # Ensure a valid return value
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total < 1e-12:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_a = best_a / total
    best_a = np.clip(best_a, 0.0, 1000.0)

    # make the solution available for possible warm starts in later runs
    globals()["height_sequence_1"] = best_a.copy()

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
