def refine_exponential_gradient(a, deadline, best_val, best_seq):

        # compute active weight
        active = conv == max_c  # would need tolerance
        # maybe also include near-active for smoothing
        mask = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)

        if k_active > 0:

"""
Hybrid optimisation for the step‑function bound

The search proceeds in three phases:

1. **Warm‑start** – try to reuse a previously found construction
   (global ``height_sequence_1``).  If none is available we fall back to a
   uniform vector.

2. **Long exponentiated‑gradient refinement** – we minimise the exact
   objective 2·n·max(conv(a,a))/sum(a)^2 on the simplex.
   The update is performed in log‑space (mirror descent w.r.t.
   KL‑divergence).  The step size decays like 1/√t and a small random
   “shake’’ is added when the method stalls, which helps to escape flat
   plateaux.

3. **Cheap stochastic hill‑climbing** – after the heavy refinement we
   perform a simple log‑space Gaussian perturbation search that accepts
   only improving moves (pure hill‑climbing).  Because the objective is
   very flat near a local optimum this final stage often squeezes a
   few thousandths out of the bound.

All heavy work respects the global deadline (default 1000 s).  The
returned sequence is non‑negative, sums to 1 (scaling does not affect the
objective) and is clipped to the evaluator’s allowed range (≤ 1000)."""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) with length len(a).
    (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv)/sum(a)^2 (sum(a) is assumed >0)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _refine_expgd(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution.
    Mirrors the update in log‑space, uses step size η_t = η0/√t,
    and injects a tiny random shake after a few seconds without progress.
    """
    eps = 1e-15
    a = a0.copy()
    loga = np.log(a + eps)

    best_seq = init_best_seq.copy()
    best_val = init_best_val
    last_improve = time.time()
    it = 0
    eta0 = 0.5                     # initial step size (tuned empirically)

    while time.time() < deadline:
        # evaluate current objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Refine] iter={it:06d} new best {val:.6f}")

        # active set where the max is attained (numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        cnt = active.sum()
        if cnt > 0:
            w[active] = 1.0 / cnt

        # sub‑gradient of max‑convolution (see derivation in the description)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule η_t = η0 / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # mirror‑descent update in the log‑domain, then take soft‑max
        loga -= eta * grad
        mx = loga.max()
        exp_log = np.exp(loga - mx)
        a = exp_log / exp_log.sum()

        # occasional random shake if stuck for >5 s
        if time.time() - last_improve > 5.0:
            loga += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing in log‑space.
    Small Gaussian perturbations of the logits are accepted only if they
    improve the *exact* objective.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        # Gaussian perturbation (scale slowly reduced with remaining time)
        remaining = deadline - time.time()
        sigma = 0.04 * max(0.05, remaining / (deadline - (deadline - 1000)))  # crude schedule
        noise = np.random.randn(n) * sigma
        logits = logits_best + noise

        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – returns a non‑negative sequence that (hopefully)
    improves upon the previously reported bound (~1.508).
    """
    np.random.seed(seed)
    now = time.time()
    deadline = now + budget_s - 0.5   # tiny safety margin

    # ----- problem size -------------------------------------------------
    n = 4096                         # size that gave the previous best
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----- warm start ---------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] > n:
            a0 = a0[:n]
        elif a0.shape[0] < n:
            a0 = np.pad(a0, (0, n - a0.shape[0]))
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-8:
        a0 = np.random.dirichlet(np.ones(n))
    else:
        a0 = a0 / s                     # normalise to sum‑1 (scale irrelevant)

    best_seq = a0.copy()
    best_val = _objective(best_seq, n, conv_fft_len)
    print(f"[Init] start value = {best_val:.6f}")

    # ----- quick random Dirichlet sweep (very cheap) --------------------
    # This may discover an even better basin before the heavy refinement.
    rnd_deadline = now + min(30.0, (deadline - now) * 0.05)  # at most 30 s or 5 % of budget
    while time.time() < rnd_deadline:
        conc = np.random.choice([0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
        cand = np.random.dirichlet(np.full(n, conc))
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Rand] new best {val:.6f}")

    # ----- long exponentiated‑gradient refinement -----------------------
    if time.time() < deadline:
        print("[Refine] start long exponential‑gradient phase")
        best_seq, best_val = _refine_expgd(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ----- cheap stochastic hill‑climbing -------------------------------
    if time.time() < deadline:
        print("[Local] final stochastic hill‑climb")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ----- final safety clipping ----------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
