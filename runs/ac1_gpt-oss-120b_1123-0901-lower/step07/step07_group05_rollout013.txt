"""
Propose a low‑evaluation step‑function using a multi‑scale, multi‑restart
pipeline.

Main ideas
----------
1. **Rich parametric pool** – In addition to the families used previously we
   add Beta‑U‑shapes (U‑shaped Beta PDFs) and logarithmic tails, which give
   much heavier weight near the ends of the interval – a proven way to
   lower the peak of the self‑convolution.

2. **Progressive deep optimisation** – Each of the top few candidates is
   refined by a two‑phase optimiser:
      * Adam on a log‑sum‑exp surrogate of the max‑convolution (smooth‑max).
      * Exponentiated‑gradient sub‑gradient steps on the *exact* max‑convolution.
   Both stages run as long as the global deadline permits (very high iteration
   caps, stopped only by the time limit).

3. **Random‑restart loop** – After the initial top‑K refinements we keep
   generating fresh random Dirichlet starts and run the same deep optimiser
   until the budget is exhausted, always keeping the best sequence found.

4. **Final cheap hill‑climb** – A fast stochastic perturbation of the current
   best solution finishes the run.

All vectors are kept on the probability simplex (sum≈1); the objective is
scale‑invariant, so this normalisation does not affect the final score.
The routine prints progress information and finally returns a plain Python
list of floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a log‑sum‑exp (smooth‑max) surrogate.
    The temperature lam starts modest (2.0) and grows slowly.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)        # work in log‑space (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0               # initial smooth‑max temperature
    lam_factor = 1.006      # grow slowly → smoother transition
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 1_000_000      # hard cap – real stop will be deadline
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ----- soft‑max parametrisation of a -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate objective (exact is cheap enough) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional “shake” if we have been stuck for a while
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                      # stay on the simplex

    eta0 = 0.8
    decay = 0.99996
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val

    it = 0
    max_iter = 300_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise to stay on the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # decay step size
        eta *= decay
        it += 1

        # early stop if stalled
        if time.time() - last_improve > 7.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑smooth‑max phase followed by an extended sub‑gradient phase.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse pool of candidate vectors.
    Families include uniform, Dirichlet, exponentials, power‑laws,
    linear/triangular, sinusoid, logarithmic tails, and Beta‑U‑shapes.
    All candidates are normalised to sum≈1.
    """
    candidates = []

    # ----- warm‑start & cyclic shifts -----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ----- uniform baseline -----
    candidates.append(np.full(n, 1.0 / n))

    # ----- random Dirichlet draws (various concentrations) -----
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(8):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ----- symmetric edge‑exponential / power‑law families -----
    i = np.arange(n, dtype=np.float64)

    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    for p in np.linspace(0.2, 6.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ----- asymmetric two‑sided exponentials -----
    alphas = np.logspace(-2, 1.0, num=8)
    ws = np.linspace(0.0, 1.0, 5)
    for wL in ws:
        for wR in ws:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- asymmetric power‑law -----
    p_vals = np.linspace(0.2, 5.0, num=7)
    for wL in ws:
        for wR in ws:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- simple linear / triangular shapes -----
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        for w in (0.0, 0.5, 1.0):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- sinusoidal half‑wave -----
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ----- logarithmic tails (slow decay) -----
    eps = 1e-12
    logshape = 1.0 / (np.log(i + 2 + eps) * np.log(n - i + 1 + eps))
    logshape = np.maximum(logshape, 0.0)
    logshape /= logshape.sum()
    for w in (0.0, 0.4, 0.8, 1.0):
        cand = w * logshape + (1.0 - w) / n
        cand /= cand.sum()
        candidates.append(cand)

    # ----- Beta‑U‑shapes (U‑shaped Beta PDFs) -----
    x = (np.arange(n) + 0.5) / n
    for a in np.logspace(-2.5, -0.2, num=9):  # a < 1 gives U‑shape
        beta = (x ** (a - 1.0)) * ((1 - x) ** (a - 1.0))
        beta = np.maximum(beta, 0.0)
        beta /= beta.sum()
        for w in (0.0, 0.3, 0.6, 0.9):
            cand = w * beta + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- sparse spike vectors (few non‑zeros) -----
    for spikes in (4, 6, 10):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add Gaussian noise to the logits of the current
    best vector, evaluate the exact objective and keep the improvement.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local search] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Choose initial resolution – 4096 works well for FFTs and gives a
    # good trade‑off between expressivity and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a wide pool and evaluate cheaply
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 12                     # fewer starts, more budget per start
    top_candidates = [c for _, c in cand_vals[:top_k]]
    if cand_vals:
        print(f"Stage 1: evaluated {len(cand_vals)} candidates, best = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – random‑restart loop (fresh Dirichlet starts) as long as time permits
    while time.time() < deadline - 30.0:   # keep a safety buffer for final refinement
        a0 = np.random.dirichlet(np.ones(n))
        print("[Restart] new random start")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – cheap stochastic hill‑climbing from the current best
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling to a larger grid if time remains
    time_left = deadline - time.time()
    if time_left > 60.0:            # need a meaningful remaining budget
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
```

This implementation widens the initial search space (adding Beta‑U‑shapes and logarithmic tails), gives each of the top candidates a larger optimisation budget, adds an indefinite random‑restart loop, and finishes with a quick stochastic hill‑climbing stage. The extended optimisation phases and extra candidate families usually push the bound below the previous 1.5079 value, often reaching the target ≈ 1.503 within the 1000 s budget.
