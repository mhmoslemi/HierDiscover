def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # choose base n (maybe 8192)
    n_base = 8192
    # ensure length within time: compute conv length and use simple loops
    conv_len = _next_pow2(2*n_base - 1)
    corr_len = _next_pow2(3*n_base - 2)

    # build candidate pool
    candidates = []

    # Dirichlet random
    for _ in range(5):
        cand = np.random.dirichlet(np.ones(n_base))
        candidates.append(cand)

    # Uniform
    candidates.append(np.full(n_base, 1.0/n_base))

    # Beta shapes
    alphas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            cand = _beta_candidate(n_base, a, b)
            candidates.append(cand)

    # Edge exponential
    for gamma in [0.1, 0.3, 0.5, 1.0, 2.0]:
        left = np.exp(-gamma * np.arange(n_base))
        right = left[::-1]
        cand = left + right
        cand = np.clip(cand, 0, None)
        cand /= cand.sum()
        candidates.append(cand)

    # Sinusoid half-wave
    cand = np.sin(np.pi * (np.arange(n_base)+0.5)/n_base)
    cand = np.clip(cand, 0, None)
    cand /= cand.sum()
    candidates.append(cand)

    # Spike patterns
    for k in [32,64,128]:
        a=np.zeros(n_base)
        pos=np.round(np.linspace(0, n_base-1, k)).astype(int)
        a[pos]=1.0/k
        candidates.append(a)

    # optionally incorporate warm start variable if exists
    if 'height_sequence_1' in globals():
        w = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        # adjust to length n_base
        if w.shape[0] > n_base:
            w = w[:n_base]
        elif w.shape[0] < n_base:
            w = np.pad(w, (0, n_base-w.shape[0]))
        w = np.maximum(w, 0)
        if w.sum()>=0.01:
            w = w / w.sum()
            candidates.append(w)

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of three stages:

1. **Rich warm‑start pool** –  a deterministic collection of
   parametric shapes (Beta‑distributions, edge‑exponential,
   sinusoid, uniform, sparse spikes) together with a possible
   previous solution ``height_sequence_1``.  All candidates are
   normalised to sum=1 and the best one (according to the exact
   evaluator) is kept.

2. **Smooth‑max Adam optimisation** –  minimise a differentiable
   soft‑max surrogate of the maximum convolution using Adam
   (mirror‑descent on the simplex).  The temperature starts
   small (≈0.05) and is multiplied each iteration (≈1.07) until
   the surrogate approaches the true maximum.

3. **Exact‑max sub‑gradient phase** –  after the temperature has
   become large we switch to the exact max and perform exponentiated‑
   gradient (multiplicative) updates with a decreasing step‑size
   ηₜ=η₀/√t.  A cheap random‑move hill‑climber is finally applied
   if time permits.

The whole search respects the supplied time budget (default 1000 s)
and always returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _beta_candidate(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on a uniform grid, normalised to sum = 1."""
    # grid points in the centre of each bin
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid exact 0 / 1 when α or β < 1 (integrable singularities)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s <= 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors."""
    cand = []

    # Warm start (if any) and a few cyclic shifts.
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # Edge‑exponential families (mass near the ends)
    for gamma in (0.1, 0.3, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # Uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Sparse equally‑spaced spikes
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    # Beta‑distribution family (covers U‑shaped, triangular, etc.)
    alphas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    betas   = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    for a in alphas:
        for b in betas:
            cand.append(_beta_candidate(n, a, b))

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # start on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09                               # a bit larger than baseline
    lam = 0.04                               # temperature
    lam_factor = 1.07

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---------- project logits to simplex (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---------- chain rule through soft‑max (mirror descent) ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter when we are close to the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 60_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size ηₜ = η₀ / √(t+1).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise – guard against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: transfer a random fraction of mass from one index
    to another; keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # problem size – we start with a relatively large grid (8192)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start from previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build a diverse pool of starts
    pool = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # quick scan of the pool (only exact evaluation, no optimisation)
    print("[Scanning] evaluating candidate pool ...")
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand = cand / cand.sum()
        # use the provided evaluator for a reliable objective
        try:
            val = evaluate_sequence(cand.tolist())
        except Exception:          # defensive – should never happen
            val = float("inf")

        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Pool  {idx:03d}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Allocate remaining time to a deep refinement of the current best
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    time_left = deadline - time.time()
    # 40 % Adam, 55 % sub‑gradient, 5 % local search (if still time)
    adam_deadline = time.time() + 0.40 * time_left
    subgrad_deadline = time.time() + 0.95 * time_left   # leaves ~5 % for local search

    # ---- Adam (smooth‑max) ------------------------------------------------
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=250_000,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved best {val_adam:.6f}")

    # ---- Sub‑gradient (exact max) ----------------------------------------
    if time.time() < subgrad_deadline:

            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            subgrad_deadline,
            max_iters=400_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:

            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation – enforce evaluator's constraints
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
