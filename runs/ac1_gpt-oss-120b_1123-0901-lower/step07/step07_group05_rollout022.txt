"""
Propose a low‑objective step‑function using a multi‑stage search:

1.  Generate a large, diverse pool of parametric candidates (uniform,
    Dirichlet, edge‑exponential / power‑law, asymmetric exponentials,
    linear / triangular, sinusoidal, sparse spikes, …).  All candidates are
    normalised to the simplex.
2.  Keep the top K candidates (by the exact objective) and run an extensive
    optimisation on each:
        – Adam on a log‑sum‑exp surrogate of the max‑convolution (smooth‑max).
        – Sub‑gradient (exponentiated‑gradient) refinement on the exact max.
    The number of iterations is dynamically limited by the global deadline.
3.  Pick the best sequence obtained and perform a simulated‑annealing /
    stochastic hill‑climbing phase with a temperature schedule to escape
    any remaining local minima.
4.  If time permits, upscale to a finer grid (n = 8192) and repeat step 2‑3.
5.  Return the best sequence as a plain Python list (values are clipped to
    [0, 1000] as required by ``evaluate_sequence``).

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for final validation, and respects the
1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective  F = 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06

    lam = 2.0               # start temperature
    lam_factor = 1.005      # slower growth → smoother final max
    lam_cap = 1e7

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----- probability simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate max (log‑sum‑exp) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:07d}  new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 2e-3
            last_improve = time.time()

        it += 1
        # safety break (prevent infinite loop if deadline far)
        if it >= 600_000:
            break

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s  # stay on simplex

    eta = 0.75
    decay = 0.99994

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:07d}  new best {val:.6f}")

        # active constraints (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (maintains non‑negativity)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # early stop if stuck
        if time.time() - last_improve > 7.0:
            break
        if it >= 250_000:
            break

    return best_seq, best_val


def _simulated_annealing_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing with Metropolis acceptance.
    Perturbs the log‑probabilities of the current best vector.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    best_seq = a.copy()
    best_val = init_best_val

    sigma = 0.05               # initial Gaussian step size in log‑space
    temp = 0.03                # Metropolis temperature
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # perturb logits
        noise = np.random.randn(n) * sigma
        logits_new = logits + noise
        mx = logits_new.max()
        exp_log = np.exp(logits_new - mx)
        a_new = exp_log / exp_log.sum()

        val_new = _objective(a_new, n, conv_fft_len)

        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            logits = logits_new
            a = a_new
            last_improve = time.time()
            print(f"[Anneal] iter={it:07d}  new best {val_new:.6f}")
        else:
            # Metropolis rule
            if np.random.rand() < math.exp(-(val_new - best_val) / max(temp, 1e-12)):
                logits = logits_new
                a = a_new

        # cool down
        sigma *= 0.9996
        temp *= 0.9997
        it += 1

        if time.time() - last_improve > 8.0:
            break
        if it >= 300_000:
            break

    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Create a diverse pool of candidate vectors.
    Families:
      • uniform
      • Dirichlet draws with many concentrations
      • symmetric edge‑exponential / power‑law
      • asymmetric two‑sided exponentials / power‑laws
      • linear / triangular
      • half‑wave sinusoid
      • sparse spikes
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # warm‑start and shifted copies
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(16):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # symmetric edge exponentials
    alphas = np.logspace(-2, 1.0, 14)
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # symmetric power‑laws
    ps = np.linspace(0.2, 6.0, 14)
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, 10)
    w_vals = np.linspace(0.0, 1.0, 6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # asymmetric power‑laws
    ps = np.linspace(0.2, 5.0, 9)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in ps:
                for pR in ps:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # linear / triangular
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate(
        [np.linspace(0, 1, n // 2, endpoint=False),
         np.linspace(1, 0, n - n // 2)]
    )
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.5, 0.9):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    candidates.append(sinus)

    # sparse spikes
    for spikes in (3, 5, 9):
        for _ in range(24):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: Gaussian noise on logits, keep improvement.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – search for a non‑negative step function that
    minimises the supplied evaluation function.  Returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Choose a moderate size (large enough to be expressive but still cheap)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – cheap generation + evaluation of a large candidate pool
    all_cands = _generate_parametric_candidates(n, warm_start)
    cand_vals = []
    for cand in all_cands:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # safety fallback
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq, n, conv_fft_len)
    else:
        cand_vals.sort(key=lambda x: x[0])
        top_k = 28
        top_candidates = [c for _, c in cand_vals[:top_k]]
        best_val, best_seq = cand_vals[0]
        print(
            f"[Stage 1] evaluated {len(cand_vals)} candidates, best = {best_val:.6f}"
        )

        # ------------------------------------------------------------------
        # Stage 2 – deep optimisation on the top candidates
        # we give each candidate a modest budget; the globally best
        # receives the remaining time.
        per_candidate_deadline = deadline - 5.0  # keep some slack for final stages
        for idx, start_vec in enumerate(top_candidates):
            if time.time() >= per_candidate_deadline:
                break
            print(f"--- Deep optimisation start #{idx + 1}/{len(top_candidates)} ---")
            # Adam + sub‑gradient
            seq, val = _adam_smooth_max(
                start_vec,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )
            seq, val = _subgradient_refinement(
                start_vec,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                seq,
                val,
            )
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Update] global best improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # Stage 3 – simulated‑annealing refinement on the current global best
        if best_seq is not None and time.time() < deadline:
            print("--- Simulated‑annealing refinement ---")
            best_seq, best_val = _simulated_annealing_refinement(
                best_seq,
                n,
                conv_fft_len,
                deadline,
                best_val,
            )

        # ------------------------------------------------------------------
        # Stage 4 – cheap local stochastic hill‑climbing
        if best_seq is not None and time.time() < deadline:
            print("--- Final local random walk ---")
            best_seq, best_val = _local_random_search(
                best_seq,
                n,
                conv_fft_len,
                deadline,
                best_val,
            )

        # ------------------------------------------------------------------
        # Stage 5 – optional upscale to a finer grid if enough time remains
        time_left = deadline - time.time()
        if time_left > 40.0:
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # interpolate the current best onto the finer grid
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            if a_big.sum() < 0.01:
                a_big = np.random.dirichlet(np.ones(n_big))
            a_big /= a_big.sum()

            print("--- Optimising on larger grid (n=8192) ---")
            # a single deep optimisation run on the larger grid
            a_big, val_big = _adam_smooth_max(
                a_big,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                best_seq,
                best_val,
            )
            a_big, val_big = _subgradient_refinement(
                a_big,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                a_big,
                val_big,
            )
            a_big, val_big = _simulated_annealing_refinement(
                a_big,
                n_big,
                conv_fft_len_big,
                deadline,
                val_big,
            )
            if val_big < best_val:
                best_seq, best_val = a_big, val_big
                n = n_big
                print(f"[Upgrade] improved on larger grid to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – in the unlikely event that nothing was found
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
