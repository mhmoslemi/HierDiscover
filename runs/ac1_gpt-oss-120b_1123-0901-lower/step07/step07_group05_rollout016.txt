def propose_candidate(seed=0,budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose base n
    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    # Warm start
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        # convert to array length n
        warm = np.asarray(warm_start, dtype=np.float64)
        if warm.shape[0] != n:
            # interpolate to n
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0)
        s = warm.sum()
        if s >= 0.01:
            warm = warm / s
        else:
            warm = None
    else:
        warm = None

    # Stage 1: generate candidate pool
    candidates = _generate_parametric_candidates(n, warm)
    # maybe add random Dirichlet draws for diversification
    for _ in range(30):
        cand = np.random.dirichlet(np.ones(n))
        candidates.append(cand)

    # Evaluate
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline: break
        # clip
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # Sort and keep top_k
    cand_vals.sort(key=lambda x:x[0])
    top_k = 12
    top_candidates = [c for _,c in cand_vals[:top_k]]
    best_seq = None
    best_val = float('inf')
    if cand_vals:
        best_val, best_seq = cand_vals[0]

    print(f"Stage1 evaluated {len(cand_vals)} candidates, best initial {best_val:.6f}")

    # Stage 2: deep optimizer for top candidates
    for idx, start in enumerate(top_candidates):
        if time.time() >= deadline: break
        print(f"Deep opt start {idx+1}/{len(top_candidates)}")
        # deep optimize; deep_optimize returns best_seq_loc, best_val_loc
        best_seq, best_val = _deep_optimize(start, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val)

    # Stage 3: local random hill climbing
    if best_seq is not None and time.time() < deadline:
        print("Local stochastic refinement")
        best_seq, best_val = _local_random_search(best_seq, n, conv_fft_len, deadline, best_val)

    # Stage 4: upscale if time remains
    if time.time() < deadline - 30:
        n_big = 8192
        conv_fft_big = _next_pow2(2*n_big-1)
        corr_fft_big = _next_pow2(3*n_big-2)
        # interpolate best to n_big
        x_old = np.linspace(0,1,n,endpoint=False)
        x_new = np.linspace(0,1,n_big,endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big,0)
        a_big /= a_big.sum()
        print("Optimization on larger grid")
        best_seq, best_val = _deep_optimize(a_big, n_big, conv_fft_big, corr_fft_big, deadline, best_seq, best_val)
        n = n_big
        conv_fft_len = conv_fft_big
        corr_fft_len = corr_fft_big

    # optional final local search on final n
    if best_seq is not None and time.time() < deadline:
        best_seq, best_val = _local_random_search(best_seq, n, conv_fft_len, deadline, best_val)

    # Clip and return list
    best_seq = np.clip(best_seq,0,1000)
    print(f"Finished: best value {best_val:.6f}, n={best_seq.shape[0]}")
    return best_seq.tolist()

"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1. Choose a moderate size `n = 4096`.  (If time permits a larger grid
   `n = 8192` is tried later.)
2. Build a *large* parametric pool of candidate probability vectors.
   Families include:
      – uniform, Dirichlet draws,
      – symmetric edge‑exponential / power‑law,
      – asymmetric two‑sided exponential / power‑law,
      – pure power‑law with positive exponent (very heavy edges),
      – linear and triangular ramps,
      – half‑sine wave, logarithmic shape,
      – a few random sparse spikes.
   All candidates are normalised to sum ≈ 1.
3. Evaluate the exact objective for every candidate (FFT‑based
   convolution, O(n log n)).  Keep the best `top_k` candidates.
4. For each of the top candidates run a *deep* optimisation:
      * Adam on a log‑sum‑exp surrogate of the max‑convolution,
        with a warm‑up learning‑rate schedule.
      * Followed by an exponentiated‑gradient sub‑gradient phase
        on the true max‑convolution.
   The optimisation stops as soon as the global deadline is reached.
5. Starting from the best vector found, perform cheap stochastic
   hill‑climbing (Gaussian perturbations in log‑space) to squeeze out
   any remaining improvement.
6. If time is still left, up‑sample the current best to `n = 8192`,
   repeat step 4 and a final hill‑climbing.
7. Return the best sequence (clipped to the evaluator’s range).

The routine respects all security checks performed by
`evaluate_sequence` and never produces negative or > 1000 entries.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(a*a)/(sum(a)²).  sum(a) is assumed =1."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0               # start temperature
    lam_factor = 1.012
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (its value is needed for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake when stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the exact max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s  # keep on the simplex

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 120_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # decay step size
        eta *= decay
        it += 1

        # early stop if no progress for a while
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam‑smooth‑max followed by an extended sub‑gradient phase.
    Returns the best sequence/value seen (updates the global best).
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse pool of candidate vectors.
    Families include uniform, Dirichlet, symmetric/asymmetric exponentials,
    power‑laws (both decreasing and increasingly heavy at the edges),
    linear/triangular ramps, half‑sine, logarithmic shape and sparse spikes.
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # ------------------------------------------------------------------
    # warm‑start & its shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ------------------------------------------------------------------
    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ------------------------------------------------------------------
    # index array
    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # symmetric edge‑exponential
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # ------------------------------------------------------------------
    # symmetric edge‑power‑law (decreasing)
    for p in np.linspace(0.2, 6.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # ------------------------------------------------------------------
    # asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=8)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ------------------------------------------------------------------
    # asymmetric two‑sided power‑law (decreasing)
    p_vals = np.linspace(0.2, 5.0, num=6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ------------------------------------------------------------------
    # **new**: power‑law with *positive* exponent (very heavy edges)
    for p in np.linspace(0.2, 4.0, num=10):
        edge = ((i + 1.0) / n) ** p + ((n - i) / n) ** p
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # ------------------------------------------------------------------
    # linear / triangular ramps
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base /= base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ------------------------------------------------------------------
    # half‑sine wave (already used but kept)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # logarithmic shape (edges dominate moderately)
    shape = np.log(i + 2.0) + np.log(n - i + 1.0)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # sparse spike vectors (few non‑zeros)
    for spikes in (3, 5, 8, 12):
        for _ in range(12):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: perturb logits with Gaussian noise,
    keep improvements.  Runs until the deadline.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local search] improved to {val:.6f}")

    return a_best, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.  Returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base grid size (fast enough for many candidates)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, warm_start.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm_start = np.interp(x_new, x_old, warm_start)
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of cheap candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    # add a handful of pure Dirichlet draws for extra diversity
    for _ in range(30):
        all_candidates.append(np.random.dirichlet(np.ones(n)))

    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep the top few for the expensive optimisation
    cand_vals.sort(key=lambda x: x[0])
    top_k = 8
    top_candidates = [c for _, c in cand_vals[:top_k]]
    if cand_vals:
        best_val, best_seq = cand_vals[0]

    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on the selected starts
    for idx, start in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic refinement
    if best_seq is not None and time.time() < deadline:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – if generous time remains, upscale to a larger grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 45.0:
        n_big = 8192
        conv_fft_big = _next_pow2(2 * n_big - 1)
        corr_fft_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            best_seq,
            best_val,
        )
        # final local refinement on the larger grid
        if best_seq is not None and time.time() < deadline:
            best_seq, best_val = _local_random_search(
                best_seq,
                n_big,
                conv_fft_big,
                deadline,
                best_val,
            )
        n = n_big
        conv_fft_len = conv_fft_big
        corr_fft_len = corr_fft_big

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
