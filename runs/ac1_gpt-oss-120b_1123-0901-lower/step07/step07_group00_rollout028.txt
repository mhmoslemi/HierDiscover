"""
Enhanced search for a low‑value step‑function.

The evaluation objective is

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

with a non‑negative vector a of length n.
The new algorithm adds a dedicated “flattening” phase that directly minimises
the ℓ₂‑distance between the convolution a∗a and the uniform target
   target = (sum a)² / (2·n‑1) .
Flattening drives every entry of the convolution towards the theoretical lower
bound, after which a short exact‑max sub‑gradient phase finishes the optimisation.
All updates stay on the probability simplex via exponentiated‑gradient
updates; the final result is always returned before the deadline expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _flatten_expgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimiser that minimises the ℓ₂‑distance of the
    convolution to the uniform target (i.e. it “flattens’’ the convolution).
    Returns best vector and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    # target for a perfectly flat convolution:
    #   Σ conv = (Σ a)² = 1  (after normalisation)
    #   there are 2·n‑1 entries, so each should be 1/(2·n‑1)
    const_target = 1.0 / (2 * n - 1)

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        # ℓ₂ loss (un‑scaled, constant offset does not affect gradient)
        err = conv - const_target
        loss = np.mean(err * err)

        # exact objective for monitoring
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Flatten it={it:06d}] loss {loss:.6e}  obj {val:.6f}")

        # gradient of ½·‖err‖² w.r.t a:
        #   d/da Σ err_k² = 2 Σ err_k ∂conv_k/∂a
        #   ∂conv_k/∂a_i = 2·(a ★ δ_{i})(k)  → we reuse cross‑correlation
        w = 2.0 * err                       # 2·err from the chain rule
        grad = 4.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad)

        # renormalise – protect against under‑flow
        s = a.sum()
        if s <= 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (same as the previous state‑of‑the‑art).

    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set for the exact max
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing that swaps a random fraction of mass between two
    coordinates and keeps the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.30)   # move up to 30 % of source

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  The routine respects the given time budget and always
    returns the best (non‑negative) sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ---- phase 1 : quick exploration of a few diverse starts ----------
    per_candidate_budget = (deadline - time.time()) * 0.20
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 2.0, time.time() + per_candidate_budget)

        # flattening (ℓ₂) phase – focuses on reducing variance of the convolution
        a_flat, _ = _flatten_expgd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=150_000,
            eta0=0.9,
        )
        # short exact‑max refinement
        a_ref, val_ref = _subgradient_optimize(
            a_flat,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=30_000,
            eta0=0.9,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[cand {cid}] candidate best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Deep refinement of the current best candidate
    if best_seq is not None and time.time() < deadline - 10:
        a_refined, val_refined = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            eta0=0.85,
        )
        if val_refined < best_val:
            best_val = val_refined
            best_seq = a_refined.copy()
            print(f"[Refine] sub‑grad long best {val_refined:.6f}")

    # ------------------------------------------------------------------
    # Very cheap random hill‑climb to squeeze a last bit of improvement
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid (8192 points) if time permits
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # flattening on the larger grid (short)
        a_big_flat, _ = _flatten_expgd_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
            eta0=0.9,
        )
        # final exact‑max refinement
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_flat,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
