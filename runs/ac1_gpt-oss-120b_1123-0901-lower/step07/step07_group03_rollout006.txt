"""
Propose a low‐value step function.

The algorithm builds a diverse set of warm‑starts (including the previous
best solution, Dirichlet samples, anti‑pair ordering, edge‑focused
shapes, etc.).  Each start is refined by a three‑stage continuous
optimisation:

1. Adam on a smooth‐max (log‑sum‑exp) surrogate.
2. Projected‑gradient descent on the exact max‑convolution (simplex
   projection after each step).
3. Stochastic local moves (swap / mass transfer).

After each stage the sequence is re‑ordered with the anti‑pair layout
(which is optimal for the centre shift) and a cheap permutation‑swap
search is performed.  If time permits the final solution is interpolated
to a finer grid (8192 points) and refined once more.

All operations respect the simplex constraint ∑a_i = 1 and a_i ≥ 0,
which makes the objective invariant to scaling.  The best sequence found
within the time budget is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    Returns the first 2·len(a)-1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    w has length 2·len(a)-1, a has length n.
    Returns a vector of length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, without the outer safety wrapper)
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)^2).  Assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al.)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum(x)=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _warm_start(n: int):
    """Retrieve the previous best sequence if available."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place sorted values opposite each other (largest ↔ smallest)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # median in centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample followed by anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _exponential_edges_candidate(n: int, lam: float = 0.001) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    shape = shape / shape.sum()
    return shape

# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Parameters are deliberately tuned for quick convergence.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Softmax → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # Small jitter every few thousand iterations to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _projected_gradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the exact max‑convolution.
    Uses the simplex projection after each gradient step.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[ProjGrad it={it:06d}] new best {val:.6f}")

        # Active set for the exact max
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Step‑size schedule (decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # Gradient step and projection onto simplex
        a = a - eta * grad_a
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing on the exact objective.
    Moves:
        – swap two positions
        – transfer a random fraction (≤40 %) from one index to another
    Acceptance: always if improvement, otherwise with probability
    exp(−Δ/Temp) where Temp decays linearly.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # Choose move type
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % of mass at i
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # linear temperature schedule
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


def _permutation_local_search(
    vals_desc: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap permutation search: start from the anti‑pair layout of the
    sorted values and repeatedly try random swaps; keep any swap that
    improves the exact objective.
    """
    a = _anti_pair_arrangement(vals_desc)            # initial layout
    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_moves:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        a2 = a.copy()
        a2[i], a2[j] = a2[j], a2[i]
        val2 = _objective(a2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        it += 1
        if it % 10000 == 0:
            print(f"[PermSwap {it:05d}] best {best_val:.6f}")
    return best_a, best_val

def _reorder_and_keep_best(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Apply the anti‑pair reordering (which is optimal for the centre shift)
    and keep whichever variant has the lower exact objective.
    """
    a_reordered = _anti_pair_arrangement(a)
    val_orig = _objective(a)
    val_reord = _objective(a_reordered)
    if val_reord < val_orig:
        return a_reordered, val_reord
    return a, val_orig

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function that minimizes
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine respects the given time budget and returns the best
    sequence (as a Python list) found.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Core parameters (primary resolution)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Assemble a diverse pool of starter vectors
    candidates = []

    # Warm start (if any)
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Uniform
    candidates.append(_uniform_candidate(n))

    # Random Dirichlet with a range of concentration parameters
    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # Anti‑pair Dirichlet samples (often strong)
    for alpha in (0.1, 0.3, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # Edge‑focused shapes
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_exponential_edges_candidate(n, lam=0.0005))

    # Simple tent shape
    candidates.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool; keep the best starter
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        print(f"[Init {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Reserve a few seconds for the final up‑sampling stage
    final_upscale_deadline = deadline - 20.0

    # ------------------------------------------------------------------
    # 3) Continuous optimisation – Adam (smooth‑max)
    adam_deadline = min(deadline - 15.0, time.time() + 250.0)
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=300_000,
        lr=0.07,
        lam_start=0.5,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # Re‑order with anti‑pair and keep best
    best_seq, best_val = _reorder_and_keep_best(best_seq)

    # ------------------------------------------------------------------
    # 4) Projected‑gradient refinement (exact max)
    projgrad_deadline = min(deadline - 10.0, time.time() + 250.0)
    a_proj, val_proj = _projected_gradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        projgrad_deadline,
        max_iters=300_000,
        eta0=0.9,
    )
    if val_proj < best_val:
        best_val = val_proj
        best_seq = a_proj.copy()
        print(f"[ProjGrad] improved to {val_proj:.6f}")

    best_seq, best_val = _reorder_and_keep_best(best_seq)

    # ------------------------------------------------------------------
    # 5) Stochastic local search (swap / mass transfer)
    if time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        a_local, val_local = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=120_000,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[LocalSearch] improved to {val_local:.6f}")

    best_seq, best_val = _reorder_and_keep_best(best_seq)

    # ------------------------------------------------------------------
    # 6) Permutation‑only hill‑climbing on the sorted values
    if time.time() < deadline - 5.0:
        perm_deadline = deadline - 2.0
        vals_desc = np.sort(best_seq)[::-1]               # descending
        a_perm, val_perm = _permutation_local_search(
            vals_desc,
            n,
            conv_fft_len,
            perm_deadline,
            max_moves=80_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[PermSearch] improved to {val_perm:.6f}")

    best_seq, best_val = _reorder_and_keep_best(best_seq)

    # ------------------------------------------------------------------
    # 7) Optional upscale to a finer grid (8192) if time permits
    if time.time() < final_upscale_deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Adam phase on the large grid
        ad_big_deadline = min(deadline - 5.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            ad_big_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.07,
        )

        # Projected‑gradient refinement on the big grid
        proj_big_deadline = min(deadline, time.time() + 200.0)
        a_big_proj, val_big_proj = _projected_gradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            proj_big_deadline,
            max_iters=200_000,
            eta0=0.9,
        )

        if val_big_proj < best_val:
            best_val = val_big_proj
            best_seq = a_big_proj.copy()
            print(f"[BigGrid] improved to {val_big_proj:.6f}")

        # Final cheap local moves on the big grid (optional, limited time)
        if time.time() < deadline - 2.0:
            local_big_deadline = deadline - 2.0
            a_big_loc, val_big_loc = _local_random_search(
                best_seq,
                n_big,
                conv_fft_len_big,
                local_big_deadline,
                max_moves=60_000,
            )
            if val_big_loc < best_val:
                best_val = val_big_loc
                best_seq = a_big_loc.copy()
                print(f"[BigLocal] improved to {val_big_loc:.6f}")

    # ------------------------------------------------------------------
    # 8) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store the result for possible warm‑starts in future runs
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
