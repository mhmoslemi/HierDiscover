"""
Hybrid optimizer for the step‑function problem.

Algorithm overview
------------------
1. Warm‑start from a previously found solution (if available).
2. Exhaustive exploration of two simple parametric families:
   * Edge‑concentrated power‑law a_i ∝ (i·(n‑i+1))**α  (α∈[‑8,‑0.05])
   * Double‑exponential a_i ∝ exp(-γ·i)+exp(-γ·(n‑1‑i)) (γ∈[0.05,10])
   The best member of each family is kept.
3. A cheap massive random‑Dirichlet sweep on a small grid (n=512).  
   The top‑k candidates are linearly interpolated to the full size
   (n=4096) and refined with a short Adam run.
4. The overall best candidate is then polished with a long
   multiplicative sub‑gradient (exponentiated‑gradient) phase.
5. Finally a very cheap simulated‑annealing hill‑climber is run for
   the remaining time budget.

All phases respect the global 1000 s budget and the routine always
returns the best sequence found.  The implementation uses only NumPy
(and the standard library) and works on a single CPU core.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """
    Exact objective  F(a) = 2·n·max(a*a) / (sum(a)^2) .
    The input may be any non‑negative vector; it will be normalised
    internally.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a = a / s
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# ----------   Parametric families   ------------------------------------
def _power_law_shape(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ (i·(n‑i+1))**α , i = 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1.0)
    # use log‑exp to avoid overflow/underflow
    shape = np.exp(alpha * np.log(prod))
    s = shape.sum()
    if s == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= s
    return shape


def _double_exp_shape(gamma: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-γ·i)+exp(-γ·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    s = shape.sum()
    if s == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= s
    return shape


# ----------------------------------------------------------------------
# ----------   Adam optimiser (smooth‑max)   ---------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (∑a=1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (∑a=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random shake
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# ----------   Sub‑gradient (exponentiated gradient)   ---------------
def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update + renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random perturbation if we have been stuck for a while
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # (1) warm start, if a previous solution exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, None)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            val = _objective(warm, conv_fft_len)
            best_seq = warm.copy()
            best_val = val
            print(f"[Warm start] value {val:.6f}")

    # ------------------------------------------------------------------
    # (2) exhaustive scan of two simple families
    # -------------------------------------------------- power‑law --------------------------------------------------
    for alpha in np.linspace(-8.0, -0.05, 160):
        if time.time() >= deadline:
            break
        a = _power_law_shape(alpha, n)
        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Power α={alpha:.3f}] new best {val:.6f}")

    # -------------------------------------------------- double‑exp --------------------------------------------------
    for gamma in np.geomspace(0.05, 10.0, 160):
        if time.time() >= deadline:
            break
        a = _double_exp_shape(gamma, n)
        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[DoubleExp γ={gamma:.3f}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # (3) massive random Dirichlet sweep on a small grid (n_small=512)
    n_small = 512
    conv_fft_len_small = _next_pow2(2 * n_small - 1)

    top_k = 10                     # keep the 10 best small candidates
    heap: list[tuple[float, np.ndarray]] = []   # stores (-val, vector)

    max_small_samples = 200_000
    samples = 0
    while time.time() < deadline - 5 and samples < max_small_samples:
        # a random concentration parameter adds diversity
        conc = np.random.rand() * 2.5 + 0.1          # in [0.1, 2.6]
        a_small = np.random.dirichlet(np.full(n_small, conc))
        conv = _full_convolution(a_small, conv_fft_len_small)
        max_c = conv.max()
        val = 2.0 * n_small * max_c                # sum(a_small)=1
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, a_small))
        else:
            if -val > heap[0][0]:                  # better than current worst
                heapq.heapreplace(heap, (-val, a_small))
        samples += 1

    # Upscale the retained candidates and give each a short Adam refinement
    top_candidates = [item[1] for item in heap]
    for idx, cand_small in enumerate(top_candidates):
        if time.time() >= deadline - 5:
            break
        # linear interpolation to full size
        x_old = np.linspace(0.0, 1.0, n_small, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        a_big = np.interp(x_new, x_old, cand_small)
        a_big = np.clip(a_big, 0.0, None)
        a_big /= a_big.sum()

        # give the Adam optimiser a few seconds
        cand_deadline = min(deadline - 5.0, time.time() + 12.0)
        a_adam, val_adam = _adam_optimize(
            a_big, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=100_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Upscaled {idx}] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # (4) long sub‑gradient refinement on the current best
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_opt2(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Subgrad] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # (5) cheap simulated‑annealing hill‑climber for the remaining budget
    if best_seq is not None and time.time() < deadline - 5:
        a_cur = best_seq.copy()
        cur_val = best_val
        T = 1e-3                 # initial temperature
        decay = 0.99995
        it = 0
        while time.time() < deadline:
            i = np.random.randint(n)
            if a_cur[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            # move up to 20 % of the mass at i
            frac = np.random.rand() * 0.2
            delta = a_cur[i] * frac
            a_new = a_cur.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv.max()
            if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / (T + 1e-12)):
                a_cur = a_new
                cur_val = val_new
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a_cur.copy()
                    print(f"[SA iter={it}] new best {cur_val:.6f}")
            T *= decay
            it += 1
        print(f"[SA] finished after {it} iterations, best {best_val:.6f}")

    # ------------------------------------------------------------------
    # (6) final safety‑net
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip for the evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
