"""
Propose a low‑evaluation step function.

The algorithm works in four stages:

1. **Random‑parametric exploration** – a very fast generator creates a
   large variety of shapes (symmetric/asymmetric exponentials,
   power‑laws, linear/triangular, sinusoid, sparse spikes, Dirichlet
   draws, etc.).  Every candidate is normalised to sum = 1 and its exact
   objective (the max‑convolution bound) is evaluated.  The best
   ~10 candidates are retained.

2. **Deep optimisation** – the current best candidate is refined by two
   successive continuous optimisers:
      * Adam on a smooth‑max (log‑sum‑exp) surrogate of the max‑convolution.
        The temperature (λ) grows slowly, allowing a smooth transition to
        the true max.  The optimisation runs as long as time permits.
      * An exponentiated‑gradient sub‑gradient method on the exact max‑convolution.
   Both phases respect the simplex constraint (non‑negative, sum = 1).

3. **Local stochastic hill‑climbing** – small Gaussian perturbations of the
   logits of the best vector are accepted if they improve the exact bound.
   This cheap refinement often yields a few‑thousandths improvement.

4. **Optional up‑sampling** – if generous time remains a finer grid
   (n = 8192) is created by linear interpolation and a short deep‑optimisation
   is performed there, possibly further lowering the bound.

All heavy loops are time‑aware: they stop as soon as the global deadline
(≈ 1000 s) is reached.  The function finally returns a plain Python list
containing the best non‑negative heights found.
"""

import time
import math
import numpy as np
import heapq

# ----------------------------------------------------------------------
# Helper linear‑algebra / convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(a*a)/(sum(a)²).  Assumes a ≥ 0."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Continuous optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 600_000,
    lr: float = 0.05,
    lam_start: float = 1.0,
    lam_factor: float = 1.005,
    lam_cap: float = 1e6,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    Returns (best_sequence, best_value).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_seq = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate objective (exact value needed for best‑track)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d}, new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # chain‑rule for logits (mirror descent on simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck for >5 s
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.8,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the exact max‑convolution.
    Returns (best_sequence, best_value).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta = eta0
    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d}, new best {val:.6f}")

        # active set = indices attaining the maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # stop early if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑smooth‑max phase followed by a sub‑gradient phase.
    Returns the best sequence/value seen.
    """
    # Adam phase (more iterations than the reference code)
    best_seq, best_val = _adam_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iter=600_000,
        lr=0.05,
        lam_start=1.0,

    )
    # Sub‑gradient phase (more iterations)
    best_seq, best_val = _subgrad_optimize(
        best_seq,

        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iter=200_000,
        eta0=0.8,
        decay=0.9999,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
# Random‑parametric candidate generation
# ----------------------------------------------------------------------
def _random_candidate(n: int) -> np.ndarray:
    """
    Produce a single random shape on the simplex.
    A wide palette of families is sampled uniformly.
    """
    i = np.arange(n, dtype=np.float64)
    family = np.random.randint(0, 10)

    if family == 0:                                   # uniform
        a = np.full(n, 1.0)

    elif family == 1:                                 # Dirichlet
        conc = 10 ** np.random.uniform(-3, 1)          # 0.001 … 10
        a = np.random.dirichlet(np.full(n, conc))

    elif family == 2:                                 # symmetric exponential
        alpha = 10 ** np.random.uniform(-3, 1)
        a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))

    elif family == 3:                                 # symmetric power‑law
        p = np.random.uniform(0.1, 8.0)
        a = (i + 1) ** (-p) + (n - i) ** (-p)

    elif family == 4:                                 # asymmetric exponential mixture
        aL = 10 ** np.random.uniform(-3, 1)
        aR = 10 ** np.random.uniform(-3, 1)
        wL = np.random.rand()
        wR = np.random.rand() * (1.0 - wL)
        wU = 1.0 - wL - wR
        left = np.exp(-aL * i)
        right = np.exp(-aR * (n - 1 - i))
        a = wL * left + wR * right + wU

    elif family == 5:                                 # asymmetric power‑law mixture
        pL = np.random.uniform(0.1, 8.0)
        pR = np.random.uniform(0.1, 8.0)
        wL = np.random.rand()
        wR = np.random.rand() * (1.0 - wL)
        wU = 1.0 - wL - wR
        left = (i + 1) ** (-pL)
        right = (n - i) ** (-pR)
        a = wL * left + wR * right + wU

    elif family == 6:                                 # linear decreasing
        a = np.linspace(1.0, 0.0, n)

    elif family == 7:                                 # triangular (mountain)
        half = n // 2
        left = np.linspace(0.0, 1.0, half, endpoint=False)
        right = np.linspace(1.0, 0.0, n - half)
        a = np.concatenate([left, right])

    elif family == 8:                                 # half‑wave sinusoid
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)

    else:                                             # sparse spikes
        spikes = np.random.choice([3, 5, 8, 12])
        idx = np.random.choice(n, size=spikes, replace=False)
        a = np.zeros(n)
        a[idx] = np.random.dirichlet(np.ones(spikes))

    # Mix with a uniform background to stay away from extreme sparsity
    mix = np.random.rand()
    a = mix * a + (1.0 - mix) / n

    # Clamp negatives (should not appear) and normalise
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Stochastic hill‑climbing on logits
# ----------------------------------------------------------------------
def _local_refinement(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: Gaussian jitter on the logits of the current best.
    Accepts a move only if it strictly improves the exact objective.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises the supplied
    `evaluate_sequence` objective.  The routine respects a ~1000 s wall‑time
    budget and returns the best sequence it has found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a working resolution (the reference used n=4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] != n:
            if warm_start.shape[0] > n:
                warm_start = warm_start[:n]
            else:
                warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – massive random parametric search, keep top‑k candidates
    top_k = 12
    top_heap = []                     # min‑heap on objective (largest at top)
    def _push_candidate(val, cand):
        if len(top_heap) < top_k:
            heapq.heappush(top_heap, (-val, cand))
        else:
            # heap[0] stores the worst (most negative) value, i.e. largest objective
            if -val > top_heap[0][0]:
                heapq.heapreplace(top_heap, (-val, cand))

    # Evaluate warm start first (if present)
    if warm_start is not None:
        val_ws = _objective(warm_start, n, conv_fft_len)
        _push_candidate(val_ws, warm_start)
        print(f"[Init] warm‑start objective = {val_ws:.6f}")

    # Random exploration – keep running while we have a healthy time buffer
    min_time_for_deep = 250.0   # seconds we want to keep for deep optimisation later
    while time.time() < deadline - min_time_for_deep:
        cand = _random_candidate(n)
        val = _objective(cand, n, conv_fft_len)
        _push_candidate(val, cand)

        # occasional progress report
        if len(top_heap) == top_k and (time.time() % 30) < 0.1:
            cur_best = -top_heap[0][0]
            print(f"[Search] current heap best = {cur_best:.6f}")

    # Extract the sorted list of top candidates (best first)
    top_candidates = [cand for (_, cand) in sorted(top_heap, key=lambda x: -x[0])]
    if not top_candidates:
        # fallback to uniform if everything failed
        top_candidates = [np.full(n, 1.0 / n)]

    best_seq = top_candidates[0].copy()
    best_val = _objective(best_seq, n, conv_fft_len)
    print(f"[Stage 1] evaluated candidates, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the current best candidate
    # (If time permits, also try the second‑best candidate)
    deep_start = time.time()
    remaining = deadline - time.time()
    # allocate roughly half of the remaining time to the top candidate,
    # half to a possible second candidate
    if remaining > 30:
        sub_deadline = time.time() + remaining * 0.55
        best_seq, best_val = _deep_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
        )
        print(f"[Deep‑opt 1] best objective = {best_val:.6f}")

    # If we still have time, try a second best start
    if len(top_candidates) > 1 and time.time() < deadline - 10:
        sec_seq = top_candidates[1].copy()
        sec_deadline = deadline - 5.0
        sec_seq, sec_val = _deep_optimize(
            sec_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sec_deadline,
        )
        print(f"[Deep‑opt 2] candidate 2 objective = {sec_val:.6f}")
        if sec_val < best_val:
            best_seq, best_val = sec_seq, sec_val
            print("[Deep‑opt] second candidate overtook the best.")

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic refinement
    if time.time() < deadline - 5.0:
        print("[Local] starting stochastic hill‑climbing")
        best_seq, best_val = _local_refinement(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – optional up‑sampling if generous time remains
    time_left = deadline - time.time()
    if time_left > 45.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Interpolate to the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        up_deadline = deadline - 5.0
        print("[Upsample] fine optimisation on n=8192")
        a_big, val_big = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            up_deadline,
        )
        # Keep the larger vector only if it improves the bound
        if val_big < best_val:
            best_seq = a_big
            best_val = val_big
            n = n_big
            print(f"[Upsample] improvement found: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final safety – ensure sequence respects evaluate_sequence's clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
