def cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)

"""
Hybrid global‑search with deep local refinement.

The algorithm proceeds in three phases under the given time budget:

1. **Diverse warm‑starts** – a pool of analytically‑motivated shapes
   (uniform, Dirichlet, power‑law, arcsine, double‑Gaussian edges,
   half‑sine, regularly spaced spikes) is assembled.  Each candidate is
   normalised to sum‑one and receives a short Adam optimisation on the
   smooth‑max surrogate (log‑sum‑exp).  The best few are retained.

2. **Intensive local optimisation** – starting from the current global
   best we run a long Adam pass (exactly the same surrogate as in the
   short runs, but with many more iterations), followed by an
   exponentiated‑gradient sub‑gradient descent that works on the true
   max‑convolution (exact max).  The sequence is periodically symmetrised
   (a ↦ (a+a[::-1])/2) which never harms the objective and often improves
   it.

3. **Fast simulated‑annealing fine‑tuning** – a mass‑transfer SA that
   updates the convolution in O(n) per elementary move.  The acceptance
   temperature follows a linear schedule and the move magnitude shrinks
   as the algorithm cools.

If enough time remains the routine works on a finer grid (n=8192); otherwise
it stays at n=4096.  The final sequence is clipped to the limits enforced by
`evaluate_sequence` and returned as a plain Python list.

All heavy convolution work is performed with FFTs for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    # warm start & a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform) and a spiky Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike families (k equally‑spaced non‑zero entries)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        pos = np.unique(pos)  # avoid duplicates
        a[pos] = 1.0 / pos.size
        cand.append(a)

    return cand

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in the log‑space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3                # temperature for smooth‑max
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # recover a from logits (simple softmax on the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c          # sum(a) == 1 by construction
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # transform to gradient w.r.t. logits (mirrored‑KL geometry)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal the temperature
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter near the end to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set of tight convolution entries
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that works directly on the convolution (O(n) per move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling from 0.1 down to 1e‑6
        T = max(1e-6, 0.1 * (1.0 - fraction))
        # shrink move magnitude as we cool
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # source index chosen proportionally to its current mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue

        # target index uniformly among the others
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass to relocate
        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution in O(n) (see comments in the original code)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
        2·n·max(a∗a) / (sum a)^2 .
    The routine respects the supplied time budget and always returns a valid
    sequence (non‑negative, finite length).  It blends a rich pool of
    warm‑starts, Adam on a smooth‑max surrogate, exact‑max exponentiated‑
    gradient descent and a fast simulated‑annealing local search.  If the
    budget allows, the optimisation is performed on a finer grid (n=8192);
    otherwise n=4096 is used.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # Choose grid size based on available time
    if budget_s > 800:
        n = 8192
    else:
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously discovered sequence if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, np.inf)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Assemble a modestly sized yet diverse pool of initial candidates
    candidates = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    start_time = time.time()
    remaining = len(candidates)

    # ------------------------------------------------------------------
    # Shallow optimisation for each candidate (quick Adam pass)
    per_cand_time = min(20.0, (deadline - time.time()) / max(1, remaining))
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        local_deadline = min(deadline - 5.0, time.time() + per_cand_time)

        a_opt, val_opt = _adam_optimize(a0, n, conv_fft_len,
                                        corr_fft_len, local_deadline,
                                        max_iters=100_000)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[cand {cid}] shallow Adam improved best to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Intensive refinement of the global best (if time permits)
    if best_seq is not None and time.time() < deadline - 15:
        # Long Adam optimisation
        a_opt, val_opt = _adam_optimize(best_seq, n, conv_fft_len,
                                        corr_fft_len, deadline,
                                        max_iters=300_000)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[global Adam] improved best to {val_opt:.6f}")

        # Exact‑max exponentiated‑gradient descent
        if time.time() < deadline - 5:
            a_opt, val_opt = _subgradient_optimize(best_seq, n,
                                                    conv_fft_len,
                                                    corr_fft_len,
                                                    deadline,
                                                    max_iters=150_000)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[global Subgrad] improved best to {val_opt:.6f}")

        # Symmetrisation (keeps sum and non‑negativity)
        sym = (best_seq + best_seq[::-1]) * 0.5
        conv_sym = _full_convolution(sym, conv_fft_len)
        val_sym = 2.0 * n * conv_sym.max()
        if val_sym < best_val:
            best_val = val_sym
            best_seq = sym.copy()
            print(f"[symmetrisation] improved best to {val_sym:.6f}")

        # Fast simulated‑annealing fine‑tuning
        if time.time() < deadline - 5:
            a_opt, val_opt = _simulated_annealing(best_seq, n,
                                                  conv_fft_len,
                                                  deadline,
                                                  best_val)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[global SA] improved best to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Final reporting (uses the judge's routine if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal best value {best_val:.6f} (length {len(best_seq)})")

    # Return a plain Python list as required by the interface
    return best_seq.tolist()
