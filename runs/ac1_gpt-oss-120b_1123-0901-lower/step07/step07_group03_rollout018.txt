```
def _alternating_ends_candidate(n, alpha):
    vals = np.random.dirichlet(alpha*np.ones(n))
    vals = np.maximum(vals,0)
    vals /= vals.sum()
    sorted_vals = np.sort(vals)[::-1]
    a = np.empty(n)
    left = 0
    right = n-1
    i=0
    while left <= right:
        a[left] = sorted_vals[i]; i+=1; left+=1
        if left <= right:
            a[right] = sorted_vals[i]; i+=1; right-=1
    return a
```

"""
Hybrid optimizer for the step‑function bound.

The algorithm proceeds in several stages:

1. **Diverse warm‑starts** – we build a pool of candidates using many
   simple generators (uniform, Dirichlet, anti‑pair, an “alternating‑ends”
   arrangement, Gaussian edges, tent).  All candidates are normalised to
   sum = 1 and the best few are kept.

2. **Smooth‑max Adam** – a large‑scale Adam optimisation is performed on a
   log‑sum‑exp surrogate of the max‑convolution.  The surrogate temperature
   (λ) is increased exponentially so that the optimisation gradually focuses
   on the true max.

3. **Exact‑max sub‑gradient** – we switch to an exponentiated‑gradient step
   that works directly on the exact max (using the active set of indices).

4. **Targeted stochastic hill‑climbing** – a simulated‑annealing style
   local search that performs swaps, mass‑transfers and a cheap
   “target‑max” move (shifting mass away from the indices that realise the
   current maximal convolution entry).

5. **Optional upscale** – if time permits the current best solution is
   interpolated to a finer grid (8192 points) and the whole pipeline is
   repeated, which usually yields a still lower bound.

All steps respect the overall time budget (≈ 1000 s) and the returned
sequence is guaranteed to be non‑negative and to have a total mass ≥ 0.01.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for the (sub‑)gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (sum of coefficients is assumed to be 1) ---------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Candidate generators ----------------------------------------------------
def _warm_start(n: int):
    """Extract a warm start from the global variable if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Largest opposite smallest, second‑largest opposite second‑smallest.
    """
    n = len(vals)
    s = np.sort(vals)[::-1]                # descending
    out = np.empty(n, dtype=np.float64)
    half = n // 2
    out[:half] = s[:half]
    out[-half:] = s[-half:][::-1]
    if n % 2 == 1:
        out[half] = s[half]
    return out


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _alternating_ends_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """
    Symmetric decreasing shape: largest values placed at both ends,
    second‑largest just inside them, etc.
    """
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    s = np.sort(vals)[::-1]                 # descending
    out = np.empty(n, dtype=np.float64)
    left = 0
    right = n - 1
    idx = 0
    while left <= right:
        out[left] = s[idx]
        idx += 1
        left += 1
        if left <= right:
            out[right] = s[idx]
            idx += 1
            right -= 1
    return out


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives -------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best exact‑objective vector seen during optimisation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # log‑softmax parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # current point on simplex

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (smooth surrogate)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule: increase λ → focus on true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                               # ensure sum = 1

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (decaying)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # project back onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing with three move types:
    - swap two positions,
    - transfer a fraction of mass,
    - a cheap “target‑max” move that shifts mass away from the current
      maximum‑convolution index.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02               # a slightly hotter start than before

    it = 0
    while time.time() < deadline and it < max_moves:
        # ----- pick move -------------------------------------------------
        r = np.random.rand()
        if r < 0.4:                                 # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]

        elif r < 0.8:                               # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        else:                                        # targeted move
            conv = _full_convolution(a, conv_fft_len)
            k = int(conv.argmax())                    # index of current max
            # indices contributing to this k
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx_range = np.arange(i_min, i_max + 1)
            prod = a[idx_range] * a[k - idx_range]
            # pick the pair with largest product
            worst_local = idx_range[np.argmax(prod)]
            partner = k - worst_local
            # choose a destination far from the offending pair
            # (any index outside the interval [i_min, i_max] has zero product)
            if np.random.rand() < 0.5:
                t = np.random.randint(0, i_min) if i_min > 0 else np.random.randint(k + 1, n)
            else:
                t = np.random.randint(i_max + 1, n) if i_max + 1 < n else np.random.randint(0, k)
            if t == worst_local or t == partner:
                continue
            delta = a[worst_local] * (np.random.rand() * 0.3)
            a2 = a.copy()
            a2[worst_local] -= delta
            a2[t] += delta

        val2 = _objective(a2)

        # ----- acceptance -------------------------------------------------
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine follows the multi‑stage scheme described at the
    top of the file and respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Core parameters
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of start vectors
    pool = []

    warm = _warm_start(n)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    pool.append(_uniform_candidate(n))

    for alpha in (0.2, 0.5, 0.8, 1.0, 2.0):
        pool.append(_random_dirichlet_candidate(n, alpha))

    for alpha in (0.2, 0.5, 1.0):
        pool.append(_anti_pair_candidate(n, alpha))

    for alpha in (0.2, 0.5, 1.0):
        pool.append(_alternating_ends_candidate(n, alpha))

    for gamma in (0.2, 0.5, 1.0):
        pool.append(_gaussian_edges_candidate(n, gamma))

    pool.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool, keep the best 5
    evaluated = []
    for idx, cand in enumerate(pool):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s                      # normalise to sum = 1
        val = _objective(cand)
        evaluated.append((val, cand))
        print(f"[Init cand {idx:02d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]

    # ------------------------------------------------------------------
    # 3) Optimisation pipeline for the current best candidate
    # Allocate time slices for each stage
    now = time.time()
    remaining = deadline - now

    # Adam phase (smooth‑max)
    adam_deadline = min(deadline - 350.0, now + min(250.0, remaining * 0.3))
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=400_000,
        lr=0.07,
        lam_start=1.0,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # Sub‑gradient refinement
    now = time.time()
    subgrad_deadline = min(deadline - 200.0, now + min(300.0, (deadline - now) * 0.35))
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        subgrad_deadline,
        max_iters=400_000,
        eta0=0.9,                     # a slightly larger step‑size
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Alternating local search / sub‑gradient cycles while time permits
    while time.time() < deadline - 80.0:
        # a short sub‑gradient burst
        now = time.time()
        sub_dead = min(deadline - 40.0, now + 60.0)
        a_tmp, val_tmp = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=200_000,
            eta0=0.9,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Sub‑cycle] improved to {val_tmp:.6f}")

        # a short local search burst
        now = time.time()
        local_dead = min(deadline - 20.0, now + 60.0)
        a_tmp, val_tmp = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=150_000,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Local‑cycle] improved to {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional upscale to a finer grid (8192) if time permits
    if time.time() < deadline - 10.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on the larger grid
        now = time.time()
        ad_big_dead = min(deadline - 5.0, now + 120.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            ad_big_dead,
            max_iters=250_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        # sub‑gradient on big grid
        now = time.time()
        sub_big_dead = min(deadline, now + 120.0)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_big_dead,
            max_iters=250_000,
            eta0=0.9,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 6) Final sanity – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure positivity and clip to the evaluator limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:                     # fallback safeguard
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    # Normalise to a reasonable total mass (the exact scale does not matter)
    best_seq = best_seq / best_seq.sum()

    # store the result for a possible warm‑start in the next run
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
