"""
Propose a low‑value step‑function for the objective

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The approach is three‑phased:

1. **Coarse parametric sweep** – families of analytically‑motivated
   shapes (Beta‑U shapes, edge‑Gaussian, mixtures with uniform) are
   generated for a relatively large grid (n = 8192).  Each candidate is
   normalised to sum‑one and its objective is evaluated via a fast FFT‑
   based convolution.  The best candidate from this sweep seeds the next
   stage.

2. **Smooth‑max Adam optimisation** – the best candidate is refined with
   Adam on the log‑sum‑exp surrogate of the max.  The temperature is
   increased very slowly, allowing the optimiser to see a smooth
   landscape early on and gradually focus on the true max.

3. **Exact‑max sub‑gradient + simulated annealing** – a final
   exponentiated‑gradient sub‑gradient descent eliminates any remaining
   bias from the surrogate, followed by a fast O(n)‑per‑move simulated
   annealing that directly manipulates the convolution.

All work respects the supplied time budget; any remaining time is used
for a short final simulated‑annealing pass.  The returned sequence is
clipped to the limits enforced by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                 # a little more conservative than the baseline
    lam = 0.05                # start temperature (smooth‑max softness)
    lam_factor = 1.01         # very slow annealing

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax representation of a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # tiny jitter near the end to shake out plateaus
        if it % 5000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-4

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast simulated‑annealing local search (O(n) per move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling
        T = max(1e-7, 0.2 * (1.0 - fraction))
        move_scale = max(0.01, 0.5 * (1.0 - fraction))

        # pick source proportionally to its current mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # propose move
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution (linear‑time slice updates)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the given evaluation
    function.  The routine performs a parametric sweep over analytically
    motivated families (U‑shaped Beta densities, edge Gaussians, mixtures)
    on a fairly large grid (n = 8192), then refines the best candidate with
    Adam, sub‑gradient descent and simulated annealing.  The final sequence
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # grid size – larger grid gives a lower theoretical bound (≈1)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Helper to evaluate the objective for a normalised vector
    def eval_obj(a: np.ndarray) -> float:
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 1) Warm‑start (optional)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # interpolate / trim to the current length
        if warm.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, warm.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # 2) Coarse parametric sweep
    best_val = float("inf")
    best_seq = None

    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)                 # (i+½)(n‑½‑i) > 0

    # ----- Beta‑U family (symmetric Beta(p,p) → p‑1 = α, α∈(-1,0)) -----
    for alpha in np.linspace(-0.95, -0.05, 181):      # 0.005 step
        if time.time() >= deadline:
            break
        shape = prod ** alpha
        a = shape / shape.sum()
        val = eval_obj(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[beta] α={alpha:.3f}  val={val:.6f}")

    # ----- Edge‑Gaussian family (two mirrors) -----
    for gamma in np.linspace(0.1, 2.0, 39):
        if time.time() >= deadline:
            break
        shape = np.exp(-gamma * i ** 2) + np.exp(-gamma * (n - 1 - i) ** 2)
        a = shape / shape.sum()
        val = eval_obj(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[gauss] γ={gamma:.3f}  val={val:.6f}")

    # ----- Random low‑concentration Dirichlet draws (exploratory) -----
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.full(n, 0.2))
        val = eval_obj(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[dirichlet] val={val:.6f}")

    # ----- Mixture of uniform and Beta‑U shapes -----
    uniform = np.full(n, 1.0 / n)
    for lam in np.linspace(0.0, 1.0, 11):            # mixing weight
        for alpha in np.linspace(-0.90, -0.10, 81):
            if time.time() >= deadline:
                break
            shape = lam * uniform + (1.0 - lam) * (prod ** alpha)
            a = shape / shape.sum()
            val = eval_obj(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[mix] λ={lam:.2f} α={alpha:.3f}  val={val:.6f}")

    # ----- Warm‑start candidate -----
    if warm is not None:
        val = eval_obj(warm)
        if val < best_val:
            best_val = val
            best_seq = warm.copy()
            print(f"[warm] val={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Refine the current best with Adam (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 30:
        a_adam, val_adam = _adam_optimize(best_seq, n,
                                          conv_fft_len, corr_fft_len,
                                          deadline, max_iters=250_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam refine] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact‑max sub‑gradient descent
    if best_seq is not None and time.time() < deadline - 20:
        a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                               conv_fft_len, corr_fft_len,
                                               deadline, max_iters=180_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Sub‑grad refine] new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Simulated‑annealing local search (if any budget left)
    if best_seq is not None and time.time() < deadline - 5:
        a_sa, val_sa = _simulated_annealing(best_seq, n,
                                            conv_fft_len,
                                            deadline,
                                            best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA final] new best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Enforce the same clipping that evaluate_sequence applies
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final report (uses the judge's routine if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Could not call evaluate_sequence ({e}); internal best {best_val:.6f}")

    return best_seq.tolist()
