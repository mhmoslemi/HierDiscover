def _solve_lp_max_sum(f, M, time_limit):
    n = len(f)
    # Build sparse matrix
    from scipy.sparse import diags
    offsets = np.arange(n)
    data = [np.full(n, f[d]) for d in range(n)]
    T = diags(data, offsets, shape=(2*n-1, n), format='csr')
    c = -np.ones(n)  # maximize sum -> minimize -sum
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=T, b_ub=b, bounds=(0, None), method='highs', options={'time_limit': time_limit})
    if not res.success:
        print("LP solver failed:", res.message)
        return None
    g = res.x
    return g

"""
Propose a step‑function that minimizes
    F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm improves upon the previous runner by adding a
linear‑programming “mass‑inflation’’ stage.  For a fixed shape a,
the convolution bound M = max(a∗a) is known.  Solving

    maximise   sum(g)
    subject to  a∗g ≤ M      (elementwise)
                g ≥ 0

gives a vector whose total mass can be larger than that of a while
still respecting the same max‑convolution bound.  After normalising
g back to unit sum the new max‑convolution drops to M / (sum(g)²),
which strictly improves the objective whenever sum(g)>1.
The LP is solved with SciPy’s high‑performance ``linprog`` (HiGHS).

The overall pipeline:

1. Build a diverse pool of start vectors (Dirichlet draws,
   anti‑pair ordering, alternating high/low, triangle, edge‑Gaussians,
   warm‑start if present) and keep the best one (exact objective).

2. Refine the best start with a smooth‑max Adam phase followed by an
   exact‑max exponentiated‑gradient sub‑gradient phase (identical to
   the previous runner but with a few more iterations).

3. Apply the LP inflation step repeatedly (as time permits).  After each
   successful inflation the current vector is re‑normalised and a short
   Adam/sub‑gradient polishing is performed.

4. Finish with a stochastic hill‑climbing (simulated‑annealing) that
   swaps entries or transfers mass.

5. If time allows, interpolate to a finer grid (8192 points) and repeat
   the Adam/sub‑gradient refinement.

All sequences are kept non‑negative, summed to 1 (scale‑invariant) and
clipped to the range accepted by ``evaluate_sequence`` before returning.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# optional SciPy imports (used only for the LP stage)

    from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = rfft(w, fft_len)
    fr = rfft(rev_a, fft_len)
    corr = irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant: sum(a) is not assumed to be 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Anti‑pair layout: largest half on the left, smallest half on the right.
    ``vals`` must already sum to 1.
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # median in centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _alternating_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """
    Interleave the sorted values: largest, smallest, second‑largest,
    second‑smallest, … .  This often yields a more balanced convolution.
    """
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    sorted_vals = np.sort(vals)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
    return a


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _build_initial_pool(n: int):
    """Return a list of diverse start vectors (all normalised to sum = 1)."""
    pool = []

    warm = _warm_start(n)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    pool.append(_uniform_candidate(n))

    for alpha in (0.3, 0.7, 1.0, 2.0):
        pool.append(_random_dirichlet_candidate(n, alpha))

    for alpha in (0.3, 0.7, 1.0):
        pool.append(_anti_pair_candidate(n, alpha))

    for alpha in (0.3, 0.7, 1.0):
        pool.append(_alternating_candidate(n, alpha))

    for gamma in (0.2, 0.5, 1.0):
        pool.append(_gaussian_edges_candidate(n, gamma))

    pool.append(_triangle_candidate(n))

    # ensure every entry is non‑negative and normalised
    cleaned = []
    for v in pool:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 0.01:            # safety fallback
            v = np.random.dirichlet(np.ones(n))
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation (exponentiated‑gradient updates).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
      – random swap of two positions
      – or transfer a fraction of mass from one index to another.
    Accepted if it improves the exact objective, or with a temperature‑
    dependent probability.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % transfer
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Linear‑programming mass‑inflation stage
def _solve_lp_max_sum(f: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve   maximise sum(g)   subject to  f∗g ≤ M , g ≥ 0.
    Returns the optimal g (raw, not normalised) or ``None`` on failure.
    Requires SciPy.
    """
    if not SCIPY_AVAILABLE:
        return None

    n = f.shape[0]
    # Build the convolution matrix T such that T @ g = f∗g.
    # T has shape (2n‑1, n).  Entry T[k, j] = f[k‑j]  (if 0≤k‑j<n, else 0).
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    T = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                # minimise –sum(g)  == maximise sum(g)
    b = np.full(2 * n - 1, M)

    # HiGHS respects a time limit via the 'time_limit' option (seconds)
    res = linprog(
        c,
        A_ub=T,
        b_ub=b,
        bounds=(0, None),
        method="highs",
        options={"time_limit": time_limit},
    )
    if not res.success:
        print("[LP] solver failed:", res.message)
        return None

    return res.x

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a pool of starters and keep the best one.
    pool = _build_initial_pool(n)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, cand.copy()

    # ------------------------------------------------------------------
    # 2) Smooth‑max Adam refinement
    adam_deadline = min(deadline - 15.0, time.time() + 250.0)
    best_seq, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=350_000,
        lr=0.07,
        lam_start=1.0,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val, best_seq = val_adam, best_seq.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient refinement
    sub_deadline = min(deadline - 10.0, time.time() + 300.0)
    best_seq, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        sub_deadline,
        max_iters=350_000,
        eta0=0.7,
    )
    if val_sub < best_val:
        best_val, best_seq = val_sub, best_seq.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Linear‑programming mass‑inflation loop (as time permits)
    while SCIPY_AVAILABLE and time.time() < deadline - 30.0:
        # current bound
        conv = _full_convolution(best_seq, conv_fft_len)
        M = conv.max()

        # allocate a modest slice of the remaining budget to the LP
        lp_time = min(8.0, (deadline - time.time()) * 0.2)
        g = _solve_lp_max_sum(best_seq, M, lp_time)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1.02:
            # not enough inflation – give up on further LP attempts
            break

        # normalise and evaluate the new vector
        a_new = g / sum_g
        val_new = _objective(a_new)
        if val_new < best_val:
            best_val, best_seq = val_new, a_new.copy()
            print(f"[LP] improved to {best_val:.6f} (sum_g={sum_g:.3f})")
        else:
            # no improvement – abort further LP iterations
            break

        # brief polishing after the LP step
        polish_deadline = min(deadline - 5.0, time.time() + 30.0)
        best_seq, val_polish = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            polish_deadline,
            max_iters=120_000,
            lr=0.04,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_polish < best_val:
            best_val, best_seq = val_polish, best_seq.copy()
            print(f"[LP+Adam] improved to {best_val:.6f}")

        # a short sub‑gradient pass
        sub_deadline2 = min(deadline - 5.0, time.time() + 30.0)
        best_seq, val_sub2 = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline2,
            max_iters=120_000,
            eta0=0.7,
        )
        if val_sub2 < best_val:
            best_val, best_seq = val_sub2, best_seq.copy()
            print(f"[LP+Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Stochastic hill‑climbing (swap / mass transfer)
    if best_seq is not None and time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        best_seq, val_local = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=180_000,
        )
        if val_local < best_val:
            best_val, best_seq = val_local, best_seq.copy()
            print(f"[Local] improved to {val_local:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional upscale to a finer grid (8192) if time permits
    if time.time() < deadline - 20.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the big grid
        ad_big_deadline = min(deadline - 5.0, time.time() + 120.0)
        a_big, val_big = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            ad_big_deadline,
            max_iters=150_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        # sub‑gradient on the big grid
        sub_big_deadline = min(deadline, time.time() + 120.0)
        a_big, val_big = _subgradient_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_big_deadline,
            max_iters=150_000,
            eta0=0.7,
        )
        if val_big < best_val:
            best_val, best_seq = val_big, a_big.copy()
            print(f"[Big] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) Guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
