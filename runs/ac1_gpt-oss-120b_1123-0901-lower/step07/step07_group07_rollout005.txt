"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1. **Candidate seeding** – generate a rich pool of symmetric
   parametric vectors (uniform, Dirichlet draws, exponentials,
   power‑laws, beta‑shapes, raised‑cosine, Gaussians, arcsine, …).
   All vectors are projected onto the simplex and quickly evaluated
   with an FFT‑based convolution.  The best few (≤12) are kept.

2. **Deep optimisation** – each selected seed is refined by two
   stages:
   * Adam on a smooth‑max surrogate of the convolution
     (log‑sum‑exp with an ever‑increasing temperature).
   * Exact sub‑gradient descent on the true max‑convolution,
     using exponentiated‑gradient updates.
   Both stages run until the time budget expires or a
   generous iteration limit is hit, keeping the best iterate.

3. **Upscaling** – the current global best is interpolated to a
   larger grid (n = 8192) and the deep optimisation is repeated
   with longer iteration limits.

4. **Targeted local refinement** – a cheap hill‑climbing loop that
   repeatedly moves a small amount of mass from the index that
   contributes most to the current maximal convolution value to a
   far‑away index.  Only improving moves are kept, and the loop
   stops when time runs out.

All FFT lengths are chosen as the next power‑of‑two for speed.
The implementation uses only NumPy and respects the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT of length
    ``fft_len`` (must be ≥ 2·len(a)-1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (sum(a)≈1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = float(conv.max())
    return 2.0 * n * max_c / (float(a.sum()) ** 2)

# Parametric candidate generation ---------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse pool of symmetric candidates.
    All vectors are normalised to have sum ≈ 1.
    """
    candidates: list[np.ndarray] = []

    # warm‑start & cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families
    for alpha in np.logspace(-2, 1.0, num=14):   # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

# Adam optimiser on smooth‑max -----------------------------------------
def _adam_smooth_max_generic(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # start in soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.2                 # start temperature (smaller → smoother)
    lam_factor = 1.02
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # soft‑max to obtain a feasible probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for progress monitoring)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[adam n={n}] it={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        # weights concentrate on entries close to the maximum
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (for the smooth surrogate) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # project gradient to the simplex‑tangent (soft‑max parameterisation)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck for >4 s
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

    max_iter: int,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution.
    Uses exponentiated‑gradient updates (keeps a ≥ 0, sum = 1).
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[subgrad n={n}] it={it:05d} new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise, protecting against numerical collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early‑stop if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Wrapper that runs both phases -----------------------------------------
def _deep_optimize_local(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    adam_iters: int = 200_000,
    subgrad_iters: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    # Phase 1 – Adam
    best_seq, best_val = _adam_smooth_max_generic(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        adam_iters,
        global_best_seq,
        global_best_val,
    )
    # Phase 2 – exact sub‑gradient descent
    best_seq, best_val = _subgradient_refinement_generic(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        subgrad_iters,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted local refinement ---------------------------------------------
def _targeted_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    current_best_val: float,
    max_attempts: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly moves a tiny amount of mass
    away from the index that contributes most to the current maximal
    convolution entry.  Only strictly improving moves are kept.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = current_best_val
    attempts = 0
    step_frac = 0.02                # percent of source mass to move

    while time.time() < deadline and attempts < max_attempts:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c / (a.sum() ** 2)

        # keep the best we have seen
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # locate a maximal convolution index
        m = int(np.argmax(conv))

        # contributions of each a[i] to this maximal entry
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j[mask]]

        src = int(np.argmax(contrib))
        if a[src] < 1e-12:
            # nothing useful to move – break early
            break

        # pick a far‑away target index
        tgt = (src + n // 2) % n
        if tgt == src:
            tgt = (src + n // 3) % n

        delta = a[src] * step_frac
        # ensure we do not create negative entries
        delta = min(delta, a[src])

        # perform the move
        a[src] -= delta
        a[tgt] += delta

        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Base grid size for the cheap seeding stage
    n_small = 4096
    conv_fft_small = _next_pow2(2 * n_small - 1)
    corr_fft_small = _next_pow2(3 * n_small - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered vector, if any
    warm_start = globals().get("height_sequence_1")
    ws_vec = None
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n_small:
            ws = ws[:n_small]
        else:
            ws = np.pad(ws, (0, n_small - ws.shape[0]), mode="constant")
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws /= ws.sum()
            ws_vec = ws

    # ------------------------------------------------------------------
    # Phase 1 – generate and pre‑select candidates
    cand_pool = _generate_parametric_candidates(n_small, ws_vec)
    evaluated = []
    for cand in cand_pool:
        if time.time() >= deadline:
            break
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n_small))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n_small, conv_fft_small)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Phase 1: evaluated {len(cand_pool)} candidates, kept {top_k}")

    # ------------------------------------------------------------------
    # Phase 2 – deep optimisation of the promising seeds
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize_local(
            start_vec,
            n_small,
            conv_fft_small,
            corr_fft_small,
            deadline,
            best_seq,
            best_val,
            adam_iters=180_000,
            subgrad_iters=40_000,
        )

    # ------------------------------------------------------------------
    # Phase 3 – if we still have a comfortable time slice, upscale
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_big = _next_pow2(2 * n_big - 1)
        corr_fft_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0.0, 1.0, n_small, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.clip(a_big, 0.0, None)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Upscaling to n=8192 and deep optimisation ---")
        best_seq, best_val = _deep_optimize_local(
            a_big,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            best_seq,
            best_val,
            adam_iters=300_000,
            subgrad_iters=80_000,
        )
        n_final = n_big
        conv_fft_final = conv_fft_big
    else:
        n_final = n_small
        conv_fft_final = conv_fft_small

    # ------------------------------------------------------------------
    # Phase 4 – final targeted local refinement
    if best_seq is not None and time.time() < deadline - 5.0:
        print("--- Final targeted refinement ---")
        best_seq, best_val = _targeted_refine(
            best_seq,
            n_final,
            conv_fft_final,
            deadline,
            best_val,
            max_attempts=70_000,
        )

    # ------------------------------------------------------------------
    # Fallback sanity check
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)

    # Clip to evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
