def make_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(2*n-1):
        j_start = max(0, i-(n-1))
        j_end = min(n-1, i) inclusive.
        # number of cols = j_end - j_start + 1
        for j in range(j_start, j_end+1):
            rows.append(i)
            cols.append(j)
            data.append(f[i-j])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def _conv_matrix_sparse(f):
    n = f.shape[0]
    # column for Toeplitz: f followed by zeros of length n-1
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row: f[0] then zeros
    r = np.concatenate([np.array([f[0]]), np.zeros(n - 1, dtype=f.dtype)])
    T = toeplitz(c, r)
    # Return as csr
    return csr_matrix(T)

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Method
------
1. **Coarse linear‑programming phase** (n≈1024)  
   – Start from a warm‑start or a random Dirichlet vector.  
   – Repeatedly (a) evaluate M = max(conv(a,a)).  
   – (b) solve a LP: maximise Σ g_j  subject to g≥0 and conv(a,g)≤M.  
   – (c) normalise g to have unit L1‑norm and blend it into a.  
   This monotone process usually pushes the bound below the previous
   best (~1.508) while being cheap (few seconds).

2. **Upsample to a fine grid (n=4096)** and perform a two‑stage
   smooth‑max Adam optimisation followed by an exact‑max exponentiated‑gradient
   sub‑gradient phase (identical to the state‑of‑the‑art code, but with a
   better warm‑start).

3. **Cheap random local moves** to escape tiny plateaus.

4. **Optional refinement on an even finer grid (n=8192)** if time remains.

All phases respect the overall deadline (default 1000 s) and always keep the
best sequence seen so far.  The returned list is clipped to the evaluator’s
accepted range (≥0, ≤1000) and has a total L1‑mass ≈1, guaranteeing a valid
score.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _max_convolution(a: np.ndarray) -> float:
    """Return max(a∗a) using an FFT‑based convolution."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    return float(conv.max())


def _conv_matrix_sparse(f: np.ndarray) -> csr_matrix:
    """
    Build the (2·n‑1)×n Toeplitz matrix that implements the linear map
    g ↦ conv(f,g).  The matrix is returned in CSR format for linprog.
    """
    n = f.shape[0]
    # first column : f followed by n‑1 zeros
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row  : f[0] followed by n‑1 zeros
    row = np.concatenate([np.array([f[0]]), np.zeros(n - 1, dtype=f.dtype)])
    T = toeplitz(col, row)          # dense (2n‑1)×n
    return csr_matrix(T)            # convert to sparse

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                 # a bit larger than before
    lam = 0.05                                 # start temperature (soft)
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----------- simplex projection (soft‑max) --------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ------------------- exact objective ------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ------------------- smooth‑max weights ----------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------- gradient of the surrogate ---------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ------------------- Adam update on logits --------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ------------------- temperature schedule --------------------
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake when we are close to the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay (very gentle)
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

    restarts = 3
    for r in range(restarts):
        if time.time() >= phase1_deadline:
            break
        # initial candidate for this restart
        if warm_coarse is not None and r == 0:
            a = warm_coarse.copy()
        else:
            a = np.random.dirichlet(np.ones(n_coarse))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # a small number of LP updates per restart
        for it in range(12):
            if time.time() >= phase1_deadline:
                break

            M = _max_convolution(a)                 # current max convolution
            val = 2.0 * n_coarse * M                 # sum(a)=1 ⇒ bound = 2·n·M
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[LP restart {r} it {it}] new best coarse {val:.6f}")

            # ----- linear‑programming step --------------------------------
            A_ub = _conv_matrix_sparse(a)           # (2n‑1)×n
            b_ub = np.full(2 * n_coarse - 1, M, dtype=np.float64)
            c_obj = -np.ones(n_coarse, dtype=np.float64)   # maximise Σg  → minimise −Σg
            bounds = [(0, None)] * n_coarse

            # give the solver a tiny time budget so we never overshoot the deadline
            remaining = phase1_deadline - time.time()
            if remaining < 0.3:
                break

            try:
                # recent versions of HiGHS accept a time_limit option
                res = linprog(
                    c_obj,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options={"time_limit": max(remaining - 0.1, 0.0)},
                )
            except Exception:                     # fallback for older SciPy versions
                res = linprog(
                    c_obj,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                )

            if not res.success:
                print(f"[LP restart {r} it {it}] linprog failed: {res.message}")
                break

            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 1e-12:
                break
            g = g0 / sum_g0                       # normalise to unit L1‑mass

            # smooth blend to avoid drastic jumps
            t = 0.5
            a = (1 - t) * a + t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    if best_seq is None:
        best_seq = np.full(n_coarse, 1.0 / n_coarse, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to a finer grid (n = 4096) and perform gradient refinements
    now = time.time()
    if now < deadline - 10.0:
        n_fine = 4096
        x_c = np.linspace(0, 1, n_coarse, endpoint=False)
        x_f = np.linspace(0, 1, n_fine, endpoint=False)
        a_fine = np.interp(x_f, x_c, best_seq)
        a_fine = np.maximum(a_fine, 0.0)
        a_fine = a_fine / a_fine.sum()

        # allocate time slices for the three fine‑grained phases
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.35
        conv_fft_len_fine = _next_pow2(2 * n_fine - 1)
        corr_fft_len_fine = _next_pow2(3 * n_fine - 2)

        a_adam, val_adam = _adam_optimize(
            a_fine,
            n_fine,
            conv_fft_len_fine,
            corr_fft_len_fine,
            adam_deadline,
            max_iters=300_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Fine Adam] improved best to {val_adam:.6f}")

        # sub‑gradient refinement
        sub_deadline = time.time() + remaining * 0.35
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n_fine,
            conv_fft_len_fine,
            corr_fft_len_fine,
            sub_deadline,
            max_iters=400_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Fine Subgrad] improved best to {val_sub:.6f}")

        # cheap random local search
        if time.time() < deadline - 5.0:
            a_loc, val_loc = _local_random_search(
                best_seq,
                n_fine,
                conv_fft_len_fine,
                deadline,
                max_moves=80_000,
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Fine Local] improved best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Optional extra refinement on a still larger grid (n = 8192)
    if time.time() + 30 < deadline:
        n_big = 8192
        x_cur = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_cur, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        big_deadline = time.time() + (deadline - time.time()) * 0.40
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=200_000,
        )

        # longer sub‑gradient phase on the huge grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=200_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big refine] improved best to {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Final safety clip and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best bound ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
