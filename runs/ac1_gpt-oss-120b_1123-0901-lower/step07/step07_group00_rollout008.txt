"""
Propose a step‑function (non‑negative vector) that minimizes
        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The procedure has three stages

1️⃣  **Fast parametric exploration (n≈2 k).**  
    Randomly generate a few hundred vectors from a rich family of
    parametric shapes (Beta‑pdf, exponential, cosine‑power,
    polynomial, Gaussian mixtures, Dirichlet, sparse spikes).  
    They are evaluated with the exact objective and the top‑k are kept.

2️⃣  **Refinement on the target size (n=4096).**  
    Each promising vector is interpolated to the target grid and
    improved by three optimisation phases that respect the global time
    budget:
        • Adam on the smooth‑max surrogate (log‑sum‑exp) – long enough
          to get close to a stationary point.
        • Exact‑max sub‑gradient descent (exponentiated‑gradient).
        • A cheap simulated‑annealing / hill‑climbing that moves a
          random fraction of mass between two coordinates, accepting
          uphill moves with a slowly decreasing temperature.

3️⃣  **Final clean‑up.**  
    The best vector found so far is refined a little longer with the
    sub‑gradient method and finally clipped to the evaluator’s allowed
    range.

All phases constantly check the remaining wall‑clock time, so the
function always returns a valid sequence before the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv)/(sum(a)^2)."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.05
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = _objective(a0, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through soft‑max (mirror‑descent form) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 40_000,
) -> tuple[np.ndarray, float]:
    """Very cheap annealed hill‑climbing (moves a random fraction of mass)."""
    a = a0.copy()
    cur_val = _objective(a, conv_fft_len)
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule (log‑linear)
    T0 = 0.2
    Tf = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = T0 * (Tf / T0) ** (it / max_moves)

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, conv_fft_len)

        # Metropolis acceptance
        if val2 < cur_val or np.random.rand() < np.exp(-(val2 - cur_val) / max(T, 1e-12)):
            a = a2
            cur_val = val2
            if cur_val < best_val:
                best_val = cur_val
                best_a = a2.copy()
                print(f"[SA move {it:06d}] improved to {cur_val:.6f}")

        it += 1

    return best_a, best_val

def _random_param_candidates(
    n: int,
    rng: np.random.Generator,
    max_candidates: int = 800,
) -> list[np.ndarray]:
    """Generate a diverse set of vectors from a handful of parametric families."""
    cand = []

    xs = np.linspace(0.0, 1.0, n, endpoint=False)

    # Dirichlet (full support)
    cand.append(rng.dirichlet(np.ones(n)))

    # Beta‑pdf family
    for _ in range(150):
        a = rng.uniform(0.1, 12.0)
        b = rng.uniform(0.1, 12.0)
        shape = (xs ** (a - 1)) * ((1 - xs) ** (b - 1))
        shape = np.where(np.isfinite(shape), shape, 0.0)
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        cand.append(shape)

    # Exponential / double‑exponential
    for _ in range(80):
        gamma = rng.uniform(0.05, 4.0)
        left = np.exp(-gamma * xs)
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Cosine‑power (raised‑cosine)
    for _ in range(80):
        p = rng.uniform(0.5, 12.0)
        shape = np.sin(np.pi * (xs + 0.5 / n)) ** p
        shape = np.maximum(shape, 0.0)
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        cand.append(shape)

    # Polynomial (x·(1‑x))‑power
    for _ in range(80):
        alpha = rng.uniform(-2.5, 2.5)
        prod = xs * (1.0 - xs)
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        cand.append(shape)

    # Gaussian mixture (1‑2 components)
    for _ in range(80):
        n_comp = rng.integers(1, 3)
        mix = np.zeros(n)
        total = 0.0
        for _ in range(n_comp):
            mu = rng.random()
            sigma = rng.uniform(0.03, 0.18)
            weight = rng.random()
            gauss = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
            mix += weight * gauss
            total += weight
        if total == 0:
            continue
        mix /= mix.sum()
        cand.append(mix)

    # Sparse equally spaced spikes (k = 32 … 512)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    # Small random perturbations of the warm‑start (if any) will be added later
    # (the warm‑start handling lives in propose_candidate).

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Fast parametric search on a modest grid (n≈2048)
    n_coarse = 2048
    conv_fft_len_coarse = _next_pow2(2 * n_coarse - 1)

    print("[Stage 1] Generating coarse candidates …")
    candidates = _random_param_candidates(n_coarse, rng, max_candidates=700)

    # optionally inject a warm‑start if the user supplied one
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_coarse:
            warm = warm[:n_coarse]
        elif warm.shape[0] < n_coarse:
            warm = np.pad(warm, (0, n_coarse - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            candidates.append(warm)

    best_coarse = None
    best_val = float("inf")
    # evaluate all coarse candidates (fast because n is moderate)
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        val = _objective(cand, conv_fft_len_coarse)
        if val < best_val:
            best_val = val
            best_coarse = cand.copy()
            print(f"[Coarse #{idx}] new best {val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Upsample the best coarse vectors and run heavy refinement
    #     target size: n_target = 4096 (as in the previous state‑of‑the‑art)
    n_target = 4096
    conv_fft_len = _next_pow2(2 * n_target - 1)
    corr_fft_len = _next_pow2(3 * n_target - 2)

    # Keep a small pool of the *best* coarse candidates for refinement.
    # We take the top‑3 (or whatever fits in the remaining time).
    pool_fine = []
    # Re‑evaluate coarse candidates with exact objective (still cheap)
    coarse_vals = [(cand, _objective(cand, conv_fft_len_coarse)) for cand in candidates]
    coarse_vals.sort(key=lambda x: x[1])
    for cand, _ in coarse_vals[:3]:
        # interpolate to the finer grid
        x_old = np.linspace(0, 1, n_coarse, endpoint=False)
        x_new = np.linspace(0, 1, n_target, endpoint=False)
        a_big = np.interp(x_new, x_old, cand)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 1e-12:
            a_big = np.random.dirichlet(np.ones(n_target))
        a_big /= a_big.sum()
        pool_fine.append(a_big)

    # If no candidate survived (unlikely), start from a uniform vector.
    if not pool_fine:
        pool_fine.append(np.full(n_target, 1.0 / n_target))

    best_seq = None
    best_seq_val = float("inf")

    # allocate fraction of remaining time for the refinement of each pool member
    while pool_fine and time.time() < deadline:
        a0 = pool_fine.pop(0)
        # Remaining budget for this candidate
        cand_deadline = min(deadline - 2.0, time.time() + (deadline - time.time()) * 0.35)

        # ------------------------------------------------------------------
        # Adam phase (smooth‑max)
        a_adam, val_adam = _adam_optimize(
            a0,
            n_target,
            conv_fft_len,
            corr_fft_len,
            cand_deadline,
            max_iters=180_000,
        )
        if val_adam < best_seq_val:
            best_seq_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Refine Adam] best {val_adam:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient phase (exact max)
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n_target,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=80_000,
                eta0=0.9,
            )
            if val_sub < best_seq_val:
                best_seq_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Refine Subgrad] best {val_sub:.6f}")

        # ------------------------------------------------------------------
        # Simulated‑annealing local search (cheap but can escape plateaus)
        if time.time() < deadline:
            a_sa, val_sa = _simulated_annealing(
                best_seq,
                n_target,
                conv_fft_len,
                deadline,
                max_moves=30_000,
            )
            if val_sa < best_seq_val:
                best_seq_val = val_sa
                best_seq = a_sa.copy()
                print(f"[Refine SA] best {val_sa:.6f}")

        # optional extra sub‑gradient pass if we still have time
        if time.time() < deadline - 5.0:
            a_sub2, val_sub2 = _subgradient_optimize(
                best_seq,
                n_target,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=120_000,
                eta0=0.85,
            )
            if val_sub2 < best_seq_val:
                best_seq_val = val_sub2
                best_seq = a_sub2.copy()
                print(f"[Extra Subgrad] best {val_sub2:.6f}")

    # ----------------------------------------------------------------------
    # 3️⃣  Final fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n_target, 1.0 / n_target, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_seq_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
