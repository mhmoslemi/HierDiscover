def generate_initial_candidates(n, warm=None):
    cand = []
    if warm is not None:
        cand.append(warm.copy())
    # uniform
    cand.append(np.full(n, 1.0/n))
    # arcsine
    i = np.arange(1, n+1)
    a = 1.0/np.sqrt(i * (n - i + 1))
    a = a / a.sum()
    cand.append(a)
    # triangular
    i = np.arange(n)
    mid = (n-1)/2
    a = 1 - np.abs(i - mid) / (mid+1)
    a = np.maximum(a, 0)
    a = a / a.sum()
    cand.append(a)
    # sine half-wave
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    cand.append(a)
    # sin^p for p in [...]
    for p in (2, 3, 4):
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        a = np.maximum(a, 0)
        a = a / a.sum()
        cand.append(a)
    # polynomial power law
    idx = np.arange(1, n+1)
    prod = idx * (n - idx + 1)

        a = prod ** alpha
        a = np.maximum(a, 0)
        if a.sum() > 0:
            a = a / a.sum()
            cand.append(a)
    # edge expo
    for gamma in (0.1, 0.3, 0.7, 1.5):
        left = np.exp(-gamma * np.arange(n))

        a = a / a.sum()

def simulated_annealing(a0, n, conv_fft_len, deadline, max_moves=100000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    current_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = current_val

    start_time = time.time()
    T0 = 0.1
    # schedule: T = T0 * (1 - (t / max_moves))
    for move in range(max_moves):
        if time.time() >= deadline:
            break
        i = np.random.randint(n)
        if a[i] < 1e-12: continue
        j = np.random.randint(n)
        if i == j: continue
        # move fraction up to 0.2
        delta = a[i] * (np.random.rand() * 0.2)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalize? sum stays same.
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)
        # accept if improve or probabilistically
        if val_new < current_val:
            accept = True
        else:
            # temperature schedule
            T = T0 * (1.0 - move / max_moves)
            prob = np.exp(-(val_new - current_val) / (T + 1e-12))
            accept = np.random.rand() < prob
        if accept:
            a = a_new
            conv = conv_new
            current_val = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                # print improvement
                if move % 5000 == 0:
                    print(f"[SA move {move}] new best {val_new:.6f}")
    return best_a, best_val

"""
Propose a non‑negative step‑function (list of floats) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The search proceeds in three stages:

1. **Coarse exploration** on a modest grid (n≈1024).  A large and diverse
   pool of start vectors is generated (uniform, arcsine, triangular,
   sin‑powers, edge‑exponential, double‑Gaussian, Dirichlet with many
   concentrations, and sparse spike patterns).  Each candidate is
   optimized with a smooth‑max Adam phase followed by a short exact‑max
   sub‑gradient phase.  The best coarse solution is retained.

2. **Upsampling & deep refinement** – the coarse best vector is linearly
   interpolated to the full resolution (n=4096) and refined with a long
   exponentiated‑gradient (sub‑gradient) run.  This stage aggressively
   drives the max of the self‑convolution down.

3. **Final stochastic polishing** – a simple simulated‑annealing
   hill‑climber swaps a random fraction of mass between two coordinates,
   accepting worsening moves with a linearly cooling temperature.  This
   helps escape any remaining local minima.

All phases respect the global deadline (default 1000 s) and the best
vector found at any moment is returned.  The implementation uses FFT‑based
convolutions for speed, works on the simplex (sum = 1), and never
produces negative entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.07,
    lam0: float = 0.05,
    lam_mul: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  The temperature
    (lam) starts at ``lam0`` and is multiplied by ``lam_mul`` each
    iteration, gradually focusing on the true max.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # start on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam0

    best_a = a0.copy()
    best_val = float("inf")

    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_mul, 1e7)

        # occasional jitter near the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay (very gentle)
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size η_t = η₀ / sqrt(t+1).  Returns the best vector seen and
    its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against under‑flow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climber that moves a random fraction of mass
    between two positions.  Accepts worsening moves with a linearly
    cooling temperature.  Returns the best vector found.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val

    # start temperature (relative to the typical objective scale)
    T0 = 0.02

    for move in range(max_moves):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if val_new < cur_val:
            accept = True
        else:
            # linear cooling schedule
            T = T0 * max(0.0, 1.0 - move / max_moves)
            prob = np.exp(-(val_new - cur_val) / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            conv = conv_new
            cur_val = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                if move % 5000 == 0:
                    print(f"[SA move={move:06d}] new best {val_new:.6f}")

    return best_a, best_val

def _generate_initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a fairly rich set of start vectors for the coarse search.
    """
    cand = []

    # warm start and simple shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # arcsine‑like (1/√(i·(n‑i+1)))
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    a = a / a.sum()
    cand.append(a)

    # triangular (tent) window
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(idx - mid) / (mid + 1.0)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    cand.append(a)

    # half‑wave sinusoid and its powers
    base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    base = np.maximum(base, 0.0)
    for p in (1, 2, 3, 4):
        a = base ** p
        a = a / a.sum()
        cand.append(a)

    # edge‑exponential families
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a = a / a.sum()
        cand.append(a)

    # edge‑polynomial (power‑law) families
    prod = (np.arange(1, n + 1, dtype=np.float64)
            * (n - np.arange(1, n + 1, dtype=np.float64) + 1.0))
    for alpha in (-2.0, -1.5, -1.2, -1.0, -0.8):
        a = np.where(prod > 0, prod ** alpha, 0.0)
        if a.sum() > 0:
            a = a / a.sum()
            cand.append(a)

    # double‑Gaussian (different widths)
    x = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.12, 0.25):
        a = (np.exp(-0.5 * (x / (sigma * n)) ** 2) +
             np.exp(-0.5 * ((x - (n - 1)) / (sigma * n)) ** 2))
        a = a / a.sum()
        cand.append(a)

    # random Dirichlet with various concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        a = np.random.dirichlet(np.full(n, conc))
        cand.append(a)

    # sparse spike patterns
    for k in (8, 16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Coarse search on a medium sized grid (n≈1024)
    n_small = 1024
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # warm‑start, if a previous solution exists
    warm_big = globals().get("height_sequence_1")
    warm_small = None
    if isinstance(warm_big, (list, np.ndarray)):
        w = np.asarray(warm_big, dtype=np.float64)
        if w.shape[0] > n_small:
            w = w[:n_small]
        elif w.shape[0] < n_small:
            w = np.pad(w, (0, n_small - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            warm_small = w / w.sum()

    # generate a rich pool of starts
    pool = _generate_initial_candidates(n_small, warm_small)
    np.random.shuffle(pool)

    best_seq = None
    best_val = float("inf")

    # allocate roughly 60 % of the time to exploration
    exploration_budget = (deadline - time.time()) * 0.60
    per_cand_budget = max(5.0, exploration_budget / max(len(pool), 1))

    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        # local deadline for this candidate
        local_deadline = min(deadline - 2.0, time.time() + per_cand_budget)

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n_small))
        a0 = a0 / a0.sum()

        # ---- Adam (smooth‑max) ------------------------------------------------
        a_adam, val_adam = _adam_optimize(
            a0,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            local_deadline,
            max_iters=150_000,
            lr=0.07,
            lam0=0.03,
            lam_mul=1.05,
        )
        # ---- short sub‑gradient refinement ------------------------------------
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                a_adam,
                n_small,
                conv_fft_len_small,
                corr_fft_len_small,
                deadline,
                max_iters=30_000,
                eta0=0.9,
            )
        else:
            a_sub, val_sub = a_adam, val_adam

        # keep the better of the two phases
        if val_sub < val_adam:
            cur_val = val_sub
            cur_a = a_sub
        else:
            cur_val = val_adam
            cur_a = a_adam

        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_a.copy()
            print(f"[Exploration cand={cid}] new best {cur_val:.6f}")

    # Fallback in the unlikely case the pool was empty
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to the full resolution (n=4096) and deep refinement
    n_big = 4096
    if best_seq.shape[0] != n_big:
        x_small = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
    else:
        a_big = best_seq.copy()

    a_big = np.maximum(a_big, 0.0)
    a_big = a_big / a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # Long sub‑gradient phase (takes the bulk of remaining time)
    if time.time() < deadline - 10:
        a_ref, val_ref = _subgradient_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=300_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] sub‑grad best {val_ref:.6f}")
        else:
            best_seq = a_big.copy()
    else:
        best_seq = a_big.copy()

    # ------------------------------------------------------------------
    # 3️⃣  Final stochastic polishing (simulated annealing)
    if time.time() < deadline - 2:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n_big,
            conv_fft_len_big,
            deadline,
            max_moves=80_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Ensure the output respects the evaluator's constraints
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
