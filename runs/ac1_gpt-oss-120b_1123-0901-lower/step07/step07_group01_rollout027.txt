```
def generate_beta_distribution(n, alpha, beta):
    x = (np.arange(n) + 0.5) / n
    pdf = np.where((x>0) & (x<1), (x**(alpha-1)) * ((1-x)**(beta-1)), 0.0)
    total = pdf.sum()
    if total == 0:
        return np.full(n, 1.0/n)
    return pdf / total
```

```
def subgradient_optimize(a0, n, max_iters, deadline):
    a = np.maximum(a0, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        total = a.sum()
    a = a / total

    # precompute FFT length for conv and correlation
    conv_fft_len = 1 << ((2*n-1).bit_length())
    corr_fft_len = conv_fft_len

    best_a = a.copy()
    best_val = compute_objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv
        fa = np.fft.rfft(a, conv_fft_len)
        conv = np.fft.irfft(fa * fa, conv_fft_len)[:2*n-1]
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # determine active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        # gradient
        # cross-correlation w ⋆ a
        # w convolve reversed a:
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = n - 1
        grad = 2.0 * corr[start:start+n]
        # step size
        eta = 0.6 / np.sqrt(it + 1)  # eta0 = 0.6
        a = a * np.exp(-eta * grad)
        # renormalize
        total = a.sum()
        if total < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            total = a.sum()
        a = a / total
        it += 1
    return best_a, best_val
```

```
    a = np.power(idx, p)
    a_sum = a.sum()
    a = a / a_sum

    # Clip to [0,1000]
    a = np.clip(a, 0.0, 1000.0)

    # evaluate
    try:
        val = evaluate_sequence(a.tolist())
        print(f"Generated polynomial shape p={p}, n={n}, value={val:.6f}")
    except Exception as e:
        print("Evaluation error:", e)

    return a.tolist()
```
