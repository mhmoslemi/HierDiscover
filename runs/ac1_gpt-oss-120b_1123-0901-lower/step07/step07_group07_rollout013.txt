       • Adam on a log‑sum‑exp (smooth‑max) surrogate of the max‑convolution.
       • Exact sub‑gradient descent on the true max‑convolution.
    Both phases run as long as the global deadline allows
    (iteration limits are generous – up to a few hundred‑thousand steps).

3.  If time permits the current best vector is interpolated to a larger
    grid (n = 8192) and the whole optimisation is repeated.
    Finally a long random mass‑transfer hill‑climb finishes the search.

All vectors are kept on the probability simplex (non‑negative,
sum‑normalised) and clipped to the evaluator’s admissible range.
The code contains many print‑outs for progress monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT of length
    ``fft_len`` (must be ≥ 2·len(a)-1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (sum(a) ≈ 1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a rich candidate pool (symmetric & asymmetric).  All candidates
    are normalised to have sum ≈ 1.
    """
    candidates: list[np.ndarray] = []

    # Warm‑start and a few cyclic shifts (if any)
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # Asymmetric exponentials (decay / growth)
    for alpha in np.logspace(-2, 1.0, num=12):
        dec = np.exp(-alpha * i)
        dec /= dec.sum()
        candidates.append(dec)                          # decreasing
        inc = np.exp(-alpha * (n - 1 - i))
        inc /= inc.sum()
        candidates.append(inc)                          # increasing

    # Edge‑exponential families (with optional uniform mix)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 0.9, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Symmetric beta‑shapes  (a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1})
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape /= shape.sum()
        candidates.append(shape)

    # Two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape /= shape.sum()
    candidates.append(shape)

    # Random sparse spikes (3‑12 spikes)
    for _ in range(12):
        spikes = np.random.randint(3, 13)
        positions = np.random.choice(n, size=spikes, replace=False)
        weights = np.random.rand(spikes)
        vec = np.full(n, 1e-6)          # tiny baseline to keep full support
        vec[positions] = weights
        vec /= vec.sum()
        candidates.append(vec)

    return candidates


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_len: int,
                   corr_len: int,
                   deadline: float,
                   best_seq: np.ndarray | None,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of max‑convolution.
    The routine runs as long as time permits (capped at 400 000 iters).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise in soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05                               # a bit more conservative
    lam = 0.2                               # start very smooth
    lam_factor = 1.015
    lam_cap = 1e7

    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max → probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & true objective
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d}  new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck > 8 s
        if time.time() - last_improve > 8.0:
            jitter = (np.random.rand(n) - 0.5) * 2e-2
            logits += jitter
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_len: int,
                          corr_len: int,
                          deadline: float,
                          best_seq: np.ndarray | None,
                          best_val: float) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient steps on the true max‑convolution.
    Runs up to 150 000 iterations or until the deadline.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    it = 0
    max_iter = 150_000
    eta0 = 0.7
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:06d}  new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # decaying step‑size (exponentiated‑gradient keeps positivity)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # modest early‑stop if nothing improves for a while
        if time.time() - last_improve > 15.0:
            # give another shot – keep going but reset timer
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_len: int,
                   corr_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Run the two‑phase optimiser (Adam → sub‑gradient) and return the best
    sequence/value seen during the run.
    """
    seq, val = _adam_optimize(a0, n, conv_len, corr_len,
                              deadline, global_best_seq, global_best_val)
    seq, val = _subgradient_optimize(seq, n, conv_len, corr_len,
                                     deadline, seq, val)
    return seq, val


# ----------------------------------------------------------------------
def _random_hill_climb(a: np.ndarray,
                       n: int,
                       conv_len: int,
                       deadline: float,
                       best_val: float,
                       max_iter: int = 25000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb: repeatedly transfer a tiny random amount
    of mass between two indices and keep the move if it improves the true
    objective.
    """
    best_seq = a.copy()
    best_local = best_val
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose a transfer proportional to the smaller of the two masses
        delta_max = min(best_seq[i], best_seq[j])
        if delta_max < 1e-14:
            continue
        delta = (np.random.rand() - 0.5) * 0.4 * delta_max   # up to ±20 %
        # keep positivity
        if delta > best_seq[j]:
            delta = best_seq[j]
        if -delta > best_seq[i]:
            delta = -best_seq[i]
        if abs(delta) < 1e-15:
            continue

        best_seq[i] += delta
        best_seq[j] -= delta

        conv = _full_convolution(best_seq, conv_len)
        val = 2.0 * n * conv.max() / (best_seq.sum() ** 2)

        if val < best_local:
            best_local = val
            # keep the improvement
        else:
            # revert
            best_seq[i] -= delta
            best_seq[j] += delta

        it += 1
    return best_seq, best_local


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Small grid first – fast exploration
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # try to grab a warm‑start from previous runs (if present)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ---- Stage 1: generate a broad candidate pool --------------------
    cand_pool = _generate_parametric_candidates(n, ws)

    # quick evaluation on the small grid
    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # fallback to a fresh Dirichlet
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising seeds (n={n})")

    # ---- Stage 2: deep optimisation on the selected seeds ----------
    for sid, seed_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{sid + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            seed_vec,
            n,
            conv_len,
            corr_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # If we still have a comfortable amount of time, move to a larger grid.
    time_left = deadline - time.time()
    if time_left > 45.0:
        n_big = 8192
        conv_len_big = _next_pow2(2 * n_big - 1)
        corr_len_big = _next_pow2(3 * n_big - 2)

        # interpolate current best to the finer grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_len_big,
            corr_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n, conv_len, corr_len = n_big, conv_len_big, corr_len_big

    # ------------------------------------------------------------------
    # Final local refinement (random hill‑climb)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("--- Final random hill‑climb refinement ---")
        best_seq, best_val = _random_hill_climb(
            best_seq,
            n,
            conv_len,
            deadline,
            best_val,
            max_iter=20000,
        )
        print(f"Final refinement best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
