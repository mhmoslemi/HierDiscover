def _incremental_local_search(a, n, conv, deadline, max_moves=None):
    best_val = 2 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and (max_moves is None or it < max_moves):
        i = np.random.randint(n)
        if a[i] < 1e-8:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        conv2[idx_2j] += delta2

      sub‑gradient steps.
  4.  Keep the best candidate and perform a long sub‑gradient run.
  5.  Apply a cheap hill‑climbing random‑move phase.
  6.  Finish with a very fast “incremental’’ mass‑transfer local
      search that updates the convolution by O(n) arithmetic per move
      (no FFT per move).  This phase can execute many hundreds of
      thousands of moves in the remaining time budget.

All vectors are normalised to sum = 1, stays non‑negative,
and the routine respects the global time limit.  The best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest set of diverse start vectors."""
    cand = []

    # warm start + a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 13):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 5.0, 8):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """A few “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient phase."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing with full FFT convolution per move."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


def _incremental_local_search(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Fast O(n) mass‑transfer hill‑climbing.
    Moves a random fraction of the source mass to a random destination.
    The convolution is updated incrementally, avoiding an FFT per move.
    """
    a = a0.copy()
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)                # length 2·n‑1
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    accepted = 0
    while time.time() < deadline and (max_moves is None or it < max_moves):
        # source index biased by current mass (larger entries move more)
        i = np.random.choice(n, p=a)
        if a[i] <= 1e-14:
            it += 1
            continue
        # destination index, uniformly random but different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % of source
        if delta <= 0.0:
            it += 1
            continue

        # candidate convolution (copy & incremental update)
        conv2 = conv.copy()
        conv2[i:i + n] -= 2.0 * delta * a
        conv2[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        delta2 = delta * delta
        conv2[idx2i] += delta2
        conv2[idx2j] += delta2
        conv2[idxij] -= 2.0 * delta2

        val2 = 2.0 * n * conv2.max()
        if val2 + 1e-15 < best_val:               # accept only improvements
            best_val = val2
            best_a = a.copy()
            best_a[i] -= delta
            best_a[j] += delta
            a[i] -= delta
            a[j] += delta
            conv = conv2
            accepted += 1
            if accepted % 5_000 == 0:
                # re‑compute convolution from scratch to avoid drift
                conv = _full_convolution(a, fft_len)
                best_val = 2.0 * n * conv.max()
                best_a = a.copy()
                print(f"[IncLocal iter={it}] re‑sync, new best {best_val:.6f}")
        it += 1
    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from a previous solution, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s >= 0.01:
            warm = warm / s
        else:
            warm = None

    candidates = _initial_candidates(n, warm) + _spike_candidates(n)

    # ------------------------------------------------------------------
    # Phase 1 – quick Adam + short sub‑gradient on each candidate
    # Reserve ~5 min for later intensive phases
    reserve_for_later = 300.0
    for cid, cand in enumerate(candidates):
        if time.time() > deadline - reserve_for_later:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # Adam – limited time per candidate (≈8 s)
        cand_deadline = min(deadline - reserve_for_later, time.time() + 8.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=80_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # Short sub‑gradient refinement (≈8 s)
        if time.time() < deadline - reserve_for_later:
            sub_deadline = min(deadline - reserve_for_later, time.time() + 8.0)
            a_sub, val_sub = _subgradient_opt2(
                a_adam, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=30_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long sub‑gradient on the current best
    if best_seq is not None and time.time() < deadline - 180.0:
        a_ref, val_ref = _subgradient_opt2(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline - 180.0,
            max_iters=300_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random local search (full convolution per move)
    if best_seq is not None and time.time() < deadline - 120.0:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline - 120.0, max_moves=300_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fast incremental mass‑transfer hill climbing
    if best_seq is not None and time.time() < deadline - 30.0:
        a_inc, val_inc = _incremental_local_search(
            best_seq, n, deadline - 30.0, max_moves=None
        )
        if val_inc < best_val:
            best_val = val_inc
            best_seq = a_inc.copy()
            print(f"[IncLocal] final best {val_inc:.6f}")

    # ------------------------------------------------------------------
    # Optional upscale to 8192 if we still have generous time left
    if time.time() + 20 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid
        big_deadline = min(deadline - 5.0, time.time() + 15.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=120_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Big‑8192] Adam best {val_big_adam:.6f}")

        # sub‑gradient refinement
        if time.time() < deadline:
            a_big_ref, val_big_ref = _subgradient_opt2(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=200_000,
            )
            if val_big_ref < best_val:
                best_val = val_big_ref
                best_seq = a_big_ref.copy()
                print(f"[Big‑8192] Subgrad best {val_big_ref:.6f}")

        # final cheap local search on 8192
        if time.time() < deadline - 5:
            a_loc2, val_loc2 = _local_random_search(
                best_seq, n_big, conv_fft_len_big, deadline, max_moves=150_000
            )
            if val_loc2 < best_val:
                best_val = val_loc2
                best_seq = a_loc2.copy()
                print(f"[Local‑8192] improved best {val_loc2:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return value
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
