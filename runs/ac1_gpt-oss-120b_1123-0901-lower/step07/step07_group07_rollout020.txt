"""
Propose a non‑negative step sequence with a very low evaluation value.

Algorithm outline
-----------------
1. **Parametric seeding** – generate a diverse pool of symmetric
   candidates (uniform, arcsine, Dirichlet draws, edge‑exponential,
   edge‑power‑law, symmetric beta, raised‑cosine).  Each candidate is
   normalised to sum 1 and is evaluated with a fast FFT based convolution.
2. **Deep optimisation** – the top few candidates are refined by
   (a) Adam on a log‑sum‑exp smooth approximation of the max‑convolution,
   followed by (b) an exponentiated‑gradient sub‑gradient method that
   directly minimises the true `max(conv)`.  The available time budget is
   shared equally among the selected starts.
3. **Upscaling** – if time permits the best vector is interpolated to a
   finer grid (n = 8192) and the deep optimiser is run again.
4. **Random local search (simulated‑annealing)** – a short hill‑climb that
   repeatedly transfers a small random mass between two indices, accepting
   uphill moves with a decreasing temperature schedule.

All vectors are kept on the probability simplex (non‑negative and sum 1).
The final sequence is clipped to the evaluator’s allowed range and returned.

The implementation uses only NumPy and the supplied ``evaluate_sequence``
function; it respects the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast evaluation of the objective (sum(a) ≈ 1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

# Adam on a smooth‑max surrogate
def _adam_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max‑convolution.
    Returns the best (sequence, value) seen before the deadline.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise in log‑softmax space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05                               # a bit smaller than before
    lam = 0.1                               # start very smooth
    lam_factor = 1.015                       # grow slowly
    lam_cap = 1e5

    it = 0
    last_improve = time.time()
    max_iter = 500_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] it={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term from chain rule
        grad_log = a * (grad_a - avg)         # gradient w.r.t. logits

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional jitter if stuck -----
        if time.time() - last_improve > 3.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

# Exponentiated‑gradient sub‑gradient on the true max‑convolution
def _subgrad_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent (exponentiated‑gradient) directly on the max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    max_iter = 200_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] it={it:06d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 8.0:
            break
        it += 1

    return best_seq, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:

    """
    i = np.arange(n, dtype=np.float64)
    candidates: list[np.ndarray] = []

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Arcsine‑type shape
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = np.clip(arcsine, 0.0, None)
    arcsine /= arcsine.sum()
    candidates.append(arcsine)

    # Warm‑start & cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for s in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, s))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # Randomised parametric families
    nrnd = 300
    for _ in range(nrnd):
        family = np.random.choice(["exp", "power", "beta", "sine"])
        if family == "exp":
            # edge exponential
            alpha = 10 ** np.random.uniform(-2.0, 1.0)   # 0.01 … 10
            w = np.random.rand()
            edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
            edge /= edge.sum()
            cand = w * edge + (1.0 - w) * np.full(n, 1.0 / n)
        elif family == "power":
            # edge power‑law
            p = np.random.uniform(0.2, 5.0)
            edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
            edge /= edge.sum()
            w = np.random.rand()
            cand = w * edge + (1.0 - w) * np.full(n, 1.0 / n)
        elif family == "beta":
            # symmetric beta‑shape
            alpha = np.random.uniform(0.1, 3.0)
            shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
            shape = np.clip(shape, 0.0, None)
            shape /= shape.sum()
            w = np.random.rand()
            cand = w * shape + (1.0 - w) * np.full(n, 1.0 / n)
        else:  # sine
            p = np.random.uniform(0.5, 12.0)
            shape = np.sin(np.pi * (i + 0.5) / n) ** p
            shape = np.clip(shape, 0.0, None)
            shape /= shape.sum()
            w = np.random.rand()
            cand = w * shape + (1.0 - w) * np.full(n, 1.0 / n)

        # Normalise and guard against pathological draws
        cand = np.clip(cand, 0.0, None)
        total = cand.sum()
        if total < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
        else:
            cand /= total
        candidates.append(cand)

    return candidates

        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
            warm_start = ws
        else:
            warm_start = None

    # ---- stage 1 : generate a large pool of candidates ----------------
    pool = _generate_candidates(n, warm_start)

    evaluated = []
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # occasional progress print
        # print(f"[pool] {idx} → {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: evaluated {len(evaluated)} candidates, kept top {top_k}")

    # ---- stage 2 : deep optimisation of the retained starts ----------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for start_id, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # allocate a fair slice of the remaining budget
        remaining = len(top_candidates) - start_id
        time_per = (deadline - time.time()) / max(remaining, 1)
        local_deadline = time.time() + time_per

        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        seq, val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            local_deadline,
            best_seq,
            best_val,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"New global best after start {start_id + 1}: {best_val:.6f}")

    # ---- stage 3 : upscale to a larger grid if time permits ----------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)

        # linear interpolation of the current best
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.clip(a_big, 0.0, None)
        a_big /= a_big.sum()

        print("--- Upscaling to n=8192 and final refinement ---")
        seq_big, val_big = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = seq_big
            n = n_big
            conv_fft_len = conv_fft_len_big
            print(f"Improved after up‑scaling: {best_val:.6f}")

    # ---- stage 4 : short simulated‑annealing hill‑climb -------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        cur_val = best_val
        best_local_seq = a.copy()
        best_local_val = cur_val

        total_sa_time = deadline - time.time() - 5.0
        sa_start = time.time()
        max_transfer_factor = 0.5        # at most 50 % of donor mass

        while time.time() < deadline - 5.0:
            elapsed = time.time() - sa_start
            frac = max(0.0, 1.0 - elapsed / total_sa_time)

            # adapt the transfer magnitude
            step = max_transfer_factor * frac + 0.001

            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            # always transfer from the larger entry to the smaller one
            donor, recv = (i, j) if a[i] > a[j] else (j, i)
            max_transfer = a[donor] * step
            if max_transfer <= 0.0:
                continue
            delta = np.random.rand() * max_transfer

            a_new = a.copy()
            a_new[donor] -= delta
            a_new[recv] += delta

            conv = _full_convolution(a_new, conv_fft_len)
            new_val = 2.0 * n * conv.max() / (a_new.sum() ** 2)

            if new_val < cur_val:
                a = a_new
                cur_val = new_val
                if new_val < best_local_val:
                    best_local_val = new_val
                    best_local_seq = a_new.copy()
            else:
                # simulated‑annealing acceptance
                T0 = 1e-3
                temp = T0 * frac
                if temp > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / temp):
                    a = a_new
                    cur_val = new_val

        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_seq
            print(f"Local SA improved best to {best_val:.6f}")

    # ---- fallback ------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        best_val = _objective(best_seq, 4096, _next_pow2(2 * 4096 - 1))

    # ---- final clipping & return ---------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete: best objective ≈ {best_val:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()
