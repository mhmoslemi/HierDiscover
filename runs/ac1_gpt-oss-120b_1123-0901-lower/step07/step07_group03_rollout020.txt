"""
Propose a low‑value step‑function for the objective
    F(a) = 2·n·max(a∗a) / (∑a)²

The algorithm works in three phases:

1) **Diverse warm‑starts** – a large pool of candidate vectors is built.
   The pool contains
   * the previous best solution (if any) and its rotations,
   * uniform vector,
   * Dirichlet samples with a wide range of concentration,
   * anti‑pair ordered Dirichlet samples (large values opposite small ones),
   * simple analytic shapes (triangular, edge‑Gaussian, power‑law).

2) **Rapid multi‑restart optimisation** – for each candidate a short
   Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) is run,
   followed by a few thousand exponentiated‑gradient sub‑gradient steps.
   The best result among all restarts is kept.

3) **Fine‑grained refinement** – on the current best vector a long Adam
   phase (higher temperature schedule), a longer sub‑gradient phase and a
   simulated‑annealing local search are performed.  If time permits the
   solution is interpolated to a finer grid (8192 points) and refined once
   more.

All operations respect the 1000 s wall‑time budget, and the routine
always returns a valid non‑negative list (clipped to the range accepted by
``evaluate_sequence``).  Progress and intermediate bounds are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution via zero‑padded FFT)
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (without the safety checks of evaluate_sequence)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators -------------------------------------------------
def _warm_start(n: int) -> np.ndarray | None:
    """Return a warm start from the global variable if it exists."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    w = np.asarray(warm, dtype=np.float64)
    if w.shape[0] > n:
        w = w[:n]
    elif w.shape[0] < n:
        w = np.pad(w, (0, n - w.shape[0]))
    w = np.clip(w, 0.0, 1000.0)
    if w.sum() >= 0.01:
        return w / w.sum()
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    """Draw from Dirichlet(alpha,…,alpha) and normalise to sum = 1."""
    a = np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)
    return a / a.sum()


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place the sorted values in anti‑pair order (large ↔ small)."""
    n = len(vals)
    sorted_desc = np.sort(vals)[::-1]
    out = np.empty_like(vals)
    half = n // 2
    out[:half] = sorted_desc[:half]          # largest half on the left
    out[-half:] = sorted_desc[-half:][::-1]  # smallest half on the right
    if n % 2:
        out[half] = sorted_desc[half]
    return out


def _anti_pair_candidate(n: int, alpha: float) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)
    return _anti_pair_arrangement(vals / vals.sum())


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (big at edges, small in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)        # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float = 1.5) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = i ** (-p)
    shape = shape / shape.sum()
    return shape

# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    The temperature λ grows multiplicatively; a larger λ makes the surrogate
    approach the true max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        exact_val = 2.0 * n * max_c
        if exact_val < best_val:
            best_val = exact_val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # transform through soft‑max Jacobian
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape flat regions
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # indicators of the active (max‑achieving) shifts
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climbing.
    Moves: swap two entries or transfer a fraction of mass.
    Acceptance: improvement or Metropolis rule with linearly decaying temperature.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # decide move type
        if np.random.rand() < 0.5:        # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                             # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of source mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5_000 == 0:
            print(f"[Local move {it:07d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Quick optimisation (used for many restarts)
def _quick_optimize(a_start: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a fast Adam → sub‑gradient pipeline on a given start vector.
    The allotted time is split roughly 60 % Adam, 30 % sub‑gradient,
    remaining time left for a brief local search.
    """
    now = time.time()
    time_adam = min(deadline - now - 0.5, (deadline - now) * 0.6)
    a_ad, val_ad = _adam_optimize(
        a_start, n, conv_fft_len, corr_fft_len,
        deadline=now + time_adam,
        max_iters=200_000,
        lr=0.09,
        lam_start=0.5,
        lam_factor=1.08,
    )
    now = time.time()
    if now >= deadline - 0.5:
        return a_ad, val_ad

    time_sub = min(deadline - now - 0.5, (deadline - now) * 0.3)
    a_sub, val_sub = _subgradient_optimize(
        a_ad, n, conv_fft_len, corr_fft_len,
        deadline=now + time_sub,
        max_iters=200_000,
        eta0=0.8,
    )
    now = time.time()
    if now >= deadline - 0.5:
        return a_sub, val_sub

    # very short local search (a few thousand moves)
    a_loc, val_loc = _local_random_search(
        a_sub, n, conv_fft_len, deadline=deadline - 0.3, max_moves=30_000
    )
    return a_loc, val_loc

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a)/(∑a)²

    The function respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a pool of start vectors (diverse shapes)
    start_pool = []

    warm = _warm_start(n)
    if warm is not None:
        start_pool.append(warm)
        # add rotated versions (shifted by quarter, half, three quarters)
        for shift in (n // 4, n // 2, 3 * n // 4):
            start_pool.append(np.roll(warm, shift))

    start_pool.append(_uniform_candidate(n))

    # Dirichlet samples with a wide range of concentration parameters
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        start_pool.append(_dirichlet_candidate(n, alpha))

    # Anti‑pair ordered Dirichlet samples (critical for low max‑conv)
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0):
        start_pool.append(_anti_pair_candidate(n, alpha))

    # Analytic shapes
    for gamma in (0.1, 0.3, 0.5, 0.8, 1.2):
        start_pool.append(_gaussian_edges_candidate(n, gamma))
    start_pool.append(_triangle_candidate(n))
    for p in (0.8, 1.2, 1.6, 2.0):
        start_pool.append(_powerlaw_candidate(n, p))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool and keep the current best
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(start_pool):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        print(f"[Pool cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val

    # ------------------------------------------------------------------

        elif gen_choice == 3:
            cand = _powerlaw_candidate(n, np.random.uniform(0.8, 2.2))
        elif gen_choice == 4:
            cand = _triangle_candidate(n)
        elif gen_choice == 5:
            cand = _uniform_candidate(n)

        cand = cand / cand.sum()

        cand_opt, val_opt = _quick_optimize(cand, n, conv_fft_len, corr_fft_len, restart_deadline)

        a_big = a_big / a_big.sum()

        # short Adam + sub‑gradient on the bigger grid
        big_deadline = min(deadline - 5.0, time.time() + 120.0)
        a_big_ad, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline=big_deadline,
            max_iters=300_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if time.time() < deadline - 10.0:
            a_big_sub, big_val = _subgradient_optimize(
                a_big_ad,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline=deadline - 2.0,
                max_iters=300_000,
                eta0=0.7,
            )
            if big_val < best_val:
                best_val = big_val
                best_seq = a_big_sub.copy()
                print(f"[Upscaled] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
