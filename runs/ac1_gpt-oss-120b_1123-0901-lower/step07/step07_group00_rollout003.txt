def window_candidates(n):
    candidates = []
    # Hann
    w = 0.5 * (1 - np.cos(2*np.pi*np.arange(n)/ (n-1)))
    w = w/ w.sum()
    candidates.append(w)
    # Hamming
    w = 0.54 - 0.46*np.cos(2*np.pi*np.arange(n)/(n-1))
    w = w / w.sum()
    candidates.append(w)
    # Blackman
    w = 0.42 - 0.5*np.cos(2*np.pi*np.arange(n)/(n-1)) + 0.08*np.cos(4*np.pi*np.arange(n)/(n-1))
    w = w / w.sum()
    candidates.append(w)
    # Kaiser with beta = 2,4,6,8,10
    for beta in (2,4,6,8,10,12):
        from scipy.special import i0
        n_arr = np.arange(n)
        alpha = (n-1)/2.0
        t = (n_arr - alpha)/alpha
        w = i0(beta * np.sqrt(1 - t**2))/i0(beta)
        w = np.maximum(w,0)
        w /= w.sum()
        candidates.append(w)
    # Tukey windows with alpha = 0.5,0.75,0.9
    for a in (0.4, 0.5, 0.6, 0.8):
        n_arr = np.arange(n)
        # Tukey window formula:

        w = np.ones(n)
        # first region

def _adam_optimize2(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    lam = 0.1
    lam_factor = 1.015
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # Softmax projection:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # Compute objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam2 it={it}] new best {val:.6f}")
        # Smooth max weights:
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # Mirror descent gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        logits -= step
        lam = min(lam * lam_factor, 1e7)
        # decay lr gently
        if it % 50000 == 0 and it > 0:
            lr *= 0.9
        it += 1
    return best_a, best_val
```

"""
Propose a non‑negative step function (list of floats) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm builds a rich set of initial patterns (Dirichlet, exponential,
polynomial, spikes, classic signal‑processing windows and Beta‑shaped pdfs),
picks the most promising few, and refines them with three optimisation
phases:

1.  Adam on a smooth‑max (log‑sum‑exp) surrogate, with a slowly increasing
    temperature schedule.
2.  Exact‑max sub‑gradient descent on the simplex (exponentiated‑gradient)
    with a decreasing step‑size.
3.  Cheap random‑move hill‑climbing.

The best solution is repeatedly perturbed and re‑optimised to escape local
minima.  If time permits a finer grid (8192 points) is tried.  The routine
always returns a feasible sequence before the deadline.
"""

import time
import numpy as np
from scipy.special import i0  # for Kaiser windows

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families (mass concentrated at the left/right)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


def _window_candidates(n: int) -> list[np.ndarray]:
    """Classic DSP windows (Hann, Hamming, Blackman, Kaiser, Tukey)."""
    cands = []

    # Hann
    w = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    cands.append(w / w.sum())

    # Hamming
    w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    cands.append(w / w.sum())

    # Blackman
    w = (0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
         + 0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1)))
    cands.append(w / w.sum())

    # Kaiser windows, several betas
    for beta in (2, 4, 6, 8, 10, 12):
        i = np.arange(n)
        alpha = (n - 1) / 2.0
        t = (i - alpha) / alpha
        w = i0(beta * np.sqrt(1 - t ** 2)) / i0(beta)
        w = np.maximum(w, 0)
        cands.append(w / w.sum())

    # Tukey windows (different taper ratios)
    for a in (0.4, 0.5, 0.6, 0.8):
        N1 = a * (n - 1) / 2.0
        N2 = (n - 1) * (1 - a / 2.0)
        i = np.arange(n)
        w = np.ones(n)
        # left taper
        left = i < N1
        w[left] = 0.5 * (1 + np.cos(np.pi * (2 * i[left] / (a * (n - 1)) - 1)))
        # right taper
        right = i > N2
        w[right] = 0.5 * (1 + np.cos(np.pi * (2 * i[right] / (a * (n - 1))
                                               - 2 / a + 1)))
        cands.append(w / w.sum())

    return cands


def _beta_candidates(n: int) -> list[np.ndarray]:
    """Beta‑distribution shaped pdfs on [0,1]."""
    x = np.linspace(0, 1, n, endpoint=False)
    cands = []
    alphas = (0.3, 0.5, 0.8, 1.0, 2.0, 5.0)
    betas = (0.3, 0.5, 0.8, 1.0, 2.0, 5.0)
    for a in alphas:
        for b in betas:
            # pdf ∝ x^{a-1} (1-x)^{b-1}
            shape = np.where(
                (x > 0) & (x < 1),
                np.power(x, a - 1) * np.power(1 - x, b - 1),
                0.0,
            )
            if shape.sum() > 0:
                shape /= shape.sum()
                cands.append(shape)
    return cands


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02
    lam_max = 5000.0

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---------- simplex projection (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        arg = lam * (conv - max_c)
        # protect against overflow
        arg = np.clip(arg, -700, 700)
        w = np.exp(arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update on logits ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        logits -= step

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, lam_max)

        # occasional learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        it += 1
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size η_t = η₀ / sqrt(t+1) (decreasing).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to required size
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a large, diverse pool of initial patterns
    pool = (
        _initial_candidates(n, warm)
        + _spike_candidates(n)
        + _window_candidates(n)
        + _beta_candidates(n)
    )

    # ------------------------------------------------------------------
    # Quick evaluation of raw candidates, keep the most promising few
    evaluated = []
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = evaluate_sequence(a.tolist())
        evaluated.append((val, a))
        if idx % 50 == 0:
            print(f"[Eval] candidate {idx} => {val:.6f}")
    evaluated.sort(key=lambda x: x[0])
    top_k = 8
    top_candidates = [a for _, a in evaluated[:top_k]]

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optimise each top candidate with Adam + a short sub‑gradient phase
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        # allocate a slice of remaining time for this candidate
        remaining = deadline - time.time()
        cand_deadline = time.time() + min(remaining * 0.2, 30.0)

        # Adam (smooth‑max) phase
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=200_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # short sub‑gradient refinement
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                a_adam,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=80_000,
                eta0=1.0,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad short best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Deep refinement of the overall best candidate
    if best_seq is not None and time.time() < deadline - 10:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            eta0=0.9,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Random‑perturbation restarts (to escape possible local minima)
    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline - 5:
            break
        noise = np.random.normal(scale=0.02, size=n)
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            a_pert = np.random.dirichlet(np.ones(n))
        else:
            a_pert = a_pert / s
        a_sub2, val_sub2 = _subgradient_optimize(
            a_pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=80_000,
            eta0=0.8,
        )
        if val_sub2 < best_val:
            best_val = val_sub2
            best_seq = a_sub2.copy()
            print(f"[restart {r}] improved to {val_sub2:.6f}")

    # ------------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sample to a finer grid (8192) if enough time remains
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=200_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
