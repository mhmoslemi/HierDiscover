"""
Propose a low‑objective step‑function using a two‑phase
soft‑max / sub‑gradient mirror‑descent.  The method works on a
large grid (default 8192 points) and repeatedly
   * computes the full self‑convolution by FFT,
   * builds a smooth probability distribution over the convolution
     entries (soft‑max with a temperature that is slowly increased),
   * takes an exponentiated‑gradient (KL‑mirror‑descent) step,
   * renormalises onto the probability simplex,
   * adaptively decays the step‑size and, when progress stalls,
     injects a tiny log‑space jitter.

The smoothed objective (soft‑max of the convolution) provides a
well‑behaved gradient that helps escape the plateaus that often
trap the plain max‑convolution sub‑gradient method used before.
When the temperature becomes large the soft‑max concentrates on the
active (maximal) indices, effectively switching to the original
max‑convolution sub‑gradient.  Multiple diverse initial seeds are
tried; the best vector found (according to the true
``evaluate_sequence`` objective) is returned.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the official objective (scale‑invariant)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _generate_initial_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Diverse seeds: uniform, Dirichlet, edge‑exponential, sinusoid, plus warm start."""
    candidates = []

    # warm start (if provided)

        candidates.append(warm_start.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws with several concentration levels
    for conc in (0.01, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0):
        for _ in range(6):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (plus a uniform mixture)
    for alpha in np.logspace(-2, 1.0, num=12):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # half‑wave sinusoid (positive part only)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    candidates.append(s)

    return candidates


def _optimize_candidate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Two‑phase soft‑max / max‑convolution mirror descent.
    Returns the best vector and its objective value observed during the run.
    """
    # initialise on the simplex
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    # algorithmic parameters
    eta = 0.6            # initial step‑size for exponentiated‑gradient
    decay_eta = 0.99993  # exponential decay of the step‑size
    min_eta = 0.001

    beta = 1.0           # soft‑max temperature (low → smooth)
    beta_mul = 1.015     # multiplicative increase each iteration
    beta_max = 300.0

    last_improve = time.time()
    iteration = 0

    while time.time() < deadline:
        # ----- convolution and objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() == 1 → just 2 n max_c

        # keep the best observed solution
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={iteration:07d}] new best {val:.6f}")

        # ----- soft‑max weights over convolution entries -----
        # Using max subtraction prevents overflow.
        m = max_c
        exp_arg = beta * (conv - m)
        # clamp exponent to avoid nan for huge beta
        exp_arg = np.clip(exp_arg, -700, 700)   # np.exp(-inf) -> 0 safely
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to pure max‑set weighting
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = w / w_sum

        # ----- gradient of the (soft) max‑convolution -----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- exponentiated‑gradient (mirror descent) update -----
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # ----- schedule updates -----
        eta = max(min_eta, eta * decay_eta)
        beta = min(beta_max, beta * beta_mul)

        # ----- occasional random jitter if stuck -----
        if time.time() - last_improve > 4.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # ~2 % log‑space noise
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

        iteration += 1

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a step‑function (non‑negative vector)
    with a low value of ``evaluate_sequence``.  Returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Grid size – 8192 points gives a good trade‑off between quality and speed.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)          # length for a*a
    corr_fft_len = _next_pow2(3 * n - 2)          # length for w★a

    # ------------------------------------------------------------------
    # Warm start (if a previous best sequence is stored in the global scope).
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Build a modest pool of initial vectors.
    candidates = _generate_initial_candidates(n, warm_start)

    # ------------------------------------------------------------------
    # Fast pre‑evaluation of all seeds – keep the most promising few.
    conv_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        conv_vals.append((val, cand))

    if not conv_vals:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    conv_vals.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [c for _, c in conv_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Deep optimisation of the selected starts.
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Optimising start #{start_id + 1}/{len(top_candidates)} (init={init_val:.6f})")
        best_seq, best_val = _optimize_candidate(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # If time permits, upscale to a finer grid (n = 16384) and refine.
    time_left = deadline - time.time()
    if time_left > 20.0 and best_seq is not None:
        n_big = n * 2
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print(f"--- Refinement on larger grid (n={n_big})")
        best_seq, best_val = _optimize_candidate(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big

    # ------------------------------------------------------------------
    # Final safety – uniform fallback if nothing succeeded.
    if best_seq is None:
        best_seq = np.full(8192, 1.0 / 8192, dtype=np.float64)

    # Clip to the range accepted by ``evaluate_sequence``.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
