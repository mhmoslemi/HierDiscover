"""
Propose a step function (non‑negative sequence) minimizing the
evaluate_sequence objective. The algorithm works in four stages:
1) generate a diverse set of parametric shapes on a large grid (n≈8192);
   keep the best few.
2) deep optimisation of each selected start using Adam on a smooth‑max
   surrogate and an extended exponentiated‑gradient sub‑gradient phase.
3) cheap stochastic hill‑climbing from the globally best vector.
4) final clipping and conversion to a plain Python list.
All timing is controlled by the supplied budget (default 1000 s).
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.1          # larger initial learning rate

    lam = 2.0               # start temperature
    lam_factor = 1.006      # slower rise, smooth transition
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        # soft‑max to obtain a valid probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used only for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via chain rule
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s  # stay on the simplex

    eta0 = 0.8
    decay = 0.99997
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # decay step size
        eta *= decay
        it += 1

        # early stop if stagnant for a while
        if time.time() - last_improve > 7.0:
            break

    return best_seq, best_val

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam + sub‑gradient phases on a single start vector.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iter=150_000,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        max_iter=200_000,
    )
    return best_seq, best_val

def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
    max_candidates: int = 800,
) -> list[np.ndarray]:
    """
    Produce a diverse pool of candidate vectors. The generation stops after
    *max_candidates* vectors have been created to keep the runtime bounded.
    """
    candidates = []

    def _maybe_append(vec):
        if len(candidates) >= max_candidates:
            return False
        candidates.append(vec)
        return True

    # warm‑start and its shifts
    if warm_start is not None:
        if not _maybe_append(warm_start.copy()):
            return candidates
        for shift in (n // 4, n // 2, 3 * n // 4):
            if not _maybe_append(np.roll(warm_start, shift)):
                return candidates

    # uniform baseline
    if not _maybe_append(np.full(n, 1.0 / n)):
        return candidates

    # Dirichlet draws
    concs = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in concs:
        for _ in range(8):   # fewer draws than before
            cand = np.random.dirichlet(np.full(n, conc))
            if not _maybe_append(cand):
                return candidates

    i = np.arange(n, dtype=np.float64)

    # symmetric edge‑exponential / power‑law
    alphas = np.logspace(-2, 1.0, num=8)
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            if not _maybe_append(shape):
                return candidates

    ps = np.linspace(0.3, 5.0, num=8)
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            if not _maybe_append(shape):
                return candidates

    # asymmetric two‑sided exponentials
    alphas2 = np.logspace(-2, 1.0, num=6)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas2:
                for aR in alphas2:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    if not _maybe_append(shape):
                        return candidates

    # linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * base + (1.0 - w) / n
            shape /= shape.sum()
            if not _maybe_append(shape):
                return candidates

    # half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    _maybe_append(sinus)

    # sparse spikes
    for spikes in (3, 5):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            if not _maybe_append(vec):
                return candidates

    return candidates

def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add Gaussian noise to the logits of the current
    best vector, evaluate the exact objective and keep the improvement.
    """
    logits_best = np.log(a_best + 1e-15)
    while time.time() < deadline:
        noise = np.random.randn(n) * 0.02   # slightly larger step than before
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val

            warm_start = np.interp(x_new, x_old, warm_start)

        else:

    # ------------------------------------------------------------------

    candidates = _generate_parametric_candidates(n, warm_start, max_candidates=800)

    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:

            s = cand.sum()

        best_val = cand_vals[0][0]

        for idx, a0 in enumerate(top_candidates):

            best_seq, best_val = _deep_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )

        # ------------------------------------------------------------------
        # Stage 3 – cheap stochastic hill‑climbing from the global best
        if best_seq is not None and time.time() < deadline:
            print("--- Local stochastic refinement ---")
            best_seq, best_val = _local_random_search(
                best_seq,
                n,
                conv_fft_len,
                deadline,
                best_val,

    if best_seq is None:

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()

"""
Search for a non‑negative step‑function that minimises the supplied
``evaluate_sequence`` objective.

The algorithm works in four stages:

1. **Diverse parametric pool** – Build a set of candidate shapes on a
   relatively large grid (default n≈8192).  The pool contains uniform,
   Dirichlet draws, symmetric/asymmetric exponentials and power‑laws,
   linear/triangular profiles, a half‑wave sinusoid and a few sparse
   spike vectors.  Generation stops after a configurable number of
   candidates (800) to keep the cost bounded.

2. **Screening** – Evaluate every candidate with an exact FFT‑based
   objective and keep the best few (top_k=3).  These become the starting
   points for the expensive optimisation.

3. **Deep optimisation** – For each start run two phases:
   * *Adam on a smooth‑max surrogate*:  the maximum of the
     self‑convolution is approximated by a log‑sum‑exp with a temperature
     that slowly grows.  Adam updates the logits of the soft‑max
     representation of the probability vector.
   * *Exponentiated‑gradient sub‑gradient*:  after the Adam phase the
     exact max‑convolution is used; the active set (indices attaining the
     maximum) receives uniform weight and the gradient is obtained via a
     cross‑correlation.  The update keeps the vector non‑negative and
     renormalises it onto the simplex.

   Both phases are given generous iteration budgets (≈150 k and 200 k)
   but stop early if the global deadline is reached.

4. **Local stochastic refinement** – A cheap hill‑climbing step perturbs
   the logits with Gaussian noise and keeps any improvement.

All timing is controlled by ``budget_s`` (default 1000 s).  The final
sequence is clipped to the evaluator’s admissible range
[0, 1000] and returned as a plain Python list.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)] for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.1                     # relatively large start

    lam = 2.0                    # temperature for smooth‑max
    lam_factor = 1.006
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val
    it = 0
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        # ----- softened max via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (only for logging and best‑track)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake when stagnant
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            if not _append(shape):
                return candidates

    # asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=6)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    if not _append(shape):
                        return candidates

    # linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * base + (1.0 - w) / n
            shape /= shape.sum()
            if not _append(shape):
                return candidates

    # half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    _append(sinus)

    # sparse spike vectors
    for spikes in (3, 5):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            if not _append(vec):
                return candidates

    return candidates


def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: perturb the logits of the current best vector,
    evaluate the exact objective and keep any improvement.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.02       # modest Gaussian noise
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.  The routine works on a
    grid of size n≈8192 (if time permits) and combines a diverse
    parametric pool, Adam‑smooth‑max optimisation, an extended
    exponentiated‑gradient sub‑gradient phase and a final stochastic
    refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Grid size (fallback to 4096 only if we run out of time later)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] != n:
            # linear interpolation to the required length
            x_old = np.linspace(0, 1, warm_start.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm_start = np.interp(x_new, x_old, warm_start)
        warm_start = np.clip(warm_start, 0.0, 1000.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate and evaluate a broad pool of cheap candidates
    candidates = _generate_parametric_candidates(n, warm_start, max_candidates=800)

    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # Extreme fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
    else:
        cand_vals.sort(key=lambda x: x[0])
        best_val = cand_vals[0][0]
        best_seq = cand_vals[0][1].copy()
        top_k = 3                               # number of starts for deep phase
        top_candidates = [c for _, c in cand_vals[:top_k]]
        print(f"Stage 1: evaluated {len(cand_vals)} candidates, best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Stage 2 – deep optimisation of the most promising starts
        for idx, a0 in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(f"--- Deep optimisation start {idx + 1}/{len(top_candidates)} ---")
            best_seq, best_val = _deep_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )

        # ------------------------------------------------------------------
        # Stage 3 – cheap stochastic hill‑climbing from the global best
        if best_seq is not None and time.time() < deadline:
            print("--- Local stochastic refinement ---")
            best_seq, best_val = _local_random_search(
                best_seq,
                n,
                conv_fft_len,
                deadline,
                best_val,
            )

    # ------------------------------------------------------------------
    # Final safety clipping (the evaluator also clips, but we stay consistent)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
