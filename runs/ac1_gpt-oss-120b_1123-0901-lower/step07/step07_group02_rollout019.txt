"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a list of heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

where `a∗a` is the linear convolution of the vector with itself.
The routine is a self‑contained optimiser that works within a single
CPU core and respects a user‑supplied time budget (default 1000 s).

Main ideas
~~~~~~~~~~
1. **Diverse warm‑starts** – uniform, Dirichlet with several
   concentration parameters, monotone exponential shapes, symmetric
   double‑exponential, Beta‑law shapes, sparse “spike’’ patterns and the
   previously found optimum (`height_sequence_1`).  This gives the optimiser
   many qualitatively different starting points.

2. **Surrogate Adam phase** – a smooth log‑sum‑exp approximation of the
   max‑convolution is optimised by Adam on the log‑softmax (i.e. on the
   simplex).  The temperature parameter is increased until the surrogate
   is essentially the true max.  The number of iterations is larger than in
   the reference implementation.

3. **Exact sub‑gradient phase** – after the Adam phase the true
   non‑smooth objective is attacked with an exponentiated‑gradient (multiplicative)
   sub‑gradient method.  The step‑size decays as 1/√t and a random shake
   is injected when progress stalls.

4. **Cheap local hill‑climbing** – a large number of random mass‑transfer
   moves are tried; a move is kept only if it improves the exact objective.
   This phase helps to escape tiny plateaus that the sub‑gradient cannot
   see.

5. **Upscaling** – if time permits the current best sequence is linearly
   interpolated to a finer grid (8192 points) and the whole pipeline
   (Adam → sub‑gradient → local search) is repeated.  A final
   up‑scale to 16384 points is attempted when still enough time is left.

All phases stop automatically when the deadline is reached, guaranteeing
that a valid (and often very good) sequence is returned before the caller
times out.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where index i is Σ_k w[k]·a[i+k-(n‑1)].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of heterogeneous start vectors."""
    cand: list[np.ndarray] = []

    # warm start and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with different concentration parameters
    for alpha in (0.1, 0.3, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # monotone exponential decays (purely one‑sided)
    for gamma in np.geomspace(0.02, 5.0, num=8):
        shape = np.exp(-gamma * np.arange(n))
        shape /= shape.sum()
        cand.append(shape)

    # reversed exponentials – increasing shape (useful when paired with the
    # decreasing one the optimiser can discover non‑symmetric optima)
    for gamma in np.geomspace(0.02, 5.0, num=8):
        shape = np.exp(gamma * np.arange(n))[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # symmetric double‑exponential (the shape from the reference code)
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 5.0, 8):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Beta‑law shapes – U‑shaped or bell‑shaped depending on parameters
    for a, b in ((0.2, 0.2), (0.5, 0.5), (0.8, 0.8), (2.0, 2.0), (5.0, 5.0)):
        x = np.linspace(0.0, 1.0, n)
        shape = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
        shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    The vector is internally stored as logits (log‑soft‑max) to stay on the
    probability simplex.  Returns the best vector encountered (sum≈1) and
    its exact objective (computed with the true max‑convolution).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise logits
    m = np.zeros_like(logits)          # 1st moment
    v = np.zeros_like(logits)          # 2nd moment

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # a bit smaller than reference (more stable)
    lam = 0.5                         # start temperature for smooth‑max
    lam_factor = 1.04                 # faster temperature increase

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random shake when we are about to run out of time
        if it % 8_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient method on the exact max‑convolution.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set – indices attaining the current maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size η_t = η0 / √t
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (mirror descent on the simplex) and renormalise
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional perturbation when we appear stuck
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a random fraction of mass from one
    index to another and keep the move if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1

    return best_a, best_val

    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin for the caller

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – base resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start (previous best) if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    candidates = _initial_candidates(n, warm) + _spike_candidates(n)

    # ------------------------------------------------------------------
    # Phase 1 – short Adam + brief sub‑gradient for every candidate
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # a few seconds of Adam for the candidate
        cand_deadline = min(deadline - 2.0, time.time() + 11.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=150_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam new best {val_adam:.6f}")

        # a short sub‑gradient refinement
        if time.time() < deadline:
            sub_deadline = min(deadline - 2.0, time.time() + 8.0)
            a_sub, val_sub = _subgradient_opt2(
                best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=30_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long sub‑gradient on the current best
    if best_seq is not None and time.time() < deadline - 2.0:
        a_ref, val_ref = _subgradient_opt2(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=0.7,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random local search
    if best_seq is not None and time.time() < deadline - 2.0:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=300_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – upscale to 8192 if time permits
    if time.time() + 45 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to finer grid
            x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid
        big_deadline = min(deadline - 2.0, time.time() + 30.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=250_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Big‑8192] Adam new best {val_big_adam:.6f}")

        # long sub‑gradient at 8192
        if time.time() < deadline:
            a_big_ref, val_big_ref = _subgradient_opt2(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=600_000,
                eta0=0.7,
            )
            if val_big_ref < best_val:
                best_val = val_big_ref
                best_seq = a_big_ref.copy()
                print(f"[Big‑8192] Subgrad best {val_big_ref:.6f}")

        # local random moves at 8192
        if time.time() < deadline - 2.0:
            a_loc2, val_loc2 = _local_random_search(
                best_seq, n_big, conv_fft_len_big, deadline, max_moves=200_000
            )
            if val_loc2 < best_val:
                best_val = val_loc2
                best_seq = a_loc2.copy()
                print(f"[Local‑8192] improved best {val_loc2:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final upscale to 16384 if still time
    if time.time() + 30 < deadline:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # very short Adam at 16384
        big_deadline = min(deadline - 2.0, time.time() + 18.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=120_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Big‑16384] Adam new best {val_big_adam:.6f}")

        # final sub‑gradient run on 16384
        if time.time() < deadline:
            a_big_ref, val_big_ref = _subgradient_opt2(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=300_000,
                eta0=0.7,
            )
            if val_big_ref < best_val:
                best_val = val_big_ref
                best_seq = a_big_ref.copy()
                print(f"[Big‑16384] Subgrad best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
