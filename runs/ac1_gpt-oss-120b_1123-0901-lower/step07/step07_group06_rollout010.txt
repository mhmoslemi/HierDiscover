def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _fft_cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns an array of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective value (assumes a non‑negative, sum≈1)."""
    conv = _fft_convolve(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float,
                     best_seq: np.ndarray | None,
                     best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of max‑convolution.
    The parameters (learning‑rate, temperature schedule, etc.) are tuned
    for a long run (up to the remaining budget).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # softmax logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    lam = 0.3          # start temperature (lower → smoother)
    lam_factor = 1.04
    lam_cap = 1e9

    last_improve = time.time()
    it = 0
    max_iter = 600_000

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate evaluation
        conv = _fft_convolve(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _fft_cross_correlation(w, a, fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional perturbation if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_exp(a: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float,
                     best_seq: np.ndarray | None,
                     best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient on the true max‑convolution.
    The step‑size decays slowly; early‑stop when no improvement for 4 s.
    """
    a = best_seq if best_seq is not None else a.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.7
    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        conv = _fft_convolve(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _fft_cross_correlation(w, a, fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # decay eta
        eta *= 0.9999

        if time.time() - last_improve > 4.0:
            break

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _projected_gradient_refine(a0: np.ndarray,
                               n: int,
                               fft_len: int,
                               deadline: float,
                               best_seq: np.ndarray | None,
                               best_val: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the exact objective with Euclidean projection onto
    the simplex after each step.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.25
    last_improve = time.time()
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        conv = _fft_convolve(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _fft_cross_correlation(w, a, fft_len)

        # gradient step and projection
        a_candidate = _project_simplex(a - lr * grad_a)

        # evaluate candidate
        conv_c = _fft_convolve(a_candidate, fft_len)
        val_c = 2.0 * n * conv_c.max()

        if val_c < val:
            a = a_candidate
            lr = min(lr * 1.01, 1.0)  # slowly expand step when improvement
        else:
            lr *= 0.68
            if lr < 1e-6:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99995
        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _deep_optimize(start_vec: np.ndarray,
                   n: int,
                   fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Full optimisation pipeline for a single start:
    Adam (smooth‑max) → exponentiated‑gradient → projected‑gradient.
    """
    best_seq, best_val = _adam_smooth_max(
        start_vec, n, fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_exp(
        start_vec, n, fft_len, deadline, best_seq, best_val
    )
    best_seq, best_val = _projected_gradient_refine(
        start_vec, n, fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Produce a large and diverse set of candidate vectors (all sum≈1)."""
    cand_list = []

    # --- warm‑start and rotations -------------------------------------------------
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand_list.append(np.roll(warm_start, shift))

    # --- uniform baseline ---------------------------------------------------------
    cand_list.append(np.full(n, 1.0 / n))

    # --- random Dirichlet draws (many concentrations) ----------------------------
    for conc in (0.01, 0.03, 0.07, 0.15, 0.3, 0.6, 1.0, 2.0, 4.0, 8.0):
        cand_list.append(np.random.dirichlet(np.full(n, conc)))

    # --- exponential families (single‑sided and symmetric) -----------------------
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 0.8, num=60):           # richer grid
        left = np.exp(-alpha * i)
        sym = left + left[::-1]                         # symmetric edges
        for mix in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = mix * sym + (1.0 - mix) / n
            cand_list.append(shape)

    # --- power‑law families -------------------------------------------------------
    for p in np.linspace(0.1, 8.0, num=60):
        pw = (i + 1.0) ** (-p) + (n - i) ** (-p)
        for mix in (0.0, 0.3, 0.6, 1.0):
            shape = mix * pw + (1.0 - mix) / n
            cand_list.append(shape)

    # --- beta‑shaped families (flexible edge/center bias) ------------------------
    alphas = np.linspace(0.1, 8.0, 15)
    betas = np.linspace(0.1, 8.0, 15)
    for a in alphas:
        for b in betas:
            beta = (i + 1) ** (a - 1) * (n - i) ** (b - 1)
            beta = np.maximum(beta, 0.0)
            beta /= beta.sum()
            cand_list.append(beta)

    # --- two‑Gaussian edges -------------------------------------------------------
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.4, 0.8):
        gauss = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        gauss /= gauss.sum()
        cand_list.append(gauss)

    # --- sinusoidal half‑wave ------------------------------------------------------
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand_list.append(s)

    # --- simple triangular / quadratic -------------------------------------------
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand_list.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand_list.append(quad)

    return cand_list


# ----------------------------------------------------------------------
def _local_perturbations(base_seq: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         best_seq: np.ndarray | None,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Stochastic local search around the current best (Gaussian jitter + short sub‑grad)."""
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.03
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s

        a_ref, val_ref = _subgradient_exp(
            a_pert, n, fft_len, deadline, best_seq, best_val
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            # drift gently back toward the centre
            a0 = 0.85 * a0 + 0.15 * base_seq
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a step‑function with minimal evaluation
    score under the 1000 s time budget. Returns the best sequence as a list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Grid size (start with 4096, upscale later if time permits)
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction (if supplied)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – build a large pool of parametric candidates
    all_cands = _generate_parametric_candidates(n, warm_start)

    # Quick evaluation (no gradient) – keep top‑K candidates
    cand_vals = []
    for cand in all_cands:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand /= s
        val = _objective(cand, n, fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 30
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation for each start
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec, n, fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Stage 3 – stochastic local perturbations around the best vector
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Local stochastic perturbations ---")
        best_seq, best_val = _local_perturbations(
            best_seq, n, fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Stage 4 – enlarge to 8192 points if enough time remains
    time_left = deadline - time.time()
    if time_left > 40.0:
        n_big = 8192
        fft_len_big = _next_pow2(2 * n_big - 1)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, fft_len_big, deadline, best_seq, best_val
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big  # remember the final size

    # ------------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
