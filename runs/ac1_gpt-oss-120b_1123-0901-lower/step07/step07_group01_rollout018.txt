"""
Hybrid LP‑Adam optimizer for the step‑function bound

The routine proceeds in three phases:

1. **LP‑descent on a modest grid (n≈2048).**
   For a normalized vector a (∑a=1) we compute M = max(a∗a) and solve the
   linear program
        minimise    t
        subject to  (a∗b)_k ≤ t   (k=0…2n‑2)
                    Σ b_i = 1
                    b_i ≥ 0 , t ≥ 0 .
   The LP gives a feasible b with a strictly smaller cross‑convolution bound.
   We mix a←(1–α)a+αb with a decreasing step‑size α.  A few iterations
   already decrease the maximal autocorrelation by a few‑percent.

2. **Adam optimisation on a smooth‑max surrogate.**
   On the (still small) grid we run Adam on the log‑sum‑exp smooth‑max of the
   convolution, slowly raising the temperature to approach the true max.
   This refines the shape found by the LP phase.

3. **Upscaling and final refinement.**
   The best small‑grid solution is interpolated to the target size
   (4096 points) and a short Adam pass finishes the optimisation.
   The final vector is clipped to the limits required by `evaluate_sequence`.

The whole pipeline respects the 1000 s budget and always returns a
non‑negative sequence with ∑a>0, so `evaluate_sequence` never returns
`np.inf`.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, max_iter: int, deadline: float):
    """
    Perform a few LP updates on a (∑a=1).  Returns the best vector seen and its
    objective value 2·n·max(a∗a).
    """
    n = a.size
    best_a = a.copy()
    best_val = np.inf
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        val = 2.0 * n * M
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP it={it}] new best {val:.6f}")

        # build convolution matrix H (shape (2n‑1, n))
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([np.array([a[0]], dtype=np.float64),
                              np.zeros(n - 1, dtype=np.float64)])
        H = scipy.linalg.toeplitz(col, row)[: 2 * n - 1, :n]

        # LP: minimise t  s.t. H·b ≤ t , Σb = 1 , b ≥ 0 , t ≥ 0
        c = np.concatenate([np.zeros(n), np.array([1.0])])
        A_ub = np.hstack([H, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, M)]

        try:
            res = scipy.optimize.linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception as e:          # pragma: no cover
            print(f"[LP it={it}] LP error: {e}")
            break

        if not res.success:
            print(f"[LP it={it}] LP failed: {res.message}")
            break

        b_opt = res.x[:n]
        t_opt = res.x[-1]

        # if the LP cannot improve the bound we stop
        if t_opt >= M - 1e-12:
            print(f"[LP it={it}] no improvement (t_opt ≈ M)")
            break

        # mixing step – decreasing step size
        alpha = 0.5 / (it + 1)
        a = (1.0 - alpha) * a + alpha * b_opt
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return best_a, best_val


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(a∗a).
    Stops when the deadline expires.
    """
    n = a0.size
    conv_len = _next_pow2(2 * n - 1)
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = 2.0 * n * np.convolve(best_a, best_a).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights  (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, conv_len)

        # Adam update on logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    """
    n = a0.size
    a = np.maximum(a0, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = 2.0 * n * np.convolve(a, a).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set mask
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, _next_pow2(3 * n - 2))

        eta = 0.6 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point required by the competition harness.

    Returns a non‑negative list of heights that (hopefully) improves on the
    current best bound (≈1.503).  The implementation follows the three‑phase
    scheme described in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ---------------------- Warm start / initial vector -----------------
    n_small = 2048                # size used for the LP phase
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.size > 0 and w.sum() > 0:
            # down‑sample / interpolate to n_small points
            x_old = np.linspace(0.0, 1.0, w.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_small, endpoint=False)
            a = np.interp(x_new, x_old, w)
        else:
            a = np.random.dirichlet(np.ones(n_small))
    else:
        a = np.random.dirichlet(np.ones(n_small))

    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n_small))
    a /= a.sum()

    # ---------------------- Phase 1: LP descent ------------------------
    lp_iters = 12
    a, best_val = _lp_improve(a, lp_iters, deadline)

    # ---------------------- Phase 2: Adam refinement -------------------
    a_adam, val_adam = _adam_optimize(a, deadline, max_iters=250_000)
    if val_adam < best_val:
        best_val = val_adam
        a = a_adam.copy()
        print(f"[Phase2] Adam improvement to {val_adam:.6f}")

    # ---------------------- Phase 3: Sub‑gradient fine tune -------------
    a_sub, val_sub = _subgradient_optimize(a, deadline, max_iters=120_000)
    if val_sub < best_val:
        best_val = val_sub
        a = a_sub.copy()
        print(f"[Phase3] Sub‑gradient improvement to {val_sub:.6f}")

    # ---------------------- Optional up‑scaling -----------------------
    if time.time() + 30 < deadline:
        n_big = 4096
        x_small = np.linspace(0.0, 1.0, a.size, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, a)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        # short Adam pass on the large grid
        a_big_adam, val_big = _adam_optimize(
            a_big, deadline, max_iters=100_000
        )
        if val_big < best_val:
            best_val = val_big
            a = a_big_adam.copy()
            print(f"[Upscale] improvement to {val_big:.6f}")

    # ---------------------- Final safety clipping --------------------
    best_seq = np.clip(a, 0.0, 1000.0).tolist()

    # report the value we think we achieved (and let the judge double‑check)
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:  # pragma: no cover
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq
