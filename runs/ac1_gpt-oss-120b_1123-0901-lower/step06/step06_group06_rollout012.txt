def _build_initial_candidates(n, best_seq=None):

    candidates = []

    # If best_seq given, pad/truncate/rescale.
    if best_seq is not None:
        h = np.asarray(best_seq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            # upscale via linear interpolation
            x_old = np.linspace(0, 1, h.shape[0])
            x_new = np.linspace(0, 1, n)
            h = np.interp(x_new, x_old, h)
        h = np.maximum(h, 0.0)
        if h.sum() > 1e-12:
            h = h / h.sum()
            candidates.append(h)

    # 1) Dirichlet random (1)
    candidates.append(np.random.dirichlet(np.ones(n)))

    # 2) Dirichlet spiky (conc <1)
    for conc in [0.2, 0.4, 0.6, 0.8]:
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 3) Dirichlet overdispersed (conc >1)
    for conc in [1.5, 2.0, 3.0]:
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4) Power-law edge shape:
    idx = np.arange(1, n+1)
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-2, -0.2, 7):
        shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # 5) Exponential decays (symmetrical)
    for gamma in [0.001, 0.003, 0.006]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n-1-i))
        shape = shape / shape.sum()
        candidates.append(shape)

    # 6) Cosine window
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n-1)))
    coswin = coswin / coswin.sum()
    candidates.append(coswin)

    # 7) Half-sine window
    sinwin = np.sin(np.pi * i / (n-1))
    sinwin = sinwin / sinwin.sum()
    candidates.append(sinwin)

    # 8) Gaussian blob
    sigma = n/8.0
    gauss = np.exp(-0.5 * ((i - (n-1)/2)/sigma)**2)
    gauss = gauss / gauss.sum()
    candidates.append(gauss)

    # 9) Arcsine shape
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    candidates.append(arcs)

    # 10) Random sparse pattern
    m = max(2, int(np.sqrt(2*n)))
    sparse = np.zeros(n)
    idxs = np.random.choice(n, size=m, replace=False)
    sparse[idxs] = np.random.rand(m) + 0.1
    sparse = sparse / sparse.sum()
    candidates.append(sparse)

    # 11) Beta distributions: (alpha,beta) in grid
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    betas = [0.2, 0.5, 1.0, 2.0, 5.0]
    x = (np.arange(n) + 0.5) / n   # midpoints
    for a in alphas:
        for b in betas:
            shape = (x**(a-1)) * ((1-x)**(b-1))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

    # Remove duplicates? Not needed.

    return candidates

def _adam_optimize(a0, n, conv_len, corr_len, deadline, best_val, best_seq):
    a = a0.copy()
    # Normalize
    if a.sum() == 0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a.sum()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # Learning rate schedule
    lr = 0.08 * np.sqrt(2048.0 / n)
    lam = 0.2
    lam_factor = 1.015
    # iteration counts based on n and remaining time; we will break anyway
    max_iters = int(40000 * 1024 / n) + 5000   # similar to earlier but maybe more
    last_improve = time.time()
    for t in range(1, max_iters+1):
        # compute a
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        # softmax weights
        w = np.exp(lam * (conv - max_c))
        # stabilize
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient
        grad = 2.0 * _corr_fft(w, a, corr_len)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature increase
        lam = min(lam * lam_factor, 1e6)
        # compute exact objective
        sum_a = a.sum()
        val = 2.0 * n * max_c / (sum_a * sum_a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")
        # random shake if stagnation
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()
        if time.time() > deadline:
            break
    return best_seq, best_val

def _subgrad_optimize(a, n, conv_len, corr_len, deadline, best_val, best_seq):
    if a.sum() == 0:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    eta = 0.55
    last_improve = time.time()
    max_iters = int(8000 * 1024 / n) + 2000
    for it in range(max_iters):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        # active indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _corr_fft(w, a, corr_len)
        # EG update
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        # compute val
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")
        if time.time() - last_improve > 5.0:
            break
        if time.time() > deadline:
            break
    return best_seq, best_val

def _hill_climb(a, n, conv_len, deadline, best_val, best_seq):
    a = a.copy()
    epsilon = 0.1
    total_steps = 5000
    for step in range(total_steps):
        # propose perturbation
        # draw random direction vector d of size n with mean zero and sum zero: generate random gaussian, then subtract mean.
        d = np.random.randn(n)
        d = d - d.mean()
        # scale to have L2 norm epsilon?
        d = d / np.linalg.norm(d) * epsilon
        a_new = a + d
        # enforce nonnegative
        a_new = np.maximum(a_new, 0.0)
        # renorm
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new = a_new / s
        # evaluate
        conv = _conv_fft(a_new, conv_len)
        val = 2.0 * n * conv.max() / (a_new.sum() ** 2)  # sum=1
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[n={n} HillClimb step={step}] new best {val:.6f}")
            epsilon = min(epsilon*1.05, 0.5)   # increase epsilon after success
        else:
            epsilon = epsilon * 0.99
        if time.time() > deadline:
            break
    return best_seq, best_val

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    global height_sequence_1

    best_seq = None
    best_val = float('inf')

    # Determine initial best_seq from existing global
    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(height_sequence_1, dtype=np.float64)
            if seq.size > 0:
                best_seq = seq / max(seq.sum(), 1e-9) # normalized
                # evaluate
                best_val = evaluate_sequence(best_seq.tolist())
                print(f"Loaded prior best value {best_val}")
        except Exception as e:
            best_seq = None
            best_val = float('inf')
    # define lengths to explore
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        # compute FFT lengths (next power of 2)
        conv_len = 1 << int(np.ceil(np.log2(2*n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(2*n - 1 + n - 1)))
        # Build candidates using best_seq from previous length (if any) and diverse shapes
        candidates = _build_initial_candidates(n, best_seq if best_seq is not None and best_seq.shape[0] <= n else None)
        # Possibly limit number of candidates if time runs out
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            # initial candidate normalized
            a = cand.astype(np.float64)
            if a.sum() == 0:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            # Adam stage
            best_seq, best_val = _adam_optimize(a, n, conv_len, corr_len, deadline, best_val, best_seq)
            # Subgradient stage
            if time.time() > deadline:
                break
            best_seq, best_val = _subgrad_optimize(best_seq if best_seq.shape[0]==n else a, n, conv_len, corr_len, deadline, best_val, best_seq)
        # Fine-tune for current length if we have best_seq of that length
        if best_seq is not None and best_seq.shape[0] == n:
            print(f"[n={n}] fine-tuning global best (val={best_val:.6f})")
            best_seq, best_val = _adam_optimize(best_seq, n, conv_len, corr_len, deadline, best_val, best_seq)
            # Hill climb
            best_seq, best_val = _hill_climb(best_seq, n, conv_len, deadline, best_val, best_seq)
        # If we have best_seq, prepare for next length: upscale to next length via interpolation (done in build candidates)
        # store global variable for next run
        height_sequence_1 = best_seq.tolist() if best_seq is not None else height_sequence_1

    # final fallback uniform
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
    print(f"Best value found: {best_val}")
    # assign to global for possible next invocation
    height_sequence_1 = best_seq.tolist()
    return best_seq.tolist()

"""
Propose a non‑negative step function (list of heights) with a low
evaluation score F(a)=2·n·max(a*a)/(sum(a)²).

The search proceeds in three phases, repeated for increasing lengths
(1024 → 2048 → 4096 → 8192 → 16384 if time permits):

1. **Diversified initialisation** – a large pool of deterministic shapes
   (Dirichlet draws, power‑law, exponential, cosine, Gaussian, arcsine,
   β‑distributions, sparse spikes) plus an up‑scaled copy of the best
   sequence found so far.

2. **Two‑stage optimisation** for each candidate  
   • *Adam* on a smooth‑max surrogate of the true max (soft‑max weighting
     of the convolution). A temperature λ grows exponentially so that the
     surrogate gradually approaches the true max.  
   • *Exact sub‑gradient* (exponentiated‑gradient) phase where the
     gradient is taken only on the active (maximal) convolution entries.

   Both stages keep the vector on the simplex via a soft‑max
   (logits) representation.

3. **Global fine‑tuning** – the current overall best sequence of the
   current length is further optimised with a long Adam run, followed by
   a cheap hill‑climbing / simulated‑annealing pass that perturbs the
   vector and accepts improvements.

All heavy work (convolution and cross‑correlation) uses FFTs
(O(n log n)).  The routine respects the supplied time budget and
returns the best sequence discovered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (a*a) and the cross‑correlation needed for
# the gradient.

def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Candidate generation ---------------------------------------------------

def _build_initial_candidates(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diversified pool of starting points for length ``n``.
    All candidates are normalised to sum = 1.
    """
    cand = []

    # 0) Upscaled copy of the previous best (if any)
    if prev_seq is not None:
        h = prev_seq.astype(np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            # linear interpolation to the new size
            x_old = np.linspace(0.0, 1.0, h.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            h = np.interp(x_new, x_old, h)
        h = np.maximum(h, 0.0)
        if h.sum() > 1e-12:
            h = h / h.sum()
            cand.append(h)

    # 1) pure Dirichlet (conc = 1)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 2) spiky Dirichlet (conc < 1)
    for conc in (0.2, 0.4, 0.6, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 3) over‑dispersed Dirichlet (conc > 1)
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # 5) symmetric exponential decay
    for gamma in (0.001, 0.003, 0.006):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)

    # 6) cosine window (single bump)
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    # 7) half‑sine window
    sinwin = np.sin(np.pi * i / (n - 1))
    sinwin = sinwin / sinwin.sum()
    cand.append(sinwin)

    # 8) discrete Gaussian
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    # 9) truncated arcsine
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    cand.append(arcs)

    # 10) random sparse pattern (≈√n spikes)
    m_spikes = max(2, int(np.sqrt(2 * n)))
    sparse = np.zeros(n, dtype=np.float64)
    idxs = np.random.choice(n, size=m_spikes, replace=False)
    sparse[idxs] = np.random.rand(m_spikes) + 0.1
    sparse = sparse / sparse.sum()
    cand.append(sparse)

    # 11) β‑distributions on a grid of parameters
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    betas = [0.2, 0.5, 1.0, 2.0, 5.0]
    x = (np.arange(n) + 0.5) / n          # mid‑point mesh on [0,1]
    for a in alphas:
        for b in betas:
            shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                cand.append(shape)

    # Shuffle to avoid systematic ordering (helps random shuffling later)
    np.random.shuffle(cand)

    # For very large n we keep only a modest subset of the pool
    if n >= 8192:
        cand = cand[:12]   # enough diversity, cheap enough

    return cand

# Optimisation primitives ------------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_len: int,
                   corr_len: int,
                   deadline: float,
                   best_val: float,
                   best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the (possibly) updated global best sequence/value.
    """
    a = a0.copy()
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # learning‑rate scaling with problem size
    lr = 0.08 * np.sqrt(2048.0 / n)

    # smooth‑max temperature
    lam = 0.2
    lam_factor = 1.015          # a bit slower growth than the previous run

    last_improve = time.time()
    # iteration budget: grow for smaller n, shrink for larger n
    max_iters = int(50000 * 1024 / n) + 8000

    for t in range(1, max_iters + 1):
        # ----- simplex projection via soft‑max -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- convolution and smooth‑max weights -----
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth‑max surrogate -----
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- exact objective (scale‑invariant) -----
        sum_a = a.sum()
        val = 2.0 * n * max_c / (sum_a * sum_a)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}]  new best {val:.6f}")

        # ----- stagnation handling -----
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_seq, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_len: int,
                      corr_len: int,
                      deadline: float,
                      best_val: float,
                      best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient phase using exponentiated gradient.
    """
    a = a0.copy()
    a = a / a.sum()
    eta = 0.55                     # step size for EG
    last_improve = time.time()
    max_iters = int(8000 * 1024 / n) + 3000

    for it in range(max_iters):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()

        # active set of indices attaining the max (tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # exact objective
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:04d}] new best {val:.6f}")

        # early abort if no improvement for a while
        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_seq, best_val


def _hill_climb(a0: np.ndarray,
                n: int,
                conv_len: int,
                deadline: float,
                best_val: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing (simulated‑annealing style) that
    perturbs the vector while keeping the simplex constraint.
    """
    a = a0.copy()
    epsilon = 0.05                      # initial perturbation magnitude
    steps = 4000

    for step in range(steps):
        # random zero‑mean perturbation
        d = np.random.randn(n)
        d -= d.mean()
        if np.linalg.norm(d) == 0:
            continue
        d = d / np.linalg.norm(d) * epsilon
        a_new = a + d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new = a_new / s

        conv = _conv_fft(a_new, conv_len)
        val = 2.0 * n * conv.max() / (a_new.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            epsilon = min(epsilon * 1.04, 0.5)   # slowly enlarge after a win
            print(f"[n={n} HillClimb step={step:04d}] new best {val:.6f}")
        else:
            epsilon *= 0.985                     # anneal otherwise

        if time.time() > deadline:
            break

    return best_seq, best_val
