        if shape.sum()>0:

    # beta shapes

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution (numpy.convolve).

Algorithm
---------
* Work on the simplex (sum=1) – we keep a “logit’’ representation so that
  positivity and the sum constraint are automatically satisfied.
* For each target length `n` we build a rich pool of deterministic
  initial shapes (Dirichlet, power‑law, beta‑type, exponential,
  Gaussian, arcsine, tent, sparse spikes …).  If a previous best
  solution (`height_sequence_1`) exists it is also used as a warm‑start.
* Every candidate is refined in two stages:
    1. Adam optimisation on a smooth‑max surrogate.  The surrogate is
       the log‑sum‑exp of the convolution, implemented via a soft‑max
       weighting `w`.  The temperature `lam` is increased slowly so that
       the surrogate converges to the true max.
    2. An exponentiated‑gradient (multiplicative sub‑gradient) phase
       that directly minimises the exact objective on the active set
       (indices where the convolution attains its current maximum).
* After all candidates for a given length have been processed we
  fine‑tune the current global best with another Adam run (longer
  schedule, smaller learning‑rate).  The whole search respects the
  provided time budget (default 1000 s).
* The best sequence found (as a plain Python list) is stored in the
  global variable `height_sequence_1` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution a*a and the cross‑correlation needed for
# the gradient of the smooth‑max surrogate.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """Construct a diversified set of starting vectors (sum = 1)."""
    cand = []

    # 0) warm‑start from previous global solution, if any
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-12:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # 1) Dirichlet families
    cand.append(np.random.dirichlet(np.ones(n)))            # concentration = 1
    for conc in (0.2, 0.4, 0.6, 0.8):                       # spiky
        cand.append(np.random.dirichlet(np.full(n, conc)))
    for conc in (1.5, 2.5, 3.5):                            # over‑dispersed
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 2) power‑law on the “triangular’’ index product
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-4.0, -0.1, 12):
        shape = np.where(prod > 0,
                         np.power(prod, alpha, where=prod > 0,
                                  out=np.zeros_like(prod)), 0.0)
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 3) beta‑type shape: (i+1)^{p‑1}·(n‑i)^{q‑1}
    i = np.arange(n, dtype=np.float64)
    for p in np.linspace(0.2, 5.0, 5):
        for q in np.linspace(0.2, 5.0, 5):
            shape = (i + 1) ** (p - 1) * (n - i) ** (q - 1)
            if shape.sum() > 0:
                cand.append(shape / shape.sum())

    # 4) two‑sided exponential decay
    for gamma in (0.0005, 0.001, 0.002, 0.005, 0.01):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # 5) discrete Gaussian (σ proportional to n)
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    cand.append(gauss / gauss.sum())

    # 6) truncated arcsine (inverse‑square‑root edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    cand.append(arcs / arcs.sum())

    # 7) smooth tent (linear rise‑then‑fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:               # odd length – add a centre point
        tent = np.concatenate([tent, [0.0]])
    cand.append(np.maximum(tent, 0) / tent.sum())

    # 8) cosine window (single bump)
    x = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    cand.append(coswin / coswin.sum())

    # 9) random sparse pattern (≈ sqrt(n) spikes)
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=m_spikes, replace=False)
    sparse[indices] = 1.0
    if sparse.sum() > 0:
        cand.append(sparse / sparse.sum())

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Respects the given time budget and returns the best sequence as a
    plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths explored – we go a bit beyond the previous 8192.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for each n.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        candidates = _build_initial_candidates(n)

        # ------------------------------------------------------------------
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1) and switch to logits.
            a = a0.copy()
            a = a / a.sum()
            logits = np.log(a + 1e-15)

            # ---------- Adam on smooth‑max surrogate ----------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            # Learning‑rate scales mildly with problem size.
            lr = 0.09 * np.sqrt(1024.0 / n)

            # Temperature (smooth‑max) schedule.
            lam = 0.15
            lam_factor = 1.020

            last_improve = time.time()
            max_iters = int(20000 * 1024 / n) + 3000   # more iterations than before

            for t in range(1, max_iters + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate (linear in w)
                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature increase – approach the true max
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a)==1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t:05d}] new best {val:.6f}")

                # gentle random shake when stagnating
                if time.time() - last_improve > 6.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---------- Exact max: exponentiated‑gradient (sub‑gradient) ----------
            a = a / a.sum()
            eta = 0.55                      # step‑size for EG
            sub_last_improve = time.time()
            max_sub_iters = 8000

            for sub_it in range(max_sub_iters):
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # active set (indices attaining the current maximum)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                w = np.zeros_like(conv)
                if active.any():
                    w[active] = 1.0 / active.sum()

                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                # avoid collapse to zero
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * np.max(_conv_fft(a, conv_len))

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} EG iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 7.0:
                    break
                if time.time() > deadline:
                    break

        # ---------- fine‑tune the current global best for this length ----------
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.045 * np.sqrt(1024.0 / n)
            lam = 0.35
            lam_factor = 1.015

            last_improve = time.time()
            fine_iters = int(40000 * 1024 / n) + 5000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 9.0:
                    # small random perturbation to escape shallow plateaus
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – a uniform vector (should never be needed).
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # expose the best sequence for the next run
    globals()["height_sequence_1"] = best_seq.tolist()

    # Return a plain Python list – the external evaluator expects a list.
    return best_seq.tolist()
