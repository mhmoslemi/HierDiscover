def propose_candidate(seed=12345, budget_s=1000):

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    # maybe load height_sequence_1 if exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # noqa: F821
    except Exception:
        base = None

    # choose lengths
    # start with some baseline
    lengths = [1500, 1800, 2100, 2400, 3000, 3600, 4200]
    lengths = [l for l in lengths if l <= 8000]
    best_seq = None
    best_val = float('inf')

    for n in lengths:
        t_remaining = deadline - time.time()
        if t_remaining < 10:
            break

        target_sum = math.sqrt(2.0 * n)
        # generate initial vectors
        inits = _make_initial_vectors(n, target_sum, rng)
        if base is not None:
            # interpolate base to this n and add
            old_x = np.linspace(0, 1, base.shape[0])
            new_x = np.linspace(0, 1, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            inits.append(interp)

        # decide iteration budget per run: roughly proportional to n
        max_iters = int(3_000_000 * 2100 / n)  # up to 3M for smallest n
        max_iters = max(200_000, max_iters)

        for idx, a0 in enumerate(inits):
            if time.time() >= deadline:
                break
            # PGD
            seq, val = _pgd_optimize(a0, n, deadline, max_iters, best_seq if best_seq is not None else a0, best_val, rng)
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"[global] new best M={best_val:.6f} at n={n}")

            # Adam style
            seq, val = _adam_optimize(a0, n, deadline, max_iters, best_seq if best_seq is not None else a0, best_val, rng)
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"[global] new best M={best_val:.6f} at n={n}")

    # final refinement
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5:
            refine_deadline = time.time() + min(remaining * 0.5, 30.0)
            best_seq, best_val = _local_refine(best_seq, len(best_seq), refine_deadline, best_val, rng)

    if best_seq is None:

        best_seq = np.full(n, target_sum / n)

    print(f"[done] best bound (max conv) = {best_val:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
* Fix the ℓ₁‑norm of the vector to  √(2 n)  (the objective is scale invariant).
* For a small set of candidate lengths (≈ 2 k–3 k) generate diverse feasible
  starting points (uniform, Dirichlet, exponential, Hann, Blackman, …).
* For each start run **Projected Sub‑Gradient Descent** on the exact
  max‑convolution.  The sub‑gradient is obtained from the active (max‑attaining)
  convolution indices and a single FFT gives the required reverse convolution.
  After each step the vector is projected onto the simplex.
* In parallel run a **soft‑max Adam optimiser** (exponentiated‑gradient on the
  simplex) which uses a temperature‑controlled smooth approximation of the
  max‑convolution.
* The best vector seen so far (lowest max‑convolution) is kept globally.
* After the allocated optimisation time a cheap stochastic refinement
  (mass‑transfer simulated‑annealing) is applied to the global best.
* The routine respects the 1000 s budget, dynamically splitting the budget
  between lengths and optimisation phases.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum} (Duchi et al.)."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho = rho_idx[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Return a short list of diverse feasible start vectors (sum = target_sum)."""
    init = []

    # uniform
    init.append(np.full(n, target_sum / n))

    # Dirichlet (Gamma(1,1))
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(target_sum * g / g.sum())

    # exponential decay, random rate
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(target_sum * x / x.sum())

    # reversed exponential
    init.append(target_sum * x[::-1] / x.sum())

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(target_sum * hann / hann.sum())

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(target_sum * black / black.sum())

    # small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n,
                                                   size=n)
    pert = np.maximum(pert, 0.0)
    init.append(target_sum * pert / pert.sum())

    # triangular shape
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(target_sum * tri / tri.sum())

    # sparse random spikes
    k_spikes = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k_spikes, replace=False)
    vals = rng.random(k_spikes)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init


def _max_convolution(a: np.ndarray, Nfft: int) -> (float, np.ndarray):
    """Exact max of a*a (linear convolution) using one FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max()), conv


def _weighted_rev_convolution(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i},
    i.e. the weighted reverse convolution needed for the sub‑gradient.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # linear part
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev

# Optimisers -----------------------------------------------------------


def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int,
                  best_seq: np.ndarray,
                  best_score: float,
                  rng: np.random.Generator) -> (np.ndarray, float):
    """
    Projected (sub‑)gradient descent on the exact max‑convolution.
    Returns the best vector observed and the corresponding max‑convolution.
    """
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    a = a * target_sum / a.sum()          # enforce simplex

    Nfft = 1 << ((3 * n - 2).bit_length())     # enough for 3n‑2 linear conv
    step0 = 0.001 * target_sum                  # base step size
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution & max
        M, conv = _max_convolution(a, Nfft)

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:8d}] new best max = {best_score:.6f}")

        # active set (hard max)
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # sub‑gradient via weighted reverse convolution
        rev = _weighted_rev_convolution(a, w, Nfft)
        grad = 2.0 * rev

        # gradient step with simple annealing
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * grad
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter to escape plateaux
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> (np.ndarray, float):
    """
    Exponentiated‑gradient Adam optimiser using a soft‑max approximation of the max‑convolution.
    Returns the best vector seen and its exact max‑convolution.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)                     # logits on the simplex

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08

    Nfft = 1 << ((3 * n - 2).bit_length())
    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current point = softmax(logits) scaled to target_sum
        logits = y - np.max(y)                     # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # convolution & max
        M, conv = _max_convolution(a, Nfft)

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n:5d} it={it:8d}] new best max = {best_score:.6f}")

        # soft‑max weights for the max (smooth max)
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()

        # gradient wrt a
        rev = _weighted_rev_convolution(a, w, Nfft)
        g_a = 2.0 * rev

        # chain rule to logits (mirror‑descent on simplex)
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # jitter
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _local_refine(a: np.ndarray,
                  n: int,
                  deadline: float,
                  best_score: float,
                  rng: np.random.Generator) -> (np.ndarray, float):
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly transfer a small fraction of mass from a random index i to j.
    Accept improvements always; else accept with a decaying simulated‑annealing probability.
    """
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for max‑convolution (2n‑1 is sufficient)
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 250_000
    step = 0

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 %–5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score
