"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm overview
------------------
* **Initialisation** – start from the previously found best sequence (if
  available) and a diverse set of random feasible vectors on the simplex
  Σa_i = √(2 n).  Both plain and *symmetric* (mirror‑image) vectors are
  generated – symmetry often yields lower peak autocorrelation.
* **Two complementary continuous optimisers**
    - *Projected Gradient Descent* (PGD) on the exact max‑convolution.
      The sub‑gradient is computed from the (soft) active set; after each
      gradient step the vector is projected onto the simplex.  An
      annealed step‑size schedule and occasional jitter avoid stagnation.
    - *Exponentiated‑gradient Adam* – optimisation on the logits of a
      soft‑max representation of the vector.  A temperature schedule yields
      a smooth approximation of the max‑convolution; the true max value is
      monitored after every iteration.
* **Heavy stochastic refinement** – a simulated‑annealing hill‑climber
  performs mass‑transfer moves (i → j) on the current global best.  Moves
  are accepted if they lower the convolution peak, otherwise with a
  temperature‑dependent probability.  The refinement runs as long as the
  time budget permits (up to several hundred thousand proposals).
* **Length search** – a handful of lengths around the known good size
  (≈ 1900) are explored (including a larger length ≈ 4000).  For each length
  the above optimisation pipeline is executed; the best overall sequence
  is returned.

All helper functions are top‑level and contain no closures.  The code obeys
the 1000 s time budget by checking the deadline frequently.
"""

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target_sum}.
    Implements the algorithm of Duchi et al. (2008) in O(n log n).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """
    Compute the exact maximum of the linear convolution a * a.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _weighted_rev_convolution(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute the weighted reverse convolution needed for the sub‑gradient:

        rev_i = Σ_k w_k * a_{k-i}   (indices outside the array are ignored)

    The calculation is performed with FFTs (O(N log N)).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)   # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part: length = len(w) + len(a) - 1
    cut_len = len(w) + a.shape[0] - 1
    conv_rev = conv_rev[:cut_len]
    # indices that map back to a (see original code)
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator, symmetric: bool = False
) -> list[np.ndarray]:
    """
    Generate a small, diverse set of feasible start vectors.
    If ``symmetric`` is True, the vector is mirrored (a_i = a_{n-1-i}).
    """
    init = []

    # 1) uniform
    vec = np.full(n, target_sum / n)
    if symmetric:
        vec = (vec + vec[::-1]) / 2.0
    init.append(vec)

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    vec = g / g.sum() * target_sum
    if symmetric:
        vec = (vec + vec[::-1]) / 2.0
        vec = vec / vec.sum() * target_sum
    init.append(vec)

    # 3) exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    vec = x / x.sum() * target_sum
    if symmetric:
        vec = (vec + vec[::-1]) / 2.0
        vec = vec / vec.sum() * target_sum
    init.append(vec)

    # 4) reversed exponential
    vec = x[::-1] / x[::-1].sum() * target_sum
    if symmetric:
        vec = (vec + vec[::-1]) / 2.0
        vec = vec / vec.sum() * target_sum
    init.append(vec)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    vec = hann / hann.sum() * target_sum
    if symmetric:
        vec = (vec + vec[::-1]) / 2.0
        vec = vec / vec.sum() * target_sum
    init.append(vec)

    # 6) sparse random spikes (helps explore highly non‑uniform shapes)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    if symmetric:
        sparse = (sparse + sparse[::-1]) / 2.0
        sparse = sparse / sparse.sum() * target_sum
    init.append(sparse)

    # 7) small random perturbation of uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(
        scale=0.01 * target_sum / n, size=n
    )
    pert = np.maximum(pert, 0.0)
    vec = pert / pert.sum() * target_sum
    if symmetric:
        vec = (vec + vec[::-1]) / 2.0
        vec = vec / vec.sum() * target_sum
    init.append(vec)

    return init


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    symmetric: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed and its exact objective (max conv).
    """
    target_sum = math.sqrt(2.0 * n)

    # Initialise on the simplex
    a = a0.copy()
    a = a * target_sum / a.sum()

    # FFT size – must accommodate linear convolution length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # Simple decreasing step‑size schedule
    # Step scale factor 0.003 gives about 0.3 % of total mass per step initially.
    step0 = 0.003 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution & current max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # Update global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- active set (hard max) -----
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ----- weighted reverse convolution (sub‑gradient) -----
        rev = _weighted_rev_convolution(a, w, Nfft)
        g = 2.0 * rev  # gradient of max conv w.r.t a

        # ----- gradient step on the simplex -----
        lr = step0 / math.sqrt(it)  # annealing
        a_tmp = a - lr * g

        if symmetric:
            # enforce mirror symmetry before projection
            a_tmp = (a_tmp + a_tmp[::-1]) / 2.0

        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- occasional jitter to avoid stagnation -----
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            if symmetric:
                jitter = (jitter + jitter[::-1]) / 2.0
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    symmetric: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logits (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.  Returns best vector seen.
    """
    target_sum = math.sqrt(2.0 * n)

    # Initialise logits from the start vector
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.09  # larger base LR for logits

    # Temperature schedule parameters
    tau_factor = 0.35
    decay_rate = max_iters / 6.0

    # FFT size
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector (softmax of logits) -----
        logits = y - np.max(y)  # stability shift
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        if symmetric:
            a = (a + a[::-1]) / 2.0
            a = a / a.sum() * target_sum  # re‑normalise (symmetry preserves sum)

        # ----- convolution and max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # fall back to hard max
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            sum_ex = ex.sum()
            w = ex / sum_ex

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev_convolution(a, w, Nfft)
        g_a = 2.0 * rev  # ∂f/∂a

        # ----- chain rule to logits (mirror‑descent on simplex) -----
        avg = np.dot(p, g_a)  # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)  # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter on logits
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            y += jitter

    return best_seq, best_score


def _local_sa_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    cur_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the simplex.
    Randomly transfers a small fraction of mass from i → j.
    Accepts worsening moves with a temperature that decays linearly.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_max = cur_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    # Determine a generous step budget based on remaining time
    # Rough estimate: each iteration ≈ 0.0004 s for n≈2000, so ~2 M steps ~800 s.
    # We'll cap at 2 M moves.
    max_steps = 2_000_000
    step = 0

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            step += 1
            continue

        if cur_seq[i] < 1e-14:
            step += 1
            continue

        # Random fraction to move, size 0.5 %–3 %
        frac = rng.uniform(0.005, 0.03)
        delta = cur_seq[i] * frac

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # Compute new max convolution (fast via FFT)
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_max = float(conv.max())

        # Acceptance rule
        if new_max < cur_max - 1e-12:
            cur_seq = new_seq
            cur_max = new_max
            if new_max < best_score_global:  # update global if improved
                global_best_seq_global[:] = cur_seq
                best_score_global = new_max
            # occasional progress report
            if step % 50000 == 0:
                print(f"[refine] new global best = {new_max:.6f} (step {step})")
        else:
            # Temperature linearly decays from a fraction of current max
            T0 = max(0.02 * cur_max, 1e-5)
            T = T0 * (1.0 - step / max_steps)
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_max = new_max
        step += 1

    return cur_seq, cur_max
