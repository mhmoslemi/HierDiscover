def smooth_max_adam(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=5000,
                   lr=0.1, init_lam=0.5, lam_factor=1.02):
    a = np.maximum(a0, 1e-12)
    a = a / a.sum()
    logits = np.log(a)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lam = init_lam
    best_val = float('inf')
    best_a = a.copy()
    last_improve = time.time()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log/exp_log.sum()
        # conv
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0/w.size)
        # gradient of smooth max wrt a
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        logits -= lr * m_hat/(np.sqrt(v_hat)+epsilon)
        # anneal lam
        lam = min(lam*lam_factor, 1e8)
        # evaluate real objective
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        # stagnation: random perturbation
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n)-0.5)*1e-3
            last_improve = time.time()
    return best_a, best_val

def subgradient_exponential(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=3000, eta0=0.5):
    a = a0 / a0.sum()
    best_val = float('inf')
    best_a = a.copy()
    eta = eta0
    last_improve = time.time()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # identify active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad_a)
        # normalization
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / s
        # evaluate
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        eta *= 0.9995
        # early stop if no improvement for some time
        if time.time() - last_improve > 10.0:
            break
    return best_a, best_val

"""
Propose a non‑negative step‑function (vector) that minimises

    F(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a multi‑scale approach:
  * start from small lengths and progressively up‑sample,
  * initialise many diverse seeds (uniform, power‑law, Gaussian, triangular,
    random Dirichlet, previous best, global variable …),
  * phase‑1 : Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) with a
    soft‑max parametrisation (logits → probabilities) to keep a on the simplex,
  * phase‑2 : exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient
    update (mirror descent w.r.t. KL‑divergence),
  * final jitter around the current global best and a last fine‑tuning pass.
The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length `fft_len`.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 3000,
    lr: float = 0.10,
    init_lam: float = 0.5,
    lam_factor: float = 1.015,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2*n*max(conv) for the best
    iterate discovered before the deadline.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lam = init_lam
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- soft‑max (re‑parameterisation) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and smooth‑max weights ----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # ---- gradient of smooth‑max w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ---- back‑propagation through soft‑max ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature annealing ----
        lam = min(lam * lam_factor, 1e8)

        # ---- objective evaluation (exact, because sum(a)=1) ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- occasional random perturbation to avoid stagnation ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.
    Returns (best_a, best_val) after at most `max_iters` iterations or when the
    deadline is reached.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = eta0
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:               # fallback to a random distribution
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995               # slowly decay step size

        # early exit if no improvement for a while
        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _generate_seeds(n: int, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Create a heterogeneous pool of seeds for length `n`.
    The pool includes uniform, interpolated old solutions, global variable,
    random Dirichlet draws and a wide variety of parametric shapes.
    """
    seeds: list[np.ndarray] = []

    def _add(arr: np.ndarray):
        if arr is None:
            return
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s > 0:
            arr = arr / s
            seeds.append(arr.astype(np.float64))

    # 1) uniform
    _add(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) warm‑start from previous global best (if same length)
    if best_seq is not None and best_seq.shape[0] == n:
        _add(best_seq.copy())

    # 3) up‑sample / down‑sample a previous best of a different size
    if best_seq is not None and best_seq.shape[0] != n:
        old = best_seq.astype(np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        _add(up)

    # 4) historic global variable, if present
    gseq = globals().get("height_sequence_1")
    if gseq is not None:
        g = np.asarray(gseq, dtype=np.float64)
        if g.shape[0] > n:
            g = g[:n]
        elif g.shape[0] < n:
            g = np.pad(g, (0, n - g.shape[0]))
        _add(g)

    # 5) a few random Dirichlet draws
    for _ in range(3):
        _add(np.random.dirichlet(np.ones(n)))

    # 6) power‑law shapes (edge‑concentrated and centre‑concentrated)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in [-3.0, -2.5, -2.0, -1.5, -1.0, -0.75, -0.5, -0.25, 0.25, 0.5]:
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        _add(shape)

    # 7) edge‑Gaussian shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.05, 0.1, 0.2, 0.4, 0.8]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        _add(shape)

    # 8) central Gaussian shapes
    mu = (n - 1) / 2.0
    for gamma in [0.01, 0.05, 0.1, 0.2]:
        shape = np.exp(-gamma * ((i - mu) ** 2))
        _add(shape)

    # 9) triangular (tent) shape
    tri = np.minimum(i + 1, n - i)
    _add(tri)

    # 10) sine‑squared bump (smooth bell)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    _add(shape)

    # 11) mixture of uniform and edge‑Gaussian
    edge = np.exp(-0.5 * (i ** 2)) + np.exp(-0.5 * ((n - 1 - i) ** 2))
    edge = edge / edge.sum()
    mix = 0.5 * np.full(n, 1.0 / n) + 0.5 * edge
    _add(mix)

    # Limit the pool size (too many seeds at very large n are costly)
    max_cand = 20 if n <= 8192 else 12
    return seeds[:max_cand]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimizing the evaluation function.
    The routine respects the wall‑clock budget and returns the best sequence
    discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq: np.ndarray | None = None
    best_val: float = float("inf")

    # Length schedule – start small (cheap warm‑starts) and grow.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    # Helper to compute iteration budgets proportional to problem size.
    def _adam_iters(n):
        base = 80000        # larger than the reference implementation
        it = int(base * 1024 / n) + 3000
        return max(1200, min(it, 20000))

    def _sub_iters(n):
        base = 8000
        it = int(base * 1024 / n) + 2000
        return max(800, min(it, 8000))

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Build candidate pool
        # ------------------------------------------------------------------
        candidates = _generate_seeds(n, best_seq)

        # ------------------------------------------------------------------
        # Optimise each candidate
        # ------------------------------------------------------------------
        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            try:
                # Phase 1 – Adam on smooth‑max surrogate
                a_adam, val_adam = _adam_optimize(
                    cand,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=_adam_iters(n),
                    lr=0.10,
                    init_lam=0.5,
                    lam_factor=1.015,
                )

                # Phase 2 – exact ℓ∞ sub‑gradient refinement
                a_ref, val_ref = _subgradient_refine(
                    a_adam,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=_sub_iters(n),
                    eta0=0.45,
                )

                # Evaluate (using the supplied evaluator – it performs its own
                # safety checks and clipping, but our vectors already respect the
                # constraints).
                val = evaluate_sequence(a_ref.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cand_id}] new global best {best_val:.6f}")

                # Occasionally keep the Adam‑only result if it is better.
                val_a = evaluate_sequence(a_adam.tolist())
                if val_a < best_val:
                    best_val = val_a
                    best_seq = a_adam.copy()
                    print(
                        f"[n={n} cand={cand_id}] new global best (Adam) {best_val:.6f}"
                    )
            except Exception as exc:
                print(f"[n={n} cand={cand_id}] exception: {exc}")

        # ------------------------------------------------------------------
        # Small random jitters around the current best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            jitter_rounds = 8
            for jitter_id in range(jitter_rounds):
                if time.time() > deadline:
                    break
                # perturb logits slightly
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, _ = _adam_optimize(
                    a0,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=int(_adam_iters(n) * 0.6),
                    lr=0.08,
                    init_lam=0.5,
                    lam_factor=1.015,
                )
                a_ref, _ = _subgradient_refine(
                    a_adam,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=int(_sub_iters(n) * 0.6),
                    eta0=0.45,
                )
                val = evaluate_sequence(a_ref.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = a_ref.copy()
                    print(
                        f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}"
                    )

    # ------------------------------------------------------------------
    # Final fine‑tuning on the overall best (if we have one)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # a last longer Adam + subgradient pass
        a_adam, _ = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=_adam_iters(n) * 2,
            lr=0.07,
            init_lam=0.5,
            lam_factor=1.010,
        )
        a_ref, _ = _subgradient_refine(
            a_adam,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=_sub_iters(n) * 2,
            eta0=0.42,
        )
        val = evaluate_sequence(a_ref.tolist())
        if val < best_val:
            best_val = val
            best_seq = a_ref.copy()
            print(f"[final fine‑tune] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished. Best value: {best_val:.6f} (length {best_seq.shape[0]}) ===")
    return best_seq.tolist()
