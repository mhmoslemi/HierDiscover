"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm Overview
------------------
We perform a multi‑scale, multi‑restart optimisation of the vector
``a`` (length ``n``) under the simplex constraint

    a_i ≥ 0 , Σ a_i = √(2 n).

The objective is the infinity‑norm of the auto‑convolution,
because with the above normalisation  ``evaluate_sequence(a) = max(a∗a)``.

The search consists of three phases:

1. **Multi‑scale seeding** – start from short lengths (≈300),
   optimise heavily, then interpolate the best vector to larger
   lengths (up to ≈3000).  This yields a smooth continuation that
   often beats a direct optimisation at the final size.

2. **Main optimiser** – for every length we run two complementary
   first‑order methods:
   * *Projected Gradient Descent with momentum* on the exact max‑convolution.
   * *Exponentiated‑gradient Adam* where the max is approximated by a
     temperature‑scaled soft‑max.  Both keep the best vector seen.

3. **Local stochastic refinement** – a simulated‑annealing hill‑climber
   that repeatedly transfers a small fraction of mass between two
   positions while preserving the simplex constraint.

All phases respect the global 1000 s budget and stop early if the
deadline is reached.  The final best vector is returned as a plain
Python list.

The implementation uses only NumPy (no external solvers) and runs
on a single thread, but the heavy FFT work is fast enough to explore
hundreds of thousands of iterations within the budget.
"""

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σ x_i = target_sum}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute the weighted reverse convolution needed for the sub‑gradient:

        rev_i = Σ_k w_k * a_{k-i}   (outside indices are ignored)

    The calculation uses FFTs (O(N log N)).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)

    # keep the linear part (length = len(w) + len(a) - 1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]

    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact maximum of the auto‑convolution a∗a using a single FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """
    Produce a diverse set of feasible starting vectors of length ``n``.
    All vectors satisfy Σ a_i = target_sum and a_i ≥ 0.
    """
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (
        a0
        - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
        + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    )
    init.append(black / black.sum() * target_sum)

    # 7) Kaiser windows with several betas
    for beta in (0.5, 2.0, 4.0, 6.0):
        kaiser = np.kaiser(n, beta)
        init.append(kaiser / kaiser.sum() * target_sum)

    # 8) cosine (half‑wave)
    cos = np.cos(np.pi * np.arange(n) / (n - 1))
    cos = np.maximum(cos, 0)
    init.append(cos / cos.sum() * target_sum)

    # 9) Gaussian shape (random sigma)
    sigma = rng.uniform(0.1, 0.4) * n
    gauss = np.exp(-0.5 * ((np.arange(n) - (n - 1) / 2) / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    # 10) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 11) sparse random spikes (few non‑zeros)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    # 12) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(
        scale=0.01 * target_sum / n, size=n
    )
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    return init

def _pgd_momentum_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent with classic momentum on the exact max‑convolution.
    Returns the best vector found and its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise on the simplex
    a = a0.copy()
    a = a * target_sum / a.sum()

    # FFT size – must accommodate linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # hyper‑parameters
    mu = 0.85                      # momentum coefficient
    step0 = 0.002 * target_sum     # base step size (~0.2 % of total mass)
    eps_active = 1e-12

    momentum = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- forward convolution & current max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD‑mom len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- active set for sub‑gradient -----
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ----- sub‑gradient via weighted reverse convolution -----
        rev = _weighted_rev(a, w, Nfft)
        g = 2.0 * rev

        # ----- momentum step -----
        lr = step0 / math.sqrt(it)
        momentum = mu * momentum - lr * g
        a_tmp = a + momentum

        # ----- projection onto simplex -----
        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- occasional jitter to escape plateaux -----
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient Adam optimiser using a temperature‑scaled soft‑max
    approximation of the max‑convolution.
    Returns the best vector observed and its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)                # logits on the simplex

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08                         # learning rate for logits

    # temperature schedule parameters
    tau_factor = 0.35                  # initial τ = factor·M
    decay_rate = max_iters / 7.0       # exponential decay divisor

    # FFT size
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- construct current a from logits (softmax) -----
        logits = y - np.max(y)                # shift for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()     # probability simplex
        a = target_sum * p

        # ----- forward convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- global best tracking -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            sum_ex = ex.sum()
            w = ex / sum_ex

        # ----- gradient wrt a (2·rev) -----
        rev = _weighted_rev(a, w, Nfft)
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on simplex) -----
        avg = np.dot(p, g_a)          # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to keep diversity
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex.
    Randomly moves a small fraction of mass from i → j.
    Accepts improvements; accepts worsening moves with a decaying
    simulated‑annealing probability.
    """
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 250_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-14:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.04)          # 1‑4 % transfer
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score
