    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape

"""
Propose a step‑function (non‑negative sequence) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²).

The implementation builds a diverse pool of starting points (including
arcsine‑type shapes, power‑law families, Gaussian bumps, etc.), runs a
smooth‑max Adam optimiser, refines it with an exponentiated‑gradient
sub‑gradient phase, applies a short smoothing phase and finally a cheap
stochastic hill‑climb.  It works on several lengths, up‑sampling the best
solution found so far, and respects the supplied time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape (Beta(½,½) on the grid)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - 0.5 - np.arange(n, dtype=np.float64)
    shape = 1.0 / np.sqrt(i * j)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # historic solution (if present)
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # random Dirichlet draws with different concentration patterns
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # explicit arcsine shape and its mixtures with uniform
    arco = _arcsine_shape(n)
    candidates.append(arco)
    for mix in (0.2, 0.5, 0.8):
        mix_cand = mix * arco + (1.0 - mix) * np.full(n, 1.0 / n)
        mix_cand /= mix_cand.sum()
        candidates.append(mix_cand)

    # power‑law families (including the arcsine exponent –0.5)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.where(prod > 0,
                         np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod)),
                         0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # symmetric exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # Hann window
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # half‑sine
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # centred Gaussians of various widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 6.0, n / 4.0):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best sequence seen and its exact (non‑smooth) value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.30
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → probability vector (soft‑max)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c        # sum(a)=1 ⇒ denominator drops out
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum != 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional random shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _run_smoothing(a0: np.ndarray, n: int, deadline: float, max_steps: int = 10):
    """
    Very short smoothing phase: repeatedly convolve with a small
    [0.25,0.5,0.25] kernel and keep the improvement if any.
    """
    best_a = a0.copy()
    conv_fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    for step in range(max_steps):
        if time.time() > deadline:
            break
        cand = np.convolve(best_a, kernel, mode='same')
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            print(f"[n={n} Smoothing step={step}] improved to {val:.6f}")
        else:
            break
    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000, no_improve_limit: int = 5_000):
    """
    Cheap stochastic local optimisation.  Moves a small mass ε from one index
    to another and accepts the move if the exact objective improves.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < no_improve_limit:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0:                     # guarded by eps bound, but stay safe
            trials += 1
            continue
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

            # ---- Optional smoothing phase ----
            a_smooth, val_smooth = _run_smoothing(best_seq, n, deadline, max_steps=6)
            if val_smooth < best_val:
                best_val, best_seq = val_smooth, a_smooth.copy()
                print(f"[n={n}] Smoothing improved global best to {val_smooth:.6f}")

            # ---- Short fine‑tune on the current global best for this length ----
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                             int(0.5 * max_adam))
                if val_fine < best_val:
                    best_val, best_seq = val_fine, a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to larger sizes with lightweight refinement
    larger_lengths = [16384, 32768, 65536]

    for n in larger_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Upsampling to length n={n} ===")
        if best_seq is None:
            continue
        up_init = _upsample(best_seq, n)
        # a tiny pool for the large‑size stage
        candidates = [up_init, _arcsine_shape(n)]

        max_adam = int(15000 * 2048 / n) + 4000
        max_subgrad = int(4000 * 2048 / n) + 800

        for cid, init_seq in enumerate(candidates):
            if time.time() > deadline:
                break

            a_adam, val_adam = _run_adam(init_seq, n, deadline, max_adam)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[n={n} cand={cid}] Adam up‑sample new global best {val_adam:.6f}")

            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad up‑sample new global best {val_sub:.6f}")

            a_smooth, val_smooth = _run_smoothing(best_seq, n, deadline, max_steps=4)
            if val_smooth < best_val:
                best_val, best_seq = val_smooth, a_smooth.copy()
                print(f"[n={n}] Upsample smoothing improved global best to {val_smooth:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap local hill‑climb (only if time permits)
    if best_seq is not None and time.time() < deadline:
        final_n = len(best_seq)
        print(f"\n=== Final hill‑climb (n={final_n}) ===")
        a_final, val_final = _local_hill_climb(best_seq, final_n, deadline,
                                               max_trials=300_000,
                                               no_improve_limit=10_000)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
