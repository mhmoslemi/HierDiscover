def build_convolution_matrix(f):
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)
        # iterate over j in range
        # for each j, i = k - j

            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data,(rows,cols)), shape=(2*n-1, n))
    return A

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm explores several lengths, generates a diverse pool of starts,
searches each start with a long exponentiated‑gradient (sub‑gradient) phase,
then applies a cheap linear‑programming “direction‑finding’’ refinement that
maximizes the total mass while keeping the convolution below the current
maximum.  Finally a tiny random local‑search cleans up the result.
All work respects the given time budget (default 1000 s) and always returns
the best sequence found.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the sub‑gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest but diverse set of starting vectors.
    """
    cand = []

    # warm start and simple rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 3, n // 2, 2 * n // 3):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge concentration
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # anti‑palindromic half‑support (no i and n‑1‑i both non‑zero)
    half = np.zeros(n, dtype=np.float64)
    half[: n // 2] = np.random.rand(n // 2)
    half[n // 2 :] = 0.0
    half /= half.sum()
    cand.append(half)

    # spike patterns (equally spaced)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) descent on the exact max.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _build_conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """
    Return the dense matrix A (shape (2*n‑1, n)) where
    (A @ g)[k] = (f * g)[k] (linear convolution, full mode).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_refine(f: np.ndarray, n: int, current_max: float, max_iter: int = 5) -> np.ndarray:
    """
    Perform a few LP‑based refinements.
    In each iteration solve:
        max   Σ_i g_i
        s.t.  (f * g)_k ≤ current_max   ∀k
              g_i ≥ 0
    Then mix f ← (1‑t)·f + t·g with t=0.2 and repeat.
    """
    t = 0.2
    g = f.copy()
    for it in range(max_iter):
        A = _build_conv_matrix_dense(f)                     # (2n‑1) × n
        b = np.full(2 * n - 1, current_max, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)                  # maximise sum(g)  → minimise –sum
        bounds = [(0, None)] * n

        # Solve with HiGHS (fast for moderate sizes)
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                               method='highs-ds', options={'disp': False})
        if not res.success:
            # If LP fails we abort further refinements.
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        g = res.x
        # Mix and renormalise (scale does not matter, keep sum=1)
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.random.dirichlet(np.ones(n))
            s = f.sum()
        f /= s

        # recompute the max convolution after mixing
        conv = _full_convolution(f, _next_pow2(2 * n - 1))
        new_max = conv.max()
        # stop early if we are not improving the bound
        if new_max >= current_max * (1.0 - 1e-6):
            break
        current_max = new_max
        print(f"[LP] refinement {it} → max={new_max:.6e}")

    return f


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (np.random.rand() * 0.4)  # move up to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local] move {it} improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # try several vector lengths; smaller lengths are fast and sometimes
    # give a better bound because the factor 2·n is smaller.
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # warm start if a previous solution exists
        warm = None
        if 'height_sequence_1' in globals():
            warm_raw = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if warm_raw.shape[0] > n:
                warm_raw = warm_raw[:n]
            elif warm_raw.shape[0] < n:
                warm_raw = np.pad(warm_raw, (0, n - warm_raw.shape[0]))
            warm = np.maximum(warm_raw, 0.0)
            if warm.sum() >= 0.01:
                warm = warm / warm.sum()
            else:
                warm = None

        candidates = _initial_candidates(n, warm)

        # ----------------------------------------------------------------
        # Run a fairly long exponentiated‑gradient (sub‑gradient) phase for
        # each candidate, but stop early if the global deadline is near.
        for cid, cand in enumerate(candidates):
            if time.time() >= deadline:
                break
            a0 = np.maximum(cand, 0.0)
            if a0.sum() < 0.01:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # Reserve a tiny slice of the remaining budget for the LP and local search
            sub_deadline = min(deadline - 5.0, time.time() + 30.0)

            a_opt, val_opt = _subgradient_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=180_000,
                eta=0.6,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[cand {cid}] Subgrad best {val_opt:.6f}")

        # ----------------------------------------------------------------
        # LP refinement on the current best for this length (if we still have time)
        if best_seq is not None and len(best_seq) == n and time.time() < deadline - 5:
            conv = _full_convolution(best_seq, conv_fft_len)
            cur_max = conv.max()
            print(f"[LP] start refinement, current max conv = {cur_max:.6e}")
            refined = _lp_refine(best_seq, n, cur_max, max_iter=6)
            conv_ref = _full_convolution(refined, conv_fft_len)
            val_ref = 2.0 * n * conv_ref.max() / (refined.sum() ** 2)
            if val_ref < best_val:
                best_val = val_ref
                best_seq = refined.copy()
                print(f"[LP] refined best {val_ref:.6f}")

        # ----------------------------------------------------------------
        # Very cheap random local moves
        if best_seq is not None and len(best_seq) == n and time.time() < deadline - 5:
            a_loc, val_loc = _local_random_search(
                best_seq,
                n,
                conv_fft_len,
                deadline,
                max_moves=25_000,
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Local] improved best {val_loc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – ensure a valid return
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512, dtype=np.float64)

    # Clip to evaluator's accepted range (0 … 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\nFinal best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
