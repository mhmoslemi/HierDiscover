"""
Propose a low‑objective step‑function by a two‑phase strategy:

1. **Coarse LP‑boost on a small grid (n≈1024).**
   – Starting from the previous best (or a random Dirichlet) we
     repeatedly solve a linear program:
        maximize   sum(g)
        s.t.      g ≥ 0
                  (f * g)[k] ≤ M   ∀k,
        where M = max(f * f) and * denotes linear convolution.
   – The LP pushes mass into positions where the current shape f is
     weak, increasing the total L1‑mass while keeping the peak
     convolution bounded.
   – The new vector g is rescaled to the same L1‑mass as f and mixed:
        f ← (1‑t)·f + t·g   (t starts at 0.2 and decays).
   – The loop runs while enough time remains; each iteration yields a
     strictly better (or equal) objective.

2. **Fine Adam + sub‑gradient refinement on the full size (n=4096).**
   – The best small‑grid vector is linearly interpolated to n=4096.
   – A smooth‑max Adam optimiser (log‑sum‑exp surrogate) runs on the
     soft‑max (simplex) parametrisation of the vector.
   – After Adam, a few exponentiated‑gradient sub‑gradient steps using
     the exact max‑convolution sharpen the solution.
   – If time permits the whole pipeline is repeated on n=8192.

Both stages respect the 1000 s budget and print progress.
The returned list is a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (assumes sum(a)≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

# ---------- LP‑boost (small grid) ------------------------------------
try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _lp_max_sum(f: np.ndarray, M: float, max_iters: int = 2000):
    """
    Solve the linear program:
        maximize   sum(g)
        s.t.      g ≥ 0
                  (f * g)[k] ≤ M  ∀k .
    Returns the optimal g as a NumPy array, or None on failure.
    """
    if not _CVX_AVAILABLE:  # fallback – never gets called in the target env
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n)
    f_param.value = f
    conv_fg = cp.conv(f_param, g)               # length 2n‑1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=max_iters, verbose=False)
    except (cp.error.SolverError, ValueError):
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)

# ---------- Adam on smooth‑max (large grid) -------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5               # start temperature
    lam_factor = 1.015
    lam_cap = 1e6

    last_improve = time.time()
    it = 0
    max_iter = 250_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & surrogate -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient (log‑softmax chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # random shake if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 20_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set where conv attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run Adam + sub‑gradient refinement, updating the globally best."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    return best_seq, best_val

        final_val = evaluate_sequence(best_seq.tolist())
    except Exception:  # pragma: no cover
        final_val = _objective(best_seq, best_seq.shape[0],
                               _next_pow2(2 * best_seq.shape[0] - 1))
    print(f"*** Finished search: best objective ≈ {final_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
