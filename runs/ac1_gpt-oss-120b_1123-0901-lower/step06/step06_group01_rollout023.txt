def _incremental_refine(a, deadline, rng):

    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_seq = a.copy()
    best_score = cur_score
    start = time.time()

    # Determine max steps based on time:
    max_steps = 10_000_000  # large but will break on time

        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # Choose fraction to move
        frac = rng.uniform(0.005, 0.05)   # small fraction
        delta = a[i] * frac
        if delta <= 0:
            continue

        # Save old slices for possible revert (if reject). But we adopt only accept if improvement else reject and revert.

        # Compute update: apply to conv using old a (pre-mod)
        # Save copies of impacted parts for revert
        slice_i = slice(i, i+n)

        # store old values
        conv_i_old = conv[slice_i].copy()
        conv_j_old = conv[slice_j].copy()
        # also store special indices
        idx_2i = 2*i
        idx_2j = 2*j
        idx_ij = i+j
        old_2i = conv[idx_2i]
        old_2j = conv[idx_2j]
        old_ij = conv[idx_ij]

        # apply linear updates
        conv[slice_i] -= 2.0 * delta * a
        conv[slice_j] += 2.0 * delta * a

        # apply quadratic updates
        conv[idx_2i] = old_2i + delta*delta
        conv[idx_2j] = old_2j + delta*delta
        conv[idx_ij] = old_ij - 2.0 * delta*delta

        # update a
        a[i] -= delta
        a[j] += delta

        # compute new max
        new_score = float(conv.max())

        # acceptance: deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[refine] new best {best_score:.6f} step {step}")

            a[j] -= delta

    # Compute conv
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_seq_local = best_seq.copy()
    best_score_local = best_score

    while time.time() < deadline:
        # Choose i, j
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.uniform(0.01, 0.04)
        delta = a[i] * frac
        # compute potential new conv max

        slice_i = slice(i, i+n)
        slice_j = slice(j, j+n)
        new_slice_i = conv[slice_i] - 2.0 * delta * a
        new_slice_j = conv[slice_j] + 2.0 * delta * a

        idx_2j = 2*j
        idx_ij = i + j

        new_val_2j = conv[idx_2j] + delta*delta
        new_val_ij = conv[idx_ij] - 2.0 * delta*delta

        max_candidate = new_slice_i.max()
        max_candidate = max(max_candidate, new_slice_j.max())

            conv[slice_j] = new_slice_j

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Choose a promising length (the length of the previously known best
   construction if available, otherwise ~1900).
2. Generate a diverse set of feasible starts (uniform, Dirichlet,
   exponential, windows, sparse spikes, plus the previous best).
3. Run two complementary gradient‑based optimisers:
   * Projected Gradient Descent (sub‑gradient of the exact max‑convolution).
   * Exponentiated‑gradient Adam (soft‑max smoothing of the max).
   Both optimisers keep the vector on the simplex Σa_i = √(2 n).
4. Starting from the globally best vector, perform a fast stochastic
   hill‑climbing refinement.  Each move transfers a small fraction of mass
   from a source index to a target index; the effect on the convolution
   is updated **incrementally** in O(n) time (no FFT per move).  Moves are
   accepted only if they strictly improve the current maximum; occasional
   jitter resets avoid stagnation.
5. Return the best sequence found within the 1000 s budget.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x_i ≥ 0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Return a short list of diverse feasible start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via a single FFT (returns length 2*len(a)-1)."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return conv.astype(np.float64)


def _max_convolution(a: np.ndarray) -> float:
    """Exact max of a*a (linear convolution)."""
    return float(_conv_fft(a).max())

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed (still normalised to sqrt(2n)).
    """
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())   # enough for linear conv of length 3n‑2
    step0 = 0.001 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and its max
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # active set – average over indices attaining the max
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # weighted reverse convolution (sub‑gradient)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev

        # gradient step + projection
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on logits with a soft‑max (temperature) smoothing of the max.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)                 # logits

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # soft‑max weights
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            sum_ex = ex.sum()
            w = ex / sum_ex

        # weighted reverse convolution
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g_a = 2.0 * rev
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _incremental_refine(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """
    Fast stochastic hill‑climbing using an *incremental* update of the
    convolution.  At each step a small fraction of mass is moved from a
    source index to a target index.  The convolution is updated in O(n)
    using simple slice arithmetic; only improving moves are kept.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    cur_max_idx = int(np.argmax(conv))

    best_seq = a.copy()
    best_score = cur_score

    step = 0
    while time.time() < deadline and step < 10_000_000:
        step += 1

        # pick source with some mass
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a modest fraction of the source mass
        frac = rng.uniform(0.01, 0.04)
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # is the current maximal index affected?
        affected = (
            (i <= cur_max_idx <= i + n - 1) or
            (j <= cur_max_idx <= j + n - 1) or
            (cur_max_idx == 2 * i) or
            (cur_max_idx == 2 * j) or
            (cur_max_idx == i + j)
        )
        base = -np.inf if affected else cur_score

        # compute candidate new maximum using only the parts that change
        slice_i = conv[i : i + n]
        slice_j = conv[j : j + n]

        new_max_i = np.max(slice_i - 2.0 * delta * a)
        new_max_j = np.max(slice_j + 2.0 * delta * a)

        new_2i = conv[2 * i] + delta * delta
        new_2j = conv[2 * j] + delta * delta
        new_ij = conv[i + j] - 2.0 * delta * delta

        cand_score = max(base, new_max_i, new_max_j, new_2i, new_2j, new_ij)

        # accept only improving moves
        if cand_score < cur_score - 1e-12:
            # apply the incremental update
            conv[i : i + n] = slice_i - 2.0 * delta * a
            conv[j : j + n] = slice_j + 2.0 * delta * a
            conv[2 * i] = new_2i
            conv[2 * j] = new_2j
            conv[i + j] = new_ij

            a[i] -= delta
            a[j] += delta

            cur_score = cand_score
            cur_max_idx = int(np.argmax(conv))

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[refine] new global best = {best_score:.6f} step {step}")

        # occasional jitter to escape plateaux
        if step % 50000 == 0:
            jitter = rng.normal(scale=0.001 * target_sum / n, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = float(conv.max())
            cur_max_idx = int(np.argmax(conv))

    return best_seq, best_score
