"""
propose_candidate
-----------------
A multi‑stage optimiser for the quadratic‑max objective

    F(a) = 2·n·max(a∗a) / (sum(a)²) ,

restricted to non‑negative sequences a with sum(a)>0.
The algorithm works as follows

1. **Fast parametrised sampling** (n≈2048):
   – uniform, sinusoid, edge‑exponential / power‑law,
   – random Dirichlet vectors,
   – random mixtures of a few Gaussian bumps.
   The 𝑘≈6 best samples (by a cheap FFT‑based objective) are kept.

2. **Annealed Adam on a smooth‑max surrogate**  
   – variables are stored as logits and passed through a soft‑max,
     guaranteeing positivity and the simplex constraint.
   – a log‑sum‑exp approximation of the max‐convolution is used;
     the temperature λ is slowly increased (annealing).
   – Adam with default hyper‑parameters adapts per‑coordinate steps.
   – occasional random “shake‑ups’’ avoid stagnation.
   The best vector seen during the run is recorded.

3. **Exact sub‑gradient refinement** (exponentiated‑gradient steps)
   using the true max‑convolution, sharpening the solution.

4. **Upscaling to a larger grid (n≈8192)** and a second round of
   the same optimiser (more iterations, tighter time budget).

5. **Final local search** – small random perturbations of the best
   logits are accepted if they improve the true objective.

All heavy work is performed with FFT‑based convolutions (O(n log n))
and every loop checks the global deadline (budget_s seconds).  The
function returns the best sequence found as a plain Python list,
clipped to the evaluator’s allowed range (≥0, ≤1000)."""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the true objective (assumes sum(a)≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)   # sum(a) ≡ 1 → denominator = 1

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Annealed Adam on the smooth‑max surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                 # start temperature
    lam_factor = 1.008        # smooth annealing
    lam_cap = 50000.0

    last_improve = time.time()
    it = 0
    max_iter = 1_000_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution -----
        conv = _full_convolution(a, conv_fft_len)
        conv = np.maximum(conv, 0.0)          # numerical safety
        max_c = conv.max()
        val = 2.0 * n * max_c                 # true objective (sum a = 1)

        # ----- record best -----
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # occasional progress report (keep it sparse)
            if it % 20000 == 0:
                print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        # w_i ∝ exp(lam * (conv_i - max_c))
        w_shift = conv - max_c
        w = np.exp(lam * w_shift)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fall back to uniform on the active set
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            w[active] = 1.0 / max(active.sum(), 1)
        else:
            w /= w_sum

        # ----- gradient wrt a (chain rule through convolution) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- gradient wrt logits (soft‑max Jacobian) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- random shake if stuck -----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient steps (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s

    eta = 0.6
    last_improve = time.time()
    max_iter = 30_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        conv = np.maximum(conv, 0.0)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            if it % 5000 == 0:
                print(f"[Subgrad n={n} it={it:05d}] new best {val:.6f}")

        # active set where conv attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:               # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,

    )
    best_seq, best_val = _subgradient_optimize(

    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Diverse family of hand‑crafted shapes."""
    candidates = []

    # warm‑start and simple shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet with various concentrations
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (mix with uniform)
    for alpha in np.logspace(-2, 1.0, num=16):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.2, 0.5, 0.8, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # two Gaussian bumps at the edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # half‑wave sinusoid (positive part only)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    candidates.append(s)

    # random mixtures of a few Gaussian bumps (more expressive)
    for _ in range(15):
        k = np.random.randint(2, 5)          # number of components
        weights = np.random.rand(k)
        weights /= weights.sum()
        mus = np.random.rand(k) * (n - 1)
        sigmas = np.random.rand(k) * (n / 8) + 1e-3
        shape = np.zeros(n)
        for w, mu, sigma in zip(weights, mus, sigmas):
            shape += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point used by the evaluation harness.
    Returns the best non‑negative step‑function sequence found
    within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Helper to keep a global record
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – small grid for cheap exploration
    n0 = 2048
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # warm start (if any) – shrink/expand to n0
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n0:
            warm_start = warm_start[:n0]
        elif warm_start.shape[0] < n0:
            warm_start = np.pad(warm_start, (0, n0 - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Generate a big pool of parametric candidates
    all_cands = _generate_parametric_candidates(n0, warm_start)

    # Fast evaluation + keep the top few
    cand_vals = []
    for idx, cand in enumerate(all_cands):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n0))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n0, conv_fft_len0)
        cand_vals.append((val, cand))

    # sort and keep the elite
    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on the small grid
    print(f"--- Deep optimisation on {len(top_candidates)} starts (n={n0}) ---")
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        best_seq, best_val = _deep_optimize(
            a0,
            n0,
            conv_fft_len0,
            corr_fft_len0,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[n={n0}] after start #{start_id+1} best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print(f"--- Optimising on larger grid (n={n_big}) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big
        best_seq = best_seq.copy()
    else:
        # fall back to the small grid for the final local search
        n = n0
        conv_fft_len = conv_fft_len0
        corr_fft_len = corr_fft_len0

    # ------------------------------------------------------------------
    # Stage 4 – final stochastic local search around the best vector
    if best_seq is not None:
        a_best = best_seq.copy()
        logits_best = np.log(a_best + 1e-15)
        iterations = 0
        last_improve = time.time()
        while time.time() < deadline and iterations < 20000:
            sigma = 0.02 * (0.99 ** (iterations // 2000))   # decay
            proposal = logits_best + np.random.randn(n) * sigma
            # simplex projection via soft‑max
            mx = proposal.max()
            exp_log = np.exp(proposal - mx)
            a_prop = exp_log / exp_log.sum()

            val_prop = _objective(a_prop, n, conv_fft_len)
            if val_prop < best_val:
                best_val = val_prop
                best_seq = a_prop.copy()
                logits_best = proposal.copy()
                last_improve = time.time()
                print(f"[LocalSearch it={iterations}] new best {val_prop:.6f}")

            iterations += 1
            if time.time() - last_improve > 8.0:
                break   # stop if stagnated

    # ------------------------------------------------------------------
    # Fallback to a safe uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip according to evaluator's constraints (≥0, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
