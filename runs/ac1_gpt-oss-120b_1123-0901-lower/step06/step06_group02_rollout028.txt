    (exponential, power‑law, Hann, Hamming, Kaiser, Gaussian, Tukey).
    Each candidate is normalised so that Σaᵢ = √(2 n) and its exact
    max‑convolution is recorded.

2.  **Soft‑max Adam minimisation** – runs of Adam on the log‑sum‑exp
    surrogate of the max‑convolution.  Hyper‑parameters (initial
    temperature, learning‑rate, etc.) are randomised for every run,
    allowing a thorough exploration of the local landscape.

3.  **Projected sub‑gradient descent** on the exact max‑convolution.
    The gradient at the current worst index is followed and the step is
    projected onto the simplex {a ≥ 0, Σa = √(2 n)}.

4.  **Simulated‑annealing hill‑climb** – tiny mass transfers between
    random positions are accepted with a Metropolis rule; this cleans
    up any remaining local irregularities.

All helper functions are defined at top level (no closures).  The
routine returns a plain Python list that can be fed directly to
`evaluate_sequence`.  The global variable `height_sequence_1` (if
present) is used as a warm start.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 1e-4,
    alpha0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed‑max surrogate.
    Returns (best_vector_on_simplex, exact_max_convolution_of_best).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to the simplex (softmax)
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stable soft‑max
        p = exp_shift / np.sum(exp_shift)           # Σ p = 1
        a = target_sum * p

        # exact max convolution (for logging)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n}  new best max={cur_max:.6f}")

        # smoothed max (log‑sum‑exp) weights
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a : 2·a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak

        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # modestly increase step after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            if np.random.rand() < 0.07:        # jitter
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.49:                     # early stop if very good
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    T0 = max(1e-3, best_val * 0.4)   # initial temperature
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.12   # up to 12 % transfer
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        if it % 2500 == 0:               # occasional re‑normalisation
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.49:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def _normalize_to_target(a: np.ndarray) -> np.ndarray:
    """Scale a to have Σa = sqrt(2·len(a)).  Returns a new array."""
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0:
        return a
    return a * (target / s)


def _evaluate_max(a: np.ndarray) -> float:
    """Exact max‐convolution value (the objective when Σa = √(2n))."""
    return float(_full_convolution_fft(a, a).max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function that minimises the
    evaluation score.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.05      # small safety margin

    best_seq = None          # numpy array
    best_val = np.inf

    # ------------------------------------------------------------------
    # Warm‑start from a globally supplied construction (if any)
    # ------------------------------------------------------------------
    hs = globals().get("height_sequence_1")
    if hs is not None:
        try:
            arr = np.asarray(hs, dtype=float)
            if arr.size > 0:
                arr = _normalize_to_target(arr)
                val = _evaluate_max(arr)
                best_seq = arr.copy()
                best_val = val
                print(f"[{time.time():.2f}] warm‑start score={best_val:.6f} n={len(arr)}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – diverse parametric random initialisations
    # ------------------------------------------------------------------
    families = ["exp", "pow", "hann", "hamming", "kaiser", "gaussian", "tukey"]
    while time.time() < deadline - 250.0:
        remaining = deadline - time.time()

        # pick length – longer lengths become more frequent later
        if remaining > 600:
            n = np.random.randint(3000, 11001)
        elif remaining > 300:
            n = np.random.randint(2500, 9500)
        else:
            n = np.random.randint(1800, 7000)

        idx = np.arange(n, dtype=float)

        shape = np.random.choice(families)
        if shape == "exp":
            tau = np.random.uniform(0.5, 3.0) * n
            a0 = np.exp(-idx / tau)
        elif shape == "pow":
            p = np.random.uniform(0.2, 2.0)
            a0 = (idx + 1.0) ** (-p)
        elif shape == "hann":
            a0 = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
        elif shape == "hamming":
            a0 = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
        elif shape == "kaiser":
            beta = np.random.uniform(0.0, 12.0)
            if n == 1:
                a0 = np.ones(1)
            else:
                t = (2.0 * idx) / (n - 1) - 1.0            # in [-1,1]
                a0 = np.i0(beta * np.sqrt(1.0 - t * t)) / np.i0(beta)
        elif shape == "gaussian":
            sigma = np.random.uniform(0.15, 0.4) * n
            centre = (n - 1) / 2.0
            a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        elif shape == "tukey":
            alpha = np.random.uniform(0.0, 1.0)
            a0 = np.ones_like(idx)
            if alpha > 0.0:
                taper_len = alpha * (n - 1) / 2.0
                # left taper
                left = idx < taper_len
                a0[left] = 0.5 * (1 + np.cos(np.pi *
                                (2 * idx[left] / (alpha * (n - 1)) - 1)))
                # right taper
                right = idx > (n - 1) - taper_len
                a0[right] = 0.5 * (1 + np.cos(np.pi *
                                 (2 * idx[right] / (alpha * (n - 1)) -
                                  2.0 / alpha + 1)))
        else:
            a0 = np.random.rand(n)

        # occasional small noise
        if np.random.rand() < 0.2:
            a0 += np.random.randn(n) * np.max(a0) * 0.01

        a0 = np.maximum(a0, 0.0)
        a0 = _normalize_to_target(a0)

        val = _evaluate_max(a0)
        if val < best_val:
            best_val = val
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_val:.6f}  n={n}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on a smoothed max‑convolution
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 150.0:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)

        # initialise logits from the current best (soft‑max parametrisation)
        x0 = np.log(best_seq + 1e-12)

        # randomise hyper‑parameters each run to diversify search
        init_tau = np.random.uniform(0.05, 0.35)
        min_tau = np.random.uniform(1e-6, 1e-4)
        alpha0 = np.random.uniform(0.01, 0.04)

        run_deadline = time.time() + min(25.0,
                                         deadline - time.time() - 30.0)

        seq_local, val_local = _optimise_adam_softmax(
            n, x0, target_sum, run_deadline,
            init_tau_factor=init_tau,
            min_tau_factor=min_tau,
            alpha0=alpha0,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] after Adam best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0    # keep some time for final hill‑climb
        seq_sub, val_sub = _targeted_subgradient(best_seq.copy(),
                                                 target_sum,
                                                 sub_deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(best_seq.copy(),
                                        target_sum,
                                        hill_deadline)
        if val_hill < best_val:
            best_val = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0, dtype=float)
        best_seq = _normalize_to_target(best_seq)
        best_val = _evaluate_max(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_val:.6f}")

    # Return as plain Python list
    return best_seq.tolist()
