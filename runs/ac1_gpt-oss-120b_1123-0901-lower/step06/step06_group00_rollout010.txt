def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def simulated_annealing(a0, n, conv_fft_len, deadline, max_iters):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val
    it = 0
    T0 = 0.01
    Tmin = 1e-6
    while time.time() < deadline and it < max_iters:
        # schedule: geometric cooling
        T = max(T0 * (0.999 ** it), Tmin)
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (0.5 * np.random.rand())
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        dE = val2 - cur_val
        if dE < 0 or np.random.rand() < np.exp(-dE / T):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
        it += 1
    return best_a, best_val

def lp_refine(a, n, deadline):
    from scipy.optimize import linprog
    # Compute max conv M
    conv_len = _next_pow2(2*n - 1)
    conv = _full_convolution(a, conv_len)
    M = conv.max()
    # Create sparse matrix A with diagonals f (a)
    from scipy.sparse import diags
    # offsets 0..n-1
    offsets = np.arange(n)
    A = diags(a, offsets, shape=(2*n - 1, n), format='csc')
    # Build constraints: A @ b <= M
    # Also b >= 0
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    # Use high precision method
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
        # Normalize to sum=1
        b = np.maximum(b,0)
        s = b.sum()
        if s > 0:
            b = b / s
        return b, 2.0 * n * _full_convolution(b, conv_len).max()
    else:
        return a, 2.0 * n * conv.max()
```

Propose a non‑negative step‑function (list of floats) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The implementation follows a multi‑phase strategy:

1️⃣ Warm‑start – a diverse pool of analytic shapes (U‑shape, spikes,
   Gaussians, sinusoid, Dirichlet) and, if available, the previous best
   solution stored in the global ``height_sequence_1`` variable.

2️⃣ Adam on a smooth‑max surrogate (log‑sum‑exp) – quickly moves a
   candidate into a promising region of the simplex.

3️⃣ Additive sub‑gradient descent with a diminishing stepsize and explicit
   Euclidean projection onto the simplex – refines the solution with
   exact max‑convolution gradients.

4️⃣ Simulated‑annealing local search – stochastic mass swaps that can
   escape shallow basins.

5️⃣ (Optional) a single linear‑programming polishing step that maximises
   the total mass under the current worst‑case convolution bound.

All phases respect the supplied time budget (default 1000 s) and the
best sequence seen is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the exact sub‑gradient.
    Returns a vector of length len(a) such that
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns the projected vector (non‑negative and sums to 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of diverse start vectors."""
    cand = []

    # warm‑start and a few rotated copies
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform‑like)
    cand.append(np.random.dirichlet(np.ones(n)))

    # Power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=float) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.3, 7):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Two‑Gaussian edge shapes (different spreads)
    i = np.arange(n, dtype=float)
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Very sparse “spike” vectors with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=float)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate
        S_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k]).
    Returns the best vector (∑a=1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # parameters on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # start temperature for the smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional shake to escape plateaus
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _additive_subgradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    alpha0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact max‑convolution.
    Uses an additive update followed by Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[AddGrad iter={it:05d}] new best {cur_val:.6f}")

        # active set (indices attaining the maximal convolution value)
        active = np.isclose(conv, max_c, atol=1e-13)
        if not active.any():
            # numerical safeguard – fall back to a uniform weight
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max‑conv (chain‑rule)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        alpha = alpha0 / math.sqrt(it + 1)

        # additive mirror‑descent step and projection onto simplex
        a = _project_simplex(a - alpha * grad)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    start_temp: float = 1e-2,
    decay: float = 0.9995,
    min_temp: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing with Metropolis acceptance.
    Moves a random fraction of mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        T = max(start_temp * (decay ** it), min_temp)

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        dE = val2 - cur_val

        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 5000 == 0:
                    print(f"[SA iter={it:05d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def _lp_polish(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    One‑shot linear‑programming refinement.
    Maximise total mass Σb_j subject to (a * b) ≤ M (entry‑wise),
    where M = max(a*a). Then renormalise b to sum = 1.
    """
    # compute current worst‑case bound
    conv = _full_convolution(a0, conv_fft_len)
    M = conv.max()

    # build sparse Toeplitz matrix A where A[k, j] = a0[k-j] (0 if out of range)
    # offsets are 0 … n‑1
    offsets = np.arange(n)
    A = diags(a0, offsets, shape=(2 * n - 1, n), format='csc')

    # linear programme: maximise 1ᵀb  ⇔ minimise -1ᵀb
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    b_ub = np.full(2 * n - 1, M)

    # we have very little time left – protect against timeout/oom
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(1, deadline - time.time())},
        )
    except Exception as e:  # pragma: no cover
        print("[LP] failed:", e)
        return a0, 2.0 * n * conv.max()

    if not res.success:
        print("[LP] optimisation did not converge")
        return a0, 2.0 * n * conv.max()

    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s < 1e-14:
        return a0, 2.0 * n * conv.max()
    b /= s

    # compute objective for polished vector
    conv_b = _full_convolution(b, conv_fft_len)
    val = 2.0 * n * conv_b.max()
    print(f"[LP] polished value {val:.6f}")
    return b, val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry‑point called by the evaluator.
    Performs a timed multi‑phase optimisation and returns the best
    non‑negative sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – the original state‑of‑the‑art used n = 4096
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] != n:
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() > 0.0:
            warm_arr = warm_arr / warm_arr.sum()
        else:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # 1) generate a modest set of starters and pick the best one quickly
    candidates = _initial_candidates(n, warm_arr) + _spike_candidates(n)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        a /= a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand={idx}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam on smooth‑max surrogate (short but aggressive)
    if best_seq is not None and time.time() < deadline - 5:
        phase_end = min(deadline - 5, time.time() + 180.0)  # up to 3 min
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase_end,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Additive sub‑gradient descent (longer, diminishing steps)
    if best_seq is not None and time.time() < deadline - 5:
        phase_end = deadline - 5
        a_add, val_add = _additive_subgradient(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase_end,
            max_iters=350_000,
            alpha0=0.9,
        )
        if val_add < best_val:
            best_val, best_seq = val_add, a_add.copy()
            print(f"[AddGrad] improved to {val_add:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing local search (exploratory)
    if best_seq is not None and time.time() < deadline - 5:
        phase_end = deadline - 5
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            phase_end,
            max_iters=300_000,
            start_temp=2e-2,
        )
        if val_sa < best_val:
            best_val, best_seq = val_sa, a_sa.copy()
            print(f"[SA] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 5) One‑shot linear‑programming polishing (if time permits)
    if best_seq is not None and time.time() < deadline - 5:
        phase_end = deadline - 5
        a_lp, val_lp = _lp_polish(best_seq, n, conv_fft_len, phase_end)
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[LP] improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # 6) (Optional) Upscale to a finer grid if we still have a comfortable margin
    if best_seq is not None and time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # brief Adam + additive refinement on the larger grid
        phase_end_big = deadline - 5
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            phase_end_big,
            max_iters=120_000,
        )
        a_big_add, val_big_add = _additive_subgradient(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            phase_end_big,
            max_iters=150_000,
        )
        if val_big_add < best_val:
            best_val, best_seq = val_big_add, a_big_add.copy()
            print(f"[Big] improved to {val_big_add:.6f}")

    # ------------------------------------------------------------------
    # Final safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
