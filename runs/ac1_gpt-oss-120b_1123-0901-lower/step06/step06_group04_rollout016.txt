        max_c = conv.max()
        # compute active mask
        active = np.isclose(conv, max_c, atol=1e-12)
        # weight w for max: average over active positions
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # compute gradient wrt a: 2 * cross_corr(w, a)

        rev_a = a[::-1]

The objective is homogeneous, so we fix sum(a)=1 and minimise 2·n·max(conv).
The search proceeds in three stages:

1.   A fast parameter sweep over simple families of symmetric shapes
     (power‑law / Beta‑like, edge Gaussians, and mixtures with the uniform
     distribution).  For each candidate the exact objective is obtained by a
     single FFT‑based convolution, which is cheap even for n≈3·10⁴.

2.   A random‑mix / jitter phase that explores convex combinations of the
     best shapes found so far and adds small log‑domain perturbations.

3.   An optional brief exponentiated‑gradient refinement (sub‑gradient on the
     ℓ∞‑norm) if time permits.

All helper functions are defined at top level (no closures) and the routine
respects the supplied wall‑clock budget (default 1000 s).  The returned
sequence is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(arr: np.ndarray, fft_len: int) -> float:
    """
    Linear convolution of ``arr`` with itself using an FFT of length ``fft_len``.
    Returns the maximum entry of the full convolution (length 2·len(arr)‑1).
    """
    fa = np.fft.rfft(arr, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * arr.shape[0] - 1]
    return float(conv.max())


def _objective(arr: np.ndarray, fft_len: int) -> float:
    """
    Exact evaluation of the objective for a non‑negative vector ``arr``.
    The vector is **not** assumed to be normalised; the formula is

        2 * n * max(conv(arr, arr)) / (sum(arr)**2)
    """
    n = arr.shape[0]
    s = float(arr.sum())
    if s <= 0.0:
        return float("inf")
    max_c = _conv_max(arr, fft_len)
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that minimises the supplied evaluation
    function.  The algorithm performs a systematic sweep over a few
    low‑dimensional families of symmetric shapes, followed by random
    mixtures and a short log‑domain jitter.  It stays within the given
    time budget and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the caller

    # ------------------------------------------------------------------
    # Fixed problem size – large enough to beat the current best bound.
    # ------------------------------------------------------------------
    n = 32768                      # 2ⁱ⁵, fits nicely into an FFT of length 2ⁱ⁶
    fft_len = _next_pow2(2 * n - 1)   # 65536 for the chosen n

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (objective = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_val = _objective(best_seq, fft_len)

    # ------------------------------------------------------------------
    # Pre‑compute the symmetric “product” array used by the power‑law family.
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)                 # 1 … n
    prod = idx * (n - idx + 1.0)                               # i·(n‑i+1)
    log_prod = np.log(prod)                                    # log‑space for stability

    # ------------------------------------------------------------------
    # 1) Pure power‑law (Beta‑like) shapes – exhaustive 1‑D grid.
    # ------------------------------------------------------------------
    for alpha in np.linspace(-3.5, -0.2, 200):   # 200 equally spaced points
        if time.time() > deadline:
            break
        shape = np.exp(alpha * log_prod)       # (i·(n‑i+1))**alpha  in log‑space
        s = shape.sum()
        if s == 0.0:
            continue
        shape /= s
        val = _objective(shape, fft_len)
        if val < best_val:
            best_val = val
            best_seq = shape.copy()
    print(f"[grid‑powerlaw] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Random mixtures of power‑law and the uniform distribution.
    # ------------------------------------------------------------------
    for _ in range(5000):
        if time.time() > deadline:
            break
        alpha = np.random.uniform(-3.5, -0.2)
        lam = np.random.rand()                     # weight for the power‑law part
        shape_pl = np.exp(alpha * log_prod)
        s = shape_pl.sum()
        if s == 0.0:
            continue
        shape_pl /= s
        shape = lam * shape_pl + (1.0 - lam) * uniform
        shape /= shape.sum()
        val = _objective(shape, fft_len)
        if val < best_val:
            best_val = val
            best_seq = shape.copy()
    print(f"[mix‑powerlaw] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Edge‑Gaussian families – systematic scan over the width parameter.
    # ------------------------------------------------------------------
    i_arr = np.arange(n, dtype=np.float64)
    for gamma in np.linspace(0.05, 2.0, 120):
        if time.time() > deadline:
            break
        shape = np.exp(-gamma * i_arr ** 2) + np.exp(-gamma * (n - 1 - i_arr) ** 2)
        shape /= shape.sum()
        val = _objective(shape, fft_len)
        if val < best_val:
            best_val = val
            best_seq = shape.copy()
    print(f"[grid‑gaussian] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Random mixtures of edge‑Gaussians with the uniform distribution.
    # ------------------------------------------------------------------
    for _ in range(2000):
        if time.time() > deadline:
            break
        gamma = np.random.uniform(0.05, 2.0)
        lam = np.random.rand()
        shape = np.exp(-gamma * i_arr ** 2) + np.exp(-gamma * (n - 1 - i_arr) ** 2)
        shape /= shape.sum()
        shape = lam * shape + (1.0 - lam) * uniform
        shape /= shape.sum()
        val = _objective(shape, fft_len)
        if val < best_val:
            best_val = val
            best_seq = shape.copy()
    print(f"[mix‑gaussian] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Local jitter around the current best (log‑domain perturbations).
    # ------------------------------------------------------------------
    jitter_rounds = 50
    for j in range(jitter_rounds):
        if time.time() > deadline:
            break
        # Work in the log‑domain to stay on the simplex after exponentiation.
        logs = np.log(np.maximum(best_seq, 1e-12))
        logs += np.random.normal(scale=0.02, size=n)
        cand = np.exp(logs - np.max(logs))
        cand /= cand.sum()
        val = _objective(cand, fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[jitter {j}] improved value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional short exponentiated‑gradient refinement (if time left).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:                         # give at least a few seconds
        max_iters = 1500
        a = best_seq.copy()
        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break
            # ------- convolution and active set -------
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-12)

            # ------- sub‑gradient of the ℓ∞‑norm -------
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
            # cross‑correlation w ★ a (linear in a) via FFT
            rev_a = a[::-1]
            fw = np.fft.rfft(w, fft_len)
            fr = np.fft.rfft(rev_a, fft_len)
            grad = 2.0 * np.fft.irfft(fw * fr, fft_len)[:n]

            # ------- exponentiated‑gradient update -------
            eta = 0.4 / np.sqrt(it)                # diminishing step size
            a = a * np.exp(-eta * grad)
            a_sum = a.sum()
            if a_sum <= 1e-12:                     # protect against collapse
                a = np.random.dirichlet(np.ones(n))
                a_sum = a.sum()
            a /= a_sum

            # ------- record improvement -------
            val = 2.0 * n * max_c          # because sum(a)=1 during the loop
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[refine it={it}] new best = {best_val:.6f}")

    print(f"[final] best objective = {best_val:.6f} (n={n})")
    return best_seq.tolist()
