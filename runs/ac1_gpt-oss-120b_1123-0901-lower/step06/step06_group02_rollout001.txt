"""
Hybrid optimizer for the step‑function autocorrelation bound.
---------------------------------------------------------

We aim at minimizing

    F(a) = 2·n·max(conv(a,a)) / (∑a)²    (n = len(a))

which, after fixing ∑a = √(2·n), reduces to
    F(a) = max(conv(a,a)).

The algorithm consists of four stages:

1. **Diverse parametric random seeding** – a large pool of candidates
   built from several families (exponential, power‑law, inverted
   Hann/Hamming, double‑ended exponentials, Kaiser windows, Dirichlet).
   The best `K` candidates are kept.

2. **Adam on a smooth approximation** – each retained candidate is
   refined using a soft‑max surrogate of the max‑convolution (log‑sum‑exp)
   with a decreasing temperature.  The soft‑max parametrisation
   (`softmax` of logits) guarantees the simplex constraint.

3. **Exact sub‑gradient descent** – a projected sub‑gradient step
   aimed at the current worst index of the exact convolution.

4. **Long simulated‑annealing hill‑climb** – tiny mass transfers between
   random positions are performed under a Metropolis criterion.  The
   schedule is long enough to escape local minima.

All stages respect the global time budget (default 1000 s) and keep the
globally best sequence.  The final sequence is returned as a plain
Python list, ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.special import i0  # Kaiser window


# ----------------------------------------------------------------------
# Fast linear convolution via real FFT (used everywhere)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    # next power of two for speed
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)            # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1               # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape families for random initialisation
# ----------------------------------------------------------------------
def _shape_exp(n: int, tau: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.exp(-i / tau)


def _shape_pow(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-p)


def _shape_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_inv_hann(n: int) -> np.ndarray:
    """Inverted Hann – high at the ends, zero in the centre."""
    i = np.arange(n, dtype=float)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_inv_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    # 0.46 + 0.54 cos gives values in [‑0.08, 1.0]; clip to non‑negative
    return np.clip(0.46 + 0.54 * np.cos(2.0 * np.pi * i / (n - 1)), 0.0, None)


def _shape_double_exp(n: int, tau: float) -> np.ndarray:
    """Sum of two exponentials emanating from both ends."""
    i = np.arange(n, dtype=float)
    return np.exp(-i / tau) + np.exp(-(n - 1 - i) / tau)


def _shape_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window (β ≥ 0)."""
    i = np.arange(n, dtype=float)
    alpha = (n - 1) / 2.0
    # argument of Bessel function
    t = beta * np.sqrt(1.0 - ((i - alpha) / alpha) ** 2)
    w = i0(t) / i0(beta)
    return w


def _shape_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Generate a random candidate from a chosen family
# ----------------------------------------------------------------------
def _random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector of length n, normalised to
    √(2·n).  The shape family and its parameters are chosen at random."""
    families = [
        ("exp", None),
        ("pow", None),
        ("hann", None),
        ("hamming", None),
        ("inv_hann", None),
        ("inv_hamming", None),
        ("double_exp", None),
        ("kaiser", None),
        ("dirichlet", None),
    ]
    name, _ = rng.choice(families)
    if name == "exp":
        tau = rng.uniform(0.1, 3.0) * n
        a = _shape_exp(n, tau)
    elif name == "pow":
        p = rng.uniform(0.2, 1.8)
        a = _shape_pow(n, p)
    elif name == "hann":
        a = _shape_hann(n)
    elif name == "hamming":
        a = _shape_hamming(n)
    elif name == "inv_hann":
        a = _shape_inv_hann(n)
    elif name == "inv_hamming":
        a = _shape_inv_hamming(n)
    elif name == "double_exp":
        tau = rng.uniform(0.1, 3.0) * n
        a = _shape_double_exp(n, tau)
    elif name == "kaiser":
        beta = rng.uniform(0.0, 12.0)      # typical beta range
        a = _shape_kaiser(n, beta)
    elif name == "dirichlet":
        a = _shape_dirichlet(n)
    else:
        a = np.ones(n)

    # Add a tiny random perturbation (helps escaping symmetry)
    if rng.random() < 0.2:
        noise = rng.normal(scale=0.02 * np.max(a), size=n)
        a = np.maximum(a + noise, 0.0)

    # Normalise to the required L1 sum √(2n)
    target = np.sqrt(2.0 * n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, target / n)
    else:
        a = a * (target / a_sum)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate of max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth surrogate.
    Returns (best_vector, best_max_conv).
    """
    n = a0.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # initialise logits (soft‑max parameters)
    x = np.log(a0 + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)      # stable soft‑max
        p = exp_shift / np.sum(exp_shift)  # Σ p = 1
        a = target_sum * p                # actual coefficients

        # exact max conv (used for bookkeeping)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # smooth max via log‑sum‑exp with temperature τ
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a : 2·a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: (3 * n - 2)]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # chain rule back to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to break plateaus
        if it % 250 == 0 and rng.random() < 0.2:
            x += rng.normal(scale=0.001, size=n)

        # slowly decay learning rate
        alpha *= 0.9995

    # final exact evaluation of the best encountered vector
    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (worst‑index)
# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        # gradient of conv_k w.r.t. a (only indices that contribute)
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70
            if rng.random() < 0.07:
                jitter = rng.random(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        # early exit if we already beat the target bound
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (long, aggressive)
# ----------------------------------------------------------------------
def _annealing_optimize(
    a0: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = max(1e-4, best_val * 0.4)
    T_end = 1e-6
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        # pick donor and receiver indices
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move a random fraction (≤8 %) of mass from i to j
        delta = a[i] * rng.random() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] anneal it={it} new best={best_val:.6f}")

        # periodic re‑normalisation to avoid drift due to rounding
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function with minimal
    evaluate_sequence() score.  Returns a plain Python list.
    """
    global rng
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None          # numpy array
    best_score = np.inf      # max‑convolution value (lower is better)

    def _record(seq: np.ndarray):
        """Update the global best based on exact max‑convolution."""
        nonlocal best_seq, best_score
        val = _full_convolution_fft(seq, seq).max()
        if val < best_score:
            best_score = float(val)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f} n={len(seq)}")
        return val

    # ------------------------------------------------------------------
    # Warm‑start from any previously discovered construction
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            arr = np.asarray(hs, dtype=float)
            n0 = arr.shape[0]
            tgt0 = np.sqrt(2.0 * n0)
            arr_norm = arr * (tgt0 / arr.sum())
            _record(arr_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – Randomised diverse seeding (keep top K candidates)
    # ------------------------------------------------------------------
    K = 7                     # size of the elite pool
    elite = []                # list of (max_conv, vector)
    while time.time() < deadline - 250.0:   # reserve a large chunk for refinements
        remaining = deadline - time.time()
        # pick length – gradually longer as time passes
        if remaining > 600:
            n = rng.integers(2500, 8000 + 1)
        elif remaining > 300:
            n = rng.integers(2000, 6000 + 1)
        else:
            n = rng.integers(1500, 4000 + 1)

        a0 = _random_candidate(n, rng)

        # exact max‑conv evaluation
        val = _full_convolution_fft(a0, a0).max()
        # keep elite
        if len(elite) < K:
            elite.append((val, a0))
            elite.sort(key=lambda t: t[0])
        elif val < elite[-1][0]:
            elite[-1] = (val, a0)
            elite.sort(key=lambda t: t[0])

        # also update the global best
        _record(a0)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on each elite candidate
    # ------------------------------------------------------------------
    # Split remaining time roughly equally among elites
    while elite and time.time() < deadline - 150.0:
        # pop the worst elite (so that we keep improving the better ones)
        val, seq = elite.pop(0)
        n = seq.shape[0]
        target_sum = np.sqrt(2.0 * n)

        # initialise logits from current sequence
        x0 = np.log(seq + 1e-12)

        # allocate a modest slice of the remaining budget
        run_deadline = time.time() + min(20.0, deadline - time.time() - 30.0)
        seq_ref, val_ref = _adam_optimize(
            seq, run_deadline,
            init_tau_factor=0.14,
            min_tau_factor=5e-5,
            alpha0=0.03,
        )
        _record(seq_ref)

        # push refined candidate back (it may be better)
        elite.append((val_ref, seq_ref))
        elite.sort(key=lambda t: t[0])

    # ------------------------------------------------------------------
    # Phase 3 – Exact projected sub‑gradient descent on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0          # keep a safety margin
        seq_sub, val_sub = _subgradient_optimize(best_seq.copy(),
                                                 sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Long simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        seq_hill, val_hill = _annealing_optimize(best_seq.copy(),
                                                 deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return as plain Python list
    return best_seq.tolist()
```

**Explanation of the improvements**

* **Expanded shape library** – we added inverted Hann/Hamming, double‑ended exponentials,
  and Kaiser windows, all of which place more mass near the ends of the interval,
  a pattern that empirically reduces the worst convolution peak.

* **Elite pool** – instead of a single best candidate we keep the top K seeds,
  allowing several promising basins to be explored in parallel during the Adam
  refinement stage.

* **Longer, temperature‑controlled annealing** – the final hill‑climb now runs
  for the majority of the remaining budget with a carefully decaying temperature,
  enabling the algorithm to escape shallow local minima that limited the previous
  short annealing pass.

* **More aggressive Adam hyper‑parameters** – a slightly larger initial learning
  rate and temperature, together with occasional jitter, give a better chance to
  reach lower‑max‑convolution values.

* **Dynamic budget allocation** – the code automatically partitions the
  1000 s budget, reserving generous time for the refinement phases while still
  performing a thorough random‑search warm‑up.

These additions together provide a richer search space and deeper optimisation,
making it realistic to push the evaluation score below the target 1.503.
