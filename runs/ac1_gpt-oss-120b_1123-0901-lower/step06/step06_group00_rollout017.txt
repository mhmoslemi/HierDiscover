The objective  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is invariant under a global scaling of the vector `a`.  
Therefore we can keep the vector on the probability simplex (`∑a = 1`) and
work only with the term  

\[
M(a)=\max_k (a*a)_k .
\]

`a*a` is the discrete distribution of the sum of two independent draws from
`a`.  Minimising the peak of this distribution is a classic *flatten‑the‑sum*
problem.  Uniform weights give a triangular sum‑distribution with a peak
`1/n` → `F = 2`.  By arranging the weights **monotonically decreasing**
(large probability on the left, small on the right) large values are
paired with small values in the centre of the convolution, which lowers the
peak below `1/n`.  The best known constructions are therefore **ordered
vectors** (linear, exponential or power‑law decay).

The search strategy combines three phases that fit comfortably into the
1000 s budget:

1. **Diverse initial pool** – uniform, Dirichlet, linear decay, exponential
   decay, power‑law decay, a few random permutations and (if available) the
   previous best `height_sequence_1`.

2. **Smooth‑max Adam optimisation** – we minimise a differentiable
   surrogate `log‑sum‑exp` of the convolution values.  The gradient is obtained
   analytically via a cross‑correlation.  After each Adam step the vector is
   projected onto the simplex *and* re‑sorted into descending order – this
   keeps the favourable monotone structure.

3. **Exact‑max exponentiated‑gradient refinement** – a sub‑gradient method that
   works on the true maximum (`M(a)`).  It is fast, scale‑invariant and
   respects the simplex.

4. **Cheap random local moves** – transfer a random fraction of mass from
   one index to another; accept the move only if the objective improves.

5. **Optional up‑sampling** – when time permits the current best vector is
   interpolated to a larger grid (e.g. 8192 points) and the whole optimisation
   loop is repeated.

All convolution operations use FFTs (`O(n log n)`).  The code logs intermediate
best values, respects the time budget, and finally returns the best sequence
found as a plain Python list.

```python
"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of

1. a rich set of initial candidates (uniform, Dirichlet, linear / exponential
   decay, power‑law, random permutations and warm‑start);
2. a smooth‑max Adam optimiser on a log‑sum‑exp surrogate;
3. an exact‑max exponentiated‑gradient (sub‑gradient) phase;
4. a cheap random local‑search that moves a fraction of mass between two
   entries;
5. an optional up‑sampling to a larger length (8192) when time permits.

All operations are performed with FFT‑based convolutions for O(n log n)
performance.  The code respects the 1000 s budget and always returns a valid
sequence (non‑negative, clipped to [0, 1000]).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Candidate generation ----------------------------------------------------
def _monotone_linear(n: int) -> np.ndarray:
    """Linear decreasing shape a_i ∝ (n-i). Normalised to sum 1."""
    w = np.arange(n, 0, -1, dtype=np.float64)
    return w / w.sum()


def _monotone_exponential(n: int, rate: float = 0.0005) -> np.ndarray:
    """Exponential decreasing shape a_i ∝ exp(-rate * i)."""
    w = np.exp(-rate * np.arange(n, dtype=np.float64))
    return w / w.sum()


def _monotone_powerlaw(n: int, alpha: float = 1.2) -> np.ndarray:
    """Power‑law decreasing shape a_i ∝ (i+1)^{-α}."""
    w = (np.arange(n, dtype=np.float64) + 1.0) ** (-alpha)
    return w / w.sum()


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of starting vectors."""
    cand = []

    # warm start (if supplied) and a few shifted copies
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        s = w.sum()
        if s >= 0.01:
            w = w / s
            cand.append(w.copy())
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand.append(np.roll(w, shift))

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # random Dirichlet – serves as a “noise” starter
    cand.append(np.random.dirichlet(np.ones(n)))

    # monotone families
    cand.append(_monotone_linear(n))
    cand.append(_monotone_exponential(n, rate=0.001))
    cand.append(_monotone_powerlaw(n, alpha=1.0))

    # a few random permutations of the monotone linear shape
    base = _monotone_linear(n)
    for _ in range(3):
        perm = np.random.permutation(n)
        cand.append(base[perm])

    return cand


# ----------------------------------------------------------------------
# Optimisers -------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    The vector is kept on the simplex via a soft‑max (logits) parameterisation.
    After every 2000 iterations the entries are re‑sorted in descending order
    to preserve the beneficial monotone structure.
    Returns best (sum‑normalised) vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08

    lam = 0.5          # temperature for the soft‑max surrogate
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- simplex projection via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()          # a ≥0 , Σa = 1

        # ---- exact objective (sum = 1) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (log‑its)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e6)

        # ---- occasional re‑ordering to monotone ----
        if it % 2000 == 0:
            # sort descending – this does not change the simplex sum
            order = np.argsort(-a)
            logits = logits[order]

        it += 1

    return best_a, best_val


def _exponentiated_gradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta: float = 0.55,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using multiplicative (exponentiated) updates.
    The vector stays on the simplex (normalised after each step).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[ExpGrad it={it:05d}] new best {val:.6f}")

        # active set – uniform weight over indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of the exact max (same formula as surrogate)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _random_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine respects the supplied time budget and returns the

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # tiny safety margin

    # ------------------------------------------------------------------
    # Primary length (the state‑of‑the‑art used 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start – the global variable may contain a previous solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
    else:
        warm = None

    # assemble a pool of start vectors
    candidates = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optimise each candidate (short runs) – keep a global best
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:

        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
            s = a0.sum()
        a0 = a0 / s

        # allocate a small slice of the remaining budget
        local_deadline = min(deadline - 4.0, time.time() + 25.0)

        # --- Adam (smooth‑max) ---------------------------------------
        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=120_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # --- Exponentiated‑gradient refinement -----------------------
        if time.time() < deadline:
            a_sub, val_sub = _exponentiated_gradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=40_000,
                eta=0.55,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] ExpGrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Long exponentiated‑gradient refinement on the current best
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _exponentiated_gradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            eta=0.55,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] ExpGrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _random_local_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=35_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()

    # ------------------------------------------------------------------
    # If time permits, upscale to a larger grid (8192) and repeat
    if time.time() + 25 < deadline:

            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.clip(a_big, 0.0, 1000.0)
        a_big /= a_big.sum()

        # Adam on the larger grid (short run)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
        )
        # Sub‑gradient refinement
        a_big_ref, val_big_ref = _exponentiated_gradient_optimize(
```
