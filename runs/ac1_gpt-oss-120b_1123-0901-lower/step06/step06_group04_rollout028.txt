"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The objective is homogeneous, so we keep the vector normalised to sum=1.
The optimiser works in two stages for each chosen length:

1.  Adam on a smooth‑max surrogate of the convolution maximum.
    The vector is represented by soft‑max logits, guaranteeing a simplex
    point.  A slowly increasing temperature (lam) turns the smooth‑max
    into the true ℓ∞‑norm.

2.  Exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient update.

A hierarchy of lengths is processed (32768 → 65536).  The best sequence
found at a smaller length is interpolated to the next length and used as
a warm‑start.  For each length we try a diverse set of seeds (uniform,
power‑law, exponential, edge‑Gaussians, random Dirichlet, linear ramps,
etc.) and also perform a few random jitter rounds around the current
global best.  All loops respect the wall‑clock deadline (default 1000 s).

The code uses only NumPy and the standard library.  No file or network
access is performed.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper functions (FFT utilities)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(len(a)-1)] .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2 * n * max(conv).
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 0.5                # start temperature
    lam_factor = 1.02

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- soft‑max (logits → probabilities) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and smooth‑max weights ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ---- gradient of smooth‑max w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- back‑propagation through soft‑max ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature annealing ----
        lam = min(lam * lam_factor, 1e8)

        # ---- exact objective (sum is 1 by construction) ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- occasional random perturbation if stagnating ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int,
) -> Tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient step.
    Returns (best_a, best_val) after at most max_iters iterations.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # fallback to a random Dirichlet
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                     # slowly decay step size

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine adapts the optimisation horizon to the supplied
    wall‑clock budget and returns the best sequence discovered as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.1

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Length hierarchy – we start from the best known length (32768) and
    # then double to 65536 to obtain a finer representation.
    # ------------------------------------------------------------------
    lengths = [32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # --------------------------------------------------------------
        # Build a heterogeneous pool of seeds
        # --------------------------------------------------------------
        candidates = []

        # 1) Interpolated previous best (if any)
        if best_seq is not None:
            old = np.asarray(best_seq, dtype=float)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, old)
            if interp.sum() > 0:
                interp = interp / interp.sum()
            else:
                interp = np.random.dirichlet(np.ones(n))
            candidates.append(interp)

        # 2) Uniform distribution
        candidates.append(np.full(n, 1.0 / n, dtype=float))

        # 3) Random Dirichlet draws
        for _ in range(2):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # 4) Edge‑concentrated power‑law shape
        idx = np.arange(1, n + 1, dtype=float)
        prod = idx * (n - idx + 1.0)
        shape = np.where(prod > 0, prod ** -1.5, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

        # 5) Monotone exponential decay
        r = 0.9995
        exp_decay = r ** np.arange(n, dtype=float)
        exp_decay = exp_decay / exp_decay.sum()
        candidates.append(exp_decay)

        # 6) Two Gaussian bumps at the edges
        i = np.arange(n, dtype=float)
        for gamma in (0.2, 0.5):
            g = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            g = g / g.sum()
            candidates.append(g)

        # 7) Linear ramp (increasing) and its reverse
        ramp = np.arange(1, n + 1, dtype=float)
        ramp = ramp / ramp.sum()
        candidates.append(ramp)

        rev_ramp = ramp[::-1]
        candidates.append(rev_ramp)

        # Trim the candidate list so that the total work stays within the budget
        max_cand = 12 if n <= 32768 else 8
        candidates = candidates[:max_cand]

        # --------------------------------------------------------------
        # Determine iteration budgets (more work for larger vectors)
        # --------------------------------------------------------------
        # For the largest size we allow ~80 k Adam steps; smaller sizes
        # receive proportionally more steps because they are cheap.
        base_adam = 80000
        max_adam_iters = max(5000, int(base_adam * 65536 / n))
        max_subgrad_iters = 3000

        # --------------------------------------------------------------
        # Optimise each candidate
        # --------------------------------------------------------------
        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            try:
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline, max_adam_iters
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline, max_subgrad_iters
                )

                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cand_id}] new global best {best_val:.6f}")
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={cand_id}] new global best (Adam) {best_val:.6f}")

            except Exception as exc:
                print(f"[n={n} cand={cand_id}] exception: {exc}")

        # --------------------------------------------------------------
        # Small random jitter around the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(4):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.03, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline, max_adam_iters // 2
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline, max_subgrad_iters // 2
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 32768
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=float)

    # expose the best sequence for possible warm‑starts by later runs
    globals()["height_sequence_1"] = best_seq.copy()

    print(f"\nBest objective value found: {best_val:.6f}")
    return best_seq.tolist()
