def propose_candidate(seed=42, budget_s=1000.0):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 0.3

    best_seq = None
    best_val = np.inf

    # Utility functions:
    def record(seq):
        nonlocal best_seq, best_val
        conv = _full_convolution_fft(seq, seq)
        val = conv.max()
        if val < best_val:
            best_val = float(val)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_val:.6f} n={len(seq)}")
        return val

    # Phase1: random initializations
    while time.time() < deadline - 200:
        n = np.random.randint(1500, 5000)
        # choose shape random
        shape_type = np.random.choice(['exp','pow','hann','hamming','dirichlet'])
        if shape_type == 'exp':
            tau = np.random.uniform(0.1,2.0)*n
            a0 = np.exp(-np.arange(n)/tau)
        elif shape_type =='pow':
            p = np.random.uniform(0.2,1.5)
            a0 = np.arange(1,n+1)**(-p)
        elif shape_type =='hann':
            i = np.arange(n)
            a0 = 0.5 - 0.5*np.cos(2*np.pi*i/(n-1))
        elif shape_type =='hamming':
            i = np.arange(n)
            a0 = 0.54 - 0.46*np.cos(2*np.pi*i/(n-1))
        else:
            a0 = np.random.dirichlet(np.ones(n))
        # add noise rarely
        if np.random.rand() < 0.2:
            a0 = np.maximum(a0 + np.random.randn(n)*0.01*np.max(a0), 0)
        # Normalize to meet target sum sqrt(2n)
        target = np.sqrt(2.0*n)
        a0 = a0 * (target / a0.sum())
        record(a0)
    
    # LP refinement
    try:
        import cvxpy as cp
        from scipy.linalg import toeplitz
        use_lp = True
    except Exception as e:
        print("CVXPY not available, skipping LP refinement")
        use_lp = False

    if use_lp and best_seq is not None:
        # Determine length based on best_seq
        while time.time() < deadline - 50:
            a = best_seq.copy()
            n = a.shape[0]
            target = np.sqrt(2.0*n)
            # compute M
            conv_a = _full_convolution_fft(a, a)
            M = conv_a.max()
            # Build Toeplitz matrix K
            first_col = np.concatenate([a, np.zeros(n-1)])
            first_row = np.concatenate([a[:1], np.zeros(n-1)])  # shape length n
            K = toeplitz(first_col, first_row)
            # Set up LP
            b = cp.Variable(n)
            constraints = [b >= 0, K @ b <= M]
            objective = cp.Maximize(cp.sum(b))
            prob = cp.Problem(objective, constraints)
            # Solve with fast solver (ECOS)
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-8, reltol=1e-8)
            except Exception:
                prob.solve(solver=cp.ECOS, warm_start=False, max_iters=5000)
            if b.value is None:
                # solver failed, break
                print("LP solve failed")
                break
            b_val = np.array(b.value).flatten()
            sum_b = b_val.sum()
            if sum_b <= 1e-12:
                # no improvement
                break
            # Rescale to target sum
            b_scaled = b_val * (target / sum_b)
            # Mixing parameter t: adaptively reduce as iterations progress
            t_mix = 0.1
            # Additional: compute new M after mixing; if M decreases, accept else reduce t
            a_new = (1 - t_mix) * a + t_mix * b_scaled
            # keep non-negativity, but already non-negative.
            # Normalize again to exact target sum
            a_new = a_new * (target / a_new.sum())
            new_val = record(a_new)
            if new_val < M - 1e-6:
                # improvement; update best_seq
                best_seq = a_new
                # No update to LP constraints? We'll recompute next iteration.
                print(f"[{time.time():.2f}] LP iteration improved to {new_val:.6f}")
                # continue
                continue
            else:
                # try reducing t
                for t_test in [0.05, 0.02, 0.01]:
                    a_new2 = (1 - t_test) * a + t_test * b_scaled
                    a_new2 = a_new2 * (target / a_new2.sum())
                    new_val2 = record(a_new2)
                    if new_val2 < M - 1e-8:
                        best_seq = a_new2
                        print(f"[{time.time():.2f}] LP iteration with smaller t improved to {new_val2:.6f}")
                        break
                else:
                    # no improvement; maybe break
                    print("[LP] No improvement, break")
                    break
    # After LP, optionally finalize with hillclimb like before
    if best_seq is not None and time.time() < deadline - 5:
        n = best_seq.shape[0]; target = np.sqrt(2.0*n)
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target, deadline)
        if val_hill < best_val:
            best_seq = seq_hill
            best_val = val_hill
            print(f"[{time.time():.2f}] final hillclimb best={best_val:.6f}")

    # Return best_seq as list
    return best_seq.tolist()

"""
Hybrid optimisation for the step‑function bound.

Algorithm
---------
1) Randomised parametric initialisations (exponential, power‑law,
   Hann/Hamming windows, Dirichlet draws) over a wide range of
   lengths.  The best candidate is kept.

2) Linear‑programming refinement (active‑set style):
   Given the current sequence a, we solve
        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b) ≤ M   (element‑wise),
   where M = max(a*a).  The solution b is scaled to the required
   L1‑norm and mixed with a (a ← (1‑t)a + t·b).  Repeating this step
   drives the peak of the self‑convolution down.

3) Targeted sub‑gradient descent on the exact max‑convolution
   (as in the previous runner) to polish the solution.

4) Final simulated‑annealing hill‑climb that swaps tiny amounts of
   mass between random positions with a Metropolis acceptance rule.

All phases respect the overall time budget (default 1000 s) and
share a global best solution.  The routine returns the best sequence
as a plain Python list.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negative values caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer at most 8 % of mass at i
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        # renormalise every few thousand steps (numerical drift)
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        # early stop if we have already crossed the target bound
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set style)
# ----------------------------------------------------------------------
def _lp_refine(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 15,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based updates.

    For the current a we compute M = max(a*a) and solve
        maximise Σ b
        s.t.   b ≥ 0,
               (a * b) ≤ M,
    where * is the linear (full) convolution.
    The solution b is scaled to the target L1‑norm and mixed back
    with a (a ← (1‑t)a + t·b).  The procedure stops early if no
    improvement is observed.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()

        # Build the convolution matrix K (shape (2n‑1, n))
        first_col = np.concatenate([a, np.zeros(n - 1)])
        first_row = np.concatenate([a[:1], np.zeros(n - 1)])
        K = toeplitz(first_col, first_row)

        # Linear program: maximise sum(b)  ⇔ minimise –sum(b)
        c = -np.ones(n)
        A_ub = K
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        # Solve with the HiGHS interior‑point solver (fast for medium‑size LPs)
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time() - 0.5)},
        )
        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed (status={res.status})")
            break

        b_raw = res.x
        sum_b = b_raw.sum()
        if sum_b < 1e-12:
            # LP gave essentially the zero vector -> nothing to improve
            break

        # Scale to the required L1‑norm
        b_scaled = b_raw * (target_sum / sum_b)

        # Mixing parameter – start relatively aggressive, then decay
        t = 0.4 if it == 1 else 0.2
        a_candidate = (1 - t) * a + t * b_scaled
        a_candidate = a_candidate * (target_sum / a_candidate.sum())  # renormalise

        cand_val = _full_convolution_fft(a_candidate, a_candidate).max()
        if cand_val < best_val - 1e-9:
            best_val = cand_val
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] LP it={it} improved max={best_val:.6f}")
        else:
            # No progress – reduce mixing weight and try one more time
            if t > 0.02:
                t = t * 0.5
                a_candidate = (1 - t) * a + t * b_scaled
                a_candidate = a_candidate * (target_sum / a_candidate.sum())
                cand_val = _full_convolution_fft(a_candidate, a_candidate).max()
                if cand_val < best_val - 1e-9:
                    best_val = cand_val
                    best_a = a_candidate.copy()
                    a = a_candidate
                    print(f"[{time.time():.2f}] LP it={it} (small t) improved max={best_val:.6f}")
                else:
                    # Stagnation – abort further LP steps
                    break
            else:
                break

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that always follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # worst peak index
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function with minimal evaluation

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf

    def _record(seq: np.ndarray):
        """Evaluate seq (using exact max‑convolution) and keep the global best."""
        nonlocal best_seq, best_score
        conv = _full_convolution_fft(seq, seq)
        score = conv.max()
        if score < best_score:
            best_score = float(score)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return score

    # ------------------------------------------------------------------
    # Phase 1 – diverse parametric random initialisations
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        remaining = deadline - time.time()
        if remaining > 600:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 4000 + 1)

        target_sum = np.sqrt(2.0 * n)

        shape = np.random.choice(["exp", "pow", "hann", "hamming", "dirichlet"])
        if shape == "exp":
            tau = np.random.uniform(0.1, 2.5) * n
            a0 = np.exp(-np.arange(n, dtype=float) / tau)
        elif shape == "pow":
            p = np.random.uniform(0.2, 1.7)
            a0 = np.arange(1, n + 1, dtype=float) ** (-p)
        elif shape == "hann":
            i = np.arange(n, dtype=float)
            a0 = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        elif shape == "hamming":
            i = np.arange(n, dtype=float)
            a0 = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
        else:
            a0 = np.random.dirichlet(np.ones(n))

        # occasional tiny noise

            noise = np.random.randn(n) * np.max(a0) * 0.02

        a0 = a0 * (target_sum / a0.sum())
        _record(a0)

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming active‑set refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:

        if val_lp < best_score:

        seq_sub, val_sub = _targeted_subgradient(best_seq.copy(),
                                                 target_sum,
                                                 sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]

                                        target_sum,

        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    return best_seq.tolist()
