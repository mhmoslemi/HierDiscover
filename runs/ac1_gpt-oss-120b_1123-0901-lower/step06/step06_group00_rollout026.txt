import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a, n):
    # a assumed non-negative, not necessarily normalized
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())

def _lp_update(f, M, n):
    # Build sparse Toeplitz matrix representing convolution with f
    # each diagonal offset i (i >=0) is filled with f[i]
    diag_data = [np.full(n - i, f[i], dtype=np.float64) for i in range(n)]
    A = diags(diag_data, offsets=range(n), shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g) -> minimize -sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        # fallback: return f unchanged
        return f
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return f
    return g / sum_g

def _lp_iterate(a0, n, conv_fft_len, deadline, max_iters=12):
    a = a0.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        g = _lp_update(a, M, n)
        # mixing factor decreases geometrically
        t = 0.5 * (0.9 ** it)
        a = (1 - t) * a + t * g
        # re‑normalise to simplex
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.random.dirichlet(np.ones(n))
    return a

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # print progress
            #print(f"[Adam iter={it}] new best {val:.6f}")
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        lam = min(lam * lam_factor, 1e6)
        it += 1
    return best_a, best_val

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000, eta=0.6):
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # if it % 2000 == 0:
            #    print(f"[Subgrad iter={it}] new best {val:.6f}")
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val

def _local_random_search(a0, n, conv_fft_len, deadline, max_moves=30_000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The algorithm proceeds in three stages:
    1. A fast low‑dimensional optimisation based on linear programming
       (maximising the total mass under a fixed convolution bound).  The LP
       runs on a compressed vector of length 2048, repeatedly mixing the
       current vector with the LP solution.
    2. The best low‑dimensional solution is interpolated to the full size
       (4096) and further refined with Adam on a smooth‑max surrogate followed
       by an exact‑max sub‑gradient phase.
    3. A cheap random‑move local search finishes the run.
    The routine respects the supplied time budget and always returns a
    feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float('inf')

    # --------------------------------------------------------------
    # Stage 0 – keep a reference to any previously discovered sequence.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 1e-6:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Stage 1 – low‑dimensional LP optimisation
    n_small = 2048
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    # simple schedule: run a few restarts until we run out of time.
    restart = 0
    while time.time() < deadline - 30 and restart < 12:
        # start from either warm vector (truncated) or a fresh Dirichlet draw
        if warm is not None and restart == 0:
            a0 = np.interp(np.linspace(0, 1, n_small, endpoint=False),
                           np.linspace(0, 1, warm.shape[0], endpoint=False),
                           warm)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.random.dirichlet(np.ones(n_small))
        per_restart_deadline = min(deadline - 5.0, time.time() + 20.0)
        a_opt = _lp_iterate(a0, n_small, conv_fft_len_small, per_restart_deadline, max_iters=15)

        # Upscale to the target length (4096) for fair comparison.
        n_big = 4096
        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, a_opt)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 1e-12:
            a_big = np.full(n_big, 1.0 / n_big)
        else:
            a_big = a_big / a_big.sum()

        val_big = evaluate_sequence(a_big.tolist())
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[restart {restart}] new best after LP up‑scale {val_big:.6f}")

        restart += 1

    # --------------------------------------------------------------
    # If no LP attempt succeeded (unlikely), fall back to a uniform start.
    if best_seq is None:
        n_big = 4096
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    # --------------------------------------------------------------
    # Stage 2 – fine‑grained optimisation on the full size.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Adam phase (smooth max)
    if time.time() < deadline - 15:
        local_deadline = min(deadline - 5.0, time.time() + 120.0)
        a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=300_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # Sub‑gradient refinement
    if time.time() < deadline - 10:
        sub_deadline = min(deadline - 5.0, time.time() + 180.0)
        a_sub, val_sub = _subgradient_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=400_000, eta=0.7)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – cheap random local moves
    if time.time() < deadline - 3:
        local_deadline = deadline
        a_loc, val_loc = _local_random_search(best_seq, n, conv_fft_len, local_deadline, max_moves=50_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Final safety – ensure the vector respects the evaluator's constraints.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Search for a non‑negative step function (a vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm works in three stages:

1️⃣  Low‑dimensional optimisation (n≈2048) by repeatedly solving a
    linear programme:
       maximise   Σ g_j
       subject to   g_j ≥ 0  and   f∗g ≤ M,
    where M = max(f∗f).  The solution g is mixed with the current f
    (geometrically decreasing mixing weight).  This quickly pushes the
    convolution peak down while keeping the ℓ₁‑norm = 1.

2️⃣  The best low‑dimensional vector is interpolated to the target length
    (4096) and refined using a smooth‑max surrogate (Adam) followed by an
    exact‑max sub‑gradient phase on the simplex.

3️⃣  A cheap random‑move hill‑climber finishes the run.

The routine respects the given time budget (default 1000 s) and always
returns a feasible sequence (non‑negative, values ≤ 1000).  Intermediate
progress is printed.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the Adam gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Linear‑programming based update (stage 1)
def _lp_update(f: np.ndarray, M: float, n: int) -> np.ndarray:
    """Solve  max Σ g  s.t. g≥0,  f∗g ≤ M .  Return the normalised solution."""
    # Build the sparse Toeplitz matrix that represents convolution with f.
    # Row k, column j contains f[k‑j] whenever 0≤k‑j<n.
    diag_data = [np.full(n - i, f[i], dtype=np.float64) for i in range(n)]
    A = diags(diag_data, offsets=range(n), shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g  ↔  minimise -Σ g
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        # fallback: keep the current vector unchanged
        return f

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return f
    return g / s


def _lp_iterate(a0: np.ndarray, n: int,
                conv_fft_len: int,
                deadline: float,
                max_iters: int = 12) -> np.ndarray:
    """Repeated LP‑mixing steps."""
    a = a0.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        g = _lp_update(a, M, n)

        # mixing weight shrinks geometrically
        t = 0.5 * (0.9 ** it)
        a = (1.0 - t) * a + t * g

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.random.dirichlet(np.ones(n))
    return a


# ----------------------------------------------------------------------
# Smooth‑max (Adam) optimisation (stage 2)
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5          # start temperature for the smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → probability vector on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient phase (stage 2)
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates on the simplex using the exact max."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # if it % 2000 == 0:
            #     print(f"[Subgrad iter={it}] new best {val:.6f}")

        # active set – indices where the maximum is attained
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap random hill climbing (stage 3)
def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 20_000) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass from one index to another."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        it += 1
    return best_a, best_val
