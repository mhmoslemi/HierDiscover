def _generate_initial_candidates(n, prev_best):
    candidates = []
    # warm start
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))
    # known global variable
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        candidates.append(_upsample(np.asarray(hseq, dtype=np.float64), n))
    # Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))
    # Dirichlet with low conc (> many zeros)
    for conc in [0.1, 0.2, 0.5]:
        candidates.append(np.random.dirichlet(np.full(n, conc)))
    # Sparse random few spikes
    for k in [max(2, n//100), max(2, n//50)]:
        idx = np.random.choice(n, k, replace=False)
        vals = np.random.rand(k) + 0.1
        seq = np.zeros(n)
        seq[idx] = vals
        seq = seq / seq.sum()
        candidates.append(seq)
    # Windows
    candidates.append(np.hanning(n))
    candidates.append(np.hamming(n))
    candidates.append(np.blackman(n))
    candidates.append(np.bartlett(n))
    # Kaiser with different beta
    for beta in [2, 5, 8, 12]:
        candidates.append(np.kaiser(n, beta))
    # symmetrical exponential shapes (like earlier)
    # but we already have.
    # previously also have power law shapes etc.
    # We'll also include earlier shapes: power law, symmetric two-sided exponentials, Gaussian, half-sine...
    # We'll reuse code from previous.

    # final normalization, filter zero-sum
    out = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 1e-12:

def _run_simulated_annealing(a0, n, deadline, max_iters=200000):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_val(a, n)
    # compute val function
    def obj(seq):
        conv = _full_convolution(seq, conv_fft_len)
        return 2.0 * n * conv.max()
    conv_len = _next_pow2(2*n-1)
    # start temperature
    T0 = 1.0
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # choose epsilon
        eps = np.random.uniform(0, min(0.1, a[i]))
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        # compute objective
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        delta = val - best_val
        # acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - it/max_iters) + 1e-12)):
            a = cand
            if val < best_val:
                best_val = val
                best_a = cand.copy()
        # optionally cooling schedule
    return best_a, best_val

"""
search for a non‑negative step‑function that minimizes

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The optimisation is scale‑invariant, so we keep the vector on the simplex
(sum = 1).  The algorithm uses several complementary stages:

1.  A large, diverse pool of starting points (warm‑start, classic windows,
    power‑law shapes, random Dirichlet draws, sparse spikes, monotone random
    vectors, etc.).
2.  Adam on a smooth‑max surrogate of the objective (soft‑max on the
    convolution) to quickly reach a good basin.
3.  An exponentiated‑gradient sub‑gradient phase on the exact max‑convolution
    (active‑set gradient) to refine the solution.
4.  A short “fine‑tune’’ Adam run on the current global best.
5.  A cheap stochastic hill‑climb that moves a small amount of mass
    between two indices and accepts improving moves.  The move size is
    increased gradually if progress stalls.

The whole search respects the supplied time budget (default 1000 s) and
stores the best sequence found in the global variable ``height_sequence_1``,
so later runs can be warm‑started.

The code deliberately uses only NumPy (no external libraries) and is
compatible with the evaluation function that will be called separately.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradient of the smooth‑max.
    Returns vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 0. warm‑start from previous best (if any)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 1. historic global variable (if present)
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        candidates.append(_upsample(np.asarray(hseq, dtype=np.float64), n))

    # 2. classic windows (all are non‑negative)
    candidates.append(np.hanning(n))
    candidates.append(np.hamming(n))
    candidates.append(np.blackman(n))
    candidates.append(np.bartlett(n))

    # 3. Kaiser windows with several betas
    for beta in (2, 5, 8, 12):
        candidates.append(np.kaiser(n, beta))

    # 4. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8, 0.05):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 5. monotone power‑law shapes (decreasing)
    idx = np.arange(n, dtype=np.float64) + 1.0
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        shape = (n - idx + 1) ** p
        shape /= shape.sum()
        candidates.append(shape)

    # 6. random monotone vectors (sort random numbers)
    for _ in range(3):
        v = np.sort(np.random.rand(n))[::-1]      # decreasing
        v = np.maximum(v, 0.0)
        if v.sum() > 1e-12:
            candidates.append(v / v.sum())

    # 7. sparse random spikes (few non‑zeros)
    for k in (max(2, n // 100), max(2, n // 50)):
        idx_sp = np.random.choice(n, k, replace=False)
        vals = np.random.rand(k) + 0.1
        seq = np.zeros(n)
        seq[idx_sp] = vals
        seq /= seq.sum()
        candidates.append(seq)

    # 8. symmetric two‑sided exponentials (already used before)
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 9. half‑sine shape
    i = np.arange(n, dtype=np.float64)
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 10. centred Gaussian bumps (several widths)
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 11. power‑law on product (as in the reference code)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 5):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Normalise and discard degenerate entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """Adam optimiser on the smooth‑max surrogate of the objective."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)               # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                                 # sensible start learning rate
    lam = 0.30                                 # temperature for smooth‑max
    lam_factor = 1.03                         # increase temperature slowly

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → point on the simplex (soft‑max)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (sum(a)=1 ⇒ denominator disappears)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights on the convolution vector
        w = np.exp(lam * (conv - max_c))          # shift by max for stability
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (make the surrogate sharper)
        lam = min(lam * lam_factor, 1e4)

        # if we are stuck, inject a tiny random shake
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """Exponentiated‑gradient sub‑gradient phase on the exact max‑convolution."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                                   # initial step size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                         # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 8.0:
            # no improvement for a while → stop early
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 250_000):
    """
    Very cheap stochastic hill‑climb.
    Moves a small amount ε from one index to another and keeps the move
    if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 8_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.07, best_a[i]))
        if eps <= 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the bound 2·n·max(a*a)/(sum(a)²).  The function respects the time budget
    ``budget_s`` (seconds) and returns the best sequence discovered as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths to explore – larger n gives more freedom but costs more time.
    lengths = [2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budgets – scale gently with length
        max_adam = int(20000 * 2048 / n) + 8000
        max_subgrad = int(6000 * 2048 / n) + 2000
        max_adam = max(max_adam, 3000)
        max_subgrad = max(max_subgrad, 800)

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ------- Adam phase (smooth‑max) -------
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # ------- Sub‑gradient phase -------
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

        # ----- short fine‑tune on the best of this length -----
        if best_seq is not None and len(best_seq) == n:
            a_fine, val_fine = _run_adam(best_seq, n, deadline, max_adam // 2)
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – return a uniform sequence if nothing was produced
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # expose the best sequence for possible warm‑starts in later runs
    globals()['height_sequence_1'] = best_seq.tolist()

    print(f"Best value achieved: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
