"""
Propose a non‑negative step‑function (vector) with a low value of

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm mixes several complementary ideas:

* a diverse pool of warm‑starts (previous best, random Dirichlet,
  edge‑concentrated power‑law, double‑Gaussian, sinusoidal, uniform,
  cosine‑window and sparse “spike” patterns);
* a long Adam phase on a smooth‑max surrogate (log‑sum‑exp) with a
  slowly increasing temperature so that the optimizer can first move
  freely and then focus on the true maximum;
* an exact‑max exponentiated‑gradient (mirror‑descent) phase with a
  decaying step‑size;
* cheap but effective “LP‑step”: solve a tiny linear program on the
  currently active convolution indices and move a little towards the
  LP solution (this corresponds exactly to the LP step described in the
  statement);
* a cheap random local‑search (swap a random fraction of mass between
  two coordinates);
* an optional up‑sampling to a larger grid (8192 points) if time permits.

All phases respect the global time budget (default 1000 s).  The
function finally returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) (sum(a) is assumed 1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of diverse start vectors."""
    cand = []

    # warm‑start (if any) and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (uniform on the simplex)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shp = np.where(prod > 0, prod ** alpha, 0.0)
        if shp.sum() > 0:
            shp /= shp.sum()
            cand.append(shp)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shp = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shp /= shp.sum()
        cand.append(shp)

    # positive half‑sine
    shp = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shp = np.maximum(shp, 0.0)
    shp /= shp.sum()
    cand.append(shp)

    # cosine (raised‑cosine) window – a classic low‑sidelobe shape
    shp = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    shp = np.maximum(shp, 0.0)
    shp /= shp.sum()
    cand.append(shp)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """
    A few “spike” vectors with k equally spaced non‑zero entries.
    Each non‑zero entry holds mass 1/k.
    """
    res = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        res.append(a)
    return res

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (sum ≈ 1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04

    # temperature (λ) starts low and grows slowly – this is the key
    lam = 0.2
    lam_factor = 1.006                # very gentle growth
    lam_max = 1e5

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # sum(a) == 1

        # exact objective (used only for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # convert to gradient w.r.t. logits (mirror‑descent on the simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_max)

        # occasional tiny perturbation to escape plateaus
        if it % 10_000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 1e-3

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size decays as η = η₀ / sqrt(t+1).
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= a_sum

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {cur_val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # re‑project onto the simplex (exact normalisation)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

    return best_a, best_val


def _lp_improve(a: np.ndarray,
                n: int,
                max_c: float,
                top_k: int = 12,
                mix: float = 0.18) -> np.ndarray | None:
    """
    Solve a tiny linear programme on the currently highest convolution
    indices and move a little towards the LP solution.
    Returns the new vector or None if the LP fails.
    """
    if top_k <= 0:
        return None

    # convolution of a with itself (sum(a) == 1)
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    # pick the indices with the largest values (the active set)
    idx = np.argpartition(conv, -top_k)[-top_k:]

    # Build sparse constraint matrix for (a * b)_i ≤ t   (i ∈ idx)
    rows = []
    cols = []
    data = []

    for row, i in enumerate(idx):
        # a_j contributes to (a * b)_i iff 0 ≤ i-j < n  → j ∈ [i-(n-1), i]
        lo = max(0, i - (n - 1))
        hi = min(n - 1, i)
        for j in range(lo, hi + 1):
            rows.append(row)
            cols.append(j)
            data.append(a[j])
        # coefficient of the slack variable t (last column) is -1
        rows.append(row)
        cols.append(n)          # t is the last variable
        data.append(-1.0)

    A_ub = csr_matrix((data, (rows, cols)), shape=(len(idx), n + 1))
    b_ub = np.zeros(len(idx))

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: b_j ≥ 0,   t ≥ 0
    bounds = [(0, None)] * n + [(0, None)]

    # minimise t  → objective vector has 1 on the last entry
    c = np.append(np.zeros(n), 1.0)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None

    b = res.x[:n]                 # optimal distribution (already Σb = 1)
    # move a little towards b
    new_a = (1.0 - mix) * a + mix * b
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: transfer a random fraction of mass from one
    index to another; keep the move if the objective improves.
    """
    a = a0.copy()
    best_val = _objective(a, n, conv_fft_len)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        cur_val = _objective(a2, n, conv_fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it}] improved to {cur_val:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns a valid
    sequence (as a Python list).  It combines Adam‑based smooth‑max
    optimisation, exact‑max exponentiated‑gradient, a tiny linear‑program
    improvement step and a cheap random local search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1) Primary length – quick exploration
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from a previously found solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # assemble a pool of diverse starters
    candidates = _initial_candidates(n, warm) + _spike_candidates(n)

    # --------------------------------------------------------------
    # 2) Optimise each candidate (short runs) – keep a global best
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # Reserve a little time for the final refinement later
        local_deadline = min(deadline - 5.0, time.time() + 25.0)

        # --- Adam on the smooth‑max surrogate -----------------------
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline,
            max_iters=200_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # --- Exact‑max sub‑gradient phase ---------------------------
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=60_000, eta0=0.6
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad best {val_sub:.6f}")

        # --- Small LP improvement (optional, cheap) ----------------
        if time.time() < deadline:
            conv = _full_convolution(best_seq, conv_fft_len)
            max_c = conv.max()
            a_lp = _lp_improve(best_seq, n, max_c, top_k=12, mix=0.15)
            if a_lp is not None:
                val_lp = _objective(a_lp, n, conv_fft_len)
                if val_lp < best_val:
                    best_val = val_lp
                    best_seq = a_lp.copy()
                    print(f"[cand {cid}] LP improve best {val_lp:.6f}")

    # --------------------------------------------------------------
    # 3) Long refinement on the currently best candidate
    if best_seq is not None and time.time() < deadline - 5:
        # longer sub‑gradient run
        a_ref, val_ref = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000, eta0=0.5
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

        # a few LP steps interleaved with sub‑gradient
        for _ in range(3):
            if time.time() >= deadline - 5:
                break
            conv = _full_convolution(best_seq, conv_fft_len)
            a_lp = _lp_improve(best_seq, n, conv.max(), top_k=15, mix=0.12)
            if a_lp is not None:
                best_seq = a_lp
                best_val = _objective(best_seq, n, conv_fft_len)
                print(f"[LP step] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=40_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # --------------------------------------------------------------
    # 5) Upsample to a larger grid if time permits (better scaling)
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best vector
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # Adam on the bigger grid (shorter run)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big, deadline,
            max_iters=180_000
        )
        # Sub‑gradient refinement
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, max_iters=180_000, eta0=0.5
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

        # One additional LP step on the big grid
        if time.time() < deadline - 5:
            conv_big = _full_convolution(best_seq, conv_fft_len_big)
            a_big_lp = _lp_improve(best_seq, n_big, conv_big.max(),
                                   top_k=20, mix=0.1)
            if a_big_lp is not None:
                val_lp_big = _objective(a_big_lp, n_big, conv_fft_len_big)
                if val_lp_big < best_val:
                    best_val = val_lp_big
                    best_seq = a_big_lp.copy()
                    print(f"[Big LP] improved best {val_lp_big:.6f}")

    # --------------------------------------------------------------
    # 6) Fallback – ensure we always return something valid
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
