def _local_search(a, n, conv_fft_len, deadline, best_seq, best_val):
    a_curr = a.copy()
    # compute conv max and val
    while time.time() < deadline:
        # propose random move
        i = np.random.randint(n)
        if a_curr[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j: continue
        # delta = fraction of a_i
        delta = a_curr[i] * (np.random.rand() * 0.1)  # up to 10% of a_i
        # move
        a_new = a_curr.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non-negative (i>=delta)
        # compute conv max via _full_convolution
        conv = _full_convolution(a_new, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a_new.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a_curr = a_new
            # optionally print
            print(f"[local search] new best {val:.6f}")
        # else optionally accept w/ probability decreasing? We'll just ignore.
    return best_seq, best_val

"""
Minimal‑max‑convolution search.

The goal is to minimise
        F(a) = 2·n·max(a*a) / (sum(a)^2)
for a non‑negative step‑function a of length n.
The algorithm consists of three stages:

1.  Generate a rich pool of parametric candidates (including many
    *asymmetric* shapes that were missing from the previous code).

2.  For the best few candidates run a two‑phase optimisation:
        – Adam on a smooth‑max (log‑sum‑exp) surrogate of the max‑convolution
          while keeping the vector on the simplex by a soft‑max.
        – Exponentiated‑gradient sub‑gradient descent on the true
          max‑convolution, now with a diminishing step size.

3.  A cheap stochastic hill‑climbing (local‑search) phase applied to
    the globally best vector in the remaining time budget.

The routine respects the 1000 s budget, prints progress and finally
returns the best sequence as a plain Python list.  All numbers are
clipped to the range accepted by ``evaluate_sequence`` (≥0, ≤1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum(a) ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,

    """
    Adam optimisation on the log‑sum‑exp surrogate of max(a*a).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # start in the log‑simplex space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                         # a little smaller than before
    lam = 0.3                         # initial temperature
    lam_factor = 1.012
    lam_cap = 1e7

    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # --- soft‑max projection onto the simplex ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- surrogate computation ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # log‑sum‑exp weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # <a,∇>
        grad_log = a * (grad_a - avg)         # ∇_logsoftmax

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # random shake if stagnant
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    Uses a diminishing step size η_t = η0 / √(t+1).
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()                         # keep on the simplex

    eta0 = 0.9
    last_improve = time.time()
    it = 0
    max_iter = 40_000

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:05d}] new best {val:.6f}")

        # active set where the max is attained (within tight tolerance)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing learning rate
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient step (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 7.0:
            break

        it += 1

    return best_seq, best_val


def _local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the true objective.
    Moves a small amount of mass between two random coordinates.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    it = 0
    max_iter = 200_000
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # take a small fraction from i and give to j
        delta = a[i] * (np.random.rand() * 0.05)   # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # quick objective (no extra clipping needed here)
        conv = _full_convolution(a_new, conv_fft_len)
        val = 2.0 * n * conv.max() / (a_new.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[Local n={n} it={it:06d}] new best {val:.6f}")

        it += 1
    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run Adam → sub‑gradient → local‑search; update global best."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    best_seq, best_val = _local_search(
        a0, n, conv_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse candidate pool.
    In addition to the symmetric families used previously we also add
    *asymmetric* families (single‑sided exponentials, power‑laws,
    logistic sigmoids, linear ramps, etc.).
    All candidates are normalised to unit L1 norm.
    """
    cand = []

    # ---- warm‑start and simple rotations ----
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm_start, shift))

    # ---- uniform ----
    cand.append(np.full(n, 1.0 / n))

    # ---- random Dirichlet with many concentrations ----
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # ---- asymmetric exponential decay ----
    for alpha in np.logspace(-3, 0, num=15):      # 0.001 … 1
        shape = np.exp(-alpha * i)
        shape /= shape.sum()
        cand.append(shape)

    # ---- asymmetric exponential growth (mirror of decay) ----
    for alpha in np.logspace(-3, 0, num=15):
        shape = np.exp(alpha * i)
        shape /= shape.sum()
        cand.append(shape)

    # ---- power‑law decay ----
    for p in np.linspace(0.2, 3.0, num=12):
        shape = (i + 1.0) ** (-p)
        shape /= shape.sum()
        cand.append(shape)

    # ---- power‑law increase ----
    for p in np.linspace(0.2, 3.0, num=12):
        shape = (i + 1.0) ** p
        shape /= shape.sum()
        cand.append(shape)

    # ---- linear ramp (increasing) ----
    shape = i + 1.0
    shape /= shape.sum()
    cand.append(shape)

    # ---- linear ramp (decreasing) ----
    shape = (n - i).astype(np.float64)
    shape /= shape.sum()
    cand.append(shape)

    # ---- logistic / sigmoid families ----
    for scale in (0.02, 0.05, 0.1, 0.25, 0.5, 1.0):
        for mu in (n * 0.25, n * 0.5, n * 0.75):
            shape = 1.0 / (1.0 + np.exp(scale * (i - mu)))
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            cand.append(shape)

    # ---- two‑Gaussian edges (symmetric, kept for completeness) ----
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # ---- sinusoidal half‑wave (positive part) ----
    shape = np.sin(np.pi * (i + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ----- work initially with n = 4096 (good trade‑off between quality and speed) -----
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----- warm start from previous best if available -----
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ----- Stage 1 – generate many plausible candidates and keep the most promising -----
    all_cands = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, c in enumerate(all_cands):
        if time.time() >= deadline:
            break
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 0.01:                     # guard against pathological samples
            c = np.random.dirichlet(np.ones(n))
            s = c.sum()
        c = c / s
        val = _objective(c, n, conv_fft_len)
        cand_vals.append((val, c))
        # keep the console a little quieter – uncomment for debugging
        # print(f"[candidate {idx}] fast val = {val:.6f}")

    cand_vals.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ----- Stage 2 – deep optimisation from each of the selected starts -----
    for sid, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation from start #{sid + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ----- Stage 3 – if time permits, move to a finer grid (n = 8192) -----
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the larger grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ----- fallback – uniform vector if nothing succeeded -----
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # final clip according to the evaluator's limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
