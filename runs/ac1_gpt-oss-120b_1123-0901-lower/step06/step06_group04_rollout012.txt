"""
Optimal step‑function search (2026 edition)

The routine builds a rich pool of *parametric* seeds (edge‑concentrated,
edge‑Gaussian, and mixed product‑Gaussian families).  All seeds are
evaluated with a fast FFT‑based convolution to keep the cheap‑screening
cost low.  The best few are refined with two complementary gradient
methods:

1.  Adam on a smooth‑max (log‑sum‑exp) surrogate of the ℓ∞‑norm.
    The parameters are the logits of a soft‑max, which guarantees a
    non‑negative vector on the simplex.

2.  Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient
    update (the same update used in the reference solution) but with
    a longer iteration budget and an adaptive step size.

After the refinement a handful of random “jitters’’ around the current
best are performed to escape possible local minima.  The algorithm
concentrates the bulk of the budget on the largest affordable length
(32768) – longer vectors give strictly smaller theoretical bounds – and
still finishes well within the 1000 s wall‑clock limit.

All heavy linear algebra is performed with FFTs (O(n log n)) so the
search can afford thousands of iterations on a 32 k‑dimensional
vector.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for the gradient.
    Result has length n where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective evaluation (fast version – used only inside the search)
# ----------------------------------------------------------------------


def _fast_objective(a: np.ndarray, conv_fft_len: int) -> float:
    """
    Compute 2·n·max(a*a) assuming sum(a)=1.
    The full convolution is performed via FFT.
    """
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2·n·max(conv) for the best
    iterate before the deadline.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0           # smooth‑max temperature
    lam_factor = 1.02   # geometric increase

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- simplex via soft‑max -------------------------------------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- convolution & smooth‑max weights ------------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0.0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # --- gradient of smooth‑max (w.r.t. a) ------------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- back‑prop through soft‑max -------------------------------------------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update on logits ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- temperature annealing ------------------------------------------------
        lam = min(lam * lam_factor, 1e8)

        # --- exact objective check ------------------------------------------------
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # --- stagnation handling --------------------------------------------------
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient update.
    Returns (best_a, best_val).
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        # sub‑gradient weight vector (uniform over active indices)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:               # safeguard against numerical collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # record improvement
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # slowly decay the step size
        eta *= 0.9995

        # stop if stuck too long
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _product_edge_candidate(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i·(n‑i+1))^α, symmetric, α < 0."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.power(prod, alpha, where=prod > 0)
    if shape.sum() == 0.0:
        return _uniform_candidate(n)
    return shape / shape.sum()


def _gaussian_edge_candidate(n: int, sigma_frac: float) -> np.ndarray:
    """Sum of two Gaussians placed at the ends."""
    i = np.arange(n, dtype=np.float64)
    sigma = max(sigma_frac, 1e-12) * n
    shape = np.exp(- (i ** 2) / (2.0 * sigma ** 2)) + np.exp(- ((n - 1 - i) ** 2) / (2.0 * sigma ** 2))
    if shape.sum() == 0.0:
        return _uniform_candidate(n)
    return shape / shape.sum()


def _mixed_product_gaussian_candidate(n: int, alpha: float, sigma_frac: float) -> np.ndarray:
    """Product‑edge shape multiplied by a centre‑depressing Gaussian."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = max(sigma_frac, 1e-12) * n
    # product edge part
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    edge = np.power(prod, alpha, where=prod > 0)
    # centred Gaussian decay
    gauss = np.exp(- ((i - center) ** 2) / (2.0 * sigma ** 2))
    shape = edge * gauss
    if shape.sum() == 0.0:
        return _uniform_candidate(n)
    return shape / shape.sum()

    return np.random.dirichlet(np.ones(n))

                # quick Adam + sub‑gradient refinement
                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len,
                    deadline, max_iters=3000
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len,
                    deadline, max_iters=2000
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[jitter {jitter_id}] NEW GLOBAL BEST = {best_val:.6f}")

        # End of this length – continue to next (if any)
        print(f"--- Finished length {n}: best value = {best_val:.6f} ---")

    # ------------------------------------------------------------------
    # Fallback safety net (should never happen, but kept for completeness)
    # ------------------------------------------------------------------
    if best_seq is None:
        print("Fallback to uniform vector (unexpected).")
        best_seq = _uniform_candidate(2048)

    # Store the final sequence in a global variable for optional reuse
    globals()["height_sequence_1"] = best_seq.copy()

    # Return as plain Python list as required by the external evaluator
    return best_seq.tolist()
