---------
1. **Diverse initial families** – exponential, power‑law, Hann,
   Hamming, Dirichlet draws and a simple “interleaved extremes”
   permutation that spreads large values over the whole support.
2. **Scale‑invariant normalisation** – every candidate is projected onto
   the simplex Σ aᵢ = √(2 n).  The evaluation then reduces to
   max(conv(a,a)).
3. **Adam on a smooth surrogate** – a soft‑max (log‑sum‑exp) replaces
   the hard max, enabling a fast gradient‑based refinement while
   preserving the simplex constraint.
4. **Multi‑active sub‑gradient descent** – instead of following a single
   worst index we descend along the (convex) hull of all active
   constraints (all indices where the convolution is within a tiny
   tolerance of the current maximum).  This yields stronger, more
   directed steps.
5. **Fast simulated‑annealing hill‑climb** – tiny mass transfers are
   accepted with a Metropolis rule to escape any remaining plateaus.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = rfft(x, n=size)
    fy = rfft(y, n=size)
    conv = irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    return float(_full_convolution_fft(a, a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) surrogate
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (on the simplex) and its exact max‑convolution."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to the simplex (a = target_sum * softmax(x))
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # stable soft‑max
        p = exp_shift / np.sum(exp_shift)      # Σ p = 1
        a = target_sum * p

        # exact max of the convolution (logging)
        cur_max = _max_convolution(a)
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n}  new best max={cur_max:.6f}")

        # smoothed max – log‑sum‑exp weights
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        conv = _full_convolution_fft(a, a)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a : 2·a⊛w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = rfft(a, n=size2)
        Fw = rfft(w_rev, n=size2)
        conv2 = irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # chain rule back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to leave plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # gradual decay of learning rate
        alpha *= 0.9995

    # final exact evaluation of the best vector
    final_val = _max_convolution(best_a)
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Multi‑active projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient_multi(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the convex hull of all active
    worst indices (those within a tiny tolerance of the current peak)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _max_convolution(a)
    # initial step – a fraction of the average mass per coordinate
    step = 0.15 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        # all indices that are (practically) active
        eps = 1e-12 * cur_max
        active = np.where(conv >= cur_max - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1, dtype=int)
            # gradient contribution (2·a_{k‑j})
            grad[js] += 2.0 * a[k - js]

        # take projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _max_convolution(cand)
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                     # modestly increase step after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70                     # shrink step on failure
            # occasional jitter
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between random positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution(a)

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer at most 8 % of the mass at i
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        # re‑normalise every few thousand steps
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper – extreme‑interleaving permutation (large values spread out)
# ----------------------------------------------------------------------
def _interleaved_extremes(seq: np.ndarray) -> np.ndarray:
    """Arrange the sorted sequence so that the largest and smallest
    values alternate (largest, smallest, second‑largest, second‑smallest,…)."""
    idx = np.argsort(-seq)               # descending order
    sorted_desc = seq[idx]               # largest → smallest
    left = 0
    right = sorted_desc.size - 1
    out = np.empty_like(seq)
    pos = 0
    while left <= right:
        out[pos] = sorted_desc[left]
        left += 1
        pos += 1
        if left <= right:
            out[pos] = sorted_desc[right]
            right -= 1
            pos += 1
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function with minimal
    evaluation score.  The routine respects the supplied time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the evaluator’s internal checks
    deadline = start_time + budget_s - 0.10

    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # tiny utility to record a candidate (uses the exact max‑conv.)
    # --------------------------------------------------------------
    def _record(seq: np.ndarray):
        nonlocal best_seq, best_val
        val = _max_convolution(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_val:.6f}  n={len(seq)}")
        return val

    # --------------------------------------------------------------
    # Warm‑start from any previously discovered construction
    # --------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            S0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (S0 / hs_arr.sum())
            _record(hs_norm)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random parametric families + extreme interleaving
    # --------------------------------------------------------------
    while time.time() < deadline - 250.0:            # reserve ~250 s for later phases
        remaining = deadline - time.time()
        # pick a length – longer sequences later in the budget
        if remaining > 600:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 4000 + 1)

        target_sum = np.sqrt(2.0 * n)

        # ------- generate a raw shape -------
        shape = np.random.choice(["exp", "pow", "hann", "hamming", "dirichlet"])
        if shape == "exp":
            tau = np.random.uniform(0.05, 0.5) * n          # a bit shorter than before
            idx = np.arange(n, dtype=float)
            raw = np.exp(-idx / tau)
        elif shape == "pow":
            p = np.random.uniform(0.2, 1.5)
            idx = np.arange(1, n + 1, dtype=float)
            raw = idx ** (-p)
        elif shape == "hann":
            i = np.arange(n, dtype=float)
            raw = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        elif shape == "hamming":
            i = np.arange(n, dtype=float)
            raw = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
        else:   # dirichlet – uniform points on the simplex
            raw = np.random.dirichlet(np.ones(n))

        # occasional tiny Gaussian perturbation
        if np.random.rand() < 0.25:
            noise = np.random.randn(n) * np.max(raw) * 0.02
            raw = np.maximum(raw + noise, 0.0)

        # ------- two variants: original and interleaved -------
        for permute in (False, True):
            a0 = raw.copy()
            if permute:
                a0 = _interleaved_extremes(a0)
            a0 = a0 * (target_sum / a0.sum())      # project onto the simplex
            _record(a0)

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on a smoothed max‑convolution
    # --------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 150.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)

        # initialise logits from the current best (soft‑max parametrisation)
        x0 = np.log(best_seq + 1e-12)

        run_deadline = time.time() + min(30.0, deadline - time.time() - 30.0)
        seq_local, val_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=5e-5,
            alpha0=0.025,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (multi‑active)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0   # keep a little margin for the final hill‑climb
        seq_sub, val_sub = _targeted_subgradient_multi(
            best_seq.copy(),
            target_sum,
            sub_deadline,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(
            best_seq.copy(),
            target_sum,
            hill_deadline,
        )
        if val_hill < best_val:
            best_val = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_val = _max_convolution(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_val:.6f}")

    # Return a plain Python list for the external evaluator
    return best_seq.tolist()
