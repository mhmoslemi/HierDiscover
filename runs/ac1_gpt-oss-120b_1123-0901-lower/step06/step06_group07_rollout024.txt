def _lbfgs_optimize(a0, n, conv_fft_len, deadline, best_seq, best_val):
    import scipy.optimize

    eps = 1e-15
    logits = np.log(a0 + eps)

    lam = 1e6  # high temperature for smoothing (approx max)

    def fun_and_grad(logits):
        # compute softmax a
        max_log = np.max(logits)
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()
        # compute convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # compute smooth max
        # compute w = exp(lam * (conv - max_c))
        # avoid overflow
        w_exp = np.exp(lam * (conv - max_c))
        w_sum = w_exp.sum()
        smooth_max = max_c + np.log(w_sum) / lam
        val = 2.0 * n * smooth_max  # sum a=1
        # gradient:
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w = w_exp / w_sum
        grad_conv = 2.0 * _cross_correlation(w, a, conv_fft_len)  # same length as a
        # gradient w.r.t logits (softmax)
        grad_log = a * (grad_conv - np.dot(a, grad_conv))

        return val, grad_log

    # define callback for checking deadline
    def callback(logits):
        if time.time() >= deadline:
            raise StopIteration

    try:
        res = scipy.optimize.minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=logits,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxfun': 300, 'maxiter': 300, 'ftol': 1e-12},
            callback=callback,
        )
    except StopIteration:
        # time up; just return current best (maybe best_seq)
        return best_seq, best_val

    # after optimization, compute final a
    log = res.x
    # reconstruct a
    max_log = np.max(log)
    exp_log = np.exp(log - max_log)
    a_opt = exp_log / exp_log.sum()
    # evaluate actual objective (max conv)
    conv = _full_convolution(a_opt, conv_fft_len)
    max_c = conv.max()
    val = 2.0 * n * max_c / (a_opt.sum() ** 2)
    if val < best_val:
        best_val = val
        best_seq = a_opt.copy()
    return best_seq, best_val

def _adam_smooth_max(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val, max_iter=150000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 1.0
    lam_factor = 1.005
    lam_cap = 1e8
    last_improve = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute a (softmax)
        max_log = logits.max()
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        # random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
    return best_seq, best_val

def _lbfgs_optimize(a_start, n, conv_fft_len, deadline, best_seq, best_val):
    try:
        from scipy import optimize
    except Exception:
        return best_seq, best_val
    eps = 1e-15
    logits = np.log(a_start + eps)
    lam = 1e6  # smoothing temperature

    def eval_and_grad(logits):
        # compute a from logits
        max_log = logits.max()
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()
        # convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # smooth max approximation
        # w = exp(lam*(conv - max_c))
        # w_sum = w.sum()
        # smooth_max = max_c + np.log(w_sum) / lam
        # best to compute weights normalized
        w_exp = np.exp(lam * (conv - max_c))
        w_sum = w_exp.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum
        smooth_max = max_c + np.log(w_sum) / lam
        val = 2.0 * n * smooth_max  # sum a=1
        # gradient wrt a
        grad_conv = 2.0 * _cross_correlation(w, a, conv_fft_len)
        # gradient wrt logits (softmax)
        grad_log = a * (grad_conv - np.dot(a, grad_conv))
        return val, grad_log

    def fun(logits):
        val, _ = eval_and_grad(logits)
        return val

    def grad(logits):
        _, g = eval_and_grad(logits)
        return g

    # callback for deadline
    def callback(logits):
        if time.time() >= deadline:
            raise StopIteration

    try:
        res = optimize.minimize(fun, logits, jac=grad, method='L-BFGS-B',
                                options={'maxiter': 300, 'ftol': 1e-12},
                                callback=callback)
    except StopIteration:
        # time ran out; just return current best
        return best_seq, best_val

    # evaluate true objective at returned point
    log = res.x
    max_log = log.max()
    exp_log = np.exp(log - max_log)
    a_opt = exp_log / exp_log.sum()
    conv = _full_convolution(a_opt, conv_fft_len)
    max_c = conv.max()
    val = 2.0 * n * max_c / (a_opt.sum() ** 2)

    if val < best_val:
        best_val = val
        best_seq = a_opt.copy()
        print(f"[L-BFGS] new best {val:.6f}")

    return best_seq, best_val

"""
Propose a low‑evaluation step function.

The algorithm consists of three stages:

1. **Diverse parametric seeding** – a large pool of shapes (exponential
   edges, power‑law edges, mixtures with a uniform component, random
   Dirichlet draws, sinusoidal half‑waves, etc.) is created.  The 8
   best‐scoring seeds (according to the true objective) are kept.

2. **Deep optimisation of each seed** – for every selected start we run
   – a long Adam optimisation on a smooth‑max surrogate,
   – an optional L‑BFGS refinement (using the same smooth‑max but with a
     very high temperature, i.e. a tight approximation of the true max),
   – an exact sub‑gradient/exponentiated‑gradient phase on the true
     maximum of the convolution.
   The number of Adam / sub‑gradient iterations is larger for the most
   promising seeds.

3. **Scale‑up** – if time remains, the current best vector is interpolated
   to a finer grid (n = 8192) and the whole optimisation pipeline is run
   again with reduced iteration budgets.

All vectors are kept on the simplex by a soft‑max parametrisation, are
forced to be non‑negative, and finally clipped to the evaluator’s allowed
range [0, 1000].  The routine respects the 1000 s budget and prints the
progress and any improvements that are found.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Create a rich set of plausible shapes (all already normalised)."""
    candidates = []

    # ------------------------------------------------------------------
    # Warm‑start copies (if any)
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ------------------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Random Dirichlet draws with a wide range of concentrations
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # ------------------------------------------------------------------
    # Edge‑exponential families (optionally mixed with uniform)
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.2, num=16):      # 0.01 … ~16
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.5, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Symmetric two‑Gaussian edges
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Positive half‑wave sinusoid raised to a power
    sin_shape = np.sin(np.pi * (i + 0.5) / n)
    sin_shape = np.maximum(sin_shape, 0.0)
    sin_shape /= sin_shape.sum()
    for exp in np.linspace(0.5, 3.0, 12):
        shape = sin_shape ** exp
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Random convex combinations of the three base families
    for _ in range(20):
        w = np.random.dirichlet([1.0, 1.0, 1.0])
        alpha = 10 ** np.random.uniform(-2, 1)
        p = np.random.uniform(0.2, 5.0)
        edge_exp = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge_exp /= edge_exp.sum()
        edge_pow = (i + 1) ** (-p) + (n - i) ** (-p)
        edge_pow /= edge_pow.sum()
        shape = w[0] * edge_exp + w[1] * edge_pow + w[2] / n
        shape /= shape.sum()
        candidates.append(shape)

    return candidates

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max‑convolution.
    The temperature (lam) is increased slowly so the surrogate converges
    to the true max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5               # start temperature
    lam_factor = 1.005      # slower growth than the previous version
    lam_cap = 1e8

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & exact objective (used for early reporting) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))          # never overflow, max_c is subtracted
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain‑rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- random shake if stagnating -----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _lbfgs_optimize(
    a_start: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    One‑shot L‑BFGS optimisation on a high‑temperature smooth‑max
    surrogate (lam ≈ 1e6).  The optimisation works in the logits space
    (soft‑max parametrisation) and automatically respects non‑negativity
    and the simplex constraint.
    """
    try:
        from scipy import optimize  # type: ignore
    except Exception:
        # SciPy not available – skip this phase
        return best_seq, best_val

    eps = 1e-15
    logits = np.log(a_start + eps)
    lam = 1e6

    def eval_val_and_grad(logits_vec: np.ndarray):
        mx = logits_vec.max()
        exp_log = np.exp(logits_vec - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # smooth‑max via log‑sum‑exp
        w_exp = np.exp(lam * (conv - max_c))
        w_sum = w_exp.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum
        smooth_max = max_c + np.log(w_sum) / lam

        val = 2.0 * n * smooth_max  # sum(a)=1 by construction

        # gradient wrt a
        grad_conv = 2.0 * _cross_correlation(w, a, conv_fft_len)

        # gradient wrt logits (soft‑max chain rule)
        grad_log = a * (grad_conv - np.dot(a, grad_conv))
        return val, grad_log

    def fun(logits_vec):
        v, _ = eval_val_and_grad(logits_vec)
        return v

    def jac(logits_vec):
        _, g = eval_val_and_grad(logits_vec)
        return g

    def callback(logits_vec):
        if time.time() >= deadline:
            # abort the optimisation cleanly
            raise StopIteration

    try:
        res = optimize.minimize(
            fun,
            logits,
            jac=jac,
            method="L-BFGS-B",
            options={"maxiter": 300, "ftol": 1e-12},
            callback=callback,
        )
    except StopIteration:
        # Ran out of time – return the best we have so far
        return best_seq, best_val

    # Re‑evaluate the exact objective at the returned point
    log = res.x
    mx = log.max()
    exp_log = np.exp(log - mx)
    a_opt = exp_log / exp_log.sum()
    conv = _full_convolution(a_opt, conv_fft_len)
    max_c = conv.max()
    val = 2.0 * n * max_c / (a_opt.sum() ** 2)

    if val < best_val:
        best_val = val
        best_seq = a_opt.copy()
        print(f"[L‑BFGS] new best {val:.6f}")

    return best_seq, best_val

def _subgradient_refinement(
    a_start: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 50_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient steps using the true max‑convolution.
    Exponentiated‑gradient updates keep the vector non‑negative and on
    the simplex.
    """
    a = a_start.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a /= a_sum

    eta = 0.6
    last_improve = time.time()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop early if stagnated
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val

def _deep_optimize_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    max_iter_adam: int = 150_000,
    max_iter_sub: int = 50_000,
) -> tuple[np.ndarray, float]:
    """
    Run the three optimisation phases (Adam → L‑BFGS → exact sub‑gradient)
    with iteration budgets that can be tuned per seed.
    """
    # --- Adam phase ----------------------------------------------------
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iter=max_iter_adam,
    )

    # --- L‑BFGS refinement (only if enough time left) -----------------
    if time.time() < deadline - 10.0:
        best_seq, best_val = _lbfgs_optimize(
            best_seq if best_seq is not None else a0,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,

    # --- Exact sub‑gradient phase --------------------------------------
    best_seq, best_val = _subgradient_refinement(
        best_seq if best_seq is not None else a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        max_iter=max_iter_sub,
    )
    return best_seq, best_val

    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ----- generate a large pool of parametric candidates -----------------
    all_candidates = _generate_parametric_candidates(n, ws)

    # ----- quick evaluation & keep the most promising --------------------
    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 8
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ----- deep optimisation of the selected starts ----------------------
    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # give the best few a larger budget
        if idx < 2:
            adam_iters = 250_000
            sub_iters = 80_000
        else:
            adam_iters = 120_000
            sub_iters = 40_000
        print(f"--- Optimising start #{idx + 1}/{len(top_candidates)} "
              f"(Adam {adam_iters}, Subgrad {sub_iters}) ---")
        best_seq, best_val = _deep_optimize_custom(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            max_iter_adam=adam_iters,
            max_iter_sub=sub_iters,
        )

    # ----- if we still have time, try a finer grid (n = 8192) -----------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the best 4096‑vector
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize_custom(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
            max_iter_adam=80_000,
            max_iter_sub=20_000,
        )
        n = n_big  # remember the final size

    # ----- fallback to a safe uniform vector if nothing succeeded -------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # ----- final clipping to the evaluator's allowed range ---------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
