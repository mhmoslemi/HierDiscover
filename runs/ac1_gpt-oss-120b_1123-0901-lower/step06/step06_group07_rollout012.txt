def long_subgradient_opt(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):

    a = a0.copy(); ensure >0.

    # step parameters
    eta0 = 0.5; decay = 0.9999 per iteration; min_eta = 0.01

    last_improve = time.time()

    it = 0

    while time.time() < deadline:

        conv = _full_convolution(a, conv_fft_len)

        max_c = conv.max()

        val = 2.0 * n * max_c / (a.sum() ** 2)  # a sum fixed 1 approx

        if val < best_val - 1e-9:
            best_val = val; best_seq = a.copy(); last_improve = time.time(); print(...)

        # active set: conv close to max
        # use tolerance relative
        tol = 1e-12
        active = np.isclose(conv, max_c, atol=tol, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            # can't happen default maybe fall back uniform?
            w = np.full_like(conv, 1.0/conv.size)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated gradient update
        eta = max(min_eta, eta0 * (decay ** it))

        a = a * np.exp(-eta * grad)

        # ensure not too small
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))

        a /= a.sum()

        # if stuck, random shock
        if time.time() - last_improve > 5.0:
            # random perturbation:
            perturb = np.random.randn(n) * 0.01  # add noise in log space
            a = a * np.exp(perturb)
            a /= a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

"""
Search for a non‑negative step‑function (probability vector) minimizing

    F(a) = 2·n·max(a∗a) / (sum(a)²)

using a multi‑start exponentiated‑gradient (mirror‑descent) on the
sub‑gradient of the max‑convolution.  A small set of diverse parametric
candidates (uniform, Dirichlet, edge‑exponential, power‑law, etc.) is
generated, the best few are selected and each is refined for a long
period with a decaying step‑size.  Random perturbations are injected if
progress stalls.  If time permits the best solution is interpolated to a
larger grid (n=8192) and refined again.

The routine respects a 1000 s budget and returns the best sequence as a
plain Python list.  All numbers are clipped to the range accepted by
``evaluate_sequence``.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective for a (assumed normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse set of candidate vectors using several parametric
    families (uniform, Dirichlet, edge‑exponential, power‑law,
    Gaussian‑edges, sinusoid).  All candidates are normalised to sum≈1.
    """
    candidates = []

    # Warm‑start and simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet vectors with various concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential families, optionally mixed with uniform
    for alpha in np.logspace(-2, 1.0, num=16):   # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law families (i+1)^(-p) + (n-i)^(-p)
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def _long_subgradient_opt(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          best_seq: np.ndarray | None,
                          best_val: float):
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Runs until the deadline is reached.  Returns the best vector/value
    observed during the run.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    eta0 = 0.5                 # initial step size
    decay = 0.99995            # exponential decay per iteration
    min_eta = 0.001

    last_improve = time.time()
    it = 0
    max_iter = 10_000_000      # effectively unlimited – loop ends by time

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:07d}] new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of the (weighted) convolution
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = max(min_eta, eta0 * (decay ** it))

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # if stuck for a few seconds, inject a small random shake
        if time.time() - last_improve > 5.0:

            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

is generated, the best few are selected and each is refined for a long
period with a decaying step‑size.  If progress stalls a small random
perturbation is injected.  When time permits the best solution is
interpolated to a larger grid (n=8192) and refined again.

The routine respects the 1000 s time budget, prints progress, and
returns the best sequence as a plain Python list.  All values are clipped
to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective for a (assumed normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of candidate vectors using several families:
    uniform, Dirichlet, edge‑exponential, edge‑power‑law, Gaussian‑edges,
    half‑wave sinusoid.  All candidates are normalised to sum≈1.
    """
    candidates = []

    # warm‑start and simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet with a few concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential (plus possible uniform mixture)
    for alpha in np.logspace(-2, 1.0, num=16):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑power‑law (i+1)^(-p) + (n-i)^(-p)
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


def _long_subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Runs until the deadline is reached.  Returns the best vector/value
    observed during the run.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    eta0 = 0.5          # initial step size
    decay = 0.99995     # decay factor per iteration
    min_eta = 0.001

    last_improve = time.time()
    it = 0
    max_iter = 10_000_000  # effectively unlimited; loop stops by deadline

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:07d}] new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of the weighted convolution
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = max(min_eta, eta0 * (decay ** it))

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # if we have been idle for a few seconds, add a small random shake
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02   # log‑space jitter
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Primary grid size – 4096 is a good compromise between quality and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start: a previous best may be stored in a global called
    # ``height_sequence_1``.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a broad set of initial candidates
    candidates = _generate_parametric_candidates(n, warm_start)

    # Add a few pure Dirichlet samples with various concentrations.
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(25):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ------------------------------------------------------------------
    # Fast evaluation of every candidate (FFT‑based).  Keep the best few.
    cand_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # safest fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Optimising from start #{start_id + 1}/{len(top_candidates)} "
              f"(init val={init_val:.6f})")
        best_seq, best_val = _long_subgradient_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – if time permits, upscale to a larger grid (n = 8192) and
    # perform a brief refinement.
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _long_subgradient_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
