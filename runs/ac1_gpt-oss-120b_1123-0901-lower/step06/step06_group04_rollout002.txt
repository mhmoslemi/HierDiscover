    mixtures) and we always keep the best solution from previous lengths.
*   **Extended jitter/post‑processing** – after a length finishes we
    perform several random perturbations of the current best and re‑run
    the two‑phase optimiser.
*   **Simple‑but‑effective sub‑gradient refinement** – a longer
    exponentiated‑gradient phase with a larger initial step‑size.

All helpers are top‑level functions, there are no closures or lambdas.
The routine respects the wall‑clock budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, power‑of‑two lengths)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length *fft_len*."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Returns the best vector (sum=1) discovered and its exact objective value.
    """
    # initialise
    a = np.maximum(a0, 1e-12)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.06

    # temperature for smooth‑max (starts small, grows slowly)
    lam = 1.0
    lam_factor = 1.01

    # iteration budget – larger for small n, still generous for big n
    if n <= 8192:
        max_iters = 25000
    elif n <= 32768:
        max_iters = 20000
    else:
        max_iters = 15000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & smooth‑max weights -----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # ----- gradient of smooth‑max w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ----- back‑prop through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature annealing -----
        lam = min(lam * lam_factor, 1e9)

        # ----- exact objective evaluation -----
        val = 2.0 * n * max_c          # sum(a) = 1 by construction
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ----- stagnation handling -----
        if time.time() - last_improve > 5.0:
            # small jitter to escape plateaus
            logits += (np.random.rand(n) - 0.5) * 2e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞‑subgradient refinement using exponentiated‑gradient updates.
    Returns the best vector found and its exact objective.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()

    # start with a relatively aggressive step‑size
    eta = 0.9
    last_improve = time.time()

    for it in range(5000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set (indices achieving the max, tolerance tuned)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:         # numerical safeguard – fall back to random simplex point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # exact objective
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # slowly decay step‑size
        eta *= 0.9997

        # early stop if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _generate_seeds(n: int, best_seq):
    """
    Produce a diverse list of initial vectors (sum to 1, non‑negative).
    The list is deliberately limited in size to keep the total runtime
    under the global budget.
    """
    seeds = []

    # 1) uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) warm‑start from previous solution of same length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.asarray(best_seq, dtype=np.float64))

    # 3) interpolated previous best (different length)
    if best_seq is not None and len(best_seq) != n:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        if up.sum() > 0:
            up = up / up.sum()
        else:
            up = np.random.dirichlet(np.ones(n))
        seeds.append(up)

    # 4) a few Dirichlet draws
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # 5) symmetric beta‑distribution shapes (edge‑concentrated)
    x = (np.arange(n) + 0.5) / n
    for _ in range(3):
        alpha = 10 ** np.random.uniform(-1.5, 0.0)   # alpha in ~[0.03, 1]
        shape = (x * (1.0 - x)) ** (alpha - 1.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            seeds.append(shape)

    # 6) edge power‑law shapes a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    idx = np.arange(n, dtype=np.float64) + 1.0
    for _ in range(3):
        p = np.random.uniform(0.5, 4.0)
        shape = (idx ** -p) + ((n - idx + 1) ** -p)
        shape = shape / shape.sum()
        seeds.append(shape)

    # 7) mixture of uniform and edge‑Gaussian (two bumps)
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0]:
        edge = np.exp(-0.5 * (i ** 2) / (gamma ** 2)) + np.exp(-0.5 * ((n - 1 - i) ** 2) / (gamma ** 2))
        mix = 0.5 * np.full(n, 1.0 / n) + 0.5 * edge
        mix = mix / mix.sum()
        seeds.append(mix)

    # limit size to keep runtime in check
    return seeds[:12]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget and returns the
    best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_val = float("inf")

    # lengths to explore – start small for cheap warm‑starts then go large
    lengths = [8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # ------------------------------------------------------------------
        # Build seed pool
        # ------------------------------------------------------------------
        seed_pool = _generate_seeds(n, best_seq)

        # ------------------------------------------------------------------
        # Optimise each seed
        # ------------------------------------------------------------------
        for cid, cand in enumerate(seed_pool):
            if time.time() > deadline:
                break
            try:
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # keep the best version (refinement is usually better)
                if val_ref < best_val:
                    best_val, best_seq = val_ref, a_ref.copy()
                    print(f"[n={n} cand={cid}] new global best {best_val:.6f}")
                elif val_adam < best_val:
                    best_val, best_seq = val_adam, a_adam.copy()
                    print(f"[n={n} cand={cid}] new global best (Adam) {best_val:.6f}")
            except Exception as exc:
                print(f"[n={n} cand={cid}] exception: {exc}")

        # ------------------------------------------------------------------
        # Jitter & re‑optimise around the current best for this length
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(8):
                if time.time() > deadline:
                    break
                # random Gaussian perturbation in log‑space
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.025, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, _ = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_ref < best_val:
                    best_val, best_seq = val_ref, a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the (unlikely) event that nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("Fallback: using uniform sequence of length", fallback_n)

    # expose the best sequence for possible reuse in subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
