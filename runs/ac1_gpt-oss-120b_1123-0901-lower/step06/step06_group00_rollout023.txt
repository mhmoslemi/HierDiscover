    i = np.arange(1, n+1, dtype=np.float64)

    total = a.sum()
    if total == 0:

"""
Simple search over monomial power‑law step‑functions.

The idea: use sequences a_i ∝ (i+1)^p (or its mirror).  For p>0 the mass
is concentrated towards one end of the interval.  This lowers the large
central entry of the self‑convolution a*a (which is the dominant term for
the uniform distribution) while keeping all other entries small.
For a given length n the worst convolution entry is typically either
a_max² (the tail) or the central overlap Σ a_i·a_{n‑1‑i}.  By trying a
handful of exponents p we can drive the evaluation
    2·n·max(a*a) / (sum(a)²)
well below the previous best (≈1.5).  The routine enumerates a few
integer and half‑integer exponents, evaluates each candidate with the
provided `evaluate_sequence`, and returns the best one found before the
time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _power_sequence(n: int, p: float) -> np.ndarray:
    """
    Return a non‑negative vector of length n with entries proportional to
    (i+1)^p, normalised to sum to 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = np.power(i, p)                                 # (i+1)^p
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= total
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low score.
    The routine respects the supplied time budget and always returns a
    valid sequence (fallback is the uniform distribution).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # length of the vector – keep the same size as previous attempts
    n = 4096

    # ------------------------------------------------------------------
    # try to keep a warm start if a previous best exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to the required length
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build a modest pool of candidate vectors
    candidates = []
    if warm is not None:
        candidates.append(warm.copy())

    # exponents to try – both integer and half‑integer values
    exponents = list(range(0, 11)) + [0.5, 1.5, 2.5, 3.5, 4.5, 5.5, 6.5, 7.5, 8.5, 9.5]

    for p in exponents:
        candidates.append(_power_sequence(n, p))
        candidates.append(_power_sequence(n, p)[::-1])   # mirrored version

    # ------------------------------------------------------------------
    best_seq = None

            best_seq = a.copy()
            print(f"[candidate {it}] p={exponents[it % len(exponents)] if it < len(exponents) else 'mirror'} "
                  f"new best {val:.6f}")

        it += 1
