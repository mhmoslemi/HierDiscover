"""
Improved search for a non‑negative step‑function minimizing

        F(a) = 2·n·max(a∗a) / (∑a)² .

The algorithm follows three stages:

1.  A rich, *symmetric* candidate pool (uniform, power‑law edges,
    double‑exponential, Gaussian bumps, edge‑only blocks, sinusoid and many
    Dirichlet draws).  All candidates are forced to be mirror‑symmetric
    about the centre – the optimal solution is expected to be symmetric,
    which also cuts the optimisation dimensionality in half.

2.  For the most promising starts a two‑phase optimiser is run:
    *Adam* on the smooth‑max (log‑sum‑exp) surrogate with a slowly rising
    temperature, followed by an exact *sub‑gradient* mirror‑descent on the
    true max‑convolution.  Hyper‑parameters (learning‑rate, temperature
    schedule, stagnation thresholds) are tuned for a longer, more careful
    descent than the reference implementation.

3.  A short *local‑refinement* stage performs random perturbations in
    the log‑space of the simplex (exponential‑map) and keeps any
    improvement.  If time permits the final distribution is up‑sampled
    to a larger support (8192) and the whole two‑phase optimiser is run
    again, often yielding a noticeable drop in the objective.

All numbers are kept non‑negative and clipped to the evaluator’s bound
(≤1000).  The routine respects the supplied time budget (default 1000 s)
and returns the best sequence encountered as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase optimiser:
    – Adam on a smooth‑max surrogate (log‑sum‑exp)
    – exact sub‑gradient mirror‑descent on the true max
    The routine updates and returns the best sequence/value found.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # log‑softmax parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                # a bit larger than the reference

    # start temperature (smooth‑max) a little lower and raise it slowly
    lam = 0.2
    lam_factor = 1.008
    max_lam = 5e5

    last_improve = time.time()
    it = 0

    # ------------------------------------------------------------------
    # Adam phase (smooth‑max surrogate)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # soft‑max → simplex (exact projection)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase the temperature (approach the real max)
        if lam < max_lam:
            lam = min(lam * lam_factor, max_lam)

        # occasional random “shake” when we have been stagnant
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1
        if it >= 400_000:          # safety bound
            break

    # ------------------------------------------------------------------
    # Sub‑gradient phase (exact max)
    # ------------------------------------------------------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(120_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:06d}] new best {val:.6f}")

        # active‑set mask (all indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                       # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 12.0:   # stop if no recent improvement
            break

    return global_best_seq, global_best_val

def _generate_candidate_pool_symmetric(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse, *mirror‑symmetric* pool of start vectors.
    Every shape is symmetrised by averaging with its reverse.
    """
    pool = []

    # Warm‑start (already symmetric in most cases)
    if warm is not None:
        pool.append(warm.copy())

    # Uniform baseline (already symmetric)
    pool.append(np.full(n, 1.0 / n))

    # Power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)                # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        shape = (shape + shape[::-1]) / 2.0    # enforce symmetry
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = (shape + shape[::-1]) / 2.0
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape = (shape + shape[::-1]) / 2.0
        shape /= shape.sum()
        pool.append(shape)

    # Single half‑wave sinusoid (already non‑negative and symmetric)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape = (shape + shape[::-1]) / 2.0
    shape /= shape.sum()
    pool.append(shape)

    # Edge‑only uniform blocks (varying width)
    for frac in np.linspace(0.05, 0.45, 10):
        k = max(1, int(frac * n))
        shape = np.zeros(n)
        shape[:k] = 0.5 / k
        shape[-k:] = 0.5 / k
        shape /= shape.sum()
        pool.append(shape)

    # Random Dirichlet draws with a range of concentration parameters
    half = (n + 1) // 2
    for alpha in np.logspace(-2, 0, 12):
        half_vec = np.random.dirichlet(np.full(half, alpha))
        if n % 2 == 0:
            shape = np.concatenate([half_vec, half_vec[::-1]])
        else:
            shape = np.concatenate([half_vec, half_vec[-2::-1]])
        shape /= shape.sum()
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def _local_refinement(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    current_best: float,
) -> tuple[np.ndarray, float]:
    """
    Very short random‑perturbation stage.
    Logits are jittered and projected back to the simplex.
    Accept any improvement; occasional uphill moves keep us from getting stuck.
    """
    if a is None:
        return None, current_best

    best_seq = a.copy()
    best_val = current_best
    it = 0
    while time.time() < deadline - 0.2 and it < 2000:
        # work in log‑space (exponential map)
        eps = 1e-12
        logits = np.log(best_seq + eps)
        # small Gaussian noise
        logits += np.random.randn(n) * 0.02
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a_new = exp_log / exp_log.sum()

        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            print(f"[local refine] new best {val:.6f}")
        else:
            # occasional uphill acceptance to escape tiny plateaus
            if np.random.rand() < 0.01:
                best_seq = a_new
        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine respects the supplied time‑budget (default 1000 s) and returns
    the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Primary support size -------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from a previously found construction, if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:

            w = None
    else:
        w = None

    # ----------------------------------------------------------------------
    # Build a rich symmetric candidate pool
    # ----------------------------------------------------------------------
    candidates = _generate_candidate_pool_symmetric(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored = []  # (value, normalised vector)

    # ----------------------------------------------------------------------
    # Quick screening of all candidates
    # ----------------------------------------------------------------------
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

        # leave a small safety margin for deep optimisation
        if time.time() > deadline - 30.0:
            break

    # Keep the most promising starts (up to five)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:5]]

    # ----------------------------------------------------------------------
    # Deep optimisation passes on the best candidates
    # ----------------------------------------------------------------------
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ----------------------------------------------------------------------
    # Final local refinement
    # ----------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_seq, best_val = _local_refinement(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, best_val
        )

    # ----------------------------------------------------------------------
    # Optional up‑sampling to a larger grid (8192) if time permits
    # ----------------------------------------------------------------------
    if time.time() + 12.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the bigger grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[big] starting optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val
        )

    # ----------------------------------------------------------------------
    # Finalisation
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list
    return best_seq.tolist()
