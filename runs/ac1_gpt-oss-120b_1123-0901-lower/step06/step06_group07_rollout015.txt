"""
Propose a low‐objective step function.

Algorithm
---------
1. **Coarse parametric sampling** – generate many candidate vectors
   (uniform, Dirichlet, random edge‑exponential / power‑law / Gaussian
   families and random mixtures).  All candidates are normalised to
   unit ℓ1‑norm and cheaply evaluated with an FFT‑based convolution.
   The best *K* candidates are retained.

2. **Deep optimisation** – each retained start is refined by
   *a)* Adam on a smooth‑max (log‑sum‑exp) surrogate of the
   convolution max, followed by *b)* exponentiated‑gradient (EG) steps
   on the true max‑convolution (sub‑gradient).  The Adam temperature
   is increased slowly; both phases run for a large but time‑capped
   number of iterations.

3. **Local perturbations** – after the deep phase we perform cheap
   random mass‑transfer moves (maintaining the ℓ1‑norm) and keep any
   improvement.

4. **Optional upscale** – if time permits we linearly interpolate the
   current best vector to a finer grid (n=8192) and run a short
   optimisation there.

All work respects the 1000 s budget; intermediate progress and the best
objective found are printed.  The returned list contains non‑negative
values (clipped to the evaluator’s [0, 1000] range).
"""

# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a).  Returns a vector of length len(a)
    where (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]  (i=0…n‑1).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

# Random parametric families
def _rand_exponential_candidate(n: int) -> np.ndarray:
    """Edge‑exponential shape with random rate and mixture weight."""
    alpha = 10 ** np.random.uniform(-3, 1)          # 0.001 … 10
    w = np.random.rand()                           # mixture with uniform
    i = np.arange(n, dtype=np.float64)
    edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    edge /= edge.sum()
    cand = w * edge + (1.0 - w) / n
    return cand


def _rand_powerlaw_candidate(n: int) -> np.ndarray:
    """Edge‑power‑law shape with random exponent and mixture weight."""
    p = np.random.uniform(0.2, 5.0)
    w = np.random.rand()
    i = np.arange(n, dtype=np.float64)
    edge = (i + 1) ** (-p) + (n - i) ** (-p)
    edge /= edge.sum()
    cand = w * edge + (1.0 - w) / n
    return cand


def _rand_gaussian_candidate(n: int) -> np.ndarray:
    """Two‑Gaussian edge shape with random width and mixture weight."""
    sigma = np.random.uniform(0.02, 0.5)            # controls spread
    w = np.random.rand()
    i = np.arange(n, dtype=np.float64)
    edge = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + np.exp(-0.5 * ((n - 1 - i) / (sigma * n)) ** 2)
    edge /= edge.sum()
    cand = w * edge + (1.0 - w) / n
    return cand

# Optimisation primitives
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 300_000,
    lr: float = 0.05,
    lam_start: float = 0.5,
    lam_factor: float = 1.018,
    lam_cap: float = 1e6,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional shake‑up if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 30_000,
    eta: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    last_improve = time.time()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:05d}] new best {val:.6f}")

        # active set (all indices attaining the max, within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # EG update – stays non‑negative
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if no progress for a while
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    adam_iters: int = 300_000,
    subgrad_iters: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) then EG sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    # Adam phase
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iter=adam_iters,
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        max_iter=subgrad_iters,
    )
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry – searches for a low‑objective step function.
    Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # Work initially with n = 4096 (good compromise between quality and speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # -----------------------------------------------------------------
    # Warm‑start from a previously stored solution, if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # -----------------------------------------------------------------
    # --------- 1️⃣  GENERATE A LARGE POOL OF CANDIDATES -------------
    candidates = []

    # deterministic families (uniform, Dirichlet, etc.)
    candidates.append(np.full(n, 1.0 / n))                     # uniform
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    if warm_start is not None:
        candidates.append(warm_start.copy())
        # rolled versions of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # random draws from the three edge families
    N_RANDOM = 1800                                           # budget for cheap candidates
    for _ in range(N_RANDOM):
        r = np.random.rand()
        if r < 0.33:
            cand = _rand_exponential_candidate(n)
        elif r < 0.66:
            cand = _rand_powerlaw_candidate(n)
        else:
            cand = _rand_gaussian_candidate(n)
        candidates.append(cand)

    # -----------------------------------------------------------------
    # Fast evaluation of the whole pool, keep the K best
    fast_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # guard against degenerate draws
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        fast_vals.append((val, cand))

    fast_vals.sort(key=lambda x: x[0])
    TOP_K = 20
    top_candidates = [c for _, c in fast_vals[:TOP_K]]

    # -----------------------------------------------------------------
    # --------- 2️⃣  DEEP OPTIMISATION OF SELECTED STARTS ----------
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        # longer inner loops than the original script
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            adam_iters=300_000,
            subgrad_iters=30_000,
        )

    # -----------------------------------------------------------------
    # --------- 3️⃣  LOCAL PERTURBATION / HILL‑CLIMBING -------------
    while time.time() < deadline - 5.0:
        if best_seq is None:
            break
        a = best_seq.copy()
        # pick two indices and transfer a tiny amount of mass
        i, j = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-3     # max ~±5e-4
        # ensure non‑negativity
        delta = np.clip(delta, -a[i], a[j])
        a[i] += delta
        a[j] -= delta
        # (ℓ1‑norm unchanged)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, a
            print(f"[Perturb] improvement to {val:.6f}")

    # -----------------------------------------------------------------
    # --------- 4️⃣  OPTIONAL UPSCALE TO n=8192 (if time) ----------
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
            adam_iters=200_000,
            subgrad_iters=20_000,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big   # remember final size

    # -----------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
