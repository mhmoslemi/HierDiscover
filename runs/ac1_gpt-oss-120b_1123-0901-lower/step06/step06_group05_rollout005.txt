def _run_lbfgs(a0, n, deadline, max_iter, lam_start=1.0, lam_factor=1.5):
    # a0 is a probability vector sum=1.
    logits = np.log(a0 + 1e-15)
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    best_a = a0.copy()
    best_val = float('inf')
    lam = lam_start

    def loss_and_grad(logits_flat):
        # compute a
        mx = np.max(logits_flat)
        exp = np.exp(logits_flat - mx)
        a = exp / np.sum(exp)

        # conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = np.max(conv)
        # smoothed max
        # compute stable softmax weights:
        w_exp = np.exp(lam * (conv - max_c))
        w_sum = np.sum(w_exp)
        w = w_exp / w_sum if w_sum != 0 else np.full_like(conv, 1.0/conv.size)
        smooth_max = max_c + np.log(w_sum) / lam

        loss = 2.0 * n * smooth_max

        # grad_a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # grad wrt logits
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)
        return loss, grad_logits.astype(np.float64)

    # define callback to update best
    def callback(logits_flat):
        nonlocal best_a, best_val, lam
        mx = np.max(logits_flat)
        exp = np.exp(logits_flat - mx)
        a = exp / np.sum(exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = np.max(conv)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[n={n} LBFGS] new best exact {val:.6f}")
        # increase lam slowly
        lam = min(lam * lam_factor, 1e4)
        # stop if time over
        if time.time() > deadline:
            raise StopIteration

    try:
        from scipy.optimize import minimize
        options = {'maxiter': max_iter, 'disp': False}
        minimize(fun=lambda x: loss_and_grad(x),
                 x0=logits,
                 method='L-BFGS-B',
                 jac=True,
                 callback=callback,
                 options=options)
    except StopIteration:
        pass
    except Exception as e:
        print(f"LBFGS exception: {e}")

"""
Search for the best non‑negative step‑function (list of heights) that minimizes

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines several complementary optimisation stages:

1. **Diverse initial pool** – warm‑starts, Dirichlet draws, power‑law,
   exponential, sinusoidal, Gaussian‑like, and sparse “spike” patterns.
2. **Adam on a smooth‑max surrogate** – fast, robust descent on a
   temperature‑annealed log‑sum‑exp approximation of the maximum.
3. **L‑BFGS‑B on the same surrogate** – quasi‑Newton steps give high‑quality
   local minima with fewer iterations.
4. **Exponentiated‑gradient sub‑gradient phase** – optimises the true
   max‑convolution using the active set of binding constraints.
5. **Stochastic hill‑climbing** – cheap mass‑transfer moves to squeeze out
   any remaining improvement.
6. **Length‑wise schedule** – searches lengths 1024 → 32768,
   up‑sampling the current best to larger sizes.

All variables are kept on the probability simplex (soft‑max of logits),
so the sum‑constraint is satisfied automatically.
The routine respects a global time‑budget (default 1000 s) and prints
progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

def _next_pow2(x: int) -> int:
    """Return the next power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector by linear interpolation and renormalise."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _random_sparse(n: int, k: int) -> np.ndarray:
    """k‑spike random distribution (Dirichlet on the selected indices)."""
    probs = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=k, replace=False)
    probs[idx] = np.random.rand(k) + 1e-6
    probs /= probs.sum()
    return probs


def _generate_initial_candidates(n: int, prev_best):
    """Create a heterogeneous pool of starting points for length ``n``."""
    candidates = []

    # 1. warm‑start from previous best (if any)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic global variable (if present)
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        candidates.append(_upsample(np.asarray(hseq, dtype=np.float64), n))

    # 3. pure Dirichlet draws (different concentration parameters)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. sparse spikes (2,3,5 non‑zero entries)
    for k in (2, 3, 5):
        candidates.append(_random_sparse(n, k))

    # 5. power‑law edge‑concentrated shapes
    idx = np.arange(n) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 6. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 7. Hann window (raised cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 8. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 9. centred Gaussian‑like bumps (different σ)
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 10. inverse power‑law (smoothly decreasing)
    shape = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** 1.2)
    shape = np.maximum(shape, 0)
    shape /= shape.sum()
    candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """Adam optimisation on the smooth‑max surrogate."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # logits = log‑softmax parameters (unconstrained)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.045
    lam = 0.30
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ---- simplex mapping ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- exact objective (sum = 1) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={t}] new best {val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # ---- gradient wrt a ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- back‑prop through soft‑max (logits) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e4)

        # ---- occasional random shake if stuck ----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_lbfgs(a0: np.ndarray, n: int, deadline: float, max_iter: int,
               lam_start: float = 0.5, lam_factor: float = 1.5):
    """Quasi‑Newton (L‑BFGS‑B) optimisation on the smooth‑max surrogate."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)          # initialise on simplex
    best_a = a0.copy()
    best_val = float('inf')
    lam = lam_start

    # --------------------------------------------------------------
    def loss_and_grad(logits_vec):
        """Return (loss, grad) for the current logits."""
        mx = np.max(logits_vec)
        exp_vec = np.exp(logits_vec - mx)
        a = exp_vec / exp_vec.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = np.max(conv)

        # smooth‑max (log‑sum‑exp)
        w_exp = np.exp(lam * (conv - max_c))
        w_sum = w_exp.sum()
        smooth_max = max_c + np.log(w_sum) / lam
        loss = 2.0 * n * smooth_max

        # gradient wrt a
        w = w_exp / w_sum if w_sum != 0 else np.full_like(conv, 1.0 / conv.size)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient wrt logits (soft‑max back‑prop)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        return loss, grad_log.astype(np.float64)

    # --------------------------------------------------------------
    def callback(logits_vec):
        nonlocal best_a, best_val, lam
        # evaluate exact objective for progress monitoring
        mx = np.max(logits_vec)
        exp_vec = np.exp(logits_vec - mx)
        a = exp_vec / exp_vec.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = np.max(conv)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LBFGS n={n}] new exact best {val:.6f}")

        # increase temperature for next iteration
        lam = min(lam * lam_factor, 1e4)

        if time.time() > deadline:
            raise StopIteration

    try:
        minimize(fun=lambda x: loss_and_grad(x),
                 x0=logits,
                 method='L-BFGS-B',
                 jac=True,
                 callback=callback,
                 options={'maxiter': max_iter, 'disp': False})
    except StopIteration:
        pass
    except Exception as exc:
        print(f"LBFGS exception (n={n}): {exc}")

    return best_a, best_val

def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """Exponentiated‑gradient sub‑gradient phase on the true max‑convolution."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """Very cheap stochastic local search: move a tiny mass ε between two indices."""
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[HillClimb n={n}] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

            # Normalise to a probability vector (soft‑max works on this)
            init = np.maximum(init, 0.0)
            s = init.sum()
            if s < 1e-12:
                init = np.random.dirichlet(np.ones(n))
            else:
                init = init / s

            # ---- Adam (smooth‑max) ----
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # ---- L‑BFGS fine‑tuning (smooth‑max) ----
            a_lbfgs, val_lbfgs = _run_lbfgs(a_adam, n, deadline,
                                            max_iter=max_lbfgs,
                                            lam_start=0.5,
                                            lam_factor=1.4)
            if val_lbfgs < best_val:
                best_val, best_seq = val_lbfgs, a_lbfgs.copy()
                print(f"[n={n} cand={cid}] LBFGS set new global best {val_lbfgs:.6f}")

            # ---- Sub‑gradient on true max ----
            a_sub, val_sub = _run_subgrad(a_lbfgs, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

        # End of length loop – attempt a short fine‑tune on the *current* best
        if best_seq is not None and len(best_seq) == n:
            a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                         int(0.5 * max_adam))
            if val_fine < best_val:
                best_val, best_seq = val_fine, a_fine.copy()
                print(f"[n={n}] Fine‑tune (Adam) improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap stochastic hill‑climb to squeeze tiny remaining gains
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\n=== Searching finished: best value ≈ {best_val:.6f}, length = {len(best_seq)} ===")
    return best_seq.tolist()
