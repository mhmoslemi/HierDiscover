   (exponential, power‑law, Dirichlet, triangular, cosine, parabolic,
   sine‑power, random block‑spikes) are generated for varied lengths.
   All are normalised to Σa = √(2 n) and evaluated; the best few are kept.

3. **Differential Evolution on the simplex** – a small population of
   log‑it vectors (so that soft‑max automatically enforces non‑negativity
   and the L¹‑normalisation) is evolved with the classic DE/rand/1/bin
   scheme.  Mutation, crossover and selection are performed on the logits;
   each trial is projected onto the simplex via a stable soft‑max.
   The best individual of the whole run is stored globally.

4. **Continuation Adam on a log‑sum‑exp surrogate** – starting from the
   best DE individual, we run Adam on a smoothed version of the
   max‑convolution (log‑sum‑exp) while gradually decreasing the smoothing
   temperature τ.  This refines the solution in the neighbourhood of the
   current optimum.

5. **Targeted projected sub‑gradient descent** – with the exact max‑
   convolution (no smoothing) we follow the gradient at the worst index
   and project back onto the simplex.

6. **Final simulated‑annealing hill‑climb** – tiny mass transfers between
   random positions are accepted according to a Metropolis rule, allowing
   the solution to escape shallow local minima.

The best sequence found (as a plain Python list) is returned.  All
intermediate progress is printed with timestamps for traceability.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)  # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (L1‑norm = s, non‑negative)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Stable soft‑max mapping from logits to a vector on the simplex of radius target_sum
# ----------------------------------------------------------------------
def _logits_to_simplex(logits: np.ndarray, target_sum: float) -> np.ndarray:
    """Map real logits to a non‑negative vector with Σ = target_sum via soft‑max."""
    max_log = np.max(logits)
    exp_shift = np.exp(logits - max_log)
    p = exp_shift / np.sum(exp_shift)
    return target_sum * p


# ----------------------------------------------------------------------
# Helper: evaluate max‑convolution of a (already scaled to Σa = √(2 n))
# ----------------------------------------------------------------------
def _max_conv_of(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  a is assumed to have the correct L1‑norm."""
    conv = _full_convolution_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Parametric families for the initial random seeding
# ----------------------------------------------------------------------
def _init_exp(n):
    tau = np.random.uniform(0.05, 3.0) * n
    idx = np.arange(n, dtype=float)
    return np.exp(-idx / tau)


def _init_pow(n):
    p = np.random.uniform(0.1, 2.0)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _init_hann(n):
    i = np.arange(n, dtype=float)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _init_hamming(n):
    i = np.arange(n, dtype=float)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def _init_triangular(n):
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(i - mid) / mid
    return np.maximum(a, 0.0)


def _init_cosine(n):
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    a = np.cos(np.pi * (i - mid) / (mid + 1e-12))
    a = (a + 1.0) / 2.0
    return np.maximum(a, 0.0)


def _init_parabolic(n):
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    a = 1.0 - ((i - mid) / mid) ** 2
    return np.maximum(a, 0.0)


def _init_sine_power(n):
    p = np.random.uniform(0.5, 4.0)
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * i / (n - 1)) ** p
    return np.maximum(a, 0.0)


def _init_block_spike(n):
    """Randomly select a subset of indices (≈10 % of n) and assign random mass."""
    k = max(2, int(0.08 * n))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = np.random.rand(k) + 0.1  # avoid exact zeros for stability
    return a


def _normalise_to_target(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a to have L1‑norm = target_sum (preserving non‑negativity)."""
    s = a.sum()
    if s <= 0:
        # fallback to uniform if something went wrong
        return np.full_like(a, target_sum / a.shape[0])
    return a * (target_sum / s)


# ----------------------------------------------------------------------
# Global recorder – keeps the best vector seen so far
# ----------------------------------------------------------------------
_best_seq = None          # numpy array
_best_val = np.inf        # max(conv) for the normalised vector


def _record_candidate(seq: np.ndarray):
    """Normalise seq, evaluate its max‑conv and update the global best."""
    global _best_seq, _best_val
    n = seq.shape[0]
    target = np.sqrt(2.0 * n)
    a = _normalise_to_target(seq, target)
    val = _max_conv_of(a)
    if val < _best_val:
        _best_val = val
        _best_seq = a.copy()
        print(f"[{time.time():.2f}] GLOBAL new best max={_best_val:.6f}  n={n}")
    return val


# ----------------------------------------------------------------------
# Differential Evolution on the simplex (log‑it representation)
# ----------------------------------------------------------------------
def _run_de(
    n: int,
    target_sum: float,
    deadline: float,
    popsize: int = 15,
    F: float = 0.5,
    Cr: float = 0.9,
) -> None:
    """Run a simple DE/rand/1/bin on the log‑it parametrisation."""
    global _best_seq, _best_val

    # Initialise population of logits
    pop = np.empty((popsize, n), dtype=float)

    # First individual seeded from the current global best (if any)
    if _best_seq is not None:
        pop[0] = np.log(_best_seq + 1e-12)
    else:
        # otherwise a random Dirichlet seed
        pop[0] = np.log(np.random.dirichlet(np.ones(n)) + 1e-12)

    # Remaining individuals: random Dirichlet draws
    for i in range(1, popsize):
        pop[i] = np.log(np.random.dirichlet(np.ones(n)) + 1e-12)

    # Evaluate initial population
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        a = _logits_to_simplex(pop[i], target_sum)
        scores[i] = _max_conv_of(a)
        _record_candidate(a)

    # DE main loop
    while time.time() < deadline:
        for i in range(popsize):
            # Choose three distinct indices different from i
            indices = list(range(popsize))
            indices.remove(i)
            a_idx, b_idx, c_idx = np.random.choice(indices, 3, replace=False)

            # Mutation
            mutant = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])

            # Crossover (binomial)
            cross_mask = np.random.rand(n) < Cr
            if not np.any(cross_mask):
                cross_mask[np.random.randint(0, n)] = True
            trial = np.where(cross_mask, mutant, pop[i])

            # Evaluate trial
            a_trial = _logits_to_simplex(trial, target_sum)
            trial_score = _max_conv_of(a_trial)

            # Selection
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                _record_candidate(a_trial)

        # small pause to avoid hogging the interpreter (optional)
        # time.sleep(0.0001)


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed‑max surrogate with continuation on τ
# ----------------------------------------------------------------------
def _optimise_adam_continuation(
    init_logits: np.ndarray,
    target_sum: float,
    deadline: float,
    alpha0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> None:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    The temperature τ is multiplied by 0.6 after each epoch
    (an epoch ≈ n updates), gradually approaching the true max.
    """
    global _best_seq, _best_val

    x = init_logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    it = 0
    alpha = alpha0

    # start with a relatively large τ (10 % of the initial max)
    # the τ schedule will be updated inside the loop
    while time.time() < deadline:
        # soft‑max mapping
        a = _logits_to_simplex(x, target_sum)

        # exact max convolution (used for τ schedule)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        _record_candidate(a)

        # temperature (smoothness) – start with 10 % of cur_max and decay
        # every (≈ 2 n) iterations
        epoch_len = max(2 * a.shape[0], 1)
        tau = max(0.05 * cur_max, 1e-6) * (0.6 ** (it // epoch_len))

        # compute softmax weights for the log‑sum‑exp surrogate
        w_raw = np.exp((conv - cur_max) / tau)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # gradient wrt a: 2·a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * a.shape[0] - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * a.shape[0] - 2]
        idx = (2 * a.shape[0] - 2) - np.arange(a.shape[0])
        grad_a = 2.0 * conv2[idx]

        # Back‑prop to logits (soft‑max Jacobian)
        dotp = np.dot(a / target_sum, grad_a)  # because a = target_sum * p
        grad_x = target_sum * (a / target_sum) * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # Simple learning‑rate decay
        alpha *= 0.9995


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> None:
    """Follow the exact gradient at the current worst index, projecting onto the simplex."""
    global _best_seq, _best_val
    a = a0.copy()
    n = a.shape[0]
    step = 0.15 * target_sum / n  # initial step size

    while time.time() < deadline:
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak
        # gradient of conv[k] w.r.t. a  (size n)
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < _best_val - 1e-12:
            a = cand
            _record_candidate(a)
            step *= 1.03   # slightly enlarge step after success
        else:
            step *= 0.70   # shrink step on failure
            # occasional tiny jitter
            if np.random.rand() < 0.05:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> None:
    """Move tiny amounts of mass between random positions with a Metropolis rule."""
    global _best_seq, _best_val
    a = a0.copy()
    n = a.shape[0]

    # initial temperature proportional to current peak
    T0 = max(1e-3, _best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - frac))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.07
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # re‑normalise occasionally (every 3000 steps) to fight numerical drift
        if it % 3000 == 0:
            a_new = _project_onto_simplex(a_new, target_sum)

        new_val = _max_conv_of(a_new)
        if new_val < _best_val:
            a = a_new
            _record_candidate(a)
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - _best_val) / (T + 1e-12)):
                a = a_new

        # optional break if already excellent
        if _best_val < 1.503:
            break


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimiser that searches for a non‑negative step‑function with
    minimal evaluation score.  The routine respects the supplied time
    budget and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    global _best_seq, _best_val
    _best_seq = None
    _best_val = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # --------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = _normalise_to_target(hs_arr, target0)
            _record_candidate(hs_norm)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – diverse parametric random initialisations
    # --------------------------------------------------------------
    while time.time() < deadline - 300.0:  # reserve ~5 min for later phases
        remaining = deadline - time.time()
        # pick a length (larger lengths later)
        if remaining > 600:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 4000 + 1)

        target_sum = np.sqrt(2.0 * n)

        # choose a parametric family
        shape = np.random.choice(
            ["exp", "pow", "hann", "hamming", "tri", "cos", "parab", "sinepow", "block"]
        )
        if shape == "exp":
            a0 = _init_exp(n)
        elif shape == "pow":
            a0 = _init_pow(n)
        elif shape == "hann":
            a0 = _init_hann(n)
        elif shape == "hamming":
            a0 = _init_hamming(n)
        elif shape == "tri":
            a0 = _init_triangular(n)
        elif shape == "cos":
            a0 = _init_cosine(n)
        elif shape == "parab":
            a0 = _init_parabolic(n)
        elif shape == "sinepow":
            a0 = _init_sine_power(n)
        else:  # block‑spike
            a0 = _init_block_spike(n)

        # optional tiny Gaussian noise
        if np.random.rand() < 0.2:
            noise = np.random.randn(n) * np.max(a0) * 0.015
            a0 = np.maximum(a0 + noise, 0.0)

        # normalise to the required L1 sum
        a0 = _normalise_to_target(a0, target_sum)
        _record_candidate(a0)

    # --------------------------------------------------------------
    # Phase 2 – Differential Evolution on the simplex (log‑it)
    # --------------------------------------------------------------
    if _best_seq is not None and time.time() < deadline - 180.0:
        n_best = _best_seq.shape[0]
        target_best = np.sqrt(2.0 * n_best)
        de_deadline = time.time() + min(150.0, deadline - time.time() - 120.0)
        _run_de(n_best, target_best, de_deadline, popsize=18, F=0.6, Cr=0.85)

    # --------------------------------------------------------------
    # Phase 3 – Continuation Adam on smoothed max
    # --------------------------------------------------------------
    if _best_seq is not None and time.time() < deadline - 120.0:
        n_best = _best_seq.shape[0]
        target_best = np.sqrt(2.0 * n_best)
        # initialise logits from current best
        init_logits = np.log(_best_seq + 1e-12)
        adam_deadline = time.time() + min(80.0, deadline - time.time() - 80.0)
        _optimise_adam_continuation(init_logits, target_best, adam_deadline,
                                    alpha0=0.03)

    # --------------------------------------------------------------
    # Phase 4 – Targeted projected sub‑gradient descent
    # --------------------------------------------------------------
    if _best_seq is not None and time.time() < deadline - 80.0:
        n_best = _best_seq.shape[0]
        target_best = np.sqrt(2.0 * n_best)
        sub_deadline = time.time() + min(60.0, deadline - time.time() - 40.0)
        _targeted_subgradient(_best_seq.copy(), target_best, sub_deadline)

    # --------------------------------------------------------------
    # Phase 5 – Final simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if _best_seq is not None and time.time() < deadline - 5.0:
        n_best = _best_seq.shape[0]
        target_best = np.sqrt(2.0 * n_best)
        hill_deadline = deadline
        _hillclimb(_best_seq.copy(), target_best, hill_deadline)

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # --------------------------------------------------------------
    if _best_seq is None:
        n_fallback = 2000
        target_fallback = np.sqrt(2.0 * n_fallback)
        _best_seq = np.full(n_fallback, target_fallback / n_fallback, dtype=float)
        _best_val = _max_conv_of(_best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={_best_val:.6f}")

    # Return a plain Python list as required by the evaluator
    return _best_seq.tolist()
