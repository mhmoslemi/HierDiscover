def _pgd_line_search(a0, n, deadline, best_seq, best_score, rng):
    target_sum = sqrt(2*n)
    a = a0.copy()
    a = a * target_sum / a.sum()
    a = _symmetrize(a)

    Nfft = 1 << ((3*n-2).bit_length())
    step = step0 = target_sum * 0.001
    for it in range(max_iters):
        if time.time()>deadline: break
        # Compute convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(...)

        # Build weight w for active indices
        eps_active = M * 1e-12
        mask = conv >= M - eps_active
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(conv)] = 1.0

        # compute subgradient rev via weighted reverse convolution
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[:len(w)+n-1]
        idx_start = 2*n-2
        rev = conv_rev[idx_start - np.arange(n)]

        grad = 2.0 * rev

        # Backtracking line search
        alpha = step
        while True:
            a_candidate = a - alpha * grad
            a_candidate = _project_onto_simplex(a_candidate, target_sum)
            a_candidate = _symmetrize(a_candidate)
            # compute new max
            A_fft_c = np.fft.rfft(a_candidate, n=Nfft)
            conv_c = np.fft.irfft(A_fft_c * A_fft_c, n=Nfft)[:2*n-1]
            M_c = conv_c.max()
            if M_c <= M - 1e-12:
                break
            alpha *= 0.5
            if alpha < 1e-12:  # cannot improve
                break

        a = a_candidate
        step = max(step * 0.99, 1e-6)  # gradually reduce
        # occasional jitter
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)
            a = _symmetrize(a)
```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Work on the simplex {a_i ≥ 0, Σ a_i = √(2 n)}.
2.  Enforce symmetry (a_i = a_{n‑1‑i}) – symmetric shapes give very low
    autocorrelation peaks.
3.  Use Projected Gradient Descent on the exact max‑convolution.
    * Sub‑gradient is obtained by the weighted reverse convolution of the
      active (max‑attaining) indices.
    * After each step the vector is projected onto the simplex and symmetrised.
    * A simple back‑tracking line search guarantees descent.
4.  Run the optimiser from many diverse initial vectors (uniform, Dirichlet,
    exponential, cosine windows, sparse spikes, and the previously best
    construction if available).
5.  After the global best is found, perform a cheap stochastic refinement:
    random mass transfers with a simulated‑annealing acceptance rule, always
    keeping the vector symmetric.
6.  The whole search respects the 1000 s budget (a 5 s safety margin).

The score returned by ``evaluate_sequence`` is exactly the maximum of the
autoconvolution (because the sum is normalised to √(2 n)), so we minimise that
maximum directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σ x_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} (exact symmetry)."""
    return (a + a[::-1]) * 0.5


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Generate a diverse set of feasible (symmetric) start vectors."""
    inits = []

    # uniform
    inits.append(np.full(n, target_sum / n))

    # Dirichlet (Gamma)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    inits.append(g / g.sum() * target_sum)

    # exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    inits.append(x / x.sum() * target_sum)

    # reversed exponential
    inits.append(x[::-1] / x[::-1].sum() * target_sum)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    inits.append(hann / hann.sum() * target_sum)

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    inits.append(black / black.sum() * target_sum)

    # small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    inits.append(pert / pert.sum() * target_sum)

    # triangular shape
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    inits.append(tri / tri.sum() * target_sum)

    # sparse random spikes
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    inits.append(sparse)

    # enforce symmetry on all candidates
    sym_inits = []
    for v in inits:
        v = _symmetrize(v)
        v = _project_onto_simplex(v, target_sum)
        sym_inits.append(v)

    return sym_inits

def _pgd_line_search(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int,
                     best_seq: np.ndarray,
                     best_score: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on max‑convolution with back‑tracking line search.
    Returns the best vector observed (as numpy array) and its max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise on the simplex and enforce symmetry
    a = a0.copy()
    a = a * target_sum / a.sum()
    a = _symmetrize(a)
    a = _project_onto_simplex(a, target_sum)

    # FFT size – must accommodate linear conv of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # basic step size (a few % of total mass)
    base_step = 0.001 * target_sum
    step = base_step

    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current max‑convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- active set (hard max) -----
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ----- weighted reverse convolution (sub‑gradient) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]
        grad = 2.0 * rev  # sub‑gradient of max(conv)

        # ----- line search -----
        alpha = step
        improved = False
        for ls in range(10):
            a_candidate = a - alpha * grad
            a_candidate = _project_onto_simplex(a_candidate, target_sum)
            a_candidate = _symmetrize(a_candidate)

            A_c_fft = np.fft.rfft(a_candidate, n=Nfft)
            conv_c = np.fft.irfft(A_c_fft * A_c_fft, n=Nfft)[: 2 * n - 1]
            M_c = float(conv_c.max())

            if M_c <= M - 1e-12:
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # no descent found – keep old a but shrink step
            step *= 0.5
            continue

        # accept candidate
        a = a_candidate
        step = max(alpha * 0.9, 1e-9)  # keep a bit of momentum

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)
            a = _symmetrize(a)

    return best_seq, best_score


def _local_refine_sym(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_score: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the symmetric simplex.
    Moves a small fraction of mass from i→j (and its mirror) and accepts
    improvements or occasional uphill steps (simulated‑annealing).
    """
    target_sum = a.sum()  # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i = rng.integers(0, n)
        if cur_seq[i] < 1e-12:
            step += 1
            continue

        # pick a destination not equal to i
        j = rng.integers(0, n)
        if j == i:
            step += 1
            continue

        # transfer a random fraction (1‑5 %) of source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # maintain symmetry after the transfer
        new_seq = _symmetrize(new_seq)
        new_seq = _project_onto_simplex(new_seq, target_sum)

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score
