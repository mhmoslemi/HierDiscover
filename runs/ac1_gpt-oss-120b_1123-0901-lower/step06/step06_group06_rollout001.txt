4. **Local random perturbations** – occasional “mass‑shifts” (simulated‑annealing‑style)

# FFT based linear convolution a*a and the cross‑correlation needed for gradients.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2*n-1 entries (linear convolution)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed to be 1)."""
    conv = _conv_fft(a, conv_len)
    max_c = conv.max()
    return 2.0 * n * max_c

            cand.append(shape / shape.sum())

    # 5) two‑sided exponential decay
    for gamma in (0.0008, 0.0015, 0.003, 0.006):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # 6) cosine window (single bump)
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    cand.append(coswin / coswin.sum())

    # 7) half‑sine window
    sinwin = np.sin(np.pi * i / (n - 1))
    cand.append(sinwin / sinwin.sum())

    # 8) discrete Gaussian
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    cand.append(gauss / gauss.sum())

    # 9) truncated arcsine (inverse‑sqrt edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    cand.append(arcs / arcs.sum())

    # 10) random sparse spikes (≈ sqrt(n) spikes)
    m_spikes = max(2, int(np.sqrt(2 * n)))
    sparse = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=m_spikes, replace=False)
    sparse[indices] = np.random.rand(m_spikes) + 0.1
    sparse = sparse / sparse.sum()
    cand.append(sparse)

    return cand

def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_len: int,
                     corr_len: int,
                     deadline: float,
                     max_iters: int,
                     lr_factor: float,
                     lam_start: float,
                     lam_factor: float,
                     patience: float,
                     verbose_prefix: str) -> tuple[np.ndarray, float]:
    """
    Adam optimise a soft‑max surrogate of max(conv).  Returns best (a, val)
    found during the run.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lr = lr_factor * np.sqrt(1024.0 / n)          # base learning‑rate
    lam = lam_start
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ---- soft‑max to respect the simplex ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- convolution and smooth‑max weighting ----
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth surrogate ----
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a) == 1)
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"{verbose_prefix}[Adam iter={t:05d}] new best {val:.6f}")

        # early‑exit on stagnation
        if time.time() - last_improve > patience:
            # Add a tiny random jitter to escape plateaus
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _exponentiated_subgrad(a0: np.ndarray,
                          n: int,
                          conv_len: int,
                          corr_len: int,
                          deadline: float,
                          max_iters: int,
                          eta0: float,
                          patience: float,
                          verbose_prefix: str) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated updates.
    Returns best sequence and value seen.
    """
    a = a0.copy()
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        # active set of indices attaining the max (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # decreasing step size
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:               # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # recompute objective (exact max)
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"{verbose_prefix}[Subgrad iter={it}] new best {val:.6f}")

        # stagnation check
        if time.time() - last_improve > patience:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _random_local_search(a0: np.ndarray,
                         n: int,
                         conv_len: int,
                         deadline: float,
                         max_seconds: float,
                         verbose_prefix: str) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic “mass‑shift” search.  Proposes a random transfer
    of a small amount of probability mass between two positions.
    Accepts only improvements (greedy) to keep it fast.
    """
    a = a0.copy()
    val = _objective(a, n, conv_len)
    best_val = val
    best_a = a.copy()
    start = time.time()

    while time.time() - start < max_seconds and time.time() < deadline:
        i, j = np.random.choice(n, size=2, replace=False)
        # move a random fraction of the smaller of a[i] and a[j]
        delta = 0.01 * min(a[i], a[j]) * np.random.rand()
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalise (numerical drift)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        val_new = _objective(a_new, n, conv_len)

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            print(f"{verbose_prefix}[Local perturb] new best {val_new:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function (list of heights) with a low
    evaluation score.  The routine respects a time budget and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – start small, then go larger if time permits.
    lengths = [1024, 2048, 4096, 8192]

    # pre‑compute FFT lengths (next power of two) for speed
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # correlation needs length conv_len + n - 1
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # ---- Build a diversified pool of seeds ----
        candidates = _build_initial_candidates(n)

        # ---- Optimise each seed ---------------------------------------
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            verbose = f"[n={n} cand={cid}] "

            # Normalise to sum = 1 (the objective is scale‑invariant)
            a0 = a0 / a0.sum()

            # ----- Stage 1: Adam on smooth‑max surrogate -----
            max_iters_adam = int(25000 * 1024 / n) + 8000   # more iterations for smaller n
            a_adam, val_adam = _adam_smooth_max(
                a0,
                n=n,
                conv_len=conv_len,
                corr_len=corr_len,
                deadline=deadline,
                max_iters=max_iters_adam,
                lr_factor=0.07,
                lam_start=0.5,
                lam_factor=1.025,
                patience=6.0,
                verbose_prefix=verbose,
            )

            # ----- Stage 2: Exact‑max exponentiated‑gradient -----
            a_sub, val_sub = _exponentiated_subgrad(
                a_adam,
                n=n,
                conv_len=conv_len,
                corr_len=corr_len,
                deadline=deadline,
                max_iters=18000,
                eta0=0.55,
                patience=8.0,
                verbose_prefix=verbose,
            )

            # ----- Stage 3: Random local perturbations (quick greedy) -----
            a_loc, val_loc = _random_local_search(
                a_sub,
                n=n,
                conv_len=conv_len,
                deadline=deadline,
                max_seconds=12.0,
                verbose_prefix=verbose,
            )

            # Keep the best of the three phases for this candidate
            cand_best_a, cand_best_val = a_loc, val_loc
            if val_adam < cand_best_val:
                cand_best_a, cand_best_val = a_adam, val_adam
            if val_sub < cand_best_val:
                cand_best_a, cand_best_val = a_sub, val_sub

            if cand_best_val < best_val:
                best_val = cand_best_val
                best_seq = cand_best_a.copy()
                print(f"{verbose}** GLOBAL NEW BEST {best_val:.6f}")

        # ------------------------------------------------------------------
        # Fine‑tune the current global best for this length (if it matches the size)
        if best_seq is not None and best_seq.shape[0] == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a_fine = best_seq.copy()
            logits = np.log(a_fine + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            lr = 0.045 * np.sqrt(1024.0 / n)
            lam = 0.6
            lam_factor = 1.015
            last_improve = time.time()
            fine_iters = int(50000 * 1024 / n) + 8000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # If nothing succeeded (should not happen) fall back to a uniform vector.
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq, fallback_n, fft_len_conv[fallback_n])

    print(f"\n*** FINAL BEST value = {best_val:.6f} with length {best_seq.shape[0]} ***")
    # Return a plain Python list, as expected by the external evaluator.
    return best_seq.tolist()
