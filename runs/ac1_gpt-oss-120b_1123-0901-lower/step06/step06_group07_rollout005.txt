"""
Improved search for a low‑evaluation step function.

Algorithm
---------

1. **Parameterised families** – Produce a large, diverse pool of candidate
   vectors (uniform, warm‑start rotations, random Dirichlet,
   edge‑exponential/power‑law, symmetric Gaussian mixtures,
   beta‑type shapes, smoothed Dirichlet, higher‑frequency half‑wave
   sinusoids).  All candidates are normalised to sum‑to‑one.

2. **Fast pre‑selection** – Evaluate every candidate with a cheap FFT‑based
   convolution and keep the best `TOP_K` (default = 20).

3. **Refinement pipeline**
   * **Adam on a smooth‑max surrogate** – minimise a log‑sum‑exp
     approximation to the true max‑convolution while keeping the vector on
     the simplex via a soft‑max re‑parameterisation.  Temperature (`lam`)
     is increased slowly so the surrogate converges to the true max.
   * **Exact sub‑gradient phase** – using the true max‑convolution,
     perform exponentiated‑gradient updates (projected onto the simplex)
     to sharpen the solution.
   * **Local stochastic hill‑climb** – after the sub‑gradient phase,
     repeatedly move a tiny amount of mass between two random indices;
     accept only improving moves.  This often discovers tiny tweaks that
     further lower the objective.

4. **Scale‑up** – If time permits, linearly interpolate the best 4096‑vector
   to 8192 entries and run the same refinement pipeline on the larger grid.

All steps respect the 1000 s time budget and keep track of the overall best
vector and its objective value.  The final vector is clipped to the range
accepted by `evaluate_sequence` and returned as a plain Python list.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)-1 using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the true objective.
    a is assumed to be a probability vector (sum≈1, non‑negative).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of max‑convolution.
    Returns the best vector and value observed during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    # start with a very smooth approximation, then sharpen
    lam = 0.5
    lam_factor = 1.02
    lam_cap = 5e5

    last_improve = time.time()
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution & true value (for occasional logging) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        # w_i ∝ exp(lam * (conv_i - max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate (chain rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- random shake if stagnating ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exact max‑convolution sub‑gradient steps using exponentiated updates.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 30_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if stagnating
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _local_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climb: move a tiny mass between two random
    entries and keep the move if it improves the objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    last_improve = time.time()
    iteration = 0
    max_iter = 200_000

    while time.time() < deadline and iteration < max_iter:
        i, j = np.random.choice(n, size=2, replace=False)
        # propose moving a small random chunk from i to j
        delta = (np.random.rand() - 0.5) * 2e-3  # symmetric around 0
        if a[i] - delta < 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        # renormalise (sum unchanged, only numerical drift)
        # evaluate
        val = _objective(a, n, conv_fft_len)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # keep the improvement
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        if time.time() - last_improve > 6.0:
            break

        iteration += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Execute the three refinement stages (Adam → sub‑gradient → hill‑climb)
    and update the global best.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    best_seq, best_val = _local_hill_climb(
        best_seq if best_seq is not None else a0,
        n,
        conv_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _gaussian_kernel(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian kernel of length 2·n‑1 (used for smoothing)."""
    k = np.arange(-n + 1, n, dtype=np.float64)
    g = np.exp(-0.5 * (k / sigma) ** 2)
    return g / g.sum()


def _smooth_vector(v: np.ndarray, sigma: float, fft_len: int) -> np.ndarray:
    """Convolve v with a Gaussian kernel (zero‑padded) using FFT."""
    n = v.shape[0]
    kernel = _gaussian_kernel(n, sigma)
    fv = np.fft.rfft(v, fft_len)
    fk = np.fft.rfft(kernel, fft_len)
    smooth = np.fft.irfft(fv * fk, fft_len)[: n]
    return smooth


def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Assemble a diverse set of candidate vectors.
    All returned vectors are normalised to sum‑to‑one.
    """
    candidates = []

    # ---- warm‑start and a few rotations ----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ---- uniform baseline ----
    candidates.append(np.full(n, 1.0 / n))

    # ---- random Dirichlet (various concentrations) ----
    for conc in (0.03, 0.07, 0.15, 0.3, 0.6, 1.2, 2.5):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # ---- edge‑exponential families, mixed with uniform ----
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=16):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # ---- edge‑power‑law families ----
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.35, 0.7, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # ---- symmetric Gaussian mixtures (two bumps) ----
    for sigma_fac in (0.02, 0.04, 0.07, 0.12, 0.20):
        sigma = sigma_fac * n
        for w in (0.2, 0.5, 0.8):
            mu1 = int(0.25 * n)
            mu2 = n - 1 - mu1
            gauss = np.exp(-0.5 * ((i - mu1) / sigma) ** 2) + np.exp(-0.5 * ((i - mu2) / sigma) ** 2)
            gauss /= gauss.sum()
            cand = w * gauss + (1.0 - w) / n
            candidates.append(cand)

    # ---- Beta‑type shapes (zero at the edges) ----
    for a_exp in (0.5, 1.0, 2.0, 4.0):
        for b_exp in (0.5, 1.0, 2.0, 4.0):
            beta_shape = (i / (n - 1) + 1e-12) ** a_exp * (1 - i / (n - 1) + 1e-12) ** b_exp
            beta_shape /= beta_shape.sum()
            candidates.append(beta_shape)

    # ---- smoothed Dirichlet (Gaussian blur) ----
    for sigma_fac in (0.01, 0.03, 0.06):
        for conc in (0.1, 0.5, 1.0):
            raw = np.random.dirichlet(np.full(n, conc))
            sigma = sigma_fac * n
            fft_len = _next_pow2(2 * n - 1)
            smooth = _smooth_vector(raw, sigma, fft_len)
            smooth = np.maximum(smooth, 0.0)
            smooth /= smooth.sum()
            candidates.append(smooth)

    # ---- higher‑frequency half‑wave sinusoids (positive part only) ----
    for k in (1, 2, 3, 4):
        s = np.sin(k * np.pi * (np.arange(n) + 0.5) / n)
        s = np.maximum(s, 0.0)
        s /= s.sum()
        candidates.append(s)

    # ---- random perturbations of the best known vector (if available) ----
    if warm_start is not None:
        for _ in range(8):
            pert = warm_start + 0.02 * (np.random.rand(n) - 0.5)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            candidates.append(pert)

    # Remove exact duplicates (within tolerance) to keep list sane
    uniq = []
    seen = set()
    for cand in candidates:
        key = tuple(np.round(cand, 8))
        if key not in seen:
            seen.add(key)
            uniq.append(cand)
    return uniq

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point: search for a step‑function with minimal evaluation score.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # slight safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Initial grid size (the sweet spot from earlier experiments)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from the previously discovered best, if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start /= warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool and keep the most promising
    all_cands = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_cands):
        if time.time() >= deadline:
            break
        # sanity normalisation
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))
        # optional progress print:
        # print(f"[candidate {idx}] quick val = {val:.6f}")

    # keep the top K for deep optimisation
    cand_vals.sort(key=lambda x: x[0])
    TOP_K = 20
    top_candidates = [c for _, c in cand_vals[:TOP_K]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on each selected start
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} (n={n}) ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – if budget permits, upscale to a finer grid
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation onto larger grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed range and return as plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
