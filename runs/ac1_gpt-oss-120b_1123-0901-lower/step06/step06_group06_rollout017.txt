def _exponentiated_gradient_opt(a, n, conv_len, corr_len, max_iter, deadline, best_val_global, best_seq_global):

       conv = _conv_fft(a, conv_len); max_c = conv.max()

       val = 2.0 * n * max_c # as sum 1

       if val < best_val_global: update.

       # compute weight w (active)
       # tolerance: 1e-12 absolute.

       active = np.isclose(conv, max_c, atol=1e-12)

       if not active.any(): break

       w = np.zeros_like(conv)

           # restart from uniform
           a = np.full(n, 1.0/n)
       else:
           a = a / s

   return a, best_val_global, best_seq_global

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')

    # base length list
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]

    # potentials for shape parameters
    exp_gammas = np.geomspace(5e-5, 0.02, 12) # adjust for larger n
    alphas = np.linspace(-2.5, -0.2, 12)
    asym_pairs = [(-2.0, -0.5), (-1.5, -0.5), (-1.0, -0.2), (-0.5, -0.2)]

    # Start with warm start from global var if present else uniform
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base_seq = None

    for n in lengths:
        if time.time() > deadline:
            break

        conv_len = 1 << int(np.ceil(np.log2(2*n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(3*n - 2)))

        # generate initial candidates
        candidates = []

        # 1: upsample previous best (if best_seq of same length else upsample)
        if best_seq is not None:
            if best_seq.shape[0] != n:
                # linear interpolation to new length
                old_n = best_seq.shape[0]
                x_old = np.linspace(0, 1, old_n)
                x_new = np.linspace(0, 1, n)
                up = np.interp(x_new, x_old, best_seq)
                candidates.append(up / up.sum())
            else:
                candidates.append(best_seq.copy())

        # 2: base_seq if exists, upsample similarly
        if base_seq is not None:
            if base_seq.shape[0] != n:
                old_n = base_seq.shape[0]
                x_old = np.linspace(0, 1, old_n)
                x_new = np.linspace(0, 1, n)
                up = np.interp(x_new, x_old, base_seq)
                candidates.append(up / up.sum())
            else:
                candidates.append(base_seq.copy() / base_seq.sum())

        # 3: uniform
        candidates.append(np.full(n, 1.0/n))

        # 4: exponential families
