def _pgd_optimize(a0, n, deadline, max_iters, best_seq, best_score, rng):
    target_sum = sqrt(2n)
    a = a0.copy()
    a = a * target_sum / a.sum()
    Nfft = next_pow2(3*n-2)
    step0 = 0.001 * target_sum
    beta = 0.9
    v = np.zeros_like(a)
    eps_active_rel = 1e-6  # relative tolerance

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = float(conv.max())
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n} it={it}] new best {best_score:.6f}")

        # active mask with tolerance:
        thresh = M * (1 - eps_active_rel)
        mask = conv >= thresh
        if mask.sum() == 0:
            mask[np.argmax(conv)] = True
        w = mask.astype(float)
        w /= w.sum()
        # gradient
        # compute reverse conv weighted by w
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        rev = rev[:len(w)+n-1]
        idx_start = 2*n-2
        rev_vec = rev[idx_start - np.arange(n)]
        g = 2.0 * rev_vec

        # momentum update
        v = beta * v + (1 - beta) * g

        # step size schedule
        lr = step0 / math.sqrt(it)
        a_new = a - lr * v
        a = _project_onto_simplex(a_new, target_sum)

        # jitter
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)
    return best_seq, best_score

     (temperature starts at 0.5·max and decays slowly).  
   – **Projected Gradient Descent (PGD)** with momentum, a tolerance‑based
     active set and a simple annealing step‑size schedule.  
   – Both optimisers run until the time budget is exhausted (they respect
     a global deadline).

4. **Stochastic refinement**  
   – Mass‑transfer hill‑climbing with simulated‑annealing acceptance.
     Sources are chosen proportionally to their mass, destinations
     uniformly, and transfers are 1‑5 % of the source mass.  The move
     that yields the smallest max‑convolution is kept.

All vectors are projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)} after each update, guaranteeing a valid
candidate for `evaluate_sequence`.  The routine respects the 1000 s
budget (a 5‑second safety margin is kept).

The function `propose_candidate` returns the best list of heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a * a (linear convolution) using a single FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Generate a short, diverse set of feasible start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0_, a1_, a2_ = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0_
             - a1_ * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2_ * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes (helps explore highly non‑uniform shapes)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Momentum, tolerant active set and occasional jitter are used.
    Returns the best vector observed and its max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    a = a * target_sum / a.sum()

    # FFT size that can hold the convolution (linear, length 2n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # step‑size schedule
    step0 = 0.001 * target_sum
    beta = 0.9                     # momentum coefficient
    v = np.zeros_like(a)          # momentum buffer
    rel_tol = 1e-6                # relative tolerance for active set

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # ----- active set (max tolerant) -----
        thresh = M * (1.0 - rel_tol)
        mask = conv >= thresh
        cnt = mask.sum()
        if cnt == 0:
            mask[np.argmax(conv)] = True
            cnt = 1
        w = mask.astype(float) / float(cnt)

        # ----- weighted reverse convolution (sub‑gradient) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        rev = rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev_vec = rev[idx_start - np.arange(n)]

        g = 2.0 * rev_vec                     # ∇_a max(a∗a)

        # ----- momentum update & gradient step -----
        v = beta * v + (1.0 - beta) * g
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * v
        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- jitter to escape plateaux -----
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑parameters (exponentiated‑gradient) with a
    temperature‑smoothed max.  The temperature starts at 0.5·M and decays
    linearly to 1e‑9.
    """
    target_sum = math.sqrt(2.0 * n)
    eps = 1e-16

    # initialise logits from the start vector
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.12                     # base learning rate for logits

    Nfft = 1 << ((2 * n - 1).bit_length())

    # initial temperature (half of initial max)
    A_fft = np.fft.rfft(target_sum * np.exp(y - np.max(y)) / np.exp(y - np.max(y)).sum(), n=Nfft)
    conv0 = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    tau0 = 0.5 * max(conv0)        # start temperature

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector (softmax of logits) -----
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # ----- forward convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # ----- temperature schedule (linear decay) -----
        tau = max(tau0 * (1.0 - it / max_iters), 1e-9)

        # ----- soft‑max weights for the smoothed max -----
        if tau < 1e-12:
            # fall back to hard max
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            sum_ex = ex.sum()
            w = ex / sum_ex

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        rev = rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev_vec = rev[idx_start - np.arange(n)]

        # ----- gradient wrt a -----
        g_a = 2.0 * rev_vec

        # ----- chain rule to logits (mirror descent on the simplex) -----
        avg = np.dot(p, g_a)                 # <p, g_a>
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # ----- occasional jitter -----
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex.
    Moves a random fraction (1‑5 %) of mass from a source (chosen
    proportionally to its weight) to a destination (uniform).  Improves
    are always accepted; worsening moves are accepted with a
    simulated‑annealing probability that decays linearly.
    """
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 500_000
    step = 0

    # pre‑compute source probabilities (updated each iteration)
    while time.time() < deadline and step < max_steps:
        # source proportional to mass, destination uniform
        prob_src = cur_seq / cur_seq.sum()
        i = rng.choice(n, p=prob_src)
        j = rng.integers(0, n, endpoint=False)
        if i == j:
            step += 1
            continue

        mass_i = cur_seq[i]
        if mass_i <= 1e-12:
            step += 1
            continue

        # transfer a random fraction of source mass
        frac = rng.uniform(0.01, 0.05)
        delta = mass_i * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improve → {cur_score:.6f} (step {step})")
        else:
            # simulated annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load historic construction if present
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # 1. Global scan over a few lengths
    candidate_lengths = [300, 500, 800, 1000, 1200, 1500, 1800, 2100, 2400, 3000]
    if base is not None:
        candidate_lengths.append(base.shape[0])
    candidate_lengths = sorted({max(300, min(int(L), 8000)) for L in candidate_lengths})

    length_best = []                     # (best_M, n, best_vector)
    local_deadline = deadline * 0.4      # about 40 % of the budget for the scan

    for n in candidate_lengths:
        if time.time() >= local_deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        Nfft = 1 << ((2 * n - 1).bit_length())
        init_vecs = _make_initial_vectors(n, target_sum, rng)

        # supplement with a few random Dirichlet draws if we have <12 vectors
        while len(init_vecs) < 12:
            g = rng.gamma(1.0, 1.0, size=n)
            init_vecs.append(g / g.sum() * target_sum)

        # if we have a historic best of different size, interpolate it
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vecs.append(interp)

        best_M_n = float("inf")
        best_a_n = None

        for a in init_vecs:
            M = _max_convolution(a, Nfft)
            if M < best_M_n:
                best_M_n = M
                best_a_n = a.copy()

        length_best.append((best_M_n, n, best_a_n))
        print(f"[scan] length {n:5d} best M = {best_M_n:.6f}")

    # ------------------------------------------------------------------
    # 2. Pick the most promising lengths (top‑2)
    length_best.sort(key=lambda x: x[0])
    promising = length_best[:2] if len(length_best) >= 2 else length_best

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 3. Intensive optimisation on the promising lengths
    for best_M_n, n, a0 in promising:
        if time.time() >= deadline:
            break
        print(f"[optim] working on length {n} (initial M = {best_M_n:.6f})")
        target_sum = math.sqrt(2.0 * n)

        # allow a fairly large number of iterations – the optimiser will stop
        # automatically when the deadline is reached
        max_iters = 2_000_000

        # Adam (exponentiated‑gradient) pass
        best_seq, best_score = _exp_gradient_optimize(
            a0=a0.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq if best_seq is not None else a0,
            best_score=best_score,
            rng=rng,
        )

        # PGD pass (momentum‑based)
        best_seq, best_score = _pgd_optimize(
            a0=a0.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # 4. Final stochastic refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
