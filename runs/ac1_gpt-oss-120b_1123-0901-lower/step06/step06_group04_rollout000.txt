"""
Propose a step‑function (non‑negative vector) that minimizes

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The objective is homogeneous, so we keep the vector on the simplex
(sum = 1).  The algorithm consists of three phases:

1. **Long Adam run on a smooth‑max surrogate** – the temperature
   (smooth‑max “λ”) is increased slowly so the surrogate approaches the
   true ℓ∞‑norm.  Adam with a soft‑max parametrisation guarantees the
   simplex constraints.

2. **Exact sub‑gradient refinement** – after the temperature is large,
   the true max‑convolution is used and an exponentiated‑gradient (mirror‑descent)
   step is applied on the simplex.

3. **Local stochastic search** – a cheap hill‑climbing / simulated‑annealing
   loop that perturbs the current best in the log‑space (logits) with
   low‑frequency noise, keeping only improvements.

The routine respects the wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / cross‑correlation helpers (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 15000,
    lr: float = 0.07,
    lam_init: float = 0.5,
    lam_factor: float = 1.015,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val).  ``best_val`` is the exact quantity
    2·n·max(conv) (i.e. the evaluation function with sum(a)=1).
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lam = lam_init
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- soft‑max re‑parametrisation ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- convolution and smooth‑max weights ---
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # --- gradient of smooth‑max (w.r.t. a) ---
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # --- back‑prop through soft‑max (logits gradient) ---
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- temperature annealing ---
        lam = min(lam * lam_factor, 1e12)

        # --- exact evaluation (since sum=1) ---
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # --- occasional jitter to escape plateaus ---
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement via exponentiated‑gradient updates.
    Returns (best_a, best_val).
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = eta0
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # numerical safeguard
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                     # slow decay

        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_trials: int = 3000,
    init_sigma: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing / simulated‑annealing in the log‑space.
    Random Gaussian perturbations are added to the logits; only
    improvements are kept.  Returns (best_a, best_val).
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = float("inf")
    # pre‑compute initial value
    conv = _full_convolution(a, fft_len=conv_fft_len)
    best_val = 2.0 * n * conv.max()

    sigma = init_sigma
    trials = 0
    while trials < max_trials and time.time() < deadline:
        trials += 1

        # perturbation in log‑space (logits)
        logits = np.log(np.maximum(best_a, 1e-12))
        # low‑frequency noise: convolve with a short Gaussian kernel
        noise = np.random.normal(scale=1.0, size=n)
        # smooth the noise (simple moving average)
        k = max(3, int(0.005 * n))   # few % of the length
        kernel = np.exp(-0.5 * (np.arange(k) - k // 2) ** 2)
        kernel /= kernel.sum()
        smooth_noise = np.convolve(noise, kernel, mode='same')
        logits += sigma * smooth_noise

        a_candidate = np.exp(logits - logits.max())
        a_candidate = a_candidate / a_candidate.sum()

        conv = _full_convolution(a_candidate, fft_len=conv_fft_len)
        val = 2.0 * n * conv.max()

        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()
            sigma = max(0.001, sigma * 0.95)   # shrink step size after success

        # slowly anneal sigma even without improvement
        if trials % 200 == 0:
            sigma = max(0.001, sigma * 0.99)

    return best_a, best_val

                    a_adam,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=max_iters_sub,
                )

                # Phase 3: Local random search around the refined point
                remaining = deadline - time.time()
                max_trials = int(min(4000, max(500, remaining / 0.0008)))
                a_loc, val_loc = _local_random_search(
                    a_sub,
                    n,
                    conv_fft_len,
                    deadline,
                    max_trials=max_trials,
                )

                # Update global best if any phase yielded improvement
                for cand_a, cand_val in [(a_adam, val_adam),
                                         (a_sub, val_sub),
                                         (a_loc, val_loc)]:
                    if cand_val < best_val:
                        best_val = cand_val
                        best_seq = cand_a.copy()
                        print(
                            f"[n={n} seed={sid}] new global best {best_val:.6f}"
                        )

            except Exception as exc:
                print(f"[n={n} seed={sid}] exception: {exc}")

        # ------------------------------------------------------------------
        # Final jitter / fine‑tuning on the current best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(4):
                if time.time() > deadline:
                    break

                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.025, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_jit, val_jit = _adam_optimize(
                    a0,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=4000,
                    lr=0.06,
                    lam_init=0.3,
                )
                a_jit, val_jit = _subgradient_refine(
                    a_jit,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=3000,
                )
                if val_jit < best_val:
                    best_val = val_jit
                    best_seq = a_jit.copy()
                    print(
                        f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}"
                    )

    # ------------------------------------------------------------------
    # Fallback safeguard
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
