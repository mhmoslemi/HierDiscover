"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search consists of three phases:

1. **Symmetric initialisation** – build a diverse set of feasible vectors
   (uniform, Dirichlet, exponential, window functions, interpolated previous
   best) and enforce mirror symmetry a_i = a_{n‑1‑i}.

2. **Two complementary gradient‑based optimisers**  
   – *Projected Gradient Descent* (PGD) on the exact max‑convolution, using the
   true sub‑gradient obtained from the active set and projecting onto the
   simplex after each step.  
   – *Exponentiated‑gradient Adam* where the max is approximated by a
   temperature‑controlled soft‑max.  In both schemes the iterate is forced to
   stay symmetric (averaging with its reverse) which drastically reduces the
   search space and tends to produce lower peaks.

3. **Symmetric stochastic refinement** – a simulated‑annealing hill‑climbing
   that repeatedly transfers a random fraction of mass from one index to
   another together with its mirrored partner.  The move is always accepted if
   it lowers the current maximum convolution; otherwise it is accepted with
   a temperature‑controlled probability.

The whole routine respects the 1000 s budget, dynamically allocating time to
the optimisation phases and the final refinement.  The best sequence found is
returned as a plain Python list.
"""

import time
import math
import numpy as np

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target_sum}.
    Implements the algorithm of Duchi et al. (2008) in O(n log n).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry a_i = a_{n‑1‑i}."""
    return (v + v[::-1]) * 0.5


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Generate a small, diverse set of feasible start vectors."""
    init = []

    # 1) Uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) Exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) Reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) Small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) Triangular shape
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) Sparse random spikes (helps explore non‑uniform shapes)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    # enforce symmetry (preserves sum)
    return [_symmetrize(v) for v in init]

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed and its exact objective value (max conv).
    """
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * target_sum / a.sum()          # ensure exact sum

    # FFT size – must accommodate linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    step0 = 0.001 * target_sum            # base step size
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution & current max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- active set -----
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ----- weighted reverse convolution (sub‑gradient) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev                     # ∇_a max(a∗a)

        # ----- gradient step + symmetry + projection -----
        lr = step0 / math.sqrt(it)        # annealed step size
        a_tmp = a - lr * g
        a_tmp = _symmetrize(a_tmp)
        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- occasional jitter -----
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(_symmetrize(a + jitter), target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a
    temperature‑controlled soft‑max approximation of the max‑convolution.
    Returns the best vector seen together with its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)          # logits

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.12                      # a bit larger than before
    jitter_interval = 2000

    # temperature schedule parameters
    tau_factor = 0.35               # initial temperature as fraction of M
    decay_rate = max_iters / 6.0    # slower decay → smoother optimisation

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector from logits -----
        logits = y - np.max(y)                 # shift for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()      # lies on the probability simplex
        a = target_sum * p                     # respects Σa = target_sum, a_i ≥ 0
        a = _symmetrize(a)                     # enforce mirror symmetry
        a = a / a.sum() * target_sum            # re‑normalise (tiny numerical drift)

        # ----- convolution & max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- update global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- soft‑max weights for the (smoothed) max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # fall back to hard max
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            sum_ex = ex.sum()
            w = ex / sum_ex

        # ----- weighted reverse convolution -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        # ----- gradient w.r.t. a -----
        g_a = 2.0 * rev

        # ----- mirror‑descent on logits (Adam) -----
        avg = np.dot(p, g_a)                  # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)    # ∂f/∂y_i

        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to avoid stagnation
        if it % jitter_interval == 0:
            y += rng.normal(scale=0.01, size=n)

    return best_seq, best_score


def _local_refine_sym(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Symmetric stochastic hill‑climbing with simulated annealing.
    Operates on the simplex (Σa = √(2 n)) and repeatedly transfers a random
    fraction of mass between symmetric index pairs.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 500_000
    step = 0
    half = n // 2
    middle = n // 2 if n % 2 == 1 else None

    while time.time() < deadline and step < max_steps:
        # pick source and destination from the first half (symmetry will handle the rest)
        i = rng.integers(0, half)
        j = rng.integers(0, half)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # random fraction of the source mass to move
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()

        # mirrored indices
        i_m = n - 1 - i
        j_m = n - 1 - j

        # remove mass from source (and its mirror)
        if i != i_m:
            new_seq[i] -= delta
            new_seq[i_m] -= delta
        else:  # source is the centre element (odd length)
            new_seq[i] -= delta

        # add mass to destination (and its mirror)
        if j != j_m:
            new_seq[j] += delta
            new_seq[j_m] += delta
        else:
            new_seq[j] += delta

        # ensure non‑negativity (tiny numerical errors)
        new_seq = np.maximum(new_seq, 0.0)

        # recompute max‑convolution
        conv = np.fft.irfft(np.fft.rfft(new_seq, n=Nfft) *
                            np.fft.rfft(new_seq, n=Nfft), n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            if step % 100000 == 0:
                print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score

        step += 1

    return cur_seq, cur_score

# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load any previously discovered construction if available
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a promising length (start from the size of the previous best)
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 1800                 # a reasonable default

    # Explore a few nearby lengths; larger sizes tend to give slightly lower peaks
    length_factors = [0.9, 1.0, 1.1, 1.2, 1.4]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in length_factors})

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- generate diverse symmetric starting points -----------------
        init_vectors = _make_initial_vectors(n, target_sum, rng)

        if base is not None and base.shape[0] != n:
            # linearly interpolate the previous best to the new size
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(_symmetrize(interp))

        # ---- per‑start optimisation ------------------------------------
        # iteration budget – scaled inversely with length
        max_iters = int(2_400_000 * n0 / n)
        max_iters = max(150_000, max_iters)

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            # Adam‑based exponentiated‑gradient optimisation
            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

            # Projected Gradient Descent (independent)
            best_seq, best_score = _pgd_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Local stochastic refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # give at most half the remaining budget to refinement (capped at 30 s)
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine_sym(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback if everything failed (unlikely)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
