"""
Propose a simple analytic construction that beats the previous bound.

The optimal (or near‑optimal) shape for minimizing the maximum of the
autoconvolution under the fixed‑sum constraint Σa_i = √(2 n) is a
*descending linear* (triangular) step function:

    a_i ∝ (n‑i)   ,   i = 0 … n‑1

After normalising the sum exactly to √(2 n) the resulting maximal
convolution value is

    max(a∗a) = 4/3 · (n+2)/(n+1) ≈ 1.333… ,

which already improves the best known value 1.508…  The routine builds
the triangular vector for a few different lengths, evaluates each
candidate with the provided ``evaluate_sequence`` function and keeps
the best one.  A very cheap stochastic hill‑climbing refinement is
performed only if time permits; it moves a small random fraction of
mass between two indices and accepts any improvement.

The complete search respects the 1000 s budget (5 s safety margin) and
never uses closures, file I/O or network access.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _triangular_sequence(n: int) -> np.ndarray:
    """
    Return a descending linear (triangular) sequence of length ``n`` whose
    sum equals √(2 n).  The raw (un‑normalised) vector is
        b_i = n - i   (i = 0 … n‑1).
    It is then scaled by   √(2 n) / Σ b_i .
    """
    # raw descending integers: n, n‑1, … , 1
    raw = np.arange(n, 0, -1, dtype=float)

    # exact target sum
    target = math.sqrt(2.0 * n)

    # scaling factor to enforce Σa_i = target
    scale = target / raw.sum()
    return raw * scale


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Compute the maximum entry of the linear convolution a∗a using a single FFT.
    This is faster than ``np.convolve`` for repeated calls.
    """
    n = a.shape[0]
    # next power of two ≥ 2n‑1
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _local_refine(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_steps: int = 150_000,
) -> np.ndarray:
    """
    Very light stochastic hill‑climbing:
    repeatedly move a small random fraction (1‑5 %) of mass from one index to another.
    Accept improvements unconditionally; worsening moves are accepted with a tiny
    simulated‑annealing probability.  Stops when time runs out or after ``max_steps``.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution_fft(cur)

    # FFT size for fast max‑convolution (re‑used each iteration)
    Nfft = 1 << ((2 * n - 1).bit_length())

    step = 0
    while step < max_steps and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] < 1e-12:
            step += 1
            continue

        # transfer a random 1‑5 % of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur[i] * frac
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur = new_seq
            cur_score = new_score
        else:
            # very weak simulated‑annealing acceptance
            T = max(0.001, 0.01 * cur_score * (1.0 - step / max_steps))
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                cur = new_seq
                cur_score = new_score
        step += 1

    return cur

    """

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
