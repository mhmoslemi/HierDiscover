"""
Propose a non‑negative step‑function minimizing

        F(a) = 2·n·max(a∗a) / (∑a)² .

Strategy
---------

1. **Small‑scale exploration** (n≈2048)  
   Build a diversified pool of trial vectors (uniform, power‑law,
   double‑exponential, Gaussian, sinusoid, random Dirichlet).  Keep the
   best one (according to an exact FFT‑based objective).

2. **Linear‑programming mixing**  
   For the current distribution `f` compute `M = max(f∗f)`.  Solve the LP

        max   ∑ g_j
        s.t.  g_j ≥ 0,
              (f ∗ g)_k ≤ M   for all k = 0,…,2n‑2.

   This pushes mass to positions where the convolution is slack.
   Normalise the solution `g`, then mix
        f ← (1‑t)·f + t·g   (t slowly decays).
   Repeat a handful of times.

3. **Upsampling & high‑precision refinement**  
   Interpolate the refined small‑scale distribution to a larger grid
   (n=4096) and run a two‑phase optimiser:
   * Adam on a smooth‑max surrogate of the max‑convolution.
   * Exact sub‑gradient mirror‑descent on the true max.

4. **Light‑weight local perturbations** (optional) – random mass swaps
   accepted only if they improve the objective.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence encountered as a plain Python list.  All
numbers are forced to be non‑negative and ≤ 1000, matching the
evaluator’s constraints.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate followed by an exact sub‑gradient
    mirror‑descent (identical to the reference implementation, with a
    few more iterations and a slightly slower temperature schedule).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.2          # start temperature (more smoothing)
    lam_factor = 1.008

    last_improve = time.time()
    it = 0

    # --------------------------- Adam phase ---------------------------
    while time.time() < deadline:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (useful for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase the temperature (approach the true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random “shake” when we have been stagnant
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 800_000:          # generous safety bound
            break

    # -------------------- Sub‑gradient phase (exact max) --------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(60_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active‑set mask (all indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:   # stop if no recent improvement
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of start vectors on a grid of size *n*."""
    pool = []

    # warm‑start and cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)          # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # random Dirichlet draws with a range of concentration parameters
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def _lp_maximize_sum_under_conv(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max ∑g_j   s.t.   g ≥ 0,  (f∗g)_k ≤ M  (linear convolution).
    Returns the optimal vector `g`.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A where A @ g = f∗g (linear conv)
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n)                     # maximise sum(g)  → minimise -sum(g)
    bounds = [(0.0, None)] * n
    b_ub = np.full(rows, M)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects the supplied time‑budget and returns the
    best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # -------------------------------------------------
    # Stage 1 – small support (n≈2048) + LP mixing
    # -------------------------------------------------
    n_small = 2048
    conv_fft_len_small = _next_pow2(2 * n_small - 1)

    # Warm start from a globally supplied construction if it exists
    warm_small = None
    w_glob = globals().get("height_sequence_1")
    if w_glob is not None:

        w_arr = np.maximum(w_arr, 0.0)
        if w_arr.sum() >= 0.01:
            warm_small = w_arr / w_arr.sum()

    # Build and scan a rich candidate pool
    candidates = _generate_candidate_pool(n_small, warm_small)

    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        a = a / a.sum()
        val = _objective(a, n_small, conv_fft_len_small)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # Progress print – can be silenced if desired
        print(f"[pool {idx:03d}] obj={val:.6f}")

    # ------------------- LP mixing loop (few iterations) -------------------
    t_mix = 0.3                     # initial mixing weight
    max_lp_iters = 20

    # -------------------------------------------------

    # -------------------------------------------------
