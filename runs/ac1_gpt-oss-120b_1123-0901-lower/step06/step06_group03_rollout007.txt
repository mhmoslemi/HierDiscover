"""
Propose a step‑function minimizing

        F(a) = 2·n·max(a∗a) / (∑a)²

using a multi‑stage meta‑heuristic:

1. **Diverse initial pool** – uniform, triangular, power‑law,
   double‑exponential, centred / off‑centre Gaussians, sinusoid,
   random Dirichlet samples and (if present) the warm‑start
   `height_sequence_1`.  All vectors are normalised to unit L1‑norm.

2. **Deep refinement** – each of the best few seeds is optimised with
   an Adam optimiser that minimises a log‑sum‑exp smooth‑max surrogate
   of the convolution, followed by an exact sub‑gradient phase on the
   true max.  The optimiser respects the simplex constraint by working
   on log‑softmax logits.  A small “shake’’ perturbation is applied if
   the search stagnates.  The routine is time‑aware and updates a
   global best across all runs.

3. **Local stochastic search** – after gradient phases a cheap random
   walk on the simplex (Gaussian perturbations + renormalisation) is
   performed until the deadline.  Any improvement updates the global
   best.

4. **Optional up‑sampling** – if time permits the best shape is
   interpolated to a finer grid (8192 points) and the whole optimisation
   is repeated, which usually yields a lower bound.

All numbers are forced to be non‑negative and ≤1000 as required by the
evaluator.  The function returns the best sequence found as a plain
Python list.
"""

import time

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    Returns the real part (numpy default type float64).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Both arguments are real; the result has length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective value (the evaluator uses the same formula).
    The vector a is assumed to be non‑negative and sum(a)>0.
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of start vectors.
    The pool contains symmetric and asymmetric shapes, covering many
    parametric families.
    """
    pool: list[np.ndarray] = []

    # ---- Warm start and simple rotations --------------------------------
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ---- Uniform baseline ------------------------------------------------
    pool.append(np.full(n, 1.0 / n))

    # ---- Triangular (linear) shape --------------------------------------
    tri = np.arange(1, n // 2 + 1)
    if n % 2 == 0:
        tri_full = np.concatenate([tri, tri[::-1]])
    else:
        tri_full = np.concatenate([tri, [n // 2 + 1], tri[::-1]])
    tri_full = np.maximum(tri_full, 0.0)
    tri_full /= tri_full.sum()
    pool.append(tri_full)

    # ---- Power‑law edge‑concentrated shapes -----------------------------
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)            # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # ---- Double‑exponential (two‑sided Laplace) families -----------------
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # ---- Gaussian bumps (centred) ---------------------------------------
    for sigma in np.linspace(n * 0.03, n * 0.4, 8):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # ---- Off‑centre Gaussians (edge bumps) ------------------------------
    for mu_factor in (0.2, 0.35, 0.65, 0.8):
        mu = mu_factor * n
        for sigma in (n * 0.05, n * 0.12):
            shape = np.exp(-0.5 * ((i - mu) ** 2) / (sigma ** 2))
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            pool.append(shape)

    # ---- Single half‑wave sinusoid ---------------------------------------
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # ---- Random Dirichlet draws with varied concentration -----------------
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    # ---- Sparse random spikes smoothed by a short Gaussian -------------
    for _ in range(6):
        k = max(3, n // 200)               # a few non‑zero entries
        indices = np.random.choice(n, size=k, replace=False)
        spikes = np.zeros(n)
        spikes[indices] = np.random.rand(k) + 0.1
        # smooth a bit
        sigma = max(1, n // 200)
        x = np.arange(-3 * sigma, 3 * sigma + 1)
        kernel = np.exp(-0.5 * (x / sigma) ** 2)
        kernel /= kernel.sum()
        smooth = np.convolve(spikes, kernel, mode='same')
        smooth = np.maximum(smooth, 0.0)
        smooth /= smooth.sum()
        pool.append(smooth)

    return pool

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    symmetrize: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by an
    exponentiated‑gradient sub‑gradient phase.
    Optionally forces symmetry (a == a[::-1]).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # work in log‑softmax space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # Hyper‑parameters (tuned for speed / stability)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.2          # start temperature
    lam_factor = 1.025

    last_improve = time.time()
    it = 0
    max_it = 800_000   # safety bound; loop also respects deadline

    # --------------------------- Adam phase ---------------------------
    while time.time() < deadline and it < max_it:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        if symmetrize:
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()

        # exact objective for logging
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # classic Adam: compute gradient w.r.t logits via soft‑max Jacobian
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach the true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random shake when we have been stagnant
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    # -------------------- Sub‑gradient phase (exact max) --------------------
    # Start from best sequence found during Adam (or from a0 if none)
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    if symmetrize:
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

    eta0 = 0.6
    sub_last_improve = time.time()
    max_sub_it = 30_000
    for sub_it in range(max_sub_it):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # identify active constraints (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (exponentiated‑gradient) update
        eta = eta0 / np.sqrt(sub_it + 1)
        a = a * np.exp(-eta * grad_a)

        # renormalise to the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if symmetrize:
            a = (a + a[::-1]) / 2.0
            a /= a.sum()

        if time.time() - sub_last_improve > 7.0:
            break

    return global_best_seq, global_best_val

def _local_search(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    current_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search on the simplex.
    Perturbs the current best vector with Gaussian noise, re‑normalises,
    and accepts moves that strictly improve the objective.
    """
    best_seq = a.copy()
    best_val = current_best_val
    amp = 0.02                     # initial perturbation amplitude

    while time.time() < deadline:
        # generate a Gaussian perturbation and project back to the simplex
        delta = np.random.normal(scale=amp, size=n)
        a_new = best_seq + delta
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0.0:
            continue
        a_new /= total

        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a_new
            amp = max(amp * 0.98, 1e-7)   # gradually shrink step‑size
            print(f"[local] new best {val:.6f} amp={amp:.2e}")
        else:
            # occasional modest increase to escape plateaus
            if np.random.rand() < 0.001:
                amp = min(amp * 1.05, 0.2)

    return best_seq, best_val

        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # 1. Build a diverse candidate pool
    candidates = _generate_candidate_pool(n, w)

    # 2. Score them quickly and keep the top few
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored: list[tuple[float, np.ndarray]] = []

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # Keep the most promising starts (up to eight)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]

    # 3. Deep optimisation passes (with and without symmetry)
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation (non‑sym)")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val, symmetrize=False
        )
        # also try a symmetric version of the same seed
        if time.time() < deadline:
            print(f"[optim start {cid}] launching deep optimisation (sym)")
            best_seq, best_val = _deep_optimize(
                a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val, symmetrize=True
            )

    # 4. Optional up‑sampling to a finer grid if time permits ----------
    if time.time() + 12 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # interpolate linearly
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[big] starting optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val, symmetrize=False
        )

        # a final stochastic refinement on the big grid
        if time.time() < deadline:
            print("[big] stochastic local search")
            best_seq, best_val = _local_search(best_seq, n_big,
                                               conv_fft_len_big,
                                               deadline, best_val)
    else:
        # If we stay on the original grid, still apply local search
        if best_seq is not None and time.time() < deadline:
            print("[local] stochastic fine‑tuning")
            best_seq, best_val = _local_search(best_seq, n,
                                               conv_fft_len,
                                               deadline, best_val)

    # ------------------------------------------------------------------
    # Final safety: ensure we have a sequence and clip to allowed range
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Optional sanity check with the official evaluator (cheap, no side‑effects)
    # print("Final evaluated bound:", evaluate_sequence(best_seq.tolist()))

    return best_seq.tolist()
