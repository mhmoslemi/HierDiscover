Pseudo:
```
def anneal_refine(a, max_iters, deadline):
   best_score = compute_max_conv(a)
   cur = a.copy()
   cur_score = best_score
   for step in range(max_iters):
        if time.time() >= deadline: break
        # random start index i with some mass > 0
        i = rng.integers(n)
        if cur[i] <= 0: continue
        # choose j (different)
        j = rng.integers(n)
        if i == j: continue
        # delta = fraction*cur[i] (or random small amount)
        delta = cur[i] * rng.uniform(0.02, 0.2) maybe bigger to explore.
        # move delta from i to j
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # compute new max conv
        new_score = conv_max(new)
        # Accept if improved or with prob
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_seq = cur.copy()
                best_score = new_score
        else:
            T = initial_T * (1 - step/max_iters)
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                cur = new
                cur_score = new_score
   return best_seq, best_score
```

def _pgd_optimize(a0, n, deadline, max_iters, best_seq, best_score, rng):
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    a = a * target_sum / a.sum()
    Nfft = 1 << ((3 * n - 2).bit_length())
    step = 0.02 * target_sum  # maybe bigger
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # conv
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        # record best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n} it={it}] new best M={M:.6f}")
        # active mask
        eps = 1e-12
        mask = conv >= M - eps
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)
        # compute weighted reverse conv (gradient)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[:len(w)+n-1]
        idx_start = 2*n-2
        rev = conv_rev[idx_start - np.arange(n)]
        grad = 2.0 * rev
        # step size schedule
        lr = step / math.sqrt(it)
        a_tmp = a - lr * grad
        a = _project_onto_simplex(a_tmp, target_sum)
        if it % 2000 ==0:
            jitter = rng.normal(scale=0.005*target_sum/n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)
    return best_seq, best_score

def _adam_optimize(a0, n, deadline, max_iters, best_seq, best_score, rng):
    target_sum = math.sqrt(2.0 * n)
    eps = 1e-16
    y = np.log(a0+eps)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9; beta2=0.999; epsadam=1e-8
    lr0 = 0.05
    Nfft = 1 << ((3*n-2).bit_length())
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft*A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        # keep best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n} it={it}] new best M={M:.6f}")
        # smoothed max weights:
        tau0 = 0.1 * M
        tau = tau0 * math.exp(-it/ (max_iters/5.0))
        if tau < 1e-12:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt,1)
        else:
            max_c = conv.max()
            ex = np.exp((conv - max_c)/tau)
            w = ex / ex.sum()
        # rev conv for grad wrt a
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[:len(w)+n-1]
        rev = conv_rev[2*n-2 - np.arange(n)]
        g_a = 2.0 * rev
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)
        m = beta1*m + (1-beta1)*g_y
        v = beta2*v + (1-beta2)*(g_y**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1 - beta2**it)
        lr = lr0/(1+1e-4*it)
        y = y - lr * m_hat / (np.sqrt(v_hat)+epsadam)
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y = y + jitter
    return best_seq, best_score

def _refine_anneal(a, n, deadline, best_score, rng):
    target_sum = a.sum()
    cur = a.copy()
    cur_score = best_score
    best_seq = cur.copy()
    best_s = cur_score
    Nfft = 1 << ((2*n-1).bit_length())
    # Determine number of steps based on time left (~ maybe 200k)
    step = 0
    max_steps = 500_000
    while time.time() < deadline and step < max_steps:
        # propose move
        i = rng.integers(n)
        if cur[i] <= 0:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # random fraction of source mass
        frac = rng.uniform(0.01, 0.2)  # maybe up to 20%
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # compute max conv
        A_fft = np.fft.rfft(new, n=Nfft)
        conv = np.fft.irfft(A_fft*A_fft, n=Nfft)[:2*n-1]
        new_score = conv.max()
        # acceptance
        if new_score < cur_score - 1e-12:
            cur = new
            cur_score = new_score
            if new_score < best_s:
                best_seq = new.copy()
                best_s = new_score
                print(f"[Refine] new best {best_s:.6f} step {step}")
        else:
            # temperature schedule
            T = max(0.01 * cur_score, 1e-5) * (1 - step / max_steps)
            if rng.random() < math.exp(-(new_score - cur_score)/ max(T, 1e-12)):
                cur = new
                cur_score = new_score
        step += 1
    return best_seq, best_s
```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search proceeds in three stages:

1. **Diverse initialisation** – a collection of analytically‑motivated vectors
   (uniform, exponential, windowed, sparse spikes, etc.) together with an
   interpolation of the previously‑found best sequence if it exists.

2. **Two complementary continuous optimisers** –  
   *Projected Gradient Descent* (sub‑gradient of the exact max‑convolution) and
   an *exponentiated‑gradient Adam* optimiser (smooth max via a soft‑max).  
   Both work on the simplex {a_i≥0, Σa_i = √(2 n)} and keep the globally best
   vector seen.

3. **Intensive stochastic refinement** – simulated‑annealing style mass
   transfers (move a random fraction of the mass from one index to another)
   that accept improvements always and occasional worsenings with a
   temperature schedule.  This phase often yields the final few‑percent gain
   required to beat the reference bound.

A global time budget (default 1000 s) is honoured: the algorithm constantly
checks the deadline and stops early if necessary, returning the best sequence
found so far.  All helper functions are top‑level and avoid closures.
"""

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x_i≥0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_max(a: np.ndarray) -> float:
    """Exact maximum of the linear convolution a*a using a single FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max())


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """Return a small, varied set of feasible start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    idx = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * idx / (n - 1)) + a2 * np.cos(
        4.0 * np.pi * idx / (n - 1)
    )
    init.append(black / black.sum() * target_sum)

    # 7) small Gaussian perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(
        scale=0.01 * target_sum / n, size=n
    )
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes (helps explore highly non‑uniform shapes)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    # 10) random sinusoidal modulation
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    sinus = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    init.append(sinus / sinus.sum() * target_sum)

    return init

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected Gradient Descent on the exact max‑convolution."""
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    a = a * target_sum / a.sum()  # ensure feasibility

    Nfft = 1 << ((3 * n - 2).bit_length())
    base_lr = 0.001 * target_sum  # base step (≈0.05 % of total mass)
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current max‑convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- record best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:7d}] new best = {M:.6f}")

        # ----- active set (hard max) -----
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ----- weighted reverse convolution → sub‑gradient -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev  # ∇_a max(a∗a)

        # ----- gradient step + projection -----
        lr = base_lr / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            jitter = rng.normal(
                scale=0.01 * target_sum / n, size=n
            )
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient Adam optimiser with a soft‑max approximation of max."""
    target_sum = math.sqrt(2.0 * n)
    eps = 1e-16
    y = np.log(a0 + eps)  # logits

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.07  # base learning rate for logits
    jitter_interval = 3000

    Nfft = 1 << ((3 * n - 2).bit_length())
    decay_rate = max_iters / 8.0
    tau_factor = 0.30  # τ₀ = τ_factor·M

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector from logits -----
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # ----- convolution and exact max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- record best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n:5d} it={it:7d}] new best = {M:.6f}")

        # ----- soft‑max weights for the smoothed max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()

        # ----- gradient w.r.t a (2·rev) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        rev = conv_rev[2 * n - 2 - np.arange(n)]
        g_a = 2.0 * rev

        # ----- chain rule back to logits (mirror‑descent on simplex) -----
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _annealing_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    cur_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing / simulated annealing.
    Randomly moves a fraction of mass from one index to another.
    Accepts improvements always; occasionally accepts worsenings with
    a temperature that linearly decays to zero.
    """
    target_sum = a.sum()
    cur = a.copy()
    best = cur.copy()
    best_val = cur_score

    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 300_000
    step = 0

    while time.time() < deadline and step < max_steps:
        i = rng.integers(0, n)
        if cur[i] <= 1e-15:
            step += 1
            continue
        j = rng.integers(0, n)
        if i == j:
            step += 1
            continue

        # move a random fraction (1 % – 20 %) of the source mass
        frac = rng.uniform(0.01, 0.20)
        delta = cur[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_val = float(conv.max())

        # deterministic improvement
        if new_val < cur_score - 1e-12:
            cur = new
            cur_score = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
                print(f"[Refine] new best = {best_val:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            if rng.random() < math.exp(-(new_val - cur_score) / max(T, 1e-12)):
                cur = new
                cur_score = new_val
        step += 1

    return best, best_val

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the surrounding harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Attempt to load a previously discovered construction
    base = None
    if "height_sequence_1" in globals():
        try:
            base = np.array(height_sequence_1, dtype=float)
        except Exception:
            base = None

    # ------------------------------------------------------------------
    # Choose a handful of promising lengths.
    # The historically good size is around 1900; we also probe a neighbourhood.
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 1900

    length_factors = [0.85, 0.95, 1.00, 1.10, 1.20]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in length_factors})

    global_best_seq = None
    global_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        print(f"\n=== Optimising length = {n} (target sum = {target_sum:.3f}) ===")

        # ------------------------------------------------------------------
        # Generate initial vectors
        init_vecs = _make_initial_vectors(n, target_sum, rng)

        # interpolate previous best if we have one of a different size
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vecs.append(interp)

        # ------------------------------------------------------------------
        # Determine a per‑start iteration budget (dynamic, but never too small)
        remaining = deadline - time.time()
        # we aim for roughly 0.3 s per 10 k iterations on a typical runner
        # -> allocate up to 300 k iters per optimiser if time permits
        max_iters = int(min(300_000, max(80_000, remaining * 200)))  # very rough heuristic

        for start_id, a0 in enumerate(init_vecs, 1):
            if time.time() >= deadline:
                break
            print(f"[Start {start_id}/{len(init_vecs)}]")

            # ---------- Adam (exponentiated‑gradient) ----------
            global_best_seq, global_best_score = _adam_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=global_best_seq if global_best_seq is not None else a0,
                best_score=global_best_score,
                rng=rng,
            )

            # ---------- Projected Gradient Descent ----------
            global_best_seq, global_best_score = _pgd_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=global_best_seq,
                best_score=global_best_score,
                rng=rng,
            )

        # ------------------------------------------------------------------
        # Local stochastic refinement on the current global best (if any)
        if global_best_seq is not None and time.time() < deadline:
            # allocate at most half the remaining time, but not more than 45 s
            refine_deadline = time.time() + min((deadline - time.time()) * 0.5, 45.0)
            global_best_seq, global_best_score = _annealing_refine(
                a=global_best_seq,
                n=global_best_seq.shape[0],
                deadline=refine_deadline,
                cur_score=global_best_score,
                rng=rng,
            )
            print(f"[After refinement for length {n}] best = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform sequence (should never be needed)
    if global_best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        global_best_seq = np.full(n, target_sum / n)
        global_best_score = _conv_max(global_best_seq)
        print("[Fallback] using uniform sequence")

    # ------------------------------------------------------------------
    # Final normalisation (guard against accumulated numeric drift)
    best_arr = np.array(global_best_seq, dtype=float)
    target = math.sqrt(2.0 * best_arr.shape[0])
    best_arr = best_arr * target / best_arr.sum()
    print(f"\n[Done] best objective = {global_best_score:.6f}, length = {best_arr.shape[0]}")
    return best_arr.tolist()
