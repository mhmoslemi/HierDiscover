"""
Propose a high‑quality step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The code builds a very diverse pool of starting shapes, runs a
smooth‑max Adam optimiser, refines it with an exponentiated‑gradient
sub‑gradient phase, performs a long fine‑tune, and finally applies a
cheap stochastic hill‑climb.  The search is time‑budget aware (default
≈1000 s) and repeatedly upsamples the current best solution to larger
lengths, allowing the expressive power of long vectors while keeping the
overall runtime well within the limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (fast integer version)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                 # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s

def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentration parameters)
    for conc in (1.0, 0.5, 0.2, 0.05):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes (negative exponents)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)                # i·(n‑i+1)
    for alpha in np.linspace(-4.0, -0.05, 14):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials with many scales
    i = np.arange(n, dtype=np.float64)
    for gamma in np.logspace(-5, -2, 12):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 12, n / 8, n / 5, n / 3):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 9. inverse‑triangular (edge‑heavy) shape
    inv_tri = 1.0 / (i + 1) + 1.0 / (n - i)
    inv_tri = np.maximum(inv_tri, 0)
    inv_tri /= inv_tri.sum()
    candidates.append(inv_tri)

    # 10. random convex combinations of two already generated shapes
    base = list(candidates)          # shallow copy
    for _ in range(6):
        a, b = np.random.choice(base, 2, replace=False)
        w = np.random.rand()
        comb = w * a + (1 - w) * b
        comb = np.maximum(comb, 0)
        s = comb.sum()
        if s > 1e-12:
            candidates.append(comb / s)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam(a0: np.ndarray, n: int, deadline: float,
              max_iters: int, lr: float = 0.05,
              lam_init: float = 0.30, lam_factor: float = 1.02):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence seen and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)       # keep us on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_init

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ----- map logits → point on the simplex (soft‑max) -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- exact objective (sum(a)=1 ⇒ denominator drops out) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # ----- smooth‑max weights (soft‑max on the convolution vector) -----
        w = np.exp(lam * (conv - max_c))     # shift by max for stability
        w_sum = w.sum()
        if w_sum != 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- back‑prop through the soft‑max (logits) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e4)

        # ----- occasional random shake if we are stuck -----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    Returns the best sequence encountered and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                     # initial step size (tuned empirically)

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where the convolution actually attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        # safeguard against collapse
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 250_000):
    """
    Very cheap stochastic hill‑climb that moves a small ε of mass from one
    index to another and keeps the move if the exact objective improves.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 6_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.02, best_a[i]))  # finer moves
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

    lengths, builds a rich pool of initial shapes (including up‑sampled
    best‑so‑far, historic solutions, Dirichlet draws, power‑law,
    exponentials, Hann, sine, Gaussian, inverse‑triangular and random
    mixes), runs an Adam optimiser with a smooth‑max surrogate,
    refines the result with an exponentiated‑gradient sub‑gradient phase,
    performs a longer fine‑tune on the current global best for the
    current length, and finally applies a stochastic hill‑climb.
    The best sequence seen overall is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths – we start small for quick gains, then move to larger n.
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budgets (scale roughly with 1/n, more generous for larger n)
        max_adam = int(30000 * 2048 / n) + 10_000
        max_subgrad = int(12000 * 2048 / n) + 4_000

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam (smooth‑max) ------------------------------------------------
            a_adam, val_adam = _run_adam(init, n, deadline,
                                         max_adam, lr=0.05,
                                         lam_init=0.30, lam_factor=1.02)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam new global best {val_adam:.6f}")

            # ---- Sub‑gradient (exponentiated‑gradient) ---------------------------
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad new global best {val_sub:.6f}")

            # ---- Extra fine‑tune on the *current* global best for this length -----
            if best_seq is not None and len(best_seq) == n:
                # a longer, low‑lr Adam run
                a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                             int(0.6 * max_adam),
                                             lr=0.02, lam_init=0.5, lam_factor=1.01)
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ----------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
