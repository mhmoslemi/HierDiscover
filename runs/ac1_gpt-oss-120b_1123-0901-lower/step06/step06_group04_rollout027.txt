def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0  # safety
    best_val = np.inf
    best_seq = None
    # lengths to try descending order for better quality early?
    lengths = [32768, 16384, 8192, 4096, 2048] # start with large length quicker?
    for n in lengths:
        if time.time() > deadline: break
        # generate initial seeds
        seeds = []
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # power law exponent grid
        for alpha in np.linspace(-2.8, -0.2, 13):
            shape = np.power(np.arange(1, n+1) * (n - np.arange(1, n+1) + 1), alpha)
            shape = shape / shape.sum()
            seeds.append(shape)
        # mixture probing random
        for _ in range(5):
            alpha = np.random.uniform(-3, 0)
            shape = np.power(np.arange(1, n+1) * (n - np.arange(1, n+1) + 1), alpha)
            shape = shape / shape.sum()
            lam = np.random.rand()
            shape = lam * np.full(n, 1.0/n) + (1-lam) * shape
            shape = shape / shape.sum()
            seeds.append(shape)
        # optionally add sine shape with exponent p
        for p in [0.5, 0.75, 1., 1.5, 2., 3.]:
            shape = np.sin(np.pi * (np.arange(n)+0.5)/n) ** p
            shape = shape / shape.sum()
            seeds.append(shape)
        # limit seeds
        max_seeds = 30 if n <= 8192 else 12
        seeds = seeds[:max_seeds]
        for seed_id, a0 in enumerate(seeds):
            if time.time() > deadline: break
            # condense to training steps
            a_best, val_best = _optimize_with_lp_and_adam(a0, n, deadline)
            if val_best < best_val:
                best_val = val_best
                best_seq = a_best.copy()
                print(f"[n={n} seed={seed_id}] new global best {best_val:.6f}")
        # after seeds we can jitter around best_seq if matches n
        if best_seq is not None and len(best_seq)==n:
            for jitter_id in range(5):
                if time.time()>deadline: break
                logits = np.log(np.maximum(best_seq,1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()
                a_best, val_best = _optimize_with_lp_and_adam(a0, n, deadline)
                if val_best < best_val:
                    best_val = val_best
                    best_seq = a_best.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")
    if best_seq is None:
        best_seq = np.full(2048,1.0/2048)
    return best_seq.tolist()

def _lp_step(f, M, epsilon=1e-12):
    import cvxpy as cp
    n = f.shape[0]
    # identify active indices
    # compute convolution using _full_convolution
    conv_f_f = _full_convolution(f, 1<<int(np.ceil(np.log2(2*n-1))))
    active_idx = np.where(conv_f_f >= M - epsilon)[0]
    if active_idx.size == 0:
        active_idx = np.array([np.argmax(conv_f_f)], dtype=int)
    # Build constraints
    A = []
    for i in active_idx:
        # Create a row vector length n
        coeff = np.zeros(n)
        j_min = max(0, i - (n-1))
        j_max = min(n-1, i)
        # assign f[j] to coeff[i-j]
        for j in range(j_min, j_max+1):
            coeff[i-j] = f[j]
        A.append(coeff)
    A = np.array(A)  # m x n
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500, abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    if g.value is None:
        return None
    g_vec = np.asarray(g.value).flatten()
    sumg = g_vec.sum()
    if sumg <=0:
        return None
    return g_vec / sumg

"""
Propose a step‑function that minimises

    F(a) = 2*n*max(a*a) / (sum(a)**2)

using a combination of:

* a fast FFT based convolution,
* Adam optimisation on a smooth‑max surrogate,
* exact ℓ∞ sub‑gradient refinement,
* a cheap linear‑program “direction” step (cvxpy) that pushes the
  current iterate towards a vector that respects the current maximal
  convolution bound,
* an extensive parametric grid (power‑law / uniform mixtures) to
  initialise the search.

The routine respects a wall‑clock budget (default 1000 s) and returns
the best vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional LP support (cvxpy).  If cvxpy is unavailable we simply skip
# the LP step – the rest of the optimiser works without it.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False


# ----------------------------------------------------------------------
# FFT based convolution utilities (they are the bottleneck of the
# optimiser, therefore we use power‑of‑two lengths for speed)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate of max(a*a).
    Returns the best vector found and its exact objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.07                     # a slightly larger step size than before

    lam = 1.0                     # smooth‑max temperature
    lam_factor = 1.03

    # iteration budget – a larger constant than in the reference code
    max_iters = int(80000 * 1024 / n) + 3000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max (ensures a stays on the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal the temperature
        lam = min(lam * lam_factor, 1e9)

        # exact objective (sum(a)=1 → value = 2*n*max_c)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional random jitter if we stagnate
        if time.time() - last_improve > 3.5:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Refines a vector using the true ℓ∞ sub‑gradient of the convolution
    maximum.  Returns the best vector seen and its exact objective.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(6000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        # uniform weighting over the active set
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                      # fallback to a random simplex point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                       # slowly decay step size

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# A tiny LP step that produces a feasible direction (only if cvxpy is
# present).  The LP: maximise Σ g_j  s.t.  f*g ≤ M  (element‑wise) and
# g ≥ 0.  Only the almost‑active convolution indices are kept as
# constraints, making the problem tiny.
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, eps: float = 1e-12) -> np.ndarray | None:
    """Return a normalized vector g (sum=1) solving the LP described
    above, or None if the LP fails or cvxpy is unavailable."""
    if not _HAS_CVXPY:
        return None

    n = f.shape[0]
    conv_ff = _full_convolution(f, fft_len=1 << int(np.ceil(np.log2(2 * n - 1))))
    active_idx = np.where(conv_ff >= M - eps)[0]
    if active_idx.size == 0:                     # fallback to the global maximum index
        active_idx = np.array([np.argmax(conv_ff)])

    # Build a small constraint matrix A where A[i,:] are the coefficients of
    # (f * g)[active_idx[i]].
    rows = []
    for i in active_idx:
        coeff = np.zeros(n)
        j_min = max(0, i - (n - 1))
        j_max = min(n - 1, i)
        # for each admissible j we have g[i-j] with weight f[j]
        for j in range(j_min, j_max + 1):
            coeff[i - j] = f[j]
        rows.append(coeff)
    A = np.vstack(rows)          # shape (m_active, n)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=500, abstol=1e-9, reltol=1e-9, feastol=1e-9,
                    verbose=False)
    except Exception:           # pragma: no cover
        return None

    if g.value is None:
        return None

    g_vec = np.asarray(g.value).flatten()
    s = g_vec.sum()
    if s <= 0:
        return None
    return g_vec / s


# ----------------------------------------------------------------------
# Combined optimisation pipeline: ADAM → sub‑gradient → LP → mixture.
# ----------------------------------------------------------------------
def _optimise_candidate(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        corr_fft_len: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a few rounds of Adam + exact sub‑gradient refinement,
    optionally followed by an LP‑based mixing step.
    Returns the best vector obtained and its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    # outer loop – limited by remaining time
    while time.time() < deadline - 0.2:
        # ----------- Adam phase ----------
        a_adam, val_adam = _adam_optimize(a, n, conv_fft_len,
                                          corr_fft_len, deadline)
        # ----------- exact sub‑gradient ----------
        a_ref, val_ref = _subgradient_refine(a_adam, n,
                                             conv_fft_len, corr_fft_len,
                                             deadline)

        # pick the better of the two
        if val_ref < best_val:
            best_val, best_a = val_ref, a_ref.copy()
        elif val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()

        a = best_a.copy()

        # -------------- LP step (if possible) --------------
        conv_ff = _full_convolution(a, conv_fft_len)
        M = conv_ff.max()
        g = _lp_step(a, M)
        if g is None:
            break          # fall back – LP not usable or failed

        # mix the current point towards the LP direction
        t = 0.18          # modest step size
        a = (1 - t) * a + t * g
        a = a / a.sum()

        # evaluate after mixing
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val, best_a = val, a.copy()
        else:
            # if mixing does not help, stop early
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – the function that the evaluation harness will call.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The routine respects a wall‑clock budget
    (default 1000 s) and returns the best sequence discovered as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    best_seq = None
    best_val = float("inf")

    # Work on increasingly large lengths – large n gives the best
    # asymptotic bounds but is also the slowest, so we start with the
    # biggest size we can afford.
    lengths = [32768, 16384, 8192, 4096, 2048]

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (no np.convolve – we use our
    # fast FFT version).  The sum of the vector is kept equal to 1,
    # therefore the objective reduces to 2·n·max(conv).
    # ------------------------------------------------------------------
    def _objective(a: np.ndarray, n_: int, fft_len: int) -> float:
        conv = _full_convolution(a, fft_len)
        return 2.0 * n_ * conv.max()

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n*** Optimising length n={n} ***")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # ------------------- Build a diverse pool of seeds -------------
        seeds = []

        # 1) uniform
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) previous global best (if of same size)
        if best_seq is not None and len(best_seq) == n:
            seeds.append(np.asarray(best_seq, dtype=np.float64))

        # 3) up/down sampled previous best
        if best_seq is not None and len(best_seq) != n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            if up.sum() > 0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            seeds.append(up)

        # 4) power‑law family + optional mixing with uniform
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)          # (i+1)*(n-i)
        alphas = np.linspace(-2.8, -0.2, 25)   # fine grid of exponents
        lams = np.linspace(0.0, 0.6, 7)       # mixing coefficient with uniform
        for alpha in alphas:
            shape = np.exp(alpha * np.log(prod))
            shape_sum = shape.sum()
            if shape_sum == 0:
                continue
            shape /= shape_sum
            for lam in lams:
                seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * shape
                seq = seq / seq.sum()
                seeds.append(seq)

        # 5) a few random Dirichlet draws (different concentration)
        for conc in [0.2, 0.5, 1.0, 2.0]:
            seeds.append(np.random.dirichlet(np.full(n, conc)))

        # 6) smooth sine‑squared bumps with an exponent
        for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
            shape = shape / shape.sum()
            seeds.append(shape)

        # Limit the total number of seeds to keep the overall runtime
        max_seeds = 30 if n <= 8192 else 14
        seeds = seeds[:max_seeds]

        # ------------------- Evaluate each seed -----------------------
        # We first compute a cheap objective to discard clearly bad seeds.
        # The top‑k promising ones (by this cheap metric) are then sent
        # through the full optimisation pipeline.
        cheap_vals = []
        for i, a in enumerate(seeds):
            if time.time() > deadline:
                break
            val = _objective(a, n, conv_fft_len)
            cheap_vals.append((val, i, a))

        cheap_vals.sort(key=lambda x: x[0])        # lower is better
        top_k = min(6, len(cheap_vals))            # only a few go deeper

        for rank in range(top_k):
            if time.time() > deadline:
                break
            _, sid, cand = cheap_vals[rank]
            print(f"[n={n} seed={sid} (rank {rank})] cheap value {_: .6f}")

            a_opt, val_opt = _optimise_candidate(cand, n,
                                                 conv_fft_len, corr_fft_len,
                                                 deadline)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} seed={sid}] ** new global best {best_val:.6f}")

        # ------------------- Jitter around the current best ------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(8):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.04, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_opt, val_opt = _optimise_candidate(a0, n,
                                                     conv_fft_len, corr_fft_len,
                                                     deadline)
                if val_opt < best_val:
                    best_val = val_opt
                    best_seq = a_opt.copy()
                    print(f"[n={n} jitter={jitter_id}] ** new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing succeeded (should never happen).
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\nFinished search – best value achieved: {best_val:.6f}")
    return best_seq.tolist()
