"""
Optimisation of a non‑negative step‑function for the objective

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm works in several stages:

1. **Multi‑resolution warm‑start** – a sequence found for a smaller
   length is interpolated to serve as a very good initial guess for the
   next larger length.  This exploits the fact that the optimal shape
   changes only slowly when the number of points grows.

2. **Adam on a smooth‑max surrogate** – the true max is approximated by
   a soft‑max with an exponentially increasing temperature.  Optimising
   the surrogate in the log‑simplex (softmax of logits) guarantees
   non‑negativity and unit L¹‑norm.

3. **Exact‑max exponentiated‑gradient (EG) phase** – after the surrogate
   has been tightened we switch to a sub‑gradient method that works
   directly on the exact max.  EG respects the simplex constraint by
   multiplicative updates.

4. **Stochastic perturbations & early‑stop** – when progress stalls a
   small random jitter is added to escape plateaus.  The search stops
   as soon as the target bound (≈1.503) is reached or the time budget
   expires.

All heavy convolutions are performed with FFTs (O(n log n)).  The
function respects the supplied time budget and returns the best
sequence discovered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution and cross‑correlation (used repeatedly)
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _smooth_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights for the smooth‑max surrogate."""
    mx = conv.max()
    w = np.exp(lam * (conv - mx))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _exact_max_weights(conv: np.ndarray, atol: float = 1e-12) -> np.ndarray:
    """Uniform weights on the (approximately) active max positions."""
    mx = conv.max()
    active = np.isclose(conv, mx, atol=atol)
    if not active.any():
        return np.zeros_like(conv)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.sum()
    return w

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin
    best_seq = None
    best_val = float("inf")

    # Lengths explored – start small, enlarge while time permits.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths needed for each n.
    conv_fft_len = {}
    corr_fft_len = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft_len[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # cross‑correlation needs conv_len + n - 1 points
        corr_fft_len[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # Helper to generate a deterministic “arcsine’’ shape (often very good)
    def arcsine_shape(n: int) -> np.ndarray:
        i = np.arange(n, dtype=np.float64)
        arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
        arcs = np.maximum(arcs, 0.0)
        arcs /= arcs.sum()
        return arcs

    # keep the best sequence from the previous size to interpolate upwards
    prev_best = None

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        conv_len = conv_fft_len[n]
        corr_len = corr_fft_len[n]

        # --------------------------------------------------------------
        # Build a small, diverse pool of start points for this length.
        candidates = []

        # 1) Interpolated previous best (if available)
        if prev_best is not None:
            m = prev_best.shape[0]
            x_old = np.linspace(0.0, 1.0, m)
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, prev_best)
            up = np.maximum(up, 0.0)
            if up.sum() < 1e-12:
                up = np.random.dirichlet(np.ones(n))
            else:
                up = up / up.sum()
            candidates.append(up)

        # 2) Pure arcsine (a strong deterministic baseline)
        candidates.append(arcsine_shape(n))

        # 3) Random Dirichlet vectors (different concentration parameters)
        candidates.append(np.random.dirichlet(np.ones(n)))          # uniform
        for conc in (0.3, 0.6, 1.2, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

        # 4) Two‑sided exponential decay (edge‑focused)
        i = np.arange(n, dtype=np.float64)
        for gamma in (0.001, 0.003, 0.006):
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 5) Power‑law on distance‑to‑edge product (spiky at borders)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in (-1.5, -1.0, -0.5):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 6) Random sparse pattern (≈ √n spikes)
        m_spikes = max(2, int(np.sqrt(2 * n)))
        sparse = np.zeros(n, dtype=np.float64)
        idxs = np.random.choice(n, size=m_spikes, replace=False)
        sparse[idxs] = np.random.rand(m_spikes)
        if sparse.sum() > 0:
            sparse = sparse / sparse.sum()
            candidates.append(sparse)

        # --------------------------------------------------------------
        # Optimise each candidate.
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()
            logits = np.log(a0 + 1e-15)      # work in log‑simplex

            # ---- Adam on smooth‑max surrogate ---------------------------------
            m_adam = np.zeros_like(logits)
            v_adam = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            # learning‑rate scales mildly with problem size
            lr = 0.07 * np.sqrt(2048.0 / n)
            lam = 0.1                      # initial temperature
            lam_factor = 1.012

            # iteration budget – more for smaller n, still bounded by time
            max_iters = int(30000 * 1024 / n) + 5000
            last_improve = time.time()

            for t in range(1, max_iters + 1):
                # soft‑max → a (lies on the simplex)
                mx_log = logits.max()
                exp_log = np.exp(logits - mx_log)
                a = exp_log / exp_log.sum()

                # convolution & smooth‑max weights
                conv = _conv_fft(a, conv_len)
                w = _smooth_max_weights(conv, lam)

                # gradient wrt a (sub‑gradient of the smooth‑max)
                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m_adam = beta1 * m_adam + (1.0 - beta1) * grad_log
                v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m_adam / (1.0 - beta1 ** t)
                v_hat = v_adam / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule – slowly tighten the surrogate
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a) == 1)
                val = 2.0 * n * conv.max()
                if val < best_val - 1e-9:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t}] new best {val:.6f}")

                # If stalled, add a tiny random jitter to logits
                if time.time() - last_improve > 6.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---- Exact‑max exponentiated‑gradient (sub‑gradient) phase -------
            a = a / a.sum()
            eta = 0.55                     # step size for EG
            sub_last = time.time()
            max_sub_iters = 8000

            for sub in range(max_sub_iters):
                conv = _conv_fft(a, conv_len)
                w = _exact_max_weights(conv)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # multiplicative EG update (mirror descent with KL‑divergence)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-14:                     # protect against collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluation
                conv = _conv_fft(a, conv_len)
                val = 2.0 * n * conv.max()
                if val < best_val - 1e-9:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cid} EG sub={sub}] new best {val:.6f}")

                # early stop if no progress for a while
                if time.time() - sub_last > 8.0:
                    break
                if time.time() > deadline:
                    break

        # Store the best sequence for the next (larger) length.
        if best_seq is not None:
            prev_best = best_seq.copy()
            # if we already passed the target bound, we can stop early
            if best_val <= 1.503:
                print(f"Target bound reached (val={best_val:.6f}); terminating.")
                break
        print(f"[n={n}] best value so far: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed).
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # expose the final sequence globally for possible warm‑starts in later runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    # Return as plain list for the external evaluator.
    return best_seq.tolist()
