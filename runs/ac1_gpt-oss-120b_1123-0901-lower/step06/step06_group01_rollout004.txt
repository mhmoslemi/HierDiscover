"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s)

1. **Length schedule** – start from the best known length (if any) and
   explore a few neighbours (± 10 %).  For each length ``n`` the target
   ℓ₁‑norm is fixed to √(2 n); the objective then equals the infinity‑norm
   of the auto‑convolution of the normalised probability vector.

2. **Diverse initialisations** – for a given ``n`` we generate a 
   portfolio of feasible starts (uniform, Dirichlet, smoothed noise,
   random spikes and an interpolated copy of the previous best).  

3. **Two complementary descent schemes**

   * *Projected Gradient Descent* (PGD) on the exact max‑convolution.
     The sub‑gradient is obtained from the active (max‑attaining) indices;
     a randomisation of the active‐set weights helps to escape plateaus.
     After each step the vector is projected on the simplex
     {x ≥ 0, Σx = √(2 n)}.

   * *Exponentiated‑gradient Adam* on a smoothed max (soft‑max with a 
     temperature that decays exponentially).  The variables are logits;
     the soft‑max guarantees feasibility automatically.

   Both optimisers run for a time‑adapted number of iterations and keep
   the best vector observed.

4. **Stochastic refinement** – a simulated‑annealing hill‑climber that
   repeatedly transfers a small random fraction of mass between two
   positions (occasionally moving larger fractions).  Improvements are
   always accepted; deteriorations are accepted with a temperature‑dependent
   probability.  The refinement runs as long as time permits.

5. **Return** the best sequence found (plain Python list).  Throughout the

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, Nfft: int) -> float:
    """Return max of linear convolution a*a using a single FFT."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _rev_weighted_convolution(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute the weighted reverse convolution needed for the sub‑gradient:

        rev_i = Σ_k w_k * a_{k-i}      (out‑of‑range indices ignored)

    The calculation uses FFTs (O(N log N)).
    """
    A = np.fft.rfft(a, n=Nfft)
    Wrev = np.fft.rfft(w[::-1], n=Nfft)      # reverse w before FFT
    rev_full = np.fft.irfft(Wrev * A, n=Nfft)
    # keep linear part (length = len(w)+len(a)-1)
    rev_full = rev_full[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = rev_full[idx_start - np.arange(a.shape[0])]
    return rev


def _make_initial_vectors(
    n: int,
    target_sum: float,
    rng: np.random.Generator,
    base: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of feasible start vectors (sum = target_sum)."""
    inits = []

    # 1) uniform
    inits.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1) i.i.d.)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    inits.append(g / g.sum() * target_sum)

    # 3) Random Gaussian‑smoothed noise (produces smooth, non‑uniform shapes)
    noise = rng.normal(size=n)
    sigma = rng.uniform(5.0, 30.0)            # width of Gaussian kernel
    # simple convolution with a Gaussian kernel using FFT
    ksize = int(4 * sigma) | 1
    x = np.arange(ksize) - ksize // 2
    gauss = np.exp(-0.5 * (x / sigma) ** 2)
    gauss /= gauss.sum()
    Nfft = 1 << ((n + ksize - 1).bit_length())
    G = np.fft.rfft(gauss, n=Nfft)
    N = np.fft.rfft(noise, n=Nfft)
    smooth = np.fft.irfft(G * N, n=Nfft)[:n]
    smooth = np.maximum(smooth, 0.0)
    if smooth.sum() > 0:
        inits.append(smooth / smooth.sum() * target_sum)

    # 4) Exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.02)
    expo = np.exp(-rate * np.arange(n))
    inits.append(expo / expo.sum() * target_sum)

    # 5) Reversed exponential
    inits.append(expo[::-1] / expo.sum() * target_sum)

    # 6) Hann window (smooth bell‑shape)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    inits.append(hann / hann.sum() * target_sum)

    # 7) Sparse random spikes (helps explore highly non‑uniform shapes)
    k = max(2, min(20, n // 80))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    inits.append(sparse)

    # 8) Interpolated copy of the previous best (if provided)
    if base is not None and base.shape[0] != n:
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        interp = np.interp(new_x, old_x, base)
        interp = np.maximum(interp, 0.0)
        if interp.sum() > 0:
            interp = interp / interp.sum() * target_sum
            inits.append(interp)

    # 9) Small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    inits.append(pert / pert.sum() * target_sum)

    # Remove near‑duplicates (optional)
    uniq = []
    for v in inits:
        if not any(np.allclose(v, u, atol=1e-12) for u in uniq):
            uniq.append(v)
    return uniq

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the *exact* max‑convolution.
    Active‑set weights are randomly perturbed to avoid plateaux.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise on the simplex (safeguard against rounding noise)
    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.0015 * target_sum            # a slightly larger base step
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution & current max -----
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- update global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:4d} it={it:7d}] new best = {best_score:.6f}")

        # ----- active set (hard max) -----
        mask = conv >= M - eps_active
        idxs = np.nonzero(mask)[0]
        cnt = len(idxs)
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # random weighting over the active set (instead of uniform)
            rnd = rng.random(cnt)
            w = np.zeros_like(conv)
            w[idxs] = rnd / rnd.sum()    # normalise

        # ----- sub‑gradient (weighted reverse convolution) -----
        rev = _rev_weighted_convolution(a, w, Nfft)
        g = 2.0 * rev                     # gradient of max(a∗a)

        # ----- gradient step with diminishing step size -----
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- occasional jitter to break symmetry -----
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on logits with a *soft‑max* approximation of the max
    convolution.  Temperature decays exponentially; the learning rate is
    schedule‑adapted.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (add tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.12                     # a slightly larger base learning rate
    jitter_interval = 2500

    # temperature schedule parameters
    tau_factor = 0.35              # τ₀ = τ_factor·M
    decay_rate = max_iters / 6.0   # slower decay → smoother optimisation

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current probability vector on the simplex -----
        logits = y - np.max(y)                  # for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # ----- convolution & current max -----
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n:4d} it={it:7d}] new best = {best_score:.6f}")

        # ----- soft‑max weights for smoothed max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # fall back to hard max (uniform over active set)
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            # stable soft‑max
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            sum_ex = ex.sum()
            w = ex / sum_ex

        # ----- gradient wrt a (2·rev) -----
        rev = _rev_weighted_convolution(a, w, Nfft)
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent style) -----
        avg = np.dot(p, g_a)                # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # ----- occasional jitter (helps escape flat regions) -----
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.02, size=n)
            y += jitter

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the simplex.
    Moves: transfer a random fraction (1‑15 %) of mass from i to j.
    Accept improvements always; otherwise accept with probability exp(-Δ/T).
    """
    cur_seq = a.copy()
    cur_score = best_score
    target_sum = cur_seq.sum()

    Nfft = 1 << ((2 * n - 1).bit_length())

    # SA schedule
    max_steps = 400_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random relative amount
        frac = rng.uniform(0.01, 0.15)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max convolution
        A = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.02 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

        n0 = 1890    # the length that produced the best known bound

    # explore a neighbourhood; keep lengths between 600 and 5000
    length_factors = [0.85, 0.90, 0.95, 1.00, 1.05, 1.10, 1.15]
    lengths = sorted(
        {
            max(600, min(int(n0 * f), 5000))
            for f in length_factors
        }
    )

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        print(f"[run] length = {n}, target ℓ₁ = {target_sum:.3f}")

        # generate diverse start vectors (including interpolated copy of base)
        init_vectors = _make_initial_vectors(n, target_sum, rng, base)

        # iteration budget per start (scaled with n, never below 180 k)
        max_iters = int(3_000_000 * 2100 / n)
        max_iters = max(180_000, max_iters)

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx + 1}/{len(init_vectors)} for n={n}")

            # ------- Exponentiated‑gradient Adam -------
            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

            # ------- Projected Gradient Descent -------
            best_seq, best_score = _pgd_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
            )

    # --------------------------------------------------------------
    # Final stochastic refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.6, 45.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # --------------------------------------------------------------
    # Fallback (unlikely)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list (evaluate_sequence will handle any final scaling)
    return best_seq.tolist()
