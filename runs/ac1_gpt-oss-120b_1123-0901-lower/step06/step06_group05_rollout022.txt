def _run_adam_opt(a0, n, deadline, max_iters, lr=0.05, lam_start=0.1, lam_factor=1.02):
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    best_a = None
    best_val = np.inf
    lam = lam_start
    last_improve = time.time()
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        # compute a
        max_log = logits.max()
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()
        # compute conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] best {val:.6f}")
        # compute smooth max weights
        # lam may be large quickly; start small
        w_raw = np.exp(lam * (conv - max_c))
        w = w_raw / w_raw.sum()
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # update Adam
        m = 0.9 * m + 0.1 * grad_log
        v = 0.999 * v + 0.001 * (grad_log**2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # update lam
        lam = min(lam * lam_factor, 1e5)
        # random perturb if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
    if best_a is None:
        best_a = a0
        best_val = 2.0 * n * _full_convolution(best_a, _next_pow2(2*n-1)).max()
    return best_a, best_val

def _stochastic_hill_climb(a0, n, deadline):
    conv_fft_len = _next_pow2(2*n-1)
    best_a = a0.copy()
    best_val = 2.0*n*_full_convolution(best_a, conv_fft_len).max()
    T0 = 0.01
    decay = 0.9999
    T = T0
    t_start = time.time()
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0, min(best_a[i], 0.05))
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0*n*conv.max()
        delta = val - best_val
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_a = cand
            best_val = val
            # Optionally reset temperature after improvement?
        T *= decay
        if T < 1e-8:
            T = 1e-8
    return best_a, best_val

def _generate_initial_candidates(n, prev_best):
    candidates = []
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        candidates.append(_upsample(np.asarray(hseq, dtype=np.float64), n))
    # uniform Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))
    # various concentrations
    for conc in (0.2, 0.5, 0.8, 0.05, 0.1):
        candidates.append(np.random.dirichlet(np.full(n, conc)))
    # power-law edges
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)
    # symmetric exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005, 0.01):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)
    # Hann and sine windows
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)
    # Gaussians
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)
    # Random sparse mask
    for sparsity in (0.05, 0.1):
        mask = np.random.rand(n) < sparsity
        if mask.sum()>0:
            vals = np.random.rand(mask.sum())
            shape = np.zeros(n)
            shape[mask] = vals
            s = shape.sum()
            if s>0:
                shape /= s
                candidates.append(shape)
    # Normalise and discard pathological entries
    cleaned = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        cleaned.append(c / s)
    return cleaned

"""
Propose a high‑quality step‑function (a non‑negative vector on the simplex)
that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)²)

The search proceeds in three stages:

1. **Coarse optimisation** on a moderate size (4096).  A diverse pool of
   initial shapes (warm‑starts, Dirichlet draws, power‑law, exponentials,
   Hann, sine, Gaussians, very sparse masks) is refined by an Adam optimiser
   on a smooth‑max surrogate, followed by an exponentiated‑gradient
   sub‑gradient phase on the exact max.

2. **Multi‑scale up‑sampling** – the current best vector is interpolated to
   larger lengths (8192 and 16384) and again optimised with the same two‑phase
   routine (more iterations for the smaller scales, fewer for the largest).

3. **Final stochastic hill‑climb** (with occasional uphill moves) to squeeze
   out any remaining improvement.

All FFT‑based convolutions and cross‑correlations are performed with NumPy.
The routine respects a user‑supplied time budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """Create a heterogeneous pool of starting points for length ``n``."""
    candidates = []

    # warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # Dirichlet draws with various concentrations (including very sparse)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.05, 0.1, 0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # symmetric two‑sided exponentials (different decay rates)
    for gamma in (0.0005, 0.001, 0.002, 0.005, 0.01):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # centred Gaussian‑like bumps (several widths)
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # very sparse random masks (simulates a few large spikes)
    for sparsity in (0.02, 0.05, 0.1):
        mask = np.random.rand(n) < sparsity
        if mask.sum() == 0:
            continue
        vals = np.random.rand(mask.sum())
        shape = np.zeros(n)
        shape[mask] = vals
        s = shape.sum()
        if s > 0:
            shape /= s
            candidates.append(shape)

    # normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _adam_opt(a0: np.ndarray, n: int, deadline: float, max_iters: int,
              lr: float = 0.05, lam_start: float = 0.1,
              lam_factor: float = 1.02, lam_max: float = 1e5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best vector seen and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)               # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # map logits → point on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a) == 1 by construction)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights (soft‑max on the convolution vector)
        w = np.exp(lam * (conv - max_c))        # shift by max for stability
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (slowly sharpen the max)
        lam = min(lam * lam_factor, lam_max)

        # occasional random perturbation if we are stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgrad_opt(a0: np.ndarray, n: int, deadline: float, max_iters: int,
                 eta0: float = 0.55) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient method applied to the *exact* max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-13)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val

def _hill_climb(a0: np.ndarray, n: int, deadline: float,
                max_trials: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic local search.
    Moves a small mass ε from one index to another; accepts the move if it improves.
    Occasionally accepts uphill moves based on a slowly cooling temperature.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (best_a.sum() ** 2)

    T0 = 0.02
    decay = 0.9999
    T = T0
    trials = 0

    while time.time() < deadline and trials < max_trials:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue

        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        # sanity check (should never be negative because of eps bound)
        if cand[i] < 0:
            trials += 1
            continue

        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max() / (cand.sum() ** 2)
        delta = val - best_val

        # accept if improvement, otherwise with probability exp(-Δ/T)
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            best_a = cand
            best_val = val
            T = T0               # reset temperature after an improvement
        else:
            T *= decay           # otherwise cool down slowly

        trials += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Main entry point.
    Returns a list of non‑negative heights that (approximately) minimise the
    evaluation function.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # ---- Stage 1 : coarse optimisation at medium scale ----------------
    coarse_n = 4096
    if time.time() < deadline:
        print(f"\n=== Coarse optimisation (n={coarse_n}) ===")
        candidates = _generate_initial_candidates(coarse_n, best_seq)
        max_adam = int(35000 * 2048 / coarse_n) + 10_000   # ≈30 k iterations
        max_sub = 12_000

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            a_adam, v_adam = _adam_opt(init, coarse_n, deadline, max_adam)
            if v_adam < best_val:
                best_val, best_seq = v_adam, a_adam.copy()
                print(f"[n={coarse_n} cand={cid}] Adam new global best {v_adam:.6f}")

            a_sub, v_sub = _subgrad_opt(a_adam, coarse_n, deadline, max_sub)
            if v_sub < best_val:
                best_val, best_seq = v_sub, a_sub.copy()
                print(f"[n={coarse_n} cand={cid}] Subgrad new global best {v_sub:.6f}")

    # ---- Stage 2 : multi‑scale up‑sampling ---------------------------
    for n in (8192, 16384):
        if time.time() > deadline:
            break
        print(f"\n=== Upsampling optimisation (n={n}) ===")
        if best_seq is not None:
            init = _upsample(best_seq, n)
        else:
            init = np.random.dirichlet(np.ones(n))

        # moderate budget – fewer iterations as the dimension grows
        max_adam = max(18000, int(25000 * 2048 / n) + 4000)
        max_sub = max(8_000, int(12_000 * 2048 / n))

        a_adam, v_adam = _adam_opt(init, n, deadline, max_adam)
        if v_adam < best_val:
            best_val, best_seq = v_adam, a_adam.copy()
            print(f"[n={n}] Adam improved global best to {v_adam:.6f}")

        a_sub, v_sub = _subgrad_opt(a_adam, n, deadline, max_sub)
        if v_sub < best_val:
            best_val, best_seq = v_sub, a_sub.copy()
            print(f"[n={n}] Subgrad improved global best to {v_sub:.6f}")

        # short fine‑tune on the current global best for this length
        if best_seq is not None and len(best_seq) == n:
            a_fine, v_fine = _adam_opt(best_seq, n, deadline,
                                        int(0.5 * max_adam))
            if v_fine < best_val:
                best_val, best_seq = v_fine, a_fine.copy()
                print(f"[n={n}] Fine‑tune improved global best to {v_fine:.6f}")

    # ---- Stage 3 : final stochastic hill‑climb -----------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final hill‑climb (n={len(best_seq)}) ===")
        a_final, v_final = _hill_climb(best_seq, len(best_seq), deadline)
        if v_final < best_val:
            best_val, best_seq = v_final, a_final.copy()
            print(f"[final] hill‑climb improved global best to {v_final:.6f}")

    # ---- Fallback ----------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # return plain Python list as required by the external evaluator
    return best_seq.tolist()
