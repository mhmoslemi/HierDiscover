def conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n))
    for i in range(n):
        # column i: f[i] placed at rows i .. i + n - 1

    return A

"""
Hybrid LP‑refinement step‑function optimizer

The evaluator is scale invariant, so we keep the L1‑norm fixed to
√(2 n) and minimise max(conv(a,a)).  The algorithm works in three
stages:

1.  Wide‑range random parametric initialisations (exponential,
    power‑law, Hann/Hamming, Dirichlet).  The best among them is kept.
2.  Linear‑programming refinement: for the current sequence f we solve
    the LP  max Σ g_j  s.t.  (f∗g)_k ≤ M  (for all shifts) and g_j ≥ 0,
    where M = max(conv(f,f)).  The solution g is scaled to the required
    L1‑norm and mixed with f (convex combination).  A back‑tracking line
    search on the mixing factor guarantees a non‑increasing worst
    convolution value.
3.  A short simulated‑annealing hill‑climb on the exact objective to
    polish the final result.

All stages respect the global 1000 s budget and the best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np
import cvxpy as cp
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Fast full (linear) convolution via real FFT (used everywhere)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_full_convolution_fft(a, a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f∗g)_k ≤ M  (∀k) , g ≥ 0.
    Returns the optimal vector g (may be all‑zero on failure).
    """
    n = f.shape[0]
    # Build the (2n‑1)×n convolution matrix A where (A g)_k = (f∗g)_k .
    c = np.concatenate([f, np.zeros(n - 1)])
    r = np.concatenate([np.array([f[0]]), np.zeros(n - 1)])
    A = toeplitz(c, r)                     # dense, shape (2n‑1, n)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]            # elementwise ≤
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is fast for dense LPs; warm_start saves time on repeated calls.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except cp.error.SolverError:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.asarray(g.value).reshape(-1)


# ----------------------------------------------------------------------
# LP‑based refinement (monotone reduction of the max‑convolution)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iters: int = 12,
               t0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above, mix the solution with the
    current vector and accept the mix only if the worst convolution does
    not increase.  Returns the best vector found and its max‑convolution.
    """
    best_a = a.copy()
    best_val = _conv_max(a)
    cur_a = a.copy()
    t = t0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _conv_max(cur_a)
        g0 = _solve_lp_max_sum(cur_a, M)
        if g0 is None:
            break
        sum_g0 = np.sum(g0)
        if sum_g0 < 1e-12:
            break

        # Scale to the required L1 norm (exactly the same as cur_a).
        g = g0 * (target_sum / sum_g0)

        # ----- back‑tracking line search on mixing factor -----
        t_try = t
        while t_try > 1e-6:
            cand = (1 - t_try) * cur_a + t_try * g
            cand = np.maximum(cand, 0.0)               # safeguard
            cand_val = _conv_max(cand)
            if cand_val <= M + 1e-9:                  # monotone accept
                break
            t_try *= 0.5

        if t_try <= 1e-6:
            # could not find a feasible step → stop
            break

        cur_a = cand
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = cur_a.copy()
            print(f"[{time.time():.2f}] LP it {it:2d}  new best {best_val:.6f}")

        # slowly increase step size when we are successful
        t = min(0.9, t_try * 1.1)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               target_sum: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """Tiny mass transfers between random positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move a small fraction (up to 8 %) of the mass at i to j
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _conv_max(a_new)
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it:5d}  new best={best_val:.6f}")

        # occasional full re‑projection onto the simplex (helps numerical drift)
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        # early exit if we have already reached the target bound
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function with minimal evaluation score.
    Returns a plain Python list of coefficients.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # tiny safety margin

    best_seq = None
    best_score = np.inf

    def _record(seq: np.ndarray):
        nonlocal best_seq, best_score
        score = _conv_max(seq)                     # equals evaluation after normalisation
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return score

    # ------------------------------------------------------------------
    # Warm‑start from any globally supplied construction
    # ------------------------------------------------------------------
    hs = globals().get("height_sequence_1")
    if hs is not None:
        hs_arr = np.asarray(hs, dtype=float)
        if hs_arr.ndim == 1 and hs_arr.size > 0:
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            _record(hs_norm)

    # ------------------------------------------------------------------
    # Phase 1 – diverse parametric random initialisations
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:      # reserve ~250 s for later phases
        remaining = deadline - time.time()
        # pick a length (larger lengths later)
        if remaining > 600:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 4000 + 1)

        target_sum = np.sqrt(2.0 * n)

        # random family
        shape = np.random.choice(["exp", "pow", "hann", "hamming", "dirichlet"])
        if shape == "exp":
            tau = np.random.uniform(0.1, 2.5) * n
            idx = np.arange(n, dtype=float)
            a0 = np.exp(-idx / tau)
        elif shape == "pow":
            p = np.random.uniform(0.2, 1.7)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p)
        elif shape == "hann":
            i = np.arange(n, dtype=float)
            a0 = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        elif shape == "hamming":
            i = np.arange(n, dtype=float)
            a0 = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
        else:                     # Dirichlet (uniform on simplex)
            a0 = np.random.dirichlet(np.ones(n))

        # optional tiny noise
        if np.random.rand() < 0.25:
            noise = np.random.randn(n) * np.max(a0) * 0.02
            a0 = np.maximum(a0 + noise, 0.0)

        # normalise to the required L1 sum
        a0 = a0 * (target_sum / a0.sum())
        _record(a0)

    # ------------------------------------------------------------------
    # Phase 2 – LP based refinement on the current global best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        refined_seq, refined_val = _lp_refine(best_seq.copy(),
                                              target_sum,
                                              deadline - 10.0,
                                              max_iters=12,
                                              t0=0.4)
        if refined_val < best_score:
            best_score = refined_val
            best_seq = refined_seq.copy()
            print(f"[{time.time():.2f}] after LP refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final simulated‑annealing hill‑climb (short polishing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_seq, hill_val = _hillclimb(best_seq.copy(),
                                        target_sum,
                                        deadline)
        if hill_val < best_score:
            best_score = hill_val
            best_seq = hill_seq.copy()
            print(f"[{time.time():.2f}] after hill‑climb final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _conv_max(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list (the evaluator expects a list)
    return best_seq.tolist()
