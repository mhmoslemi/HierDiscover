def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    target_val = 1.503  # stop when reached

    # choose length(s)
    lengths = [4096, 2048, 8192]  # prioritize medium length
    # We'll cycle length selection
    length_idx = 0

    # Precompute conv_fft_len per length
    fft_lens = {n: _next_pow2(2*n-1) for n in lengths}
    # Main loop
    while time.time() < deadline:
        n = lengths[length_idx % len(lengths)]
        length_idx += 1
        conv_fft_len = fft_lens[n]

        # generate candidate
        # sample K as proportion
        prop = np.exp(np.random.uniform(np.log(0.1), np.log(0.5)))  # between 10% and 50%
        K = max(2, int(round(prop * n)))
        # random indices
        idx = np.random.choice(n, K, replace=False)
        if np.random.rand() < 0.5:
            # equal weights
            weights = np.full(K, 1.0 / K)
        else:
            alpha = np.random.choice([0.2, 0.5, 1.0, 2.0])
            weights = np.random.dirichlet(np.full(K, alpha))
        a = np.zeros(n, dtype=np.float64)
        a[idx] = weights

        # compute objective via FFT
        fa = np.fft.rfft(a, conv_fft_len)
        conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]
        max_c = conv.max()
        s = a.sum()
        val = 2.0 * n * max_c / (s * s) if s > 0 else float('inf')

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n}] new best {val:.6f} (K={K})")
            if best_val <= target_val:
                print("Target reached, breaking.")

"""
Monte‑Carlo sparse search for a low‑peak self‑convolution step function.

The objective is
    F(a) = 2·n·max(a*a) / (sum(a)²)

for a non‑negative vector `a` of length `n`.  The search works as follows:

1. Fix a few candidate lengths (2048 … 16384).  Large `n` gives more
   expressive power, but convolutions become more expensive.
2. Repeatedly draw a *sparse* random vector:
   - Choose a proportion `p` in (0.1, 0.5) and set K = ⌈p·n⌉.
   - Pick K distinct indices and put either equal weights
     (1/K) or a Dirichlet‑distributed random split on those indices.
   - All other entries stay exactly zero.
   The resulting vector is already on the feasible simplex (sum = 1)
   and has many zeroes, which dramatically lowers the peak of the
   self‑convolution.
3. Compute the objective cheaply with an FFT based convolution.
   Keep the best vector seen so far; stop early once the target bound
   (1.503) is surpassed.
4. If time permits, run a cheap stochastic hill‑climb on the current
   best vector to squeeze out any remaining improvement.
5. Return the best sequence as a plain Python list.

The whole routine respects the supplied time budget (default 1000 s)
and never returns an empty or invalid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)²)."""
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    conv = _full_convolution_fft(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


def _random_sparse_sequence(n: int) -> np.ndarray:
    """
    Generate a random sparse non‑negative sequence of length ``n``.
    The number of non‑zeros K is a random proportion of n (≈10‑50 %).
    The non‑zero values are either equal or drawn from a Dirichlet
    distribution (different concentration parameters are tried).
    """
    # proportion p ∈ [0.1, 0.5] drawn log‑uniformly
    p = np.exp(np.random.uniform(np.log(0.1), np.log(0.5)))
    K = max(2, int(round(p * n)))
    idx = np.random.choice(n, K, replace=False)

    if np.random.rand() < 0.5:
        # equal masses
        weights = np.full(K, 1.0 / K, dtype=np.float64)
    else:
        # Dirichlet split with a random concentration
        alpha = np.random.choice([0.2, 0.5, 1.0, 2.0])
        weights = np.random.dirichlet(np.full(K, alpha)).astype(np.float64)

    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = weights
    return seq


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, deadline: float,
                     max_trials: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local optimisation.
    Repeatedly moves a tiny amount ε from one index to another and
    keeps the move if it reduces the exact objective.
    """
    n = a0.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution_fft(best_a, fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution_fft(cand, fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
        else:
            no_improve += 1
        trials += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2·n·max(a*a) / (sum(a)²)

    The search is dominated by a random sparse Monte‑Carlo phase,
    followed by a tiny hill‑climb refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # Target value – stop early once we are under it
    TARGET_VAL = 1.503

    best_seq = None
    best_val = float("inf")

    # Lengths to try (larger lengths allow a lower theoretical bound)
    lengths = [2048, 4096, 8192, 16384]
    # Pre‑compute FFT sizes for each length
    fft_lens = {n: _next_pow2(2 * n - 1) for n in lengths}
    # Simple round‑robin over lengths
    length_cycle = 0

    # --------------------------  Monte‑Carlo phase  --------------------
    while time.time() < deadline:
        n = lengths[length_cycle % len(lengths)]
        length_cycle += 1
        fft_len = fft_lens[n]

        a = _random_sparse_sequence(n)
        val = _objective_fft(a, n, fft_len)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n}] new best {val:.6f}")
            if best_val <= TARGET_VAL:
                print("Target achieved – exiting Monte‑Carlo loop.")
                break

    # ---------------------------  refinement  -------------------------
    if best_seq is not None and time.time() < deadline:
        # Perform a quick hill‑climb on the current best sequence.
        remaining = deadline - time.time()
        # At most a few hundred thousand tiny moves – will stop early
        # if time runs out.
        a_ref, val_ref = _local_hill_climb(best_seq, time.time() + remaining,
                                           max_trials=200_000)
        if val_ref < best_val:
            best_val, best_seq = val_ref, a_ref
            print(f"[final hill‑climb] improved to {val_ref:.6f}")

    # ------------------------------ fallback ---------------------------
    if best_seq is None:
        # should never happen, but keep a safe uniform default
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external grader
    return best_seq.tolist()
