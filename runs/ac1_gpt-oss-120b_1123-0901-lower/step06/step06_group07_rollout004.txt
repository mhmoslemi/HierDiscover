"""
Propose a low‑objective step‑function using a multi‑stage global‑local search.

Algorithm
---------
1. **Generate diverse seeds** – a rich palette of parametric families
   (uniform, Dirichlet, edge‑exponential/power‑law, two‑cluster,
   alternating high/low, half‑support, etc.) together with the previous
   best if available.  All candidates are normalised to sum ≈ 1.

2. **Quick screening** – evaluate every seed with a fast FFT‑based
   surrogate for the objective (the exact same formula as
   ``evaluate_sequence``).  Keep the best few (default 12).

3. **Deep optimisation** – for each selected seed run an Adam optimiser
   on a smooth‑max surrogate (log‑sum‑exp) of the convolution peak,
   followed by an exact sub‑gradient refinement using exponentiated
   updates.  Hyper‑parameters (learning‑rate, temperature schedule,
   number of iterations) are tuned aggressively and the loop aborts
   as soon as the global deadline is reached.

4. **Scale up** – if time permits, interpolate the current best vector
   to a finer grid (n = 8192, optionally 16384) and repeat the deep
   optimisation on the larger problem.

5. **Random local search** – a cheap hill‑climbing stage that repeatedly
   moves a tiny amount of mass between two random coordinates and keeps
   any improvement.  This can squeeze out the last few hundredths of the
   objective.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found as a plain Python list of non‑negative
floats (clipped to the evaluator’s allowed range).

"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective (same as evaluate_sequence) – cheap because convolution
    is performed with FFT.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # start in log‑space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                     # a bit larger than before

    lam = 0.4                                      # initial temperature
    lam_factor = 1.012                             # slower increase → smoother
    lam_cap = 5e6

    last_improve = time.time()
    it = 0
    max_iter = 350_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate evaluation ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam iter={it:06d}  best={val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)          # <∇, a>
        grad_log = a * (grad_a - avg)    # gradient w.r.t. logits

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- random shake if stuck ----
        if time.time() - last_improve > 4.5:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    # initialise from the best sequence seen so far (or a0)
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on the simplex

    eta0 = 0.65                      # initial learning rate
    last_improve = time.time()
    max_iter = 30_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad iter={it:05d}  best={val:.6f}")

        # ---- active set (all entries within small absolute tolerance) ----
        eps_active = 1e-12
        active = np.isclose(conv, max_c, atol=eps_active, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            # fall back to a single peak if numerics are pathological
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- decreasing step size (simple schedule) ----
        eta = eta0 / np.sqrt(it + 1)

        # ---- exponentiated‑gradient update (keeps a ≥ 0) ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- early stop if stagnating ----
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) + exact sub‑gradient refinement.
    Updates the global best.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list:
    """
    Produce a diverse set of seeds.
    In addition to the families used before, we now include:
        • two‑cluster spikes,
        • alternating high/low patterns,
        • half‑support (mass only on one side),
        • random Dirichlet with extreme concentrations.
    All vectors are normalised to sum ≈ 1.
    """
    candidates = []

    # ---------- warm start & simple rotations ----------
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ---------- uniform ----------
    candidates.append(np.full(n, 1.0 / n))

    # ---------- random Dirichlet (various concentrations) ----------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # ---------- edge‑exponential ----------
    for alpha in np.logspace(-2, 1.2, num=20):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # ---------- edge‑power‑law ----------
    for p in np.linspace(0.2, 5.5, num=20):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # ---------- two‑Gaussian edges ----------
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # ---------- half‑support (mass only on one side) ----------
    for decay in (0.02, 0.05, 0.1, 0.2):
        left = np.exp(-decay * i)
        left[n // 2 :] = 0.0
        left /= left.sum()
        candidates.append(left)

    # ---------- alternating high/low ----------
    for high_frac in (0.9, 0.8, 0.7):
        high = high_frac / (n // 2)
        low = (1.0 - high_frac) / (n // 2)
        alt = np.empty(n, dtype=np.float64)
        alt[0::2] = high
        alt[1::2] = low
        alt /= alt.sum()
        candidates.append(alt)

    # ---------- two‑cluster spikes ----------
    for cluster_sz in (4, 8, 16, 32):
        for spacing in (n // 4, n // 3, n // 2):
            vec = np.zeros(n, dtype=np.float64)
            # left cluster
            start_l = np.random.randint(0, n // 2 - cluster_sz)
            vec[start_l : start_l + cluster_sz] = np.random.rand(cluster_sz)
            # right cluster
            start_r = start_l + spacing
            if start_r + cluster_sz <= n:
                vec[start_r : start_r + cluster_sz] = np.random.rand(cluster_sz)
            vec = np.maximum(vec, 0.0)
            if vec.sum() < 0.01:
                continue
            vec /= vec.sum()
            candidates.append(vec)

    # ---------- sinusoidal half‑wave ----------
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def _random_local_search(a0: np.ndarray, n: int, conv_fft_len: int,
                         deadline: float, best_seq: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a tiny amount of mass between two random
    coordinates and keep any improvement.  Runs until the deadline.
    """
    a = best_seq.copy()
    eps = 1e-4                     # base step size (relative to total mass)
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to transfer (bounded to keep positivity)
        transfer = eps * rng.random()
        if a[i] < transfer:
            continue
        a_new = a.copy()
        a_new[i] -= transfer
        a_new[j] += transfer
        # keep normalisation (sum unchanged)
        conv = _full_convolution(a_new, conv_fft_len)
        val = 2.0 * n * conv.max() / (a_new.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a_new
            a = a_new
            print(f"[{n}] Local search improved to {val:.6f}")
    return best_seq, best_val

            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Final cheap random‑local‑search stage to squeeze out the last bit
    if best_seq is not None and time.time() < deadline:
        print("--- Final random local search ---")
        best_seq, best_val = _random_local_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
