def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    lengths = [int(1900), 2100, 2300]? Actually we may use set [1500, 1800, 2100, 2400].

    best_seq = None; best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline: break
        target = math.sqrt(2*n)

        # param families
        param_families = []
        # exponential basis
        for r in np.linspace(0.85, 0.999, 30):
            def gen_exp(n, r=r):
                idx = np.arange(n)
                a = r**idx
                return a
            param_families.append(gen_exp)

        # power law decreasing
        for p in np.arange(-4.5, 5.5, 0.5):
            def gen_pow(n, p=p):
                idx = np.arange(1, n+1)  # avoid zero power down to negative infinite
                a = idx**p
                return a
            param_families.append(gen_pow)

        # symmetric valley: a_i = ((i+1)*(n-i))**p for negative p
        for p in np.arange(-4, -0.5, 0.5):
            def gen_valley(n, p=p):
                i = np.arange(n)
                a = ((i+1)*(n-i))**p
                return a
            param_families.append(gen_valley)

        # Gaussian
        for sigma_frac in [0.2,0.3,0.4,0.5]:
            sigma = sigma_frac * n
            def gen_gauss(n, sigma=sigma):
                i = np.arange(n)
                mu = (n-1)/2.0
                a = np.exp(- (i - mu)**2 / (2 * sigma**2))
                return a
            param_families.append(gen_gauss)

        # sin shape
        def gen_sin(n):
            i = np.arange(n)
            a = np.sin(np.pi * (i+0.5) / n)
            return a
        param_families.append(gen_sin)

        # maybe add uniform and random to have baseline
        def gen_uniform(n):
            return np.ones(n)
        param_families.append(gen_uniform)

        # now evaluate each generator
        for gen in param_families:
            if time.time() >= deadline:
                break
            a = gen(n).astype(float)
            a = np.maximum(a, 0.0)  # ensure non-negative
            if a.sum() == 0:
                continue
            a = a * (target / a.sum())
            # compute conv max
            Nfft = 1 << ((2*n - 1).bit_length())
            A_fft = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
            max_b = float(conv.max())
            score = 2.0 * n * max_b / (target**2)   # should equal max_b
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[candidate] n={n} gen={gen.__name__} score={score:.6f}")

    # after scanning, perform local refinement
    if best_seq is not None and time.time() < deadline:
        # use projected gradient descent (PGD) for more improvement
        # reuse earlier PGD function scaled with allowed iterations based on remaining time
        remaining = deadline - time.time()
        # allocate half to PGD
        pgd_deadline = time.time() + remaining * 0.5
        # compute max_iters based on length: say 500k max
        max_iters = int(min(5_000_000, 200_000 * best_seq.shape[0] / 2000))
        best_seq, best_score = _pgd_optimize(best_seq.copy(), best_seq.shape[0], pgd_deadline,
                                            max_iters, best_seq, best_score, rng)

        # final local stochastic refinement with SA
        if time.time() < deadline:
            refinement_deadline = deadline
            best_seq, best_score = _local_refine(best_seq, best_seq.shape[0], refinement_deadline, best_score, rng)

    # return as list
    return best_seq.tolist()

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Explore a few low‑dimensional parametric families** (exponential,
   power‑law, symmetric “valley”, Gaussian, sinusoidal, uniform).  
   Each candidate is normalised so that Σa = √(2 n) – this makes the
   objective equal to the maximum of the self‑convolution.
   All candidates are evaluated with a single FFT‑based convolution;
   the best one is kept.

2. **Fine‑tune the best candidate** with a Projected Gradient Descent
   (PGD) on the exact max‑convolution.  The sub‑gradient is obtained from
   the active (max‑attaining) convolution entries, a step is taken and
   the vector is projected onto the simplex {a_i≥0, Σa_i = √(2 n)}.
   An annealing schedule and occasional Gaussian jitter improve
   convergence.

3. **Stochastic refinement** – a cheap simulated‑annealing hill‑climber
   that repeatedly moves a small fraction of mass from one index to
   another.  Improvements are always accepted; worsening moves are
   accepted with a temperature‑controlled probability.

The whole search respects the 1000 s budget and dynamically allocates
time among the three phases.  The final sequence (plain Python list) is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = target_sum}.
    O(n log n) implementation (Duchi et al. 2008).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact maximum of the linear convolution a*a (via one FFT)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed and its exact objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise on the simplex
    a = a0.copy()
    a = a * target_sum / a.sum()

    # FFT size – must accommodate linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.001 * target_sum          # base step size
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution & current max
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # active set (hard max)
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # weighted reverse convolution (sub‑gradient)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev                     # ∇ max(a∗a)

        # gradient step + projection
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter to avoid plateaux
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing with simulated annealing on the simplex.
    Mass is repeatedly transferred between two random indices.
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a small random fraction of source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

    best_score = float("inf")

    # Phase 1 – parametric sweep
    for n in lengths:

        target_sum = math.sqrt(2.0 * n)
        print(f"[phase‑1] exploring length {n}")

        families = []

        families.append(0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1)))

            a0

        )

        # optional – use previous best if lengths match
        if base is not None and base.shape[0] == n:
            families.append(base)

        # ----- evaluate all candidates ---------------------------------
        Nfft = 1 << ((2 * n - 1).bit_length())
        for cand in families:
            if time.time() >= deadline:
                break
            a = cand.astype(float)
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                continue
            a = a * (target_sum / a.sum())      # normalise to √(2 n)

            # fast max‑convolution
            A_fft = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
            max_b = float(conv.max())
            score = max_b                        # because 2 n·max_b / (√(2 n))² = max_b

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[phase‑1] new global best length {n} score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected gradient descent fine‑tuning
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        pgd_deadline = time.time() + remaining * 0.55   # allocate ~55% to PGD
        n = best_seq.shape[0]
        # iteration budget roughly proportional to length, capped at 3 M
        max_iters = int(min(3_000_000, 2_500_000 * 2100 / n))
        max_iters = max(200_000, max_iters)
        best_seq, best_score = _pgd_optimize(
            a0=best_seq.copy(),
            n=n,
            deadline=pgd_deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic refinement (simulated annealing)
    if best_seq is not None and time.time() < deadline:
        refine_deadline = deadline
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback (unlikely)

        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    # final sanity check (optional)

        final_val = evaluate_sequence(best_seq.tolist())

    except Exception:
        print(f"[done] best score (max‑conv) = {best_score:.6f}, length = {len(best_seq)}")

    return best_seq.tolist()
