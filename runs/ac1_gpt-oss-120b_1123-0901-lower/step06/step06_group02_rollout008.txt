def generate_parametric_sequence(shape, n, rng):
    idx = np.arange(n, dtype=float)
    if shape == "exp":
        tau = rng.uniform(0.01 * n, 5.0 * n)
        a = np.exp(-idx / tau)
    elif shape == "exp_pow":
        tau = rng.uniform(0.01 * n, 5.0 * n)
        p = rng.uniform(0.5, 3.0)
        a = np.exp(-(idx / tau) ** p)
    elif shape == "pow":
        p = rng.uniform(0.2, 2.0)
        a = (idx + 1) ** (-p)
    elif shape == "pow_shift":
        p = rng.uniform(0.2, 2.0)
        shift = rng.uniform(0.0, 5.0)
        a = (idx + shift) ** (-p)
    elif shape == "geometric":
        r = rng.uniform(0.9, 0.999)
        a = r ** idx
    elif shape == "beta":
        alpha = rng.uniform(0.0, 5.0)
        beta = rng.uniform(0.0, 5.0)
        a = (idx + 1) ** alpha * (n - idx) ** beta
    elif shape == "triangular":
        a = np.minimum(idx + 1, n - idx)
    elif shape == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
    elif shape == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    else:
        a = np.ones(n, dtype=float)
    # add small noise optionally
    if rng.random() < 0.3:
        noise = rng.normal(scale=0.02 * a.max(), size=n)
        a = np.maximum(a + noise, 0.0)
    # special handling: ensure no all zero.
    if a.sum() == 0:
        a = np.ones(n, dtype=float)
    # Normalize to target sum sqrt(2n)
    target = np.sqrt(2.0 * n)
    a = a * (target / a.sum())
    return a
```

"""
Hybrid search for a low‑peak auto‑convolution step function.

The objective (after fixing Σaᵢ = √(2n)) equals max(conv(a,a)).
We therefore minimise the peak of the auto‑convolution.

Algorithm outline
-----------------
1.  Initialise a global best candidate (uniform, previous global variable
    `height_sequence_1`, and many random parametric families).  Lengths
    are drawn up to 15000, and families include exponentials, power‑laws,
    beta‑shaped, geometric, triangular and window functions.
2.  Refine the current best with Adam on a log‑sum‑exp (soft‑max) surrogate
    of the max‑convolution.  The parameters are the logits of a softmax,
    which automatically respects the simplex constraint.
3.  Occasionally upscale the best shape (linear interpolation to a longer
    length) and re‑optimise – longer sequences have more degrees of freedom
    and can lower the peak further.
4.  Targeted projected sub‑gradient descent on the exact max‑convolution:
    follow the gradient at the worst index and project back onto the
    simplex.
5.  Final simulated‑annealing hill‑climb that moves small amounts of mass
    between positions, accepting moves with a Metropolis rule.
6.  If nothing is found a simple uniform construction is returned.

All phases respect the overall time budget (default 1000 s) and print
progress information.  The returned value is a plain Python list of
non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return full convolution x*y using real FFT (non‑negative output)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based indexing
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameterised families for random initialisation
# ----------------------------------------------------------------------
def _generate_parametric_sequence(shape: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative raw sequence (not yet normalised) for a given shape."""
    idx = np.arange(n, dtype=float)

    if shape == "exp":
        tau = rng.uniform(0.01 * n, 5.0 * n)
        a = np.exp(-idx / tau)

    elif shape == "exp_pow":
        tau = rng.uniform(0.01 * n, 5.0 * n)
        p = rng.uniform(0.5, 3.0)
        a = np.exp(- (idx / tau) ** p)

    elif shape == "pow":
        p = rng.uniform(0.2, 2.0)
        a = (idx + 1) ** (-p)

    elif shape == "pow_shift":
        p = rng.uniform(0.2, 2.0)
        shift = rng.uniform(0.0, 5.0)
        a = (idx + shift) ** (-p)

    elif shape == "geometric":
        r = rng.uniform(0.90, 0.999)
        a = r ** idx

    elif shape == "beta":
        # a_i ∝ (i+1)^α * (n-i)^β, large at the ends, small in the centre
        alpha = rng.uniform(0.0, 5.0)
        beta = rng.uniform(0.0, 5.0)
        a = (idx + 1) ** alpha * (n - idx) ** beta

    elif shape == "triangular":
        a = np.minimum(idx + 1, n - idx)

    elif shape == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))

    elif shape == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))

    else:                               # fallback uniform
        a = np.ones(n, dtype=float)

    # Tiny Gaussian noise to break symmetries (optional)
    if rng.random() < 0.3:
        noise = rng.normal(scale=0.02 * a.max(), size=n)
        a = np.maximum(a + noise, 0.0)

    # Guard against all‑zero vectors (unlikely)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=float)

    return a


def _random_sequence(rng: np.random.Generator) -> np.ndarray:
    """Sample a random length and shape, normalise to the required L1 sum."""
    # Length choice – longer sequences generally allow a lower peak.
    # We cap at 15000 to keep FFTs cheap.
    n = rng.integers(4000, 15001)      # inclusive of 4000, exclusive of 15001
    shape = rng.choice(["exp", "exp_pow", "pow", "pow_shift",
                        "geometric", "beta", "triangular", "hann", "hamming"])
    raw = _generate_parametric_sequence(shape, n, rng)
    target = np.sqrt(2.0 * n)
    seq = raw * (target / raw.sum())
    return seq


# ----------------------------------------------------------------------
# Soft‑max Adam optimisation on a smoothed max‑convolution
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    logits0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best vector (on the simplex) and its exact max‑convolution value."""
    x = logits0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # Map logits → simplex via soft‑max
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # numeric stability
        p = exp_shift / np.sum(exp_shift)      # Σp = 1
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()

        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # Log‑sum‑exp smoothing weights
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient w.r.t. a: 2·a ⊛ rev(w)
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # Back‑prop into logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and rng.random() < 0.2:
            x += rng.standard_normal(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current peak
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70
            if rng.random() < 0.07:    # occasional jitter
                jitter = rng.random(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:            # early exit if target reached
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Metropolis hill‑climb moving tiny amounts of mass between positions."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # Temperature schedule
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * rng.random() * 0.08   # move up to 8 % of donor mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if rng.random() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Upscaling helper – linear interpolation from old to new length
# ----------------------------------------------------------------------
def _upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate seq to length new_n and renormalise to √(2·new_n)."""
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    interp = np.interp(x_new, x_old, seq)
    target = np.sqrt(2.0 * new_n)
    return interp * (target / interp.sum())


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimisation to produce a non‑negative step function whose
    max‑convolution is as small as possible (equivalently, minimise
    evaluate_sequence()).  Returns a plain Python list.
    """
    global rng
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.15    # safety margin

    best_seq = None
    best_score = np.inf      # this stores the exact max‑convolution value

    def _record(seq: np.ndarray):
        """Evaluate seq (exact max‑convolution) and keep the global best."""
        nonlocal best_seq, best_score
        val = _full_convolution_fft(seq, seq).max()
        if val < best_score:
            best_score = float(val)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return val

    # ------------------------------------------------------------------
    # Warm‑start from any previously saved construction
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            _record(hs_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – diverse parametric random initialisations (long sequences)
    # ------------------------------------------------------------------
    while time.time() < deadline - 300.0:   # reserve ~5 min for refinements
        seq = _random_sequence(rng)
        _record(seq)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on a smoothed max‑convolution
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 180.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        # Initialise logits from the current best (soft‑max parametrisation)
        logits0 = np.log(best_seq + 1e-12)

        # Give each Adam run a modest time slice (≈20 s)
        run_deadline = time.time() + min(22.0, deadline - time.time() - 30.0)
        seq_adam, val_adam = _optimise_adam_softmax(
            n, logits0, target_sum, run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=5e-5,
            alpha0=0.025,
        )
        if val_adam < best_score:
            best_score = val_adam
            best_seq = seq_adam.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_score:.6f}")

        # Occasionally try to enlarge the length (up‑scaling) – longer shapes
        # often admit a lower peak.
        if rng.random() < 0.25 and time.time() < deadline - 200.0:
            new_n = int(n * rng.uniform(1.10, 1.40))
            new_seq = _upscale_sequence(best_seq, new_n)
            _record(new_seq)          # keep it if better
            # Run a quick Adam on the up‑scaled candidate
            logits_up = np.log(new_seq + 1e-12)
            up_deadline = time.time() + min(18.0, deadline - time.time() - 30.0)
            seq_up, val_up = _optimise_adam_softmax(
                new_n, logits_up, np.sqrt(2.0 * new_n), up_deadline,
                init_tau_factor=0.12,
                min_tau_factor=5e-5,
                alpha0=0.025,
            )
            if val_up < best_score:
                best_score = val_up
                best_seq = seq_up.copy()
                print(f"[{time.time():.2f}] after up‑scale+Adam best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0
        seq_sub, val_sub = _targeted_subgradient(best_seq.copy(),
                                                 target_sum,
                                                 sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(best_seq.copy(),
                                        target_sum,
                                        hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return as plain Python list for the evaluator
    return best_seq.tolist()
