Implementation of projection onto simplex:

def proj_simplex(v, s=1):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def _pgd_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000, init_step=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    step = init_step
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # Could print progress
        # subgradient, uniform over active set
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient descent step
        a_new = a - step * grad
        # project onto simplex
        a_new = _proj_simplex(a_new, s=1.0)
        # maybe adapt step by line search: if objective increased, halve step
        conv_new = _full_convolution(a_new, conv_fft_len)
        max_c_new = conv_new.max()
        val_new = 2.0 * n * max_c_new
        if val_new > val:
            # backtrack step size
            step *= 0.5
        else:
            step = min(step * 1.01, 1.0)  # slowly increase step
            a = a_new
    return best_a, best_val

"""
Two‑phase optimiser for the step‑function problem.

Phase 1 – smooth‑max optimisation:
    The true objective 2·n·max(a*a) / (∑a)² is replaced by a
    log‑sum‑exp (smooth‑max) surrogate with temperature λ.
    For a sequence a on the simplex we optimise the surrogate with
    L‑BFGS‑B, increasing λ geometrically so the surrogate converges to the
    true maximum.  The optimisation is performed on the log‑probability
    (logits) of a, i.e. a = softmax(logits), which automatically enforces
    non‑negativity and the simplex constraint.

Phase 2 – exact‑max sub‑gradient refinement:
    With the best sequence from Phase 1 we run a mirrored‑gradient
    (exponentiated‑gradient) scheme on the exact max‑convolution until the
    time budget expires.

A cheap random local‑search is finally applied.  The routine respects a
user‑supplied time budget (default 1000 s) and always returns a feasible
(non‑negative) list of coefficients.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)] .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    """
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return a vertex of the simplex
        w = np.zeros_like(v)
        w[np.argmax(v)] = s
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _smooth_obj_and_grad(logits: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         lam: float):
    """
    Smooth‑max (log‑sum‑exp) surrogate and its gradient w.r.t. logits.
    Returns (value, gradient) where the gradient is taken w.r.t. the
    logits (the variables passed to the optimiser).
    """
    eps = 1e-15
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()                     # a is on the simplex

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # soft‑max weights over the convolution vector
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # smooth surrogate value
    smooth_max = max_c + np.log(w_sum) / lam
    val = 2.0 * n * smooth_max

    # gradient w.r.t. a (exact for the surrogate)
    grad_a = 2.0 * _cross_correlation(w_norm, a, corr_fft_len)

    # chain rule through the soft‑max (logits → a)
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    return val, grad_logits.astype(np.float64)


def _l_bfgs_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     lambdas=None,
                     maxiter_per_stage: int = 20):
    """
    Optimises the smooth‑max surrogate with L‑BFGS‑B.
    The temperature λ is increased through the list `lambdas`.
    Returns the best sequence (sums to 1) and its exact objective value.
    """
    if lambdas is None:
        lambdas = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 3000.0]

    logits = np.log(a0 + 1e-15)
    best_a = a0.copy()
    best_val = float("inf")

    for lam in lambdas:
        if time.time() >= deadline:
            break
        args = (n, conv_fft_len, corr_fft_len, lam)
        # minimise the smooth surrogate; jac=True makes SciPy expect (f, g)
        res = minimize(_smooth_obj_and_grad,
                       logits,
                       args=args,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage,
                                'disp': False})
        logits = res.x
        # Transform back to a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective for the current a
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new exact best {val:.6f}")

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta: float = 0.6):
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates
    (mirrored descent on the simplex).  Returns the best vector seen
    and its exact objective value.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad iter={it:05d}] new best {val:.6f}")

        # Active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # Re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 30_000):
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    index to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 3000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget and always returns the
    best sequence found (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – start with a modest length and upscale if time permits
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Construct a small pool of diverse starting vectors
    candidates = []

    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Random Dirichlet (5 draws)
    for _ in range(5):
        candidates.append(np.random.dirichlet(np.ones(n)))

    # Sinusoidal half‑wave
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    candidates.append(s)

    # Power‑law edge shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 3):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optimise each candidate (short runs) – keep the globally best
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # Phase 1 – smooth‑max L‑BFGS optimisation
        a_lbfgs, val_lbfgs = _l_bfgs_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            lambdas=[0.5, 2.0, 10.0, 50.0, 200.0, 800.0],
            maxiter_per_stage=15,
        )
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"[cand {cid}] L‑BFGS exact best {val_lbfgs:.6f}")

        # Phase 2 – exact‑max sub‑gradient refinement (short)
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=30_000,
                eta=0.6,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad short best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Long sub‑gradient refinement on the best candidate
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=200_000,
            eta=0.6,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=30_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a larger grid if time permits (more degrees of freedom)
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # Interpolate linearly to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # Phase 1 on the larger grid (shorter run)
        a_big_lbfgs, val_big_lbfgs = _l_bfgs_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            lambdas=[0.5, 2.0, 10.0, 50.0, 200.0],
            maxiter_per_stage=12,
        )
        # Phase 2 refinement
        if time.time() < deadline:
            a_big_ref, val_big_ref = _subgradient_optimize(
                a_big_lbfgs,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=120_000,
                eta=0.6,
            )
            if val_big_ref < best_val:
                best_val = val_big_ref
                best_seq = a_big_ref.copy()
                print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
