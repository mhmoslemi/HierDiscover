    best_a = a.copy()

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm follows a multi‑scale scheme:

* A hierarchy of lengths (1024 → 2048 → 4096 → 8192).  
  For each length we generate a diverse set of start‐up shapes
  (Dirichlet draws, power‑law, exponential, cosine, Gaussian,
  truncated‑arcsine, sparse spikes and – if available – an up‑sampled
  solution from the previous size).

* Each candidate is refined by three phases
  1. Adam optimisation on a smooth‑max surrogate (softmax over the
     convolution).  The temperature (λ) grows slowly so that the
     optimisation initially sees a smooth landscape and later focuses
     on the true maximum.
  2. Exponentiated‑gradient sub‑gradient descent that directly minimises
     the exact max (active‑set weighting).  This phase is able to
     escape the bias of the smooth approximation.
  3. A short “fine‑tune” Adam run with a reduced learning‑rate.

* After the multi‑scale pass a lightweight simulated‑annealing / hill‑climb
  refinement is applied to the global best sequence for the remaining
  time budget.

All convolution and cross‑correlation operations are performed via FFT
routines (O(n log n)).  The routine respects a time budget (default 1000 s)
and returns the best sequence discovered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (a*a) and cross‑correlation (w ★ a)
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·len(a)−1 entries (linear convolution)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective_val(a: np.ndarray, n: int, conv_len: int) -> float:
    """
    Exact evaluation of the objective (sum of a is assumed to be 1).
    Returns 2·n·max(conv) .
    """
    conv = _conv_fft(a, conv_len)
    return 2.0 * n * conv.max()

def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified list of start‑up vectors, normalised to sum to 1.
    """
    cand = []

    # 1) warm‑start from previous global solution if present
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-8:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # 2) plain Dirichlet (uniformly random)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 3) spiky Dirichlet (conc < 1)
    for conc in (0.2, 0.4, 0.6):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) over‑dispersed Dirichlet (conc > 1)
    for conc in (1.5, 2.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 5) edge‑concentrated power‑law (prod = i·(n‑i+1))
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.4, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 6) symmetric two‑sided exponential
    for gamma in (0.001, 0.002, 0.004):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # 7) cosine window (single bump)
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    cand.append(coswin / coswin.sum())

    # 8) half‑sine window
    sinwin = np.sin(np.pi * i / (n - 1))
    cand.append(sinwin / sinwin.sum())

    # 9) discrete Gaussian
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    cand.append(gauss / gauss.sum())

    # 10) truncated arcsine (inverse‑sqrt edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    cand.append(arcs / arcs.sum())

    # 11) random sparse pattern (≈ sqrt(n) spikes)
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=m_spikes, replace=False)
    sparse[indices] = 1.0
    if sparse.sum() > 0:
        cand.append(sparse / sparse.sum())

    return cand

def _adam_phase(a0: np.ndarray, n: int, conv_len: int, corr_len: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best vector found (sum = 1) and its objective value.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    # Adam hyper‑parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # learning‑rate scales with problem size
    lr = 0.12 * np.sqrt(2048.0 / n)

    # temperature (smooth‑max) schedule
    lam = 0.3
    lam_factor = 1.01  # gentle increase

    best_val = np.inf
    best_a = a.copy()
    last_improve = time.time()

    # iteration budget: more iterations for smaller n, but we grant
    # a generous ceiling for the largest size as well.
    max_iters = int(50000 * 1024 / n) + 20000

    for t in range(1, max_iters + 1):
        # ---- soft‑max (log‑space) to enforce simplex ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weighting
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (linear in w)
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature update
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a)==1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional random shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline - 0.2:
            break

    return best_a, best_val

def _subgrad_expg_phase(a0: np.ndarray, n: int, conv_len: int, corr_len: int,
                       best_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase that works on the exact max.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()

    eta = 0.55                # step‑size
    max_iters = 8000
    last_improve = time.time()

    for it in range(max_iters):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()

        # active set (indices attaining the max, tolerance tiny)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline - 0.2:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _fine_tune_phase(a0: np.ndarray, n: int, conv_len: int, corr_len: int,
                     best_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    A short Adam fine‑tune with a smaller learning‑rate and slower λ growth.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lr = 0.045 * np.sqrt(2048.0 / n)
    lam = 0.4
    lam_factor = 1.012

    max_iters = int(60000 * 1024 / n) + 5000
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum != 0.0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 10.0:
            # gentle random jitter
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        if time.time() > deadline - 0.2:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_anneal(start_seq: np.ndarray,
                  start_val: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing / simulated‑annealing refinement that moves a tiny
    amount of mass between two random coordinates.
    """
    best_a = start_seq.copy()
    best_val = start_val
    n = best_a.size

    # temperature schedule (linear cooling)
    T0 = 1e-3
    Tmin = 1e-6

    while time.time() < deadline - 0.1:
        # fraction of remaining time determines current temperature
        remaining = deadline - time.time()
        T = Tmin + (T0 - Tmin) * (remaining / (deadline - time.time() + 1e-12))

        # propose a random transfer
        i, j = np.random.choice(n, 2, replace=False)
        if best_a[i] < 1e-12:
            continue
        delta = (np.random.rand() * 0.02) * best_a[i]   # up to 2 % of source mass
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # renormalise (should stay ~1 but we do it for safety)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new /= s

        # evaluate
        conv = _conv_fft(a_new, 1 << int(np.ceil(np.log2(2 * n - 1))))
        val = 2.0 * n * conv.max()

        # accept if improvement or with Metropolis probability
        if val < best_val - 1e-12:
            best_a, best_val = a_new, val
        else:
            prob = np.exp(-(val - best_val) / T)
            if np.random.rand() < prob:
                best_a, best_val = a_new, val

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  A multi‑scale hierarchical optimisation (Adam → sub‑gradient →
    fine‑tune → hill‑climb) is performed under the given time budget.
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – we end at 8192 because larger FFTs become costly.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2((2 * n - 1) + n - 1)))

        # start‑up candidates
        candidates = _build_initial_candidates(n)

        # add an up‑sampled version of the current global best (if any)
        if best_seq is not None and len(best_seq) != n:
            # linear interpolation to new size
            prev_n = len(best_seq)
            x_old = np.linspace(0, 1, prev_n)
            x_new = np.linspace(0, 1, n)
            up = np.interp(x_new, x_old, best_seq)
            up = np.maximum(up, 0.0)
            if up.sum() < 1e-12:
                up = np.random.dirichlet(np.ones(n))
            else:
                up = up / up.sum()
            candidates.append(up)

        # optimise each candidate
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            a0 = np.maximum(cand, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            else:
                a0 /= a0.sum()

            # Phase 1 – Adam on smooth‑max
            a_adam, val_adam = _adam_phase(a0, n, conv_len, corr_len, deadline)

            # Phase 2 – exponentiated‑gradient sub‑gradient
            a_sub, val_sub = _subgrad_expg_phase(a_adam, n, conv_len, corr_len,
                                                val_adam, deadline)

            # Phase 3 – fine‑tune Adam
            a_fine, val_fine = _fine_tune_phase(a_sub, n, conv_len, corr_len,
                                                val_sub, deadline)

            # update global best
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n} cand={cid}] new global best {best_val:.6f}")

    # final hill‑climb on the best sequence found so far
    if best_seq is not None:
        print(f"\n=== Final local refinement (n={len(best_seq)}) ===")
        best_seq, best_val = _local_anneal(best_seq, best_val, deadline)
        print(f"Final best value after refinement: {best_val:.6f}")

    # expose the last best solution globally (helps a future run)
    global height_sequence_1
    height_sequence_1 = best_seq.tolist() if best_seq is not None else None

    # Return as a plain list for the external evaluator.
    return best_seq.tolist()
