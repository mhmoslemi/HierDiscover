def shape_from_params(param, n, upscale=False):
    # param length K
    K = len(param)
    # positivity
    pos = np.exp(param)  # ensures >0
    # positions for param points: indices at linspace [0, n-1] inclusive
    x = np.linspace(0, n - 1, K)
    # create fine grid of length n
    xi = np.arange(n)
    a = np.interp(xi, x, pos)
    # ensure non-negatives
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a

"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(a∗a) / (∑a)²

The method differs from the previous “Adam‑surrogate” pipeline:

* A very compact low‑dimensional parametrisation (log‑weights at K control
  points) is optimised by a simple (μ,λ)‑Evolution Strategy (ES).  The ES works
  on the log‑space so that the generated heights are automatically non‑negative.
* After a coarse optimisation (n≈1024, K≈32) the best parameters are up‑sampled
  and refined on a finer grid (n≈4096, K≈64).  The optimisation uses the fast
  FFT‑based convolution for the objective, which is orders of magnitude faster
  than ``evaluate_sequence``.
* Finally a short mirror‑descent / exponentiated‑gradient sub‑gradient phase
  (identical to the one used in the reference code) tightens the result.
* The whole routine respects the supplied time budget, prints progress and
  returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast, exact for the linear convolution)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the exact sub‑gradient.
    w has length 2·n‑1, a has length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a)>0 assumed)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Low‑dimensional representation of a step function
def _shape_from_params(param: np.ndarray, n: int) -> np.ndarray:
    """
    Convert a low‑dimensional log‑parameter vector to a normalised step function.
    - param  : shape (K,) – log‑weights at K evenly spaced control points.
    - n      : length of the desired full vector.
    """
    # keep the log‑weights in a reasonable range to avoid overflow
    param = np.clip(param, -30.0, 30.0)
    pos = np.exp(param)                # strictly positive values
    K = pos.shape[0]
    # positions of the K control points on the fine grid [0, n‑1]
    xs = np.linspace(0, n - 1, K)
    grid = np.arange(n)
    # linear interpolation onto the full grid
    a = np.interp(grid, xs, pos)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Evolution Strategy optimisation (very small footprint)
def _es_optimize(
    mean_init: np.ndarray,
    sigma_init: float,
    n: int,
    K: int,
    deadline: float,
    best_val: float = float("inf"),
    best_param: np.ndarray | None = None,
    popsize: int = 20,
) -> tuple[np.ndarray, float]:
    """
    Simple (μ,λ)‑ES on the log‑parameter vector.
    The algorithm keeps only the current best solution (μ=1) and a fixed
    population size λ.  The step‑size σ follows the classic 1/5‑rule.
    """
    mean = mean_init.copy()
    sigma = sigma_init
    conv_fft_len = _next_pow2(2 * n - 1)

    iteration = 0
    while time.time() < deadline and sigma > 1e-4 and iteration < 3000:
        # generate population
        pop = [mean + sigma * np.random.randn(K) for _ in range(popsize)]
        vals = []
        for param in pop:
            a = _shape_from_params(param, n)
            vals.append(_objective(a, n, conv_fft_len))

        # order by objective (smaller is better)
        order = np.argsort(vals)
        pop = [pop[i] for i in order]
        vals = [vals[i] for i in order]

        # update global best
        if vals[0] < best_val - 1e-12:
            best_val = vals[0]
            best_param = pop[0].copy()
            print(f"[ES n={n} K={K} iter={iteration:04d}] new best {best_val:.6f}")

        # success‑rate for the 1/5‑rule
        # success = fraction of candidates that improve over the previous best
        success = sum(v < best_val for v in vals) / popsize

        # adapt σ
        if success > 0.2:
            sigma *= 1.2
        else:
            sigma *= 0.85

        # recompute the centre as the mean of the top 20 % (or just the elite)
        elite_cnt = max(1, popsize // 5)
        elite = np.stack(pop[:elite_cnt])
        mean = elite.mean(axis=0)

        # occasional tiny random shake to avoid dead‑ends
        if iteration % 30 == 0:
            mean += 0.01 * np.random.randn(K)

        iteration += 1

    return best_param, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iter: int = 30000,
) -> np.ndarray:
    """
    Perform the exact max‑convolution sub‑gradient updates used in the
    reference implementation.  The routine stops early when no improvement
    has been seen for several seconds or when the time budget expires.
    """
    a = a0.copy()
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    last_improve = time.time()
    eta0 = 0.6

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[subgrad n={n} it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent step (exponentiated‑gradient on the simplex)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if not improving for a while
        if time.time() - last_improve > 6.0:
            break

    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the given metric.
    The algorithm consists of:

    1. a coarse ES on n≈1024 (K=32),
    2. up‑sampling and fine ES on n≈4096 (K=64),
    3. a short exact sub‑gradient refinement,
    4. an optional final up‑sample to n≈8192 if time permits.

    The best sequence encountered is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -------------------------------------------------------------
    # 1️⃣  Coarse optimisation (n≈1024, K=32)
    n_coarse = 1024
    K_coarse = 32
    sigma_coarse = 1.0

    # warm start: if a previous construction is available, downsample it
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        # interpolate warm array onto the coarse grid

        xs_src = np.arange(warm_arr.size)
        a_warm = np.interp(xs_warm, xs_src, warm_arr)
        a_warm = np.maximum(a_warm, 0.0)
        s = a_warm.sum()
        if s > 0.01:
            a_warm /= s
        else:
            a_warm = np.full(n_coarse, 1.0 / n_coarse)

        init_param_coarse = np.log(a_warm + 1e-12)
    else:

    best_param_coarse, best_val_coarse = _es_optimize(

        deadline=deadline,
        best_val=float("inf"),
        best_param=None,
        popsize=20,
    )

    # -------------------------------------------------------------

    n_fine = 4096

        n=n_fine,

        deadline=deadline,

        best_param=None,
