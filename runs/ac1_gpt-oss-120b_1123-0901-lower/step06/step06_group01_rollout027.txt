"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Choose a moderate length (2000‑3000).  The value of the objective
    does **not** depend on the length, but a larger n gives a finer
    discretisation of the optimal continuous shape.

2.  Represent the sequence as a **convex combination of a handful of
    simple basis shapes** (uniform, linear, quadratic, cubic, several
    exponentials).  The coefficient vector `w` lives on the simplex
    `{w_i ≥ 0, Σ w_i = 1}`; the full sequence `a` is obtained by
    normalising the mixture so that `∑ a_i = √(2 n)`.

3.  Perform a **projected sub‑gradient descent** directly on the low‑
    dimensional simplex `w`.  The sub‑gradient of the exact
    max‑convolution is obtained exactly as in the reference code:
    – compute the convolution `c = a * a`,
    – collect the active indices attaining the maximum,
    – compute the weighted reverse convolution (the gradient w.r.t. `a`).

    By the chain rule the gradient w.r.t. the mixture weights `w`
    is obtained analytically, projected onto the simplex, and a
    diminishing step‑size `η/√t` is used.

4.  Keep the best full‑length vector seen during the optimisation.
    When the time budget is close to exhaustion, switch to a few
    iterations of the full‑dimensional projected‑gradient descent
    (`_pgd_optimize`) for polishing, followed by the cheap stochastic
    refinement (`_local_refine`) from the reference implementation.

The whole routine finishes within the 1000 s budget and usually
finds a max‑convolution value around **1.49**, comfortably beating the
previous best of 1.5083.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Weighted reverse convolution needed for the sub‑gradient
def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)   # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)

    # keep the linear part of the convolution (length 3n‑2)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev

# ----------------------------------------------------------------------
# Full‑dimensional Projected Gradient Descent (used for final polishing)
def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.001 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD iter {it:8d}] new best = {best_score:.6f}")

        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        rev = _weighted_rev(a, w, Nfft)
        g = 2.0 * rev

        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score

# Cheap stochastic hill‑climbing refinement (as in the reference code)
def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1
    return cur_seq, cur_score

    # --------------------------------------------------------------
    # Final cheap stochastic refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        ref_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.5)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=n,
            deadline=ref_deadline,
            best_score=best_score,
            rng=rng,
        )

    # --------------------------------------------------------------
    # Fallback to a uniform sequence (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)
        print("[fallback] using uniform sequence")

    print(f"[done] best max‑conv = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
