"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a∗a) / (∑a)² .

The search proceeds in several phases:

1. **Diverse pool** – a set of candidate distributions of length *n* is
   generated (uniform, triangular, quadratic, double‑exponential,
   Gaussian, sinusoidal, and many random Dirichlet draws).  Every vector is
   projected onto the probability simplex and its exact objective is
   evaluated with an FFT‑based convolution.  The best few (≤ 8) are kept.

2. **Smooth‑max Adam** – each kept candidate is refined by Adam on the
   log‑sum‑exp surrogate of the maximum convolution.  The variables are
   logits of a soft‑max (hence the simplex constraint is satisfied
   automatically).  The temperature *λ* of the smooth‑max is increased
   steadily, driving the surrogate towards the true max.

3. **Exact sub‑gradient phase** – after the Adam phase we switch to a
   mirror‑descent sub‑gradient method that works directly on the true
   max‑convolution.  The update uses exponentiated‑gradient steps
   (multiplicative updates) followed by a projection onto the simplex.

4. **Upsampling & final polishing** – the best distribution is linearly
   interpolated to a larger support (4096 points) and the whole optimisation
   (Adam + sub‑gradient) is repeated.  Finally a tiny Gaussian smoothing
   sweep is performed and the best smoothed version is returned.

All steps are time‑aware: the routine stops as soon as the supplied budget
(≈ 1000 s) is exhausted and returns the best sequence found so far.
The code uses only NumPy (and optionally SciPy for a Gaussian kernel) and
contains no file or network I/O.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective (sum(a) is assumed > 0).
    Because the evaluation function is scale‑invariant we keep sum(a)=1.
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Reference: Duchi et al., “Efficient Projections onto the l1‑Ball”.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Log‑sum‑exp (smooth‑max) weights.
    """
    mx = conv.max()
    w = np.exp(lam * (conv - mx))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return w


def _maybe_gaussian_kernel(sigma: float) -> np.ndarray:
    """Return a 1‑D Gaussian kernel with the given sigma (in index units)."""
    if sigma <= 0:
        return np.array([1.0])
    # kernel size = 6σ rounded up to odd
    k = int(np.ceil(sigma * 6))
    if k % 2 == 0:
        k += 1
    x = np.arange(k) - (k - 1) / 2.0
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    return kernel


def _apply_gaussian(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Convolve seq with a Gaussian kernel of width sigma (same length output)."""
    if sigma <= 0:
        return seq.copy()
    kernel = _maybe_gaussian_kernel(sigma)
    # use 'same' mode to keep length
    return np.convolve(seq, kernel, mode='same')

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors."""
    pool = []

    # warm‑start (if present) and simple rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # 1. Uniform
    pool.append(np.full(n, 1.0 / n))

    # 2. Triangular (linear taper to the edges)
    i = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(i - (n - 1) / 2.0) * 2.0 / (n - 1)
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    pool.append(tri)

    # 3. Quadratic (i·(n‑i) shape)
    prod = (i + 1) * (n - i)
    quad = prod ** 0.5
    quad /= quad.sum()
    pool.append(quad)

    # 4. Cubic‑bump (raised cosine)
    cos_bump = (1 + np.cos(2 * np.pi * (i + 0.5) / n)) / 2.0
    cos_bump = np.maximum(cos_bump, 0.0)
    cos_bump /= cos_bump.sum()
    pool.append(cos_bump)

    # 5. Double‑exponential families
    for lam in np.linspace(0.1, 4.0, 12):
        dexp = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        dexp = np.maximum(dexp, 0.0)
        dexp /= dexp.sum()
        pool.append(dexp)

    # 6. Gaussian bumps with varying width
    for sigma in np.linspace(n * 0.02, n * 0.25, 10):
        gauss = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        gauss /= gauss.sum()
        pool.append(gauss)

    # 7. Single half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    pool.append(sinus)

    # 8. Random Dirichlet draws (wide concentration range)
    for alpha in np.logspace(-2, 0, 14):
        rnd = np.random.dirichlet(np.full(n, alpha))
        pool.append(rnd)

    return pool

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max convolution.
    The variables are logits of a soft‑max, guaranteeing non‑negativity
    and unit‑sum after the soft‑max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # a slightly larger start learning rate
    lam = 0.8                         # start temperature for smooth‑max
    lam_factor = 1.04                 # increase factor (moderately aggressive)

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ------------------------ simplex projection via soft‑max ------------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # a lies on the simplex

        # exact objective for logging and for global bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ---------------------- smooth‑max surrogate ----------------------
        w = _smooth_weights(conv, lam)         # shape (2n‑1)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂/∂a of surrogate
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)           # chain rule through soft‑max

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase smooth‑max temperature
        lam = min(lam * lam_factor, 1e6)

        # occasional random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        # safety bound on iterations (prevent runaway in pathological cases)
        if it > 600_000:
            break

    return global_best_seq, global_best_val


def _subgradient_mirror_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exact max‑convolution sub‑gradient descent using an exponentiated‑gradient
    mirror step (KL‑divergence) followed by a simplex projection.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    eta0 = 0.9                     # initial step size (log‑space)
    sub_last_improve = time.time()

    for sub_it in range(40_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:05d}] new best {val:.6f}")

        # active set (indices achieving the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # adaptive step size – decays like 1/√t
        eta = eta0 / math.sqrt(sub_it + 1)

        # exponentiated‑gradient (mirror‑descent) update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex (numerically stable)
        s = a.sum()
        if s < 1e-12:                    # collapse guard
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 8.0:
            # no improvement for a while → stop early
            break

    return global_best_seq, global_best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑smooth‑max phase followed by the exact sub‑gradient phase.
    The function updates and returns the best sequence/value encountered
    across all calls.
    """
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_mirror_descent(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation metric.  The routine respects the given time‑budget and
    returns the best sequence (as a plain Python list) found before the
    deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Primary support size and warm‑start handling
    # ------------------------------------------------------------------
    n = 1024                                      # first pass (fast)
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # 2. Build a diverse candidate pool
    # ------------------------------------------------------------------
    candidates = _generate_candidate_pool(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored = []  # (value, vector)

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-3:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # keep the most promising starts (up to eight)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]

    # ------------------------------------------------------------------
    # 3. Deep optimisation passes (Adam + exact sub‑gradient)
    # ------------------------------------------------------------------
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation (n={n})")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling to a finer grid
    # ------------------------------------------------------------------
    if time.time() + 15 < deadline:
        n_big = 4096
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the larger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[upsample] starting optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # 5. Tiny Gaussian smoothing sweep (helps flatten peaks)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        sigmas = [0.0, 2.0, 4.0, 6.0, 8.0, 12.0, 16.0]  # in index units
        best_local_seq = best_seq
        best_local_val = best_val
        n_final = best_seq.shape[0]
        conv_len = _next_pow2(2 * n_final - 1)

        for sigma in sigmas:
            smoothed = _apply_gaussian(best_seq, sigma)
            smoothed = np.maximum(smoothed, 0.0)
            if smoothed.sum() == 0:
                continue
            smoothed /= smoothed.sum()
            val = _objective(smoothed, n_final, conv_len)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = smoothed
                print(f"[smooth sigma={sigma:.1f}] new best {val:.6f}")

        best_seq, best_val = best_local_seq, best_local_val

    # ------------------------------------------------------------------
    # 6. Finalise – clip to evaluator limits and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
