def project_onto_simplex(v):
    # ensures v >=0, sum=1
    # algorithm from Duchi: sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Note: ensures sum=1.

def _proj_subgradient_opt(a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val):
    a = a0.copy()
    # ensure sum = 1
    a = np.clip(a, 0, None)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0/n)
    else:
        a = a / total

    best_local_val = global_best_val
    best_local_seq = a.copy()
    # track improvement
    last_improve = time.time()
    for it in range(1, 500000): # large bound
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)  # a sum = 1
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[proj subgrad n={n} iter={it}] new best {val:.6f}")
        # active mask
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            # maybe due to floating compute; set w all zeros? but use random?
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # step size
        eta = 0.5 / np.sqrt(it)  # can adjust
        a_new = a - eta * grad
        # projection onto simplex
        a = _project_simplex(a_new)
        # Symmetrize
        a = (a + a[::-1]) / 2.0
        # optionally re-project onto simplex again to maintain sum 1 (symmetrization preserves sum)
        # continue
        if time.time() - last_improve > 5.0:
            break
    return global_best_seq, global_best_val

_Projection function:

def _project_simplex(v):
    """Return the projection of v onto the probability simplex."""
    # algorithm from Duchi et al 2008.
    v = v.astype(np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _random_local_search(a, n, conv_fft_len, deadline, global_best_seq, global_best_val):
    # We'll run until deadline or max steps.
    a_curr = a.copy()
    conv_curr = _full_convolution(a_curr, conv_fft_len)
    best_val_local = global_best_val
    last_improve = time.time()
    steps = 0
    while time.time() < deadline and steps < 50000:
        # choose random indices distinct
        i, j = np.random.choice(n, size=2, replace=False)
        # propose moving epsilon from i to j
        eps = (np.random.rand() * 0.02) * a_curr[i]  # up to 2% of mass at i
        if eps < 1e-12:
            continue
        a_new = a_curr.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # ensure non-neg
        if a_new[i] < 0:
            continue
        # compute new conv via FFT
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        # sum = 1 by construction
        if val_new < best_val_local:
            best_val_local = val_new
            a_curr = a_new
            conv_curr = conv_new
            global_best_seq = a_new.copy()
            global_best_val = val_new
            last_improve = time.time()
            print(f"[local search step {steps}] new best {val_new:.6f}")
        steps += 1
        if time.time() - last_improve > 3.0:
            break
    return global_best_seq, global_best_val

def _smoothmax_adam(a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val):
    eps = 1e-15
    logits = np.log(a0 + eps)
    # symmetrize logits initially
    logits = (logits + logits[::-1]) / 2.0
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.015
    it=0
    last_improve = time.time()
    best_seq = global_best_seq
    best_val = global_best_val

    while time.time() < deadline:
        # compute a via softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # symmetrize a just for evaluation? Actually a from logits is already symmetric due to symmetric logits; but due to numerical error we can symmetrize anyway:
        a = (a + a[::-1]) / 2.0
        # ensure sum=1 (still)
        # compute objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        # track best
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it}] new best {val:.6f}")

        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum) or np.isinf(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # with softmax chain rule: grad logits = a * (grad_a - avg)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # symmetrize logits
        logits = (logits + logits[::-1]) / 2.0

        # increase lambda
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter if no improvement for some time
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 500_000:
            break

    return best_seq, best_val

        a /= total

    best_val = global_best_val

    last_improve = time.time()
    for it in range(1, 200_000):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjSubgrad n={n} iter={it}] new best {val:.6f}")

        # active mask
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            # all entries far? fallback uniform weights
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = 0.6 / np.sqrt(it)  # step size

        # project onto simplex

        if time.time() - last_improve > 6.0:

"""
Hybrid optimiser for the step‑function evaluation problem.

The algorithm builds a diverse pool of initial distributions (uniform,
power‑law, double‑exponential, Gaussian, sinusoid, Dirichlet draws and an
optional warm‑start).  The best few are refined by three complementary
phases that respect the simplex constraint and enforce symmetry:

1. **Adam on a smooth‑max surrogate** – a log‑sum‑exp approximation of the
   max‑convolution is minimised.  Logits are kept symmetric, guaranteeing a
   symmetric step‑function at every iteration.

2. **Projected sub‑gradient descent on the exact max** – the true max‑convolution
   gradient is computed, an adaptive step size is used and the iterate is
   projected onto the probability simplex (exact Euclidean projection).
   After each step the vector is symmetrised.

3. **Stochastic local refinement** – small random mass transfers between
   positions are evaluated with a cheap FFT convolution; improvements are
   accepted, providing a fine‑grained escape from plateaus.

If time permits the final distribution is interpolated to a larger support
(8192) and the three phases are repeated, often yielding a further drop in the
objective.  All numbers are kept non‑negative and the routine respects the
supplied wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length() if x > 0 else 0)


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smoothmax_adam(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    deadline: float,
                    global_best_seq: np.ndarray | None,
                    global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Symmetric logits are maintained throughout.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    logits = (logits + logits[::-1]) / 2.0          # enforce symmetry

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.018

    best_seq = global_best_seq
    best_val = global_best_val

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # soft‑max → simplex, already symmetric because logits are symmetric
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep logits symmetric
        logits = (logits + logits[::-1]) / 2.0

        # raise temperature – approach the true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 500_000:          # safety bound
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _proj_subgradient_opt(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          global_best_seq: np.ndarray | None,
                          global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    After each gradient step the iterate is projected onto the probability
    simplex and symmetrised.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    a = (a + a[::-1]) / 2.0      # enforce symmetry

    best_seq = global_best_seq
    best_val = global_best_val

    last_improve = time.time()
    for it in range(1, 200_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjSubgrad n={n} iter={it:06d}] new best {val:.6f}")

        # active set of constraints (where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = 0.6 / np.sqrt(it)
        a = _project_simplex(a - eta * grad)

        # preserve symmetry
        a = (a + a[::-1]) / 2.0

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _random_local_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         global_best_seq: np.ndarray | None,
                         global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Stochastic fine‑tuning: move a tiny amount of mass from one index to another.
    Each trial is evaluated with a fresh FFT convolution; accepted moves are kept.
    """
    a = a0.copy()
    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    steps = 0

    # pre‑compute a mask of indices that currently hold non‑negligible mass
    while time.time() < deadline and steps < 30_000:
        # pick a source index with some mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n)  # destination can be any index
        if i == j:
            continue

        # move up to 5 % of the source mass
        eps = (np.random.rand() * 0.05) * a[i]
        if eps <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()

        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
            print(f"[Local n={n} step={steps:05d}] new best {val_new:.6f}")

        steps += 1
        if time.time() - last_improve > 4.0:
            break

    return best_seq, best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of starting vectors (including arcsine‑type shape)."""
    pool = []

    # warm‑start and cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1)               # i·(n‑i+1)

    # power‑law edge‑concentrated families
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    sinusoid /= sinusoid.sum()
    pool.append(sinusoid)

    # arcsine‑type shape (heavy edges, light centre)
    x = (np.arange(n) + 0.5) / n
    arcsine = 1.0 / np.sqrt(x * (1.0 - x) + 1e-12)
    arcsine = np.clip(arcsine, 0.0, None)
    arcsine /= arcsine.sum()
    pool.append(arcsine)

    # random Dirichlet draws with varying concentration
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the supplied metric.
    The routine respects the supplied wall‑clock budget and returns the best
    sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Primary support size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start (if a previous construction is present)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # Build candidate pool and quick screen
    candidates = _generate_candidate_pool(n, w)
    scored = []                     # (objective, normalised vector)
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:               # degenerate – redraw
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[candidate {idx:03d}] obj={val:.6f}")
        if time.time() >= deadline:
            break

    # Keep the best few for deep optimisation
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:6]]

    # ------------------------------------------------------------------
    # Deep optimisation passes (Adam → projected sub‑gradient → local search)
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching Adam phase")
        best_seq, best_val = _smoothmax_adam(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching projected sub‑gradient")
        best_seq, best_val = _proj_subgradient_opt(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching local refinement")
        best_seq, best_val = _random_local_search(
            best_seq, n, conv_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger grid
    if time.time() + 15 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation of the best 4k‑vector to 8k
        x_small = np.linspace(0, 1, n, endpoint=False)
        x_big   = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq if best_seq is not None else np.full(n, 1.0 / n))
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[big] starting Adam phase on enlarged grid")
        best_seq_big, best_val_big = _smoothmax_adam(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, None, float("inf")
        )
        if time.time() >= deadline:
            # time ran out – keep whatever we have
            pass
        else:
            print("[big] starting projected sub‑gradient")
            best_seq_big, best_val_big = _proj_subgradient_opt(
                best_seq_big, n_big, conv_fft_len_big, corr_fft_len_big,
                deadline, best_seq_big, best_val_big
            )
            if time.time() < deadline:
                print("[big] starting local refinement")
                best_seq_big, best_val_big = _random_local_search(
                    best_seq_big, n_big, conv_fft_len_big,
                    deadline, best_seq_big, best_val_big
                )
        # Keep the better of the two resolutions
        if best_seq_big is not None and best_val_big < best_val:
            best_seq, best_val = best_seq_big, best_val_big

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
