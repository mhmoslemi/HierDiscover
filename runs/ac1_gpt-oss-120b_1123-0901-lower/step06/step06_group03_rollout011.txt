def _final_subgradient(a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val):
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    # parameters: max subiterations = 20000
    sub_it = 0
    last_improve = time.time()
    while time.time() < deadline and sub_it < 20000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[final subgrad n={n} iter={sub_it}] new best {val:.6f}")

        # compute active mask
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = 0.6 / np.sqrt(sub_it + 1)
        a = a * np.exp(-eta * grad_a)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a = a / a_sum
        # Stall detection
        if time.time() - last_improve > 5.0:
            break
        sub_it += 1
    return global_best_seq, global_best_val

"""
Multi‑start smooth‑max L‑BFGS with continuation and final sub‑gradient polish.

The search proceeds as follows:

1. Build a diverse pool of candidate step‑functions (uniform, power‑law,
   double‑exponential, Gaussian bumps, sinusoid, Dirichlet draws and an
   optional warm‑start).  Every candidate is normalised to unit sum.

2. Keep the K best candidates (by the exact objective) and, for each of them,
   run a continuation L‑BFGS optimiser:
   - Parameters are the logits of a soft‑max (guaranteeing non‑negativity
     and unit sum).
   - The objective is a smooth‑max surrogate of the max‑convolution,
     using the log‑sum‑exp trick for numerical stability.
   - After each L‑BFGS run the temperature λ is increased (continuation)
     so the surrogate approaches the true max.
   - The best exact solution found so far is stored globally.

3. A short mirror‑descent (exponentiated‑gradient) sub‑gradient phase
   refines the best solution on the exact max‑convolution.

4. If time permits the best distribution is interpolated to a larger grid
   (8192 points) and steps 2‑3 are repeated.

All numbers are forced to be non‑negative and bounded by 1000, and the
routine respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the target metric (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a heterogeneous set of start vectors (includes a parametric sweep)."""
    pool = []

    # Warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform distribution
    pool.append(np.full(n, 1.0 / n))

    # Power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)                     # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # Single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Random Dirichlet draws with various concentration parameters
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool

class _LbfgsSmoothMax:
    """
    Callable object for scipy.optimize.minimize.

    Implements the smooth‑max surrogate of max(conv) together with its gradient
    with respect to the logits of a soft‑max parametrisation.
    """

    def __init__(self, n: int, conv_fft_len: int, corr_fft_len: int, lam: float):
        self.n = n
        self.conv_fft_len = conv_fft_len
        self.corr_fft_len = corr_fft_len
        self.lam = lam
        self.eps = 1e-15

    def __call__(self, logits: np.ndarray):
        """Return (scalar objective, gradient w.r.t. logits)."""
        # soft‑max → feasible a (unit sum, non‑negative)
        mx = logits.max()
        exp_shift = np.exp(logits - mx)
        a = exp_shift / exp_shift.sum()

        # linear convolution a*a
        conv = _full_convolution(a, self.conv_fft_len)

        # smooth‑max via log‑sum‑exp trick
        max_c = conv.max()
        exp_w = np.exp(self.lam * (conv - max_c))
        w_sum = exp_w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
            smooth_max = max_c
        else:
            w = exp_w / w_sum
            smooth_max = max_c + np.log(w_sum) / self.lam

        obj = 2.0 * self.n * smooth_max

        # gradient of surrogate w.r.t. a
        # ∇_a smooth_max = 2 * (w ★ a)
        grad_a_smooth = 2.0 * _cross_correlation(w, a, self.corr_fft_len)

        # gradient of the full objective w.r.t. a: d(2n*smooth_max)/da
        grad_a = 4.0 * self.n * _cross_correlation(w, a, self.corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        return float(obj), grad_logits.astype(np.float64)


# ----------------------------------------------------------------------
def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Continuation L‑BFGS on the smooth‑max surrogate.
    The temperature λ is increased geometrically; after each phase the exact
    objective is evaluated and the global best is updated.
    """
    # initialise logits from the starting vector
    logits = np.log(a0 + 1e-15)

    # continuation schedule
    lam = 0.5
    lam_factor = 1.2
    lam_max = 1e6

    while time.time() < deadline and lam <= lam_max:
        # build the callable for the current λ
        fun = _LbfgsSmoothMax(n, conv_fft_len, corr_fft_len, lam)

        # L‑BFGS options – keep the number of function evaluations modest
        options = {"maxiter": 500, "maxfun": 500, "disp": False}

        # run the optimiser
        res = scipy.optimize.minimize(fun, logits, method="L-BFGS-B", jac=True,
                                      options=options)

        logits = res.x

        # evaluate the exact objective for logging / global update
        a = np.exp(logits - logits.max())
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[L‑BFGS n={n} λ={lam:.2e}] new best {val:.6f}")

        # increase temperature for the next continuation step
        lam = min(lam * lam_factor, lam_max)

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _final_subgradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Short mirror‑descent (exponentiated‑gradient) phase that works on the exact
    max‑convolution.  It starts from ``a0`` and stops after a few seconds of
    stagnation or when the time budget expires.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    eta0 = 0.6
    last_improve = time.time()

    for it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[subgrad n={n} iter={it}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent step on simplex (exponentiated gradient)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # stop if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine respects the supplied time‑budget and returns the best sequence
    encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # primary support size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm‑start from a previously discovered construction
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, np.ndarray)):
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            # linear interpolation to the required length
            x_old = np.linspace(0, 1, w.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            w = np.interp(x_new, x_old, w)
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # build a diverse pool of start vectors
    candidates = _generate_candidate_pool(n, w)

    # add a bunch of extra random Dirichlet draws for extra diversity
    for _ in range(200):
        cand = np.random.dirichlet(np.full(n, 0.2))
        candidates.append(cand)

    # evaluate all candidates quickly (exact objective) and keep the best few
    scored = []
    for cand in candidates:
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-8:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))

    scored.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [vec for _, vec in scored[:top_k]]

    # ------------------------------------------------------------------
    # global best trackers
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # optimisation of the top candidates with L‑BFGS + sub‑gradient polish
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[start L‑BFGS run {cid}/{len(top_candidates)-1}]")
        best_seq, best_val = _lbfgs_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        # final exact sub‑gradient polishing
        if time.time() < deadline:
            best_seq, best_val = _final_subgradient(
                best_seq if best_seq is not None else a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )

    # ------------------------------------------------------------------
    # optional up‑sampling to a larger grid if time permits
    if time.time() + 20 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the larger grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[upsample] running L‑BFGS on enlarged support")
        best_seq, best_val = _lbfgs_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if time.time() < deadline:
            best_seq, best_val = _final_subgradient(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                best_seq,
                best_val,
            )

    # ------------------------------------------------------------------
    # fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Clamp to evaluator’s allowed range and return as a plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
