def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    
    lengths = [1024, 2048, 4096, 8192]
    
    # Precompute FFT lengths:
    fft_len_conv, fft_len_corr = {}, {}
    for n in lengths:
        conv_len = 2*n -1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n -1)))
    
    for n in lengths:
        if time.time() > deadline:
            break
        print(f'Optimising length {n}')
        conv_len = fft_len_conv[n]; corr_len = fft_len_corr[n]
        
        # build candidates
        candidates = _build_initial_candidates(n)
        # add padded previous best if exists
        if best_seq is not None:
            # pad/truncate to length n
            cand_prev = np.zeros(n)
            old_n = len(best_seq)
            if old_n >= n:
                cand_prev = np.asarray(best_seq[:n])
            else:
                cand_prev[:old_n] = best_seq
                # maybe fill rest with zeros
            # normalize
            if cand_prev.sum() < 1e-12:
                cand_prev = np.random.dirichlet(np.ones(n))
            else:
                cand_prev /= cand_prev.sum()
            candidates.append(cand_prev)
            
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break
            # initial a sum=1
            a = a0 / a0.sum()
            # apply symmetrization
            a = 0.5 * (a + a[::-1])
            # start logits
            logits = np.log(a + 1e-15)
            
            # Adam parameters
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            # learning-rate scaling
            base_lr = 0.15 
            lr = base_lr * np.sqrt(1024.0 / n)
            # temperature schedule
            lam = 0.3
            lam_factor = 1.02
            
            last_improve_time = time.time()
            max_iters = int(80000 * 1024 / n)  # more iterations for smaller n
            max_iters = max(max_iters, 2000)
            # Guarantee not too large for larger n; maybe limit to 20000 for n=8192
            max_iters = min(max_iters, 40000)
            
            for it in range(max_iters):
                if time.time() > deadline:
                    break
                # compute a from logits
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                # impose symmetry after each step
                a = 0.5 * (a + a[::-1])
                # convolution and weight
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # compute smooth max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum
                # compute gradient via correlation
                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                # gradient wrt logits
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                
                # Adam update
                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
                m_hat = m / (1 - beta1 ** (it+1))
                v_hat = v / (1 - beta2 ** (it+1))
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
                
                # optionally clip logits to  range
                # maybe not needed
                
                # increase temperature
                lam = min(lam * lam_factor, 1e5)
                
                # compute objective using direct formula (since sum=1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve_time = time.time()
                    print(f"[n={n} cand={cid} iter={it}] new best {val:.6f}")
                
                # jitter if stagnated
                if time.time() - last_improve_time > 8.0:
                    # small random perturbation at logits
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve_time = time.time()
                    
            # subgradient phase (exponentiated gradient)
            # start from current a (already normalized) 
            a = a / a.sum()
            eta = 0.5
            sub_last = time.time()
            max_sub_iters = int(10000 * 1024 / n) + 2000
            max_sub_iters = min(max_sub_iters, 20000)
            for sub_it in range(max_sub_iters):
                if time.time() > deadline:
                    break
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # active indices near max
                active = conv >= max_c - 1e-12
                if not active.any():
                    w = np.zeros_like(conv)
                else:
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                # exponentiated gradient update
                a = a * np.exp(-eta * grad_a)
                # renormalize to sum=1
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s
                # symmetrize
                a = 0.5 * (a + a[::-1])
                # evaluate
                val = 2.0 * n * max_c
                if val < best_val - 1e-9:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cid} subgrad={sub_it}] new best {val:.6f}")
                if time.time() - sub_last > 10.0:
                    break
        # after candidate loop, optionally fine-tune best for this length further
        if best_seq is not None and len(best_seq) == n:

"""
Propose a non‑negative step function with a low value of

        F(a) = 2·n·max(a*a) / (sum(a)²)

The implementation follows these ideas:

* A rich pool of deterministic and random initial shapes (including
  Dirichlet draws, power‑law “U‑shapes”, exponential tails,
  cosine / Gaussian windows and symmetric random spikes).
* A three‑stage optimisation for each candidate:
  1. Adam on a smooth‑max surrogate (temperature grows exponentially).
  2. Exponentiated‑gradient (sub‑gradient) phase on the true max.
  3. A final fine‑tuning Adam pass with a very high temperature.
* Every iteration a symmetric copy of the sequence is enforced
  (a[i] = a[n‑1‑i]), which matches the empirical structure of good minima
  and reduces the effective dimensionality.
* Stagnation is broken by a tiny random jitter of the logits.
* The algorithm respects the supplied time budget (default 1000 s) and
  stops as soon as the target value 1.503 is reached.

The code is self‑contained – only NumPy and the already‑imported
`evaluate_sequence` are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution a*a and the cross‑correlation needed for
# the gradient of the smooth‑max surrogate.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of a with itself using an FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified list of starting points for the optimisation.
    All candidates are normalised to sum to 1.
    """
    cand = []

    # 1) Dirichlet (uniform and concentrated)
    cand.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8, 1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 2) Edge‑concentrated power‑law (U‑shapes)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 13):
        shape = np.where(prod > 0,
                         np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod)),
                         0.0)
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 3) Two‑sided exponential decay
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.0005, 0.001, 0.002, 0.004):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # 4) Cosine / sine windows (single bump)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    cand.append(coswin / coswin.sum())
    sinwin = np.sin(np.pi * i / (n - 1))
    cand.append(sinwin / sinwin.sum())

    # 5) Discrete Gaussian (σ scaled with n)
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    cand.append(gauss / gauss.sum())

    # 6) Truncated arcsine (inverse‑sqrt edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    cand.append(arcs / arcs.sum())

    # 7) Random sparse symmetric spikes (≈√n spikes)
    m_spikes = max(2, int(np.sqrt(2 * n)))
    base = np.zeros(n, dtype=np.float64)
    half = (n + 1) // 2
    # pick indices only from the first half and mirror them
    idxs = np.random.choice(half, size=m_spikes, replace=False)
    values = np.random.rand(m_spikes) + 0.1
    base[idxs] = values
    base[-np.array(idxs) - 1] = values  # mirror
    if base.sum() > 0:
        cand.append(base / base.sum())

    # 8) Linear tent shape (rise‑fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:   # odd size – add centre
        tent = np.concatenate([tent, [0.0]])
    cand.append(np.maximum(tent, 0) / tent.sum())

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    The routine respects the given time budget and returns the best sequence
    discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")
    target_val = 1.503   # stop early if we beat this bound

    # Lengths to explore – the larger the length, the more expressive the shape.
    lengths = [1024, 2048, 4096, 8192]

    # Pre‑compute FFT lengths (next power of two) for each problem size.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # ---- build a diversified list of initial candidates -----------------
        candidates = _build_initial_candidates(n)

        # add the current global best (padded or truncated) as an extra seed
        if best_seq is not None:
            old = np.asarray(best_seq, dtype=np.float64)
            if old.shape[0] >= n:
                cand = old[:n].copy()
            else:
                cand = np.pad(old, (0, n - old.shape[0]))
            # normalise – if the sum vanished we fallback to a random Dirichlet
            if cand.sum() < 1e-12:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            candidates.append(cand)

        # ------------------------------------------------------------------
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # initialise on the simplex via logits (soft‑max parametrisation)
            a = a0 / a0.sum()
            a = 0.5 * (a + a[::-1])                # enforce symmetry
            logits = np.log(a + 1e-15)

            # --------------------------------------------------------------
            # Phase 1 – Adam on a smooth‑max surrogate
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)

            # learning‑rate scales mildly with problem size
            lr = 0.12 * np.sqrt(1024.0 / n)
            lam = 0.25                # initial temperature
            lam_factor = 1.020        # geometric growth

            last_improve = time.time()
            # iteration budget that roughly equals 2·10⁴·1024/n (capped)
            max_iters = int(20000 * 1024 / n) + 4000
            max_iters = min(max_iters, 30000)

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # soft‑max conversion
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                a = 0.5 * (a + a[::-1])       # keep symmetry

                # convolution and smooth‑max weights
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate (linear in w)
                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule – approach the true max
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a)=1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t:05d}] new best {val:.6f}")
                    if best_val <= target_val:
                        print("Target reached – exiting early.")
                        return best_seq.tolist()

                # if stuck, add a tiny random jitter to logits
                if time.time() - last_improve > 8.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

            # --------------------------------------------------------------
            # Phase 2 – Exponentiated‑gradient (exact sub‑gradient) phase
            a = a / a.sum()
            eta = 0.55
            sub_last_improve = time.time()
            max_sub_iters = int(8000 * 1024 / n) + 2500
            max_sub_iters = min(max_sub_iters, 12000)

            for sub_it in range(max_sub_iters):
                if time.time() > deadline:
                    break
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # active indices where the max is attained (tight constraints)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # exponentiated‑gradient update (mirror descent on the simplex)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # avoid collapse to zero
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s
                a = 0.5 * (a + a[::-1])           # enforce symmetry again

                val = 2.0 * n * max_c
                if val < best_val - 1e-9:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} Subgrad iter={sub_it}] new best {val:.6f}")
                    if best_val <= target_val:
                        print("Target reached – exiting early.")
                        return best_seq.tolist()

                if time.time() - sub_last_improve > 10.0:
                    break   # stagnated → stop sub‑gradient phase

        # ------------------------------------------------------------------
        # Final fine‑tuning of the *global* best for this length (if it matches n)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07 * np.sqrt(1024.0 / n)
            lam = 0.45
            lam_factor = 1.015

            fine_iters = min(fine_iters, 80000)

                if time.time() > deadline:

                a = exp_x / exp_x.sum()
                a = 0.5 * (a + a[::-1])

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()

                    logits += (np.random.rand(n) - 0.5) * 2e-3
                    last_improve = time.time()

    # Fallback – uniform vector if everything failed (should never happen).
    if best_seq is None:

        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    return best_seq.tolist()
