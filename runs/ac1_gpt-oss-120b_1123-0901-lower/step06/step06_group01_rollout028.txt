            cur_score = new_score

            T = max(cur_score*0.01, 1e-6) * (1 - t/n_iters)

                cur_score = new_score

"""
A three‑stage optimiser for the ``evaluate_sequence`` objective.

Stage 1 – a cheap parametric sweep.
    The family a_i ∝ (min(i, n‑1‑i)+1)^p concentrates mass at the
    ends of the interval.  For several lengths and many exponents p
    we evaluate the exact objective and keep the best vector.

Stage 2 – projected‑gradient descent (exact max‑convolution).
    Starting from the best vector of stage 1 we run a sub‑gradient
    descent with Euclidean projection onto the simplex Σa_i = √(2 n).
    This refines the shape and typically improves the bound by a few
    percent.

Stage 3 – cheap stochastic hill‑climbing.
    Small random mass transfers are accepted if they improve the
    objective (or with a simulated‑annealing probability).  This
    polishing step is fast and often yields the final gain.

The whole routine respects the supplied time budget (default 1000 s)
and returns a list of non‑negative heights that minimises
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x | x_i ≥ 0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a * a (linear convolution) using one FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Weighted reverse convolution needed for the sub‑gradient:

        rev_i = Σ_k w_k * a_{k‑i}
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


# ----------------------------------------------------------------------
# Shape generators ------------------------------------------------------


def _distance_power_shape(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law shape
        a_i ∝ (min(i, n‑1‑i)+1)^p .
    p is typically negative, which concentrates mass near the ends.
    """
    dist = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
    base = dist.astype(float) ** p
    # avoid NaNs when p is very negative and dist is zero (not possible here)
    base = np.where(np.isfinite(base), base, 0.0)
    target_sum = math.sqrt(2.0 * n)
    a = base / base.sum() * target_sum
    return a


def _exponential_shape(n: int, rate: float) -> np.ndarray:
    """Simple exponential decay a_i ∝ exp(-rate·i)."""
    x = np.exp(-rate * np.arange(n))
    target_sum = math.sqrt(2.0 * n)
    a = x / x.sum() * target_sum
    return a

# Optimisers -----------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected Gradient Descent on the exact max‑convolution."""
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.001 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current max‑convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # active set (hard max)
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # sub‑gradient via weighted reverse convolution
        rev = _weighted_rev(a, w, Nfft)
        g = 2.0 * rev

        # gradient step + projection
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    move a small fraction of mass from i to j.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 200_000
    step = 0

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Stage 1 – parametric sweep
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400, 2800, 3200]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[stage‑1] exploring length {n}")

        # 1) distance‑power family
        for p in np.linspace(-2.0, -0.05, 36):   # 36 values ≈ step 0.054
            if time.time() >= deadline:
                break
            a = _distance_power_shape(n, p)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                print(f"  new best (power p={p:.3f}) = {best_score:.6f}")

        # 2) simple exponential decays (different rates)
        for rate in np.linspace(0.003, 0.02, 18):
            if time.time() >= deadline:
                break
            a = _exponential_shape(n, rate)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                print(f"  new best (exp rate={rate:.4f}) = {best_score:.6f}")

        # 3) a few random Dirichlet vectors (just to diversify)
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = rng.gamma(shape=1.0, scale=1.0, size=n)
            a = a / a.sum() * math.sqrt(2.0 * n)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                print(f"  new best (random) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected gradient descent refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_n
        remaining = deadline - time.time()
        # crude estimate: ~0.0006 s per PGD iteration for n≈2000
        it_per_sec = int(1.6 / 0.0006)  # around 1600 it/s
        max_iters = int(min(1_500_000, remaining * it_per_sec))
        max_iters = max(150_000, max_iters)   # guarantee a reasonable effort
        print(f"[stage‑2] PGD refinement: n={n}, max_iters={max_iters}")

        best_seq, best_score = _pgd_optimize(
            a0=best_seq.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        print(f"[stage‑3] local stochastic refinement (until {refine_deadline:.2f})")
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best value = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
