"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Parameterised families** – generate candidate shapes from a few
   low‑dimensional families (exponential, power‑law, shifted power‑law,
   symmetric beta‑type).  For each length we normalise the vector to the
   target sum √(2 n) and compute the exact objective using one FFT.
   The best vector seen so far is kept.

2. **Local stochastic refinement** – starting from the best candidate,
   repeatedly move a random fraction of mass from one index to another.
   Improvements are always accepted; worsening moves are accepted with a
   simulated‑annealing probability.  This cheap hill‑climber is able to
   escape plateaux and further lower the maximum of the self‑convolution.

3. **Projected Gradient Descent (optional)** – a few PGD iterations on the
   exact max‑convolution are performed as a final polish step.

The whole search respects the 1000 s budget, checking the deadline after
each candidate and each refinement step.  The final sequence (plain
Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σ x = target_sum} (Duchi et al.)."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a * a (linear convolution) using one FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected Gradient Descent on the exact max‑convolution."""
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    a = a * target_sum / a.sum()
    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.001 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        mask = conv >= M - eps_active
        cnt = mask.sum()
        w = mask.astype(float) / max(cnt, 1.0)

        # weighted reverse convolution (sub‑gradient)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev  # sub‑gradient of max(a∗a)

        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex.  Randomly move a
    small fraction of mass from i to j, always accept improvements and
    accept worsening moves with a cooling simulated‑annealing schedule.
    """
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 300_000
    step = 0

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # move 1‑5 % of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1
    return cur_seq, cur_score

            M = float(conv.max())
            s = float(vec.sum())
            return 2.0 * n * M / (s * s)

        def consider(candidate):
            """Evaluate candidate, update global best if improved."""
            nonlocal best_seq, best_score
            score = obj_of(candidate)
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f"[new best] length {n:4d}  score {score:.6f}")

        # ------------------ exponential decay -------------------------
        for lam in np.linspace(0.0005, 0.02, 30):
            if time.time() >= deadline:
                break
            a = np.exp(-lam * np.arange(n, dtype=float))
            a = a / a.sum() * target_sum
            consider(a)

        # ------------------- pure power law -------------------------
        for p in np.linspace(0.30, 0.80, 25):
            if time.time() >= deadline:
                break
            i = np.arange(1, n + 1, dtype=float)
            a = i ** (-p)
            a = a / a.sum() * target_sum
            consider(a)

        # ------------- shifted power‑law (two parameters) ------------
        alphas = np.logspace(-1, 2, 20)       # 0.1 … 100
        for alpha in alphas:
            if time.time() >= deadline:
                break
            for p in np.linspace(0.35, 0.65, 15):
                if time.time() >= deadline:
                    break
                i = np.arange(n, dtype=float)
                a = (i + alpha) ** (-p)
                a = a / a.sum() * target_sum
                consider(a)

        # ------------------- symmetric beta‑type --------------------
        for a_par in np.linspace(0.0, 5.0, 11):
            if time.time() >= deadline:
                break
            x = np.arange(n, dtype=float) / (n - 1)
            a = (x ** a_par) * ((1 - x) ** a_par) + 1e-12
            a = a / a.sum() * target_sum
            consider(a)

    # ------------------------------------------------------------------
    # Local stochastic refinement on the best sequence (if any)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # use at most half of the remaining time for refinement
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # (Optional) short PGD polishing – a few thousand iterations are cheap
    if best_seq is not None and time.time() < deadline:
        try:
            best_seq, best_score = _pgd_optimize(
                a0=best_seq,
                n=best_seq.shape[0],
                deadline=deadline,
                max_iters=200_000,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
            )
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Fallback to uniform if everything failed (unlikely)
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = obj_of(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
