"""
A fast multi‑scale evolutionary optimiser for the step‑function problem.

The algorithm works in three stages:

1. **Exploratory GA** – a small population (≈30) of diverse
   candidates (warm‑start, Dirichlet draws, edge‑biased power laws,
   double‑Gaussian edges, sinusoidal half‑waves, uniform and spike
   patterns) is evolved with log‑it mutation, Dirichlet mixing and
   simple crossover.  The best individual is kept after each generation.
2. **Exact‑max sub‑gradient refinement** – the best GA individual is
   refined by exponentiated‑gradient (multiplicative) updates on the
   true objective  `2·n·max(a*a)/(sum a)²`.  The update uses the active
   set of indices attaining the current maximum and a cross‑correlation
   to obtain a sub‑gradient.
3. **Graduated non‑convexity** – the procedure is repeated on a
   geometrically increasing grid size (2048 → 4096 → 8192) by linear
   interpolation of the current best sequence.  Each refinement gets
   its own time slice; any remaining budget is spent on a final
   sub‑gradient pass.

All operations are performed with FFT‑based convolutions (`O(n log n)`);
the code respects the supplied time budget and always returns a
non‑negative sequence.  Progress and current best values are printed
to the console.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by the optimiser:
        F(a) = 2·n·max(a*a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Candidate generators (warm‑start, random, analytic shapes)
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest, diverse pool of start vectors."""
    cand = []

    # warm start (if supplied) and its rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # simple positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike patterns (equally spaced non‑zero entries)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Simple evolutionary optimiser (GA)
# ----------------------------------------------------------------------
def _evolve_population(
    pop: list[np.ndarray],
    n: int,
    deadline: float,
    elite_size: int = 5,
    offspring_per_elite: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Evolutionary loop with mutation on log‑it, Dirichlet mixing and
    occasional crossover.  Returns the best vector seen and its objective.
    """
    # initialise scores
    scores = [_objective(a) for a in pop]
    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    sigma = 0.5          # log‑it noise magnitude
    mix_alpha = 0.4      # weight of random Dirichlet mixing
    it = 0

    while time.time() < deadline:
        # ----- selection -----
        order = np.argsort(scores)[:elite_size]
        elite = [pop[i] for i in order]

        # ----- generate offspring -----
        new_pop = []
        for parent in elite:
            for _ in range(offspring_per_elite):
                # log‑it mutation
                logit = np.log(parent + 1e-12) + sigma * np.random.randn(n)
                child = np.exp(logit)
                child /= child.sum()

                # Dirichlet mixing (helps exploration)
                if np.random.rand() < 0.5:
                    rand = np.random.dirichlet(np.ones(n))
                    child = (1.0 - mix_alpha) * child + mix_alpha * rand

                # occasional crossover with another elite member
                if np.random.rand() < 0.3:
                    partner = elite[np.random.randint(elite_size)]
                    lam = np.random.rand()
                    child = lam * child + (1.0 - lam) * partner

                child = np.maximum(child, 0.0)
                child /= child.sum()
                new_pop.append(child)

        # fill the rest of the population with fresh Dirichlet draws
        while len(new_pop) < len(pop):
            new_pop.append(np.random.dirichlet(np.ones(n)))

        # evaluate new population
        scores = [_objective(a) for a in new_pop]

        # update global best
        cur_best = int(np.argmin(scores))
        cur_score = scores[cur_best]
        if cur_score < best_score:
            best_score = cur_score
            best_a = new_pop[cur_best].copy()
            print(f"[GA iter={it:05d}] new best {best_score:.6f}")

        # replace old population
        pop = new_pop

        # slowly decay exploration parameters
        sigma *= 0.996
        mix_alpha *= 0.996
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient optimisation (multiplicative updates)
# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact objective.
    Returns the best vector encountered and its objective value.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad iter={it:05d}] new best {val:.6f}")

        # active set where the maximum is attained
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # project back onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------

            print(f"[Scale {cur_n}] GA best {best_score:.6f}")

        # ------------------------------------------------------------------
        # Exact‑max sub‑gradient refinement (as much time as left)
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            conv_fft_len = _next_pow2(2 * cur_n - 1)
            corr_fft_len = _next_pow2(3 * cur_n - 2)
            ref_a, ref_score = _subgradient_optimize(
                best_a,
                cur_n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=200_000,
                eta=0.6,
            )
            if ref_score < best_score:
                best_score, best_a = ref_score, ref_a.copy()
                print(f"[Scale {cur_n}] Subgrad best {best_score:.6f}")

        # ------------------------------------------------------------------
        # Decide whether to move to a finer grid
        # ------------------------------------------------------------------
        if cur_n >= 8192 or time.time() + 30.0 > deadline:
            break
        cur_n = min(8192, cur_n * 2)

    # ------------------------------------------------------------------
    # Final sanitisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s < 0.01:
        best_a = np.random.dirichlet(np.ones(best_a.shape[0]))
        s = best_a.sum()
    best_a = best_a / s
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Final best approx value {best_score:.6f} (length {len(best_a)})")
    return best_a.tolist()
