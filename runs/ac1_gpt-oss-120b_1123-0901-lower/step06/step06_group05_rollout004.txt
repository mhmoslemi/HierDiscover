"""
Improved search for a low‑max‑auto‑correlation step function.

The objective is

    F(a) = 2·n·max(a * a) / (sum(a)^2),

where * denotes linear convolution, a_i ≥ 0 and n = len(a).
Because the objective is scale‑invariant we optimise on the probability simplex
(sum(a)=1) and return a normalised vector.

The algorithm builds on the previous Adam + sub‑gradient pipeline but adds:

* a progressive multi‑resolution schedule (1024 → 32768),
* richer initial families (double Gaussian bumps, triangles, different Dirichlet
  concentrations),
* longer Adam runs with a slower temperature increase,
* an adaptive learning‑rate schedule,
* a final, more aggressive hill‑climb that also moves larger ε‑values,
* automatic time‑budget handling – the loop cuts short any phase when the
  deadline is reached.

All heavy work is done with FFT‑based convolutions (O(n log n)).  The code
conforms to the required signature `propose_candidate(seed: int = 0,
budget_s: float = 1000) -> list[float]`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradient computation.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)] ,
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize a non‑negative vector to ``new_n`` points by linear interpolation
    and renormalise it to sum to 1.
    """
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                         # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, warm_start):
    """
    Produce a small yet diverse pool of starting points for length ``n``.
    ``warm_start`` may be ``None`` – in that case the pool contains only
    generic shapes.
    """
    candidates = []

    # 1️⃣ warm‑start (if available)
    if warm_start is not None:
        candidates.append(_upsample(warm_start, n))

    # 2️⃣ historic global variable (if the runner already set it)
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        candidates.append(_upsample(np.asarray(hseq, dtype=np.float64), n))

    # 3️⃣ random Dirichlet draws
    candidates.append(np.random.dirichlet(np.ones(n)))            # uniform
    candidates.append(np.random.dirichlet(np.full(n, 0.2)))        # spiky
    candidates.append(np.random.dirichlet(np.full(n, 0.8)))        # smoother

    # 4️⃣ edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.5, -1.5, -0.8):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5️⃣ exponentials (two‑sided)
    for gamma in (0.001, 0.003, 0.006):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6️⃣ Hann window
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7️⃣ centred Gaussian bumps (single and double)
    mu = (n - 1) / 2.0
    for sigma in (n / 12, n / 6):
        g = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        g /= g.sum()
        candidates.append(g)
    # double bump
    for sigma in (n / 12, n / 6):
        g1 = np.exp(-0.5 * ((i - mu / 2) / sigma) ** 2)
        g2 = np.exp(-0.5 * ((i - 3 * mu / 2) / sigma) ** 2)
        db = g1 + g2
        db /= db.sum()
        candidates.append(db)

    # 8️⃣ symmetric triangle
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(float)
    tri /= tri.sum()
    candidates.append(tri)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _adam_optimize(a0: np.ndarray, n: int, deadline: float,
                   max_iters: int, start_lr: float = 0.05,
                   start_temp: float = 0.30, temp_factor: float = 1.02):
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best vector (sum = 1) seen and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = start_lr
    lam = start_temp

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ---- map logits → simplex (soft‑max) ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- exact objective (sum(a)=1) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] new best {val:.6f}")

        # ---- smooth‑max weights (soft‑max) ----
        w = np.exp(lam * (conv - max_c))       # shift for stability
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # ---- gradient of weighted sum (surrogate) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- back‑prop through the soft‑max (logits) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- learning‑rate decay (very mild) ----
        if t % 2000 == 0:
            lr *= 0.95

        # ---- temperature schedule ----
        lam = min(lam * temp_factor, 1e5)

        # ---- random shake if stalled ----
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _subgrad_opt(a0: np.ndarray, n: int, deadline: float,
                 max_iters: int, init_eta: float = 0.55):
    """
    Exponentiated‑gradient sub‑gradient optimisation on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it}] new best {val:.6f}")

        # active set (positions attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = init_eta / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 7.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 250_000, no_improve_limit: int = 7_000):
    """
    Very cheap stochastic local search.
    Moves a random mass ε from one index to another, accepting only improving moves.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < no_improve_limit:
        i, j = np.random.choice(n, 2, replace=False)
        # allow larger jumps when the source is large
        eps = np.random.uniform(0.0, min(0.08, best_a[i]))
        if eps <= 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[HillClimb n={n}] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The search proceeds via a multi‑resolution schedule, Adam optimisation on a
    smooth‑max surrogate, an exponentiated‑gradient sub‑gradient phase, and a
    final stochastic hill‑climb.  It respects the supplied time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float('inf')

    # Progressive lengths – start small for cheap exploration, then expand.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # 1️⃣ Build a modest but diverse candidate pool
        # --------------------------------------------------------------
        candidates = _generate_initial_candidates(n, best_seq)

        # --------------------------------------------------------------
        # 2️⃣ Adaptive iteration budgets (scale roughly with 1/n)
        # --------------------------------------------------------------
        # we reserve ~70 % of the remaining time for the Adam / sub‑grad phases,
        # the rest for a final hill‑climb.
        time_left = deadline - time.time()
        # crude split: main phase gets 0.7 * remaining, hill‑climb later.
        phase_deadline = time.time() + 0.70 * time_left

        max_adam = max(5_000, int(15_000 * 2048 / n) + 5_000)
        max_sub  = max(2_000, int( 5_000 * 2048 / n) + 1_000)

        for cid, init in enumerate(candidates):
            if time.time() > phase_deadline:
                break

            # -------------------------------------------------
            # Adam (smooth‑max)
            # -------------------------------------------------
            a_adam, val_adam = _adam_optimize(
                init, n, phase_deadline, max_adam,
                start_lr=0.07,
                start_temp=0.25,
                temp_factor=1.015)

            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # -------------------------------------------------
            # Sub‑gradient (exponentiated‑gradient)
            # -------------------------------------------------
            a_sub, val_sub = _subgrad_opt(best_seq if best_seq is not None else a_adam,
                                          n, phase_deadline, max_sub,
                                          init_eta=0.70)

            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

            # -------------------------------------------------
            # Short fine‑tune on the current global best for this length
            # -------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _adam_optimize(best_seq, n, phase_deadline,
                                                  max_iters=int(0.5 * max_adam),
                                                  start_lr=0.05,
                                                  start_temp=0.30,
                                                  temp_factor=1.02)
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

        # --------------------------------------------------------------
        # End of length – keep best sequence for up‑sampling.
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final cheap local hill‑climb on the overall best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
