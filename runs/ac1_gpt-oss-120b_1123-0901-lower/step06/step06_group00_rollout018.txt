def subgradient_optimize_decrease(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=500000, eta0=0.5):
    a = np.maximum(a0, 0.0)
    a = a / max(a.sum(),1e-12)
    best_a = a.copy()
    best_val = np.inf
    it=0
    while time.time() < deadline and it < max_iters:
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set weight
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * cross_correlation(w, a, corr_fft_len)
        # adaptive step size
        eta = eta0 / np.sqrt(it+1)
        a = a * np.exp(-eta * grad)
        a = a / max(a.sum(), 1e-12)
        it+=1
    return best_a,best_val
```

def annealed_local_search(a0, n, conv_fft_len, deadline, max_moves=50000):
    a = a0.copy()
    conv = full_convolution(a, conv_fft_len)
    best_val = 2.0*n*conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j: continue
        delta = a[i] * (np.random.rand() * 0.5) # up to 50% of source
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = full_convolution(a2, conv_fft_len)
        val2 = 2.0*n*conv2.max()
        # accept if better or with prob exp(-(val2-val)/T)
        if val2 < best_val or np.random.rand() < np.exp(-(val2 - best_val) / (0.01 + 0.001*it)):
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
        it+=1
    return best_a,best_val
```

"""
Hybrid optimizer for the step‑function bound

The routine searches for a non‑negative vector a (∑a = 1) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)² ,

where a∗a is the ordinary linear convolution.
The algorithm proceeds in stages:

1. Build a modest pool of diverse start vectors (uniform, Dirichlet,
   edge‑concentrated, Gaussian, sinusoidal, spiked patterns).  If a
   previous solution ``height_sequence_1`` exists it is used as a warm‑start.

2. For each start vector run a limited Adam optimisation on the smooth‑max
   surrogate (log‑sum‑exp) with a slowly rising temperature.  The iterate
   is always projected onto the probability simplex and forced to be
   symmetric (a[i] = a[n‑1‑i]).

3. Refine the best candidate with a long sub‑gradient phase on the *exact*
   max‑convolution.  The step size decays as η₀/√t and the iterate again
   stays on the symmetric simplex via multiplicative (exponentiated‑gradient)
   updates.

4. Perform a cheap simulated‑annealing local search that swaps a random
   fraction of mass between two coordinates; the move is accepted if it
   improves the objective or with a temperature‑controlled probability.
   Symmetry is re‑imposed after each move.

5. If time permits, interpolate the best 4096‑vector to length 8192,
   run a short Adam phase followed by a sub‑gradient refinement and keep
   the best among all lengths.

All stages respect the global time budget (default 1000 s) and the routine
always returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest pool of diverse start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    cand.append(np.random.dirichlet(np.ones(n)))                # Dirichlet
    # power‑law edge shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))                           # uniform
    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Spike patterns with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:

    Adam on the log‑sum‑exp (smooth‑max) surrogate.
    Returns the best vector (∑≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # start temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")

        a = (a + a[::-1]) * 0.5

        # exact objective (sum(a)=1)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)

        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):

        else:

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.clip(a_big, 0.0, 1000.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
        )
        # sub‑gradient refinement
        a_big_ref, val_big = _subgradient_optimize_decay(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
            eta0=0.3,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – always return a valid vector
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value approx {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
