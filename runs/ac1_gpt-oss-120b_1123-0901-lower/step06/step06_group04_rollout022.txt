"""
propose_candidate
-----------------

A two‑phase stochastic optimisation that searches for a non‑negative step
function with a low value of

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The routine works on a single length (32768 by default) but can fall back to
a smaller size if the time budget is tight.  It builds a diverse pool of seed
vectors (uniform, edge‑concentrated power‑law / exponential shapes,
two‑Gaussian edges, sinusoidal bumps, random Dirichlet draws and the previous
global best if available).  Each seed is refined by

1. Adam optimisation on a smooth‑max surrogate of the ℓ∞‑norm of the
   convolution (log‑sum‑exp with a slowly increasing temperature).  The
   parametrisation is a soft‑max of free logits, therefore the iterate always
   stays in the probability simplex.

2. Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.

Both phases are run for a higher number of iterations than the earlier
implementation (≈ 8000 Adam steps and 5000 sub‑gradient steps for the
32768‑dimensional problem).  A few jittered restarts around the current best
are also performed.  All work respects the supplied wall‑clock budget; the
best sequence discovered before the deadline is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based, works for any power‑of‑two length)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = None,
    lam_start: float = 0.5,
    lam_factor: float = 1.01,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The iterate is kept on the simplex via a soft‑max of logits.
    Returns the best (vector, value) pair encountered before the deadline.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    if max_iters is None:
        # a generous budget that scales with dimension
        max_iters = int(120_000 * 1024 / n) + 5_000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- soft‑max (project onto probability simplex) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution + smooth‑max weights -----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ----- gradient of smooth‑max w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ----- back‑prop through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature annealing (smooth‑max approaches true max) -----
        lam = min(lam * lam_factor, 1e8)

        # ----- exact objective (sum(a)=1) -----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ----- occasional random perturbation if stagnating -----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000,
    eta_start: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient update.
    Returns the best (vector, value) pair found before the deadline.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = eta_start
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # mask of (almost) active constraints
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # fallback in the unlikely event of underflow
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # slowly decay the step size
        eta *= 0.9995

        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators (edge‑concentrated families)
# ----------------------------------------------------------------------


def _edge_powerlaw(n: int, frac: float, alpha: float) -> np.ndarray:
    """
    Symmetric edge shape a_i ∝ (i+1)^(-alpha) on the first `w` indices,
    mirrored to the right side.  `frac` is the fraction of the full length
    used for each edge (0 < frac ≤ 0.5).
    """
    w = int(frac * n)
    idx = np.arange(1, w + 1, dtype=np.float64)
    shape = np.power(idx, -alpha)
    seq = np.zeros(n, dtype=np.float64)
    seq[:w] = shape
    seq[-w:] = shape[::-1]
    total = seq.sum()
    if total > 0:
        seq /= total
    return seq


def _edge_exponential(n: int, frac: float, gamma: float) -> np.ndarray:
    """
    Symmetric edge shape a_i ∝ exp(-gamma * i) on the first `w` indices.
    """
    w = int(frac * n)
    i = np.arange(w, dtype=np.float64)
    shape = np.exp(-gamma * i)
    seq = np.zeros(n, dtype=np.float64)
    seq[:w] = shape
    seq[-w:] = shape[::-1]
    total = seq.sum()
    if total > 0:
        seq /= total
    return seq


def _two_gaussian_edges(n: int, sigma: float) -> np.ndarray:
    """
    Superposition of two Gaussian bumps centred at the two ends.
    """
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / sigma) ** 2)
    right = np.exp(-0.5 * ((n - 1 - i) / sigma) ** 2)
    seq = left + right
    seq /= seq.sum()
    return seq


def _sine_squared_bump(n: int) -> np.ndarray:
    """A smooth sinusoidal bump that is zero at both boundaries."""
    seq = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    seq /= seq.sum()
    return seq

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget (seconds) and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer framework
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a working length.  If the budget is already small we fall back
    # to a shorter vector to guarantee at least some progress.
    # ------------------------------------------------------------------
    if deadline - time.time() < 120:          # less than two minutes left?
        n = 8192
    else:
        n = 32768

    print(f"\n=== Optimising length n={n} ===")
    conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    # ------------------------------------------------------------------
    # Build a heterogeneous pool of seeds
    # ------------------------------------------------------------------
    candidates = []

    # 1) uniform distribution
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) previous global best if available (could be of different size)
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] != n:
            # resample linearly to the new length
            x_old = np.linspace(0.0, 1.0, h.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            h = np.interp(x_new, x_old, h)
        h = np.maximum(h, 0.0)
        if h.sum() > 0:
            h = h / h.sum()
        else:
            h = np.random.dirichlet(np.ones(n))
        candidates.append(h)

    # 3) a few random Dirichlet draws
    for _ in range(4):
        candidates.append(np.random.dirichlet(np.ones(n)))

    # 4) Edge‑concentrated power‑law families
    for frac in [0.03, 0.05, 0.07, 0.10, 0.15]:
        for alpha in [0.5, 1.0, 1.5, 2.0, 2.5]:
            candidates.append(_edge_powerlaw(n, frac, alpha))

    # 5) Edge‑concentrated exponential families
    for frac in [0.04, 0.08, 0.12]:
        for gamma in [0.02, 0.05, 0.1, 0.2]:
            candidates.append(_edge_exponential(n, frac, gamma))

    # 6) Two Gaussian edges (different widths)
    for sigma in [0.8 * n, 0.4 * n, 0.2 * n, 0.1 * n]:
        candidates.append(_two_gaussian_edges(n, sigma))

    # 7) Smooth sine‑squared bump
    candidates.append(_sine_squared_bump(n))

    # 8) Mixture of uniform and edge Gaussian
    i = np.arange(n, dtype=np.float64)
    edge = np.exp(-0.5 * (i ** 2) / (0.15 * n) ** 2) + np.exp(-0.5 * ((n - 1 - i) ** 2) / (0.15 * n) ** 2)
    edge = edge / edge.sum()
    mix = 0.6 * np.full(n, 1.0 / n) + 0.4 * edge
    mix = mix / mix.sum()
    candidates.append(mix)

    # ------------------------------------------------------------------
    # Remove possible duplicates and keep a manageable number of seeds
    # ------------------------------------------------------------------
    # Convert each candidate to a canonical tuple (rounded) for deduplication
    uniq = {}
    for c in candidates:
        key = tuple(np.round(c, 12))
        if key not in uniq:
            uniq[key] = c
    candidates = list(uniq.values())
    # Limit to at most 40 seeds (still cheap with FFTs)
    candidates = candidates[:40]

    # ------------------------------------------------------------------
    # Optimisation loop over all seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    for cand_id, cand in enumerate(candidates):
        if time.time() > deadline:
            break

        try:
            # Phase 1 – Adam on the smooth‑max surrogate
            a_adam, val_adam = _adam_optimize(
                cand,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=None,              # let the function decide (≈ 8000 iters)
                lam_start=0.5,
                lam_factor=1.01,
            )

            # Phase 2 – exact ℓ∞ sub‑gradient refinement
            a_ref, val_ref = _subgradient_refine(
                a_adam,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=5_000,
                eta_start=0.45,
            )

            # Keep the best among the two refined results
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[seed={cand_id}] new global best {best_val:.6f}")
            elif val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[seed={cand_id}] new global best (Adam) {best_val:.6f}")

        except Exception as exc:
            print(f"[seed={cand_id}] exception: {exc}")

        # ------- occasional jittered restarts around the current best -------
        if best_seq is not None and len(best_seq) == n:
            # at most two jitter attempts per seed (budget aware)
            for jitter_id in range(2):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                jitter_start = np.exp(logits - np.max(logits))
                jitter_start /= jitter_start.sum()

                a_jit, val_jit = _adam_optimize(
                    jitter_start,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=3_500,           # a shorter run for jittered starts
                    lam_start=0.5,
                    lam_factor=1.01,
                )
                a_jit_ref, val_jit_ref = _subgradient_refine(
                    a_jit,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=3_000,
                    eta_start=0.45,
                )
                if val_jit_ref < best_val:
                    best_val = val_jit_ref
                    best_seq = a_jit_ref.copy()
                    print(
                        f"[seed={cand_id} jitter={jitter_id}] new global best {best_val:.6f}"
                    )

    # ------------------------------------------------------------------
    # Fallback safety net (very unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Return as a plain Python list – this is what the external evaluator expects
    return best_seq.tolist()
