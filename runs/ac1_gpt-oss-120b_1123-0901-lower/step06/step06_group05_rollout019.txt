```
def _run_adam(a0, n, deadline, max_iters):
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1,beta2,eps = 0.9,0.999,1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.01
    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()
    for t in range(1, max_iters+1):
        # compute a
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # compute conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")
        # compute smoothed max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient w.r.t a:
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature schedule
        lam = min(lam * lam_factor, 1e4)
        # random shake if stalled
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        if time.time() > deadline:
            break
    return best_a, best_val
```

```
def _generate_sparse_candidates(n, prev_best, num=3):
    candidates = []
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))
    # create sparse patterns
    for _ in range(num):
        K = max(2, int(np.random.uniform(np.sqrt(n), np.sqrt(n)*4)))
        # ensure K ≤ n
        if K > n:
            K = n
        idx = np.random.choice(n, K, replace=False)
        weights = np.random.dirichlet(np.ones(K))
        seq = np.zeros(n)
        seq[idx] = weights
        candidates.append(seq)
    # comb pattern
    comb = np.zeros(n)
    comb[::2] = 1.0
    comb /= comb.sum()
    candidates.append(comb)
    # random binary (0/1) normalized
    binary = np.random.binomial(1, 0.5, size=n).astype(float)
    if binary.sum() == 0:
        binary[np.random.randint(n)] = 1.0
    binary /= binary.sum()
    candidates.append(binary)
    return candidates
```

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

The objective is homogeneous – scaling the whole vector does not change the score –  
so we can restrict the search to the simplex (`∑a_i = 1`).  
The problem is non‑convex because the max‑convolution is a piece‑wise linear
function of \(a\).  A practical approach is:

1. **Smooth‑max surrogate** – replace the hard max by a temperature‑controlled
   soft‑max.  Its gradient can be written analytically using a single
   cross‑correlation (`_cross_correlation`).  
   Optimisation of the surrogate is performed with the Adam optimiser.

2. **Sub‑gradient phase** – after the surrogate has been mostly minimised we
   switch to the exact max‑convolution and use an exponentiated‑gradient
   sub‑gradient step (the true objective is still homogeneous, therefore the
   simplex projection by normalisation is enough).

3. **Diverse initialisation** – for each length we create a pool that contains  
   * a warm‑start obtained by up‑sampling the best sequence found so far,  
   * a few classical window shapes (Gaussian, Hann, exponential, power‑law),  
   * several random Dirichlet vectors, and most importantly **sparse candidates**
     (few non‑zero indices) which tend to give a flatter autocorrelation.

4. **Curriculum over the length** – start with modest lengths (2048–4096) to
   obtain a good shape quickly, then up‑sample and refine on larger
   resolutions (8192, 16384).  Larger `n` gives more degrees of freedom,
   allowing the convolution to approach its theoretical lower bound
   (\(F\ge 2n/(2n-1)\approx 1\)).

5. **Final cheap hill‑climb** – a stochastic local move that transfers a tiny
   amount of mass between two positions and keeps the move if the exact
   objective improves.

All heavy work is done with FFT‑based convolutions (`_full_convolution`,
`_cross_correlation`), thus each iteration costs only *O(n log n)* and
fits comfortably into the 1000 s budget.

The implementation below follows this plan, adds a few extra candidates,
and dynamically scales the number of iterations with the available time.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""Propose a step‑function (non‑negative height sequence) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The optimiser works on a smooth‑max surrogate (Adam) → exact sub‑gradient
(Exponentiated‑gradient) → final cheap hill‑climb.  A diverse pool of initial
shapes (including sparse “Sidon‑like” candidates) and a curriculum over the
length (2048 → 4096 → 8192 → 16384) allow us to beat the previous best
(~1.5086) well within the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean


def _generate_sparse_candidates(n: int, prev_best, num: int = 4):
    """Create a few sparse candidates (few non‑zero entries)."""
    candidates = []
    # keep the warm‑start as a sparse version as well
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    for _ in range(num):
        # pick a support size between sqrt(n) and 4·sqrt(n)
        K = int(np.random.uniform(np.sqrt(n), 4 * np.sqrt(n)))
        K = max(2, min(K, n))
        idx = np.random.choice(n, K, replace=False)
        weights = np.random.dirichlet(np.ones(K))
        seq = np.zeros(n)
        seq[idx] = weights
        candidates.append(seq)

    # a simple comb pattern (mass on every other index)
    comb = np.zeros(n)
    comb[::2] = 1.0
    comb /= comb.sum()
    candidates.append(comb)

    # random binary mask normalised
    mask = np.random.binomial(1, 0.5, size=n).astype(float)
    if mask.sum() == 0:
        mask[np.random.randint(n)] = 1.0
    mask /= mask.sum()
    candidates.append(mask)

    return candidates


# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best sequence seen during the run and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)          # parametrise the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.30                         # start temperature
    lam_factor = 1.01                  # very slow increase → smoother optimisation

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ---- map logits → point on the simplex (soft‑max) ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- exact objective (sum(a)=1 ⇒ denominator drops out) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # ---- smooth‑max weights (soft‑max on the convolution vector) ----
        w = np.exp(lam * (conv - max_c))          # stability shift by max_c
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ---- gradient of the surrogate (via cross‑correlation) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- back‑prop through the soft‑max (logits) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e4)

        # ---- occasional random shake if we are stuck ----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the *exact* max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                      # initial step‑size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where the convolution actually attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                        # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search.
    Moves a tiny mass ε from one index to another and keeps the move if it
    reduces the *exact* objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The algorithm uses a coarse‑to‑fine curriculum over the length,
    a heterogeneous pool of start points (including sparse patterns),
    Adam optimisation on a smooth‑max surrogate, an exact sub‑gradient
    phase, and a final cheap hill‑climb.  It respects the supplied time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # lengths – we go up to 16384; larger lengths give more expressive power
    # but also cost more CPU time.
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # build a heterogeneous set of starting points
        init_pool = _generate_initial_candidates(n, best_seq)
        init_pool += _generate_sparse_candidates(n, best_seq, num=3)

        # iteration budgets – roughly scale with 1/n (more iterations for small n)
        max_adam = int(50000 * 2048 / n) + 8_000   # a generous budget
        max_subgrad = int(12000 * 2048 / n) + 2_000

        for cid, init in enumerate(init_pool):
            if time.time() > deadline:
                break

            # ---- Adam (smooth‑max) ---------------------------------
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam new global best {val_adam:.6f}")

            # ---- Sub‑gradient (exponentiated‑gradient) ---------------
            a_sub, val_sub = _run_subgrad(best_seq, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad new global best {val_sub:.6f}")

            # ---- Short fine‑tune on the current global best -----------
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                             int(0.5 * max_adam))
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq,
                                               len(best_seq),
                                               deadline,
                                               max_trials=500_000)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nBest value achieved: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
