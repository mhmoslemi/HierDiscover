def pgd_nesterov(a0, n, target_sum, deadline, max_iters, rng):
    a = a0.copy()
    a = a * target_sum / a.sum()
    a_prev = a.copy()
    best_seq = a.copy()
    best_obj = compute_obj(a)

    # FFT size
    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.005 * target_sum
    beta = 0.9
    eps_active = 1e-12

    for it in range(1, max_iters+1):
        if time.time() >= deadline: break

        # Nesterov extrapolation
        y = a + beta * (a - a_prev)

        # compute convolution of y
        A_fft = np.fft.rfft(y, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n - 1]
        M = conv.max()

        # active mask
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv); w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / cnt

        # weighted reverse conv for subgradient
        # but we can compute rev via parameter: conv_rev = irfft( rfft(w[::-1]) * A_fft, n=Nfft)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[:len(w)+n-1]
        idx_start = 2*n - 2
        rev = conv_rev[idx_start - np.arange(n)]

"""
Optimisation of a non‑negative step function for the given evaluation metric.

The algorithm consists of three stages:

1️⃣  **Fast projected Nesterov‑gradient descent** on the exact
    max‑convolution.  The sub‑gradient is obtained from the active
    convolution indices; after each gradient step the vector is projected
    onto the simplex {a_i ≥ 0, Σ a_i = √(2 n)}.  A Nesterov momentum term
    and an annealed step‑size accelerate convergence.

2️⃣  **Simulated‑annealing mass transfer** starting from the best vector of
    stage 1.  Randomly moves a small fraction of mass from one index to
    another; improvements are always kept, worse moves are accepted with
    a temperature that decays linearly.

3️⃣  **Fine‑grained stochastic refinement** (tiny mass transfers) to
    polish the solution in the remaining time.

All operations use FFTs (O(n log n)) to keep the per‑iteration cost low.
The total running time is limited to the provided budget (≈1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target_sum}.
    Implements the algorithm of Duchi et al. (2008) in O(n log n).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _fft_size_for_conv(n: int, extra_factor: int = 3) -> int:
    """
    Return a power‑of‑2 length suitable for linear convolutions of length
    up to (extra_factor·n‑2).  The default extra_factor=3 works for the
    reverse‑weighted convolution needed for the sub‑gradient.
    """
    size = extra_factor * n - 2
    return 1 << ((size).bit_length())


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a * a (linear convolution) using one FFT."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _objective_from_max(n: int, max_conv: float, total_sum: float) -> float:
    """Exact value of evaluate_sequence for a given max convolution."""
    return 2.0 * n * max_conv / (total_sum * total_sum)


def _subgradient_and_max(a: np.ndarray, Nfft: int, eps_active: float = 1e-12):
    """
    Return (grad, max_val) where grad is a sub‑gradient of
    max(a∗a) w.r.t. a (using the active set) and max_val = max(a∗a).
    """
    n = a.shape[0]

    # forward convolution and its maximum
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    M = float(conv.max())

    # active weights (uniform on the active indices)
    mask = conv >= M - eps_active
    cnt = int(mask.sum())
    if cnt == 0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = mask.astype(float) / cnt

    # reverse‑weighted convolution = sub‑gradient of the max‑convolution
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]

    grad = 2.0 * rev
    return grad, M

# Optimisers ------------------------------------------------------------


def _pgd_nesterov(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with Nesterov acceleration on the exact
    max‑convolution.  Returns the best vector observed and its exact
    objective value.
    """
    # initialise on the simplex (exactly)
    a = a0.copy()
    a = a * target_sum / a.sum()

    a_prev = a.copy()
    best_seq = a.copy()
    # the objective depends only on max convolution (sum is constant)
    Nfft = _fft_size_for_conv(n, extra_factor=3)
    eps_active = 1e-12

    # hyper‑parameters
    step0 = 0.005 * target_sum      # base step (≈0.5 % of the total mass)
    beta = 0.9                      # Nesterov momentum
    best_obj = _objective_from_max(n, _max_convolution(a, Nfft), target_sum)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # Nesterov extrapolation
        y = a + beta * (a - a_prev)

        # sub‑gradient of max‑convolution at y
        grad, M = _subgradient_and_max(y, Nfft, eps_active)

        # step size (inverse‑sqrt schedule)
        lr = step0 / math.sqrt(it)

        # gradient step and projection
        a_new = y - lr * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # shift variables for the next iteration
        a_prev = a
        a = a_new

        # evaluate
        cur_obj = _objective_from_max(n, M, target_sum)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[PGD‑Nesterov] n={n:5d} it={it:8d} best={best_obj:.6f}")

        # occasional jitter to escape plateaux
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_obj


def _simulated_annealing_refine(
    a: np.ndarray,
    deadline: float,
    cur_obj: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.
    Randomly moves a fraction (1‑20 %) of mass from a source index to a
    target index; improvements are always kept, worse moves are accepted
    with a linearly decaying temperature.
    """
    n = a.shape[0]
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_val = cur_obj
    Nfft = _fft_size_for_conv(n, extra_factor=2)   # only forward conv needed

    step = 0
    while time.time() < deadline:
        # pick source proportional to current mass (more mass → more chances)
        if cur_seq.sum() == 0:
            break
        prob = cur_seq / cur_seq.sum()
        i = rng.choice(n, p=prob)
        j = rng.integers(0, n)
        if i == j or cur_seq[i] < 1e-14:
            continue

        # random transfer fraction 1‑20 %
        frac = rng.uniform(0.01, 0.20)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate new objective (fast via FFT)
        M = _max_convolution(new_seq, Nfft)
        new_val = _objective_from_max(n, M, target_sum)

        # deterministic improvement
        if new_val < cur_val - 1e-12:
            cur_seq = new_seq
            cur_val = new_val
            print(f"[SA] step {step:7d} improvement → {cur_val:.6f}")
        else:
            # simulated‑annealing acceptance
            # temperature drops from 0.1·cur_val to ~0
            T = 0.1 * cur_val * max(0.0, 1.0 - step / 800_000.0)
            prob_accept = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob_accept:
                cur_seq = new_seq
                cur_val = new_val
        step += 1

    return cur_seq, cur_val


def _fine_local_refine(
    a: np.ndarray,
    deadline: float,
    cur_obj: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic hill‑climbing (mass fractions 0.5‑3 %).
    Very cheap, runs until time expires.
    """
    n = a.shape[0]
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_val = cur_obj
    Nfft = _fft_size_for_conv(n, extra_factor=2)

    step = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-14:
            continue
        frac = rng.uniform(0.005, 0.03)
        delta = cur_seq[i] * frac
        if delta <= 0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        M = _max_convolution(new_seq, Nfft)
        new_val = _objective_from_max(n, M, target_sum)

        if new_val < cur_val - 1e-12:
            cur_seq = new_seq
            cur_val = new_val
            print(f"[local] step {step:7d} improve → {cur_val:.6f}")
        step += 1

    return cur_seq, cur_val

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.

    The routine follows the three‑stage scheme described at the top of the file.
    It respects the total time budget, dynamically allocating iterations to each
    phase.  The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin for the surrounding harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine the problem size (length of the step function)
    base = None
    try:
        # ``height_sequence_1`` may be supplied by the surrounding environment
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        pass

    if base is not None:
        n = int(base.shape[0])
        # initialise with the supplied construction (renormalised)
        a0 = base.copy()
        target_sum = math.sqrt(2.0 * n)
        a0 = a0 * target_sum / a0.sum()
    else:
        # fall‑back: choose a length that historically works well
        n = 1900
        target_sum = math.sqrt(2.0 * n)
        # a simple exponential decay as a generic start point
        rate = rng.uniform(0.001, 0.008)
        a0 = np.exp(-rate * np.arange(n))
        a0 = a0 / a0.sum() * target_sum

    # --------------------------------------------------------------
    # Phase 1 – accelerated projected gradient descent (majority of budget)
    pgd_deadline = start_time + 0.70 * (deadline - start_time)  # ~70 % of time
    best_seq, best_obj = _pgd_nesterov(
        a0=a0,
        n=n,
        target_sum=target_sum,
        deadline=pgd_deadline,
        rng=rng,
        max_iters=10_000_000,
    )
    print(f"[phase‑1] best objective = {best_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing mass transfer (remaining time, ≈20 %)
    if time.time() < deadline:
        sa_deadline = start_time + 0.90 * (deadline - start_time) + start_time
        best_seq, best_obj = _simulated_annealing_refine(
            a=best_seq,
            deadline=sa_deadline,
            cur_obj=best_obj,
            rng=rng,
        )
        print(f"[phase‑2] after SA objective = {best_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – fine‑grained local hill climbing (final ~10 %)
    if time.time() < deadline:
        refine_deadline = deadline
        best_seq, best_obj = _fine_local_refine(
            a=best_seq,
            deadline=refine_deadline,
            cur_obj=best_obj,
            rng=rng,
        )
        print(f"[phase‑3] final objective = {best_obj:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain list)
    return best_seq.tolist()
