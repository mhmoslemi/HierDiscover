def _pgd_optimize_momentum(a0, n, deadline, max_iters, best_seq, best_obj, rng):
    target_sum = sqrt(2n)
    a = a0.copy()
    a = a * target_sum / a.sum()
    Nfft = 1 << ((3*n-2).bit_length())
    momentum = np.zeros_like(a)
    mu = 0.9
    base_lr = 0.001 * target_sum
    
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute convolution via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        # update best
        if M < best_obj:
            best_obj = M
            best_seq = a.copy()
            print(f"[PGD-mom len {n} iter {it}] new best = {best_obj:.6f}")
        # active mask
        eps = 1e-12
        mask = conv >= M - eps
        cnt = mask.sum()
        w = mask.astype(float) / max(cnt,1)
        # reverse weighted convolution for gradient
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]
        grad = 2.0 * rev
        lr = base_lr / math.sqrt(it)
        momentum = mu * momentum - lr * grad
        a_new = a + momentum
        a = _project_onto_simplex(a_new, target_sum)

def _refine_simulated_anneal(a, n, deadline, best_obj, rng):
    target_sum = a.sum()
    cur = a.copy()
    cur_obj = best_obj
    Nfft = 1 << ((2*n-1).bit_length())
    steps = 0
    max_steps = 500_000
    while time.time() < deadline and steps < max_steps:
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i == j:
            continue
        # compute amount to move: random fraction of source mass
        if cur[i] < 1e-12:
            continue
        frac = rng.uniform(0.001, 0.02)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # compute conv max quickly with FFT
        A_fft = np.fft.rfft(new, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        new_obj = float(conv.max())
        if new_obj + 1e-12 < cur_obj:
            cur = new
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = cur.copy()
                print(f"[refine] improvement to {cur_obj:.6f}")
        else:
            # acceptance probability
            T = max(0.001 * cur_obj, 1e-5) * (1.0 - steps / max_steps)
            if rng.random() < math.exp(-(new_obj - cur_obj) / max(T,1e-12)):
                cur = new
                cur_obj = new_obj
        steps += 1
    return cur, cur_obj
```

# Simplex projection (Euclidean) ----------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target_sum}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial vectors --------------------------------------------------------
def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Return a list of diverse feasible start vectors (sum = target_sum)."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (γ = 1)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 7) sparse random spikes (few large entries)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init


# ----------------------------------------------------------------------
# Optimiser utilities ----------------------------------------------------
def _fft_size_for_len(l):
    """Next power‑of‑two size that can accommodate a linear convolution of length l."""
    return 1 << ((l - 1).bit_length())


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of the linear self‑convolution a*a using a single FFT."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _active_weights(conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Return the weight vector w used for sub‑gradient (averaged over active slots)."""
    M = conv.max()
    mask = conv >= M - eps
    cnt = mask.sum()
    if cnt == 0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = mask.astype(float) / float(cnt)
    return w


def _reverse_weighted_conv(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (indices outside are ignored) via FFT.
    """
    A = np.fft.rfft(a, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    tmp = np.fft.irfft(W_rev * A, n=Nfft)
    tmp = tmp[: len(w) + a.shape[0] - 1]                     # linear part only
    idx_start = 2 * a.shape[0] - 2
    rev = tmp[idx_start - np.arange(a.shape[0])]
    return rev

# Projected Gradient Descent with momentum -------------------------------
def _pgd_optimize_momentum(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Accelerated PGD on the exact max‑convolution.
    Returns the best vector observed and its max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * target_sum / a.sum()                # initialise on the simplex
    Nfft = _fft_size_for_len(3 * n - 2)

    mu = 0.9                                    # momentum coefficient
    velocity = np.zeros_like(a)
    step0 = 0.001 * target_sum                  # base step size
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current max‑convolution
        M = _max_convolution(a, Nfft)

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD  len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # sub‑gradient of the max (average over active indices)
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        w = _active_weights(conv, eps=eps_active)

        rev = _reverse_weighted_conv(a, w, Nfft)
        grad = 2.0 * rev                         # ∇_a max(a∗a)

        # momentum step + projection
        lr = step0 / math.sqrt(it)               # simple annealing
        velocity = mu * velocity - lr * grad
        a_tmp = a + velocity
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional random jitter to escape plateaux
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient Adam (soft‑max smoothed max) --------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logits (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.
    Returns the best vector seen and its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)
    eps = 1e-16
    y = np.log(a0 + eps)                       # initialise logits

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    # temperature schedule for the soft‑max
    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    Nfft = _fft_size_for_len(3 * n - 2)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # soft‑max of logits → probability vector on the simplex
        logits = y - np.max(y)                 # for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p                     # satisfies Σa = target_sum

        # exact max‑convolution (used for the true objective)
        M = _max_convolution(a, Nfft)

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # soft‑max weights for the smoothed max
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            w = _active_weights(np.convolve(a, a)[:2 * n - 1])
        else:
            conv = np.convolve(a, a)[:2 * n - 1]
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()

        # gradient w.r.t. a (2·reverse‑conv)
        rev = _reverse_weighted_conv(a, w, Nfft)
        g_a = 2.0 * rev

        # chain rule to logits (mirrored‑gradient on the simplex)
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # jitter
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local stochastic refinement (simulated annealing) ----------------------
def _refine_simulated_anneal(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing on the simplex:
    repeatedly transfer a small random fraction of mass from a random index i
    to a random index j.  Accept improvements; otherwise accept with a decaying
    simulated‑annealing probability.
    """
    target_sum = a.sum()               # already √(2 n)
    cur = a.copy()
    cur_score = best_score

    Nfft = _fft_size_for_len(2 * n - 1)

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur[i] < 1e-12:
            step += 1
            continue

        # transfer a moderate random fraction of source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # compute new max‑convolution (fast FFT)
        M = _max_convolution(new, Nfft)

        if M + 1e-12 < cur_score:               # deterministic improvement
            cur = new
            cur_score = M
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(M - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new
                cur_score = M
        step += 1

    return cur, cur_score

# Main entry point --------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    # try to reuse the previous construction if it exists
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a single length (the previously good one or a fallback)
    if base is not None:
        n = int(base.shape[0])
    else:
        n = 1890                       # historically good size

    # clamp length to a reasonable range
    n = max(300, min(n, 5000))

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Populate the pool of initial vectors
    init_vectors = _make_initial_vectors(n, target_sum, rng)

    # additionally interpolate the previous best to the current length
    if base is not None and base.shape[0] != n:
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        interp = np.interp(new_x, old_x, base)
        interp = np.maximum(interp, 0.0)
        interp = interp / interp.sum() * target_sum
        init_vectors.append(interp)

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # iteration budget – generous but still respects the overall time limit
    max_iters = max(150_000, int(2_500_000 * 1890 / n))

    print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

    for idx, a0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break
        print(f"  start {idx + 1}/{len(init_vectors)} for length {n}")

        # Adam based optimisation
        best_seq, best_score = _adam_optimize(
            a0=a0.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq if best_seq is not None else a0,
            best_score=best_score,
            rng=rng,
        )

        # Projected Gradient Descent with momentum
        best_seq, best_score = _pgd_optimize_momentum(
            a0=a0.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Final stochastic refinement (uses any remaining time)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate up to half of the remaining budget for refinement,
        # but never more than 45 seconds
        refine_deadline = time.time() + min(remaining * 0.5, 45.0)
        best_seq, best_score = _refine_simulated_anneal(
            a=best_seq,
            n=n,
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback sanity – should never be needed
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
