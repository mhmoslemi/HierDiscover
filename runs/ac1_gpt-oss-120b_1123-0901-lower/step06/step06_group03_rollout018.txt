def _proj_simplex(v):
    # projects onto probability simplex sum=1, v >=0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (assumes sum(a) > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero or negative – return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp (smooth max) surrogate.
    Returns updated global_best_seq / best_val.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise on simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                          # a tad larger than the reference
    lam = 0.5                          # temperature for smooth‑max
    lam_factor = 1.010                 # slower increase → smoother path
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # soft‑max → simplex point
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (useful for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter if we stall
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        # hard cut‑off (prevents infinite loops if deadline is huge)
        if it >= 800_000:
            break

    return global_best_seq, global_best_val


def _subgradient_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    """
    # start from the best point we have (or from a0)
    a = global_best_seq if global_best_seq is not None else a0.copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    eta0 = 0.6
    last_improve = time.time()

    it = 0
    while time.time() < deadline and it < 120_000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:06d}] new best {val:.6f}")

        # active set mask – all positions attaining the max (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient (mirror descent on simplex)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)          # stays non‑negative
        a_sum = a.sum()
        if a_sum < 1e-12:                       # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        # occasional “shake” if we have not improved for a while
        if time.time() - last_improve > 8.0:
            # small random re‑allocation of 0.5 % of the mass
            mask = np.random.rand(n) < 0.005
            if mask.any():
                delta = a[mask] * 0.5
                a[mask] -= delta
                a += delta.sum() / n
                a = _proj_simplex(a)          # re‑project to simplex
            last_improve = time.time()

        it += 1

    return global_best_seq, global_best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by an exact sub‑gradient phase.
    The function updates and returns the best sequence/value found.
    """
    best_seq, best_val = _adam_phase(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_phase(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Construct a diverse set of start vectors.
    Families:
        – warm‑start (and cyclic shifts)
        – uniform
        – power‑law edge concentrated
        – double‑exponential (two‑sided Laplace)
        – symmetric Beta (U‑shaped)
        – generalized Gaussian (|x‑½|^p)
        – single half‑wave sinusoid
        – random Dirichlet draws (various concentrations)
    """
    pool = []

    # ------------------------------------------------------------------
    # Warm‑start & shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # Uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Power‑law edge‑concentrated (α negative)
    idx = np.arange(1, n + 1, dtype=np.float64)            # 1 … n
    prod = idx * (n - idx + 1.0)                           # i·(n‑i+1)
    for alpha in np.linspace(-5.0, -0.05, 25):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # ------------------------------------------------------------------
    # Double‑exponential (two‑sided Laplace)
    i = np.arange(n, dtype=np.float64)
    for lam in np.logspace(-2, 1, 20):                     # 0.01 … 10
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # Symmetric Beta (U‑shaped) :  a_i ∝ (x)^{α-1}(1‑x)^{α‑1}
    x = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n
    for alpha in np.logspace(-1, 0.7, 15):                 # 0.1 … 5
        shape = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
        # avoid infinities at the borders when α<1
        shape[np.isinf(shape)] = 0.0
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # ------------------------------------------------------------------
    # Generalised Gaussian (|x-½|^p)   – narrower for larger p
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        sigma = n * 0.25
        shape = np.exp(-0.5 * (np.abs(i - (n - 1) / 2) ** p) / (sigma ** p))
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # Single half‑wave sinusoid (positive half‑sine)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # ------------------------------------------------------------------
    # Random Dirichlet draws (various concentration parameters)
    for conc in np.logspace(-2, 0, 20):
        shape = np.random.dirichlet(np.full(n, conc))
        shape /= shape.sum()
        pool.append(shape)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects the supplied time‑budget and returns the
    best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Primary support size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from a previously stored construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # Build candidate pool & quick screening
    candidates = _generate_candidate_pool(n, w)
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored = []   # (value, normalised vector)

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:                         # avoid almost‑zero vectors
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # Keep the most promising starts (up to ten)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:10]]

    # ------------------------------------------------------------------
    # Deep optimisation passes (Adam + exact sub‑gradient)
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[optim start {cid}] best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling stages (each roughly doubles the support)
    while time.time() + 12 < deadline:
        n_big = n * 2
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation of the current best to the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq if best_seq is not None else np.full(n, 1.0 / n))
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print(f"[upsample] n={n_big} – starting deep optimisation")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[upsample] new best {best_val:.6f}")

        # Prepare for a possible second up‑sample
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Final guard – if optimisation somehow failed, fall back to uniform
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
