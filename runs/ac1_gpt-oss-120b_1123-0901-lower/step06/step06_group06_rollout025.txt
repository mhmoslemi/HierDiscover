def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    n = 8192
    # compute FFT lengths
    conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    def eval_obj(a):
        c = _conv_fft(a, conv_len)
        max_c = c.max()
        return 2.0 * n * max_c

    # Build initial candidates list
    candidates = []
    # 1. warm from global
    if 'height_sequence_1' in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n-h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() > 0:
            candidates.append(h / h.sum())
    # 2. random Dirichlet (conc=1)
    candidates.append(np.random.dirichlet(np.ones(n)))
    # 3. random sparse (sqrt(n) spikes)
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n)
    idx = np.random.choice(n, size=m_spikes, replace=False)
    sparse[idx] = 1.0
    sparse = sparse / sparse.sum()
    candidates.append(sparse)
    # 4. one-sided exponential
    gamma = 0.001
    expo = np.exp(-gamma * np.arange(n))
    expo = expo / expo.sum()
    candidates.append(expo)
    # 5. power-law decreasing
    alpha = 1.5
    power = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), alpha)
    power = power / power.sum()
    candidates.append(power)
    # 6. asymmetric beta-like: a_i ∝ (i+1)^{p} (n-i)^{q}
    p = -0.5
    q = 0.5
    beta_asym = np.power(np.arange(1, n+1, dtype=np.float64), p) * np.power(np.arange(n, 0, -1, dtype=np.float64), q)
    beta_asym = np.maximum(beta_asym, 0)
    if beta_asym.sum()>0:
        candidates.append(beta_asym / beta_asym.sum())
    # 7. linear decreasing ramp (tent reversed)
    ramp = np.linspace(1, 0, n)  # start high, linear down to 0
    ramp = ramp / ramp.sum()
    candidates.append(ramp)
    # 8. half-sine (peak left)
    i = np.arange(n, dtype=np.float64)
    sin_left = np.sin(np.pi * i / (2 * (n-1)))  # from 0 to 1 at mid? Actually left half.
    sin_left = sin_left / sin_left.sum()
    candidates.append(sin_left)

    # Optimization loops
    for cid, a0 in enumerate(candidates):
        if time.time() > deadline:
            break
        # Ensure non-negative and normalized
        a = np.maximum(a0, 0.0)
        if a.sum() == 0:
            continue
        a = a / a.sum()
        logits = np.log(a + 1e-15)

        # Adam hyperparameters scaled for n
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.04 * np.sqrt(2048.0 / n)  # smaller lr for large n
        lam = 0.1
        lam_factor = 1.015  # slower increase

        m = np.zeros_like(logits)
        v = np.zeros_like(logits)

        # iteration budget adaptive: base 20000 for n=4096, scale linearly
        base_iters = 20000
        max_iters = int(base_iters * n / 4096)

        last_improve = time.time()
        for t in range(1, max_iters+1):
            # softmax
            mx = logits.max()
            exp_x = np.exp(logits - mx)
            a = exp_x / exp_x.sum()

            # conv
            conv = _conv_fft(a, conv_len)
            max_c = conv.max()
            # compute smooth max weights
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient
            grad_a = 2.0 * _corr_fft(w, a, corr_len)

            # backprop through softmax
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # increase temperature
            lam = min(lam * lam_factor, 1e5)

            # evaluate objective
            val = 2.0 * n * max_c
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[n={n} cand={cid} Adam iter={t}] new best {val:.6f}")

            # random shake when stagnated >10 sec
            if time.time() - last_improve > 10.0:
                logits += (np.random.rand(n) - 0.5) * 5e-3
                last_improve = time.time()

            if time.time() > deadline:
                break

        # Exponentiated gradient refinement
        if best_seq is None:
            a = a / a.sum()
        else:
            a = best_seq.copy()
        a = a / a.sum()
        eta = 0.5
        sub_last_improve = time.time()
        max_sub_iters = 8000
        for sub_it in range(max_sub_iters):
            conv = _conv_fft(a, conv_len)
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-12)
            if active.any():
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
            else:
                w = np.zeros_like(conv)
            grad_a = 2.0 * _corr_fft(w, a, corr_len)
            a = a * np.exp(-eta * grad_a)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a = a / s
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                sub_last_improve = time.time()
                print(f"[n={n} Subgrad iter={sub_it}] new best {val:.6f}")
            if time.time() - sub_last_improve > 12.0:
                break
            if time.time() > deadline:
                break

    # If still not enough, do simple random perturbation local search around best_seq:
    if best_seq is not None:
        a = best_seq.copy()
        a = a / a.sum()
        for step in range(20000):
            # propose perturbation: select 10 random indices
            idx = np.random.choice(n, size=10, replace=False)
            delta = (np.random.randn(10) * 1e-4)
            a[idx] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            val = 2.0 * n * _conv_fft(a, conv_len).max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[local perturb step={step}] new best {val:.6f}")
            if time.time() > deadline:
                break

    if best_seq is None:
        best_seq = np.full(4096, 1.0/4096)

    return best_seq.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution (numpy.convolve).

Algorithm
---------
* Work with a relatively large length (n=8192).  Larger n gives a
  finer search space while still fitting easily into memory.
* Build a diversified pool of simple parametric shapes (Dirichlet,
  one‑sided exponential, power‑law, asymmetric Beta‑like, linear ramp,
  sparse spikes, the previous best if available).  All shapes are
  normalised to have ℓ¹‑norm = 1, which makes the objective independent of
  scaling.
* For each candidate run a two‑phase optimisation:
    1. Adam on a *smooth‑max* surrogate of the true max.  The surrogate
       uses a soft‑max weighting (log‑sum‑exp) with a temperature that
       grows very slowly so the optimiser sees a smooth landscape for a
       long time.  The variables are kept in the log‑softmax (logits)
       domain, which automatically respects non‑negativity and the
       simplex constraint.
    2. An exact sub‑gradient phase (exponentiated‑gradient) that updates
       the probabilities directly with the true max‑gradient.
* Whenever a new best value is found we log it and keep the corresponding
  sequence.
* After the two phases a cheap random‑perturbation local search is run
  for a few seconds to try to escape shallow plateaus.
* The whole routine respects the time budget (default 1000 s) and
  returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """
    Exact objective (scaling‑invariant; sum(a) == 1).
    """
    conv = _conv_fft(a, conv_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    The function is limited by ``budget_s`` seconds and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Problem size – we concentrate most of the time on n = 8192.
    n = 8192
    conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))      # for a*a
    corr_len = 1 << int(np.ceil(np.log2(3 * n - 2)))      # for cross‑corr

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate and possibly store a new best.
    def consider(candidate: np.ndarray, val: float, label: str):
        nonlocal best_seq, best_val
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[{label}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Build a list of diverse initial shapes (all normalised to sum = 1).
    initial_shapes = []

    # 1) Warm‑start from a previously found solution, if present.
    if "height_sequence_1" in globals():
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.shape[0] > n:
            hs = hs[:n]
        else:
            hs = np.pad(hs, (0, n - hs.shape[0]))
        hs = np.maximum(hs, 0.0)
        if hs.sum() > 0:
            hs = hs / hs.sum()
            initial_shapes.append(hs)

    # 2) Pure Dirichlet (uniform on the simplex).
    initial_shapes.append(np.random.dirichlet(np.ones(n)))

    # 3) Sparse pattern – roughly sqrt(n) spikes.
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=m_spikes, replace=False)
    sparse[idx] = 1.0
    sparse = sparse / sparse.sum()
    initial_shapes.append(sparse)

    # 4) One‑sided exponential decay.
    gamma = 0.001
    expo = np.exp(-gamma * np.arange(n, dtype=np.float64))
    expo = expo / expo.sum()
    initial_shapes.append(expo)

    # 5) Power‑law decreasing (α > 0).
    alpha = 1.5
    power = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), alpha)
    power = power / power.sum()
    initial_shapes.append(power)

    # 6) Asymmetric “beta‑like’’ shape: a_i ∝ (i+1)^p * (n‑i)^q.
    p, q = -0.5, 0.5
    i_vec = np.arange(1, n + 1, dtype=np.float64)
    beta_asym = np.power(i_vec, p) * np.power(n - i_vec + 1.0, q)
    beta_asym = np.maximum(beta_asym, 0.0)
    beta_asym = beta_asym / beta_asym.sum()
    initial_shapes.append(beta_asym)

    # 7) Linear decreasing ramp (high on the left, zero on the right).
    ramp = np.linspace(1.0, 0.0, n, dtype=np.float64)
    ramp = ramp / ramp.sum()
    initial_shapes.append(ramp)

    # 8) Half‑sine (starts at 0, peaks at the middle‑left, then falls).
    half_sin = np.sin(np.pi * np.arange(n, dtype=np.float64) / (2 * (n - 1)))
    half_sin = half_sin / half_sin.sum()
    initial_shapes.append(half_sin)

    # ------------------------------------------------------------------
    # Optimisation for each starting shape.
    for cid, init_a in enumerate(initial_shapes):
        if time.time() > deadline:
            break

        # Normalise and move to the log‑softmax (logits) domain.
        a = np.maximum(init_a, 0.0)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        logits = np.log(a + 1e-15)

        # Adam hyper‑parameters (scaled for the problem size).
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.04 * np.sqrt(2048.0 / n)          # a bit smaller for n=8192
        lam = 0.08                               # initial smooth‑max temperature
        lam_factor = 1.012                       # very gradual increase

        m = np.zeros_like(logits)
        v = np.zeros_like(logits)

        # Number of Adam iterations – tuned to fit into the remaining budget.
        base_iters = 25000
        max_iters = int(base_iters * n / 4096)   # ~50 k for n=8192

        last_improve = time.time()

        for t in range(1, max_iters + 1):
            # Soft‑max to stay on the simplex.
            mx = logits.max()
            exp_x = np.exp(logits - mx)
            a = exp_x / exp_x.sum()

            # Convolution and smooth‑max weights.
            conv = _conv_fft(a, conv_len)
            max_c = conv.max()
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # Gradient of the smooth‑max surrogate.
            grad_a = 2.0 * _corr_fft(w, a, corr_len)

            # Back‑prop through the soft‑max.
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam update.
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # Increase temperature (approach true max).
            lam = min(lam * lam_factor, 1e6)

            # Exact objective (sum(a) == 1 by construction).
            val = 2.0 * n * max_c
            consider(a, val, f"n={n} cand={cid} Adam iter={t}")

            # Simple “shake’’ if we have been idle for a while.
            if time.time() - last_improve > 10.0:
                logits += (np.random.rand(n) - 0.5) * 5e-3
                last_improve = time.time()

            if time.time() > deadline:
                break

        # ------------------------------------------------------------------
        # Exact sub‑gradient phase (exponentiated gradient).
        a = a / a.sum()
        eta = 0.55
        sub_last_improve = time.time()
        max_sub_iters = 8000

        for sub_it in range(max_sub_iters):
            conv = _conv_fft(a, conv_len)
            max_c = conv.max()

            # Active set of indices attaining the max (within tolerance).
            active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
            if active.any():
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
            else:
                w = np.zeros_like(conv)

            grad_a = 2.0 * _corr_fft(w, a, corr_len)

            # Multiplicative update (exact exponentiated‑gradient step).
            a = a * np.exp(-eta * grad_a)
            s = a.sum()
            if s < 1e-12:                               # protect against collapse
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a = a / s

            val = 2.0 * n * max_c
            consider(a, val, f"n={n} cand={cid} Subgrad iter={sub_it}")

            if time.time() - sub_last_improve > 12.0:
                break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Final cheap local‑perturbation search around the best sequence.
    if best_seq is not None:
        a = best_seq.copy()
        a = a / a.sum()
        perturb_start = time.time()
        while time.time() - perturb_start < 30.0 and time.time() < deadline:
            # Randomly pick a small set of indices and move a tiny amount of mass.
            idx = np.random.choice(n, size=10, replace=False)
            delta = (np.random.rand(10) - 0.5) * 1e-4
            a[idx] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            val = _objective(a, n, conv_len)
            consider(a, val, "local perturb")
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed).
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Return as a plain Python list – the external evaluator expects a list.
    return best_seq.tolist()
