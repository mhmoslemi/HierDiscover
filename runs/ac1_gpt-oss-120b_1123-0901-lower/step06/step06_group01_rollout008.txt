def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto
    {x | x_i ≥ 0, Σ x_i = target_sum}.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho_arr = u - cssv / ind
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max(a * a) using a single FFT."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _subgradient_and_max(a: np.ndarray,
                         Nfft_max: int,
                         Nfft_rev: int,
                         eps_active: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Compute the exact sub‑gradient of the max‑convolution and the current
    maximal convolution value.

    cnt = mask.sum()

        w[np.argmax(conv)] = 1.0
    else:
        w = mask.astype(float) / float(cnt)

    # ----- weighted reverse convolution (sub‑gradient) -----
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft_rev)
    # reuse A_fft padded to Nfft_rev
    A_rev_fft = np.fft.rfft(a, n=Nfft_rev)
    conv_rev = np.fft.irfft(W_rev_fft * A_rev_fft, n=Nfft_rev)
    conv_rev = conv_rev[: len(w) + n - 1]                     # length = 3 n‑2
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    g = 2.0 * rev
    return g, M


def _interpolate(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``old`` to length ``new_n``."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    return new


# ----------------------------------------------------------------------
# Core optimisation routine --------------------------------------------


def _gradient_guided_sa(initial: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 300_000,
                        grad_every: int = 15) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided simulated annealing on the simplex.
    Returns the best sequence seen and its max‑convolution value.
    """
    n = initial.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # FFT sizes (powers of two) – pre‑computed for speed
    Nfft_max = 1 << ((2 * n - 1).bit_length())          # for a*a
    Nfft_rev = 1 << ((3 * n - 2).bit_length())          # for weighted reverse conv

    # current state
    cur = initial.copy()
    cur_val = _max_convolution(cur, Nfft_max)

    # global best
    best = cur.copy()
    best_val = cur_val

    # temperature schedule (linear decay)
    T_start = 0.02 * best_val
    T_end = 0.001 * best_val

    # pre‑select a few top / bottom indices after each gradient computation
    top_k = min(7, n)
    bot_k = min(7, n)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # recompute sub‑gradient every ``grad_every`` iterations
        if it % grad_every == 0:
            grad, M = _subgradient_and_max(cur, Nfft_max, Nfft_rev)
            # source = large gradient, destination = small gradient
            src_candidates = np.argpartition(-grad, top_k - 1)[:top_k]
            dst_candidates = np.argpartition(grad, bot_k - 1)[:bot_k]
            src = int(rng.choice(src_candidates))
            dst = int(rng.choice(dst_candidates))
        else:
            # occasionally a completely random move for extra exploration
            if rng.random() < 0.03:
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                while dst == src:
                    dst = rng.integers(0, n)

        # decide fraction of mass to move
        if cur[src] <= 0.0:
            continue
        frac = rng.uniform(0.015, 0.07)                 # 1.5% – 7% of donor mass
        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        # propose new sequence
        new_seq = cur.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # evaluate
        new_val = _max_convolution(new_seq, Nfft_max)

        # acceptance test
        improve = new_val < cur_val - 1e-12
        if improve:
            cur = new_seq
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best = new_seq.copy()
                best_val = new_val
                print(f"[SA iter {it:7d}] new global best = {best_val:.6f}")
        else:
            # simulated‑annealing acceptance
            T = T_start + (T_end - T_start) * (it / max_iters)
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new_seq
                cur_val = new_val

        # occasional jitter to shake the system
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.008 * target_sum / n, size=n)
            cur = cur + jitter
            cur = np.maximum(cur, 0.0)
            cur = cur * target_sum / cur.sum()
            cur_val = _max_convolution(cur, Nfft_max)
            if cur_val < best_val - 1e-12:
                best = cur.copy()
                best_val = cur_val
                print(f"[jitter iter {it:7d}] new global best = {best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine is deterministic given ``seed``.
    It returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load any previously discovered construction
    try:
        base = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        print("[info] loaded previous construction")
    except Exception:
        base = None
        print("[info] no previous construction found")

    # Choose a base length (historically good size ~1900)
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 1900

    # Explore a handful of nearby lengths
    length_factors = [0.90, 0.95, 1.00, 1.05, 1.10]
    lengths = sorted({max(500, min(int(n0 * f), 8000)) for f in length_factors})

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- initialise -------------------------------------------------
        if base is not None:
            init_vec = _interpolate(base, n)
        else:
            init_vec = np.full(n, target_sum / n)

        init_vec = np.maximum(init_vec, 0.0)
        init_vec = init_vec * target_sum / init_vec.sum()

        # Dynamically allocate an iteration budget roughly proportional to n
        # – we aim for about 2‑3 × 10⁵ effective moves per length.
        max_iters = int(300_000 * 2000 / n)
        max_iters = max(120_000, max_iters)

        print(f"[run] length={n:5d}  max_iters≈{max_iters:,}")

        # ---- run the gradient‑guided SA --------------------------------
        best_seq, best_val = _gradient_guided_sa(
            initial=init_vec,
            deadline=deadline,
            rng=rng,
            max_iters=max_iters,
            grad_every=15,
        )

        # ---- keep global best -------------------------------------------
        if best_val < global_best_val - 1e-12:
            global_best_val = best_val
            global_best_seq = best_seq.copy()
            print(f"[global] new best across lengths = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – ensure the vector respects the simplex constraint
    if global_best_seq is None:
        # fallback to a safe uniform sequence
        n = n0
        target_sum = math.sqrt(2.0 * n)
        global_best_seq = np.full(n, target_sum / n)
        global_best_val = _max_convolution(global_best_seq, 1 << ((2 * n - 1).bit_length()))
        print("[fallback] using uniform sequence")

    # Verify (optional) – useful during development
    # cur_val = _max_convolution(global_best_seq, 1 << ((2 * global_best_seq.shape[0] - 1).bit_length()))
    # print(f"[verify] max conv = {cur_val:.6f}")

    print(f"[done] best value = {global_best_val:.6f}, length = {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
