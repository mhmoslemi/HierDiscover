#!/usr/bin/env python
"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

where ``a*a`` is the linear convolution (numpy.convolve).  The optimisation
uses three complementary phases:

1. **Rich initial pool** – deterministic shapes (uniform, Dirichlet,
   power‑law, exponential, Gaussian, cosine, arcsine, block‑patterns) and a
   warm‑start from the globally best sequence seen so far (``height_sequence_1``).

2. **Large‑scale Adam optimisation** on a smooth‑max surrogate.  The
   surrogate is the log‑sum‑exp of the convolution values; its gradient is
   obtained efficiently with FFT‑based cross‑correlation.  The temperature
   (smoothness) is increased aggressively so that the surrogate converges to
   the true max.

3. **Exact sub‑gradient (exponentiated‑gradient) phase** that directly
   minimises the true objective.  A simple adaptive step‑size is used and
   occasional random shakes escape shallow plateaus.

The routine respects a time budget (default 1000 s), stops gracefully when the
budget expires and returns the best sequence found as a plain ``list``.  The
global variable ``height_sequence_1`` is updated with the final best sequence,
allowing future calls to start from this solution.

The implementation is deliberately CPU‑light – all heavy convolutions are
FFT‑based (``O(n log n)``) and no external libraries beyond NumPy are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the linear part (length 2·len(a)‑1)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Output length = len(a);  (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)].
    """
    n = a.shape[0]
    rev_a = a[::-1]                               # reversed a
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """
    Exact objective (sum(a) is assumed to be 1).
    Returns 2·n·max(conv(a,a)).
    """
    conv = _conv_fft(a, conv_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Diverse initial candidates --------------------------------------------
def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """Create a list of plausible starting vectors (all normalised to sum‑1)."""
    cand = []

    # ------------------------------------------------------------------
    # 0) Warm‑start from the global best if it exists
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.clip(h, 0.0, None)
        if h.sum() < 1e-12:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # ------------------------------------------------------------------
    # 1) Uniform vector – the trivial lower‑bound baseline
    cand.append(np.full(n, 1.0 / n))

    # 2) Dirichlet draws with several concentrations
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 3) Edge‑concentrated power‑law (raised triangle)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 4) Symmetric exponential decay
    for gamma in (0.0005, 0.001, 0.003):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # 5) Cosine window (single bump)
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    cand.append(coswin / coswin.sum())

    # 6) Half‑sine window (zero at both ends)
    sinwin = np.sin(np.pi * i / (n - 1))
    cand.append(sinwin / sinwin.sum())

    # 7) Truncated Gaussian (σ adapts to n)
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    cand.append(gauss / gauss.sum())

    # 8) Truncated arcsine (inverse‑sqrt edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    cand.append(arcs / arcs.sum())

    # 9) Block‑pattern – uniform block occupying half the domain
    block = np.zeros(n, dtype=np.float64)
    block[: n // 2] = 1.0
    cand.append(block / block.sum())

    # 10) Two separated blocks (binary pattern)
    blk = np.zeros(n, dtype=np.float64)
    s = n // 8
    blk[:s] = 1.0
    blk[3 * s:4 * s] = 1.0
    cand.append(blk / blk.sum())

    # 11) Random sparse spikes (≈√n spikes)
    m_spikes = max(2, int(np.sqrt(n)))
    sparse = np.zeros(n, dtype=np.float64)
    sparse[np.random.choice(n, size=m_spikes, replace=False)] = 1.0
    cand.append(sparse / sparse.sum())

    # 12) Mixture: 70 % uniform + 30 % Gaussian
    mix = 0.7 * np.full(n, 1.0 / n) + 0.3 * gauss
    cand.append(mix / mix.sum())

    return cand


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_opt(a: np.ndarray,
              n: int,
              conv_len: int,
              corr_len: int,
              deadline: float,
              max_iters: int = 60000,
              lr_factor: float = 0.08,
              lam0: float = 0.2,
              lam_factor: float = 1.02) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max.
    Returns the final vector and its (exact) objective value.
    """
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lr = lr_factor * np.sqrt(1024.0 / n)
    lam = lam0
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ----- soft‑max projection to the simplex -----------------------
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- convolution & smooth‑max weights -------------------------
        conv = _conv_fft(a, conv_len)
        # log‑sum‑exp surrogate of max
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (quadratic form) ---------------
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through the soft‑max (mirror‑descent style)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -------------------------------------
        lam = min(lam * lam_factor, 1e6)

        # ----- exact objective -----------------------------------------
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ----- stagnation handling -------------------------------------
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


def _subgrad_opt(a: np.ndarray,
                 n: int,
                 conv_len: int,
                 corr_len: int,
                 deadline: float,
                 max_iters: int = 8000,
                 eta: float = 0.55) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (sub‑gradient) optimisation directly on the true max.
    Returns the best vector seen and its objective.
    """
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(max_iters):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()

        # active set of indices attaining the maximum (with tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:  # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # exact objective
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


def _local_shake(a: np.ndarray, magnitude: float = 5e-3) -> np.ndarray:
    """Tiny random perturbation of the logits (used to escape plateaus)."""
    n = a.shape[0]
    logits = np.log(a + 1e-15)
    logits += (np.random.rand(n) - 0.5) * magnitude
    exp_x = np.exp(logits - logits.max())
    return exp_x / exp_x.sum()


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation value.
    The routine respects the given time budget and returns the best sequence
    (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Length schedule – we focus most of the time on the two largest sizes.
    lengths = [1024, 2048, 4096, 8192]
    # Pre‑compute FFT lengths for each n.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _next_pow2(conv_len)
        # Correlation uses a vector of length conv_len + n - 1
        fft_len_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # --------------------------------------------------------------
        # Build a diversified pool of starting points.
        candidates = _build_initial_candidates(n)

        # --------------------------------------------------------------
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a = a0 / a0.sum()
            # ----------------------------------------------------------
            # Phase 1 – Adam on a smooth‑max surrogate
            max_iters_adam = int(40000 * 1024 / n) + 8000   # a few hundred k iters for small n
            a, val = _adam_opt(a, n, conv_len, corr_len, deadline,
                               max_iters=max_iters_adam,
                               lr_factor=0.09,
                               lam0=0.2,
                               lam_factor=1.018)

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} cand={cid:02d} Adam] new best {val:.6f}")

            # ----------------------------------------------------------
            # Phase 2 – exact sub‑gradient (exponentiated‑gradient)
            a, val = _subgrad_opt(a, n, conv_len, corr_len, deadline,
                                  max_iters=7000,
                                  eta=0.55)

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} cand={cid:02d} Subgrad] new best {val:.6f}")

            # ----------------------------------------------------------
            # Occasionally perform a cheap random shake to diversify.
            if np.random.rand() < 0.07:
                a_shake = _local_shake(a, magnitude=1e-2)
                a, val = _subgrad_opt(a_shake, n, conv_len, corr_len,
                                      deadline, max_iters=2000, eta=0.55)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} cand={cid:02d} Shake] new best {val:.6f}")

        # ------------------------------------------------------------------
        # Global fine‑tune for the current length (if we already have a
        # candidate of this length).
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a = best_seq.copy()
            a, val = _adam_opt(a, n, conv_len, corr_len, deadline,
                               max_iters=int(60000 * 1024 / n) + 5000,
                               lr_factor=0.045,
                               lam0=0.4,
                               lam_factor=1.012)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} fine‑tune] new best {val:.6f}")

            # short sub‑gradient polish
            a, val = _subgrad_opt(a, n, conv_len, corr_len, deadline,
                                  max_iters=4000, eta=0.5)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} polish] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed).
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("[fallback] using uniform vector")

    # expose the best found sequence globally for the next call
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    return best_seq.tolist()
