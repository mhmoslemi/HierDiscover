def _annealed_mirror_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _full_convolution_fft(a,a).max()
    tau_factor = 0.2   # initial factor relative to max conv
    min_tau = 1e-6
    tau = None
    step = 0.1  # base step size for exponentiated gradient
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if tau is None:
            tau = max(tau_factor * cur_max, min_tau)
        else:
            tau = max(tau * 0.99, min_tau)  # exponential decay
        # compute softmax weights
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of smoothed max
        # we need gradient of sum_k w_k * conv_k = 2 * a **conv w_rev? Actually gradient = 2 * (a convolve w_rev)
        w_rev = w[::-1]
        size2 = 1 << ((2 * a.shape[0] - 2).bit_length())
        # Use FFT to compute gradient efficiently:
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        grad = np.fft.irfft(Fa * Fw, n=size2)[: 3*a.shape[0] - 2]  # length 3n-2
        idx = (2 * a.shape[0] - 2) - np.arange(a.shape[0])
        grad_a = 2.0 * grad[idx]
        # Mirror descent update: log space
        eta = step / np.sqrt(it)  # diminishing step
        # update multiplicative factor
        a_exp = a * np.exp(-eta * grad_a)
        # renormalize to simplex
        a_new = a_exp * (target_sum / a_exp.sum())
        # evaluate
        val = _full_convolution_fft(a_new, a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            # optionally reset step or tau? we keep decaying.
        a = a_new
        # occasional random perturbation to escape local minima
        if it % 200 == 0 and np.random.rand() < 0.1:
            # add small Gaussian noise then renorm
            noise = np.random.randn(a.shape[0]) * a.mean() * 0.01
            a = np.maximum(a + noise, 0.0)
            a = a * (target_sum / a.sum())
        if best_val < 1.503:
            break
    return best_a, best_val

"""
Hybrid search for low‑autocorrelation step functions.

The evaluation metric (after normalising Σaᵢ = √(2 n)) reduces to
max(conv(a,a)).  The algorithm has four stages:

1. Random parametric families (exponential, power‑law, Hann, Hamming,
   Gaussian, Dirichlet) over a wide range of lengths.  The best
   candidate is recorded.

2. Deterministic‑annealing mirror descent on the simplex.
   We minimise a smoothed max‑convolution using a temperature τ that
   decays exponentially.  The update is performed in log‑space
   (multiplicative exponential‑gradient) and the vector is projected
   back onto the ℓ₁‑simplex after each step.

3. Exact sub‑gradient descent on the current worst convolution index,
   with an adaptive step size and projection onto the simplex.

4. A simple simulated‑annealing hill‑climb that moves tiny amounts of
   mass between random positions.

All phases respect the supplied time budget (default 1000 s) and the
best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper to update the global best solution
# ----------------------------------------------------------------------
def _try_update_best(seq: np.ndarray,
                     cur_best_seq: np.ndarray | None,
                     cur_best_val: float) -> tuple[np.ndarray, float]:
    """Return the updated best sequence/value pair."""
    val = _full_convolution_fft(seq, seq).max()
    if val < cur_best_val:
        print(f"[{time.time():.2f}] GLOBAL new best={val:.6f}  n={len(seq)}")
        return seq.copy(), float(val)
    return cur_best_seq, cur_best_val

# Random parametric initialisers
# ----------------------------------------------------------------------
def _generate_random_sequence(n: int) -> np.ndarray:
    """Create a random non‑negative sequence of length n and normalise it."""
    shape = np.random.choice(
        ["exp", "pow", "hann", "hamming", "gaussian", "dirichlet"]
    )
    if shape == "exp":
        tau = np.random.uniform(0.1, 2.5) * n
        a = np.exp(-np.arange(n, dtype=float) / tau)
    elif shape == "pow":
        p = np.random.uniform(0.2, 1.7)
        a = np.arange(1, n + 1, dtype=float) ** (-p)
    elif shape == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "hamming":
        i = np.arange(n, dtype=float)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "gaussian":
        sigma = np.random.uniform(0.1, 0.5) * n
        i = np.arange(n, dtype=float) - n / 2.0
        a = np.exp(-(i ** 2) / (2.0 * sigma ** 2))
    else:  # dirichlet

# ----------------------------------------------------------------------
def _annealed_mirror_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror descent on the simplex that minimises a
    temperature‑smoothed max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    it = 0
    tau = None               # temperature
    step_factor = 0.12       # base step size for the multiplicative update

        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()

        # initialise / decay temperature
        if tau is None:
            tau = max(0.2 * cur_max, 1e-6)
        else:
            tau = max(tau * 0.98, 1e-6)   # exponential cooling

        # soft‑max weights (log‑sum‑exp surrogate)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (2 a ⊛ w_rev)
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        # multiplicative (mirror‑descent) update
        eta = step_factor / np.sqrt(it)
        mult = np.exp(-eta * grad)
        a_new = a * mult

        # projection back onto the ℓ₁‑simplex
        a_new = a_new * (target_sum / a_new.sum())

        # evaluate new candidate
        val = _full_convolution_fft(a_new, a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] mirror it={it} new best={best_val:.6f}")

        a = a_new

        # occasional jitter to escape tiny plateaus
        if it % 300 == 0 and np.random.rand() < 0.1:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.maximum(a + noise, 0.0)
            a = a * (target_sum / a.sum())

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that always follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # modest increase after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               target_sum: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Hybrid optimiser that respects the supplied time budget and returns the
    best non‑negative step‑function (as a Python list) found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any
    # ------------------------------------------------------------------
    hs = globals().get("height_sequence_1")
    if hs is not None:
        hs_arr = np.asarray(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        best_seq, best_score = _try_update_best(hs_norm, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 1 – diverse random initialisations
    # ------------------------------------------------------------------
    # keep roughly 560 s for the later refinement phases
    while time.time() < deadline - 560.0:
        remaining = deadline - time.time()
        # choose a length (larger lengths earlier)
        if remaining > 700:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 400:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 4000 + 1)

        a0 = _generate_random_sequence(n)
        best_seq, best_score = _try_update_best(a0, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 2 – annealed mirror descent on the current best
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback uniform if we somehow have no candidate yet
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
    else:
        n = best_seq.shape[0]
        target = np.sqrt(2.0 * n)

    md_deadline = deadline - 300.0   # reserve ~300 s for sub‑grad & hill‑climb
    if time.time() < md_deadline:
        seq_md, val_md = _annealed_mirror_descent(best_seq, target, md_deadline)
        best_seq, best_score = _try_update_best(seq_md, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 3 – targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    sub_deadline = deadline - 70.0    # keep a little room for the final hill‑climb
    if time.time() < sub_deadline:
        seq_sg, val_sg = _targeted_subgradient(best_seq.copy(),
                                               target,
                                               sub_deadline)
        best_seq, best_score = _try_update_best(seq_sg, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 4 – final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq_hc, val_hc = _hillclimb(best_seq.copy(),
                                    target,
                                    deadline)
        best_seq, best_score = _try_update_best(seq_hc, best_seq, best_score)

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    print(f"[{time.time():.2f}] FINAL best max={best_score:.6f}  n={len(best_seq)}")
    return best_seq.tolist()
