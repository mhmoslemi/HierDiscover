def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = np.inf

    # lengths to try: start with moderate (2048) to find shape quickly
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Optimising n={n} ---")
        # get initial pool (10 candidates)
        candidates = _generate_initial_candidates(n, best_seq)
        # optionally add mixture of best two previously found
        if best_seq is not None and len(best_seq)==n:
            candidates.append(best_seq.copy())
        # limit to top 10
        candidates = candidates[:10]

        # iteration counts scaling: bigger n less loops
        max_adam = max(30000 * 2048 // n, 5000)  # at least 5k
        max_sub = max(12000 * 2048 // n, 2000)
        # for each candidate run variants
        for idx, init in enumerate(candidates):
            if time.time()>deadline:
                break
            # 1) Adam
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[n={n} cand={idx}] Adam new global best {val_adam:.6f}")

            # 2) Subgrad
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_sub)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[n={n} cand={idx}] Subgrad new global best {val_sub:.6f}")

            # 3) LP improvement (optional)
            if n <= 8192:  # keep small to avoid memory
                a_lp, val_lp = _lp_refine(best_seq, n, deadline)
                if val_lp is not None and val_lp < best_val:
                    best_val, best_seq = val_lp, a_lp.copy()
                    print(f"[n={n}] LP refine improved to {val_lp:.6f}")

            # 4) fine-tune small Adam on current best (same length)
            if best_seq is not None and len(best_seq)==n:
                a_fine, val_fine = _run_adam(best_seq, n, deadline, max_adam//2)
                if val_fine < best_val:
                    best_val, best_seq = val_fine, a_fine.copy()
                    print(f"[n={n}] fine-tune improved to {val_fine:.6f}")

        # after finishing candidates for this n, attempt hill climb on best
        if best_seq is not None and len(best_seq)==n and time.time()<deadline:
            a_hc, val_hc = _local_hill_climb(best_seq, n, deadline, max_trials=200_000)
            if val_hc < best_val:
                best_val, best_seq = val_hc, a_hc.copy()
                print(f"[n={n}] final hill-climb improved to {val_hc:.6f}")

    # final global hill climb
    if best_seq is not None and time.time()<deadline:
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline, max_trials=500_000)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f"[final] global hill-climb improved to {val_final:.6f}")

    # fallback uniform
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0/n_fallback)
        best_val = evaluate_sequence(best_seq.tolist())
        print("fallback uniform produced val", best_val)

    print(f"Best value found: {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a high‑quality non‑negative step‑function (vector of heights) that
minimises

    F(a) = 2·n·max(a∗a) / (∑a)²

where * denotes the linear convolution.  The algorithm consists of
    • a rich pool of initial shapes (uniform, Dirichlet, power‑law,
      exponentials, Hann, Gaussian, etc.) and an up‑sampled best‑so‑far;
    • a long Adam optimisation on a smooth‑max surrogate;
    • an exponentiated‑gradient sub‑gradient phase on the true max‑convolution;
    • a small linear‑programming “direction‑finding’’ step that solves
      max ∑g_j  s.t.  a∗g ≤ M (M = max(a∗a)), then mixes the solution;
    • a short fine‑tune and a cheap stochastic hill‑climb.
The whole search respects a user supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (fast integer version)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

# Optimisation primitives
# --------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """Adam optimiser on the smooth‑max surrogate."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.30          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → point on the simplex (soft‑max)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (sum(a)=1 ⇒ denominator drops out)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional random shake if we are stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """Exponentiated‑gradient sub‑gradient phase on the true max‑convolution."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """Very cheap stochastic local search."""
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

# Linear‑programming refinement (small‑scale, active‑set)
# --------------------------------------------------------------
def _lp_refine(a: np.ndarray, n: int, deadline: float, max_constraints: int = 500):
    """
    Solve   max Σ g_j   s.t.   a∗g ≤ M,   g ≥ 0
    where M = max(a∗a).  The solution g is scaled to sum to 1 and
    mixed back into ``a`` (or replaces it) if it yields a lower objective.
    Only the most critical convolution constraints (those close to M) are
    kept, keeping the LP small.
    Returns (new_a, new_val) or (a, None) if no improvement.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, conv_fft_len)
    M = conv.max()

    # Choose constraints where conv is near the maximum.
    # Keep at most ``max_constraints`` rows.
    # Prefer the absolute largest values.
    idx = np.argsort(-conv)[:max_constraints]
    rows = len(idx)

    # Build sparse matrix A where A[k, j] = a_{k-j} (if in range)
    data = []
    row_ind = []
    col_ind = []
    for r, k in enumerate(idx):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # i runs over a-index = k - j
        i_vals = np.arange(j_min, j_max + 1)
        j_vals = k - i_vals
        a_vals = a[i_vals]
        data.extend(a_vals.tolist())
        row_ind.extend([r] * len(i_vals))
        col_ind.extend(j_vals.tolist())

    A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    b = np.full(rows, M)

    # maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n)
    bounds = [(0, None)] * n

    # Linprog with the HiGHS interior point method handles sparse matrices efficiently.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

    if not res.success or res.x is None:
        return a, None

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return a, None

    g_norm = g / sum_g        # enforce Σ g = 1 (same normalization as a)

    # Evaluate the candidate.  First try the pure g_norm, otherwise blend.
    for t in (1.0, 0.75, 0.5, 0.25):
        cand = (1 - t) * a + t * g_norm
        conv_c = _full_convolution(cand, conv_fft_len)
        val_c = 2.0 * n * conv_c.max()
        if val_c < 2.0 * n * M - 1e-12:   # genuine improvement
            return cand, val_c

    return a, None

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects a time budget (`budget_s` seconds) and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths to explore – start small, finish with a large expressive one.
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Build a moderate pool of diversified initial points.
        candidates = _generate_initial_candidates(n, best_seq)
        # Ensure we have at most ten distinct starts (speed vs diversity trade‑off)
        candidates = candidates[:10]

        # Scale iteration budgets inversely with n (larger n → cheaper iterations)
        max_adam = max(30_000 * 2048 // n, 5_000)   # at least 5k
        max_sub = max(12_000 * 2048 // n, 2_000)

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ----- Adam (smooth‑max) -----
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # ----- Sub‑gradient (exponentiated‑gradient) -----
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_sub)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

            # ----- Small LP refinement (only for moderate n) -----
            if n <= 8192 and time.time() < deadline:
                a_lp, val_lp = _lp_refine(best_seq, n, deadline)
                if val_lp is not None and val_lp < best_val:
                    best_val, best_seq = val_lp, a_lp.copy()
                    print(f"[n={n}] LP refine improved global best to {val_lp:.6f}")

            # ----- Fine‑tune on the current best of this length -----
            if best_seq is not None and len(best_seq) == n and time.time() < deadline:
                a_fine, val_fine = _run_adam(best_seq, n, deadline, max_adam // 2)
                if val_fine < best_val:
                    best_val, best_seq = val_fine, a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

        # ----- Local stochastic hill‑climb for this length -----
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            a_hc, val_hc = _local_hill_climb(best_seq, n, deadline, max_trials=200_000)
            if val_hc < best_val:
                best_val, best_seq = val_hc, a_hc.copy()
                print(f"[n={n}] Final hill‑climb improved global best to {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Global cheap hill‑climb (last chance to squeeze a bit more)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Global final hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline,
                                               max_trials=500_000)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f"[final] Global hill‑climb improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = float('inf')
        print("Fallback uniform vector used (unexpected).")

    print(f"\nBEST RESULT: value={best_val:.6f}, length={len(best_seq)}")
    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
