def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math
    try:
        from scipy.sparse import diags
        from scipy.optimize import linprog
        use_lp=True
    except Exception:
        use_lp=False

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # choose base length
    n = 1024
    eps = 1e-9

    # warm start
    warm = globals().get('height_sequence_1')
    if warm is not None and len(warm) >= n:
        a0 = np.asarray(warm[:n], dtype=np.float64)
    else:
        a0 = np.random.dirichlet(np.ones(n))

    a0 = np.clip(a0, eps, None)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = float('inf')
    iteration = 0

    while time.time() < deadline - 1.0:
        # evaluate current
        val = evaluate_sequence(best_a.tolist())
        if val < best_val:
            best_val = val
            # print progress
            print(f"[Iter {iteration}] Current best {val:.6f}")
        # Compute M = max convolution
        conv = np.convolve(best_a, best_a)
        M = conv.max()
        # Linear programming step if possible
        if use_lp:
            # Build sparse matrix A
            offsets = np.arange(n, dtype=int)
            A = diags(best_a, offsets, shape=(2*n-1, n), format='csc')
            # Solve LP
            c = -np.ones(n)  # maximize sum(b)
            b_ub = np.full(2*n-1, M, dtype=np.float64)
            bounds = [(0, None)] * n
            # We may limit iteration for LP solving?
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
            if res.success:
                b = res.x
                # avoid tiny values
                b = np.clip(b, eps, None)
                b /= b.sum()
                # update mixing factor
                t = 0.2 / (1 + iteration * 0.01)
                best_a = (1 - t) * best_a + t * b
            else:
                print("LP failed:", res.message)
                # fall back to simple random move
                # random perturb
                i, j = np.random.choice(n, 2, replace=False)
                delta = best_a[i] * (np.random.rand() * 0.5)
                best_a[i] -= delta
                best_a[j] += delta
        else:
            # fallback: random local move
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                delta = best_a[i] * (np.random.rand() * 0.5)
                best_a[i] -= delta
                best_a[j] += delta
        # clip and renorm
        best_a = np.clip(best_a, eps, None)
        best_a /= best_a.sum()
        iteration += 1

    # after loop, upscale to larger size if time leaves
    if time.time() < deadline - 5:
        n_big = 4096
        # Interpolate
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_a)
        a_big = np.clip(a_big, eps, None)
        a_big /= a_big.sum()
        # Do a short Adam refinement similar to earlier code (reuse functions)
        # We'll reuse _adam_optimize defined earlier (but need to include definitions).
        # Or maybe not. For simplicity, we just return a_big.
        # Evaluate to see if improved
        val_big = evaluate_sequence(a_big.tolist())
        if val_big < best_val:
            best_val = val_big
            best_a = a_big

    # Final clip
    best_seq = np.clip(best_a, 0.0, 1000.0).tolist()
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

Algorithm
---------
1. Warm‑start with a previous solution (global `height_sequence_1`) or a
   random Dirichlet vector.
2. Repeatedly improve the shape on a modest dimension (n≈1024) by
   *computing the current maximal convolution* M = max(f∗f),
   *building a sparse Toeplitz matrix* that implements the linear map
   b ↦ f∗b,
   *solving a linear program* that maximises the total mass of a new
   distribution b while keeping f∗b ≤ M component‑wise.
   The LP pushes mass into positions where f is small; a convex combination
   `f←(1‑t)f+t·b` (with a decreasing step‑size t) strictly reduces the
   convolution maximum.
3. Once the time budget is almost exhausted, linearly interpolate the best
   1024‑vector to the target length 4096.
4. Perform a short Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate
   followed by an exact‑max sub‑gradient phase (the same machinery that the
   previous state‑of‑the‑art used).  This polishes the interpolated vector.
5. Return the best sequence found; it is guaranteed to be non‑negative and
   lies inside the evaluator’s accepted range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # start temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:05d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if it % 5_000 == 0 and time.time() - deadline > -5:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2_000 == 0:
                print(f"[Subgrad iter={it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float, eps: float = 1e-9) -> np.ndarray:
    """
    One LP‑based improvement step.
    Solve   max Σ b_j   s.t.  (f * b)[k] ≤ M  for all k,  b ≥ 0.
    Return the normalised solution b (sum = 1).  If LP fails, return f.
    """
    n = f.size
    # Build the sparse Toeplitz matrix A where A_{k,j}=f_{k-j} (0≤k-j<n)
    from scipy.sparse import diags

    data = []
    offsets = []
    # each offset d corresponds to a diagonal with constant value f[d]
    for d in range(n):
        length = n - d
        if length <= 0:
            continue
        data.append(np.full(length, f[d]))
        offsets.append(d)
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

    # Linear program: maximise sum(b)  ⇔ minimise -∑b
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use HiGHS (fast) if available
    from scipy.optimize import linprog

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fallback: tiny random perturbation
        print("[LP] solve failed, using random perturbation:", res.message)
        return f
    b = res.x
    # ensure positivity and normalise
    b = np.clip(b, eps, None)
    if b.sum() == 0:
        return f
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # small dimension for the fast LP phase
    n_small = 1024
    eps = 1e-9

    warm = globals().get("height_sequence_1")
    if warm is not None and len(warm) >= n_small:
        a = np.asarray(warm[:n_small], dtype=np.float64)
    else:
        a = np.random.dirichlet(np.ones(n_small))

    a = np.clip(a, eps, None)
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline - 5.0:
        # evaluate current candidate
        cur_val = evaluate_sequence(best_a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            print(f"[LP loop iter={it}] new best {cur_val:.6f}")

        # current max convolution M
        conv = np.convolve(best_a, best_a)
        M = conv.max()

        # LP improvement
        b = _lp_improve(best_a, M, eps=eps)

        # mixing step with decreasing step size
        t = 0.25 / (1.0 + it * 0.01)   # starts ~0.25, slowly shrinks
        best_a = (1.0 - t) * best_a + t * b
        best_a = np.clip(best_a, eps, None)
        best_a /= best_a.sum()

        it += 1
        if it >= 30:          # a few dozen LP solves are plenty
            break

    # ------------------------------------------------------------------
    # Upscale to the target size (4096) and polish with Adam+sub‑grad
    if time.time() < deadline - 5.0:
        n_big = 4096
        conv_fft_len = _next_pow2(2 * n_big - 1)
        corr_fft_len = _next_pow2(3 * n_big - 2)

        # linear interpolation
        x_old = np.linspace(0.0, 1.0, n_small, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_a)
        a_big = np.clip(a_big, eps, None)
        a_big /= a_big.sum()

        # short Adam phase on the larger grid
        a_adam, val_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=80_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_a = a_adam.copy()
            print(f"[After Adam] improved best {val_adam:.6f}")

        # sub‑gradient refinement
        keep_deadline = deadline
        a_sub, val_sub = _subgradient_optimize(
            best_a,
            n_big,
            conv_fft_len,
            corr_fft_len,
            keep_deadline,
            max_iters=120_000,
            eta=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_a = a_sub.copy()
            print(f"[After Subgrad] improved best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_a is None or best_a.size == 0:
        best_a = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_a, 0.0, 1000.0).tolist()
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq
