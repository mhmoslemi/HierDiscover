"""
Propose a non‑negative step‑function (a list of heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

where “∗” is linear convolution.
The algorithm runs within a wall‑clock budget (default 1000 s) and
returns the best sequence found.

Main ideas
---------
* Use a convex formulation: the objective is the ℓ∞‑norm of a
  quadratic map, thus we can apply first‑order methods safely.
* Optimise each candidate with two phases
  – Adam on a smooth‑max surrogate (log‑sum‑exp) to get into a good basin.
  – Exact ℓ∞ sub‑gradient refinement via an exponentiated‑gradient step.
* Generate a diverse pool of seeds:
  – Uniform, Dirichlet, power‑law, arcsine (β(½,½)) and other β‑distributions.
  – Random‑phase “spectrally flat’’ sequences (constant |FFT|, random phases).
  – Upsampled / down‑sampled version of the current best.
  – Small jitter around the incumbent.
* Process increasing lengths up to 65536 (the larger the length the lower
  the theoretical bound).  For the biggest sizes we keep the number of
  candidates small to stay within the time budget.
* All updates keep the vector on the probability simplex, guaranteeing
  non‑negativity and sum‑to‑one, which matches the scaling used inside
  `evaluate_sequence`.

The code is deliberately self‑contained, uses only NumPy and the standard
library, and prints intermediate progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2·n·max(conv).
    """
    eps = 1e-14
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.08

    # start with a fairly smooth surrogate and then sharpen it
    lam = 0.3
    lam_factor = 1.12
    lam_max = 1e6

    # number of iterations scales roughly with 1 / n
    max_iters = max(1500, int(25000 * 2048 / n) + 1500)

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and smooth‑max weights ----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ---- gradient of smooth‑max wrt a ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ---- back‑prop through soft‑max ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_max)

        # ---- exact objective (sum = 1) ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-4
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient step.
    Returns (best_a, best_val) after at most 4000 iterations or deadline.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(4000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # indices where the maximum is attained (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient multiplicative update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:  # numerical fallback
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluate
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995      # gentle decay

        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val

def _gen_uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _gen_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Random Dirichlet draw."""
    draw = np.random.dirichlet(np.full(n, alpha))
    return draw.astype(np.float64)


def _gen_powerlaw(n: int, exponent: float) -> np.ndarray:
    """
    Symmetric power‑law shape:
        shape[i] ∝ (i+1)*(n-i)   ^ exponent
    The exponent is negative for edge‑concentrated shapes.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, exponent), 0.0)
    if shape.sum() == 0:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape = shape / shape.sum()
    return shape


def _gen_arcsine(n: int) -> np.ndarray:
    """
    Discrete approximation of the Beta(½,½) (arcsine) density:
        f(x) ∝ 1/√(x·(1‑x)),   x∈(0,1)
    This yields strong edge concentration which empirically flattens the
    self‑convolution.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # avoid 0 and 1 exactly
    shape = 1.0 / np.sqrt(x * (1.0 - x))
    shape = shape / shape.sum()
    return shape


def _gen_beta(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete approximation of a Beta(a,b) density on {1,…,n}.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑beta to avoid underflow for extreme parameters
    logpdf = (a - 1) * np.log(x) + (b - 1) * np.log(1 - x)
    # shift to avoid -inf at the boundaries
    maxlog = np.max(logpdf)
    shape = np.exp(logpdf - maxlog)
    shape = shape / shape.sum()
    return shape


def _gen_spectral_random(n: int) -> np.ndarray:
    """
    Generate a sequence whose zero‑padded FFT magnitude is constant.
    Random phases are used; the resulting real signal is shifted to
    become non‑negative and normalised.
    """
    L = _next_pow2(2 * n - 1)
    # Hermitian symmetry for a real signal
    half = L // 2 + 1
    phases = np.random.uniform(0, 2 * np.pi, size=half)
    mag = np.ones_like(phases)  # flat magnitude
    freqs = mag * np.exp(1j * phases)
    # enforce conjugate symmetry manually (rfft already returns half‑spectrum)
    signal = np.fft.irfft(freqs, n=L)[:n]  # take first n samples
    # shift to be non‑negative
    signal = signal - signal.min()
    if signal.sum() <= 0:
        signal = np.random.dirichlet(np.ones(n))
    else:
        signal = signal / signal.sum()
    return signal.astype(np.float64)


def _upsample_or_downsample(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation between old and resized.
    """
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, old)
    if up.sum() > 0:
        up = up / up.sum()
    else:
        up = np.random.dirichlet(np.ones(new_n))
    return up


def _build_candidate_pool(n: int, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a heterogeneous set of starting points for length n.
    """
    candidates = []

    # 1. uniform
    candidates.append(_gen_uniform(n))

    # 2. best of same length (warm‑start)
    if best_seq is not None and best_seq.shape[0] == n:
        candidates.append(best_seq.astype(np.float64))

    # 3. up‑/down‑sample best of different size
    if best_seq is not None and best_seq.shape[0] != n:
        candidates.append(_upsample_or_downsample(best_seq, n))

    # 4. arcsine (Beta(½,½))
    candidates.append(_gen_arcsine(n))

    # 5. other Beta shapes (more or less edge heavy)
    candidates.append(_gen_beta(n, 0.3, 0.3))
    candidates.append(_gen_beta(n, 0.7, 0.7))

    # 6. power‑law edge shapes
    for exp in (-2.5, -2.0, -1.5, -1.0):
        candidates.append(_gen_powerlaw(n, exp))

    # 7. random Dirichlet draws
    for _ in range(2):
        candidates.append(_gen_dirichlet(n, alpha=1.0))

    # 8. spectral‑random candidate
    candidates.append(_gen_spectral_random(n))

    # 9. simple two‑Gaussian blend (edges)
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # Limit pool size for very large n to keep runtime reasonable
    max_pool = 12 if n <= 32768 else 6
    return candidates[:max_pool]

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the supplied wall‑clock budget (seconds)
    and returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0  # safety margin

    best_seq = None          # ndarray
    best_val = float("inf")  # the objective value 2·n·max(conv)

    # Process a cascade of increasing lengths; larger lengths give a lower
    # theoretical bound but are more expensive.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        candidates = _build_candidate_pool(n, best_seq)

        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            try:
                # Phase 1 – Adam on smooth‑max surrogate
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )
                # Phase 2 – exact ℓ∞ sub‑gradient refinement
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # Keep the best among the two phases
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cand_id}] new global best {best_val:.6f}")
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={cand_id}] new global best (Adam) {best_val:.6f}")

            except Exception as exc:
                print(f"[n={n} cand={cand_id}] exception: {exc}")

        # --------------------------------------------------------------
        # Jitter the incumbent of the current length to escape local minima
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            for jitter_id in range(4):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.018, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, _ = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # If time still permits, try one final upscale to the next power of two
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # attempt to double the length via interpolation and a short refinement
        cur_n = best_seq.shape[0]
        new_n = cur_n * 2
        if new_n <= 65536:
            print(f"\n=== Final upscale from {cur_n} to {new_n} ===")
            up_seq = _upsample_or_downsample(best_seq, new_n)
            conv_fft_len = _next_pow2(2 * new_n - 1)
            corr_fft_len = _next_pow2(3 * new_n - 2)

            a_adam, _ = _adam_optimize(
                up_seq, new_n, conv_fft_len, corr_fft_len, deadline
            )
            a_ref, val_ref = _subgradient_refine(
                a_adam, new_n, conv_fft_len, corr_fft_len, deadline
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[final upscale] new global best {best_val:.6f}")

    # --------------------------------------------------------------
    # Safety net – fall back to a uniform sequence if nothing found
    # --------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("[fallback] returning uniform sequence")

    # expose the best sequence as a global for potential future runs
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"\n=== Finished. Best value {best_val:.6f} with length {best_seq.shape[0]} ===")
    return best_seq.tolist()
