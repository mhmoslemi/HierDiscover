"""
Hybrid optimizer for the step‑function max‑convolution problem.

The objective for a non‑negative vector a of length n with
Σa = √(2n) is simply max(a * a) (linear convolution).
The code proceeds in several stages while respecting a 1000 s budget:

1. Warm‑start from any previously discovered global construction.
2. Diverse random initialisation (including exponential, power‑law,
   Hann/Hamming windows, Dirichlet draws and a *random‑phase* construction
   obtained from an inverse FFT of complex Gaussian noise).  Each candidate
   is projected onto the simplex and the best is kept.
3. Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The surrogate temperature is annealed and the logits are updated in the
   soft‑max parametrisation, guaranteeing feasibility at every step.
4. Targeted projected sub‑gradient descent that follows the exact worst
   convolution index (the same idea as in the reference solution, but with
   a slightly larger initial step and a more aggressive step‑size schedule).
5. Deterministic “peak‑balancing’’ refinement: for the current worst index k,
   the contribution of each coefficient a_i to that peak is w_i = a_{k‑i}
   (zero outside the support).  Mass is transferred from the index with the
   largest w_i to the index with the smallest w_i, projected onto the simplex,
   and accepted only if it reduces the peak.  This directly flattens the
   autocorrelation.
6. Final stochastic hill‑climb (simulated annealing) to polish any remaining
   irregularities.

All helper functions are top‑level, no closures or lambdas are used.
The routine prints progress and returns the best sequence found as a plain
Python list, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ---------------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ---------------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


# ---------------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ---------------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1       # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------------------
# Adam optimisation on a smoothed‑max (log‑sum‑exp) surrogate
# ---------------------------------------------------------------------------
def _adam_optimize(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.15,
    min_tau_factor: float = 5e-6,
    alpha0: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    Returns the best feasible vector (on the simplex) and its exact
    max‑convolution value.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- map logits to the simplex via softmax ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stability
        p = exp_shift / np.sum(exp_shift)       # Σ p = 1
        a = target_sum * p                       # feasible a

        # ---- exact max of the convolution (for book‑keeping) ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ---- smoothed max (log‑sum‑exp) weights ----
        tau = max(init_tau_factor * cur_max * (0.995 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective w.r.t. a ----
        # grad_a = 2 * (a * w) * reversed (convolution)
        # Use FFT: conv(a, w_rev) gives exactly the needed term
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ---- back‑prop to logits (chain rule through softmax) ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.15:
            x += np.random.randn(n) * 0.002

        # slowly decay learning rate
        alpha *= 0.9994

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ---------------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ---------------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the current worst convolution index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.25 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.03
            if it % 150 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.68
            # occasional jitter
            if np.random.rand() < 0.06:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ---------------------------------------------------------------------------
# Deterministic peak‑balancing refinement (local move guided by worst index)
# ---------------------------------------------------------------------------
def _balance_peak(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small amount of mass from the coefficient that
    contributes most to the current worst peak towards the one that contributes
    least, always staying on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    it = 0
    # a modest transfer fraction; will be reduced automatically if ineffective
    base_delta = target_sum * 0.001

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(best_a, best_a)
        k = int(conv.argmax())                # index of current worst peak

        # contribution weights for each coefficient to that peak
        w = np.zeros(n)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        idx = np.arange(start, end + 1, dtype=int)
        w[idx] = best_a[k - idx]             # w_i = a_{k-i} inside support

        i_max = int(np.argmax(w))
        i_min = int(np.argmin(w))
        if i_max == i_min:
            # pick second‑best min to ensure movement
            sorted_idx = np.argsort(w)
            for idx_candidate in sorted_idx:
                if idx_candidate != i_max:
                    i_min = idx_candidate
                    break

        delta = min(best_a[i_max], base_delta)
        if delta <= 0:
            break

        a_new = best_a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new = _project_onto_simplex(a_new, target_sum)

        new_val = _full_convolution_fft(a_new, a_new).max()
        if new_val < best_val:
            best_a = a_new
            best_val = new_val
            # slowly increase the transfer magnitude when progress is made
            base_delta = min(base_delta * 1.07, target_sum * 0.005)
            if it % 100 == 0:
                print(f"[{time.time():.2f}] balance it={it} improved to {best_val:.6f}")
        else:
            # shrink delta if not improving
            base_delta *= 0.85

        if best_val < 1.503:
            break
    return best_a, best_val


# ---------------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing)
# ---------------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.35)
    T_end = 5e-6
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer at most 5 % of mass at i
        delta = a[i] * np.random.rand() * 0.05
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 3000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ---------------------------------------------------------------------------
# Main entry point required by the evaluator
# ---------------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function with minimal
    evaluation score (i.e. minimal max convolution).  The routine respects
    the supplied time budget and returns the best sequence found as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf

    def _record(seq: np.ndarray):
        """Evaluate a sequence (via exact max‑convolution) and keep the global best."""
        nonlocal best_seq, best_score
        cur_val = _full_convolution_fft(seq, seq).max()
        if cur_val < best_score:
            best_score = float(cur_val)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return cur_val

    # ------------------------------------------------------------------
    # Warm‑start from any previously discovered global construction
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            _record(hs_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – diverse random initialisations (including random‑phase)
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:   # reserve ~250 s for later phases
        remaining = deadline - time.time()
        # pick a length – gradually favour larger n as time runs out
        if remaining > 600:
            n = np.random.randint(2500, 9000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 7500 + 1)
        else:
            n = np.random.randint(1500, 6000 + 1)

        target_sum = np.sqrt(2.0 * n)

        shape = np.random.choice(
            ["exp", "pow", "hann", "hamming", "dirichlet", "randomphase"]
        )
        if shape == "exp":
            tau = np.random.uniform(0.2, 2.5) * n
            idx = np.arange(n, dtype=float)
            a0 = np.exp(-idx / tau)
        elif shape == "pow":
            p = np.random.uniform(0.2, 1.7)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p)
        elif shape == "hann":
            i = np.arange(n, dtype=float)
            a0 = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        elif shape == "hamming":
            i = np.arange(n, dtype=float)
            a0 = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
        elif shape == "dirichlet":
            a0 = np.random.dirichlet(np.ones(n))
        else:  # randomphase
            # Create a random complex spectrum and take the magnitude of the real IFFT
            # This yields a non‑negative vector with a rich frequency content.
            rand_complex = np.random.randn(n) + 1j * np.random.randn(n)
            a0 = np.abs(np.real(np.fft.ifft(rand_complex)))
            # Ensure not all zeros (extremely unlikely)
            if a0.sum() == 0.0:
                a0 = np.ones(n)

        # optional tiny additive noise
        if np.random.rand() < 0.23:
            noise = np.random.randn(n) * np.max(a0) * 0.015
            a0 = np.maximum(a0 + noise, 0.0)

        # normalise to the required L1 sum
        a0 = a0 * (target_sum / a0.sum())
        _record(a0)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on a smoothed max‑convolution
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 150.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        # initialise logits from current best (soft‑max parametrisation)
        x0 = np.log(best_seq + 1e-12)

        run_deadline = time.time() + min(30.0, deadline - time.time() - 30.0)
        seq_local, val_local = _adam_optimize(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.18,
            min_tau_factor=2e-5,
            alpha0=0.055,
        )
        if val_local < best_score:
            best_score = val_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0   # keep a buffer for the next phases
        seq_sub, val_sub = _subgradient_descent(
            best_seq.copy(),
            target_sum,
            sub_deadline,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Deterministic peak‑balancing refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        bal_deadline = deadline - 15.0
        seq_bal, val_bal = _balance_peak(
            best_seq.copy(),
            target_sum,
            bal_deadline,
        )
        if val_bal < best_score:
            best_score = val_bal
            best_seq = seq_bal.copy()
            print(f"[{time.time():.2f}] after balancing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(
            best_seq.copy(),
            target_sum,
            hill_deadline,
        )
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
