def evaluate_shape(a):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    # ensure nonnegative and sum >0
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf

"""
Multi‑scale search for a non‑negative step function that minimises

    F(a) = 2 * n * max(a * a) / (sum(a)**2)

The algorithm proceeds in three stages:

1. **Coarse exploration** – for a series of lengths (starting at 512 and
   ending at the target length 32768) a large pool of structured candidates
   is generated.  Families include exponential edges, power‑law edges,
   truncated uniform edge blocks, sine‑power bumps and asymmetric exponentials.
   The objective (computed cheaply via FFT) is evaluated for every candidate
   and only the best few (≈8‑12) are kept.

2. **Refinement** – each retained candidate is refined by two gradient‑based
   phases:
   * Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) of the
     convolution maximum.
   * Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient step.
   Both phases respect the simplex constraint (non‑negative entries that
   sum to 1) and stop as soon as the global time budget (default 1000 s)
   is exhausted.

3. **Jitter** – after a length is processed, the current best of that length
   is jittered (log‑it perturbed) a few times and re‑optimised, allowing the
   search to escape shallow local minima.

The routine respects the wall‑clock deadline, prints progress and returns the
best sequence found as a plain Python list.  It works with the supplied
`evaluate_sequence` function without any additional I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Fast evaluation of the objective (no clamping)
# ----------------------------------------------------------------------


def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv)/sum(a)^2 for a non‑negative vector a."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a)
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Adam optimisation on smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth‑max of the convolution.

    Returns the best iterate (as a probability vector) together with its
    exact objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0               # temperature for smooth‑max
    lam_factor = 1.02

    # iteration budget grows when n is small; we also cap it to avoid overruns
    max_iters = int(60000 * 1024 / n) + 2000
    best_val = np.inf
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and smooth‑max weights ----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # ---- gradient of smooth‑max w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ---- back‑prop through soft‑max ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature annealing ----
        lam = min(lam * lam_factor, 1e8)

        # ---- exact objective (since sum(a)=1) ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- occasional random perturb to escape stagnation ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Exact max‑subgradient descent with an exponentiated‑gradient step."""
    a = a0 / a0.sum()
    best_val = np.inf
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                      # fallback to a random simplex point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                       # gentle decay

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

        for _ in range(4):
            p_l = np.random.uniform(0.01, 2.0)
            p_r = np.random.uniform(0.01, 2.0)
            raw_candidates.append(_shape_asym_exp(n, p_l, p_r))

        # --------------------------------------------------------------
        # Evaluate all raw candidates quickly and keep the most promising
        # --------------------------------------------------------------
        evaluated = []
        for cand in raw_candidates:
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-12:
                continue
            cand = cand / cand.sum()
            val = _fast_objective(cand)
            evaluated.append((val, cand))

        evaluated.sort(key=lambda x: x[0])
        top_k = 12 if n <= 8192 else 8
        top_candidates = [c for _, c in evaluated[:top_k]]

        # --------------------------------------------------------------
        # Refine each top candidate with Adam + exact sub‑gradient
        # --------------------------------------------------------------
        for cid, cand in enumerate(top_candidates):
            if time.time() > deadline:
                break
            try:
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                # Keep the best result seen so far
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cid}] new global best {best_val:.6f}")
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={cid} (adam)] new global best {best_val:.6f}")
            except Exception as exc:
                print(f"[n={n} cand={cid}] exception: {exc}")

        # --------------------------------------------------------------
        # Small jitter passes around the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(5):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.03, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guards against empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _fast_objective(best_seq)

    print(f"\nFinal best objective: {best_val:.6f}")
    return best_seq.tolist()
