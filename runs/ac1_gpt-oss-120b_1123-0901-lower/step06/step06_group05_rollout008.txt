"""propose_candidate – multi‑scale hybrid optimisation for low‑autocorrelation step
functions.

The routine explores several sequence lengths, builds a diverse set of start
vectors (dense shapes, random Dirichlet draws and sparse “spike” patterns),
optimises them with an Adam optimiser on a smooth‑max surrogate, refines the
solution by an exponentiated‑gradient sub‑gradient phase on the exact max
convolution, then applies cheap stochastic local search (hill‑climb + simulated
annealing).  All operations use FFT‑based convolutions for speed; the total run
time respects the supplied budget (default 1000 s) and returns the best
non‑negative sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

# Candidate generation ---------------------------------------------------
def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates_extended(n: int, prev_best):
    """
    Same as ``_generate_initial_candidates`` but also adds a few sparse
    “spike” patterns (random support of size proportional to n) which have
    proven useful for reducing the max‑autocorrelation.
    """
    base = _generate_initial_candidates(n, prev_best)

    # Sparse patterns: equal mass on a randomly chosen support.
    for frac in (0.01, 0.02, 0.05, 0.1):
        K = max(2, int(frac * n))
        for _ in range(3):
            idx = np.random.choice(n, K, replace=False)
            vec = np.zeros(n, dtype=np.float64)
            vec[idx] = 1.0 / K
            base.append(vec)

    return base

# Adam optimiser on a smooth‑max surrogate -------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best sequence seen during the run and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # we work in the simplex via a soft‑max of logits
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.03                      # a bit more gentle
    lam = 0.3                      # smoothing temperature
    lam_factor = 1.03              # slowly increase the temperature

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → point on the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (sum(a)=1 → denominator disappears)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights (soft‑max on the convolution vector)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum != 0.0:
            w /= w_sum
        else:
            w[:] = 1.0 / w.size

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

# Exponentiated‑gradient sub‑gradient phase on the true max ---------------
def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    EG sub‑gradient optimiser directly on the exact max‑convolution.
    Returns the best sequence encountered and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                     # initial step size (chosen experimentally)

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

# Cheap stochastic hill‑climb -------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search.
    Repeatedly moves a small mass ε from one index to another and keeps the
    move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            trials += 1
            continue
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

# Simple simulated‑annealing refinement ---------------------------------
def _simulated_annealing(a0: np.ndarray, n: int, deadline: float):
    """
    Light‑weight simulated‑annealing phase on the exact objective.
    Proposes random ε‑mass transfers and accepts worsening moves with a
    temperature that decays exponentially.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_val = 2.0 * n * _full_convolution(best_a, conv_fft_len).max()
    cur_a = best_a.copy()
    cur_val = best_val

    T = best_val * 0.1   # initial temperature
    decay = 0.9999

    while time.time() < deadline and T > 1e-6:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, cur_a[i]))
        if eps == 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= eps
        cand[j] += eps

        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()

        if val < cur_val or np.random.rand() < np.exp((cur_val - val) / T):
            cur_a, cur_val = cand, val
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Anneal] new best {val:.6f}")

        T *= decay

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The algorithm runs a multi‑scale hybrid optimiser (Adam → sub‑gradient →
    hill‑climb → annealing) under the supplied time budget and returns the best
    sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # Length schedule – larger n gives more expressive power but costs more.
    lengths = [4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Build a diverse set of start vectors (dense + sparse)
        candidates = _generate_initial_candidates_extended(n, best_seq)

        # Budgets scale roughly with 1/n so we stay inside the total time limit.
        max_adam = max(4_000, int(20_000 * 1024 / n))
        max_subgrad = max(2_000, int(8_000 * 1024 / n))

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam (smooth‑max) -------------------------------------------------
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # ---- Sub‑gradient (exponentiated‑gradient) -------------------------------
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

        # ---- Short fine‑tune on the current best for this length ------------------
        if best_seq is not None and len(best_seq) == n:
            a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                         max(1_000, max_adam // 2))
            if val_fine < best_val:
                best_val, best_seq = val_fine, a_fine.copy()
                print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ----------------------------------------------------------------------
    # Final cheap stochastic improvement (hill‑climb + annealing) ---------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_hc, val_hc = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_hc < best_val:
            best_val, best_seq = val_hc, a_hc.copy()
            print(f"[final] hill‑climb improved global best to {val_hc:.6f}")

        if time.time() < deadline:
            print(f"\n=== Final simulated annealing (n={len(best_seq)}) ===")
            a_sa, val_sa = _simulated_annealing(best_seq, len(best_seq), deadline)
            if val_sa < best_val:
                best_val, best_seq = val_sa, a_sa.copy()
                print(f"[final] annealing improved global best to {val_sa:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 4096
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
