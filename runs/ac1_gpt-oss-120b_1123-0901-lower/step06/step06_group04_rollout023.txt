def optimize_from_initial(a0, n, conv_fft_len, corr_fft_len, deadline):
    # Transform a0 to logits
    eps=1e-12
    a0 = np.maximum(a0, eps)
    a0 = a0 / a0.sum()
    logits = np.log(a0)

    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    # Parameter schedule
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 25.0, 50.0, 100.0, 500.0]  # increase
    for beta in betas:
        if time.time()>deadline: break
        # define objective function closure capturing beta
        def fun(logits_vec):
            # compute a, conv
            mx = logits_vec.max()
            exp_log = np.exp(logits_vec - mx)
            a = exp_log / exp_log.sum()
            conv = full_convolution_fft(a, conv_fft_len)
            # compute smoothmax and value
            m = conv.max()
            exp_beta = np.exp(beta * (conv - m))
            sum_exp = exp_beta.sum()
            smooth_max = (m + np.log(sum_exp) / beta)
            val = 2.0 * n * smooth_max
            return val

        def jac(logits_vec):
            # similar as above, compute gradient
            mx = logits_vec.max()
            exp_log = np.exp(logits_vec - mx)
            a = exp_log / exp_log.sum()
            conv = full_convolution_fft(a, conv_fft_len)
            m = conv.max()
            exp_beta = np.exp(beta * (conv - m))
            sum_exp = exp_beta.sum()
            p = exp_beta / sum_exp
            # gradient wrt a
            grad_a = 2.0 * _cross_correlation(p, a, corr_fft_len)  # shape n
            grad_F_a = 2.0 * n * grad_a
            avg = np.dot(a, grad_F_a)
            grad_log = a * (grad_F_a - avg)
            return grad_log

        res = minimize(fun, logits, jac=jac, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12})
        logits = res.x
        # compute a for current
        # check objective with exact evaluate_sequence (not surrogate) maybe compute exact after each beta
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

def cross_correlation(w, a, corr_fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, corr_fft_len)
    fr = np.fft.rfft(rev_a, corr_fft_len)
    corr = np.fft.irfft(fw * fr, corr_fft_len)
    start = n - 1
    return corr[start:start + n]

        corr_fft_len = 1 << int(np.ceil(np.log2(3*n-2)))

        # generate seeds
        seeds = []
        # uniform
        seeds.append(np.full(n, 1.0/n, dtype=np.float64))

        # previous best
        prev = globals().get('height_sequence_1')
        if prev is not None:
            old = np.asarray(prev, dtype=np.float64)
            if old.shape[0] != n:
                x_old = np.linspace(0, 1, old.shape[0])
                x_new = np.linspace(0, 1, n)
                new = np.interp(x_new, x_old, old)
                old = new
            seeds.append(old / np.maximum(old.sum(), 1e-12))

        # edge power law shapes
        idx = np.arange(1, n+1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)

                seeds.append(shape)

"""
Optimise a non‑negative step‑function (probability vector) to minimise
    F(a)=2·n·max(a*a) / (sum a)^2
using a smooth‑max surrogate and L‑BFGS‑B, followed by an exact
ℓ∞ sub‑gradient (exponentiated‑gradient) refinement.  A large, diverse
set of seeds (uniform, edge‑concentrated, random Dirichlet, warm‑starts)
is explored within the allotted time budget.  The best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the gradient."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _lbfgs_objective(logits, beta, n, conv_fft_len, corr_fft_len):
    """Smooth‑max surrogate of the objective (value only)."""
    mx = np.max(logits)
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()
    conv = _full_convolution(a, conv_fft_len)
    m = conv.max()
    # log‑sum‑exp with temperature beta
    exp_beta = np.exp(beta * (conv - m))
    sum_exp = exp_beta.sum()
    smooth_max = m + np.log(sum_exp) / beta
    return 2.0 * n * smooth_max


def _lbfgs_jac(logits, beta, n, conv_fft_len, corr_fft_len):
    """Gradient of the smooth‑max surrogate w.r.t. logits."""
    mx = np.max(logits)
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()
    conv = _full_convolution(a, conv_fft_len)
    m = conv.max()
    exp_beta = np.exp(beta * (conv - m))
    sum_exp = exp_beta.sum()
    p = exp_beta / sum_exp                     # soft‑max weights over the 2n‑1 convolutions
    # gradient w.r.t. a (smooth max part)
    grad_a = 2.0 * _cross_correlation(p, a, corr_fft_len)
    # gradient of the full objective 2·n·smooth_max
    grad_F_a = 2.0 * n * grad_a
    avg = np.dot(a, grad_F_a)
    grad_log = a * (grad_F_a - avg)            # chain rule through the soft‑max parameterisation
    return grad_log


def _subgradient_refine(a0: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float):
    """
    Exact ℓ∞ sub‑gradient refinement using exponentiated‑gradient
    updates.  The routine stops after a few thousand iterations or
    when the wall‑clock budget is exhausted.
    Returns (best_a, best_val).
    """
    a = a0 / a0.sum()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (indices achieving the maximum up to tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _optimize_with_lbfgs(a0: np.ndarray, n: int,
                         conv_fft_len: int, corr_fft_len: int,
                         deadline: float):
    """
    Run a cascade of L‑BFGS optimisations on the smooth‑max surrogate,
    starting from ``a0``.  After the annealing schedule a short exact
    sub‑gradient refinement is performed.  Returns the best pair
    (vector, exact objective value).
    """
    eps = 1e-12
    a0 = np.maximum(a0, eps)
    a0 = a0 / a0.sum()
    logits = np.log(a0)

    best_a = a0.copy()
    best_val = 2.0 * n * _full_convolution(best_a, conv_fft_len).max()

    # temperature schedule – increasingly sharp approximation of the max
    betas = [0.5, 2.0, 10.0, 50.0, 200.0]

    for beta in betas:
        if time.time() > deadline:
            break
        # a modest iteration budget – L‑BFGS is already quasi‑Newton
        max_iter = 80
        res = minimize(_lbfgs_objective,
                       logits,
                       args=(beta, n, conv_fft_len, corr_fft_len),
                       jac=_lbfgs_jac,
                       method='L-BFGS-B',
                       options={'maxiter': max_iter,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        logits = res.x

        # evaluate the *exact* objective for the current iterate
        mx = np.max(logits)
        a = np.exp(logits - mx)
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

    a_ref, val_ref = _subgradient_refine(best_a, n,
                                          conv_fft_len, corr_fft_len,
                                          deadline)
    if val_ref < best_val:
        best_val = val_ref
        best_a = a_ref

    return best_a, best_val

                x_new = np.linspace(0.0, 1.0, n)
                old = np.interp(x_new, x_old, old)
            old = np.maximum(old, 0.0)
            if old.sum() > 0:
                old = old / old.sum()
            else:
                old = np.random.dirichlet(np.ones(n))
            seeds.append(old)

        # edge‑concentrated power‑law shapes
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in [-3.0, -2.5, -2.0, -1.5]:
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                seeds.append(shape)

        # two‑step edge mass (first/last 5 % constant)
        r = max(1, n // 20)
        shape = np.zeros(n, dtype=np.float64)
        shape[:r] = 1.0 / r
        shape[-r:] = 1.0 / r
        shape = shape / shape.sum()
        seeds.append(shape)

        # mixture of uniform and Gaussian edges
        i = np.arange(n, dtype=np.float64)
        edge = np.exp(-0.5 * (i ** 2)) + np.exp(-0.5 * ((n - 1 - i) ** 2))
        edge = edge / edge.sum()
        mix = 0.5 * np.full(n, 1.0 / n) + 0.5 * edge
        mix = mix / mix.sum()
        seeds.append(mix)

        # random Dirichlet draws
        for _ in range(5):
            seeds.append(np.random.dirichlet(np.ones(n)))

        # -----------------------------------------------------------------
        # Optimise every seed
        # -----------------------------------------------------------------
        for sid, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            try:
                a_opt, val_opt = _optimize_with_lbfgs(seed_vec, n,
                                                      conv_fft_len,
                                                      corr_fft_len,
                                                      deadline)
                if val_opt < best_val:
                    best_val = val_opt
                    best_seq = a_opt.copy()
                    print(f"[n={n} seed={sid}] new global best {best_val:.6f}")
            except Exception as exc:
                print(f"[n={n} seed={sid}] exception: {exc}")

        # -----------------------------------------------------------------
        # Jitter around the current best of this length
        # -----------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter in range(6):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()
                try:
                    a_opt, val_opt = _optimize_with_lbfgs(a0, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          deadline)
                    if val_opt < best_val:
                        best_val = val_opt
                        best_seq = a_opt.copy()
                        print(f"[n={n} jitter={jitter}] new global best {best_val:.6f}")
                except Exception as exc:
                    print(f"[n={n} jitter={jitter}] exception: {exc}")

        # expose for the next length
        globals()['height_sequence_1'] = best_seq.tolist() if best_seq is not None else None

    # -----------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    return best_seq.tolist()
