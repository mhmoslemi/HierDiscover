def solve_lp_for_g(a, M):
    n = len(a)
    # Build sparse matrix A of shape (2*n-1, n)
    rows = []
    cols = []
    data = []
    for i in range(2*n-1):
        start = max(0, i - (n-1))
        end = min(i, n-1)
        for j in range(start, end+1):
            # a[j] * g[i-j]
            rows.append(i)
            cols.append(i-j) # because g index = i - j
            data.append(a[j])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ g <= M
    # also g >= 0

    g = cp.Variable(n)
    constraints = [A @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    g_opt = g.value
    if g_opt is None or np.isnan(g_opt).any():
        return None
    # Normalize to sum=1
    s = g_opt.sum()
    if s <= 0:
        return None
    g_norm = g_opt / s
    return g_norm

def conv_fft(a, fft_len):
    # compute full linear convolution a*a using FFT
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)
    return conv[:2*len(a)-1]

def corr_fft(w, a, fft_len):
    # cross-correlation w★a
    # w length = 2n-1 (same as conv)
    n = len(a)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = n-1
    return corr[start:start+n]

def project_simplex(v):
    # onto simplex (nonneg sum=1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? as fallback uniform
        return np.full_like(v, 1.0/len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        if time.time() > deadline:

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The implementation improves on the previous reference by

* adding a richer set of initial shapes (beta‑distribution windows,
  Gaussian, random sparse spikes, interpolated warm‑starts),
* using a longer Adam phase with an aggressive smooth‑max temperature,
* a decreasing‑step‑size exponentiated‑gradient (EG) phase,
* a final fine‑tune Adam pass that runs as long as the time budget allows,
* a lightweight simplex‑projection routine (used only for safety).

All heavy work (convolution, cross‑correlation) is performed with FFTs
(`np.fft.rfft/irfft`) in O(n log n).  The routine respects the given
budget (default 1000 s) and returns the best sequence discovered as a
plain Python list.
"""

import time
import math
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow_two(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << int(math.ceil(math.log2(max(1, x))))


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the linear part
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Simplex projection (used only as a safeguard) -------------------------

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_len: int,
                   corr_len: int,
                   deadline: float,
                   max_iters: int = 20000,
                   lr: float = 0.07,
                   lam_start: float = 0.2,
                   lam_factor: float = 1.018) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of ``max(a*a)``.
    Returns the best (a, value) pair found.
    """
    # work in the log‑soft‑max (logits) domain – this automatically enforces
    # non‑negativity and Σa = 1.
    logits = np.log(np.maximum(a0, 1e-15))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start

    best_val = float("inf")
    best_a = a0.copy()
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---------- soft‑max mapping ----------
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---------- smooth‑max weights ----------
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ---------- gradient ----------
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature annealing – move towards the true max
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a) == 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # gentle random shake if we have stalled
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _eg_optimize(a0: np.ndarray,
                 n: int,
                 conv_len: int,
                 corr_len: int,
                 deadline: float,
                 max_iters: int = 6000,
                 eta0: float = 0.55) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (sub‑gradient) phase.
    Uses a decreasing step‑size η_t = η0 / √t.
    """
    a = a0.copy()
    best_val = float("inf")
    best_a = a.copy()
    last_imp = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        else:
            w[:] = 1.0 / conv.size

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:                # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_imp = time.time()

        if time.time() - last_imp > 8.0:
            break

    return best_a, best_val

def _build_initial_candidates(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Creates a diversified pool of starting points.
    ``previous`` may be a sequence from a smaller length – it will be
    interpolated to the current size and added as a warm‑start.
    """
    cand = []

    # warm‑start from a previous solution (if supplied)
    if previous is not None:
        # linear interpolation to the new length
        x_old = np.linspace(0.0, 1.0, previous.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        h = np.interp(x_new, x_old, previous)
        h = np.maximum(h, 0.0)
        if h.sum() > 0:
            h = h / h.sum()
            cand.append(h)

    # 1) random Dirichlet (concentration 1)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 2) over‑dispersed Dirichlet (≈ uniform)
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 3) spiky Dirichlet (conc < 1)
    for conc in (0.2, 0.4, 0.6):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) beta‑distribution windows (flexible edge/center behaviour)
    for _ in range(4):
        alpha = 0.1 + 5.0 * np.random.rand()
        beta = 0.1 + 5.0 * np.random.rand()
        i = np.arange(n, dtype=np.float64)
        x = i / (n - 1)
        shape = (x ** alpha) * ((1.0 - x) ** beta)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # 5) Gaussian bump (σ scaled with n)
    sigma = n / 8.0
    i = np.arange(n, dtype=np.float64)
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    if gauss.sum() > 0:
        gauss = gauss / gauss.sum()
        cand.append(gauss)

    # 6) random sparse spike pattern (≈ √n spikes)
    m_spikes = max(2, int(np.sqrt(n)))
    for _ in range(2):
        sparse = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, size=m_spikes, replace=False)
        sparse[idx] = np.random.rand(m_spikes) + 0.1
        if sparse.sum() > 0:
            sparse = sparse / sparse.sum()
            cand.append(sparse)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence (list of heights) that minimises
    the objective defined in ``evaluate_sequence``.

    The function respects the supplied time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – we start small, finish with the largest feasible.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # Keep the current best in a NumPy array for easy up‑sampling.
    best_np = None

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = _next_pow_two(2 * n - 1)
        corr_len = _next_pow_two(2 * n - 1 + n - 1)

        # ------------------------------------------------------------------
        # Build a modest set of diversified initial candidates.
        candidates = _build_initial_candidates(n, best_np)

        # ------------------------------------------------------------------
        # Optimise each candidate in turn.
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
                s = a0.sum()
            a0 = a0 / s

            # ----- Phase 1: Adam on smooth‑max surrogate -----------------
            max_iters_1 = int(30000 * 1024 / n) + 5000
            lr_1 = 0.07 * math.sqrt(2048.0 / n)
            a, val = _adam_optimize(a0, n, conv_len, corr_len,
                                    deadline,
                                    max_iters=max_iters_1,
                                    lr=lr_1,
                                    lam_start=0.2,
                                    lam_factor=1.02)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                best_np = a.copy()
                print(f"[n={n} cand={cid} Adam] new best {val:.6f}")

            # ----- Phase 2: Exponentiated‑gradient sub‑gradient ------------
            a, val = _eg_optimize(a, n, conv_len, corr_len,
                                  deadline,
                                  max_iters=8000,
                                  eta0=0.6)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                best_np = a.copy()
                print(f"[n={n} cand={cid} EG] new best {val:.6f}")

        # ------------------------------------------------------------------
        # Global fine‑tune for the current length (if we have a solution)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            max_iters_fine = int(60000 * 1024 / n) + 10000
            lr_fine = 0.045 * math.sqrt(2048.0 / n)
            a, val = _adam_optimize(best_seq, n, conv_len, corr_len,
                                    deadline,
                                    max_iters=max_iters_fine,
                                    lr=lr_fine,
                                    lam_start=0.4,
                                    lam_factor=1.012)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                best_np = a.copy()
                print(f"[n={n} fine‑tune] new best {val:.6f}")

        print(f"[n={n}] best value so far: {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final post‑processing: a short extra Adam run on the overall best.
    if best_seq is not None:
        n = len(best_seq)
        conv_len = _next_pow_two(2 * n - 1)
        corr_len = _next_pow_two(2 * n - 1 + n - 1)
        print("\n=== Final polishing of the global best ===")
        a, val = _adam_optimize(best_seq, n, conv_len, corr_len,
                                deadline,
                                max_iters=50000,
                                lr=0.03 * math.sqrt(2048.0 / n),
                                lam_start=0.5,
                                lam_factor=1.015)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"Final polish improved value to {val:.6f}")

    # ----------------------------------------------------------------------
    # Safety fallback – should never be needed.
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\n=== SEARCH COMPLETE – best objective ≈ {best_val:.6f} ===")
    return best_seq.tolist()
