def _apgd_optimize(a0, n, target_sum, deadline, max_iters, best_seq, best_score, rng):
    a = a0.copy()
    a = a * target_sum / a.sum()
    Nfft = 1 << ((3*n - 2).bit_length())
    # initial conv
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
    M = conv.max()
    # temperature start as M
    tau = M
    tau_min = 1e-9
    # schedule: tau *= decay_factor each iteration (or each epoch)
    decay = 0.9995   # per iteration
    # step size schedule: lr = init_lr / sqrt(it)
    init_lr = 0.02 * target_sum   # maybe as earlier
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # recompute conv every iteration (maybe we can reuse but conv changes)
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = float(conv.max())
        # update best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[APGD n {n} iter {it}] new best = {best_score:.6f}")
        # compute softmax weights
        if tau > 1e-9:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()
        else:
            # hard max
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)

        # weighted reverse convolution
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n -1]
        idx_start = 2*n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev
        # momentum
        # maybe use simple Nesterov: a = project(a - lr * g + momentum*prev_delta)
        # but we need store previous delta:
        lr = init_lr / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # schedule temperature
        tau = max(tau * decay, tau_min)

        # occasional jitter
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)
    return best_seq, best_score
```

"""
search for a non‑negative step function minimizing ``evaluate_sequence``.

Approach
--------
1.  **Annealed Projected Gradient Descent (APGD)**
    – minimise a smooth approximation of the max‑convolution:
      Fτ(a)=τ·log∑exp((a∗a)/τ).  The gradient is obtained via a single
      weighted reverse FFT.  After each step the vector is projected onto
      the simplex {a_i≥0, Σa_i=√(2 n)}.
    – τ is slowly decreased (geometric decay) so the optimisation
      gradually focuses on the true maximum.

2.  **Gradient‑guided Simulated Annealing**
    – after APGD a stochastic hill‑climber moves a small fraction of mass
      from indices with a large gradient (high contribution to the max)
      to indices with a small gradient.  Improvements are always accepted,
      worsening moves are accepted with an annealing probability.

The whole routine respects the 1000 s budget, dynamically allocating time
to the optimisation and the final refinement.  If a previous construction
is available via the global ``height_sequence_1`` it is interpolated to the
current length and used as an additional start point.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto {x≥0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initial vectors -------------------------------------------------------
def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """A small, diverse set of feasible start vectors."""
    init = []

    # uniform
    init.append(np.full(n, target_sum / n))

    # Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # exponential decay with random rate
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (
        a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
        + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    )
    init.append(black / black.sum() * target_sum)

    # small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(
        scale=0.01 * target_sum / n, size=n
    )
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # triangular shape (peak in centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # sparse random spikes
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init


# ----------------------------------------------------------------------
# Annealed Projected Gradient Descent -----------------------------------
def _apgd_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Annealed PGD on the soft‑max approximation of max (a∗a).

    Returns the best vector observed (in terms of the exact max‑convolution)
    and its max value.
    """
    a = a0.copy()
    a = a * target_sum / a.sum()               # enforce the simplex

    # FFT size that can hold a linear convolution of length 3·n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # step‑size and temperature schedules
    step0 = 0.001 * target_sum                  # base step size
    tau = None                                  # temperature (softmax τ)
    decay_tau = 0.9995                          # geometric decay per iteration
    eps_active = 1e-12

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- convolution & current max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[APGD n {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- temperature initialisation -----
        if tau is None:
            tau = M

        # ----- soft‑max weights (smooth max) -----
        if tau > 1e-12:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()
        else:
            # hard max when temperature is negligible
            mask = conv >= M - eps_active
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)

        # ----- weighted reverse convolution (gradient) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev                              # ∇_a max(a∗a) (soft version)

        # ----- gradient step with diminishing step size -----
        lr = step0 / math.sqrt(it + 1.0)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- temperature decay -----
        tau = max(tau * decay_tau, 1e-12)

        # ----- occasional jitter to escape plateaux -----
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Gradient‑guided simulated annealing refinement ------------------------
def _local_refine_adapt(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that preferentially moves mass from indices with
    large gradient (high contribution to the max) to indices with small
    gradient.  Acceptance follows a simple simulated‑annealing schedule.
    """
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for plain convolution (max evaluation)
    Nfft_conv = 1 << ((2 * n - 1).bit_length())
    # FFT size for gradient (rev) evaluation when needed
    Nfft_rev = 1 << ((3 * n - 2).bit_length())

    max_steps = 600_000
    step = 0
    grad_interval = 4000          # recompute gradient every this many steps
    rev = None

    while time.time() < deadline and step < max_steps:
        # recompute gradient periodically
        if step % grad_interval == 0 or rev is None:
            A_fft = np.fft.rfft(cur_seq, n=Nfft_rev)
            conv = np.fft.irfft(A_fft * A_fft, n=Nfft_rev)[: 2 * n - 1]
            M = conv.max()
            mask = conv >= M - 1e-12
            w = mask.astype(float) / max(mask.sum(), 1)
            W_rev_fft = np.fft.rfft(w[::-1], n=Nfft_rev)
            conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft_rev)
            conv_rev = conv_rev[: len(w) + n - 1]
            idx_start = 2 * n - 2
            rev = conv_rev[idx_start - np.arange(n)]

        # source – index with largest gradient, target – smallest gradient
        src = int(np.argmax(rev))
        dst = int(np.argmin(rev))

        # fallback to random move if the chosen source is empty
        if cur_seq[src] < 1e-12:
            src, dst = rng.integers(0, n, size=2)
            if src == dst or cur_seq[src] < 1e-12:
                step += 1
                continue

        # transfer a random fraction (2 % – 15 %) of the source mass
        frac = rng.uniform(0.02, 0.15)
        delta = cur_seq[src] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # evaluate new max‑convolution (exact)
        A_fft = np.fft.rfft(new_seq, n=Nfft_conv)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft_conv)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score

        step += 1

    return cur_seq, cur_score

    print(f"[done] best score ≈ {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
