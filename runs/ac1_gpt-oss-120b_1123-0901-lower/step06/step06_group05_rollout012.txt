            w /= w_sum

        logits -= step

            # small random perturbation to logits
            logits += (np.random.rand(n) - 0.5) * 2e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

"""
Propose a step‑function (list of non‑negative heights) that minimises

    F(a) = 2·n·max(a * a) / (sum(a)²)

using a multi‑stage optimisation:
  * generate a diverse pool of initial candidates (Dirichlet, power‑law, Hann,
    Gaussian, warm‑starts …)
  * run a custom Adam optimiser on a smooth‑max surrogate of the objective
  * refine with an exponentiated‑gradient sub‑gradient phase on the true max
  * build a small evolutionary population and repeatedly mix / mutate the
    best individuals, each child undergoing a short Adam+sub‑gradient refinement
  * finish with a cheap stochastic hill‑climb that moves a small mass ε
    between two positions.
The routine respects the supplied time budget (default 1000 s) and always
returns a plain Python list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (fast integer version)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam_custom(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence seen during the run and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.2                     # relatively aggressive
    lam = 0.5                     # temperature for smooth‑max
    lam_factor = 1.07             # accelerate temperature increase

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ----- map logits → point on the simplex (soft‑max) -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- exact objective (sum(a)=1 ⇒ denominator drops out) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t}] new best {val:.6f}")

        # ----- smooth‑max weights (soft‑max on the convolution vector) -----
        w = np.exp(lam * (conv - max_c))          # shift by max for stability
        w_sum = w.sum()
        if w_sum != 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- back‑prop through the soft‑max (logits) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e5)

        # ----- occasional random shake if we are stuck -----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_subgrad_custom(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    Returns best sequence and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.3                     # initial stepsize

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        else:
            grad_a = np.zeros_like(a)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                        # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      max_trials: int = 200_000):
    """
    Cheap stochastic local search.
    Moves a small mass ε from one index to another and keeps the move if it reduces
    the exact objective.  Stops after ``max_trials`` or after a few thousand
    consecutive failures.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 8_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The algorithm proceeds through several lengths, builds a diverse
    population of candidates, refines each with Adam + sub‑gradient,
    performs a lightweight evolutionary search, and finishes with a
    stochastic hill‑climb.  The best sequence encountered within the time
    budget is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths to explore – larger values give richer families but cost more.
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # 1) Build a heterogeneous initial pool (warm‑start, random, analytic)
        candidates = _generate_initial_candidates(n, best_seq)

        # add a few synthetic extreme shapes
        unit_mid = np.zeros(n)
        unit_mid[n // 2] = 1.0
        candidates.append(unit_mid)

        unit_edge = np.zeros(n)
        unit_edge[0] = 1.0
        candidates.append(unit_edge)

        two_lumps = np.zeros(n)
        two_lumps[n // 4] = 0.5
        two_lumps[3 * n // 4] = 0.5
        candidates.append(two_lumps)

        # ------------------------------------------------------------------
        # 2) Initialise a small population (keep at most pop_sz best)
        pop_sz = 8
        population = []                     # list of (value, sequence)

        # budgets for the basic optimiser (scaled with n)
        max_adam = max(5_000, int(20_000 * 2048 / n) + 2_000)
        max_subgrad = max(2_000, int(5_000 * 2048 / n) + 1_000)

        # limit the number of full‑scale candidates (time‑budget safeguard)
        max_cands = 6
        for cid, init in enumerate(candidates[:max_cands]):
            if time.time() > deadline:
                break

            # guarantee a proper simplex point
            init = np.asarray(init, dtype=np.float64)
            if init.sum() == 0:
                init = np.random.dirichlet(np.ones(n))
            else:
                init = init / init.sum()

            # ---- Adam (smooth‑max) ----
            a_adam, _ = _run_adam_custom(init, n, deadline, max_adam)

            # ---- Sub‑gradient (exponentiated‑gradient) ----
            a_sub, val_sub = _run_subgrad_custom(a_adam, n, deadline, max_subgrad)

            # store in population
            if len(population) < pop_sz:
                population.append((val_sub, a_sub))
                population.sort(key=lambda x: x[0])
            else:
                if val_sub < population[-1][0]:
                    population[-1] = (val_sub, a_sub)
                    population.sort(key=lambda x: x[0])

            # update global best
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[n={n} cand={cid}] New global best {val_sub:.6f}")

        # ------------------------------------------------------------------
        # 3) Light‑weight evolutionary refinement (mix / mutate / fine‑tune)
        # Parameters for child optimisation
        child_adam_iters = max(300, int(max_adam * 0.1))
        child_subgrad_iters = max(150, int(max_subgrad * 0.1))
        sigma = 0.02                     # mutation strength

        while time.time() < deadline - 5.0:   # keep a margin for final climb
            if len(population) < 2:
                break
            # select two parents (biased towards better individuals)
            probs = np.arange(len(population))[::-1] + 1
            probs = probs / probs.sum()
            p1_idx, p2_idx = np.random.choice(len(population), 2, replace=False, p=probs)
            p1 = population[p1_idx][1]
            p2 = population[p2_idx][1]

            # crossover + mutation
            child = (p1 + p2) * 0.5
            child = child * (1.0 + sigma * np.random.randn(n))
            child = np.maximum(child, 0.0)
            s = child.sum()
            if s < 1e-12:
                child = np.random.dirichlet(np.ones(n))
                s = child.sum()
            child = child / s

            # brief Adam + sub‑gradient refinement
            child_adam, _ = _run_adam_custom(child, n, deadline, child_adam_iters)
            child_sub, child_val = _run_subgrad_custom(child_adam, n, deadline, child_subgrad_iters)

            # update global best
            if child_val < best_val:
                best_val, best_seq = child_val, child_sub.copy()
                print(f"[n={n} Evol] Global best improved to {child_val:.6f}")

            # possibly insert child into the population
            if child_val < population[-1][0]:
                population[-1] = (child_val, child_sub.copy())
                population.sort(key=lambda x: x[0])

        # ------------------------------------------------------------------
        # end of this length – continue with the next (larger) one
        # ------------------------------------------------------------------

    # ----------------------------------------------------------------------
    # 4) Final cheap hill‑climb on the overall best candidate
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq),
                                               deadline, max_trials=300_000)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final
            print(f"[final hill‑climb] improved to {val_final:.6f}")

    # ----------------------------------------------------------------------
    # 5) Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
