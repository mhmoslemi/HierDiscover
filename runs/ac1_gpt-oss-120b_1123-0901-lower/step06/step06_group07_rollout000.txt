"""
Multi‑scale optimisation for the step‑function bound.

The algorithm works in three phases:

1. **Coarse (n≈1024)** – generate a wide variety of parametric candidates,
   evaluate them cheaply, keep the best few and run a long Adam optimisation
   on a smooth‑max surrogate (soft‑max of the convolution).  This gives a
   good “shape” quickly.

2. **Medium (n≈4096)** – up‑sample the best coarse solution and continue the
   Adam optimisation, now with a tighter temperature schedule.  Afterwards a
   few exponentiated‑gradient sub‑gradient steps (true max‑convolution) sharpen
   the result.

3. **Fine (n≈8192)** – if time permits, repeat the previous step on a doubled
   grid.  The final vector is clipped to the evaluator’s allowed range and
   returned as a plain list.

All optimisation stays on the probability simplex via a soft‑max (logits)
parameterisation; the surrogate’s temperature β is increased gradually so the
optimiser first sees a smooth landscape and finally focuses on the true max.
The sub‑gradient phase uses the exact active set of the convolution peak.

The routine respects the 1000 s budget, prints progress and always returns a
feasible sequence (uniform vector as a fallback).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)-1 using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective evaluator (fast, assumes a already normalised)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Adam optimisation on smooth‑max surrogate
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp (smooth‑max) surrogate.
    The parameters follow a slowly increasing temperature β.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in log‑space => positivity
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    beta = 0.5          # starting temperature
    beta_factor = 1.018
    beta_cap = 5e5

    it = 0
    max_iter = 200_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ---- project logits → probability vector (soft‑max) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and surrogate ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # true objective (used for record‑keeping)
        true_val = 2.0 * n * max_c / (a.sum() ** 2)

        if true_val < best_val:
            best_val = true_val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {true_val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_corr(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)          # ∂L/∂logits

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        beta = min(beta * beta_factor, beta_cap)

        # ---- occasional random shake if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

# Sub‑gradient refinement using the exact max‑convolution
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 15_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_corr(w, a, corr_fft_len)

        # exponentiated‑gradient update (maintains positivity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            # no improvement for a while → stop early
            break

    return best_seq, best_val

# Overall deep optimisation (Adam + sub‑gradient)
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

# Candidate generation (diverse parametric families)
def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    cand = []

    # warm‑start + shifts
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm_start, shift))

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # random Dirichlet with a spread of concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (mixed with uniform)
    for alpha in np.logspace(-2, 1.0, num=20):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=20):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # two‑Gaussian edges
    for gamma in (0.05, 0.15, 0.3, 0.6, 1.2, 2.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # single‑spike at each end (tiny block of mass)
    for block in (1, 2, 4, 8, 16):
        vec = np.zeros(n)
        mass = 0.5
        vec[:block] = mass / block
        vec[-block:] = mass / block
        # distribute remaining mass uniformly
        remaining = 1.0 - mass
        vec[block:-block] = remaining / (n - 2 * block)
        cand.append(vec)

    # half‑sinusoid (used previously)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    return cand


# ----------------------------------------------------------------------
# Upsampling (linear interpolation between normalised positions)
def _upsample(vec: np.ndarray, old_n: int, new_n: int) -> np.ndarray:
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    if up.sum() < 1e-12:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up /= up.sum()
    return up


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Warm start: possibly a previously discovered best vector
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.ndim != 1:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Phase 1 – low resolution optimisation (n = 1024)
    n0 = 1024
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # generate a rich set of candidates (fast evaluation)
    candidates0 = _generate_candidates(n0, warm_start)

    cand_vals = []
    for idx, cand in enumerate(candidates0):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n0))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n0, conv_fft_len0)
        cand_vals.append((val, cand))

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k0 = 12
    top_starts0 = [c for _, c in cand_vals[:top_k0]]

    # deep optimisation for each start (low‑res)
    for start_id, a0 in enumerate(top_starts0):
        if time.time() >= deadline:
            break
        print(f"=== Low‑res Adam + subgradient start {start_id + 1}/{len(top_starts0)} ===")
        best_seq, best_val = _deep_optimize(
            a0,
            n0,
            conv_fft_len0,
            corr_fft_len0,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Phase 2 – medium resolution (n = 4096) if time permits
    n1 = 4096
    conv_fft_len1 = _next_pow2(2 * n1 - 1)
    corr_fft_len1 = _next_pow2(3 * n1 - 2)

    if time.time() < deadline - 15.0:
        if best_seq is not None:
            a1 = _upsample(best_seq, n0, n1)
        else:
            a1 = np.random.dirichlet(np.ones(n1))
        print("--- Medium‑res optimisation (n=4096) ---")
        best_seq, best_val = _deep_optimize(
            a1,
            n1,
            conv_fft_len1,
            corr_fft_len1,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Phase 3 – fine resolution (n = 8192) if still time
    n2 = 8192
    conv_fft_len2 = _next_pow2(2 * n2 - 1)
    corr_fft_len2 = _next_pow2(3 * n2 - 2)

    if time.time() < deadline - 15.0:
        if best_seq is not None:
            # best_seq could be size n0 or n1 – upsample accordingly
            a2 = _upsample(best_seq, best_seq.shape[0], n2)
        else:
            a2 = np.random.dirichlet(np.ones(n2))
        print("--- Fine‑res optimisation (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a2,
            n2,
            conv_fft_len2,
            corr_fft_len2,
            deadline,
            best_seq,
            best_val,
        )
        # keep larger vector only if it actually improved the bound
        if best_seq is not None and best_seq.shape[0] == n2 and best_val < float("inf"):
            pass  # best_seq already refers to the fine‑res solution

    # ------------------------------------------------------------------
    # Fallback to a safe uniform vector
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f}, length = {best_seq.shape[0]} ***")
    return best_seq.tolist()
