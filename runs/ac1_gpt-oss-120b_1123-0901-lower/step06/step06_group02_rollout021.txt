def _full_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 2).bit_length())
    Fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)[:2*n-1]
    # Clip negative due to rounding
    np.maximum(conv, 0, out=conv)
    return conv

def conv_max(a):

"""
Hybrid optimizer for the step‑function bound.

Goal: minimise 2·n·max(conv(a,a))/sum(a)².
Since the objective is scale‑invariant we fix Σa = √(2 n) and
minimise  M = max(conv(a,a)).  The search consists of

1.  Very fast random sampling of a wide family of parametrised shapes
    (exponential, power‑law, combined, Hann / Hamming windows,
    Gaussian, sine‑power and Dirichlet draws) for lengths up to 25 000.
    The best candidate is kept and, when a new record is found,
    a short targeted sub‑gradient step is applied.

2.  A longer projected sub‑gradient descent that always follows the
    current worst convolution index (the exact max‑convolution gradient).

3.  A final simulated‑annealing hill‑climb that moves tiny amounts of
    mass between random positions using a Metropolis acceptance rule.

All helpers are defined at module level (no closures) and the routine
respects the supplied time budget.  The returned object is a plain
Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of a real vector with itself."""
    n = x.shape[0]
    size = 1 << ((2 * n - 2).bit_length())        # next power of two ≥ 2n‑1
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    # Clip possible tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  a is assumed to be already normalised."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (s ≥ 0)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted in descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1              # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (raw, positive, not normalised)
# ----------------------------------------------------------------------
def _shape_exponential(n: int, tau: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.exp(-i / tau)


def _shape_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-p)


def _shape_combined(n: int, p: float, tau: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return (i ** (-p)) * np.exp(-(i - 1) / tau)


def _shape_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def _shape_sine_power(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.sin(np.pi * (i + 0.5) / n) ** power


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the current worst convolution index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a)
        k = int(conv.argmax())            # index of current worst peak

        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        grad[i] = a[j]
        # double contribution when i == j (the diagonal terms)
        if k % 2 == 0:
            ic = k // 2
            if i_start <= ic <= i_end:
                grad[ic] += a[ic]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _conv_max(cand)
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                     # modestly increase step after success
            if it % 200 == 0:

        if best_val < 1.503:
            break
    return best_a, best_val

# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               target_sum: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # transfer at most 8 % of the mass at i
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _conv_max(a_new)
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:

                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

    return best_a, best_val
