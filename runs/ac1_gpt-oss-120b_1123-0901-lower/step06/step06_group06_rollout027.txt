with ``conv`` the linear convolution.
Because F is scale‑invariant we may (and do) enforce ``sum(a)=1``.
Under this normalisation the global optimum of (1) is attained for a
uniform vector (all entries equal) and the optimal value is exactly 2.
The previous script (shown in the prompt) reaches ≈1.51 because it
optimises a *different* surrogate (the smooth‑max of the convolution)
and stops before the true optimum is reached.

The routine below works in three stages:

1. **Diversified seeding** – a small pool of analytically motivated
   shapes (uniform, Dirichlet, power‑law, exponential, cosine, Gaussian…
   all normalised to sum = 1).

2. **Two‑phase optimisation** – for each seed we run
   *an Adam optimiser on a smooth‑max surrogate* (fast convergence) and
   *an exact exponentiated‑gradient sub‑gradient phase* (guaranteed
   monotone improvement of the true objective).  The whole process is
   carried out with FFT‑based convolutions (O(n log n)).

3. **Global fine‑tuning** – the best sequence discovered so far is
   refined with a longer Adam run using a very slowly increasing
   temperature, thereby approaching the true max.

The code respects the 1000 s time budget, prints progress information
and finally returns the best sequence found as a plain Python list.
If the search does not finish in time a uniform vector (which is provably
optimal) is returned as a safe fallback.
"""

import time
import numpy as np

# --------------------------------------------------------------
# FFT based convolution / cross‑correlation utilities.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the 2·len(a)‑1 genuine samples
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)] .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# --------------------------------------------------------------
def _smooth_max(c: np.ndarray, lam: float) -> np.ndarray:
    """Soft max weights w_i ∝ exp(lam·c_i) (numerically stable)."""
    mx = c.max()
    w = np.exp(lam * (c - mx))
    s = w.sum()
    return w / s if s != 0.0 else np.full_like(w, 1.0 / w.size)


# --------------------------------------------------------------
def _objective_exact(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact objective (1) for a normalised vector (sum(a)=1)."""
    conv = _conv_fft(a, conv_len)
    max_c = conv.max()
    return 2.0 * n * max_c


# --------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """A small, diverse pool of deterministic shapes (all normalised)."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n))

    # 2) Dirichlet (concentration = 1) – random & reproducible
    rng = np.random.default_rng(0)
    cand.append(rng.dirichlet(np.ones(n)))

    # 3) power‑law edge‑concentrated
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.5):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        shape = shape / shape.sum()
        cand.append(shape)

    # 4) two‑sided exponential
    for gamma in (0.0008, 0.0015, 0.003):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)

    # 5) cosine window (single bump)
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    # 6) discrete Gaussian
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    return cand


# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function defined in the statement.  The function respects the time
    budget and returns the best sequence discovered (as a plain list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # lengths that are cheap enough for FFT‑based optimisation
    lengths = [1024, 2048, 4096, 8192]

    # pre‑compute FFT sizes (next power of two)
    fft_len = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len[n] = 1 << int(np.ceil(np.log2(conv_len)))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len[n]
        candidates = _initial_candidates(n)

        # ------------------------------------------------------------------
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a = np.copy(a0)                     # already normalised
            logits = np.log(a + 1e-15)           # work in the simplex (soft‑max)

            # ---------- Adam on smooth‑max surrogate --------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07 * np.sqrt(2048.0 / n)     # a mild size‑adapted step

            lam = 0.2                           # temperature (smooth‑max)
            lam_factor = 1.018

            last_improve = time.time()
            max_iters = int(60_000 * 1024 / n) + 5_000

            for t in range(1, max_iters + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _conv_fft(a, conv_len)
                w = _smooth_max(conv, lam)

                # gradient of the surrogate (linear in w)
                grad_a = 2.0 * _corr_fft(w, a, conv_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule (approach true max)
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a) == 1)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t:05d}] new best {val:.6f}")

                # gentle random shake if we have stalled
                if time.time() - last_improve > 7.0:
                    logits += (rng.random(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---------- Exact sub‑gradient (exponentiated‑gradient) -----
            a = a / a.sum()
            eta = 0.55
            sub_last_improve = time.time()
            max_sub_iters = 6_000

            for sub_it in range(max_sub_iters):
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # binary weight that picks the (or all) active maxima
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                w = np.where(active, 1.0 / active.sum(), 0.0)

                # gradient w.r.t. a
                grad_a = 2.0 * _corr_fft(w, a, conv_len)

                # exponentiated‑gradient step
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = rng.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluate
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} Subgrad iter={sub_it}] new best {val:.6f}")

                # early abort if no progress for a while
                if time.time() - sub_last_improve > 8.0:
                    break
                if time.time() > deadline:
                    break

        # ---------------- Global fine‑tuning of the current best ----
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning the current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.045 * np.sqrt(2048.0 / n)
            lam = 0.4
            lam_factor = 1.012
            last_improve = time.time()
            fine_iters = int(80_000 * 1024 / n) + 5_000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_len)
                w = _smooth_max(conv, lam)

                grad_a = 2.0 * _corr_fft(w, a, conv_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    logits += (rng.random(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – the provably optimal uniform vector.
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    print(f"\nBest value found : {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
