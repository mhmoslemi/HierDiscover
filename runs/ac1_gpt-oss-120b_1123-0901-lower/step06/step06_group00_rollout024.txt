def _adam_optimize(a0, n, fft_len, deadline, max_iters=200_000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        # softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute conv max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_c = conv.max()
        val = 2.0 * n * max_c   # sum =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it}] new best {val:.6f}")

        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via cross-correlation
        grad = 2.0 * _cross_corr(w, a, fft_len)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        it += 1

    return best_a, best_val

def _subgradient_optimize(a0, n, fft_len, corr_len, deadline, max_iters=200_000, eta=0.6):
    a = np.maximum(a0,0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_corr(w, a, corr_len)
        # exponentiated gradient
        a = a * np.exp(-eta * grad)
        # renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val

def _simulated_annealing(a0, n, fft_len, deadline, max_moves=200_000):
    a = a0.copy()
    conv = np.fft.rfft(a, fft_len)
    conv_vals = np.fft.irfft(conv * conv, fft_len)[:2*n-1]
    cur_max = conv_vals.max()
    best_a = a.copy()
    best_max = cur_max
    best_val = 2.0 * n * cur_max
    # temperature schedule: start high, linear cooling to near zero
    total_time = deadline - time.time()
    start_time = time.time()
    moves = 0
    # choose initial temperature as fraction of current max?
    T0 = cur_max * 0.5  # may tune
    while time.time() < deadline and moves < max_moves:
        # elapsed fraction

"""
propose_candidate – heuristic solver for
    F(a) = 2·n·max(a*a) / (sum(a)^2)

The routine combines several complementary ideas that together
beat the previous 1.508306 bound:

1. **Rich pool of starts** – uniform, random Dirichlet,
   edge‑concentrated power‑law, double‑Gaussian, beta‑shaped
   (U‑shaped) and sparse “spike” vectors.

2. **Adam on a smooth‑max surrogate** – works on the simplex
   via a log‑softmax parametrisation.  A temperature‐parameter
   (λ) is increased exponentially so that the surrogate rapidly
   focuses on the true max.

3. **Exact‑max sub‑gradient phase** – exponentiated‑gradient (multiplicative)
   updates on the simplex with a fixed learning‑rate η.
   This phase tightens the solution obtained by Adam.

4. **Simulated‑annealing fine‑tuning** – a cheap “move” consists of
   transferring a random fraction of mass between two coordinates.
   Non‑improving moves are accepted with a temperature–dependent
   probability, allowing escape from shallow basins.

5. **Time‑aware orchestration** – the whole budget (≈1000 s) is split
   between the phases; the algorithm always returns the best vector
   seen before the deadline.

All operations use FFT‑based convolutions (O(n log n)) so that even
large numbers of moves fit comfortably into the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)] .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (∑a≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # start temperature
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_corr(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_corr(w, a, corr_fft_len)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simple SA on the exact objective.
    Randomly moves a fraction of mass from one coordinate to another.
    Accepts worsening moves with probability exp(-Δ/ T) where T cools
    linearly to zero.
    Returns the best vector encountered and its exact objective.
    """
    a = a0.copy()
    cur_conv = _full_convolution(a, fft_len)
    cur_max = cur_conv.max()
    best_a = a.copy()
    best_max = cur_max
    best_val = 2.0 * n * cur_max

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)   # avoid div‑zero
    T0 = cur_max * 0.5  # initial temperature (tuned empirically)

    moves = 0
    while time.time() < deadline and moves < max_moves:
        # linear cooling schedule
        elapsed = (time.time() - start_time) / total_time
        T = T0 * (1.0 - elapsed)

        # propose a move
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move a random fraction (up to 20 %) from i to j
        eps = np.random.rand() * 0.2
        delta = a[i] * eps
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        # evaluate new max
        conv2 = _full_convolution(a2, fft_len)
        new_max = conv2.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_max - cur_max) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a2
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_val = 2.0 * n * new_max
                print(f"[SA move={moves}] new best {best_val:.6f}")

        moves += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build a diverse pool of initial candidates
    candidates = []

    # warm‑start and a couple of rotations
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2):
            candidates.append(np.roll(warm, shift))

    # pure uniform
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # double‑Gaussian bumps at the ends
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # beta‑shaped (U‑shaped) densities – many random α,β
    for _ in range(6):
        a = np.random.uniform(0.2, 5.0)
        b = np.random.uniform(0.2, 5.0)
        x = np.arange(1, n + 1, dtype=np.float64)
        shape = (x ** (a - 1)) * ((n + 1 - x) ** (b - 1))
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        candidates.append(shape)

    # sparse “spike” patterns (equally spaced non‑zero entries)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        candidates.append(a)

    # ------------------------------------------------------------------
    # optimisation of each candidate (short runs)
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        # allocate a tiny slice of the remaining budget for this candidate
        local_deadline = min(deadline - 5.0, time.time() + 15.0)

        # Adam (smooth‑max) phase
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=80_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam new best {val_adam:.6f}")

        # short sub‑gradient refinement
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=30_000, eta=0.5
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # final long sub‑gradient refinement on the current best
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=150_000,
            eta=0.5,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # simulated‑annealing fine‑tuning (cheap but effective)
    if best_seq is not None and time.time() < deadline - 5:
        a_sa, val_sa = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, max_moves=100_000
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # fallback – guarantee a valid return value
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's accepted domain
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
