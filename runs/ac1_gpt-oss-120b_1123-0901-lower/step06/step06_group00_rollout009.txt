def _simplex_projection(v):
    # Project v onto {x>=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm is a lightweight, time‑bounded search consisting of four phases:

1. **Broad parametric search** – generate many diverse candidates
   (beta‑like, exponential‑edge, power‑law edge, centred Gaussian,
   powered sinusoid, uniform, spike patterns and a warm‑start if available)
   and keep the one with the smallest exact objective.

2. **Projected sub‑gradient descent** – starting from the best candidate,
   minimise the exact max‑convolution objective on the simplex.
   The sub‑gradient is obtained from the active set of maximal
   convolution entries, a diminishing step size `1/√t` is used and each
   iterate is projected onto the probability simplex.

3. **Optional up‑scaling** – if sufficient time remains, the current
   solution is linearly interpolated to a finer grid (twice as many
   points) and the sub‑gradient refinement is repeated.

4. **Cheap random local search** – final hill‑climbing moves that shift
   a random fraction of mass from one coordinate to another, keeping the
   move if the objective improves.

The routine respects the supplied time budget (default 1000 s) and
always returns a valid sequence.  All heavy lifting (convolution,
cross‑correlation, simplex projection) is performed with NumPy/FFT for
`O(n log n)` performance.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """
    Exact objective 2·n·max(a∗a) (the denominator is 1 because a is
    normalised to sum=1).
    """
    conv = _full_convolution(a, conv_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _search_initial_candidates(n: int, conv_len: int, deadline: float):
    """
    Generate a large pool of low‑dimensional parametric candidates,
    evaluate them exactly and keep the best.
    """
    best_seq = None
    best_val = float("inf")

    def consider(vec: np.ndarray):
        nonlocal best_seq, best_val
        if vec.ndim != 1:
            return
        s = vec.sum()
        if s < 1e-12:
            return
        vec = vec / s
        vec = np.maximum(vec, 0.0)
        val = _objective(vec, n, conv_len)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
            print(f"[initial] new best {val:.6f}")

    # Warm start (if supplied)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.size > n:
            w = w[:n]
        elif w.size < n:
            w = np.pad(w, (0, n - w.size))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            consider(w)
            for sh in (n // 4, n // 2, 3 * n // 4):
                consider(np.roll(w, sh))

    # Uniform
    consider(np.full(n, 1.0 / n))

    # Random Dirichlet (some entropy variety)
    for _ in range(30):
        if time.time() >= deadline:
            break
        consider(np.random.dirichlet(np.ones(n)))

    # Spike patterns (equally spaced k‑spikes)
    for k in (32, 64, 128, 256, 512):
        if time.time() >= deadline:
            break
        a = np.zeros(n, dtype=np.float64)
        positions = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[positions] = 1.0 / k
        consider(a)

    # ---- Beta‑like shapes -------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        alpha = np.random.uniform(-4.0, 2.0)
        beta = np.random.uniform(-4.0, 2.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        rev = n - i + 1.0
        shape = (i ** alpha) * (rev ** beta)
        if np.isfinite(shape).all() and shape.sum() > 0:
            consider(shape)

    # ---- Exponential edge shapes -----------------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        lam = np.random.uniform(0.1, 5.0)
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        consider(shape)

    # ---- Power‑law edge shapes --------------------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.5, 5.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        rev = n - i + 1.0
        shape = (i ** -p) + (rev ** -p)
        consider(shape)

    # ---- Centre Gaussian shapes -------------------------------------------
    for _ in range(150):
        if time.time() >= deadline:
            break
        sigma = np.random.uniform(0.05, 0.4) * n
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        shape = np.exp(-((i - centre) ** 2) / (2 * sigma ** 2))
        consider(shape)

    # ---- Powered sinusoid (positive half‑wave) ----------------------------
    for _ in range(150):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.5, 6.0)
        i = np.arange(n, dtype=np.float64)
        s = np.sin(np.pi * (i + 0.5) / n)
        s = np.maximum(s, 0.0) ** p
        consider(s)

    # Fallback uniform if nothing survived
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_len)

    return best_seq, best_val


# ----------------------------------------------------------------------
def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
):
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[PGD iter={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(w, a, corr_len)

        # diminishing step size
        step = 1.0 / np.sqrt(it + 1)

        # gradient step + simplex projection
        a = _simplex_projection(a - step * grad)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 30_000,
):
    """
    Light‑weight hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if it reduces the exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_len)
    moves = 0
    while time.time() < deadline and moves < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)  # up to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = _objective(a2, n, conv_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            print(f"[local] improved to {val2:.6f}")
        moves += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 0: basic parameters
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # Phase 1 – broad parametric search
    best_seq, best_val = _search_initial_candidates(n, conv_len, deadline)
    print(f"After initial search best = {best_val:.6f}")

    # Phase 2 – projected sub‑gradient refinement on the current best
    if time.time() < deadline - 5:
        best_seq, val = _pgd_optimize(
            best_seq, n, conv_len, corr_len, deadline, max_iters=500_000
        )
        if val < best_val:
            best_val = val
            print(f"After PGD refinement best = {best_val:.6f}")

    # Phase 3 – optional up‑scaling to a finer grid
    if time.time() + 30.0 < deadline:
        n_big = 8192
        conv_len_big = _next_pow2(2 * n_big - 1)
        corr_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("Upscaling to larger grid and running PGD")
        a_big, val_big = _pgd_optimize(
            a_big, n_big, conv_len_big, corr_len_big, deadline, max_iters=300_000
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big
            n = n_big
            conv_len = conv_len_big
            corr_len = corr_len_big
            print(f"Upscaled solution improved best = {best_val:.6f}")

    # Phase 4 – cheap random local search (final polishing)
    if time.time() < deadline - 5:
        best_seq, val = _local_random_search(best_seq, n, conv_len, deadline)
        if val < best_val:
            best_val = val
            print(f"After local search best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks – ensure the returned sequence is valid for
    # `evaluate_sequence`.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s < 0.01:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / s

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
