        w = np.exp((conv - cur_max) / tau)

"""
Hybrid random‑search + gradient‑based optimizer for the step‑function bound.

* Phase 0 – optional warm‑start from a global construction (height_sequence_1).
* Phase 1 – massive random sampling of parametric families (exponential,
  power‑law, Gaussian, Hann/Hamming/Blackman windows, Dirichlet draws etc.).
  Lengths from 1500 … 12000 are explored; every candidate is projected onto the
  simplex Σaᵢ = √(2 n) and evaluated with the exact max‑convolution.
* Phase 2 – Adam optimisation on a log‑sum‑exp smoothed version of the
  max‑convolution (soft‑max parametrisation a = S·softmax(x)).  The best
  candidate from Phase 1 is refined repeatedly until the time budget runs
  low.
* Phase 3 – deterministic projected sub‑gradient descent that follows the
  exact worst convolution index (the same method as in the reference code).
* Phase 4 – Simulated‑annealing hill‑climb that moves tiny amounts of mass
  between random positions with a Metropolis acceptance criterion.

All phases respect the overall 1000 s budget and stop early as soon as a
solution with evaluation score ≤ 1.503 is found.  The routine returns the
best sequence (as a plain Python list) found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate negligible negatives from numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a * a)[k] for the full linear convolution."""
    return _full_convolution_fft(a, a).max()


def _objective(a: np.ndarray) -> float:
    """Exact evaluation function (identical to evaluate_sequence)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp (soft‑max) surrogate of max‑conv
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth max‑convolution surrogate.

    Returns the best coefficient vector (on the simplex) and its exact
    max‑convolution value (not the surrogate).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- map logits to the simplex (a = target_sum * softmax(x)) ----
        mx = np.max(x)
        exp_shift = np.exp(x - mx)          # stability
        p = exp_shift / np.sum(exp_shift)   # Σp = 1
        a = target_sum * p

        # ---- exact max of the convolution (for logging & return) ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ---- smoothed max (log‑sum‑exp) weights ----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient wrt a: 2·a ⊛ rev(w) ----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ---- back‑propagation to logits ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    # final exact evaluation of the best vector
    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the current worst convolution index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution(a)
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # worst index
        # gradient of conv_k with respect to a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _max_convolution(cand)
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # increase step after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70                       # shrink on failure
            if np.random.rand() < 0.07:        # occasional jitter
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution(a)

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # move at most 8 % of the mass at i
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # re‑normalise every few thousand steps to fight drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Random parametric initialisation helper
# ----------------------------------------------------------------------
def _random_initial_sequence(shape: str, n: int) -> np.ndarray:
    """Generate a non‑negative vector of length n according to *shape*."""
    if shape == "exp_decay":
        alpha = np.random.uniform(0.5, 5.0)
        tau = alpha * np.sqrt(n)
        idx = np.arange(n, dtype=float)
        a = np.exp(-idx / tau)
    elif shape == "exp_rise":
        alpha = np.random.uniform(0.5, 5.0)
        tau = alpha * np.sqrt(n)
        idx = np.arange(n, dtype=float)
        a = np.exp(-(n - 1 - idx) / tau)
    elif shape == "power":
        p = np.random.uniform(0.2, 2.0)
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** (-p)
    elif shape == "linear":
        a = np.linspace(1.0, 0.0, n)
    elif shape == "gaussian":
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(0.1 * n, 0.5 * n)
        idx = np.arange(n, dtype=float)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    elif shape == "random_dirichlet":
        a = np.random.dirichlet(np.ones(n))
    elif shape == "random_noise":
        a = np.abs(np.random.randn(n))
    elif shape == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "hamming":
        i = np.arange(n, dtype=float)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "blackman":
        i = np.arange(n, dtype=float)
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    else:                     # fallback – uniform
        a = np.ones(n)

    # optional tiny Gaussian noise (sometimes helps escaping plateaus)
    if np.random.rand() < 0.2:
        noise_level = np.random.rand() * 0.05
        a = np.maximum(a + noise_level * np.max(a) * np.random.randn(n), 0.0)

    return a

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimiser for the step‑function bound.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf

    def _record(seq: np.ndarray):
        """Evaluate *seq*, keep it if it improves the global best."""
        nonlocal best_seq, best_score
        score = _objective(seq)
        if score < best_score:
            best_score = float(score)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best score={best_score:.6f}  n={len(seq)}")
        return score

    # ------------------------------------------------------------------
    # Warm‑start from a global construction if available
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            if hs_arr.size > 0:
                n0 = hs_arr.shape[0]
                target0 = np.sqrt(2.0 * n0)
                hs_norm = hs_arr * (target0 / hs_arr.sum())
                _record(hs_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – massive parametric random initialisations
    # ------------------------------------------------------------------
    shape_options = [
        "exp_decay", "exp_rise", "power", "linear", "gaussian",
        "random_dirichlet", "random_noise", "hann", "hamming", "blackman"
    ]

    # keep a generous chunk for later refinements
    while time.time() < deadline - 300.0:
        # pick a length – longer sequences become more valuable later
        remaining = deadline - time.time()
        if remaining > 600:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 4000 + 1)

        target_sum = np.sqrt(2.0 * n)

        shape = np.random.choice(shape_options)
        a0 = _random_initial_sequence(shape, n)

        # normalise to the required L1 sum (scale invariance)
        if a0.sum() == 0:
            continue
        a0 = a0 * (target_sum / a0.sum())
        _record(a0)

    # ------------------------------------------------------------------
    # Helper to compute remaining time
    # ------------------------------------------------------------------
    def _time_left() -> float:
        return deadline - time.time()

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on smoothed max‑convolution
    # ------------------------------------------------------------------
    while best_seq is not None and _time_left() > 150.0:
        if best_score <= 1.503:
            break
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        x0 = np.log(best_seq + 1e-12)

        # give each Adam run at most 30 s (or whatever fits)
        run_dead = time.time() + min(30.0, _time_left() - 5.0)
        seq_opt, val_opt = _optimise_adam_softmax(
            n, x0, target_sum, run_dead,
            init_tau_factor=0.12,
            min_tau_factor=5e-5,
            alpha0=0.025,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and _time_left() > 80.0 and best_score > 1.503:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_dead = time.time() + min(40.0, _time_left() - 5.0)
        seq_sub, val_sub = _targeted_subgradient(best_seq.copy(),
                                                 target_sum,
                                                 sub_dead)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and _time_left() > 15.0 and best_score > 1.503:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_dead = time.time() + min(15.0, _time_left() - 0.1)
        seq_hill, val_hill = _hillclimb(best_seq.copy(),
                                        target_sum,
                                        hill_dead)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------

    if best_seq is None:

        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    return best_seq.tolist()
