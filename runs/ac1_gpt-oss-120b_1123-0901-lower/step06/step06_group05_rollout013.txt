"""
Propose a candidate step‑function sequence that minimises

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm follows a multiscale optimisation:

1.  Choose several lengths (2048 … 32768).  A good solution for a
    smaller length is up‑sampled and used as a warm‑start for the next
    length.
2.  Build a heterogeneous pool of initial shapes:
    • up‑sampled previous best,
    • historic solution (global ``height_sequence_1``),
    • random Dirichlet draws with various concentrations,
    • power‑law, exponential, Hann, sine, Gaussian families,
    • **Beta‑distribution** shapes (new).
3.  Optimise each candidate with an Adam optimiser on a *smooth* Lp‑norm
    surrogate of the max‑convolution.  The exponent *p* is increased
    gradually (4 → 8 → 16 → 32 → 64 → 128) so the landscape becomes
    sharper only after a good region has been found.
4.  Refine the Adam result with a short exponentiated‑gradient sub‑gradient
    phase on the exact maximum.
5.  If the current best has the same length, perform a brief fine‑tune
    (Adam with the same schedule).
6.  After all lengths have been explored, run a cheap stochastic hill‑climb
    that moves a small mass ε between two indices while keeping the
    sequence on the simplex.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.  All entries are
non‑negative; the routine works only with NumPy and standard library
functions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers ----------------------------------------------------
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (fast integer version)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s

# Candidate generation ---------------------------------------------------

def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 9. **Beta‑distribution** shapes – new family
    #    shape ∝ x^{α‑1} (1‑x)^{β‑1}
    x = np.linspace(0, 1, n)
    beta_params = [
        (0.5, 0.5), (0.8, 0.8), (2.0, 2.0),
        (3.0, 3.0), (0.3, 2.0), (2.0, 0.3)
    ]
    for a, b in beta_params:
        # avoid NaNs at the borders for α<1 or β<1 by adding a tiny epsilon
        eps = 1e-12
        log_shape = (a - 1) * np.log(x + eps) + (b - 1) * np.log(1 - x + eps)
        shape = np.exp(log_shape)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam_pnorm(a0: np.ndarray, n: int, deadline: float,
                    schedule: list[tuple[int, int]]) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the Lp‑norm surrogate of the objective.
    The schedule is a list of (p, iters) tuples, e.g. [(4,2000), (8,2000), …].
    Returns the best sequence found and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Initialise logits for a soft‑max (ensures we stay on the simplex)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                     # base learning rate
    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()
    t = 0                         # global iteration counter for bias correction

    for p, iters in schedule:
        for _ in range(iters):
            t += 1

            # map logits → point on the simplex (soft‑max)
            mx = logits.max()
            exp_x = np.exp(logits - mx)
            a = exp_x / exp_x.sum()

            # exact objective (sum(a)=1 ⇒ denominator drops out)
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c
            if val < best_val:
                best_val = val
                best_a = a.copy()
                last_improve = time.time()
                print(f"[n={n} p={p} iter={t}] new best {val:.6f}")

            # ---- smooth p‑norm and its gradient ----
            #  Lp = (∑ conv_i^p)^{1/p}
            #  ∇ Lp = (1/Lp^{p‑1}) * ∑ conv_i^{p‑1} ∇conv_i
            #  ∇conv_i = 2 * (a ★ a)[i]  (computed via cross‑correlation)
            conv_pow_p = np.power(conv, p, where=conv > 0, out=np.zeros_like(conv))
            sum_pow = conv_pow_p.sum()
            if sum_pow <= 0:
                # fallback: act like a pure max‑gradient
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            else:
                Lp = sum_pow ** (1.0 / p)
                w = np.power(conv, p - 1, where=conv > 0, out=np.zeros_like(conv))
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len) / (Lp ** (p - 1))

            # back‑prop through the simplex mapping (soft‑max on logits)
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional jitter if we get stuck
            if time.time() - last_improve > 8.0:
                logits += (np.random.rand(n) - 0.5) * 1e-2
                last_improve = time.time()

            if time.time() > deadline:
                return best_a, best_val

    return best_a, best_val


def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                      # initial step size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where the convolution actually attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                        # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 150_000):
    """
    Very cheap stochastic local search.
    Repeatedly moves a small mass ε from one index to another and keeps the
    move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 8_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    the evaluation function.  The routine respects the supplied time budget
    (default 1000 s) and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths – larger ones give more expressive power but cost more time
    lengths = [2048, 4096, 8192, 16384, 32768]

    # ------------------- Main optimisation loop -----------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Build a modest but diverse set of starting points
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budgets – roughly scale with inverse of length
        base_adam = int(2500 * 2048 / n) + 5_000   # a healthier budget
        max_subgrad = int(5_000 * 2048 / n) + 2_000

        # Adam schedule (p, iters) – increase p gradually
        schedule = [
            (4,  base_adam // 4),
            (8,  base_adam // 4),
            (16, base_adam // 4),
            (32, base_adam // 4),
            (64, base_adam // 4),
            (128, base_adam // 4),
        ]

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam on Lp‑norm surrogate ----
            a_adam, val_adam = _run_adam_pnorm(init, n, deadline, schedule)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # ---- Sub‑gradient (exponentiated‑gradient) ----
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

            # ---- Short fine‑tune on the *current* global best of this length ----
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _run_adam_pnorm(best_seq, n, deadline,
                                                   [(p, max(1, it // 2)) for p, it in schedule])
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
```

This implementation enriches the candidate pool with Beta‑distribution shapes, uses a multi‑stage Lp‑norm surrogate with an Adam optimiser (gradually sharpening the surrogate), adds longer runs for larger problem sizes, and finishes with a more robust hill‑climb.  
All operations respect the 1000 s budget and output a non‑negative list suitable for `evaluate_sequence`.
