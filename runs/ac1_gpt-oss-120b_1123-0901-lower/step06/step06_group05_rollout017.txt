"""
Multi‑scale non‑negative step‑function optimizer.

The algorithm builds a hierarchy of problem sizes (256 → … → 16384).  
For each size it constructs a diverse pool of initial shapes:
    * the current best sequence up‑sampled,
    * historic global variable `height_sequence_1`,
    * Dirichlet random draws,
    * symmetric windows (Hann, sine, Gaussian, power‑law),
    * *asymmetric* families (geometric, monotone power‑law, Beta‑type),
    * a few hand‑crafted ramps.
Each candidate is refined by:
    1. Adam optimiser on a smooth‑max (soft‑max) surrogate of the max‑convolution.
    2. Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    3. A short fine‑tune run on the globally best candidate for the current length.
After the multi‑scale phase a cheap stochastic hill‑climb (mass transfers) is
performed to squeeze out any remaining improvement.

All operations stay on the simplex (the sequence is normalised to sum to 1),
so the evaluation metric simplifies to `2·n·max(conv)`.  The routine respects
the supplied time budget and returns the best sequence found as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (fast integer version)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


# ----------------------------------------------------------------------
def _generate_initial_candidates_v2(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. Random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. Symmetric window families (already proved useful)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    i = np.arange(n, dtype=np.float64)
    # symmetric exponentials (already in the previous code)
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # Half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # Gaussian‑like bumps
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # 5. Asymmetric families – monotone power‑law (increasing & decreasing)
    for power in (0.5, 1.0, 1.5, 2.0):
        inc = np.arange(1, n + 1, dtype=np.float64) ** power
        inc /= inc.sum()
        dec = inc[::-1]                     # decreasing version
        candidates.append(inc)
        candidates.append(dec)

    # 6. Geometric (exponential) shapes
    for base in (0.999, 0.995, 0.99, 0.97, 0.95):
        geo = base ** np.arange(n, dtype=np.float64)
        geo /= geo.sum()
        candidates.append(geo)
        candidates.append(geo[::-1])

    # 7. Beta‑type shapes (x**(a-1)*(1-x)**(b-1))
    for a, b in [(0.5, 5), (5, 0.5), (2, 2), (0.2, 0.2), (5, 5)]:
        x = np.linspace(0, 1, n)
        shape = np.where((x > 0) & (x < 1),
                         (x ** (a - 1)) * ((1 - x) ** (b - 1)),
                         0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean


# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best sequence seen during the run and its exact value
    (with sum(a)=1 the exact value is 2·n·max(conv)).
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # initialise logits = log(a) (soft‑max parametrisation)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07               # slightly larger to move fast early on
    lam = 0.30               # temperature for smooth‑max
    lam_factor = 1.025       # slowly increase temperature

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ---- map logits → probability simplex (soft‑max) ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- exact objective (sum(a)=1 ⇒ denominator drops out) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # ---- smooth‑max weights (soft‑max on the convolution vector) ----
        w = np.exp(lam * (conv - max_c))          # shift by max for numerical stability
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e4)

        # ---- occasional random shake if we are stuck ----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the exact max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                      # initial step size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where convolution actually attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                        # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if we stop improving for a while
        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 250_000):
    """
    Very cheap stochastic local search.
    Moves a small amount of probability mass ε from one index to another
    and keeps the move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        # take a fraction (up to 5 %) of the donor mass
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0:
            trials += 1
            continue
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The algorithm proceeds in several scales (256 … 16384), continuously
    up‑sampling the best known sequence and refining it with Adam,
    exponentiated‑gradient sub‑gradients and a final hill‑climb.
    It returns the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # Multi‑scale hierarchy (feel free to adjust if time is tight)
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Build a pool of start points (diverse + warm‑start)
        candidates = _generate_initial_candidates_v2(n, best_seq)

        # iteration budgets – larger n ⇒ fewer iterations (but still enough)
        max_adam = max(3_000, int(12_000 * 256 / n) + 5_000)
        max_subgrad = max(2_000, int(5_000 * 256 / n) + 2_000)

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam (smooth‑max) ----
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam new global best {val_adam:.6f}")

            # ---- Sub‑gradient (exponentiated‑gradient) ----
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad new global best {val_sub:.6f}")

        # ------------------------------------------------------------------
        # Fine‑tune the best sequence found for the current length
        if best_seq is not None and len(best_seq) == n:
            a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                         int(max_adam * 0.4))
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 256
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
