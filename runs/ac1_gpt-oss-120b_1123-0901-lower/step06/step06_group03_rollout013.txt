def _lp_optimize(a, M, deadline):
    n = a.shape[0]
    # Build convolution matrix
    rows = 2*n - 1
    A = np.zeros((rows, n))
    for i in range(n):
        A[i:i+n, i] = a[i]
    # define variable
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(rows)]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    # solve
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    g_val = g.value
    if g_val is None:
        return None
    # clip negative to zero
    g_val = np.maximum(g_val, 0.0)
    # if sum is zero fallback
    if g_val.sum() < 1e-12:
        return None
    return g_val

def _deep_optimize_small(a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val):
    # Parameter settings similar to earlier but with maybe less iterations
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5
    lam_factor = 1.02

    it = 0
    last_improve = time.time()
    # Adam phase
    while time.time() < deadline:
        # softmax mapping to simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # compute objective
        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it}] new best {val:.6f}")

        # smooth-max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t a using cross-correlation
        grad_a = 2.0 * _corr_fft(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it >= 300_000:
            break

    # Subgradient phase
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val

"""
Propose a non‑negative step function that (approximately) minimises

        F(a) = 2·n·max(a∗a) / (∑a)² .

The search works in three phases:

1. **Coarse search on a small grid (n≈256).**  
   A diversified pool of candidate shapes (uniform, power‑law,
   double‑exponential, Gaussian, sinusoid, random Dirichlet) is built,
   normalised to unit sum and screened with a fast FFT‑based objective.
   The best eight candidates are kept.

2. **Refinement of each candidate.**  
   – *Linear‑program improvement*: for a candidate a we compute
     M = max(a∗a) and solve the LP  
       maximise ∑g subject to a∗g ≤ M , g ≥ 0 .
     The LP pushes mass to positions that do not increase the worst‑case
     convolution, thereby increasing the effective sum for the same
     bound.  The LP solution is mixed back into the candidate (soft
     update).  
   – *Smooth‑max Adam + exact sub‑gradient*: a mirror‑descent style
     optimiser (the same scheme used in the reference code) further
     reduces the max‑convolution while keeping the vector on the simplex.

3. **Upsampling & final polishing.**  
   The best small‑grid vector is interpolated to a larger support
   (n≈4096) and the Adam + sub‑gradient optimiser is run again.  The final
   vector is clipped to the evaluator’s admissible range and returned.

All steps respect the supplied time budget (default 1000 s) by checking
the wall‑clock time after each major operation.  The algorithm requires
only NumPy and CVXPY (ECOS solver) and runs on a single CPU core.
"""

import time
import numpy as np

# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (the sum of a is assumed >0)."""
    conv = _conv_fft(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s ** 2))


# ------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors for a given support size."""
    pool = []

    # optional warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)          # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # random Dirichlet draws with a range of concentration parameters
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


# ------------------------------------------------------------
def _lp_improve(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise ∑g   s.t.   a∗g ≤ M ,   g ≥ 0 .
    Returns the optimal g (not normalised) or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("[LP] cvxpy not available:", e)
        return None

    n = a.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A where (A @ g)[k] = (a∗g)[k]
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(rows)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is fast for this size; increase max_iters if needed
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        print("[LP] problem status:", prob.status)
        return None

    g_val = g.value
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)

def _deep_optimize_small(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by an exponentiated‑gradient
    sub‑gradient phase (identical to the reference implementation, but with
    tighter iteration limits suitable for the available time budget).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # start temperature for the smooth‑max
    lam_factor = 1.02

    it = 0
    last_improve = time.time()

    # ---------------------------- Adam phase ----------------------------
    while time.time() < deadline:
        # Map logits to the probability simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (useful for logging)
        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _corr_fft(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Raise the temperature
        lam = min(lam * lam_factor, 1e6)

        # Small random shake if we are stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it >= 300_000:          # safety bound
            break

    # ------------------------ Sub‑gradient phase ----------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # Active‑set mask (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_fft_len)

        # Decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:   # stop if no recent improvement
            break

    return global_best_seq, global_best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that minimises the supplied evaluation metric.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # spare 0.5 s for cleanup

    # ------------------------ small support -------------------------
    n_small = 256
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # optional warm‑start from a previously discovered construction
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n_small:
            w = w[:n_small]
        else:
            w = np.pad(w, (0, n_small - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # Build a diverse pool of start vectors
    candidates = _generate_candidate_pool(n_small, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored = []  # holds (objective, normalised vector)

    # ------------------- quick screening of the pool ----------------
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n_small))
            s = a.sum()
        a = a / s
        val = _objective(a, n_small, conv_fft_len_small)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() >= deadline:
            break
        # print progress (optional)
        # print(f"[screen {idx:03d}] obj={val:.6f}")

    # keep the most promising starts (up to eight)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]

    # ------------------- refinement of each top candidate ------------
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] refining candidate")
        a = a0.copy()

        # ---- Linear‑program improvement (few iterations) ----
        for lp_iter in range(2):
            if time.time() >= deadline:
                break
            conv = _conv_fft(a, conv_fft_len_small)
            M = conv.max()
            g = _lp_improve(a, M)
            if g is None:
                continue
            g_norm = g / g.sum()
            # soft update toward the LP solution
            t = 0.3
            a = (1.0 - t) * a + t * g_norm
            a = np.maximum(a, 0.0)
            a /= a.sum()
            val_lp = _objective(a, n_small, conv_fft_len_small)
            if val_lp < best_val:
                best_val = val_lp
                best_seq = a.copy()
                print(f"[LP {cid}/{lp_iter}] new best {val_lp:.6f}")

        # ---- Adam + sub‑gradient refinement ----
        a_opt, val_opt = _deep_optimize_small(
            a, n_small, conv_fft_len_small, corr_fft_len_small,
            deadline, best_seq, best_val
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[optim end {cid}] best improved to {val_opt:.6f}")

    # ------------------- optional up‑sampling -----------------------
    if time.time() + 5 < deadline:
        n_big = 4096
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_small = np.linspace(0, 1, n_small, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_small, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[big] start optimisation on enlarged support")
        a_big_opt, val_big = _deep_optimize_small(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_opt.copy()
            print("[big] new best after up‑sampling", best_val)

    # ----------------------- finalisation ---------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print("Final best objective:", best_val, "length:", best_seq.shape[0])
    return best_seq.tolist()
