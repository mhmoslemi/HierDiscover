def _run_proj_gd(a0, n, deadline, max_iters):
    # a0 must sum to 1
    a = np.copy(a0)
    best_a = a.copy()
    best_val = np.inf
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)
    for t in range(1, max_iters+1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute subgradient:
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        eta = 1.0 / np.sqrt(t)  # or maybe 0.5?

        a_new = a - eta * grad
        # project onto simplex
        a = _simplex_projection(a_new)

        if time.time() > deadline:
            break

    return best_a, best_val

def _simplex_projection(v):
    # Project onto simplex {x>=0, sum=1}
    # Implementation from Duchi: sort descending.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that minimizes

    F(a) = 2·n·max(a * a) / (sum(a)^2)

with n = len(a).  The algorithm is a diversified, multi‑scale optimizer:

1.  Several lengths are tried (2048 … 65536).  Larger n gives more freedom
    but each iteration is more expensive, so the iteration budget is scaled
    with 1/n.
2.  For each length a pool of heterogeneous initial candidates is built:
    - up‑sampled best‑so‑far,
    - warm‑start from a historic solution (global ``height_sequence_1``),
    - random Dirichlet draws with different concentrations,
    - analytic shapes (power‑law, exponentials, Hann, sine, Gaussians),
    - random mixtures of a few Gaussian bumps.
3.  Every candidate is refined by three stages:
    a) Projected‑gradient descent on the *exact* max‑convolution (simplex
       projection after each step).  This gives a fast, guaranteed‑feasible
       descent direction.
    b) Adam optimisation on a smooth‑max surrogate (logits → soft‑max)
       – useful to escape shallow basins.
    c) A short exponentiated‑gradient sub‑gradient phase (as in the
       reference implementation) for a final exact‑max fine‑tune.
4.  The globally best sequence (over all lengths) is further polished by a
    cheap stochastic hill‑climb that moves a tiny amount of mass between
    two positions whenever it improves the objective.
5.  All steps respect the supplied time budget (default 1000 s) and any
    sequence returned is a plain Python list of non‑negative numbers.

The implementation uses only NumPy and the standard library and runs on a
single process (up to 2 CPU threads may be used internally by NumPy’s FFT).
"""

import time
import numpy as np

# Utility functions

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the max‑convolution.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   (n = len(a))
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to sum exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w

def _random_multi_gaussian(n: int, k: int = 3) -> np.ndarray:
    """Generate a mixture of k Gaussian bumps with random centres and widths."""
    positions = np.random.choice(n, k, replace=False)
    sigmas = np.random.uniform(n / 20.0, n / 5.0, size=k)
    x = np.arange(n, dtype=np.float64)
    shape = np.zeros(n, dtype=np.float64)
    for pos, sigma in zip(positions, sigmas):
        shape += np.exp(-0.5 * ((x - pos) / sigma) ** 2)
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape /= total
    return shape


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a diverse pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet draws with various concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 3.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. power‑law shapes (edge‑concentrated)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.4, 5):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 9. random mixture of a few Gaussians
    for _ in range(3):
        candidates.append(_random_multi_gaussian(n, k=np.random.randint(2, 5)))

    # Normalise and discard pathological entries
    cleaned = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        cleaned.append(c / s)
    return cleaned

def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best sequence seen and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07                # slightly larger than the baseline
    lam = 0.30               # temperature for the smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → point on the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (sum(a)=1 ⇒ denominator drops out)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional random shake if we are stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the exact max‑convolution.
    Returns the best sequence seen and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                      # initial step size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                  # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


def _run_proj_gd(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Projected gradient descent on the *exact* max‑convolution.
    The iterate is kept on the probability simplex via Euclidean projection.
    Returns the best sequence seen and its objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={it}] new best {val:.6f}")

        # sub‑gradient of the max‑convolution
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule (decreasing like 1/√t)
        eta = 0.6 / np.sqrt(it)    # 0.6 works well in practice
        a_new = a - eta * grad
        a = _simplex_projection(a_new)

        if time.time() - last_improve > 8.0:
            # stuck – a tiny random perturbation may help
            a += (np.random.rand(n) - 0.5) * 5e-3
            a = _simplex_projection(a)
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 250_000):
    """
    Cheap stochastic hill‑climb: move a small ε from one index to another
    whenever it improves the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  Returns the best sequence found before the
    allocated time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths – increasing expressive power; the larger ones are tried only
    # if time permits.
    lengths = [2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budgets – scale roughly as 1/n
        max_pg_iters = max(2_000, int(12_000 * 2048 / n))
        max_adam_iters = max(3_000, int(9_000 * 2048 / n))
        max_subgrad_iters = max(4_000, int(6_000 * 2048 / n))

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # 1. Projected gradient descent (exact max)
            a_pg, val_pg = _run_proj_gd(init, n, deadline, max_pg_iters)
            if val_pg < best_val:
                best_val = val_pg
                best_seq = a_pg.copy()
                print(f"[n={n} cand={cid}] PGD set new global best {val_pg:.6f}")

            # 2. Adam fine‑tune on smooth‑max
            a_adam, val_adam = _run_adam(a_pg, n, deadline, max_adam_iters)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # 3. Sub‑gradient (exponentiated‑gradient) refinement
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad_iters)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

        # 4. If we have a global best of this length, give it a short extra
        #    Adam run (acts as a fine‑tune on the *current* best).
        if best_seq is not None and len(best_seq) == n:
            a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                         max(1_000, max_adam_iters // 2))
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap stochastic local search
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] Hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
