"""
propose_candidate
-----------------

Search for a non‑negative step‑function (vector) that minimizes

    F(a) = 2·n·max(a * a) / (sum(a)^2)

The algorithm consists of three phases:

1. **Multi‑start exploration (n = 4096)**  
   – Generate many diverse random initial vectors (Dirichlet, piecewise‑constant,
     random sinusoid mixtures, power‑law shapes, the previous solution if any).  
   – For each start run a short Adam optimisation on a smooth‑max surrogate,
     followed by a brief exponentiated‑gradient sub‑gradient phase.  
   – Keep the best vector found.

2. **Deep refinement on a larger grid (n = 8192)**  
   – Linearly interpolate the best vector from phase 1 to length 8192.  
   – Run a long Adam optimisation (temperature slowly increased) and a
     longer sub‑gradient phase with an adaptive step size.

3. **Cheap local random hill‑climbing** to squeeze out any remaining gain.

All optimisation steps respect the global time budget (default 1000 s) and
always return a valid sequence (non‑negative, sum > 0.01).  The code uses only
NumPy and the provided ``evaluate_sequence`` for final scoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry: v = (v + rev(v)) / 2 and renormalise."""
    v = (v + v[::-1]) * 0.5
    s = v.sum()
    if s > 0:
        v /= s
    return v


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best simplex vector seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    lam = 0.5          # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex (projection onto probability simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # optional symmetry projection – helps the search
        a = _symmetrize(a)

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over convolution values)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – slowly focus on the true maximum
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter when we seem to stall
        if it % 4000 == 0 and time.time() > deadline - 20:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    An adaptive step‑size (η_t = η0 / sqrt(t+1)) is used.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set = indices attaining the max (with tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # adaptive multiplicative update
        eta = eta0 / math.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # symmetry projection (helps the landscape)
        a = _symmetrize(a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: transfer a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def _random_piecewise_constant_candidate(n: int) -> np.ndarray:
    """Create a piecewise‑constant distribution with a random number of blocks."""
    k = np.random.randint(2, 256)          # number of blocks
    probs = np.random.rand(k)
    probs /= probs.sum()
    # allocate roughly equal lengths, then adjust remainder
    lengths = np.full(k, n // k, dtype=int)
    leftovers = n - lengths.sum()
    if leftovers > 0:
        extra_pos = np.random.choice(k, leftovers, replace=False)
        lengths[extra_pos] += 1
    a = np.empty(n, dtype=np.float64)
    pos = 0
    for blk, ln in zip(probs, lengths):
        if ln > 0:
            a[pos:pos + ln] = blk / ln
        pos += ln
    return a


def _random_sinusoid_candidate(n: int) -> np.ndarray:
    """Mixture of a few sinusoids, shifted to be non‑negative."""
    x = np.arange(n, dtype=np.float64) / n
    p = np.random.randint(1, 5)            # number of sinusoid components
    shape = np.zeros_like(x)
    for _ in range(p):
        amp = np.random.rand()
        freq = np.random.uniform(0.0, 0.5)
        phase = np.random.uniform(0.0, 2 * np.pi)
        shape += amp * np.sin(2 * np.pi * freq * x + phase)
    shape = np.maximum(shape, 0.0)
    # avoid the all‑zero case
    if shape.sum() == 0:
        shape = np.random.rand(n)
    shape /= shape.sum()
    return shape


def _random_powerlaw_candidate(n: int) -> np.ndarray:
    """Edge‑concentrated power‑law shape (the classic “tent” with random exponent)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1)
    alpha = np.random.uniform(-2.0, -0.5)
    shape = np.where(prod > 0, prod ** alpha, 0.0)
    if shape.sum() == 0:
        shape = np.random.rand(n)
    shape /= shape.sum()
    return shape

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found as a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5       # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – many cheap restarts on the modest grid (n=4096)
    n_small = 4096
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # warm start from a previously saved solution, if present
    warm = globals().get("height_sequence_1")
    warm_arr = None
    if isinstance(warm, (list, tuple, np.ndarray)):
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
            if warm_arr.ndim == 1:
                if warm_arr.shape[0] != n_small:
                    if warm_arr.shape[0] > n_small:
                        warm_arr = warm_arr[:n_small]
                    else:
                        warm_arr = np.pad(warm_arr, (0, n_small - warm_arr.shape[0]))
                warm_arr = np.maximum(warm_arr, 0.0)
                if warm_arr.sum() >= 0.01:
                    warm_arr /= warm_arr.sum()
                else:
                    warm_arr = None
        except Exception:
            warm_arr = None

    # helper to run a short optimisation on a given start vector
    def _run_short_opt(a_start: np.ndarray) -> tuple[np.ndarray, float]:
        """Run a brief Adam + sub‑gradient on a_start (n_small)."""
        s = a_start.sum()
        if s < 1e-12:
            a0 = np.random.dirichlet(np.ones(n_small))
        else:
            a0 = a_start / s

        # allocate a small time slice (≈10 s) – enough for a few thousand iters
        local_deadline = min(deadline - 4.0, time.time() + 10.0)

        a_adam, _ = _adam_optimize(
            a0,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            local_deadline,
            max_iters=30_000,      # will stop earlier if time runs out
        )
        # a few sub‑gradient iterations
        if time.time() < deadline - 3:
            a_sub, val_sub = _subgradient_optimize(
                a_adam,
                n_small,
                conv_fft_len_small,
                corr_fft_len_small,
                deadline,
                max_iters=20_000,
                eta0=0.6,
            )
        else:
            a_sub = a_adam
            conv = _full_convolution(a_sub, conv_fft_len_small)
            val_sub = 2.0 * n_small * conv.max()
        return a_sub, val_sub

    # generate candidates until we need to reserve time for the next phases
    cand_index = 0
    while time.time() < deadline - 120.0:   # keep ~2 min for refinement & final steps
        if cand_index == 0 and warm_arr is not None:
            cand = warm_arr.copy()
        elif cand_index % 4 == 1:
            cand = np.random.dirichlet(np.ones(n_small))
        elif cand_index % 4 == 2:
            cand = _random_piecewise_constant_candidate(n_small)
        elif cand_index % 4 == 3:
            cand = _random_sinusoid_candidate(n_small)
        else:
            cand = _random_powerlaw_candidate(n_small)

        cand_index += 1
        a_opt, val_opt = _run_short_opt(cand)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Phase 1 cand {cand_index}] new global best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a larger grid (n=8192) and run a deep optimisation
    if best_seq is not None and time.time() < deadline - 60.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the best small‑grid vector
        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # long Adam phase – give it a sizeable chunk of the remaining budget
        print("[Phase 2] starting deep Adam optimisation on larger grid")
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=250_000,
        )

        # long sub‑gradient phase
        if time.time() < deadline - 10.0:
            print("[Phase 2] starting sub‑gradient refinement")
            a_big_ref, val_big_ref = _subgradient_optimize(
                a_big_adam,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=300_000,
                eta0=0.6,
            )
        else:
            a_big_ref = a_big_adam
            conv = _full_convolution(a_big_ref, conv_fft_len_big)
            val_big_ref = 2.0 * n_big * conv.max()

        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Phase 2] new best after refinement {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap local random search on the final vector
    if best_seq is not None and time.time() < deadline - 5.0:
        n_final = best_seq.shape[0]
        conv_fft_len_final = _next_pow2(2 * n_final - 1)
        print("[Phase 3] performing final local random search")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n_final,
            conv_fft_len_final,
            deadline,
            max_moves=40_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase 3] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the domain accepted by evaluate_sequence()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
