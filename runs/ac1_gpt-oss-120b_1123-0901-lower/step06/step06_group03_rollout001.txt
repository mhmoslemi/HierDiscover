"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Returns the best sequence/value seen during this phase.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # work in log‑space (exponential map)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                 # a bit larger than the reference
    lam = 0.2                                 # start temperature
    lam_factor = 1.025                        # grow faster

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < 1_000_000:
        # Soft‑max → point on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # Random shake if stagnating
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return global_best_seq, global_best_val


def _pgd_exact_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent directly on the exact max‑convolution.
    Uses a decaying step size and Euclidean projection onto the simplex.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    it = 0
    last_improve = time.time()
    eta0 = 0.6

    while time.time() < deadline and it < 30_000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            # numerical fallback
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        # early stop if stagnated
        if time.time() - last_improve > 6.0:
            break

        it += 1

    return global_best_seq, global_best_val


def _eg_subgradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase (same as reference) used for
    final polishing.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()

    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} EG  iter={sub_it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(sub_it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val


def _local_random_perturb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing style mass‑transfer moves.
    Randomly selects two indices i,j and transfers a small amount δ.
    Accepts only if the objective improves.
    """
    a = a0.copy()
    a = a / a.sum()
    rng = np.random.default_rng()

    it = 0
    last_improve = time.time()
    while time.time() < deadline and it < 200_000:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.random() * 1e-4  # tiny transfer
        if a[i] < delta:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # no need to renormalise – sum unchanged

        conv = _full_convolution(a_candidate, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a_candidate.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a_candidate.copy()
            a = a_candidate
            last_improve = time.time()
            print(f"[n={n} Perturb iter={it:06d}] new best {val:.6f}")
        else:
            # occasional random reset when stuck
            if time.time() - last_improve > 5.0:
                a = np.random.dirichlet(np.ones(n))
                a = a / a.sum()
                last_improve = time.time()

        it += 1

    return global_best_seq, global_best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Construct a diverse set of starting vectors."""
    pool = []

    # Warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # Power‑law edge‑concentrated families (prod = i·(n‑i+1))
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑exponential (two‑sided Laplace)
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian bumps
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # Single sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Random Dirichlet draws
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    # --------- Arcsine‑type shapes (1/√(i·(n‑i))) ----------
    eps = 1e-12
    arcs = 1.0 / np.sqrt((idx) * (n - idx + 1.0) + eps)
    arcs /= arcs.sum()
    pool.append(arcs)

    # --------- Two‑bump mixtures (two shifted arcsine peaks) ----------
    for _ in range(8):
        p = np.random.rand()  # centre proportion in (0,1)
        w = np.random.rand()
        left = np.arange(n) / (n - 1)
        bump = np.exp(-0.5 * ((left - p) ** 2) / (0.02 ** 2))
        bump += np.exp(-0.5 * ((left - (1 - p)) ** 2) / (0.02 ** 2))
        bump = np.maximum(bump, eps)
        bump = bump ** -0.5  # 1/sqrt shape
        bump = w * bump + (1 - w) * np.random.dirichlet(np.ones(n))
        bump /= bump.sum()
        pool.append(bump)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine respects the supplied time‑budget (default 1000 s) and returns
    the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1. Choose a base support size (start modestly, enlarge later)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm‑start from a previously discovered construction if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # --------------------------------------------------------------
    # 2. Generate a diverse candidate pool and screen them quickly
    candidates = _generate_candidate_pool(n, w)
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    screened = []

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        screened.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # Keep only the most promising starts (up to eight)
    screened.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in screened[:8]]

    # --------------------------------------------------------------
    # 3. Deep optimisation passes for each top candidate
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] Adam → PGD → EG")
        # Adam on smooth‑max
        best_seq, best_val = _adam_smooth_max(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )
        # Projected gradient descent on the exact max
        best_seq, best_val = _pgd_exact_max(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )
        # Exponentiated‑gradient sub‑gradient polishing
        best_seq, best_val = _eg_subgradient(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # --------------------------------------------------------------
    # 4. Optional up‑sampling to a larger grid if budget allows
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Interpolate the current best to the larger grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[big] Adam → PGD → EG on enlarged grid")
        best_seq, best_val = _adam_smooth_max(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big, deadline, best_seq, best_val
        )
        best_seq, best_val = _pgd_exact_max(
            best_seq, n_big, conv_fft_len_big, corr_fft_len_big, deadline, best_seq, best_val
        )
        best_seq, best_val = _eg_subgradient(
            best_seq, n_big, conv_fft_len_big, corr_fft_len_big, deadline, best_seq, best_val
        )

        # One more tiny random‑perturbation phase (cheap, helps escape plateaus)
        best_seq, best_val = _local_random_perturb(
            best_seq, n_big, conv_fft_len_big, deadline, best_seq, best_val
        )

        # If we still have time, go one step further (16384)
        if time.time() + 30 < deadline:
            n_big2 = 16384
            conv_fft_len_big2 = _next_pow2(2 * n_big2 - 1)
            corr_fft_len_big2 = _next_pow2(3 * n_big2 - 2)

            x_old = np.linspace(0, 1, n_big, endpoint=False)
            x_new = np.linspace(0, 1, n_big2, endpoint=False)
            a_big2 = np.interp(x_new, x_old, best_seq)
            a_big2 = np.maximum(a_big2, 0.0)
            a_big2 /= a_big2.sum()

            print("[bigger] Adam → PGD → EG")
            best_seq, best_val = _adam_smooth_max(
                a_big2, n_big2, conv_fft_len_big2, corr_fft_len_big2,
                deadline, best_seq, best_val
            )
            best_seq, best_val = _pgd_exact_max(
                best_seq, n_big2, conv_fft_len_big2, corr_fft_len_big2,
                deadline, best_seq, best_val
            )
            best_seq, best_val = _eg_subgradient(
                best_seq, n_big2, conv_fft_len_big2, corr_fft_len_big2,
                deadline, best_seq, best_val
            )
            best_seq, best_val = _local_random_perturb(
                best_seq, n_big2, conv_fft_len_big2, deadline, best_seq, best_val
            )

    # --------------------------------------------------------------
    # 5. Final fallback if nothing was produced
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    elapsed = time.time() - start_time
    print(f"Search finished in {elapsed:.1f}s – best value ≈ {best_val:.6f}")

    return best_seq.tolist()
```

This implementation enriches the search space, adds a faster Adam schedule, introduces a projected‑gradient descent phase on the exact max‑convolution, and finishes with a cheap random perturbation step. It also automatically upsamples to larger supports (8192 → 16384) when time permits, which typically yields a noticeable reduction of the bound, aiming to push the final value below the target 1.503.
