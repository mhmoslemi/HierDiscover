"""
Propose a step‑function (non‑negative vector) that minimises

      F(a) = 2 * n * max(a * a) / (sum(a)**2)

where * is linear convolution.  The objective is homogeneous, so we may
normalise the sum to 1.  Empirically the best vectors are strongly
asymmetric: a_i ∝ (n‑i)^p with a moderate integer exponent p.
For a fixed length n the maximum of the convolution is attained at the
central lag k=n‑1 and equals

      max(a*a) = (p+1)^2·B(p+1,p+1) / n   (B = beta‑function)

so that   F(p) = 2·(p+1)^2·B(p+1,p+1).

For n≈3·10⁴ this value decreases from 2 (uniform, p=0) down to ≈4/3
(p=1) and then to ≈0.6 (p=2), ≈0.23 (p=3) and ≈0.08 (p=4).  For p≈7
the term a₀² becomes dominant and the bound rises again, so the optimum
lies at modest p (≈2…5).  The search below evaluates a short list of
integer exponents, mixes each shape with the uniform distribution,
keeps the best candidate and finishes with a couple of gradient‑refinement
steps (Adam on a smooth‑max surrogate and an exact ℓ∞ sub‑gradient
refinement).  All inner evaluations use an FFT‑based convolution, which
is orders of magnitude faster than the naïve np.convolve used by the
external `evaluate_sequence` function.

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation of the objective (FFT based, no clamping)
# ----------------------------------------------------------------------


def _fast_objective(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)^2 using FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Convolution / cross‑correlation helpers (exact same as in the
# previous solution – required by the gradient phases)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives (identical to the previous implementation)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0               # smooth‑max temperature
    lam_factor = 1.02

    # iteration budget scaled with vector length
    max_iters = int(40000 * 1024 / n) + 2000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to enforce the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        lam = min(lam * lam_factor, 1e8)

        # exact objective (sum(a)=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional jitter to escape plateaus
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                      # fallback
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                       # decay

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    function.  The routine first explores the analytic family
    a_i ∝ (n‑i)^p (p = 0…12) and mixtures with the uniform vector.
    The best candidate is then refined by Adam + exact sub‑gradient.
    The final sequence (list of floats) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Analytic family search (very cheap)
    # ------------------------------------------------------------------
    lengths = [32768]   # a good trade‑off between quality and speed
    p_candidates = list(range(0, 13))          # 0 … 12
    mix_weights = [0.2, 0.4, 0.6, 0.8]          # mixture with uniform

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Analytic search for n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))

        # descending integers  n, n-1, … , 1  (float)
        idx_desc = np.arange(n, 0, -1, dtype=np.float64)

        uniform = np.full(n, 1.0 / n, dtype=np.float64)

        for p in p_candidates:
            if time.time() > deadline:
                break
            shape = np.power(idx_desc, p, dtype=np.float64)
            shape_sum = shape.sum()
            if shape_sum == 0:
                continue
            shape /= shape_sum

            # pure power‑law shape
            val = _fast_objective(shape, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = shape.copy()
                print(f"[p={p}] new best {best_val:.6f}")

            # mixtures with the uniform vector
            for w in mix_weights:
                if time.time() > deadline:
                    break
                mix = w * shape + (1.0 - w) * uniform
                mix /= mix.sum()
                val = _fast_objective(mix, conv_fft_len)
                if val < best_val:
                    best_val = val
                    best_seq = mix.copy()
                    print(f"[p={p} w={w:.1f}] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional refinement (few seconds, still far below the budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))
        print("\n--- Refinement phase (Adam + sub‑gradient) ---")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        a_ref, val_ref = _subgradient_refine(
            a_adam, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"Refinement improved to {best_val:.6f}")
        elif val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"Adam alone improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(2048, 1.0 / 2048, dtype=np.float64)
        print("Fallback to uniform 2048‑vector.")

    print(f"\n=== Final result: objective ≈ {best_val:.6f} (length {len(best_seq)}) ===")
    return best_seq.tolist()
