"""

        F(a) = 2·n·max(a∗a) / (∑a)²

import time
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum(a) is assumed ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Mirror‑descent on a smooth‑max surrogate -------------------------------
def _smooth_max_mirror_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the log‑sum‑exp surrogate
    of the max‑convolution.  Returns the best vector/value seen during the
    run.
    """
    eps = 1e-15
    # initialise logits = log(a0)  (softmax will enforce the simplex)
    logits = np.log(a0 + eps)

    # Adam‑style learning‑rate schedule is not needed – a plain
    # multiplicative update works very well for the exponential
    # parametrisation.
    step = 0.1
    temperature = 0.5               # start temperature for smooth‑max
    temp_factor = 1.02               # geometric increase

    last_improve = time.time()
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        # ---------- map logits → probability vector (simplex) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)          # avoid overflow
        a = exp_log / exp_log.sum()            # a ≥ 0, Σa = 1

        # ---------- evaluate true objective (for bookkeeping) ----------
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)   # a.sum()==1 ⇒ 2·n·max

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} mirror iter={it:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        max_c = conv.max()
        w = np.exp(temperature * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---------- mirror‑descent (exponentiated‑gradient) update ----------
        #   log‑space gradient: ∂/∂log a_i = a_i (∂L/∂a_i – ⟨a,∂L/∂a⟩)
        g_center = np.dot(a, grad_a)
        grad_log = a * (grad_a - g_center)

        logits += step * grad_log          # ascent on the surrogate
        temperature = min(temperature * temp_factor, 1e6)

        # ---------- occasional random shake if stuck ----------
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val

# Exact sub‑gradient polishing -------------------------------------------
def _exact_subgradient_polish(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    A short exponentiated‑gradient phase that uses the genuine max‑convolution
    (active set weighting) to sharpen the solution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # enforce simplex

    eta = 0.6
    last_improve = time.time()
    it = 0
    max_iter = 15_000

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() == 1

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} subgrad iter={it:05d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0 and on the simplex)
        a = a * np.exp(-eta * grad_a)
        a_sum = a.sum()
        if a_sum < 1e-12:                     # re‑initialise if we collapsed
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val

def _blocked_uniform(n: int, block_len: int, offset: int) -> np.ndarray:
    """
    Uniform mass on a contiguous block of length `block_len` (clipped to the grid)
    placed at `offset`.  The rest is zero.  Returned vector is normalised.
    """
    vec = np.zeros(n, dtype=np.float64)
    end = min(n, offset + block_len)
    vec[offset:end] = 1.0
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vec / total


def _double_exponential(n: int, left_alpha: float, right_alpha: float) -> np.ndarray:
    """
    a_i ∝ exp(-left_alpha * i) + exp(-right_alpha * (n-1-i)).
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-left_alpha * i) + np.exp(-right_alpha * (n - 1 - i))
    return a / a.sum()


def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse list of candidate vectors.
    """
    candidates = []

    # 1. warm‑start and simple shifts

    # 2. uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # 3. random Dirichlets (various concentrations)

    # 4. edge‑exponential families with optional uniform mix
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.2, 20):

        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # 5. double‑exponential (different decay left/right)
    for left in np.logspace(-2, 1.0, 12):
        for right in np.logspace(-2, 1.0, 12):
            candidates.append(_double_exponential(n, left, right))

    # 6. blocked‑uniform families (different block sizes / positions)
    for blk_len in (int(0.05 * n), int(0.15 * n), int(0.3 * n), int(0.5 * n)):

            candidates.append(_blocked_uniform(n, blk_len, off))

    # 7. power‑law edges
    for p in np.linspace(0.2, 5.0, 15):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8):
            candidates.append(w * edge + (1.0 - w) / n)

    # 8. sinusoidal half‑wave
    half = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    half = np.maximum(half, 0.0)
    half /= half.sum()
    candidates.append(half)

    return candidates

            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep the K best for the expensive phase
    cand_vals.sort(key=lambda x: x[0])
    K = 20
    top_candidates = [c for _, c in cand_vals[:K]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the retained starts
    for start_id, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"=== Optimising start #{start_id + 1}/{len(top_candidates)} (n={n}) ===")
        best_seq, best_val = _optimise_one_start(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – optional refinement on a doubled grid
    time_left = deadline - time.time()
    if time_left > 45.0:  # enough time for a full second stage
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # simple linear interpolation
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Second‑grid optimisation (n=8192) ---")
        best_seq, best_val = _optimise_one_start(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – if everything failed, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the range required by the evaluator.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best value ≈ {best_val:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
