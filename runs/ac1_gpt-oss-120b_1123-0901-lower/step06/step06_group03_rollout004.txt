"""
Propose a step‑function that (hopefully) beats the current best bound ~1.5085.

Main ideas
---------
* Build a richer pool of starting shapes (more extreme power‑law edges,
  double‑exponential tails, mixture of Gaussians, random Dirichlet draws).
* Run a two‑phase local optimiser on each promising start:
    - Adam on a smooth‑max surrogate (log‑sum‑exp) with a slower temperature
      schedule (lam grows by only 0.5 % per iteration) and a larger iteration
      budget.
    - Exact sub‑gradient phase (exponentiated‑gradient mirror descent) that
      runs until the time budget expires.
* After the first optimisation on n=4096 we up‑sample the current best
  distribution to a larger grid (8192 and, if time permits, 16384) and redo
  the same two‑phase optimisation – a finer discretisation often yields a
  slightly smaller supremum of the convolution.
* Throughout we keep a global best sequence/value and constantly
  re‑initialise the optimiser from that best point after a long stagnation
  period (random “shake”).
* All vectors are kept on the probability simplex (soft‑max parametrisation)
  so the scale invariance of the objective is preserved.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smoothmax(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Returns the best (seq, value) seen during this phase.
    """
    eps = 1e-15
    # initialise logits = log(a) (soft‑max will enforce the simplex)
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    # start temperature (lam) close to 0 → very smooth; increase slowly
    lam = 0.2
    lam_factor = 1.005  # slower growth than the original code

    last_improve = time.time()
    it = 0
    max_iters = 800_000   # generous safety bound

    while time.time() < deadline and it < max_iters:
        # ---- soft‑max to obtain a on the simplex ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # objective (exact) for logging / global best update
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:07d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        # w ∝ exp(lam * (conv - max_c))
        # This approximates the indicator of the max location.
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # derivative of the soft‑max (logits → a)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random perturbation if we have been stagnant
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_phase(
    a_start: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (mirror‑descent with exponentiated updates) on the
    true max‑convolution objective.
    """
    a = a_start.copy()
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    last_improve = time.time()
    it = 0
    max_iters = 200_000

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # active set: positions attaining the maximal convolution value
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent step (exponentiated gradient)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # re‑project onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if we have not improved for a while
        if time.time() - last_improve > 8.0:
            break

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient on a given start vector.
    The routine returns the global best (seq, val) seen across the whole run.
    """
    # ---- Adam on smooth‑max ----
    best_seq, best_val = _adam_smoothmax(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    # ---- Exact sub‑gradient ----
    best_seq, best_val = _subgradient_phase(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of start vectors.
    """
    pool: list[np.ndarray] = []

    # warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 5, n // 2, 3 * n // 5):
            pool.append(np.roll(warm, shift))

    # 1. uniform
    pool.append(np.full(n, 1.0 / n))

    # 2. power‑law edges (more extreme exponents)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)               # i·(n‑i+1)
    for alpha in np.linspace(-4.0, -0.1, 20):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # 3. double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.05, 4.0, 20):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # 4. Gaussian bumps of varying width
    for sigma in np.linspace(n * 0.02, n * 0.45, 12):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # 5. Sinusoidal half‑wave (already present but keep)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # 6. Mixture of two Gaussians (different means, same sigma)
    for sigma in (n * 0.07, n * 0.15, n * 0.30):
        m1, m2 = n * 0.25, n * 0.75
        g1 = np.exp(-0.5 * ((i - m1) ** 2) / (sigma ** 2))
        g2 = np.exp(-0.5 * ((i - m2) ** 2) / (sigma ** 2))
        shape = 0.5 * g1 + 0.5 * g2
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # 7. Random Dirichlet draws (wide range of concentration)
    for alpha in np.logspace(-3, 0, 15):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the given evaluation metric.  The routine respects a time budget (default
    1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Primary support size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start if a previous construction is available
    warm_start = globals().get("height_sequence_1")
    if isinstance(warm_start, (list, np.ndarray)):
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # Build a diverse candidate pool and evaluate them quickly
    candidates = _generate_candidate_pool(n, w)
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored = []   # list of (value, vector)

    print(f"[Setup] evaluating {len(candidates)} initial candidates on n={n}")
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if (idx + 1) % 20 == 0 or idx == len(candidates) - 1:
            print(f"[Screen] processed {idx + 1}/{len(candidates)} – current best {best_val:.6f}")

    # Keep the most promising starts (up to 12)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:12]]

    # ------------------------------------------------------------------
    # Deep optimisation passes on the chosen starts
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[Optim start {cid}] launching deep optimisation (n={n})")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[Optim start {cid}] best after this start {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger grid if time permits
    def _upsample_and_opt(old_seq: np.ndarray, new_n: int) -> tuple[np.ndarray, float]:
        """Interpolate a sequence to size new_n and run deep optimisation."""
        fft_len_conv = _next_pow2(2 * new_n - 1)
        fft_len_corr = _next_pow2(3 * new_n - 2)
        x_old = np.linspace(0.0, 1.0, old_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        a_new = np.interp(x_new, x_old, old_seq)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        print(f"[Upsample] new size {new_n}, starting deep optimisation")
        return _deep_optimize(
            a_new, new_n, fft_len_conv, fft_len_corr, deadline, best_seq, best_val
        )

    # First enlargement to 8192
    if time.time() + 12 < deadline:
        best_seq, best_val = _upsample_and_opt(best_seq, 8192)
        print(f"[After 8192] best val {best_val:.6f}")

    # Second enlargement to 16384 if still time left
    if time.time() + 12 < deadline:
        best_seq, best_val = _upsample_and_opt(best_seq, 16384)
        print(f"[After 16384] best val {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity: ensure non‑negativity and evaluator bounds
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[Finish] best objective {best_val:.6f} with length {best_seq.shape[0]}")
    return best_seq.tolist()
