def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # conv length = fft_len, but first part contains needed conv; we need up to 2*len(a)-1
    return conv[: 2*a.shape[0] - 1]
```

**_cross_correlation**: Need correlation of w with a (size n). Implementation earlier: rev_a = a[::-1]; compute via FFT.

```
def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]
```

def _adam_smooth_max(a0, n, conv_fft_len, corr_fft_len, deadline, max_iter=200000, best_seq=None, best_val=float('inf')):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.01
    lam_cap = 1e6
    last_improve = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        # softmax transform
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # conv & val
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n} Adam iter={it}] new best {val:.6f}")
        # smoothmax weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # grad
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        lam = min(lam * lam_factor, lam_cap)
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        it += 1
    return best_seq, best_val
```

def _subgradient_refinement(a0, n, conv_fft_len, corr_fft_len, deadline, max_iter=20000, best_seq=None, best_val=float('inf')):
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s
    eta = 0.6
    last_improve = time.time()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n} Subgrad iter={it}] new best {val:.6f}")
        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        if time.time() - last_improve > 5.0:
            break
    return best_seq, best_val
```

"""
propose_candidate
-----------------
A multi‑stage stochastic optimiser for the “max‑auto‑convolution / sum²”
objective.  The algorithm works in three phases:

1. **Diverse warm‑starts** – generate thousands of candidate shapes
   (uniform, edge‑exponential / power‑law families, beta‑like
   distributions, random Dirichlet draws and random piece‑wise linear
   profiles).  The best few (by a cheap FFT‑based bound) are kept.

2. **Exploratory optimisation** – each retained start is run through a
   short Adam optimiser on a smooth‑max surrogate of the convolution
   peak (log‑sum‑exp).  The optimiser uses a soft‑max to stay on the
   simplex and injects occasional random “shakes’’ to escape plateaus.
   A few exponentiated‑gradient sub‑gradient steps (true max) follow
   each Adam run.

3. **Refinement & up‑scaling** – the globally best vector is refined
   with a longer Adam / sub‑gradient run.  If time permits the solution
   is interpolated to a finer grid (n = 8192) and the whole deep‑optim
   routine is repeated, giving the algorithm a chance to improve the
   bound further on a larger problem.

All heavy loops respect a global deadline (1000 s – 0.5 s safety) and
periodically print progress.  The returned sequence is clipped to
the evaluator’s allowed range (≥0, ≤1000) and converted to a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear (a*a) convolution using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of  F(a)=2·n·max(a∗a)/(sum(a)²)  (sum≈1 after normalisation).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

# Candidate families ----------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produces a rich collection of analytically‑parameterised vectors.
    """
    i = np.arange(n, dtype=np.float64)
    candidates = []

    # warm start and its circular shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws with many concentration values
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # Edge‑exponential families (denser grid)
    for alpha in np.logspace(-3, 1.0, num=30):  # 0.001 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge = edge / edge.sum()
        for w in (0.0, 0.15, 0.3, 0.45, 0.6, 0.75, 0.9, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law families (denser grid)
    for p in np.linspace(0.2, 6.0, num=30):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = edge / edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Beta‑like shapes (x^{α-1}(1−x)^{β-1})
    xs = (i + 0.5) / n
    for alpha in np.logspace(-2, 0.5, num=12):
        for beta in np.logspace(-2, 0.5, num=12):
            shape = (xs ** (alpha - 1.0)) * ((1.0 - xs) ** (beta - 1.0))
            shape = np.maximum(shape, 0.0)
            shape = shape / shape.sum()
            candidates.append(shape)

    # Two‑Gaussian edges
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    candidates.append(shape)

    return candidates


def _random_piecewise_linear_candidate(n: int) -> np.ndarray:
    """
    Generates a random piece‑wise linear profile:
    choose a few break‑points, assign random heights, linearly interpolate.
    """
    num_bp = np.random.randint(5, 15)               # number of break‑points
    pts = np.sort(np.random.choice(n, num_bp, replace=False))
    heights = np.random.rand(num_bp) + 1e-3         # stay strictly positive
    arr = np.interp(np.arange(n), pts, heights,
                    left=heights[0], right=heights[-1])
    arr = np.maximum(arr, 0.0)
    arr /= arr.sum()
    return arr

# Optimisers ------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iter: int = 200_000,
                     best_seq: np.ndarray | None = None,
                     best_val: float = float('inf')) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) surrogate.
    The iterate lives on the simplex via a soft‑max of log‑probabilities.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5               # start temperature
    lam_factor = 1.01       # slower increase than the baseline
    lam_cap = 1e6

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & true objective (for logging) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain‑rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional random shake if stuck -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(start_vec: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            max_iter: int = 30_000,
                            best_seq: np.ndarray | None = None,
                            best_val: float = float('inf')) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient steps on the genuine max‑convolution.
    """
    a = start_vec.copy()
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        total = a.sum()
    a /= total

    eta = 0.6
    last_improve = time.time()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:                 # avoid numerical collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if no progress for a while
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float,
                   adam_iters: int,
                   subgrad_iters: int) -> tuple[np.ndarray, float]:
    """
    One deep optimisation run: Adam (smooth‑max) → sub‑gradient.
    """
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len,
        deadline,
        max_iter=adam_iters,
        best_seq=global_best_seq,
        best_val=global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        best_seq, n, conv_fft_len, corr_fft_len,
        deadline,
        max_iter=subgrad_iters,
        best_seq=best_seq,
        best_val=best_val,
    )
    return best_seq, best_val

# Utility to upscale a vector ------------------------------------------------
def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n points on [0,1)."""
    old_n = seq.shape[0]
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total < 1e-12:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq /= total
    return new_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Base problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start from previous best if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # 1) generate a large pool of diverse candidates
    candidates = _generate_parametric_candidates(n, warm_start)

    # add a few random piece‑wise linear shapes for extra diversity
    for _ in range(200):
        candidates.append(_random_piecewise_linear_candidate(n))

    # --------------------------------------------------------------
    # Quick evaluate all candidates (using the cheap FFT bound)
    cand_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep the most promising seeds
    cand_vals.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # --------------------------------------------------------------
    # 2) short exploratory optimisation on each seed
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    short_adam_iters = 80_000
    short_subgrad_iters = 10_000

    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Short optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            adam_iters=short_adam_iters,
            subgrad_iters=short_subgrad_iters,
        )

    # --------------------------------------------------------------
    # 3) long refinement on the current global best
    if time.time() < deadline - 10.0 and best_seq is not None:
        print("--- Long refinement on current best ---")
        best_seq, best_val = _deep_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            adam_iters=250_000,
            subgrad_iters=30_000,
        )

    # --------------------------------------------------------------
    # 4) if time permits, move to a finer grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 60.0:   # need a little budget for the larger problem
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            a_big = _interpolate_sequence(best_seq, n_big)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
            adam_iters=200_000,
            subgrad_iters=20_000,
        )
        # keep the larger vector only if it truly improves the bound
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
