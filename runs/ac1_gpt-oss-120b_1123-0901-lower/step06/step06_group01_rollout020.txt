"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

The search consists of three stages:

1. **High‑resolution optimisation on a modest size** (n≈600‑800).
   A projected‑gradient descent with Nesterov momentum and an Adam‑style
   optimiser on the log‑it representation are run for many iterations
   (the cheap size makes millions of cheap FFT steps feasible).

2. **Upscaling** the best‑found shape to a larger length (≈2500‑3500).
   The shape is interpolated, re‑normalised to Σa=√(2 n) and
   monotonicity is enforced (sorting descending).  A short burst of the
   same two optimisers is applied to the upscaled vector.

3. **Stochastic refinement** – a simulated‑annealing mass‑transfer
   hill‑climber that moves a random fraction of mass between two
   positions.  The convolution is updated with a full FFT each step
   (fast enough for a few hundred‑thousand steps) and worsening moves
   are occasionally accepted according to a decaying temperature.

All steps respect the global time budget (the harness supplies it as
``budget_s``).  The best sequence discovered (as a plain Python list)
is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target_sum}.
    Implementation of Duchi et al. (2008) – O(n log n).
    """
    if target_sum <= 0.0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, v.size + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, Nfft: int) -> float:
    """Exact maximum of a linear convolution a*a using a single FFT."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """
    Produce a small yet diverse set of feasible start vectors.
    All vectors sum to ``target_sum`` and are non‑negative.
    """
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0003, 0.008)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (
        a0
        - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
        + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    )
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(
        scale=0.02 * target_sum / n, size=n
    )
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular (peak centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes – useful for exploring non‑smooth shapes
    k = max(2, min(15, n // 150))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent with Nesterov momentum on the exact
    max‑convolution.  Returns the best vector seen and its exact objective.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise on the simplex
    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())          # >= 3n‑2
    step0 = 0.0015 * target_sum                      # base step
    beta = 0.9                                        # momentum factor
    v = np.zeros_like(a)                            # momentum accumulator
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact convolution & current max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # ----- active set (hard max) -----
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ----- weighted reverse convolution (sub‑gradient) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]
        g = 2.0 * rev                               # ∇_a max(a∗a)

        # ----- momentum + projection step -----
        v = beta * v + g
        lr = step0 / math.sqrt(it)                  # annealed step size
        a_tmp = a - lr * v
        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- occasional jitter to avoid stagnation -----
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a
    soft‑max approximation of the max‑convolution.
    Returns the best vector seen and its exact objective.
    """
    target_sum = math.sqrt(2.0 * n)
    eps = 1e-16

    # initialise logits from the start vector (avoid log(0))
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.07

    Nfft = 1 << ((3 * n - 2).bit_length())

    # temperature schedule (soft‑max smoothing)
    tau_factor = 0.35
    decay_rate = max_iters / 6.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- current vector (softmax of logits) ----
        logits = y - np.max(y)                 # shift for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()      # lies on probability simplex
        a = target_sum * p

        # ---- convolution & current max (exact) ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- keep global best ----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # ---- soft‑max weights for the max ----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # hard max – avoid numerical issues
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()

        # ---- weighted reverse convolution (gradient of smoothed max) ----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g_a = 2.0 * rev                         # ∇_a (soft)max

        # ---- chain rule to logits (mirror descent on simplex) ----
        avg = np.dot(p, g_a)                    # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)      # ∂f/∂y_i

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # ---- occasional jitter to shake out of plateaux ----
        if it % 4500 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    move a small random fraction of mass from i to j and accept
    improvements always, otherwise accept with a decaying simulated‑annealing
    probability.
    """
    target_sum = a.sum()          # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())   # length needed for a*a

    # annealing schedule
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-14:
            step += 1
            continue

        # transfer a random fraction (1 %‑5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # final sanity check – enforce the simplex constraint once more
    if best_seq is not None:
        target_final = math.sqrt(2.0 * best_seq.shape[0])
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq * target_final / best_seq.sum()
    else:
        # very unlikely fallback
        n = n_target
        target_final = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_final / n)

    print(
        f"[done] best score = {best_score:.6f}, final length = {best_seq.shape[0]}"
    )
    return best_seq.tolist()
