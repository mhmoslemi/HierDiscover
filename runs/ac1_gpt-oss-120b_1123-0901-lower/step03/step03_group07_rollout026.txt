def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Determine n: maybe 2048 or 4096; we pick 2048 for speed.
    n = 2048

    # Precompute FFT size: next power of two >= 2n-1
    fft_size = 1 << ((2 * n - 1).bit_length())

    # Precompute other constants for Adam
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.06

    best_val = float('inf')
    best_seq = None

    restart = 0
    while time.time() < deadline:
        restart += 1

        # choose mode for initialization
        mode = restart % 4
        if mode == 0:
            a = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            r = np.random.uniform(0.7, 0.99)
            a = np.power(r, np.arange(n, dtype=np.float64))
            a = a / a.sum()
        elif mode == 2:
            i = np.arange(n, dtype=np.float64)
            a = 1.0 - np.sin(np.pi * i / (n - 1))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            # start from global best if available and compatible length
            try:
                seq = height_sequence_1
                if isinstance(seq, list) and len(seq) >= n:
                    # take first n values and renormalize
                    a = np.array(seq[:n], dtype=np.float64)
                else:
                    # fallback uniform
                    a = np.full(n, 1.0 / n)
            except NameError:
                a = np.full(n, 1.0 / n)

        # ensure positivity
        a = np.maximum(a, 1e-12)
        u = np.log(a)

        m = np.zeros_like(u)
        v = np.zeros_like(u)

        max_iter = 25000
        last_improve = time.time()
        it = 0
        while it < max_iter and time.time() < deadline:
            it += 1

            # Compute probabilities from logits
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a_sum = a.sum()
            a = a / a_sum

            # Compute full convolution via FFT
            # Use np.fft.rfft
            fa = np.fft.rfft(a, fft_size)
            conv = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
            kmax = int(conv.argmax())
            max_conv = conv[kmax]

            # gradient w.r.t a for the max convolution term
            # compute shift
            grad_a = np.zeros_like(a)
            start = max(0, kmax - (n - 1))
            end = min(kmax, n - 1)
            if start <= end:
                idx = np.arange(start, end + 1)
                grad_a[idx] = 2.0 * a[kmax - idx]

            # convert to gradient w.r.t logits (softmax Jacobian)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = base_lr * (0.9995 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # periodic evaluation every 50 iters
            if it % 50 == 0:
                # compute objective using existing conv max and sum
                val = 2.0 * n * max_conv / (a_sum * a_sum)   # same as _objective
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d} new best = {val:.6f}")

            # shake if stagnating (no improvement > xx seconds)
            if time.time() - last_improve > 1.0:
                # add small random perturbation to logits
                u += (np.random.rand(n) - 0.5) * 1e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        # after inner loop, ensure we have candidate
        if best_seq is not None:
            print(f"[Restart {restart}] finished it={it}, best so far = {best_val:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()

"""
Propose a step‑function that minimizes

    F(a) = 2·n·max_k (a * a)[k] / (∑ a_i)²

using a hard‑max Adam optimiser.  The method works directly on the
simplex (log‑space parametrisation) and at each iteration computes the
exact self‑convolution via FFT, finds the index where the convolution
attains its maximum and uses the corresponding sub‑gradient
∇ₐ maxₖ(a*a)[k] = 2·a_{k‑i}.  Adam with a decaying learning‑rate performs
the updates, occasional “shakes’’ avoid stagnation and several
different initial families (Dirichlet, exponential, U‑shaped and the
previous global best, if available) give diverse starting points.
Running until the time budget expires yields the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _hardmax_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the maximal convolution entry w.r.t. the vector ``a``.
    For the argmax index k we have
        (a*a)[k] = Σ_i a_i a_{k-i},
    and ∂/∂a_i = 2·a_{k-i} (0 if the index is out of range).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.
    The optimiser runs until the time budget is exhausted and keeps the
    best sequence seen across a few diversified starts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3          # safety margin

    # ------------------------------------------------------------------
    # Length of the step function.
    # 4096 is a power of two → fast FFT, still large enough to be expressive.
    n = 4096

    # FFT length (next power of two ≥ 2·n‑1) for the linear self‑convolution.
    fft_size = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.07          # a bit larger than the previous code

    # ------------------------------------------------------------------
    # Build a few initial families (including the previous best if present)
    init_vectors = []

        init_vectors.append(np.full(n, 1.0 / n))

    # 2) Dirichlet (random simplex point)
    init_vectors.append(np.random.dirichlet(np.ones(n)))

    i = np.arange(n, dtype=np.float64)
    ushape = 1.0 - np.sin(np.pi * i / (n - 1))

    return best_seq.tolist()
