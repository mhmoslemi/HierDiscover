Algorithm
---------
1.  Exploration phase (cheap):

      is kept.

2.  Refinement phase (gradient‑based):
    * Starting from the best candidate of each length run a soft‑max‑Adam
      optimiser that minimises a smooth approximation of the maximal
      self‑convolution.
    * The gradient of the smoothed max is obtained analytically via a
      single weighted reverse convolution (FFT).  Adam updates are projected
      onto the feasible set (non‑negative, Σa = √(2 n)).
    * Temperature τ is annealed exponentially; occasional Gaussian jitter
      helps escape plateaus.
    * The optimiser stops when the global time budget (≈ 1000 s) is exhausted
      or the iteration limit is reached.

3.  The globally best sequence (lowest exact objective) is returned as a
    plain Python list.

All heavy work uses NumPy FFTs; the code respects the 1000 s wall‑clock
limit and does not perform any I/O.
"""

import time
import math
import numpy as np


def _objective(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence (without safety checks)."""
    n = seq.shape[0]
    # next power‑of‑two for convolution
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(seq, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    max_conv = float(conv.max())
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values; returns (weights, smooth max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k * a_{k-i}   (indices outside range are ignored)
    using FFTs.  ``a`` and ``w`` are zero‑padded to ``Nfft`` beforehand.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]          # keep valid part
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int = 1_000_000,
):
    """
    Adam optimiser on the smoothed maximal self‑convolution.
    Returns the (possibly) improved best_seq / best_score.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    # temperature schedule parameters
    tau_factor = 0.15          # start with a smaller temperature than the previous run
    decay_rate = 15000.0      # faster cooling

    # FFT length (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and iteration < max_iters:
        # 1) exact convolution via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # 2) smooth max
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # 3) weighted reverse convolution (gradient of smooth max)
        rev = _weighted_rev(a, w, Nfft)

        # 4) gradient of the (smooth) objective respecting the L1‑norm
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection onto feasible set (a≥0, Σa = √(2 n))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # 6) jitter occasionally
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # 7) exact objective (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as an initial reference.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        n0 = base.shape[0]
        target_sum0 = math.sqrt(2.0 * n0)
        base = base * (target_sum0 / base.sum())
        sc = _objective(base)
        best_seq = base.copy()
        best_score = sc
        print(f"[init] using height_sequence_1 length={n0}, score={sc:.6f}")
    except Exception:
        pass  # no prior sequence – we start from scratch

    # ------------------------------------------------------------------
    # Lengths to test (keep them modest for speed, but include a larger one).
    lengths = [500, 1000, 2000, 4000, 8000]
    lengths = [l for l in lengths if l >= 300]   # safety

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        print(f"\n[phase 1] length = {n}")

        # ------- 1) parametric grid search ---------------------------------
        # exponential decay a_i ∝ r**i
        r_vals = np.linspace(0.97, 0.9999, 15)
        for r in r_vals:
            shape = r ** np.arange(n, dtype=float)
            shape = shape / shape.sum() * target_sum
            sc = _objective(shape)
            if sc < best_score:
                best_score = sc
                best_seq = shape.copy()
                print(f"  exp r={r:.5f}  -> {sc:.6f}")

        # power‑law decay a_i ∝ (1 - i/(n-1))**p
        p_vals = np.linspace(0.5, 10.0, 15)
        idx = np.arange(n, dtype=float)
        for p in p_vals:
            shape = np.power(1.0 - idx / (n - 1), p)
            shape = shape / shape.sum() * target_sum
            sc = _objective(shape)
            if sc < best_score:
                best_score = sc
                best_seq = shape.copy()
                print(f"  pow p={p:.2f}  -> {sc:.6f}")

        # short plateau (first k entries uniform, rest zero)
        for k_frac in np.linspace(0.05, 0.5, 8):
            k = max(2, int(k_frac * n))
            shape = np.zeros(n, dtype=float)
            shape[:k] = 1.0 / k
            shape = shape * target_sum          # already sums to target_sum
            sc = _objective(shape)
            if sc < best_score:
                best_score = sc
                best_seq = shape.copy()
                print(f"  plateau k={k} -> {sc:.6f}")

        # few random vectors
        for _ in range(5):
            shape = np.random.rand(n)
            shape = shape / shape.sum() * target_sum
            sc = _objective(shape)
            if sc < best_score:
                best_score = sc
                best_seq = shape.copy()
                print(f"  random -> {sc:.6f}")

        # ------- 2) local refinement with Adam -----------------------------
        if best_seq is None:
            # fallback to uniform if nothing was generated (should not happen)
            best_seq = np.full(n, target_sum / n)
            best_score = _objective(best_seq)

        # allocate iteration budget based on size (larger n → fewer iters)
        if n <= 2000:
            max_iters = 2_000_000
        elif n <= 4000:
            max_iters = 1_000_000
        else:
            max_iters = 500_000

        # run the optimiser (it respects the global deadline)
        best_seq, best_score = _optimize_length(
            n=n,
            init_a=best_seq,
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
            max_iters=max_iters,
        )

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found.
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _objective(best_seq)

    print(f"\n[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
