def _optimise_one_length_adam(...):
    # parameters
    target_sum = sqrt(2*n)
    a = _project_onto_simplex(init, target_sum)
    best_seq = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    while time.time()<run_deadline:
        t_iter += 1
        # compute conv, max
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # compute softmax weight with tau schedule (similar)
        tau = max(0.1*max_val * (0.99**t_iter), 0.001*max_val)
        w = np.exp((conv - max_val)/tau)
        w = w / w.sum()
        # gradient
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]
        # Adam updates
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)  # learning rate tuned
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        # compute score
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum**2)
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n} new best {best_score:.6f}")
        a = a_new
        # occasional random shake
        if np.random.rand() < 0.005:
            noise = np.random.randn(n) * target_sum * 0.001 / n
            a = _project_onto_simplex(np.maximum(a+noise,0.0), target_sum)
```

    a = _project_onto_simplex(init, target_sum)
    best_score = evaluate_sequence(a.tolist())

"""
Hybrid optimizer for the non‑negative step‑function problem.

The routine combines two complementary strategies:

1. **Adam‑style soft‑max gradient descent** – a smooth approximation of
   the max‑convolution is used to obtain a dense gradient, which is then
   updated with Adam momentum and projected onto the simplex
   Σ a_i = √(2 n).  This quickly finds good local minima for many
   sequence lengths.

2. **LP‑based Frank‑Wolfe step** – for a given sequence f we compute
   M = max(conv(f,f)).  A linear program maximises Σ g_i subject to
   conv(f,g) ≤ M (element‑wise) and g ≥ 0.  After scaling g to the
   required ℓ₁‑norm we move a little towards g:
   f ← (1‑t) f + t g.  Repeating this step strictly reduces the
   objective and often yields a sharper improvement than pure gradient
   descent.

The outer loop performs many random restarts with different lengths
and initial shapes (power‑law, optionally tapered).  Occasionally the
current global best is warm‑started.  The best sequence seen before the
budget expires is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives from rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Soft‑max (Adam) gradient descent for a fixed length
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth‑max approximation of
    max(conv(a,a)).  Returns the best sequence found and its exact score
    (via ``evaluate_sequence``).
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    while time.time() < run_deadline:
        # ------------------------------------------------------------------
        # 1) self‑convolution and current maximum
        # ------------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # ------------------------------------------------------------------
        # 2) smooth‑max weights (soft‑max) – temperature τ decays smoothly
        # ------------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (0.99 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau          # shift for stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # 3) gradient of Σ w_k conv_k  (cross‑correlation)
        # ------------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                  # mapping to Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # ------------------------------------------------------------------
        # 4) Adam update on the simplex (followed by projection)
        # ------------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # ------------------------------------------------------------------
        # 5) exact evaluation & bookkeeping
        # ------------------------------------------------------------------
        score_new = evaluate_sequence(a_new.tolist())
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n} Adam new best = {best_score:.6f}")

        a = a_new

        # occasional shake‑up to get out of plateaus
        if np.random.rand() < 0.005:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming (Frank‑Wolfe / alternating) optimisation for a fixed length
# ----------------------------------------------------------------------
def _optimise_one_length_lp(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    t_initial: float = 0.5,
    max_iter: int = 20,
) -> tuple[np.ndarray, float]:
    """
    Alternating LP + mixing scheme.
    Starts from ``init`` (projected onto the simplex) and repeatedly:
        * compute M = max(conv(f,f))
        * solve LP  max Σ g_i   s.t. conv(f,g) ≤ M , g ≥ 0
        * scale g to the required ℓ₁‑norm and mix f ← (1‑t) f + t g
    The mixing factor t is gradually reduced.
    Returns the best sequence encountered and its exact score.
    """
    target_sum = np.sqrt(2.0 * n)
    f = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = f.copy()
    best_score = evaluate_sequence(f.tolist())

    t = t_initial
    iteration = 0

    # Pre‑allocate the constraint matrix (dense – acceptable for n ≤ ~2000)
    # A[k, i] = f_i   if i ≤ k < i+n   else 0
    while time.time() < run_deadline and iteration < max_iter:
        # 1) self‑convolution → current bound M
        conv_f = _full_convolution_fft(f, f)
        M = conv_f.max()

        # 2) build the (2n‑1)×n constraint matrix
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        # each column i contributes f over rows i … i+n‑1
        for i in range(n):
            A[i:i + n, i] = f

        # 3) linear programme: maximise Σ g_i  ⇔ minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b = np.full(rows, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            # LP failed – fall back to gradient descent
            break

        g0 = res.x
        s_g0 = g0.sum()
        if s_g0 <= 0:
            break

        # 4) rescale to the required ℓ₁‑norm
        g = g0 * (target_sum / s_g0)

        # 5) mixing step
        f = (1.0 - t) * f + t * g
        # (convex combination of two feasible points keeps the sum = target_sum)

        # 6) evaluate
        score_f = evaluate_sequence(f.tolist())
        if score_f < best_score:
            best_score = score_f
            best_seq = f.copy()
            print(f"[{time.time():.2f}] n={n} LP iter={iteration} new best = {best_score:.6f}")

        # tighten mixing factor a little each iteration
        t = max(0.01, t * 0.92)
        iteration += 1

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if available.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        s = evaluate_sequence(hs_arr.tolist())
        if s < best_score:
            best_score = s
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start (global) score = {best_score:.6f}")
    except Exception:
        pass  # no pre‑existing sequence

    # ------------------------------------------------------------------
    # Main restart loop – each restart receives a slice of the remaining time.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # pick a problem size – early on we explore a wide interval,
        # later we concentrate on moderate sizes to finish quickly.
        if remaining > 300:
            n = np.random.randint(800, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(600, 3000 + 1)
        else:
            n = np.random.randint(400, 2200 + 1)

        # ------------------- create an initial shape --------------------
        p = np.random.uniform(0.3, 0.7)                # power‑law exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)

        # optional exponential tail for added flexibility
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.4, 2.0) * n
            init *= np.exp(-idx / tau_exp)

        init += np.random.rand(n) * 1e-4               # tiny jitter

        # occasional warm‑start from the overall best
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                init = np.concatenate([bs, pad])

        # ----------------------------------------------------------------
        # allocate a time slice for this restart
        # ----------------------------------------------------------------
        slice_len = min(8.0, remaining * 0.5)          # never exceed the global deadline
        run_deadline = min(deadline, time.time() + slice_len)

        # ----------------------------------------------------------------
        # Choose optimisation strategy
        # ----------------------------------------------------------------
        use_lp = (n <= 2000) and (np.random.rand() < 0.45)   # LP is cheap for moderate n
        if use_lp:
            seq_local, score_local = _optimise_one_length_lp(
                n,
                init,
                run_deadline,
                t_initial=0.5,
                max_iter=12,
            )
        else:
            seq_local, score_local = _optimise_one_length_adam(
                n,
                init,
                run_deadline,
                init_tau_factor=0.12,
                min_tau_factor=0.001,
                lr=0.05,
                beta1=0.9,
                beta2=0.999,
                epsilon=1e-8,
            )

        # ----------------------------------------------------------------
        # Update the global best if we improved
        # ----------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # early exit if we comfortably beat the target bound
            if best_score < 1.49:
                break

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (unlikely) return a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        s_target = np.sqrt(2.0 * n)
        best_seq = np.full(n, s_target / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
