def smooth_gradient_descent(a_init, target_sum, max_time, start_time):
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_val = _max_conv(a)   # actual max conv (not smoothed)
    # Choose initial smoothing alpha (e.g., 5) and increase gradually.
    alpha = 5.0
    alpha_inc = 1.5  # increase each iteration?
    lr = 0.1
    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        # compute softmax weights
        # For numerical stability subtract max_val
        exp_vals = np.exp(alpha * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        w_rev = w[::-1]
        # convolution of a and w_rev
        # compute FFT length
        L = 1 << ((len(a) + len(w_rev) - 1).bit_length())
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w_rev, n=L)
        conv_rev = np.fft.irfft(A * W, n=L)[:len(a) + len(w_rev) - 1]
        grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]
        # step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        # maybe adapt lr
        a = a_new
        # keep best based on true max conv
        cur_val = conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SGD new best = {best_val:.6f} (n={n})")
        # schedule: increase alpha gradually to focus on max; reduce lr maybe
        alpha = min(100.0, alpha * 1.02)
        lr *= 0.9999
    return best_a, best_val

    start = time.time()

    best_seq = None

    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    # maybe choose random lengths each iteration
    while time.time() < deadline:

        target_sum = np.sqrt(2.0 * n)
        # generate seed

            # use previous best of maybe different size by resizing
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size > n:

"""
Propose a non‑negative step function (list of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation function normalises the sequence to Σa = √(2n), therefore the
objective reduces to `max(conv(a,a))`.  The algorithm works in three stages:

1. **Diverse initialisations** – random “U‑shaped’’ seeds, noisy power‑law
   shapes and, when available, the best sequence found so far (global variable
   ``height_sequence_1``).  Each seed is projected onto the simplex
   {x_i ≥ 0, Σx_i = √(2n)}.

2. **Smooth gradient descent** – we minimise a smoothed version of the
   max‑convolution using a soft‑max (log‑sum‑exp) with a gradually increasing
   temperature.  The gradient of the smoothed objective can be written as a
   convolution, so an FFT gives an O(n log n) update.  After each step the
   vector is projected back onto the simplex.

3. **Refinement** – a fast projected sub‑gradient method (the one used in the
   previous solution) followed optionally by a short simulated‑annealing run
   to escape shallow local minima.

The outer loop restarts the whole pipeline with different lengths (≈1500‑5000)
and different seed parameters, always keeping the best candidate found so far.
The total running time is limited to the supplied budget (default 1000 s).

The function ``propose_candidate`` returns a plain Python list of floats ready
to be passed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s`` (non‑negative, sum=s)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _conv_full_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors via FFT (full length)."""
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate (the part we actually need – the max of the self‑convolution).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a, a)) – the objective when Σa = √(2n)."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass transfers between two positions.
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Simple SA that respects the simplex; returns best vector seen."""
    a = a_init.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    # temperature schedule (5 % of the current value as start)
    T0 = max(0.01, cur_score * 0.05)
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        # move a random fraction (up to 50 %) of a[i] from i to j
        frac = np.random.rand() * 0.5
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        if delta_score <= 0.0 or np.random.rand() < math.exp(-delta_score / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the (hard) max‑convolution (as before).
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Refine a vector using a sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # active (max‑tying) indices
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Smooth (log‑sum‑exp) gradient descent.
# --------------------------------------------------------------------------- #
def smooth_gradient_descent(a_init: np.ndarray,
                            target_sum: float,
                            max_time: float,
                            start_time: float):
    """
    Minimises a smoothed max‑convolution using a soft‑max with increasing
    temperature.  Gradient is obtained via a convolution, allowing an
    O(n log n) update.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    # start with a moderate smoothing; it will be increased slowly.
    alpha = 5.0          # temperature inverse
    lr = 0.12            # base learning rate

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)                # length 2n‑1
        max_val = conv.max()

        # soft‑max weights (log‑sum‑exp gradient)
        exp_vals = np.exp(alpha * (conv - max_val))
        w = exp_vals / exp_vals.sum()           # sums to 1

        # gradient = 2 * convolution( a , w_rev ) evaluated at appropriate offsets
        w_rev = w[::-1]
        conv_rev = _conv_full_fft_two(a, w_rev)   # length 3n‑2
        grad = 2.0 * conv_rev[(2 * n - 2) - np.arange(n)]

        # gradient step + projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new
        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] SGD new best = {best_score:.6f} (n={n})")

        # schedule: slowly sharpen the soft‑max and decay learning rate
        alpha = min(120.0, alpha * 1.025)
        lr *= 0.9997   # very gentle decay

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")                # this holds max(conv)

    # ------------------------------------------------------------------- #
    # Helper to evaluate a candidate (fast – we already keep Σa = √(2n)).
    # ------------------------------------------------------------------- #
    def _evaluate(a):
        return _max_conv(a)

    # ------------------------------------------------------------------- #
    # Outer loop – many restarts with varying lengths and seeds.
    # ------------------------------------------------------------------- #
    # Possible lengths (larger lengths give more freedom, but cost more time)
    length_pool = [1500, 1800, 2100, 2400, 2700, 3000, 3400, 3800, 4200, 4600, 5000]

    while time.time() < deadline:
        # Choose length (sometimes reuse current best length to refine)
        if best_seq is not None and np.random.rand() < 0.18:
            n = len(best_seq)
        else:
            n = int(np.random.choice(length_pool))

        target_sum = math.sqrt(2.0 * n)      # Σ a_i = √(2n)

        # --------------------------------------------------------------- #
        # Initialise a candidate.
        # --------------------------------------------------------------- #
        use_old = False
        if "height_sequence_1" in globals() and np.random.rand() < 0.45:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
                use_old = True
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
                use_old = True
        if not use_old:
            # Randomised power‑law “U‑shape’’ with small noise.
            p = np.random.uniform(0.4, 2.2)            # exponent
            shift = np.random.uniform(0.0, 4.0)        # shift to avoid singularity
            idx = np.arange(n, dtype=float)
            init = (1.0 / (idx + 1 + shift) ** p) + (1.0 / (n - idx + shift) ** p)
            init += np.random.randn(n) * (0.005 * init.mean())
            init = np.maximum(init, 0.0)

        a = _project_onto_simplex(init.astype(float), target_sum)

        cur_score = _evaluate(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Allocate time for this restart: split between smooth GD and refinement.
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        # Roughly half the remaining time (but at most 12 s) for smooth GD.
        phase_time = min(12.0, time_left * 0.5)

        # ---- Smooth gradient descent (soft‑max) ---------------------- #
        a_sgd, val_sgd = smooth_gradient_descent(a, target_sum,
                                                 phase_time, time.time())
        if val_sgd < best_score:
            best_score = val_sgd
            best_seq = a_sgd.copy()
            print(f"[{time.time():.2f}] After SGD best = {best_score:.6f} (n={n})")

        # ---- Sub‑gradient refinement --------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left)   # up to 6 s for the classic GD

        a_gd, val_gd = subgradient_optimize(a_sgd, target_sum,
                                            phase_time, time.time())
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD best = {best_score:.6f} (n={n})")

        # ---- Optional short simulated‑annealing for final shake ------- #
        time_left = deadline - time.time()
        if time_left > 2.0:
            phase_time = min(3.0, time_left)
            a_sa, val_sa = simulated_annealing(best_seq, target_sum,
                                               phase_time, time.time())
            if val_sa < best_score:
                best_score = val_sa
                best_seq = a_sa.copy()
                print(f"[{time.time():.2f}] After SA best = {best_score:.6f} (n={n})")

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
