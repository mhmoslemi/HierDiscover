def lp_step(f, M, target_sum):
    n = len(f)
    # Determine active conv entries (within tolerance)
    conv_f = _conv_full_fft(f)
    max_val = conv_f.max()
    tol = max_val * 1e-8  # small relative tolerance
    active_idx = np.where(conv_f >= max_val - tol)[0]
    if active_idx.size == 0:
        return None, max_val
    # Build sparse matrix A_ub
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # for j ranging low..high: f[k-j] coefficient for g_j
        # use vectorized maybe
        js = np.arange(low, high+1, dtype=int)
        coeffs = f[k - js]  # f_i where i = k - j
        rows.extend([row]*len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), max_val)  # M same as max_val
    c = -np.ones(n)  # maximize sum
    bounds = [(0, None)] * n
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, max_val
    sum_g = -res.fun
    if sum_g <= target_sum:
        # cannot improve
        return None, max_val
    # scaling factor
    lam = target_sum / sum_g
    g = lam * res.x
    # Combine f and g using some t
    t = 0.5  # can be tuned
    new_f = (1 - t) * f + t * g
    # Ensure feasibility: project onto simplex
    new_f = _project_onto_simplex(new_f, target_sum)
    new_score = _max_conv(new_f)
    return new_f, new_score

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

def project_onto_simplex(v, s):
    if s < 0:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def conv_full_fft(a):
    n = a.shape[0]
    n_fft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:2*n - 1]
    return conv

def max_conv(a):
    return float(conv_full_fft(a).max())

def lp_refine(f, target_sum, cur_score, time_deadline):
    # attempt to improve f using LP based on active max conv entries
    n = f.shape[0]
    while time.time() < time_deadline:
        conv_f = conv_full_fft(f)
        M = conv_f.max()
        # active indices
        tol = 1e-12 * M if M > 0 else 1e-12
        active = np.where(conv_f >= M - tol)[0]
        if active.size == 0:
            break
        # Build sparse matrix
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active):
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            js = np.arange(low, high + 1, dtype=int)
            coeffs = f[k - js]  # f_i where i = k - j
            rows.extend([row_idx] * len(js))
            cols.extend(js.tolist())
            data.extend(coeffs.tolist())
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M, dtype=float)
        c = -np.ones(n, dtype=float)  # maximize sum(g)
        bounds = [(0, None)] * n
        # Solve LP
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            break
        if not res.success:
            break
        sum_g = -res.fun
        if sum_g <= target_sum * (1 + 1e-9):
            # No potential improvement
            break
        # scaling factor
        lam = target_sum / sum_g
        g = lam * res.x
        # try a few mixing coefficients
        best_local_score = max_conv(f)
        best_local = f
        for alpha in (0.2, 0.4, 0.6, 0.8):
            cand = (1 - alpha) * f + alpha * g
            cand = project_onto_simplex(cand, target_sum)
            cand_score = max_conv(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local = cand
        if best_local_score < cur_score - 1e-9:
            f = best_local
            cur_score = best_local_score
        else:
            # No improvement, exit loop
            break
    return f, cur_score

def subgradient_optimize(a_init, target_sum, max_time, start_time):
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = max_conv(a)
    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    
    while time.time() < start_time + max_time:
        conv = conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        max_grad = grad.max()
        if max_grad <= 0.0:
            break
        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9
        a = a_new
        base_lr *= 0.99995
    return best_vec, best_score

"""
Hybrid LP+subgradient optimizer for min max convolution of a nonnegative
step function under fixed L1 norm.  The algorithm repeatedly draws a
candidate length, builds a U‑shaped start, improves it by a small
linear‑programming step that respects the current maximal convolution,
then refines the result with a projected sub‑gradient descent.  The
best sequence found before the time budget expires is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >=0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv

def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_full_fft(a).max())

def _lp_refine(f: np.ndarray, target_sum: float,
               cur_score: float, deadline: float):
    """Iteratively improve f by solving a tiny LP on the active constraints."""
    n = f.shape[0]
    # keep track of the best score encountered inside this refinement
    best_f = f.copy()
    best_score = cur_score
    while time.time() < deadline:
        conv_f = _conv_full_fft(f)
        M = conv_f.max()
        # choose an active set – indices whose value is within relative tolerance of the max
        rel_tol = 1e-12
        tol = max(rel_tol * M, 1e-12)
        active = np.where(conv_f >= M - tol)[0]
        if active.size == 0:
            break
        # build sparse matrix A_ub such that A_ub @ g <= M encodes (f*g)_k <= M for active k
        rows = []
        cols = []
        data = []
        for r, k in enumerate(active):
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            js = np.arange(low, high + 1, dtype=int)
            coeffs = f[k - js]                 # coefficient of g_j in (f*g)_k
            rows.extend([r] * len(js))
            cols.extend(js.tolist())
            data.extend(coeffs.tolist())
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M, dtype=float)
        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success:
            break
        sum_g = -res.fun
        if sum_g <= target_sum * (1 + 1e-9):
            # we cannot increase the total mass, stop.
            break
        # scaling factor to bring the sum back to target_sum
        lam = target_sum / sum_g
        g = lam * res.x
        # try a few convex combinations and keep the best
        candidate = f
        candidate_score = M
        for alpha in (0.2, 0.4, 0.6, 0.8):
            cand = (1 - alpha) * f + alpha * g
            cand = _project_onto_simplex(cand, target_sum)
            cand_score = _max_conv(cand)
            if cand_score < candidate_score - 1e-12:
                candidate = cand
                candidate_score = cand_score
        if candidate_score < best_score - 1e-12:
            best_score = candidate_score
            best_f = candidate.copy()
        # if we made progress, continue from the new point; otherwise exit.
        if candidate_score < M - 1e-12:
            f = candidate
        else:
            break
    return best_f, best_score

def _subgradient_optimize(a_init: np.ndarray,
                          target_sum: float,
                          max_time: float,
                          start_time: float):
    """Projected sub‑gradient descent on the exact L∞‑objective."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)
    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")
        # active set – all indices whose value is within machine epsilon of max
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        grad = np.zeros_like(a)
        # the sub‑gradient of sum_{k∈active} conv_k
        for k in active:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        max_grad = grad.max()
        if max_grad <= 0.0:
            break
        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        # occasional random shake to escape shallow minima
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9
        a = a_new
        base_lr *= 0.99995   # gentle decay
    return best_vec, best_score

"""
Hybrid LP + sub‑gradient optimizer.

The algorithm repeatedly picks a length `n`, builds a simple U‑shaped start,
projects it onto the simplex Σa = √(2 n) and then performs two refinement
stages:

1. **LP refinement** – using the currently active (maximal) convolution
   indices a tiny linear program is solved:
   maximise Σ gᵢ  subject to gᵢ ≥ 0,
   (f * g)ₖ ≤ M for every active k, where M = max(f * f).
   The solution g is scaled back to Σg = √(2 n) and mixed with the
   current vector `f`.  The convex combination with the best reduction of
   the maximum convolution is kept.  This step mimics the linear‑programming
   update described in the state‑of‑the‑art method but works with a tiny
   active set, making it fast enough for many restarts.

2. **Projected sub‑gradient descent** – a fast local optimiser that
   directly reduces the L∞‑norm of the autocorrelation by following the
   sub‑gradient of the sum of the active convolution entries, with heavy‑ball
   momentum and occasional random shakes.

Finally a tiny hill‑climbing phase is applied to squeeze a little more
out of the time budget.  The best sequence found before the overall time
budget expires is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast full convolution via a real FFT.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the autocorrelation a*a."""
    return float(_conv_full_fft(a).max())

# --------------------------------------------------------------------------- #

               cur_score: float, deadline: float):

    n = f.shape[0]

        active = np.where(conv_f >= M - tol)[0]
        if active.size == 0:
            break

        rows, cols, data = [], [], []
        for r, k in enumerate(active):

            rows.extend([r] * len(js))

                         bounds=bounds, method='highs', options={'presolve': True})

            break

            # No chance to increase total mass → stop.
            break

        # Scale back to satisfy the L1‑norm constraint.
        lam = target_sum / sum_g
        g = lam * res.x

        # Try a few convex combinations and keep the best.
        candidate = f
        candidate_score = M
        for alpha in (0.2, 0.4, 0.6, 0.8):
            cand = (1 - alpha) * f + alpha * g
            cand = _project_onto_simplex(cand, target_sum)
            cand_score = _max_conv(cand)
            if cand_score < candidate_score - 1e-12:
                candidate = cand
                candidate_score = cand_score

        if candidate_score < best_score - 1e-12:
            best_score = candidate_score
            best_f = candidate.copy()

        # Continue from the improved point; otherwise stop.
        if candidate_score < M - 1e-12:
            f = candidate
        else:
            break

    return best_f, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the exact L∞ objective.
# --------------------------------------------------------------------------- #
def _subgradient_optimize(a_init: np.ndarray,
                          target_sum: float,
                          max_time: float,
                          start_time: float):
    """Refine a vector using a sub‑gradient method with momentum."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # active set – entries that tie the max (within machine epsilon)
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        # sub‑gradient of Σ_{k∈active} (a*a)_k
        for k in active:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake to escape shallow minima
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995   # gentle decay

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Tiny hill‑climbing (accept‑only‑improvements) for the very last seconds.
# --------------------------------------------------------------------------- #
def _hill_climb(a0: np.ndarray, target_sum: float, time_budget: float):
    """Simple random hill‑climber that only accepts strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)
    best = a.copy()
    best_score = cur_score
    deadline = time.time() + time_budget

    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        # move a random fraction (≤30 %) of mass from i to j
        delta = a[i] * np.random.rand() * 0.3
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, target_sum)
        cand_score = _max_conv(cand)
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
    return best, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """Search for a low‑value non‑negative step function within `budget_s` seconds."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5           # safety margin

    best_seq = None
    best_score = float("inf")

    # quick evaluator: because we keep Σa = √(2 n) the objective equals max(conv)
    def _quick_eval(a):
        return _max_conv(a)

    while time.time() < deadline:
        # Choose a length. Occasionally reuse the current best length.
        if best_seq is not None and np.random.rand() < 0.3:
            n = len(best_seq)
        else:
            n = np.random.randint(800, 2501)   # moderate size for speed + quality
        target_sum = np.sqrt(2.0 * n)

        # Initial candidate – either a known good seed or a noisy U‑shape
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)      # large at the ends, small in the middle
            init = np.random.rand(n) * weight

        a = _project_onto_simplex(init.astype(float), target_sum)

        cur_score = _quick_eval(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # ------------------- Phase 1 : LP refinement ------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left * 0.5)
        lp_deadline = time.time() + phase_time
        a_lp, score_lp = _lp_refine(a, target_sum, cur_score, lp_deadline)
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time():.2f}] After LP best = {best_score:.6f} (n={n})")

        # ------------------- Phase 2 : sub‑gradient descent ------------ #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        gd_time = min(6.0, time_left)
        a_gd, score_gd = _subgradient_optimize(a_lp, target_sum,
                                               gd_time, time.time())
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD best = {best_score:.6f} (n={n})")

        # ------------------- Phase 3 : final hill‑climb --------------- #
        time_left = deadline - time.time()
        if time_left > 1.0:   # give a few seconds to the cheap hill‑climber
            a_hc, score_hc = _hill_climb(a_gd, target_sum, time_left)
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.2f}] After HC best = {best_score:.6f} (n={n})")

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # Fallback – uniform vector (should never be needed).
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list, as required by the grader.
    return best_seq.tolist()
