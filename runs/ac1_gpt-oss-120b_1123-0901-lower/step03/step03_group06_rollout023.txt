"""
Propose a low‑peak self‑convolution step function.

The algorithm works on the probability vector ``p`` (∑p = 1) and
optimises it for a fixed length ``n`` using an exponentiated‑gradient
mirror‑descent with a soft‑max approximation of the max‑convolution.
A momentum term (Adam‑like) and occasional jitter help escape plateaus.
After a long EG run the best probability is refined by a simple
stochastic hill‑climber that moves a small amount of mass between two
coordinates.  All operations are performed with FFTs, so a single
iteration costs about O(n log n) and fits easily into the 1000 s budget.

The final sequence ``a = sqrt(2 n)·p`` is returned; its score equals the
maximum entry of its full auto‑convolution (the value returned by
``evaluate_sequence``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full linear self‑convolution (p∗p) using a single FFT of size N.
    Returns the real array of length 2·len(p)‑1.
    """
    P = np.fft.rfft(p, n=N)          # FFT of p, zero‑padded to N
    conv = np.fft.irfft(P * P, n=N)  # element‑wise product → convolution
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute  rev_i = Σ_k w_k·p_{k‑i}   (i = 0 … n‑1)
    This is the correlation of ``p`` with the weight vector ``w``.
    ``w`` must have length 2·n‑1.
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)            # FFT of p (size N)
    W_rev = np.fft.rfft(w[::-1], n=N)  # FFT of reversed weights
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]   # keep the linear part
    # slice that corresponds to the centred correlation and flip back
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _optimize_eg(p_start: np.ndarray,
                 n: int,
                 max_iters: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector (sum = 1) found and its
    objective value 2·n·max(p∗p).
    """
    # ------------------- hyper‑parameters --------------------------
    beta_start = 0.1                # start of soft‑max temperature
    beta_end = 60.0                 # final temperature (more extreme)
    beta_len = max_iters // 2       # linear schedule length
    jitter_interval = 4000
    jitter_scale = 0.003
    momentum_decay = 0.9            # classic momentum factor
    # --------------------------------------------------------------

    p = p_start.copy()
    logp = np.log(p + 1e-30)        # stay in log‑space for positivity
    N = _next_pow2(3 * n - 2)       # size sufficient for both convolutions

    # momentum buffer
    m_buf = np.zeros_like(p)

    best_score = float("inf")
    best_p = p.copy()

    for it in range(1, max_iters + 1):
        # time guard
        if time.time() >= deadline:
            break

        # ----- self‑convolution (p∗p) -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        # keep the best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[n {n:5d} iter {it:7d}] new local best = {best_score:.6f}")

        # ----- soft‑max weights -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))               # 0 < w_k ≤ 1

        # ----- weighted reverse correlation -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smoothed max -----
        grad = 4.0 * n * rev                         # ∇_p (2·n·M_β)

        # ----- momentum (Adam‑like) -----
        m_buf = momentum_decay * m_buf + (1.0 - momentum_decay) * grad

        # ----- step‑size (decays slowly) -----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # ----- EG update in log‑space -----
        logp -= eta * m_buf

        # ----- occasional jitter to escape plateaus -----
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        # (subtracting the max avoids overflow)
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


# ----------------------------------------------------------------------
def _local_refine(p: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    A cheap stochastic hill‑climber that moves a small amount of mass
    between two coordinates while the score improves.
    """
    # Pre‑compute the FFT size for the pure convolution (no weights)
    N = _next_pow2(2 * n - 1)

    def _score(pvec: np.ndarray) -> float:
        conv = _full_convolution_fft(pvec, N)
        return 2.0 * n * conv.max()

    best_p = p.copy()
    best_sc = _score(best_p)

    # step size relative to the current mass at the source index
    rel_step = 0.02
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a random fraction (±rel_step) of the source mass
        eps = (np.random.rand() * 2.0 - 1.0) * rel_step
        delta = eps * best_p[i]
        if delta > best_p[i] or best_p[i] - delta < 0:
            continue
        cand = best_p.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_p = cand
        # slowly shrink the step size to fine‑tune later on
        it += 1
        if it % 2000 == 0:
            rel_step *= 0.95
            rel_step = max(rel_step, 1e-4)

    return best_p, best_sc


# ----------------------------------------------------------------------

    The method starts from an existing construction (if available), runs a

        best_score = evaluate_sequence(best_seq.tolist())

    return best_seq.tolist()
