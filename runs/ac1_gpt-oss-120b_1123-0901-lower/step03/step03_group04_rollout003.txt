import time, math, numpy as np

def _conv_full_fft(a):
    n = a.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)
    return conv[:2*n-1]

def _smooth_grad(a, w):
    # a: shape (n,)
    # w: shape (2*n-1,) sum(w)=1
    n = a.shape[0]
    a_rev = a[::-1]
    L = len(w) + n - 1   # = 3n-2
    N = 1 << ((L - 1).bit_length())
    W = np.fft.rfft(w, n=N)
    A_rev = np.fft.rfft(a_rev, n=N)
    conv_rev = np.fft.irfft(W * A_rev, n=N)
    grad = 2.0 * conv_rev[n-1 : n-1 + n]
    return grad

def _adam_optimize_smooth(init_a, target_sum, deadline, max_iters=5000,
                          lr0=0.2, beta1=0.9, beta2=0.999, eps=1e-8,
                          alpha_start=0.1, alpha_end=20.0):
    a = np.maximum(init_a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    conv = _conv_full_fft(a)
    best_val = conv.max()
    n = a.shape[0]

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # convolution a*a
        conv = _conv_full_fft(a)
        M = conv.max()
        # schedule temperature (smoothing)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # compute softmax weights
        # to improve stability subtract M
        w = np.exp(alpha * (conv - M))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of smooth max
        grad = _smooth_grad(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # learning rate decay
        lr = lr0 / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex (non‑negative + fixed L1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # remember best (according to true max)
        cur_val = M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
    return best_a, best_val

def _local_search(a, target_sum, steps, deadline):
    n = a.shape[0]
    best = a.copy()
    best_val = _conv_full_fft(best).max()
    for step in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer small proportion from i to j
        delta = best[i] * 0.02
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection not needed because sum unchanged and non‑negative (i may become negative)
        # ensure non‑negative
        if cand[i] < 0:
            # reflect the deficit back to j
            cand[j] += cand[i]  # cand[i] negative, adds negative
            cand[i] = 0.0
        # after adjust, ensure sum still target (it will)
        cur_val = _conv_full_fft(cand).max()
        if cur_val < best_val:
            best = cand
            best_val = cur_val
    return best, best_val

"""
Propose a non‑negative step‑function (list of heights) with a low
``evaluate_sequence`` value.

The algorithm:
 * Try several problem sizes (original length and a few larger multiples).
 * For each size run a smooth‑maximum Adam optimiser.
   – Instead of the hard max we minimise the soft‑max
     (log‑sum‑exp) of the autocorrelation.  The smoothing parameter α is
     increased from a very gentle value (≈0.1) to a strong one (≈20),
     allowing the optimiser to first move in a smooth landscape and then
     focus on the true peak.
   – The gradient of the soft‑max is obtained analytically:
       ∂/∂a_i  1/α·log∑_k e^{α·c_k} = 2·∑_k w_k·a_{k‑i},
     where w_k are the normalised soft‑max weights and c_k are the
     convolution entries.  Convolutions are evaluated with FFTs (O(n log n)).
   – After each Adam step we project onto the feasible set:
     non‑negative entries and ‖a‖₁ = √(2 n) (simple scaling).
 * Finish each run with a cheap stochastic local search that moves a
   small amount of mass between two random positions if it improves the
   true peak.
 * Keep the best sequence seen across all lengths and restarts and return
   it before the time budget (≈1000 s) expires.
"""

import time
import math
import numpy as np


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a power‑of‑two FFT (real input)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


def _smooth_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution.
    a : (n,)
    w : (2n-1,) – soft‑max weights, sum to 1.
    Returns grad_i = 2·∑_k w_k·a_{k‑i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]                              # reversed a
    L = w.shape[0] + n - 1                      # = 3n‑2
    N = 1 << ((L - 1).bit_length())             # FFT length
    W = np.fft.rfft(w, n=N)
    A_rev = np.fft.rfft(a_rev, n=N)
    conv_rev = np.fft.irfft(W * A_rev, n=N)
    # slice that corresponds to indices i = 0 … n‑1
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


def _adam_optimize_smooth(init_a: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 8000,
                          lr0: float = 0.2,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          alpha_start: float = 0.1,
                          alpha_end: float = 20.0):
    """
    Adam optimiser on the smooth‑max objective.
    Returns the best vector (according to the true peak) and its peak value.
    """
    a = np.maximum(init_a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _conv_full_fft(a).max()
    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and soft‑max weights --------------------------------
        conv = _conv_full_fft(a)
        M = conv.max()

        # linearly increase the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        w = np.exp(alpha * (conv - M))          # stability through subtraction of M
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth‑max -----------------------------------------
        grad = _smooth_grad(a, w)

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set (non‑negative, fixed L1) ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ----- store best (according to true max) -----------------------------
        cur_val = M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


def _local_search(a: np.ndarray,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Simple stochastic local search: move a tiny fraction of mass from one
    index to another whenever the move reduces the true peak of the autocorrelation.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _conv_full_fft(best).max()

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = best[i] * 0.02                    # 2 % of the source entry
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # keep the sum unchanged and stay non‑negative
        if cand[i] < 0:
            continue  # shouldn't happen because delta ≤ 2 % of a_i

        cur_val = _conv_full_fft(cand).max()
        if cur_val < best_val:
            best = cand
            best_val = cur_val

    return best, best_val
