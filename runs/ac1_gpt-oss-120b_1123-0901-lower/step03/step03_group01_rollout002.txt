def lp_improve(a, target_sum, t, deadline):
    n = a.shape[0]
    M = _max_conv(a)
    # build constraint matrix
    rows = 2*n-1
    # Build A as dense np array
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = a[i]   # thanks to broadcast
    # define cvxpy variable
    import cvxpy as cp
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    # optionally we may also enforce sum(b) >= 0? Not needed.
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    # Solve with ECOS and maybe a time limit using maximum iterations or solver options
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000) # default
    except Exception as e:
        return a   # if solves fails, return unchanged

    if b.value is None:
        return a

    b_opt = np.maximum(np.array(b.value).flatten(), 0.0)  # ensure nonneg
    sum_b = b_opt.sum()
    if sum_b < 1e-12: 
        return a
    g = b_opt * (target_sum / sum_b)
    # combine
    a_new = (1 - t) * a + t * g
    # project onto simplex to enforce sum precisely
    a_new = _project_onto_simplex(a_new, target_sum)
    return a_new

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three phases:

1. **Cheap parametric search** – random “U‑shaped’’ families
   a_i ∝ (i+1)^(-p) + (n-i)^(-p)  (power law) or a_i ∝ exp(-λ·dist(i))
   are generated for many lengths `n` and parameters (`p`, `λ`).  After
   normalising to the simplex ∑a = √(2n) the best candidate is kept.

2. **Local refinement** – the best vector is polished with a
   projected sub‑gradient descent (the same method used in the reference
   implementation) for the remaining time budget.

3. **Optional LP step** – for moderate sizes (n ≤ 1500) a single linear
   program (maximise ∑b  s.t.  a*b ≤ M, b ≥ 0) is solved with CVXPY.
   The resulting direction is mixed into the current vector and projected
   back onto the simplex.  This often reduces the maximal convolution a
   little further.

All steps respect the constraints a_i ≥ 0 and Σ a_i = √(2n); therefore the
objective equals `max(conv(a,a))`.  The code is written with only top‑level
functions, uses NumPy, SciPy (for FFT) and CVXPY when available, and prints
progress information.  The returned value is a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – Σ x_i = s , x_i ≥ 0
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full convolution using a real FFT (length 2*n‑1)
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two ≥ 2*n‑1 for efficiency
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Objective = maximal entry of the auto‑convolution (since Σa = √(2n))
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Parametric “U‑shape’’ generators
# --------------------------------------------------------------------------- #
def _make_power_shape(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return a symmetric power‑law shape a_i ∝ (i+1)^(-p)+(n-i)^(-p)."""
    idx = np.arange(n, dtype=float)
    w = (idx + 1) ** (-p) + (n - idx) ** (-p)
    # ensure strict positivity (helps the projection)
    w = np.maximum(w, 1e-12)
    return _project_onto_simplex(w, target_sum)


def _make_exp_shape(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Return a symmetric exponential shape a_i ∝ exp(-lam·dist(i))."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    w = np.exp(-lam * d.astype(float))
    w = np.maximum(w, 1e-12)
    return _project_onto_simplex(w, target_sum)


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass transfers
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Perform a short SA run (mass transfers) and return best vector."""
    a = a_init.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    T0 = max(0.01, cur_score * 0.05)   # start temperature
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        if delta_score <= 0.0 or np.random.rand() < math.exp(-delta_score / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f}")

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the active convolution entries
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Refine a vector using a projected sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f}")

        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta * momentum - eta * grad
        a_new = a + momentum

        # enforce feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Optional linear‑programming improvement (LP) – works for modest n
# --------------------------------------------------------------------------- #
def _lp_improve(a: np.ndarray, target_sum: float, mix: float, deadline: float) -> np.ndarray:
    """
    Solve   maximise   Σ b_j
    subject to   a * b ≤ M   (convolution bound) ,   b ≥ 0.
    Returns the convex combination (1‑mix)*a + mix*g, where g is the
    normalised LP solution.  If CVXPY is unavailable or n is too large,
    the function returns a unchanged.
    """
    n = a.shape[0]

    # Limit memory usage – only attempt LP for moderate n
    if n > 1500:
        return a

    try:
        import cvxpy as cp
    except Exception:
        return a

    M = _max_conv(a)

    # Build dense Toeplitz matrix A where (A b)_k = Σ_i a_i b_{k-i}
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a[i]

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        return a

    if b.value is None:
        return a

    b_opt = np.maximum(np.array(b.value).ravel(), 0.0)
    sum_b = b_opt.sum()
    if sum_b < 1e-12:
        return a

    # Normalise to the required simplex radius
    g = b_opt * (target_sum / sum_b)

    # Convex combination
    a_new = (1.0 - mix) * a + mix * g
    a_new = _project_onto_simplex(a_new, target_sum)
    return a_new
