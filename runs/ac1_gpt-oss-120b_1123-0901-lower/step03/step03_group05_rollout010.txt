def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    global height_sequence_1

    base_seq = np.array(height_sequence_1, dtype=float) if 'height_sequence_1' in globals() else np.ones(2100, dtype=float)
    # Compute base length n0
    n0 = len(base_seq)
    # Normalize base so sum = sqrt(2*n0)
    target_sum0 = math.sqrt(2*n0)
    base_seq = base_seq * (target_sum0 / base_seq.sum())
    best_seq = base_seq.copy()
    best_score = compute_score(best_seq)
    print(f"[init] length={n0} score={best_score:.6f}")

    # Additional candidate families: generate some param families for various lengths
    length_factors = [0.8, 1.0, 1.2, 1.4, 1.6, 2.0] # add 0.8 (smaller)
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [l for l in lengths if l >= 200 and l <= 8000]

    # loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2 * n)
        # number of restarts
        restarts = 5
        for r in range(restarts):
            if time.time() >= deadline: break
            # initialize a
            if r == 0:
                # interpolate base
                old_x = np.linspace(0,1,n0)
                new_x = np.linspace(0,1,n)
                a = np.interp(new_x, old_x, base_seq)
            elif r == 1:
                # exponential shape
                lam = np.random.uniform(0.0001, 0.01)  # small lambda
                a = np.exp(-lam * np.arange(n))
                # scale to target sum
            elif r == 2:
                # power law shape
                p = np.random.uniform(0.2, 2.0)
                a = 1.0 / (np.arange(1, n+1) ** p)
            else:
                # random Dirichlet
                a = np.random.rand(n)
            a = a - np.min(a)  # ensure non-negative
            tot = a.sum()
            if tot > 0:
                a = a * (target_sum / tot)
            else:
                a = np.full(n, target_sum / n)

            # run subgradient descent (few thousand iterations)
            a, score = subgradient_descent(a, target_sum, deadline, max_iters=20000)

            # run cheap local search (move mass)
            a, score = local_search(a, target_sum, deadline, max_moves=5000)

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[new best] len={n} score={score:.6f}")

    # final fallback: uniform if needed
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2*n)
        best_seq = np.full(n, target_sum/n)
        best_score = compute_score(best_seq)

    print(f"[done] best score = {best_score:.6f} length={best_seq.shape[0]}")
    return best_seq.tolist()

def conv_max_fft(a):
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())

"""
Propose a step‑function (non‑negative vector) that minimises the
evaluate_sequence score.  The algorithm combines three ideas:

1.  A fast exact evaluator based on FFT (the score equals the maximal
    autocorrelation value when the L1‑norm is fixed to √(2 n)).
2.  A large, timed search over several lengths.  For each length we try
    a handful of deterministic initialisations (interpolated previous
    best, exponential, power‑law) and a few random Dirichlet starts.
3.  For every start we run a simple projected sub‑gradient descent
    (gradient = reversed vector for the currently active shift) with a
    back‑tracking line‑search, followed by a cheap coordinate‑transfer
    hill‑climb.  The best sequence found before the time budget expires
    is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation a*a via FFT and returns (max value, index,
    full convolution array).  All calculations are done in double
    precision.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx, conv


def rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, returns the vector r where
    r[i] = a[k - i] if the index is inside the support, otherwise 0.
    This is exactly the sub‑gradient of the max‑convolution at shift k.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale the non‑negative vector v to have the prescribed L1‑norm.
    Very cheap because v is already non‑negative.
    """
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target_sum / v.shape[0])
    return v * (target_sum / s)


def compute_score(a: np.ndarray) -> float:
    """
    With the L1‑norm normalised to sqrt(2n) the evaluate_sequence value
    is exactly the maximal convolution entry.
    """
    max_val, _, _ = conv_max_fft(a)
    return max_val


# ----------------------------------------------------------------------
def subgradient_descent(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 30000,
    init_alpha: float = 0.02,
):
    """
    Projected sub‑gradient descent on the objective max_k (a*a)[k].
    Uses a back‑tracking line‑search and occasional Gaussian jitter
    to leave plateaus.
    """
    n = a.shape[0]
    alpha = init_alpha
    best_score = compute_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        cur_max, cur_idx, _ = conv_max_fft(a)

        # sub‑gradient for the currently active shift
        grad = rev_for_shift(a, cur_idx)

        improved = False
        cur_alpha = alpha
        # back‑tracking line‑search (max 5 trials)
        for _ in range(5):
            a_new = a - cur_alpha * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = project_onto_simplex(a_new, target_sum)

            new_max = compute_score(a_new)
            if new_max < cur_max - 1e-12:
                a = a_new
                cur_max = new_max
                improved = True
                break
            cur_alpha *= 0.5

        if improved:
            best_score = min(best_score, cur_max)
            # slowly increase step size when we make progress
            alpha = min(alpha * 1.05, 0.1)
        else:
            # add a tiny random jitter to shake the iterate
            jitter = np.random.randn(n) * (target_sum / (n * 2000.0))
            a = np.maximum(a + jitter, 0.0)
            a = project_onto_simplex(a, target_sum)
            # shrink step size after a failure
            alpha = max(alpha * 0.7, 1e-6)

    return a, best_score


# ----------------------------------------------------------------------
def local_search(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_moves: int = 20000,
):
    """
    Very cheap hill‑climb: repeatedly move a small amount of mass from a
    random donor index to a random receiver index if the maximal
    convolution value improves.
    """
    n = a.shape[0]
    best_score = compute_score(a)

    for mv in range(max_moves):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 5 % of the donor's mass
        delta = a[i] * np.random.rand() * 0.05
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_onto_simplex(a_new, target_sum)

        new_score = compute_score(a_new)
        if new_score + 1e-12 < best_score:
            a = a_new
            best_score = new_score

    return a, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Main entry point.  Searches over a handful of lengths (including

    found before the time budget expires is returned as a plain list.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Main optimisation loop
    for n in lengths:
        if time.time() >= deadline:

            )

        best_score = compute_score(best_seq)
