def _conv_max_fft(a):
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:2*n-1]
    max_val = float(conv.max())
    return max_val, conv

def _subgradient(a, active_idxs):
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad

def _adam_optimize(init_a, n, target_sum, deadline, max_iters=10_000_000):
    a = init_a.copy()
    # Normalization
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.06
    iteration = 0
    jitter_interval = 3000
    best_vec = a.copy()
    best_val = None

    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # convolution and active set
        M, conv = _conv_max_fft(a)
        if best_val is None or M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[adam] n={n} iter={iteration} new best M={best_val:.6f}")

        # active indices with relative tolerance
        tol = max(1e-12, M * 1e-8)
        active = np.where(conv >= M - tol)[0]

        # subgradient
        rev = _subgradient(a, active)

        # gradient of objective assuming sum constraint (same as before)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = base_lr / (1 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

    return best_vec, best_val if best_val is not None else np.inf

"""
Propose a non‑negative step function that aims to minimise
``evaluate_sequence``.

Algorithm
---------
* Load a previously found construction (``height_sequence_1``) if it exists,
  otherwise start from a uniform vector.
* For several problem sizes (original length and a few multiples) run a
  *refinement loop* that alternates two complementary optimisation steps:
    1. **Linear‑programming step** – solve
         maximise   Σ g_i
         subject to g_i ≥ 0  and   (f * g)(k) ≤ M   for all shifts k,
       where f is the current sequence and M = max (f * f).  The solution
       pushes mass into positions that do not increase the peak of the
       convolution.  The LP is solved with CVXPY/ECOS.
    2. **Adam sub‑gradient step** – minimise the peak of the autocorrelation
       of the blended sequence using the exact sub‑gradient of the max‑convolution.
       The variables are projected onto the non‑negative orthant and the
       L¹‑norm is forced to √(2 n) after each update.
  * After each LP/Adam pair the best sequence seen so far is stored.
  * A cheap stochastic local‑search (mass transfer between two random positions)
    is applied as a final polishing step.
* The outer loop stops when the global time budget (≈1000 s) is exhausted.
* The best sequence discovered is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional import of CVXPY – used for the linear‑programming refinement.
# If unavailable the code falls back to the pure Adam optimiser.
try:
    import cvxpy as cp
except Exception as exc:  # pragma: no cover
    cp = None
    print("[warn] cvxpy could not be imported – LP refinement will be disabled.", exc)


# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Return the maximal value of the full convolution a*a and the full
    convolution array.  Uses FFT with zero‑padding to a power of two.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of the max‑convolution at the current point.
    For each active shift k the contribution a_{i‑k} is added.
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx                     # partner indices for shift k
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size > 0:            # average over the active set
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 10_000_000):
    """
    Adam optimiser that reduces the peak value of a*a.
    The L¹‑norm is projected back to ``target_sum`` after each step.
    Returns the best vector seen and its maximum convolution value.
    """
    a = init_a.copy()
    # Normalise to the prescribed L¹‑norm
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.06                # a bit larger than the original script
    iteration = 0
    jitter_interval = 3000

    best_vec = a.copy()
    best_val = None

    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # 1) convolution and active set
        M, conv = _conv_max_fft(a)

        # keep track of the best seen so far
        if best_val is None or M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[adam] n={n:5d} iter={iteration:7d} new best M={best_val:.6f}")

        # active set – use a relative tolerance to capture near‑max shifts
        tol = max(1e-12, M * 1e-8)
        active = np.where(conv >= M - tol)[0]

        # 2) sub‑gradient of the max term
        rev = _subgradient(a, active)

        # derivative of the full objective (the prefactor 2·n/(sum²) is constant)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # 3) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4) projection onto the feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # 5) jitter to escape flat plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

    return best_vec, best_val if best_val is not None else np.inf


# ----------------------------------------------------------------------
def _solve_lp_max_g(a: np.ndarray,
                    M: float,
                    target_sum: float,
                    deadline: float,
                    warm_start: np.ndarray | None = None):
    """
    Solve the linear program
        maximise   Σ g_i
        subject to g_i ≥ 0  and   (a * g)(k) ≤ M   for all shifts k.
    Returns a feasible g (scaled to ``target_sum``) or ``None`` if
    solving fails or time is exhausted.
    """
    if cp is None:                     # fallback – LP not available
        return None

    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        g.value = warm_start

    # convolution a * g (full length 2n‑1).  a is a constant, g is variable.
    conv = cp.conv(cp.Constant(a), g)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv <= M])

    # Give the solver a modest iteration budget derived from remaining time.
    time_left = deadline - time.time()
    if time_left <= 0:
        return None
    max_iters = int(max(200, min(5000, time_left * 100)))   # heuristic

    try:
        prob.solve(solver=cp.ECOS, max_iters=max_iters, warm_start=True, verbose=False)
    except Exception as e:            # pragma: no cover
        print("[lp] solver exception:", e)
        return None

    if g.value is None:
        print("[lp] no solution returned.")
        return None

    g_arr = np.array(g.value).astype(float).flatten()
    g_arr = np.maximum(g_arr, 0.0)
    s = g_arr.sum()
    if s > 0:
        g_arr *= (target_sum / s)
    else:
        g_arr = np.full(n, target_sum / n)

    return g_arr


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap stochastic hill‑climbing: repeatedly move a tiny amount of
    mass from a random source index to a random destination.
    The move is kept only if it reduces the maximal convolution value.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = best[i] * 0.01   # move 1 % of the source mass
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is preserved, non‑negativity already ensured

        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best, best_val = cand, M

    return best, best_val


# ----------------------------------------------------------------------
def _refine_sequence(init_a: np.ndarray,
                     target_sum: float,
                     deadline: float):
    """
    Alternating LP + Adam refinement for a single length.
    Returns a refined sequence and its maximal convolution value.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_M, _ = _conv_max_fft(best_a)

    outer = 0
    warm_g = None
    while time.time() < deadline and outer < 6:
        outer += 1
        M, _ = _conv_max_fft(a)

        # 1) linear‑programming step
        g = _solve_lp_max_g(a, M, target_sum,
                            deadline,
                            warm_start=warm_g)
        if g is None:                     # LP unavailable or failed
            break
        warm_g = g.copy()

        # 2) blend the new candidate
        t = 0.3
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / a.shape[0]

        # 3) short Adam polishing (max 30 s or remaining time)
        sub_deadline = time.time() + min(30.0, deadline - time.time())
        a, M_adam = _adam_optimize(a, a.shape[0],
                                   target_sum,
                                   sub_deadline,
                                   max_iters=500_000)

        # 4) keep the best sequence seen so far
        M_new, _ = _conv_max_fft(a)
        if M_new < best_M:
            best_M, best_a = M_new, a.copy()
            print(f"[refine] outer {outer} new best M={best_M:.6f}")

    # final cheap local search (if time permits)
    if time.time() < deadline:
        ls_steps = 30_000
        ls_seq, ls_val = _local_search(best_a,
                                       best_a.shape[0],
                                       target_sum,
                                       steps=ls_steps,
                                       deadline=deadline)
        if ls_val < best_M:
            best_M, best_a = ls_val, ls_seq.copy()
            print(f"[refine] post‑LS improved M={best_M:.6f}")

    return best_a, best_M

    """
    np.random.seed(seed)
    start = time.time()

    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------

    try:

    except Exception:

        print("[info] no previous construction – using uniform start")

    n0 = base.shape[0]
    target0 = math.sqrt(2.0 * n0)
    if base.sum() > 0:
        base *= (target0 / base.sum())
    else:
        base[:] = target0 / n0

    # Global best tracker
    best_seq = base.copy()
    best_val, _ = _conv_max_fft(best_seq)
    print(f"[init] length={n0} initial M={best_val:.6f}")

    # ------------------------------------------------------------------

    lengths = sorted({max(300, min(8000, int(n0 * f))) for f in length_factors})
    print("[info] exploring lengths:", lengths)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts for this size

            # Initialise a candidate for this length

                    old_pos = np.linspace(0, 1, base.shape[0])

                init = np.maximum(init, 0.0)
