"""
Propose a step‑function (non‑negative vector) that minimizes the
evaluate_sequence objective.

Algorithm
---------
1. Normalise every candidate to the “canonical’’ L1‑norm
   sum(a) = sqrt(2·n).  For this normalisation the objective equals the
   maximal autocorrelation value max((a∗a)).
2. For a given length `n` we run a *smoothed* gradient descent:
   *   f_α(a) = (1/α)·log∑_k exp(α·(a∗a)_k)  →  max‑convolution as α→∞.
   *   ∇f_α(a)  =  correlation(a, w)  where w_k ∝ exp(α·(a∗a)_k)
       (the weights are a soft‑max over the convolution entries).
   *   The correlation is computed by one FFT‑based convolution
       (a * reverse(w)).
   *   Adam (first/second moment) updates are performed on the gradient
       of f_α.  α is increased slowly (continuation) so that the
       optimisation starts with a smooth landscape and ends with an
       accurate approximation of the true max‑convolution.
3. After each Adam step we project onto the feasible set
   (non‑negative & fixed L1‑norm).
4. Occasionally a small random jitter is added to escape plateaus.
5. Whenever a new lower maximal autocorrelation is observed, the global
   best vector is updated.
6. A cheap coordinate‑transfer hill climber (move a tiny amount of mass
   from one index to another) is run after the Adam phase to polish the
   solution.
7. The search is performed for several lengths (the original length
   and a few scaled versions) and for a few random restarts per length.
   The best vector found before the time budget expires is returned.

The whole routine respects the 1000‑second budget, works with only
numpy/scipy and does not require any external files.
"""

import time
import math
import numpy as np
from scipy.signal import fftconvolve

# --------------------------------------------------------------
def _proj_nonneg_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto non‑negative orthant and renormalise to the given sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])
    return a


# --------------------------------------------------------------
def _full_autocorr(a: np.ndarray) -> np.ndarray:
    """Full linear autocorrelation a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    # next power‑of‑two >= 2·n‑1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
def _correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation of a (len n) with w (len 2·n‑1):
    (a ⋆ w)[i] = Σ_j a_j·w[i+j],   i=0..n‑1.
    Implemented as convolution with the reversed weight vector.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2·n‑1
    N = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    corr_full = np.fft.irfft(A * W_rev, n=N)
    start = m - 1                # index where linear correlation starts
    return corr_full[start:start + n]


# --------------------------------------------------------------
def _adam_smooth(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 40_000,
    lr0: float = 0.08,
    alpha_start: float = 1.0,
    alpha_inc: float = 0.001,
    alpha_max: float = 40.0,
    jitter_every: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Returns possibly improved (best_seq, best_score).
    """
    a = _proj_nonneg_sum(a_init.copy(), target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha = alpha_start

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        # ----- convolution & current max ---------------------------------
        conv = _full_autocorr(a)
        max_b = conv.max()

        # ----- update global best ---------------------------------------
        if max_b < best_score:
            best_score = max_b
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:6d}] new best max={best_score:.6f}")

        # ----- soft‑max weights -----------------------------------------
        # stability: shift by the maximum
        shift = max_b
        exp_arg = alpha * (conv - shift)
        # clamp to avoid overflow/underflow
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (correlation) -----------------------------------
        grad = _correlation(a, w)          # ∇f_α(a), shape (n,)

        # ----- Adam update (descent) -----------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step                     # gradient descent step

        # ----- projection ------------------------------------------------
        a = _proj_nonneg_sum(a, target_sum)

        # ----- occasional jitter ----------------------------------------
        if iteration % jitter_every == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = _proj_nonneg_sum(a + jitter, target_sum)

        # ----- increase smoothing parameter -------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

    return best_seq, best_score

def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_trials: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: repeatedly transfer a small fraction of mass
    from one index to another, accepting only moves that strictly lower
    the maximal autocorrelation.
    """
    a = a_start.copy()
    trial = 0
    while time.time() < deadline and trial < max_trials:
        trial += 1
        i = np.random.randint(n)                 # donor
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n - 1)             # receiver ≠ donor
        if j >= i:
            j += 1
        # transfer up to 5 % of donor’s mass
        delta = a[i] * np.random.rand() * 0.05
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # projection is not needed – we keep non‑negativity by construction
        max_b = _full_autocorr(new_a).max()
        if max_b < best_score - 1e-12:
            a = new_a
            best_score = max_b
            print(f"[local] improvement to {best_score:.6f} (trial {trial})")
    return a, best_score

    value.  Returns the best sequence found as a Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # small safety margin

    # ----- load previous construction (if any) ------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # ----- global bookkeeping -----------------------------------------
    best_seq = None
    best_score = float("inf")

    # ----- initialise from the previous construction ------------------
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = _proj_nonneg_sum(base, target_sum0)
    init_score = _full_autocorr(base).max()   # same as evaluate_sequence
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, score={best_score:.6f}")

    # ----- lengths to explore (scaled versions of the base length) -----
    length_factors = [0.8, 1.0, 1.2, 1.4, 1.7, 2.0]
    lengths = sorted({max(100, int(n0 * f)) for f in length_factors})
    # clip to a reasonable maximum (larger lengths are cheap enough)
    lengths = [L for L in lengths if L <= 12_000]

    # ----- main optimisation loop --------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ---- initialisation ----------------------------------------
            if r == 0:
                # interpolate the current best vector onto the new grid
                old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, best_seq)
                a0 = np.maximum(a0, 0.0)
            else:
                a0 = np.random.rand(n)

            a0 = _proj_nonneg_sum(a0, target_sum)

            # ---- Adam on smoothed objective ---------------------------
            best_seq, best_score = _adam_smooth(
                a_init=a0,
                n=n,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iter=40_000,
                lr0=0.08,
                alpha_start=1.0,
                alpha_inc=0.001,
                alpha_max=40.0,
                jitter_every=2500,
            )

            # ---- cheap local refinement --------------------------------
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_trials=20_000,
                )

    # ----- safety fallback ---------------------------------------------
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _full_autocorr(best_seq).max()
        print("[fallback] uniform sequence used")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Convert to plain Python list (evaluate_sequence will clamp if needed)
    return best_seq.tolist()
