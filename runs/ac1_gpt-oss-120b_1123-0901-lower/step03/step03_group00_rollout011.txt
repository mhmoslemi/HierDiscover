"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

The routine works in two phases:

1.  ADAM optimisation on a smooth‑max surrogate (identical in spirit to the
    previous solution but with a longer run, a slightly larger learning rate
    and a more aggressive λ‑schedule).  The sequence is kept on the
    probability simplex by using a soft‑max parametrisation.

2.  Greedy local random perturbations.  Starting from the best sequence
    obtained above we repeatedly move a small amount of probability mass
    from one index to another, updating the full self‑convolution
    incrementally (O(n) per move).  Only moves that strictly improve the
    exact objective are kept; after a period of stagnation the step‑size
    is increased (“shake’’) to escape flat regions.

Both phases respect the given time budget (default 1000 s).  The final
sequence is returned as a plain Python list; it is guaranteed to be
non‑negative, and the objective is evaluated exactly only at the very end.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions – all defined at top level (no closures)

def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) using real FFT.
    Returns a real array of length 2·n‑1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)], i = 0…n‑1.
    Implemented via FFT (see the original code for details).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def objective_from_conv(conv: np.ndarray, a_sum: float) -> float:
    """Exact objective used by evaluate_sequence for a given convolution."""
    n = (conv.shape[0] + 1) // 2
    return float(2.0 * n * conv.max() / (a_sum ** 2))


def exact_objective(a: np.ndarray) -> float:
    """Compute the exact evaluation function (no extra clipping)."""
    conv = full_convolution(a)
    return objective_from_conv(conv, a.sum())


# --------------------------------------------------------------
# Phase‑1: Adam optimisation on the smooth‑max surrogate

def adam_optimize(
    a0: np.ndarray,
    max_iter: int,
    deadline: float,
    lam_start: float = 0.5,
    lam_factor: float = 1.04,
    lr: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(a*a).
    Returns the best sequence (numpy array) and its exact objective.
    """
    n = a0.shape[0]
    # FFT sizes (powers of two) – reuse for the whole run.
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # w length + a length - 1

    # initialise logits from a0 (must be strictly positive)
    logits = np.log(a0 + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = None
    best_val = np.inf
    lam = lam_start
    last_improve = time.time()

    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        # ----- softmax → probabilities (lies on simplex) -----
        a = softmax(logits)

        # ----- self‑convolution (used for smooth‑max) ----------
        conv = full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()

        # ----- smooth‑max weights --------------------------------
        # w ∝ exp(λ·(conv - max_c))  (numerically stable)
        w = np.exp(lam * (conv - max_c))
        w /= w.sum()

        # ----- gradient of the surrogate w.r.t. a ----------------
        grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

        # ----- back‑prop through softmax (gradient w.r.t. logits) ----------
        avg = np.dot(a, grad_a)                       # ⟨a,∇φ⟩
        grad_logits = a * (grad_a - avg)              # ∇_x φ

        # ----- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule ----------
        lam = min(lam * lam_factor, 1e6)

        # ----- exact objective (fast, because Σa = 1) ----------
        val = 2.0 * n * max_c  # a.sum() == 1 exactly (up to FP error)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={t:05d}  val={val:.6f}")

        # ----- stagnation handling ----------
        if time.time() - last_improve > 6.0:
            # small random shake of the logits to escape flat plateaus
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam] shake at iter={t}")

    # final exact evaluation (in case rounding drifted)
    if best_a is not None:
        best_val = exact_objective(best_a)
        print(f"[Adam] final best = {best_val:.6f}")

    return best_a, best_val


# --------------------------------------------------------------
# Phase‑2: Greedy local random perturbations (O(n) per move)

def local_search(
    a: np.ndarray,
    best_val: float,
    deadline: float,
    epsilon_start: float = 0.02,
    epsilon_decay: float = 0.9995,
    stagnation_time: float = 8.0,
):
    """
    Greedy hill‑climbing that moves a small amount of mass from a random source
    index to a random target index.  The full convolution is updated
    incrementally, so each iteration costs O(n).  Only improving moves are kept.
    """
    n = a.shape[0]
    L = 2 * n - 1
    conv = full_convolution(a)           # current convolution
    epsilon = epsilon_start
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        # pick source (must have some mass) and target
        i = np.random.randint(0, n)
        if a[i] <= 1e-16:          # cannot take from an (almost) empty site
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue

        # step size – at most a[i] (to keep non‑negative)
        max_move = a[i]
        eps = max_move * epsilon * np.random.rand()
        if eps < 1e-12:
            continue

        # keep a copy of the entries we will modify (needed for revert)
        ai_old = a[i]
        aj_old = a[j]

        # ----- incremental update of the convolution -----------------
        # store a copy of a before modification for the slice update
        a_old = a.copy()

        # self‑convolution change:   Δc = -2·eps·a (shift by i)  + 2·eps·a (shift by j)
        conv[i:i + n] -= 2.0 * eps * a_old
        conv[j:j + n] += 2.0 * eps * a_old

        # δ*δ terms (i and j are the moved indices)
        conv[2 * i] += eps * eps
        conv[i + j] -= 2.0 * eps * eps
        conv[2 * j] += eps * eps

        # ----- apply the move on the probability vector -----------------
        a[i] = ai_old - eps
        a[j] = aj_old + eps

        # ----- evaluate the new objective --------------------------------
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() stays ≈ 1

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Local] iter={it:06d}  val={val:.6f}")

        else:
            # ---- revert the move (undo the incremental updates) -----
            conv[i:i + n] += 2.0 * eps * a_old
            conv[j:j + n] -= 2.0 * eps * a_old
            conv[2 * i] -= eps * eps
            conv[i + j] += 2.0 * eps * eps
            conv[2 * j] -= eps * eps
            a[i] = ai_old
            a[j] = aj_old

        # ----- step‑size schedule -----
        epsilon = max(epsilon * epsilon_decay, 1e-5)

        # ----- stagnation shake -----
        if time.time() - last_improve > stagnation_time:
            # enlarge the step a bit to escape flat region
            epsilon = min(epsilon * 10.0, epsilon_start)
            last_improve = time.time()
            print(f"[Local] shake at iter={it}")

    return best_seq, best_val

# Main entry point required by the competition framework

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The routine respects a fixed time budget, prints progress, and finally
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  8192 gave the best result in the previous run.
    n = 8192

    # ------------------------------------------------------------------
    # Initialise a candidate.
    # Prefer the globally supplied construction if it matches the length.
    if "height_sequence_1" in globals():
        cand = np.array(height_sequence_1, dtype=np.float64)
        if cand.shape[0] >= n:
            cand = cand[:n]
        else:
            cand = np.pad(cand, (0, n - cand.shape[0]))
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            # fallback to a random Dirichlet point
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = cand / s
        print("[Init] using height_sequence_1 (trim/pad) as start")
    else:
        # Gaussian‑shaped start (symmetric, smooth)
        sigma = n / 8.0
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a0 = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[Init] Gaussian start (σ = n/8)")

    # ------------------------------------------------------------------
    # Phase‑1 : ADAM on smooth‑max
    phase1_deadline = time.time() + (budget_s * 0.65)   # ~65 % of the budget
    a_adam, val_adam = adam_optimize(
        a0,
        max_iter=50000,
        deadline=phase1_deadline,
        lam_start=0.5,
        lam_factor=1.04,
        lr=0.07,
    )

    # ------------------------------------------------------------------
    # Phase‑2 : Greedy local search
    if a_adam is None:
        a_adam = a0
        val_adam = exact_objective(a_adam)

    phase2_deadline = deadline
    best_seq, best_val = local_search(
        a_adam,
        best_val=val_adam,
        deadline=phase2_deadline,
        epsilon_start=0.02,
        epsilon_decay=0.9995,
        stagnation_time=8.0,
    )

    # ------------------------------------------------------------------
    # In the unlikely case that local search never accepted a move,
    # fall back to the best sequence obtained from ADAM.
    if best_seq is None:
        best_seq = a_adam
        best_val = exact_objective(best_seq)

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the provided evaluator (if needed)
    # (conversion to list forces the external call to be safe.)
    print(f"[Result] final objective = {best_val:.6f}, length = {n}")

    return best_seq.tolist()
