"""
Search for a non‑negative step‑function (probability vector) that minimises
    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm works directly on the simplex by representing a as a soft‑max
of unconstrained logits.  In each iteration we:
- compute the exact linear convolution a*a,
- identify the lag(s) where the maximum is attained,
- form a sub‑gradient of the max‑function by averaging the gradients of those
  active lags,
- back‑propagate through the soft‑max and update the logits with Adam,
- occasionally shake the logits when no improvement has been seen for a while.

A few restarts are performed for several problem sizes (up to 16384).  The
best sequence encountered is returned as a plain Python list.
"""
import time
import numpy as np

def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Run a time‑bounded optimisation and return the best sequence found.
    The returned sequence is a Python list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # numpy array
    best_val = float("inf")

    # problem sizes – larger n gives a lower theoretical bound (≈1)
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # pre‑compute FFT lengths that will be reused
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # number of independent restarts for this length
        restarts = 5 if n <= 4096 else 3 if n <= 8192 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ----- initialise logits ------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                init_a = best_seq.copy()
                init_a /= init_a.sum()
                x = np.log(init_a + 1e-15)      # logits corresponding to init_a
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand.sum()
                x = np.log(cand + 1e-15)
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random start")

            # ----- Adam state -------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.1                     # initial learning rate
            decay_every = 2000           # decay interval (iterations)

            # ----- optimisation loop ------------------------------------------------
            last_improve = time.time()
            max_iters = 40000 if n <= 4096 else 25000 if n <= 8192 else 20000
            t = 0

            while time.time() < deadline and t < max_iters:
                # logits → probabilities (soft‑max)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)          # for numerical stability
                a = exp_x / exp_x.sum()            # a_i ≥ 0, Σ a_i = 1

                # linear self‑convolution
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c                # exact objective (Σa=1)

                if val < best_val - 1e-12:

                    best_seq = a.copy()
                    last_improve = time.time()

                inner = np.dot(a, grad_a)

                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay
                if t % decay_every == 0:
                    lr *= 0.7

                # stagnation handling – add a tiny random jitter
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

            # ----- optional local greedy refinement ---------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small amount from i to j
                    eps_move = min(a_local[i], 0.01) * (np.random.rand() * 0.5 + 0.5)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")
    # -------------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but we fall back to a uniform vector
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)   # = 2

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
