        init = np.maximum(init, 0.0)

        f = init.copy()

            M, conv = _conv_max_fft(f)
            # build A_ub
            try:

                from scipy.optimize import linprog

                break

            c = -np.ones(n)  # maximize sum(g)
            bounds = [(0, None)] * n

"""
Propose a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The method combines three ideas:

1. **LP‑guided refinement** (for moderate lengths only):
   Starting from a feasible vector `f` (∑f = √(2n)), we solve the linear
   program

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M  for all k       (M = max(f*f))
                    g_i ≥ 0

   The solution `g` is rescaled to the required L1‑norm and mixed
   into `f` by a small step `t`.  Repeating this few times provably never
   increases the peak autocorrelation and often yields a strong
   reduction.  The LP is built with a sparse Toeplitz matrix (one diagonal
   per offset) and solved with SciPy’s HiGHS solver.

2. **Adam optimiser on the exact sub‑gradient** of the max‑convolution.
   After the LP stage we run a standard Adam optimiser (the code from the
   previous baseline) for a few thousand iterations, projecting onto the
   feasible simplex after each step.

3. **Cheap stochastic local search** that moves a tiny amount of mass
   between two random positions if the peak improves.

The whole search respects the 1000 s wall‑clock budget, tries several
lengths (original size and a few larger multiples) and many random
initialisations, and always keeps the best vector seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return the maximal value of the full convolution a*a and the array."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """Exact sub‑gradient of the max‑convolution (active‑set version)."""
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5_000_000):
    """Adam optimiser on the exact sub‑gradient of the max‑convolution."""
    a = init_a.copy()
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.06          # a touch larger than the previous baseline
    iteration = 0
    jitter_interval = 3000
    best_vec = a.copy()
    best_val, _ = _conv_max_fft(a)

    while time.time() < deadline and iteration < max_iters:
        # 1. convolution and active set
        M, conv = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # 2. sub‑gradient of the max term
        rev = _subgradient(a, active)

        # derivative of the whole objective (after normalisation)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # 3. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4. projection onto feasible set (non‑negative & correct L1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # 5. occasional jitter to escape flat plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # 6. keep the best solution seen so far
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Simple stochastic local search: move a tiny amount of mass from one index
    to another if the peak autocorrelation improves.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)
    for step in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] <= 0:
            continue
        delta = best[i] * 0.02
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M
    return best, best_val


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               target_sum: float,
               max_iters: int,
               deadline: float):
    """
    Linear‑programming guided refinement (see doc‑string).  Returns a vector
    with a (usually) lower peak max‑convolution than the input `f`.
    """
    try:
        from scipy.sparse import diags
        from scipy.optimize import linprog
    except Exception as e:
        print("[lp] SciPy not available – skipping LP refinement.")
        return f

    best_f = f.copy()
    best_val, _ = _conv_max_fft(best_f)
    t = 0.2        # mixing parameter
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M, _ = _conv_max_fft(best_f)

        # Build the Toeplitz constraint matrix: A_ub @ g <= M
        # A_ub has one diagonal per offset d (0 … n‑1) filled with f[d].
        # Using `diags` we create a sparse CSC matrix with O(n²) non‑zeros,
        # which is acceptable for moderate n (≈ 2100).
        diags_data = [np.full(n, best_f[d]) for d in range(n)]
        A_ub = diags(diags_data, offsets=np.arange(n), shape=(2 * n - 1, n),
                     format='csc')

        # maximise Σ g_i  →  minimise –Σ g_i
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs')
        if not res.success:
            print(f"[lp] iteration {it+1} failed: {res.message}")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break
        # Rescale to the required L1 norm
        g = g0 * (target_sum / sum_g0)

        # Mix the new vector into the current one
        best_f = (1.0 - t) * best_f + t * g
        # (sum remains target_sum automatically)
        cur_val, _ = _conv_max_fft(best_f)
        if cur_val < best_val:
            best_val = cur_val
            print(f"[lp] length {n} iter {it+1} -> new M = {best_val:.6f}")
        # slowly shrink the mixing factor
        t = max(0.02, 0.9 * t)
    return best_f
