"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Initialise a sequence using the (discrete) arcsine law, which is known
  to give an almost flat self‑convolution.  The sequence is normalised
  such that  Σa = √(2 n) (the scaling in ``evaluate_sequence`` makes the
  objective equal to the maximal convolution value).
* For a carefully chosen list of lengths (from a few hundred up to a few
  thousand) run a long Adam optimiser on the exact sub‑gradient of the
  max‑convolution.  The optimiser projects onto the feasible set
  (non‑negative, correct ℓ₁‑norm) after every step.
* After each Adam run a cheap stochastic local search moves a tiny
  amount of mass between random positions and accepts the move iff it
  reduces the peak of the convolution.
* The best vector found so far is kept globally.  When we move to a
  larger length the current best is linearly up‑sampled and mixed with
  a fresh arcsine start – this gives a good warm‑start for the next
  optimisation.
* The whole routine respects the supplied wall‑clock budget (≈1000 s)
  and prints progress information.  The returned list is the best
  sequence discovered before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Compute the full linear convolution a*a using an FFT of appropriate
    size and return the maximal entry together with the whole convolution
    (real array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())           # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A                                    # element‑wise product
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]    # keep the valid part
    return float(conv.max()), conv

# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of the maximal convolution entry.
    For each active shift k the derivative w.r.t. a_i equals a_{k‑i}.
    The returned vector is the average over all active shifts.
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx                # partner indices
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size:
        grad /= float(active_idxs.size)
    return grad

# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5_000_000):
    """
    Adam optimisation of the maximal convolution.
    Returns the best vector seen and its maximal convolution value.
    """
    a = init_a.copy()
    # Normalise to the required ℓ₁‑norm
    s = a.sum()
    if s > 0.0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08                # a bit larger than the original code
    jitter_every = 3000
    best_vec = a.copy()
    best_M, _ = _conv_max_fft(best_vec)

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # 1. current convolution and active set
        M, conv = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # 2. sub‑gradient of the max‑convolution
        rev = _subgradient(a, active)

        # Full gradient of the objective 2*n*max/target_sum**2.
        # After projection the denominator stays constant, therefore we
        # differentiate only the numerator.
        coeff = 4.0 * n / (target_sum * target_sum)   # = 2  (since target_sum² = 2n)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # 3. Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4. projection onto the feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # 5. occasional jitter to escape flat plateaus
        if iteration % jitter_every == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # 6. keep the best vector
        if M < best_M:
            best_M = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best M = {best_M:.6f}")

    return best_vec, best_M

# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap stochastic local search: transfer a tiny amount of mass
    from one index to another if the move reduces the maximal convolution.
    """
    best = a.copy()
    best_M, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if best[i] <= 1e-12:
            continue

        # transfer a random fraction of the source mass (up to 5 %)
        frac = np.random.rand() * 0.05
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # sum unchanged, non‑negativity preserved by construction
        M, _ = _conv_max_fft(cand)
        if M < best_M:
            best = cand
            best_M = M

    return best, best_M

# ----------------------------------------------------------------------
def _init_arcsine(n: int) -> np.ndarray:
    """
    Initialise a vector using the discrete arcsine law:
        a_i ∝ 1 / sqrt( (i+0.5)*(n-0.5-i) )
    This shape is known to give an almost flat self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a

# ----------------------------------------------------------------------
def _upsample_vector(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a vector to a new length.
    """
    if old.shape[0] == new_n:
        return old.copy()
    old_x = np.linspace(0.0, 1.0, old.shape[0])
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, old)

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best list of heights found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Load an existing construction if present, otherwise start from arcsine
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        print("[info] loaded previous construction of length", base.shape[0])
    except Exception:
        base = _init_arcsine(2100)
        print("[info] no previous construction – using arcsine start (len=2100)")

    # Normalise to the proper ℓ₁‑norm (sqrt(2 n))
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())

    # ------------------------------------------------------------------
    # Global best
    best_seq = base.copy()
    best_M, _ = _conv_max_fft(best_seq)
    print(f"[init] length={n0}, initial max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Choose a family of lengths – start small and double up to a few thousand
    length_factors = [0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({max(300, min(12000, int(n0 * f))) for f in length_factors})
    print("[info] exploring lengths:", lengths)

    # ------------------------------------------------------------------
    # Main optimisation loop over lengths
    for idx_len, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        restarts = 2 if n > 4000 else 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ---- initialise -------------------------------------------------
            if n == n0 and r == 0:
                init = best_seq.copy()
                print(f"[run] length={n} (reuse base) restart {r+1}/{restarts}")
            else:
                # up‑sample the current global best if possible
                if best_seq.shape[0] != n:
                    init = _upsample_vector(best_seq, n)
                else:
                    init = best_seq.copy()

                # mix a bit of fresh arcsine shape (helps escape local minima)
                mix = 0.3
                init = mix * _init_arcsine(n) + (1.0 - mix) * init
                # add tiny random jitter
                init += np.random.randn(n) * 0.001 * target_sum / n
                init = np.maximum(init, 0.0)
                init *= (target_sum / init.sum())

            # ---- Adam optimisation -----------------------------------------
            time_left = deadline - time.time()
            max_time = min(180.0, time_left * 0.6)     # give each run a budget
            sub_deadline = time.time() + max_time

            opt_seq, opt_M = _adam_optimize(init, n, target_sum, sub_deadline,
                                            max_iters=5_000_000)

            # ---- cheap local search ----------------------------------------
            if time.time() < deadline:
                ls_steps = int(5e4)                     # 50 k cheap moves
                ls_seq, ls_M = _local_search(opt_seq, n, target_sum,
                                             steps=ls_steps,
                                             deadline=deadline)
                if ls_M < opt_M:
                    opt_seq, opt_M = ls_seq, ls_M

            # ---- update global best ----------------------------------------
            if opt_M < best_M:
                best_M = opt_M
                best_seq = opt_seq.copy()
                print(f"[global] new best max‑conv = {best_M:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Final refinement on the globally best vector (if we still have time)
    if time.time() < deadline:
        n_best = best_seq.shape[0]
        target_sum_best = math.sqrt(2.0 * n_best)
        # few extra Adam iterations with a smaller learning rate
        sub_deadline = time.time() + min(60.0, (deadline - time.time()) * 0.8)
        refined_seq, refined_M = _adam_optimize(best_seq, n_best,
                                                target_sum_best,
                                                sub_deadline,
                                                max_iters=2_000_000)
        if refined_M < best_M:
            best_seq, best_M = refined_seq, refined_M
            print(f"[refine] improved max‑conv = {best_M:.6f}")

        # final exhaustive local search (still cheap)
        if time.time() < deadline:
            final_seq, final_M = _local_search(best_seq, n_best,
                                               target_sum_best,
                                               steps=200_000,
                                               deadline=deadline)
            if final_M < best_M:
                best_seq, best_M = final_seq, final_M
                print(f"[final] improved max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Report the official score
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best max‑conv = {best_M:.6f}, official score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
