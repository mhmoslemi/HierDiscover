"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.
The algorithm explores a range of lengths and uses two complementary
optimisers:

1.  A stochastic Adam optimiser with a sub‑gradient of the maximal
    convolution.  The optimiser can be constrained to a *periodic*
    pattern (a short template repeated many times) – this adds a new
    search dimension that was not explored in the reference code.

2.  A cheap hill‑climbing local search that moves a little amount of
    mass between two positions (or a small batch of positions) while
    keeping the L1‑norm fixed.

Multiple restarts, random initialisations (including interpolated
previous best and random periodic templates) and occasional jitter are
used to escape plateaus.  The best sequence found before the 1000 s
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation a*a via FFT and returns (max value, index, whole array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())            # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """For shift index k, return the reversed vector rev with rev[i]=a[k-i] (if in range)."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _subgradient_average(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Average of the reversed vectors for all active shift indices."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size:
        rev /= float(active.size)
    return rev


# ----------------------------------------------------------------------
def _periodic_average(a: np.ndarray, period: int) -> np.ndarray:
    """
    Enforce a strict period: every entry that shares the same index
    modulo ``period`` gets the same value – the average of that group.
    """
    if period is None or period <= 1:
        return a
    n = a.shape[0]
    idx = np.arange(n) % period
    sums = np.bincount(idx, weights=a, minlength=period)
    cnts = np.bincount(idx, minlength=period)
    avg = sums / np.maximum(cnts, 1.0)
    return avg[idx]


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    period: int = None,
    max_iter: int = 3_000_000,
    jitter_interval: int = 3000,
    random_active_prob: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser aimed at minimising the maximal auto‑convolution.
    ``period`` (if not None) forces the iterate to stay in a periodic
    subspace of period ``period``.
    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.06
    iteration = 0
    sqrt2 = math.sqrt(2.0)

    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        M, _, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        # stochastic sub‑gradient: either a random active shift or the average
        if active.size and np.random.rand() < random_active_prob:
            k = np.random.choice(active)
            rev = _rev_for_shift(a, k)
        else:
            rev = _subgradient_average(a, active)

        # gradient of the objective under the L1‑normalisation constraint
        const = sqrt2 * M / math.sqrt(a.shape[0])
        grad = 2.0 * rev - const

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # feasible set: non‑negative & normalisation
        a = np.maximum(a, 0.0)
        if period is not None:
            a = _periodic_average(a, period)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:   # extremely unlikely – fall back to uniform
            a = np.full_like(a, target_sum / a.shape[0])

        # occasional jitter to break plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(*a.shape) * 0.001 * target_sum / a.shape[0]
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            if period is not None:
                a = _periodic_average(a, period)

        # keep the best so far
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {a.shape[0]:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 35_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: repeatedly move a tiny amount of mass from one
    index to another (keeping the L1‑norm fixed).  Accepts a move only
    if the maximal convolution value decreases.
    """
    a = a_start.copy()
    n = a.shape[0]
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        eps = a[i] * np.random.rand() * 0.04   # up to 4 % transfer
        if eps <= 0:
            continue
        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps
        M_new, _, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (prop {proposal})")
    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The search mixes:
      * Adam optimisation (with optional periodic template)
      * A final cheap local‑search refinement
    across several lengths and random restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Load the previously discovered construction if present
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # ------------------------------------------------------------------
    # Normalise the initial vector (sum = sqrt(2 n))
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start from the known length and explore larger ones
    length_factors = [1.0, 1.3, 1.6, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    max_len = 8000
    lengths = [l for l in lengths if 300 <= l <= max_len]

    # occasional periods to try (None means unrestricted)
    period_options = [None, 2, 3, 4, 5, 6]

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # three restarts per length: one interpolated, two random
        for r in range(3):
            if time.time() >= deadline:
                break

            # ------------------- initialise -------------------------------
            if r == 0:
                # linear interpolation of the existing best onto the new grid
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, base)
                init_a = np.maximum(init_a, 0.0)
            else:
                # either a pure random Dirichlet draw or a random periodic template
                if np.random.rand() < 0.5:
                    # random Dirichlet‑like start (positive numbers, renormalised)
                    init_a = np.random.rand(n)
                else:
                    # random periodic start
                    period = np.random.choice(period_options[1:])   # pick 2‑6
                    tmpl = np.random.rand(period)
                    init_a = np.tile(tmpl, (n // period) + 1)[:n]
                init_a = np.maximum(init_a, 0.0)

            # normalise to target L1‑norm
            total = init_a.sum()
            if total > 0:
                init_a = init_a * (target_sum / total)
            else:
                init_a = np.full(n, target_sum / n)

            # possibly decide on a period for the optimiser (None = free)
            chosen_period = None
            if np.random.rand() < 0.3:            # 30 % of restarts use a period
                chosen_period = np.random.choice(period_options[1:])

            # ------------------- Adam optimisation -----------------------
            best_seq, best_score = _adam_optimize(
                a_init=init_a,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                period=chosen_period,
                max_iter=3_000_000,
                jitter_interval=3000,
                random_active_prob=0.35,
            )

            # ------------------- cheap local search ----------------------
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=25_000,
                )

    # ------------------------------------------------------------------
    # Final sanity check (in case nothing was found)
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
