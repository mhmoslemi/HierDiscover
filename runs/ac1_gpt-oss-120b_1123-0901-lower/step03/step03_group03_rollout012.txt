       ``rev = (w_rev * a)`` is a weighted reverse convolution.
   *   Adam updates are taken, then projected onto the feasible set
       ``a ≥ 0`` and ``∑a = S = √(2·n)``.
   *   A small Gaussian jitter is added every few thousand steps
       to escape plateaus.
   *   The temperature ``τ`` starts as a fraction of the current maximum
       and decays exponentially (slow decay → smoother landscape early,
       sharper later).

   The current exact maximum (computed **before** the Adam step) is
   recorded; the associated vector is saved as the current best.
4.  After the allocated budget for the current length finishes,
    the best vector found is kept.  Finally a short refinement is run
    on the overall best vector.
5.  The function respects the 1000 s wall‑clock limit, printing
    progress and returning the best sequence as a plain ``list``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns (weights, smoothed maximum).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute  rev_i = Σ_k w_k·a_{k‑i}   (indices outside [0,n) are ignored)
    using a single FFT pair.
    """
    # FFT of a (zero‑padded to length Nfft)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of the reversed weight vector (still length 2·n‑1)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Linear convolution via inverse FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep only the valid part (length = (2·n‑1) + n – 1 = 3·n‑2)
    conv_rev = conv_rev[: (2 * a.shape[0] - 1) + a.shape[0] - 1]
    # The needed slice is the centre part of length n
    start = 2 * a.shape[0] - 2
    rev = conv_rev[start - np.arange(a.shape[0])]
    return rev


def _initialise_sequence(base: np.ndarray, n: int, kind: str) -> np.ndarray:
    """
    Produce an initial non‑negative vector of length ``n``.
    ``kind`` can be:
        'base'      – linear interpolation of ``base`` (if sizes differ)
        'uniform'   – constant entries
        'triangular'– ramp up then down
        'exp'       – exponential random values
    The result is *not* normalised; the caller will enforce the L1‑norm.
    """
    if kind == "base":
        if n == base.shape[0]:
            return base.copy()
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, base)
        return np.maximum(a, 0.0)

    if kind == "uniform":
        return np.ones(n, dtype=float)

    if kind == "triangular":
        half = n // 2
        inc = np.arange(1, half + 1, dtype=float)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = np.arange(half, 0, -1, dtype=float)
        a = np.concatenate([inc, dec])
        # pad with a zero if needed (odd length)
        if a.shape[0] < n:
            a = np.append(a, 0.0)
        return a

    if kind == "exp":
        return np.random.exponential(scale=1.0, size=n)

    raise ValueError(f"unknown initialisation kind: {kind}")


# ----------------------------------------------------------------------
# Core optimiser (soft‑max Adam)
# ----------------------------------------------------------------------


def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int,
    *,
    lr0: float = 0.10,
    tau_factor: float = 0.30,
    decay_rate: float = 50000.0,
    jitter_interval: int = 2000,
    jitter_scale: float = 5e-3,
):
    """
    Runs the Adam optimiser for a single length ``n``.
    Returns the (possibly) improved ``best_seq`` and its score.
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the initial vector
    a = init_a.astype(float)
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    Nfft = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2·n‑1

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # ------------------------------------------------------------------
        # 1) evaluate exact convolution of the *current* vector
        # ------------------------------------------------------------------
        a_curr = a.copy()
        A_fft = np.fft.rfft(a_curr, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ------------------------------------------------------------------
        # 2) record if this is a new global best
        # ------------------------------------------------------------------
        if M < best_score:
            best_score = M
            best_seq = a_curr.copy()
            print(f"[len {n:5d} iter {iteration:9d}] new best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3) soft‑max smoothing of the max
        # ------------------------------------------------------------------
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # ------------------------------------------------------------------
        # 4) gradient of the smooth objective (via weighted reverse conv)
        # ------------------------------------------------------------------
        rev = _weighted_rev(a_curr, w, Nfft)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # ------------------------------------------------------------------
        # 5) Adam step
        # ------------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a_curr - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # 6) projection onto the feasible set (a ≥ 0, Σa = √(2·n))
        # ------------------------------------------------------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ------------------------------------------------------------------
        # 7) occasional jitter to help escaping shallow plateaus
        # ------------------------------------------------------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (jitter_scale * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Load previous construction (if any); fall back to a uniform vector
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # use a modest length that works well for a quick start
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())

    # initialise global best
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to investigate (original + enlargements, capped at 8000)
    # ------------------------------------------------------------------
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0]
    lengths = sorted(
        {min(int(n0 * f), 8000) for f in length_factors if int(n0 * f) >= 300}
    )

    # ------------------------------------------------------------------
    # Initialisation kinds we will try for each length
    # ------------------------------------------------------------------
    init_kinds = ["base", "uniform", "triangular", "exp"]

    # ------------------------------------------------------------------
    # Main optimisation loops
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[run] length={n}")

        # Determine a reasonable iteration budget for this length
        # More iterations for smaller n (cheaper per iteration)
        max_iters = int(2_000_000 * n0 / n)
        max_iters = max(800, max_iters)  # at least a few hundred steps

        for kind in init_kinds:
            if time.time() >= deadline:
                break

            # Prepare initial vector for this length/kind
            init_a = _initialise_sequence(base, n, kind)

            # Normalise it to the correct L1‑norm
            target_sum = math.sqrt(2.0 * n)
            init_a = init_a * (target_sum / init_a.sum())

            print(f"  [init kind={kind:11s}] max_iters≈{max_iters}")

            best_seq, best_score = _optimize_length(
                n=n,
                init_a=init_a,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iters=max_iters,
            )

    # ------------------------------------------------------------------
    # Final short refinement on the current global best (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_best = best_seq.shape[0]
        print("[refine] final fine‑tuning")
        best_seq, best_score = _optimize_length(
            n=n_best,
            init_a=best_seq,
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
            max_iters=500_000,
            lr0=0.05,
            tau_factor=0.15,
            decay_rate=30000.0,
            jitter_interval=3000,
            jitter_scale=2e-3,
        )

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
