"""
Low‑peak self‑convolution optimizer.

The evaluation function depends only on the normalized probability vector
p_i = a_i / sum(a).  The objective is 2·n·max_k (p∗p)_k.
We therefore optimise p ∈ Δ_n (the simplex) and finally return a = p·√(2n).

Algorithm
---------
* A multi‑scale schedule tries several lengths (500 … 12000).
* For each length we generate a small portfolio of initial candidates:
  – up‑sampled current best,
  – a few parametrised families (edge‑heavy exponentials, power‑law,
    mixed edge+uniform),
  – a random Dirichlet draw.
* Each candidate is refined by an Exponentiated‑Gradient (EG) loop
  on the log‑probabilities.
  – The objective is smoothed with a soft‑max (temperature β that
    grows from 0.1 to 50).
  – The gradient uses the exact weighted correlation
    rev_i = Σ_k α_k·p_{k‑i} (α is the soft‑max weight, normalised).
  – Log‑probabilities are updated, periodically jittered and projected
    back onto the simplex.
* The best among all runs is kept as the global optimum.
* The final sequence is returned as a_i = p_i·√(2n).

The implementation is pure NumPy, uses FFTs for O(n log n) convolutions
and stays well within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Full aperiodic self‑convolution (p∗p) using a single FFT of size N.
    Returns real array of length 2·len(p)‑1."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """Correlation rev_i = Σ_k w_k·p_{k‑i} (i=0..n‑1) via FFT.
    w has length 2·n‑1 (full convolution size)."""
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    # reversed weights
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]   # keep linear part
    rev = conv_rev[L - n : L][::-1]    # extract needed slice and reverse
    return rev


def _max_conv(p: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of p."""
    N = _next_pow2(3 * len(p) - 2)
    conv = _full_convolution_fft(p, N)
    return float(conv.max())


def _score(p: np.ndarray) -> float:
    """Objective value 2·n·max(p∗p).  (Same as evaluate_sequence for a = p·√(2n))."""
    n = len(p)
    return 2.0 * n * _max_conv(p)


# ----------------------------------------------------------------------
def _eg_optimize(
    n: int,
    p_start: np.ndarray,
    global_best: float,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation on the simplex for a fixed length.
    Returns the best probability vector found and its objective value.
    """
    # ----- hyper‑parameters -----
    beta_start = 0.1
    beta_end = 50.0                # stronger emphasis on the true max
    beta_ramp = int(0.6 * max_iters)  # iterations over which beta grows
    jitter_interval = 3000
    jitter_scale = 1e-3
    # ---------------------------

    p = p_start.copy()
    # work in log‑space to keep positivity automatically
    logp = np.log(p + 1e-30)

    N = _next_pow2(3 * n - 2)          # large enough for both convolutions
    best_local_score = float("inf")
    best_local_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- convolution ----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            # occasional logging (quiet if many iterations)
            if it % 5000 == 0 or it == 1:
                print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ---- soft‑max weighting ----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp)
        w = np.exp(beta * (conv - M))          # w_max = 1
        w_sum = w.sum()
        if w_sum == 0.0:      # numerical safety
            w_sum = 1.0
        # ---- weighted correlation (sub‑gradient of smoothed max) ----
        rev = _weighted_rev_fft(p, w, N) / w_sum

        # ---- gradient of the smoothed objective ----
        grad = 4.0 * n * rev

        # ---- step‑size (decaying) ----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # ---- EG update in log‑space ----
        logp -= eta * grad

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ---- project back onto the simplex ----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)   # avoid overflow
        p = exp_shift / exp_shift.sum()

        # early exit if we already beat the global best by a comfortable margin
        if best_local_score < global_best * 0.99:
            # continue a little to possibly improve a bit more
            pass

    return best_local_p, best_local_score

def _random_candidate(n: int) -> np.ndarray:
    """
    Generate a random probability vector using a small collection of
    parametrised families (edge‑heavy exponentials, power‑law, mixed).
    """
    typ = np.random.choice(["power", "exp", "mix"])
    if typ == "power":
        # p_i ∝ (x·(1‑x)+ε)^a , a ∈ [‑3, 2]
        a = np.random.uniform(-3.0, 2.0)
        x = (np.arange(n) + 0.5) / n
        shape = (x * (1.0 - x) + 1e-12) ** a
    elif typ == "exp":
        # p_i ∝ exp(‑c·dist_to_edge)
        c = np.random.uniform(0.5, 5.0)          # scale, larger ⇒ steeper edge
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        shape = np.exp(-c * dist / n)           # normalise by n to keep c moderate
    else:  # mix
        c = np.random.uniform(0.5, 5.0)
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        edge = np.exp(-c * dist / n)
        uniform = np.ones(n)
        mix = np.random.rand()
        shape = mix * edge + (1.0 - mix) * uniform

    # ensure positivity and normalise to a probability vector
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0.0:
        shape = np.ones(n)
    return shape / shape.sum()
