def grad_submax(a):
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    # Choose a random index among max indices
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-12))
    k = np.random.choice(idxs)
    # Compute gradient g: g_i = 2*a[k-i] if in range
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    # for i in [start, end], k - i in [0, n-1]
    i_vals = np.arange(start, end+1)
    j_vals = k - i_vals
    g[i_vals] = 2.0 * a[j_vals]
    return g, max_c

def run_adam_opt(n, init_a, time_deadline, max_iters=200000):
    x = np.log(init_a + 1e-12)  # initial logits
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    best_local_val = np.inf
    best_local_a = None
    t = 0
    last_improve = time.time()
    while time.time() < time_deadline and t < max_iters:
        a = softmax(x)
        g, max_c = grad_submax(a)
        # compute objective
        val = 2.0 * n * max_c  # sum a =1
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={t}] new best local: {val:.6f}")
        # compute grad wrt logits
        avg = np.dot(a, g)
        grad_x = a * (g - avg)
        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # stagnation shake
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[n={n} iter={t}] shake due to stagnation")
    return best_local_a, best_local_val

def run_de_opt(n, init_pop, time_deadline, max_gens=20000, popsize=10):
    # init_pop: list of numpy arrays (each sum to 1)
    # fill up to popsize
    pop = []
    for a in init_pop:
        pop.append(a.copy())
    while len(pop) < popsize:
        a = np.random.dirichlet(np.ones(n))
        pop.append(a)
    # compute objective for each
    obj_vals = [2.0 * n * _full_convolution(a).max() for a in pop]
    best_idx = np.argmin(obj_vals)
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]
    F = 0.8
    CR = 0.9
    last_improve = time.time()
    generation = 0
    while time.time() < time_deadline and generation < max_gens:
        generation += 1
        for i in range(popsize):
            # choose three distinct indices not i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # Clip negative
            mutant = np.maximum(mutant, 0.0)
            # Crossover
            cross = np.random.rand(n) < CR
            # ensure at least one from mutant
            jrand = np.random.randint(n)
            cross[jrand] = True
            trial = np.where(cross, mutant, pop[i])
            s = trial.sum()
            if s == 0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s
            # evaluate
            val = 2.0 * n * _full_convolution(trial).max()
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[n={n} gen={generation} i={i}] new best DE: {val:.6f}")
        if time.time() - last_improve > 10.0:
            # no improvement for a while, break
            break
    return best_a, best_val

Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

under a_i ≥ 0.  The routine combines three complementary optimisation
phases:

1. **Adam on a sub‑gradient of max(a∗a)** – the exact sub‑gradient is
   obtained from the convolution index where the maximum occurs.
2. **Differential‑Evolution (DE) global search** – a small population
   explores the simplex; individuals are generated, clipped to ≥0,
   renormalised and kept if they improve the exact objective.
3. **(1+1)‑Evolution‑Strategy (ES) fine‑tuning** – Gaussian perturbations
   of the current best are accepted when they improve the score.

The algorithm respects the supplied time‑budget (default 1000 s), works
for any length up to several thousand entries and keeps the best
sequence found.  It also warm‑starts from the previously discovered
global variable ``height_sequence_1`` if it exists.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all pure, top‑level (no closures)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – maps logits to a probability vector (sum=1)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a real array.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence for an arbitrary vector.
    Equivalent to 2·n·max(a∗a) / (sum(a)²).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


def grad_submax(a: np.ndarray):
    """
    Sub‑gradient of max(a∗a) w.r.t. a.
    Returns (gradient, current max value).
    The gradient follows the exact formula ∂max/∂a_j = 2·a_{k⁠*‑j},
    where k* is a convolution index attaining the maximum.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    # pick a random index among all maxima to avoid bias
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)

    # gradient g_j = 2 * a_{k-j} for all valid j
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def run_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
):
    """
    Adam optimiser on the sub‑gradient of max(a∗a) while keeping a on the simplex.
    Returns (best_a, best_value) discovered before *deadline*.
    """
    # initialise logits from the probability vector
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥0, Σa = 1
        grad_a, max_c = grad_submax(a)     # exact sub‑gradient
        val = 2.0 * n * max_c               # exact objective (sum(a)=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits in a reasonable range to avoid overflow
        x = np.clip(x, -50.0, 50.0)

        # shake when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    if best_a is None:                      # fallback (should not happen)
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val


def run_de_opt(
    n: int,
    init_pop: list,
    deadline: float,
    popsize: int = 10,
    max_gens: int = 20_000,
):
    """
    Simple Differential Evolution on the simplex.
    Returns (best_a, best_value) found before *deadline*.
    """
    # ------------------------------------------------------------------
    # initialise population
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    # evaluate
    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8   # differential weight
    CR = 0.9  # crossover probability
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)               # enforce non‑negativity

            # binomial crossover (ensure at least one mutant component)
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        # stop early if no improvement for a while
        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
):
    """
    (1+1) Evolution Strategy – simple Gaussian perturbations with
    automatic step‑size control.  Operates directly on the simplex.
    Returns the best vector discovered before *deadline*.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02                     # initial mutation strength
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                               # success → keep
            sigma = min(sigma * 1.05, 0.2)           # slightly increase step size
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                           # decay on failure

        # occasional reset if stuck
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The function respects a total time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq = None
    best_val = np.inf

    # lengths – larger n gives more freedom, but also more expensive.
    # We start with the largest feasible size and fall back to smaller ones
    # if time runs out.
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ---------- build a few sensible initial candidates ----------
        init_candidates = []

        # warm‑start from the globally best sequence (if length matches)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        # use the provided historic construction if available
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.01:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # monotone decreasing (linear) shape
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        # uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # --- Adam optimisation (multiple restarts) ---
        restarts = 4
        for r in range(restarts):
            if time.time() > deadline:
                break
            if r < len(init_candidates):
                a0 = init_candidates[r]
            else:
                a0 = np.random.dirichlet(np.ones(n))

            # allocate a modest time slice for each Adam run
            slice_time = min((deadline - time.time()) * 0.15, 30.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_adam_opt(n, a0, sub_deadline)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} restart={r}] GLOBAL best improved to {best_val:.6f}")

        # --- Differential Evolution (global search) ---
        if time.time() < deadline:
            de_time = min((deadline - time.time()) * 0.35, 120.0)
            sub_deadline = time.time() + de_time
            de_pop = init_candidates[:3]    # a small warm start set
            a_de, val_de = run_de_opt(n, de_pop, sub_deadline)
            if val_de < best_val:
                best_val = val_de
                best_seq = a_de.copy()
                print(f"[n={n}] DE GLOBAL best now {best_val:.6f}")

        # --- (1+1) ES fine‑tuning ---
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.30, 60.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n}] ES GLOBAL best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – return a simple uniform vector if nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
