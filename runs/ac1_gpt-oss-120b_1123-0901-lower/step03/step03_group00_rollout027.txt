def simplex_projection(v):
    # v is np.ndarray of shape (n,)
    # returns projection onto probability simplex.
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho
    rho = np.where(u - cssv/ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) using the FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    ``fft_len`` must be a power of two ≥ len(w)+len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.shape[0]) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def _logits_from_probs(p: np.ndarray) -> np.ndarray:
    """Convert a probability vector to unconstrained logits."""
    return np.log(p + 1e-15)


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed ≈ 1)."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # denominator ~ 1 because we keep the vector normalised
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Return a sequence (list of non‑negative floats) that minimises the
    evaluation function.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]   # larger sizes give more freedom
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)   # len(w)=2n-1, len(a)=n → len=3n-2

        # number of independent restarts for this length
        restarts = 5

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------- initialise logits -------------------------
            if r == 0 and best_seq is not None and len(best_seq) == n:
                # warm‑start from the best sequence found so far (if size matches)
                init_a = np.array(best_seq, dtype=np.float64)
                init_a /= init_a.sum()
                x = _logits_from_probs(init_a)
                print("[restart 0] warm start from previous best")
            elif "height_sequence_1" in globals():
                # try to use the historic construction (trim/pad as needed)
                cand = np.array(globals()["height_sequence_1"], dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= s
                x = _logits_from_probs(cand)
                print("[restart 0] start from height_sequence_1")
            else:
                # pure random start
                x = np.random.randn(n) * 0.1

            # ------------------- Adam state -------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.06

            # smooth‑max temperature λ
            lam = 0.5
            lam_factor = 1.03
            max_lam = 1e5

            t = 0
            last_improve = time.time()
            max_iters = 40000

            while time.time() < deadline and t < max_iters:
                # ----- logits → probabilities (softmax) -----
                max_x = np.max(x)
                e_x = np.exp(x - max_x)
                a = e_x / e_x.sum()   # a ≥ 0, Σa = 1

                # ----- self‑convolution -----
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()

                # ----- smooth‑max weights -----
                w = np.exp(lam * (conv - max_c))
                w /= w.sum()          # Σw = 1

                # ----- gradient of the surrogate w.r.t. a -----
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # ----- back‑prop through softmax (gradient w.r.t. logits) -----
                avg = np.dot(a, grad_a)               # ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)           # ∇_x φ

                # ----- Adam update -----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule -----
                lam = min(lam * lam_factor, max_lam)

                # ----- exact objective (for bookkeeping) -----
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

                # ----- stagnation handling -----
                if time.time() - last_improve > 4.0:
                    # Perturb logits a little to escape flat regions
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

                # ----- optional learning‑rate decay -----
                if t % 5000 == 0:
                    lr *= 0.9

    # ------------------------------------------------------------------
    # Phase 2 – tiny Differential Evolution (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        n = len(best_seq)
        conv_fft_len = _next_pow2(2 * n - 1)
        pop_size = 10
        pop = [best_seq.copy()]
        for _ in range(pop_size - 1):
            pop.append(np.random.dirichlet(np.ones(n)))
        best_pop = best_seq.copy()
        best_pop_val = best_val

        F = 0.8
        CR = 0.9

        print("\n--- Differential Evolution refinement ---")
        while time.time() < deadline - 5.0:
            for i in range(pop_size):
                # choose three distinct individuals other than i
                idxs = list(range(pop_size))
                idxs.remove(i)
                a, b, c = pop[np.random.choice(idxs, 3, replace=False)]

                mutant = a + F * (b - c)
                mutant = np.maximum(mutant, 0.0)
                mutant = _simplex_projection(mutant)

                # binomial crossover
                trial = np.where(np.random.rand(n) < CR, mutant, pop[i])
                trial = _simplex_projection(trial)

                trial_val = _objective(trial, conv_fft_len)
                target_val = _objective(pop[i], conv_fft_len)

                if trial_val < target_val:
                    pop[i] = trial
                    if trial_val < best_pop_val:
                        best_pop_val = trial_val
                        best_pop = trial.copy()
                        print(f"[DE] improved to {best_pop_val:.6f}")

                if time.time() >= deadline - 5.0:
                    break
            # early exit if population stagnates
            if time.time() - last_improve > 15.0:
                break

        # adopt DE result if it beats the current best
        if best_pop_val < best_val:
            best_val = best_pop_val
            best_seq = best_pop.copy()
            print(f"[DE] global best updated to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simple hill climbing / mass transfer
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = len(best_seq)
        conv_fft_len = _next_pow2(2 * n - 1)
        a = best_seq.copy()
        cur_val = best_val
        print("\n--- Local hill‑climbing ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] == 0.0:
                continue
            # transfer up to 5 % of the mass at i to j
            delta = a[i] * (np.random.rand() * 0.05)
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # (still lies on the simplex)
            new_val = _objective(a_new, conv_fft_len)
            if new_val < cur_val:
                a, cur_val = a_new, new_val
                best_seq, best_val = a.copy(), cur_val
                print(f"[hill] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – extremely unlikely to be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\n=== Finished. Best value = {best_val:.6f} (length {len(best_seq)}) ===")
    return best_seq.tolist()
