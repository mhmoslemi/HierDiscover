"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

which, after the normalization Σa_i = √(2 n), is exactly the
maximum value of the self‑convolution.  The optimizer uses a
projected sub‑gradient method on the *exact* max (no smooth‑max
approximation).  At each iteration the active set of indices where
the convolution attains its current maximum is built, a sub‑gradient
is obtained as a weighted cross‑correlation and the sequence is
updated with a simple line‑search.  After each step the vector is
projected back onto the simplex {x ≥ 0, Σx = √(2 n)}.  Many independent
restarts with random lengths and diverse initial shapes (power‑law,
exponential, warm‑starts) are performed within the time budget.
The best sequence found is returned as a plain Python list.

The algorithm improves on the previous soft‑max based code by:
* using the true max‑subgradient (sharper descent direction);
* an explicit back‑tracking line‑search that guarantees a monotone
  decrease of the objective;
* a modest momentum term;
* more aggressive exploration of the space of lengths and
  initialisations.

It runs entirely with NumPy and the provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (sum = s, non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Optimisation of a single length using exact max‑subgradient descent
# ----------------------------------------------------------------------
def _optimise_one_length_subgradient(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    step_factor: float = 0.8,
    step_decay: float = 0.9995,
    min_step_factor: float = 1e-4,
    momentum_beta: float = 0.8,
    epsilon_active: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for a fixed length `n`.
    Returns the best sequence (as a NumPy array) and its evaluation
    obtained from ``evaluate_sequence``.
    """
    target_sum = np.sqrt(2.0 * n)                 # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # best so far (objective equals the maximum convolution value)
    conv = _full_convolution_fft(a, a)
    best_score = conv.max()
    best_seq = a.copy()

    # momentum buffer
    velocity = np.zeros_like(a)

    iteration = 0
    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # ---- active set of indices where the max is attained ----
        active = np.where(conv >= max_val - epsilon_active)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / float(active.size)          # uniform weights over the active set

        # ---- sub‑gradient of the weighted max ----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)       # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # mapping to obtain Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # ---- momentum ----
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            direction = velocity
        else:
            direction = grad

        # ---- step‑size (scaled so that the update magnitude is comparable to max_val) ----
        max_abs = np.max(np.abs(direction)) + 1e-12
        eta = step_factor * max_val / max_abs

        # ---- trial update with projection onto the simplex ----
        a_candidate = a - eta * direction
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        # ---- evaluate candidate ----
        conv_candidate = _full_convolution_fft(a_candidate, a_candidate)
        score_candidate = conv_candidate.max()

        # ---- accept if we made progress, otherwise shrink step ----
        if score_candidate < best_score - 1e-12:
            a = a_candidate
            best_seq = a_candidate.copy()
            best_score = score_candidate
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")
        else:
            # no improvement – reduce the step factor
            step_factor *= 0.5
            if step_factor < min_step_factor:
                step_factor = min_step_factor
            # Occasionally give the algorithm a tiny nudge to escape plateaus
            if np.random.rand() < 0.01:
                noise = np.random.randn(n) * (target_sum * 5e-4 / n)
                a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # continue without updating a (the current a stays unchanged)
            continue

        # ---- decay the (global) step factor and proceed ----
        step_factor *= step_decay
        iteration += 1

        # occasional random shake‑up to explore new regions
        if np.random.rand() < 0.005:
            shake = np.random.randn(n) * (target_sum * 1e-3 / n)
            a = _project_onto_simplex(np.maximum(a + shake, 0.0), target_sum)
            step_factor *= 0.9

    # Final evaluation using the provided evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score
