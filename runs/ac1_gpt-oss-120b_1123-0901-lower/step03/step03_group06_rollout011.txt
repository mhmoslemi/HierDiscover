def correlation_fft(p, w, N):
    # w length L = len(w)
    # Compute conv_rev = ifft(fft(p) * fft(w_rev))
    # conv_rev length N
    # Then rev = conv_rev[L-1 : L-1 + n] (i.e., indices L-1 to L-1+n-1)
    P = np.fft.rfft(p, n=N)
    W = np.fft.rfft(w[::-1], n=N)
    conv = np.fft.irfft(P * W, n=N)
    L = len(w)
    rev = conv[L-1 : L-1 + len(p)]
    return rev

"""
Propose a low‑peak self‑convolution step function.

The core optimizer is an exponentiated‑gradient (mirror descent on the
simplex) with a few enhancements over the reference implementation:

* Adam‑style momentum on the log‑probabilities.
* A more aggressive temperature schedule (β grows up to 30).
* Periodic jitter in log‑space to escape flat plateaus.
* Multi‑scale schedule with several random restarts per length.
* A cheap local “hill‑climbing” phase after the global search.

All operations are done on the probability vector `p` (Σp = 1), because
the evaluation function simplifies to  ``2·n·max(p∗p)`` after this
normalisation.  The final step heights are recovered by scaling with
√(2 n).
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Return the full (linear) self‑convolution of `p`.
    Result has length 2·len(p)‑1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k * p_{k-i}   for i = 0 … n‑1.
    This is a correlation of `p` with the weight vector `w`.
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    # slice the linear part and reverse the order as explained in the
    # analysis section.
    rev = conv_rev[L - n : L][::-1]
    return rev


def _objective(p: np.ndarray, n: int) -> float:
    """Return 2·n·max(p∗p).  This equals the value returned by evaluate_sequence
    after the standard normalisation (Σa = √(2n))."""
    N = _next_pow2(3 * n - 2)
    conv = _full_convolution_fft(p, N)
    return 2.0 * n * np.max(conv)

    n: int,

    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:

    Parameters

        Length of the vector.

        Initial probability vector (sum = 1, shape (n,)).

        Absolute time after which the routine must stop.

        the deadline is reached).

        Probability vector achieving the lowest objective seen during the run.
    best_score : float
        Corresponding value of 2·n·max(p∗p).
    """
    # --------------------- hyper‑parameters -------------------------
    beta_start = 0.1
    beta_end = 30.0                # final temperature – larger than the reference
    beta_len = max_iters // 2      # linear schedule length
    jitter_interval = 3000
    jitter_scale = 0.002
    eta0 = 1.0 / (4.0 * n)         # same scale as the reference code
    eta_decay = 1e-4
    momentum = 0.9                 # Adam‑style momentum factor
    eps = 1e-12                    # for numerical stability
    # -----------------------------------------------------------------

    p = p_start.copy()
    logp = np.log(p + eps)         # work in log‑space (mirror descent)
    v = np.zeros_like(p)          # velocity for momentum

    N = _next_pow2(3 * n - 2)      # FFT size that covers both convolutions

    best_score = float("inf")
    best_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n:5d} it {it:7d}] new best = {cur_score:.6f}")

        # ----- soft‑max weights (smooth max) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))          # w_max = 1
        # normalise for a proper probability‑like weight (helps conditioning)
        w /= w.sum()

        # ----- correlation (sub‑gradient of the smoothed max) -----
        rev = _weighted_rev_fft(p, w, N)

        # gradient of 2·n·M_β w.r.t. p
        grad = 4.0 * n * rev

        # ----- Adam‑style momentum in log‑space -----
        v = momentum * v - eta0 / (1.0 + it * eta_decay) * grad
        logp += v

        # ----- jitter – occasional random perturbation -----
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex (soft‑max) -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)     # avoid overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs a multi‑scale EG optimiser with several restarts and a
    final local refinement.  It respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Load a previously discovered construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # fall back to a modest uniform start
        base = np.full(500, 1.0, dtype=float)

    # Normalise to the canonical sum S = √(2·n)  (probability vector)
    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    p_best = base / S0                     # Σp = 1
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Multi‑scale schedule.  The lengths are chosen to span small to large
    # while keeping the overall runtime reasonable.
    length_schedule = [500, 1000, 1500, 2000, 2500, 3000, 5000, 8000]
    length_schedule = [l for l in length_schedule if 300 <= l <= 8000]
    length_schedule = sorted(set(length_schedule))

    # Main optimisation loop over the chosen lengths.
    for idx, n in enumerate(length_schedule):
        if time.time() >= deadline:
            break

        # Estimate how much time we can still spend on this length.
        remaining_lengths = len(length_schedule) - idx
        time_left = deadline - time.time()
        per_len_budget = max(5.0, time_left / remaining_lengths)   # at least a few seconds

        # Decide the number of restarts (more restarts for the smaller lengths).
        if n <= 2000:
            n_restarts = 4
        elif n <= 5000:
            n_restarts = 3
        else:
            n_restarts = 2

        print(f"[run] length={n}, allowed time≈{per_len_budget:.1f}s, restarts={n_restarts}")

        # Prepare a base probability vector to be interpolated.
        p_global = best_seq / math.sqrt(2.0 * best_seq.shape[0])

        for r in range(n_restarts):
            if time.time() >= deadline:
                break

            # ----- initialise p for this restart -----
            if r == 0:
                # deterministic up‑sample of the current global best
                old_x = np.linspace(0.0, 1.0, p_global.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                p_start = np.interp(new_x, old_x, p_global)
                p_start = np.maximum(p_start, 0.0)
                p_start /= p_start.sum()
            elif r == 1:
                # a fresh random Dirichlet vector
                rnd = np.random.rand(n)
                p_start = rnd / rnd.sum()
            else:
                # jittered copy of the up‑sampled best
                p_start = p_global.copy()
                if p_start.shape[0] != n:
                    old_x = np.linspace(0.0, 1.0, p_start.shape[0])
                    new_x = np.linspace(0.0, 1.0, n)
                    p_start = np.interp(new_x, old_x, p_start)
                # add a small Gaussian perturbation and re‑normalise
                p_start += 0.02 * np.random.randn(n)
                p_start = np.maximum(p_start, 0.0)
                p_start /= p_start.sum()

            # ----- run the optimiser for this (length, restart) pair -----
            local_deadline = min(deadline, time.time() + per_len_budget)
            p_opt, _ = _optimize_eg_length(
                n=n,
                p_start=p_start,
                deadline=local_deadline,
                max_iters=250_000,
            )

            # ----- evaluate the candidate (full evaluation) -----
            a_opt = p_opt * math.sqrt(2.0 * n)          # scale back to step heights
            cur_score = evaluate_sequence(a_opt.tolist())

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                p_best = p_opt.copy()
                print(f"[len {n}] NEW GLOBAL BEST = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final local refinement: small random mass transfers on the best vector.
    # This is cheap and often squeezes a little more performance out.
    if time.time() < deadline:
        n = best_seq.shape[0]
        p = best_seq / math.sqrt(2.0 * n)   # back to probability domain
        cur_score = best_score
        max_local_trials = 20000
        trial = 0
        while time.time() < deadline and trial < max_local_trials:
            trial += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a tiny fraction of mass from i to j
            delta = 0.005 * p[i]               # 0.5 % of the source mass
            if delta <= 1e-12:
                continue
            p_new = p.copy()
            p_new[i] -= delta
            p_new[j] += delta
            # ensure non‑negativity (should already hold)
            p_new = np.maximum(p_new, 0.0)
            p_new /= p_new.sum()

            # fast objective (no clipping)
            new_score = _objective(p_new, n)

            # accept if strictly better
            if new_score < cur_score:
                p = p_new
                cur_score = new_score
                best_seq = p * math.sqrt(2.0 * n)
                best_score = evaluate_sequence(best_seq.tolist())
                print(f"[local] improvement at trial {trial}: {best_score:.6f}")

    # --------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Convert to plain Python list as required by the harness.
    return best_seq.tolist()
