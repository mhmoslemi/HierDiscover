"""
A two‑stage optimiser for the step‑function bound.
Stage 1: a smoothed‑max Adam optimiser (the same idea as the previous
solution) runs a limited number of iterations for several lengths.
Stage 2: a greedy coordinate‑descent / simulated‑annealing refinement
that keeps the L¹‑norm fixed by moving a small amount of mass from a
large entry to a small entry.  Convolutions are updated with FFTs, and
full recomputation is performed periodically to avoid drift.
The best sequence found before the 1000 s wall‑clock limit is returned.
"""

import time
import math
import numpy as np

def _softmax_weights(conv: np.ndarray, tau: float):
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau

def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev

    """Stable soft‑max of convolution values."""
    maxc = conv.max()

    """Compute rev_i = Σ_k w_k a_{k‑i} via a single FFT."""

    The exact objective value (max (a∗a)) is used to update the global best.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 4000

    tau_factor = 0.20
    decay_rate = 25000.0

    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and iteration < max_iters:
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        rev = _weighted_rev(a, w, Nfft)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _coordinate_refine(a: np.ndarray, deadline: float):
    """
    Greedy coordinate‑descent with occasional full recomputation.
    Moves a small amount of mass from a big entry to a small entry while
    preserving the L¹ norm; accepts only improving steps (with a tiny
    probability of uphill moves to escape plateaus).
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    Nfft = 1 << ((2 * n - 1).bit_length())

    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    cur_max = float(conv.max())
    best_seq = a.copy()
    best_score = cur_max

    iteration = 0
    max_iters = 500_000
    recompute_every = 5000
    T0 = cur_max * 0.02

    while time.time() < deadline and iteration < max_iters:
        probs = a / a.sum()
        i = np.random.choice(n, p=probs)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        step_factor = np.random.uniform(0.001, 0.02)
        delta = step_factor * a[i]
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue

        A_fft_new = np.fft.rfft(a_new, n=Nfft)
        conv_new = np.fft.irfft(A_fft_new * A_fft_new, n=Nfft)[: 2 * n - 1]
        new_max = float(conv_new.max())

        improve = new_max < cur_max
        temperature = T0 * max(0.0, 1.0 - iteration / max_iters)
        accept = improve or (np.random.rand() < math.exp(-(new_max - cur_max) / (temperature + 1e-12)))
        if accept:
            a = a_new
            conv = conv_new
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_seq = a.copy()
                print(f"[refine] iter {iteration:7d} new best = {best_score:.6f}")

        iteration += 1

        if iteration % recompute_every == 0:
            A_fft = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
            cur_max = float(conv.max())
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[refine] recompute iter {iteration:7d} new best = {best_score:.6f}")

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load the previously discovered construction, fall back to uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.full(2100, 1.0, dtype=float)

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – original and a couple of larger multiples.
    lengths = sorted({
        n0,
        min(8000, int(n0 * 1.5)),
        min(8000, int(n0 * 2.0)),
    })
    lengths = [l for l in lengths if l >= 300]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Initialise by linear interpolation of the base shape.
        if n == n0:
            init_a = base.copy()
        else:
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            init_a = np.interp(new_x, old_x, base)
            init_a = np.maximum(init_a, 0.0)

        # Normalise the L¹‑norm.
        target_sum = math.sqrt(2.0 * n)
        init_a = init_a * (target_sum / init_a.sum())

        # ------------------------------------------------------------------
        # Stage 1 – smoothed‑max Adam optimisation.
        max_iters = int(2_000_000 * n0 / n)
        max_iters = max(1_000, max_iters)
        print(f"[run] length={n}, Adam iters≈{max_iters}")

        best_seq, best_score = _optimize_length(
            n=n,
            init_a=init_a,
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
            max_iters=max_iters,
        )

        # ------------------------------------------------------------------
        # Stage 2 – coordinate‑descent refinement.
        time_left = deadline - time.time()
        if time_left > 5.0:
            refine_deadline = time.time() + min(30.0, time_left / 2)
            refined_seq, refined_score = _coordinate_refine(best_seq.copy(), refine_deadline)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"[final] length={n} after refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen).
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
