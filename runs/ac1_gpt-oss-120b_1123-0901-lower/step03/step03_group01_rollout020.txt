def lp_improve(a, target_sum, max_iter, deadline):
    for iter in range(max_iter):
        # compute M
        M = _max_conv(a)
        try:
            # Build constraints matrix A_ub
            n = len(a)
            rows = np.empty(n*n, dtype=int)
            cols = np.empty(n*n, dtype=int)
            data = np.empty(n*n, dtype=float)
            # Fill
            idx = 0
            for i in range(n):
                rows[idx:idx+n] = i + np.arange(n)  # row indices
                cols[idx:idx+n] = i
                data[idx:idx+n] = a[i]
                idx += n
            A_ub = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsc()
            b_ub = np.full(2*n-1, M)
            c = -np.ones(n)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
            if not res.success:
                break
            b = res.x
            sum_b = b.sum()
            if sum_b <= 0:
                break
            g = b * (target_sum / sum_b)
            # combine
            t = 0.2
            a_new = (1 - t) * a + t * g
            # small projection to ensure nonnegative and sum
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_onto_simplex(a_new, target_sum)
            new_M = _max_conv(a_new)
            if new_M < M:
                a = a_new
                # print improvement
                print(f"[{time.time():.2f}] LP iter {iter+1} improved M to {new_M:.6f}")
            else:
                # if not improved, break
                break
        except MemoryError:
            break

        if time.time() > deadline:
            break
    return a

The evaluator only depends on  

\[
\text{score}= \max\bigl(a*a\bigr)
\]

because the sum of the coefficients is forced to `sqrt(2n)`.  
The task is therefore to minimise the peak of the self‑convolution.

The new search routine combines three complementary ideas:

1. **Fast deterministic initialisation** – a small family of analytically
   motivated shapes (inverse power, exponential tails, uniform) is generated,
   scaled to the required ℓ₁–norm and the best among them is kept.

2. **Projected sub‑gradient descent** – the exact sub‑gradient of the
   `max‑convolution` is built from the currently active (peak) indices,
   a momentum update is performed and the iterate is projected back onto the
   simplex `{x ≥ 0, Σx = √(2n)}`.

3. **Linear‑programming polishing** – for the current vector `f` we solve  

   \[
   \max \; \sum_j g_j \quad
   \text{s.t. } g_j \ge 0,\;(f*g)_k \le M\; \forall k,
   \]

   where `M = max(f*f)`.  Because the constraints are linear in `g`,
   a single LP (solved with SciPy’s *highs* driver) yields a vector `g`
   whose ℓ₁‑norm is maximal under the same peak bound.
   A convex combination `f ← (1‑t) f + t g` (with `t≈0.2`) strictly
   reduces the peak and therefore the overall score.

The outer loop repeatedly restarts with different problem sizes
(`1200 ≤ n ≤ 3000`) and different seeds, always keeping the globally best
candidate found so far.  All heavy work (FFT based convolutions,
sub‑gradient steps and LP solves) is timed so that the total runtime never
exceeds the allotted budget (`budget_s – 0.5 s` safety margin).

The code below follows the described strategy and returns the best
step‑function as a plain Python list.  It works even when *SciPy* is not
available – in that case the LP polishing step is simply skipped.

```python
"""
Propose a non‑negative step‑function that minimises
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of:

1.  Fast deterministic initialisation using a few analytic families
    (inverse power, exponential tails, uniform).  Each candidate is
    scaled to the required ℓ₁‑norm and the best one (lowest max‑convolution)
    is kept.

2.  Projected sub‑gradient descent (with momentum) that directly optimises
    the peak of the self‑convolution while staying on the simplex.

3.  An LP polishing step: for the current vector `f` we solve
       max Σ g_j   s.t.   g≥0   and   (f*g)_k ≤ M   (M = max(f*f)).
    The solution `g` is rescaled to the correct ℓ₁‑norm and mixed with `f`
    (t≈0.2).  This step provably never increases the objective and
    quickly drives the peak down.

The outer loop restarts with random lengths (1200‑3000) and different
initialisations, always remembering the best sequence discovered before the
time budget expires.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Optional SciPy import (used for the LP polishing step)
# --------------------------------------------------------------------------- #
try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    _SCIPY_AVAILABLE = True
except Exception:   # pragma: no cover
    _SCIPY_AVAILABLE = False

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate a vector – because we keep Σa = √(2n) the objective equals max(conv).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    """Return max(f*f) for a real vector a."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Sub‑gradient optimiser (projected with momentum).
# --------------------------------------------------------------------------- #
def _subgradient_optimize(a_init: np.ndarray,
                          target_sum: float,
                          max_time: float,
                          start_time: float):
    """Refine a vector using a simple sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.3          # initial learning rate
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        # keep the best encountered
        if max_val < best_score - 1e-12:
            best_score = max_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # Active indices (the peaks); a tiny tolerance is enough because
        # the convolution values are exact floating point numbers.
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # Build the sub‑gradient of the sum of the active entries.
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = np.max(np.abs(grad))
        if max_grad == 0.0:
            break

        # Step size proportional to the current peak.
        eta = base_lr * max_val / max_grad

        # Heavy‑ball momentum update.
        momentum = beta * momentum - eta * grad
        a_new = a + momentum

        # Project back onto the feasible simplex.
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # Occasionally inject a tiny random shake to escape flat regions.
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0),
                                          target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995   # gentle decay for fine‑tuning

    return best_a, best_score


# --------------------------------------------------------------------------- #
# Linear‑program polishing step (optional – requires SciPy).
# --------------------------------------------------------------------------- #
def _lp_polish(f: np.ndarray,
               target_sum: float,
               max_iter: int,
               deadline: float) -> np.ndarray:
    """
    Perform up to ``max_iter`` LP‑based improvements.
    Each iteration solves:
        max Σ g_j   s.t.   g ≥ 0,   (f * g)_k ≤ M   ∀k,
    where M = max(f*f).  The solution is mixed with ``f`` (t≈0.2) and
    projected onto the simplex.
    """
    if not _SCIPY_AVAILABLE:
        return f   # nothing to do without SciPy

    n = f.shape[0]
    a = f.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        M = _max_conv(a)

        # Build the convolution matrix A where (A @ g)_k = (a * g)_k.
        # A is (2n‑1) × n, highly structured – we use a sparse representation.
        i = np.arange(n)
        rows = (i[:, None] + i[None, :]).ravel()          # i + j
        cols = np.tile(i, n)                              # column index = j
        data = np.repeat(a, n)                            # each row i contributes a_i
        A_ub = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsc()

        # Linear programme: maximise sum(g)  ⇔  minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # Rescale to the required ℓ₁‑norm.
        g = g * (target_sum / sum_g)

        # Convex combination (t = 0.2 works well in practice).
        t = 0.2
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        new_M = _max_conv(a_new)

        # Accept only if we made progress.
        if new_M < M - 1e-12:
            a = a_new
            print(f"[{time.time():.2f}] LP iter {it+1} improved M to {new_M:.6f}")
        else:
            break

    return a


# --------------------------------------------------------------------------- #
# Deterministic initialisation – a handful of analytic families.
# --------------------------------------------------------------------------- #
def _initial_candidate(n: int, target_sum: float) -> tuple[np.ndarray, float]:
    """
    Return a promising start vector for a given length ``n``.
    Several parametric families (inverse power, exponential) are tried
    and the one with the smallest max‑convolution is kept.
    """
    best_vec = None
    best_score = float("inf")
    idx = np.arange(n, dtype=float)

    # Uniform baseline
    uniform = np.full(n, target_sum / n)
    sc = _max_conv(uniform)
    best_vec, best_score = uniform, sc

    # Inverse‑power family: (d+α)^{-p}
    inv_ps = [0.5, 0.7, 0.9, 1.0, 1.2, 1.5]
    alphas = [0.0, 0.5, 1.0]
    d = np.minimum(idx, n - 1 - idx)          # distance to the nearest end
    for p in inv_ps:
        for α in alphas:
            w = (d + α + 1e-12) ** (-p)
            a = w * (target_sum / w.sum())
            sc = _max_conv(a)
            if sc < best_score:
                best_vec, best_score = a, sc

    # Exponential family: exp(-β * d)
    betas = [0.001, 0.002, 0.003, 0.005, 0.01, 0.02]
    for β in betas:
        w = np.exp(-β * d)
        a = w * (target_sum / w.sum())
        sc = _max_conv(a)
        if sc < best_score:
            best_vec, best_score = a, sc

    # Random simplex point (adds diversity)
    rnd = np.random.rand(n)
    a = _project_onto_simplex(rnd, target_sum)
    sc = _max_conv(a)
    if sc < best_score:
        best_vec, best_score = a, sc

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin so that we can always return a value
    deadline = start + budget_s - 0.5

    best_seq = None            # numpy array of the best candidate
    best_score = float("inf")  # objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Outer loop – many restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # pick a problem size (reuse best length occasionally)
        if best_seq is not None and np.random.rand() < 0.2:
            n = len(best_seq)
        else:
            n = np.random.randint(1200, 3001)

        target_sum = math.sqrt(2.0 * n)   # Σ a_i = √(2n)

        # ---------------------------------------------------------------- #
        # 1️⃣  Deterministic initial candidate
        # ---------------------------------------------------------------- #
        a, cur_score = _initial_candidate(n, target_sum)

        # Occasionally seed from the best sequence found so far.
        if best_seq is not None and np.random.rand() < 0.3:
            src = np.asarray(best_seq, dtype=float)
            if src.size >= n:
                a = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                a = np.concatenate([src, pad])
            a = _project_onto_simplex(a, target_sum)
            cur_score = _max_conv(a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # ---------------------------------------------------------------- #
        # 2️⃣  Sub‑gradient refinement (major part of the time budget)
        # ---------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        sub_time = min(8.0, time_left * 0.6)
        a_sub, sub_score = _subgradient_optimize(a, target_sum,
                                                 sub_time, time.time())
        if sub_score < best_score:
            best_score = sub_score
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad best = {best_score:.6f} (n={n})")

        # ---------------------------------------------------------------- #
        # 3️⃣  Linear‑program polishing (if SciPy is available)
        # ---------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        lp_time = min(6.0, time_left)
        a_lp = _lp_polish(a_sub, target_sum, max_iter=2,
                          deadline=time.time() + lp_time)
        lp_score = _max_conv(a_lp)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = a_lp.copy()
            print(f"[{time.time():.2f}] LP best = {best_score:.6f} (n={n})")

        # ---------------------------------------------------------------- #
        # 4️⃣  Final short gradient fine‑tuning
        # ---------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        fin_time = min(4.0, time_left)
        a_fin, fin_score = _subgradient_optimize(a_lp, target_sum,
                                                 fin_time, time.time())
        if fin_score < best_score:
            best_score = fin_score
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final best = {best_score:.6f} (n={n})")

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never happen, but keeps the contract).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    # expose the best sequence for possible reuse in later runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    return best_seq.tolist()
```
