"""
Propose a non‑negative step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where ``a*a`` is the linear convolution of ``a`` with itself.  The algorithm
combines three ideas:

1. **Fast parametric sweep** – simple families (exponential, power‑law) are
   evaluated on several lengths.  These give good starting points.

2. **Stochastic sub‑gradient Adam** – the exact max‑convolution index is
   obtained with a single FFT, the gradient of that single index is a
   cheap shift of the current vector; Adam updates are performed on the
   unconstrained logits of a soft‑max parametrisation (guaranteeing
   ``a_i ≥ 0`` and ``∑a_i = 1``).

3. **Multi‑scale warm‑starts** – after a length has been optimised the
   best vector is interpolated to a larger length, providing a warm start
   for the next optimisation phase.

The routine respects the supplied time budget (default 1000 s) and prints
progress.  If nothing is found a uniform fallback is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures)

def logits_from_probs(p: np.ndarray) -> np.ndarray:
    """Log‑it conversion; p must be non‑negative and sum > 0."""
    p = np.maximum(p, 1e-15)           # avoid log(0)
    return np.log(p)


def probs_from_logits(x: np.ndarray) -> np.ndarray:
    """Soft‑max (numerically stable).  Returns a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def exact_objective(a: np.ndarray) -> float:
    """
    Compute the exact evaluation (the same as evaluate_sequence) for a
    vector whose entries are already non‑negative.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def grad_a_for_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the kth convolution coefficient w.r.t. a.
    For linear convolution:  ∂c_k/∂a_i = 2·a_{k-i}  (if 0≤k-i<n, else 0).
    """
    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    # i runs from lo..hi, j = k-i runs from k-lo .. k-hi (still in range)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if v.shape[0] == new_n:
        return v.copy()
    old_x = np.linspace(0.0, 1.0, v.shape[0])
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, v)
    up = np.maximum(up, 0.0)
    up_sum = up.sum()
    if up_sum < 1e-12:
        return np.full(new_n, 1.0 / new_n)
    return up / up_sum


def parametric_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i   (0<r<1) and normalised to sum 1."""
    a = np.power(r, np.arange(n, dtype=float))
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


def parametric_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)   (p>0) and normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Core optimiser (stochastic sub‑gradient Adam)

def adam_optimize(
    n: int,
    init_a: np.ndarray | None,
    deadline: float,
    best_val_ref: list,
    best_seq_ref: list,
    max_iters: int = 30000,
    lr: float = 0.1,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    stagnation_seconds: float = 5.0,
) -> None:
    """
    Optimise a length‑n vector using Adam on the logits of a soft‑max.
    The gradient is the sub‑gradient of the *max* convolution coefficient,
    i.e. the gradient for the single index achieving the current maximum.
    Updates are performed in‑place on the mutable containers
    ``best_val_ref`` and ``best_seq_ref`` (single‑element lists).
    """
    # Initialise logits
    if init_a is not None:
        a0 = np.maximum(init_a, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        x = logits_from_probs(a0)
    else:
        a0 = np.random.dirichlet(np.ones(n))
        x = logits_from_probs(a0)

    # Adam state
    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)

    t = 0
    last_improve = time.time()
    # Pre‑compute FFT length (used every iteration)
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))

    while time.time() < deadline and t < max_iters:
        # ---- logits → probabilities (soft‑max) ----
        a = probs_from_logits(x)          # guarantees a_i≥0, ∑a_i=1

        # ---- convolution (single FFT) ----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        idx_max = int(conv.argmax())

        # ---- sub‑gradient for the *max* coefficient ----
        grad_a = grad_a_for_index(a, idx_max)   # shape (n,)

        # ---- back‑prop through soft‑max ----
        avg = float(np.dot(a, grad_a))
        grad_x = a * (grad_a - avg)             # ∇_x φ

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- exact objective (no rounding errors) ----
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # ---- keep the best overall sequence ----
        if val < best_val_ref[0]:
            best_val_ref[0] = val
            best_seq_ref[0] = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={t:05d}] new best {val:.6f}")

        # ---- stagnation handling ----
        if time.time() - last_improve > stagnation_seconds:
            # small random shake to logits to escape plateaus
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[n={n} iter={t}] shake (stagnation)")

    # End of optimisation for this length


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the evaluation function within the given time budget.

    The method performs a quick parametric sweep, followed by a fast
    Adam‑based stochastic sub‑gradient optimiser on several problem sizes.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Containers for the best result found so far
    best_val = [float("inf")]
    best_seq = [None]   # will hold a numpy array

    # ------------------------------------------------------------------
    # 1) Parametric families (exponential & power‑law) on a set of lengths
    # ------------------------------------------------------------------
    param_lengths = [256, 512, 1024, 2048, 4096, 8192]
    for n in param_lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Parametric sweep for n={n} ---")

        # Exponential family (r close to 1 gives near‑uniform)
        for r in np.linspace(0.90, 0.9999, 40):
            a = parametric_exponential(n, r)
            val = exact_objective(a)
            if val < best_val[0]:
                best_val[0] = val
                best_seq[0] = a.copy()
                print(f"[exp r={r:.5f}] new best {val:.6f}")

        # Power‑law family
        for p in np.linspace(0.2, 3.0, 30):
            a = parametric_powerlaw(n, p)
            val = exact_objective(a)
            if val < best_val[0]:
                best_val[0] = val
                best_seq[0] = a.copy()
                print(f"[pow p={p:.3f}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Multi‑scale Adam optimisation with stochastic sub‑gradient
    # ------------------------------------------------------------------
    # Lengths are processed from small → large; larger problems inherit the
    # best solution from the previous size (interpolated to the new size).
    opt_lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    for n in opt_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Prepare a warm‑start candidate for this length:
        init_candidate = None
        if best_seq[0] is not None:
            if len(best_seq[0]) == n:
                init_candidate = best_seq[0]
            elif len(best_seq[0]) > n:
                init_candidate = best_seq[0][:n]
            else:
                init_candidate = upsample_vector(best_seq[0], n)

        # Use the previously supplied global (if any) as an alternative start
        if init_candidate is None and "height_sequence_1" in globals():
            cand = np.array(height_sequence_1, dtype=float)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s >= 0.01:
                init_candidate = cand / s

        # Number of independent restarts for this length
        restarts = 5
        for r in range(restarts):
            if time.time() > deadline:
                break
            print(f"[n={n} restart={r}] start")
            adam_optimize(
                n=n,
                init_a=init_candidate,
                deadline=deadline,
                best_val_ref=best_val,
                best_seq_ref=best_seq,
                max_iters=30000,
                lr=0.1,
                stagnation_seconds=5.0,
            )
            # After the first restart keep the same init (the best so far)
            init_candidate = best_seq[0]

    # ------------------------------------------------------------------
    # Fallback if nothing succeeded (should be rare)
    # ------------------------------------------------------------------
    if best_seq[0] is None:
        n_fallback = 2000
        best_seq[0] = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        best_val[0] = exact_objective(best_seq[0])
        print("Fallback to uniform vector.")

    print(f"\n*** BEST VALUE FOUND: {best_val[0]:.6f}  (length={len(best_seq[0])}) ***")
    # Convert to plain Python list for the external evaluator
    return best_seq[0].tolist()
