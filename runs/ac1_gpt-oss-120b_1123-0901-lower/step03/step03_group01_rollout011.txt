"""
Proposes a step‑function (non‑negative heights) minimizing
2*n*max(conv(a,a))/sum(a)^2.

Algorithm:
1. Randomly choose a length n and initialise a U‑shaped vector.
2. Perform a short simulated‑annealing phase to escape bad local minima.
3. Repeatedly solve the linear programme
      maximise   Σ b_j
      subject to b_j ≥ 0, (a * b)_k ≤ M
   where M = max(a*a).  The solution b maximises the total mass that can be
   added without raising any convolution entry above the current peak.
   After scaling b to the required ℓ₁‑norm we move a a fraction t towards b.
   This is the “alternating‑LP” step described in the task statement.
4. End with a few iterations of a projected sub‑gradient descent on the
   active convolution entries (the method used previously).
5. Keep the best vector seen over many independent restarts until the
   time budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution (a * a) using a real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv

def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_conv_full_fft(a).max())

def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Simple SA that moves a random fraction of mass between two positions."""
    a = a_init.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    T0 = max(0.01, cur_score * 0.05)
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        frac = np.random.rand() * 0.5
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)
        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        if delta_score <= 0.0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")
    return best_vec, best_score

def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Projected sub‑gradient descent on the active convolution entries."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score

def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Construct matrix A such that (A @ b) = a * b (full convolution)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # Fill column by column: column j contributes a shifted downwards.
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve  max Σ b_j  s.t. b ≥ 0 and (a * b)_k ≤ M.

    Returns the optimal b (numpy array) or None if the LP fails.
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)
    # Right‑hand side
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Objective (minimise -Σ b_j)
    c = -np.ones(n, dtype=float)

    # Bounds: b_j ≥ 0
    bounds = [(0, None)] * n

    # Use HiGHS interior‑point method, which accepts dense matrices.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    else:
        # If the solver fails we simply return None; the caller will abort the
        # LP phase for this restart.
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.

    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = float("inf")

    # Helper to evaluate a candidate (max convolution)
    def _evaluate(a):
        return _max_conv(a)

    # Pre‑fetch possible warm‑start from a previous run.
    warm_start = None
    if 'height_sequence_1' in globals():
        warm_start = np.asarray(height_sequence_1, dtype=float)

    while time.time() < deadline:
        # Choose a length. Occasionally reuse the length of the current best.
        if best_seq is not None and np.random.rand() < 0.2:
            n = len(best_seq)
        else:
            n = np.random.randint(600, 1801)   # 600 – 1800 seems a good trade‑off

        target_sum = np.sqrt(2.0 * n)

        # -------------------------------------------------------------
        # Initialise a candidate.
        # -------------------------------------------------------------
        if warm_start is not None and np.random.rand() < 0.5:
            if warm_start.size >= n:
                a0 = warm_start[:n].copy()
            else:
                pad = np.random.rand(n - warm_start.size) * 1e-3
                a0 = np.concatenate([warm_start, pad])
        else:
            # U‑shaped seed – larger near the ends, smaller in the centre.
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)
            a0 = np.random.rand(n) * weight

        a = _project_onto_simplex(a0.astype(float), target_sum)

        cur = _evaluate(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # -------------------------------------------------------------
        # Phase 1 – short simulated annealing to get out of very bad spots.
        # -------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        sa_time = min(4.0, time_left * 0.2)
        a_sa, score_sa = simulated_annealing(a, target_sum, sa_time, time.time())
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After SA best = {best_score:.6f} (n={n})")

        # -------------------------------------------------------------
        # Phase 2 – alternating linear‑program refinement.
        # -------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        lp_time = min(12.0, time_left * 0.6)   # allocate most of the remaining time
        lp_start = time.time()
        iter_no = 0
        a_lp = a_sa.copy()
        while time.time() - lp_start < lp_time:
            M = _max_conv(a_lp)
            b_opt = solve_lp_max_sum(a_lp, M)
            if b_opt is None:
                break
            sum_b = b_opt.sum()
            if sum_b <= 0:
                break
            # Rescale to the required ℓ₁‑norm.
            b_rescaled = b_opt * (target_sum / sum_b)

            # Convex combination – we keep a relatively aggressive step.
            t = 0.5
            a_lp = (1 - t) * a_lp + t * b_rescaled
            a_lp = _project_onto_simplex(a_lp, target_sum)

            cur_score = _max_conv(a_lp)
            iter_no += 1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_lp.copy()
                print(f"[{time.time():.2f}] LP iter {iter_no} best = {best_score:.6f} (n={n})")
            # simple stopping criterion: stop if improvement stalls
            if iter_no >= 30:
                break

        # -------------------------------------------------------------
        # Phase 3 – projected sub‑gradient descent for fine tuning.
        # -------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        gd_time = min(4.0, time_left)
        a_gd, gd_score = subgradient_optimize(a_lp, target_sum, gd_time, time.time())
        if gd_score < best_score:
            best_score = gd_score
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD final best = {best_score:.6f} (n={n})")

        # Early exit if we have already reached the target bound.
        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target achieved, stopping early.")
            break

    # -------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen).
    # -------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return plain Python list as required.
    return best_seq.tolist()

"""
Proposes a non‑negative step‑function that (approximately) minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

which, under the usual normalisation Σ a_i = √(2n), is exactly the maximum
entry of the self‑convolution.  The algorithm mixes three ideas:

1.  A short simulated‑annealing stage to escape very bad local minima.
2.  An alternating linear‑program refinement (the “LP‑step” from the
    description): for the current vector a with peak M = max(a*a) we solve

          maximise   Σ b_j
          subject to b_j ≥ 0,
                     (a * b)_k ≤ M  for all k,

    obtain a feasible b, scale it to the required ℓ₁‑norm and move a a
    fraction towards b.  Repeating this step rapidly lowers the peak.
3.  A projected sub‑gradient descent on the active convolution entries for
    fine‑grained polishing.

The whole procedure is repeatedly restarted with different lengths
(600‑1800 elements) and random initialisations; the best vector found
before the time budget expires is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate the objective (max of the self‑convolution).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass transfers between two positions.
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Return (best_vector, best_score) after a short SA phase."""
    a = a_init.copy()
    n = a.shape[0]

    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    T0 = max(0.01, cur_score * 0.05)   # start temperature (≈5 % of current value)
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        # move a random fraction (≤50 %) of the mass from i to j
        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        a_new = _project_onto_simplex(a_new, target_sum)
        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the active convolution entries.
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Refine a vector using a simple sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional noise to avoid stagnation
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Build the full convolution matrix A such that (A @ b) = a * b.
# --------------------------------------------------------------------------- #
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Return a dense matrix A with rows = 2·n‑1, cols = n."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a
    return A


# --------------------------------------------------------------------------- #
# Solve   max Σ b_j   s.t. b ≥ 0 and (a * b) ≤ M.
# --------------------------------------------------------------------------- #
def solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve the LP that maximises the ℓ₁‑norm of b under the convolution bound.
    Returns the optimal b (numpy array) or None if the solver fails.
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)
    # Slightly relax the RHS to avoid infeasibility caused by round‑off.
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=float)
    c = -np.ones(n, dtype=float)               # maximise Σb  ⇔ minimise −Σb
    bounds = [(0, None)] * n

    # HiGHS (default) is fast for dense problems.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    else:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None            # best candidate (numpy array)
    best_score = float("inf")  # best objective value seen so far

    # Optional warm‑start from a previously found construction.
    warm_start = None
    if 'height_sequence_1' in globals():
        warm_start = np.asarray(height_sequence_1, dtype=float)

    while time.time() < deadline:
        # ---------------------------------------------------------
        # Choose problem size.
        # ---------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.2:
            n = len(best_seq)
        else:
            n = np.random.randint(600, 1801)   # 600‑1800 elements is a good trade‑off

        target_sum = np.sqrt(2.0 * n)

        # ---------------------------------------------------------
        # Initialise a candidate.
        # ---------------------------------------------------------
        if warm_start is not None and np.random.rand() < 0.5:
            if warm_start.size >= n:
                a0 = warm_start[:n].copy()
            else:
                pad = np.random.rand(n - warm_start.size) * 1e-3
                a0 = np.concatenate([warm_start, pad])
        else:
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)   # U‑shaped seed
            a0 = np.random.rand(n) * weight

        a = _project_onto_simplex(a0.astype(float), target_sum)

        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # ---------------------------------------------------------
        # Phase 1 – short simulated annealing.
        # ---------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        sa_time = min(4.0, time_left * 0.2)
        a_sa, score_sa = simulated_annealing(a, target_sum, sa_time, time.time())
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After SA best = {best_score:.6f} (n={n})")

        # ---------------------------------------------------------
        # Phase 2 – alternating LP refinement.
        # ---------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        lp_time = min(12.0, time_left * 0.6)
        lp_start = time.time()
        a_lp = a_sa.copy()
        iter_no = 0
        max_lp_iters = 15                     # keep the LP phase cheap
        while time.time() - lp_start < lp_time and iter_no < max_lp_iters:
            M = _max_conv(a_lp)
            b_opt = solve_lp_max_sum(a_lp, M)
            if b_opt is None:
                break
            sum_b = b_opt.sum()
            if sum_b <= 0:
                break
            b_rescaled = b_opt * (target_sum / sum_b)

            t = 0.5                     # aggressive convex combination
            a_lp = (1 - t) * a_lp + t * b_rescaled
            a_lp = _project_onto_simplex(a_lp, target_sum)

            cur_score = _max_conv(a_lp)
            iter_no += 1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_lp.copy()
                print(f"[{time.time():.2f}] LP iter {iter_no} best = {best_score:.6f} (n={n})")

        # ---------------------------------------------------------
        # Phase 3 – projected sub‑gradient descent for fine tuning.
        # ---------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        gd_time = min(4.0, time_left)
        a_gd, gd_score = subgradient_optimize(a_lp, target_sum, gd_time, time.time())
        if gd_score < best_score:
            best_score = gd_score
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD final best = {best_score:.6f} (n={n})")

        # ---------------------------------------------------------
        # Early exit if the target bound has been reached.
        # ---------------------------------------------------------
        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target achieved, stopping early.")
            break

    # -------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed).
    # -------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
