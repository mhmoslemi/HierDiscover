    p = p_start.copy()
    logp = np.log(p + 1e-30)
    N = _next_pow2(3 * n - 2)
    best_local_score = float('inf')
    best_local_p = p.copy()
    it = 0
    beta_start = 0.1
    beta_end = 50.0 # or param
    beta_ramp = max_iters // 2 # increase over half iterations
    last_improve_iter = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # conv
        P = np.fft.rfft(p, n=N)
        conv = np.fft.irfft(P * P, n=N)[:2*n-1]
        M = conv.max()
        cur_score = 2.0 * n * M
        # track best
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            last_improve_iter = it
            print(f"[len {n} iter {it}] new local best = {cur_score:.6f}")
        # beta schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp)
        # softmax weights
        w = np.exp(beta * (conv - M))
        # rev via FFT correlation
        # compute convolution of p with reversed w
        W_rev = np.fft.rfft(w[::-1], n=N)
        rev_full = np.fft.irfft(P * W_rev, n=N)
        L = w.shape[0]  # 2n-1
        rev = rev_full[L - n : L][::-1]
        # gradient
        grad = 4.0 * n * rev
        # step size
        eta = (1.0/(4.0 * n)) / (1.0 + it * 1e-4)  # same as earlier
        logp -= eta * grad
        # jitter
        if jitter_interval and it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale
        # normalize to simplex: exponentiate shifted
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()
        # early stop if no improvement for a long time relative to best global maybe
        if it - last_improve_iter > 20000 and cur_score > best_score_global:
            # allow some patience
            break

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.
The search uses a multiscale exponentiated‑gradient (EG) optimiser on the
probability simplex.  For a given length n we optimise a probability vector
p (Σp = 1) and later scale it to a = √(2 n)·p, because with this normalisation
the evaluation reduces to ``2·n·max(p∗p)``.  The EG update is performed in
log‑space and the gradient of a smooth approximation to the max (soft‑max
with a linearly increasing temperature β) is obtained by one FFT for the
self‑convolution and a second FFT for the weighted correlation.  Several
initialisations (interpolated previous best, simple power‑law shapes and a
few random Dirichlet draws) are tried at each scale.  After the multiscale
phase a final high‑β refinement is performed on the best candidate found.
The routine respects the 1000 s time budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Full self‑convolution (p∗p) using a single FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i}  for i=0..n−1.
    This is a correlation of p with w and is obtained via FFT.
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep linear part
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _optimize_eg_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    beta_end: float = 50.0,
    max_iters: int = 300_000,
    jitter_interval: int = 5000,
    jitter_scale: float = 0.003,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector (sum = 1) and its objective
    ``2·n·max(p∗p)``.
    """
    # --- parameters ----------------------------------------------------
    beta_start = 0.1
    beta_ramp = max_iters // 2                     # linear increase of β
    # initialise
    p = p_start.copy()
    logp = np.log(p + 1e-30)                       # avoid log(0)
    N = _next_pow2(3 * n - 2)                      # size for both FFTs
    best_local_score = float("inf")
    best_local_p = p.copy()
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution (p∗p) ---------------------------------
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M                      # exact objective for p

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            last_improve = it
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- soft‑max weight (smooth approximation of max) ----------
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp)
        w = np.exp(beta * (conv - M))                # ≤ 1, max = 1

        # ----- weighted rev (correlation) -----------------------------
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smoothed max ---------------------------
        grad = 4.0 * n * rev                         # ∇_p (2·n·M_β)

        # ----- EG update in log‑space --------------------------------
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)   # same schedule as before
        logp -= eta * grad

        # occasional jitter to escape plateaus
        if jitter_interval and (it % jitter_interval == 0):
            logp += np.random.randn(n) * jitter_scale

        # ----- projection onto the simplex ----------------------------
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)           # avoid overflow
        p = exp_shift / exp_shift.sum()

        # early stop if stagnating for a long while
        if it - last_improve > 30_000 and cur_score > best_local_score * 1.001:
            break

    return best_local_p, best_local_score


# ----------------------------------------------------------------------
def _initial_probabilities(n: int, base_p: np.ndarray) -> list[np.ndarray]:
    """
    Produce a small collection of initial probability vectors for length n.
    Includes the interpolated base, a few power‑law shapes and a random
    Dirichlet draw.
    """
    inits = []

    # 1) Interpolated previous best (if lengths differ)
    old_x = np.linspace(0.0, 1.0, base_p.shape[0])
    new_x = np.linspace(0.0, 1.0, n)
    p_interp = np.interp(new_x, old_x, base_p)
    p_interp = np.maximum(p_interp, 0.0)
    p_interp /= p_interp.sum()
    inits.append(p_interp)

    # 2) Power‑law shapes (distance to nearest endpoint)
    d = np.minimum(np.arange(n) + 1, n - np.arange(n))
    for gamma in (0.2, 0.5, 1.0, 1.5):
        w = d.astype(float) ** (-gamma)
        w /= w.sum()
        inits.append(w)

    # 3) Random Dirichlet (few copies)
    for _ in range(2):
        rnd = np.random.dirichlet(np.ones(n))
        inits.append(rnd)

    return inits
