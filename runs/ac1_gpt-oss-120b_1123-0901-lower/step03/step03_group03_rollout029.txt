    n=len(a)

    Nfft = 1 << ((2 * n - 1).bit_length())

def compute_max_conv(seq):
    a = np.asarray(seq, dtype=float)
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    M = conv.max()
    return float(M)

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Approach
--------
1. **Coarse parametric sweep** – generate families of monotone decreasing
   sequences (power‑law a_i = (i+1)^‑p, exponential a_i = exp(-α·i) and a
   product of both).  All candidates are normalised so that Σa = √(2·n); the
   exact objective (max‑convolution) is computed with a single FFT.
2. **Randomised starts** – draw a few random vectors, sort them descending
   (to keep large values at the left side where the reverse convolution is
   small) and normalise.
3. **Local refinement** – run the soft‑max Adam optimiser (the same method
   used in the reference solution) from the best candidate found above.
   Several short optimiser runs with different random seeds are performed
   while respecting the time budget.
4. The best sequence seen is returned as a plain list.

The whole search respects the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the exact objective (max self‑convolution) for a normalised vector
def _max_convolution(seq: np.ndarray) -> float:
    """Return max(a∗a) for a non‑negative vector with Σa = √(2·n)."""
    a = np.asarray(seq, dtype=float)
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Soft‑max utilities (used by the Adam optimiser)
def _softmax_weights(conv: np.ndarray, tau: float):
    """Return the soft‑max weights w (sum to 1) and the smoothed max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k·a_{k‑i} (zero‑padding outside the range)
    using FFTs.
    """
    # FFT of a (already zero‑padded to length Nfft)
    A_fft = np.fft.rfft(a, n=Nfft)
    # reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # convolution
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


# ----------------------------------------------------------------------
# Adam optimiser (identical to the reference but with a few tunable knobs)
def _adam_optimize(
    a0: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 0.05,
    tau_factor: float = 0.20,
    decay_rate: float = 20000.0,
) -> tuple[np.ndarray, float]:
    """Run the Adam optimiser on a single start vector."""
    n = a0.shape[0]
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    iteration = 0
    jitter_interval = 5000

    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and iteration < max_iters:
        # exact convolution a*a
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # smooth max
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # weighted reverse convolution (gradient of smooth max)
        rev = _weighted_rev(a, w, Nfft)

        # gradient of the objective (see reference)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # update global best (exact objective = M after normalisation)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[{n:5d} it {iteration:9d}] new best = {best_score:.6f}")

    return best_seq, best_score

    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load previous construction if available; otherwise start from uniform
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore

    except Exception:

    best_seq = base.copy()
    best_score = _max_convolution(best_seq)
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – original and a few multiples (capped at 8000)
    length_factors = [1.0, 1.25, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [min(l, 8000) for l in lengths if l >= 300]

    # ------------------------------------------------------------------
    # --- 1. PARAMETRIC SWEEP -------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # a) power‑law a_i = (i+1)^-p
        p_grid = np.linspace(0.35, 1.2, 87)   # about 0.01 step
        idx = np.arange(1, n + 1, dtype=float)
        for p in p_grid:
            if time.time() >= deadline:
                break
            a = idx ** (-p)
            a = a * (target_sum / a.sum())
            score = _max_convolution(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[grid p] n={n} p={p:.3f} -> {score:.6f}")

        # b) exponential a_i = exp(-α·i)
        alpha_grid = np.linspace(0.0, 0.02, 41)
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            a = np.exp(-alpha * np.arange(n, dtype=float))
            a = a * (target_sum / a.sum())
            score = _max_convolution(a)
            if score < best_score:

        # c) combined power‑law * exponential
        for p in np.linspace(0.45, 0.8, 8):
            for alpha in np.linspace(0.001, 0.01, 5):
                if time.time() >= deadline:

                a = (idx ** (-p)) * np.exp(-alpha * np.arange(n, dtype=float))

                score = _max_convolution(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()

    # ------------------------------------------------------------------

    for n in lengths:
        if time.time() >= deadline:

        target_sum = math.sqrt(2.0 * n)
        for _ in range(30):          # 30 random attempts per length
            if time.time() >= deadline:
                break
            # generate a random decreasing vector
            a = np.sort(np.random.rand(n))[::-1]          # descending
            a = a * (target_sum / a.sum())
            score = _max_convolution(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[rand] n={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # --- 3. LOCAL REFINEMENT (soft‑max Adam) ---------------------------------
    # Keep running short optimiser runs while time permits.
    while time.time() < deadline - 10.0:    # leave a few seconds for final return
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)

        # create a slightly perturbed start (helps escaping shallow minima)
        jitter = np.random.randn(n) * (1e-2 * target_sum / n)
        start_a = np.maximum(best_seq + jitter, 0.0)
        start_a = start_a * (target_sum / start_a.sum())

        # decide on a generous iteration budget (scaled to remaining time)
        secs_left = deadline - time.time()
        # empirical ≈ 2000 iterations per second for n≈2000
        max_iters = int(min(2_000_000, secs_left * 1800))

        print(f"[optim] length={n}, max_iters≈{max_iters}")
        best_seq, best_score = _adam_optimize(
            start_a,
            best_seq,
            best_score,
            deadline,

            tau_factor=0.25,
            decay_rate=15000.0,
        )
        # if we already hit the target stop early
        if best_score < 1.503:

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain list)
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
