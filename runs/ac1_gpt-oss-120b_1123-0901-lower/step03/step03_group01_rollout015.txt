           best_seq = a.copy()

       # Adam optimizer with smoothed max
       m = np.zeros_like(a)
       v = np.zeros_like(a)
       t = 0
       while time.time() < deadline:
           conv = _conv_full_fft(a)
           max_conv = conv.max()
           if max_conv < best_val:
               best_val = max_conv
               best_seq = a.copy()
               print(...)

           alpha = 5 + 195 * frac   # from 5 to 200

           exp_scaled = np.exp(alpha * shifted)

           # compute gradient via convolution of reversed a with p
           a_rev = a[::-1]
           # determine n_fft for convolution
           n_fft = 1 << ((len(a_rev) + len(p) - 1).bit_length())
           A_rev = np.fft.rfft(a_rev, n=n_fft)
           P = np.fft.rfft(p, n=n_fft)
           conv_rev = np.fft.irfft(A_rev * P, n=n_fft)[:len(a_rev)+len(p)-1]
           offset = len(p) - 1
           grad = 2.0 * conv_rev[offset: offset + n]

           # Adam step
           t += 1
           beta1 = 0.9
           beta2 = 0.999
           m = beta1 * m + (1 - beta1) * grad
           v = beta2 * v + (1 - beta2) * (grad ** 2)
           m_hat = m / (1 - beta1 ** t)

           # enforce nonnegativity
           a_new = np.maximum(a_new, 0.0)
           # project onto simplex
           a_new = _project_onto_simplex(a_new, target_sum)

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

Since we keep Σ a_i = √(2 n) the objective equals max(conv(a, a)).
The algorithm below builds a candidate by

* a fast Euclidean projection onto the simplex {x ≥ 0, Σx = √(2n)},
* a smoothed approximation of the max‑convolution (log‑sum‑exp) and
  its exact gradient (computed with FFT‑based convolutions),
* an Adam‑style projected gradient descent on the smoothed objective,
* occasional random perturbations and a simple multi‑restart scheme.

The code runs for the allotted time (default 1000 s) and returns the
best sequence found.  No file I/O, no external solvers – only NumPy
and the standard library.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2·n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using real FFT (length == 2*len(a)-1)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Smoothed max‑convolution and its exact gradient.
# --------------------------------------------------------------------------- #
def _smooth_max_and_gradient(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Return (max_conv, grad) where max_conv = max(a*a) and grad is the
    gradient of the smoothed objective
        (1/alpha) * log( Σ exp(alpha * (a*a)_k) )
    with respect to ``a`` (projected onto the full domain).
    """
    # full convolution
    conv = _conv_full_fft(a)
    max_c = float(conv.max())

    # soft‑max weights (stable: shift by max)
    shifted = conv - max_c
    exp_vals = np.exp(alpha * shifted)          # ≤ 1, no overflow
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:                          # extreme under‑flow – fall back to uniform
        probs = np.full_like(exp_vals, 1.0 / exp_vals.size)
    else:
        probs = exp_vals / sum_exp

    # gradient = 2 * (a_rev * probs)_{n-1 ... n-1+n-1}
    a_rev = a[::-1]
    n = a.shape[0]
    # convolution length = n + (2n-1) - 1 = 3n-2
    n_fft = 1 << ((len(a_rev) + len(probs) - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=n_fft)
    P = np.fft.rfft(probs, n=n_fft)
    conv_rev = np.fft.irfft(A_rev * P, n=n_fft)[: len(a_rev) + len(probs) - 1]

    offset = n - 1
    grad = 2.0 * conv_rev[offset: offset + n]
    return max_c, grad


# --------------------------------------------------------------------------- #
# Adam‑style projected gradient descent on the smoothed max‑convolution.
# --------------------------------------------------------------------------- #
def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   time_limit: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Starting from ``a0`` (already on the simplex) run Adam optimisation
    until ``time_limit`` seconds have elapsed.  Returns the best vector
    seen together with its true max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_val = float("inf")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning rate – roughly one “pixel” of mass per step
    base_lr = 0.5 * target_sum / n

    it = 0
    while time.time() < start_time + time_limit:
        # schedule for the smoothing parameter alpha (grow from 5 to ~200)
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / time_limit)
        alpha = 5.0 + 195.0 * frac

        max_c, grad = _smooth_max_and_gradient(a, alpha)

        # keep the best true value (not the smoothed one)
        if max_c < best_val:
            best_val = max_c
            best_vec = a.copy()
            print(f"[{time.time():.2f}] new best max‑conv = {best_val:.6f} (n={n})")

        # Adam update (step is taken opposite to the gradient)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # enforce feasibility: non‑negativity + simplex
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, target_sum)

        # occasional tiny stochastic shake to help escape flat regions
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.002 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    return best_vec, best_val


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    The method uses many random restarts, an Adam optimiser on a
    smoothed max‑convolution and simple stochastic perturbations.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5    # tiny safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    while time.time() < deadline:
        # ------------------------------------------------------------------- #
        # Choose a problem size.
        # ------------------------------------------------------------------- #
        if best_seq is not None and np.random.rand() < 0.2:
            n = len(best_seq)
        else:
            n = np.random.randint(2500, 5501)      # a few thousand entries

        target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2n)

        # ------------------------------------------------------------------- #
        # Initialise a candidate – “U‑shaped” seed with a tiny random part.
        # ------------------------------------------------------------------- #
        idx = np.arange(n, dtype=float)
        weight = 1.0 / (idx + 1.0) + 1.0 / (n - idx)   # larger near the ends
        init = weight + 0.001 * np.random.rand(n)     # a bit of noise
        a0 = _project_onto_simplex(init, target_sum)

        # quick evaluation of the raw start point
        cur_val = _conv_full_fft(a0).max()
        if cur_val < best_score:
            best_score = cur_val
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] init best = {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------- #
        # Allocate time for this restart (≈ remaining budget split evenly).
        # ------------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0.0:
            break
        phase_time = min(12.0, time_left)    # up to 12 s per restart

        # Optimisation stage
        a_opt, val_opt = _adam_optimize(a0, target_sum,
                                        phase_time, time.time())
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] after Adam best = {best_score:.6f} (n={n})")

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
