def smooth_gradient(a, gamma):
    n = a.shape[0]
    conv = _conv_full_fft(a)
    max_conv = conv.max()
    # weights
    exp_w = np.exp(gamma * (conv - max_conv))
    Z = exp_w.sum()
    p = exp_w / Z  # shape (2n-1)
    # correlation via FFT
    a_rev = a[::-1]
    # pad to length N = next power of two >= len(p) + len(a_rev) - 1
    N = 1 << ((len(p) + n - 1).bit_length())
    P = np.fft.rfft(p, N)
    A_rev = np.fft.rfft(a_rev, N)
    corr = np.fft.irfft(P * A_rev, N)  # length N
    # gradient slice
    grad = 2.0 * corr[n-1:n-1+n]
    # return gradient and max_conv
    return grad, max_conv
```

def smooth_adam_optimize(a_init, target_sum, max_time, start_time):
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    gamma = 1.0
    while time.time() < start_time + max_time:
        grad, max_conv = smooth_gradient(a, gamma)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)   # 0.1 may be tuned
        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, target_sum)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth Adam new best {best_score:.6f} (n={n})")
        # accelerate gamma
        gamma = min(gamma * 1.02, 50.0)
        # occasionally random jitter
        if np.random.rand() < 0.01:
            a = _project_onto_simplex(np.maximum(a + np.random.randn(n) * (0.01*target_sum/n), 0.0), target_sum)
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

*   The search works in two stages.
    1.  **Randomised parametrised initialisation** – a large number of
        candidates are created from simple analytic families (exponential,
        power‑law, a product of both and a pure random vector).  Their
        lengths and parameters (decay rate, power, offset) are sampled
        randomly.  Each candidate is projected onto the simplex
        Σa = √(2 n), a_i ≥ 0, and immediately evaluated.  The best of those
        become the current global best.
    2.  **Smooth Adam optimisation** – the best candidate of the current
        restart is refined with a first‑order method.  The non‑smooth
        objective `max(conv)` is approximated by a soft‑max
        (log‑sum‑exp) with a temperature parameter γ.  The gradient of the
        smooth surrogate can be written as a correlation of the soft‑max
        weights with the current vector, which is computed efficiently by
        FFTs.  An Adam‑style update (momentum + adaptive step) is applied
        and the iterate is projected back onto the simplex.  γ is slowly
        increased during the optimisation so the method gradually focuses
        on the true maximum.

*   The outer loop repeatedly restarts the whole procedure with
    different lengths (≈800–3000).  All restarts share a common time
    budget (≈1000 s).  Whenever a better sequence is found its value and
    size are printed.  If the target bound 1.503 is reached the search
    stops early.

The code contains only top‑level functions, uses NumPy and the standard
`time` module, and respects the sandbox limits (no I/O, no external
solvers).  The final result is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s`` (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length, real FFT implementation)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a."""
    return float(_conv_full_fft(a).max())


def _objective(a: np.ndarray) -> float:
    """Exact objective (same as evaluate_sequence for Σa = √(2n))."""
    n = a.shape[0]
    maxc = _max_conv(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * maxc / (s ** 2))


# --------------------------------------------------------------------------- #
# Parametric initialisers
# --------------------------------------------------------------------------- #
def _init_exponential(n: int, target_sum: float, alpha: float, offset: int = 0) -> np.ndarray:
    """a_i = exp(-alpha * (i‑offset) / n) for i ≥ offset, else 0; then project."""
    idx = np.arange(n) - offset
    a = np.where(idx >= 0, np.exp(-alpha * idx / n), 0.0)
    if a.sum() > 0:
        a = a * (target_sum / a.sum())
    a = _project_onto_simplex(a, target_sum)
    return a


def _init_powerlaw(n: int, target_sum: float, p: float, offset: int = 0) -> np.ndarray:
    """a_i = 1/(i‑offset+1)^p for i ≥ offset, else 0; then project."""
    idx = np.arange(n) - offset
    a = np.where(idx >= 0, 1.0 / ((idx + 1) ** p), 0.0)
    if a.sum() > 0:
        a = a * (target_sum / a.sum())
    a = _project_onto_simplex(a, target_sum)
    return a


def _init_combined(n: int, target_sum: float, alpha: float, p: float, offset: int = 0) -> np.ndarray:
    """product of exponential and power‑law decay, plus optional offset."""
    idx = np.arange(n) - offset
    mask = idx >= 0
    a = np.zeros(n)
    a[mask] = np.exp(-alpha * idx[mask] / n) * ((idx[mask] + 1) ** (-p))
    if a.sum() > 0:
        a = a * (target_sum / a.sum())
    a = _project_onto_simplex(a, target_sum)
    return a


def _init_random(n: int, target_sum: float) -> np.ndarray:
    """Uniform random start, projected onto the simplex."""
    a = np.random.rand(n)
    a = _project_onto_simplex(a, target_sum)
    return a


# --------------------------------------------------------------------------- #
# Smoothed maximum (log‑sum‑exp) gradient using FFT based correlation.
# --------------------------------------------------------------------------- #
def _smooth_gradient(a: np.ndarray, gamma: float):
    """
    Return (grad, max_conv) where grad is the gradient of
    f(a)= (1/γ)·log Σ exp(γ·conv)  and max_conv is the current maximum
    convolution value.
    """
    n = a.shape[0]
    conv = _conv_full_fft(a)
    max_c = conv.max()
    # soft‑max weights, stabilised by subtracting the max
    exp_w = np.exp(gamma * (conv - max_c))
    p = exp_w / exp_w.sum()                     # shape (2n‑1,)

    # correlation p * a  (g_i = 2·Σ_k p_k·a_{k-i})
    a_rev = a[::-1]
    N = 1 << ((len(p) + n - 1).bit_length())
    P = np.fft.rfft(p, N)
    A_rev = np.fft.rfft(a_rev, N)
    corr = np.fft.irfft(P * A_rev, N)          # length N
    grad = 2.0 * corr[n - 1:n - 1 + n]         # slice that corresponds to i=0..n‑1
    return grad, max_c


# --------------------------------------------------------------------------- #
# Adam optimiser on the smoothed objective.
# --------------------------------------------------------------------------- #
def _smooth_adam_optimize(a_init: np.ndarray,
                          target_sum: float,
                          max_time: float,
                          start_time: float):
    """
    Run a few seconds of Adam on the smoothed max‑convolution.
    Returns (best_vector, best_objective).
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    # base learning rate: a small fraction of the average entry size
    lr = 0.04 * target_sum / n
    gamma = 1.0                     # start with a soft approximation

    while time.time() < start_time + max_time:
        grad, _ = _smooth_gradient(a, gamma)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # feasibility projection
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, target_sum)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam new best = {best_score:.6f} (n={n})")

        # increase the temperature slowly – we move from a smooth landscape
        # toward the true max‑convolution.
        gamma = min(gamma * 1.02, 80.0)

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < 0.01:
            a = a + np.random.randn(n) * (0.01 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a, 0.0), target_sum)

    return best_a, best_score
