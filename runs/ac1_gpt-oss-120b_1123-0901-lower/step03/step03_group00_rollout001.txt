    if np.sum(v) == 1 and np.all(v >= 0):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1   # index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)

"""
propose_candidate
-----------------

Searches for a non‑negative step‑function (probability vector) that
minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

where ``a*a`` is the linear convolution of ``a`` with itself.
The algorithm works on the probability simplex (sum(a)=1) and
optimises a smooth‑max surrogate of the max‑convolution using
Adam updates directly on the simplex (via a projection step).
A fast temperature schedule makes the surrogate approach the true
maximum, while occasional random “shakes’’ avoid stagnation.
After the gradient phase a short random‑local‑search refines the
best solution found.

The routine respects a global time budget (default 1000 s) and
returns the best sequence (as a plain Python list) discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index (largest i where u_i - cssv_i / i > 0)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

                w_sum = w.sum()
                if w_sum == 0.0:
                    # Numerical safety – fall back to uniform weight.
                    w = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    w /= w_sum

                # ----- gradient of the smooth surrogate w.r.t a -----
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ----- Adam update on a (before projection) -----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # ----- project back onto the simplex -----
                a = _project_simplex(a_new)

                # ----- temperature schedule -----
                lam = min(lam * lam_factor, 1e6)

                # ----- exact objective for monitoring -----
                val = _exact_obj(a, n)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

                    # Early exit if we already beat the target.
                    if best_val < 1.503:
                        print("Target achieved – terminating early.")
                        deadline = time.time()  # forces outer loops to stop
                        break

                # ----- stagnation handling -----
                if time.time() - last_improve > 6.0:
                    # Small random perturbation (additive noise before projection)
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # End of one restart
        # End of lengths loop

    # ------------------------------------------------------------------
    # Final local random refinement (if any time left)
    # --------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        refine_deadline = deadline
        # Simple hill‑climbing: move a tiny fraction of mass between two coordinates.
        while time.time() < refine_deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # Transfer up to 0.5 % of the total mass from i to j.
            delta = 0.005 * a[i] * np.random.rand()
            if delta == 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            # Re‑project (should stay on simplex but protects against rounding)
            a_prop = _project_simplex(a_prop)
            val = _exact_obj(a_prop, n)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq, n_fallback)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return plain Python list as required by the external evaluator.
    return best_seq.tolist()
