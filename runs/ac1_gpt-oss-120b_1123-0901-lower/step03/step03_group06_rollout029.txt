The evaluation metric is 2·n·max(conv)/sum(a)^2 which, after fixing
sum(a)=√(2n), equals the maximum entry of the full auto‑convolution.
We minimise this quantity by optimising a probability vector p (∑p=1,
p≥0) and then scaling a = √(2n)·p.

Algorithm
---------
* Multi‑start exponentiated‑gradient (EG) mirror descent on the simplex.
  For a candidate p we compute the full self‑convolution (FFT) and
  obtain its maximum M.
* A soft‑max weight w_k = exp(β·(conv_k−M)) concentrates the
  sub‑gradient on the active indices; β is increased from a small to a
  large value (0.1 → 200) so that the gradient eventually approximates
  the true sub‑gradient of the max‑operator.
* EG updates are performed in log‑space and projected back onto the
  probability simplex.
* After the EG phase a short stochastic hill‑climbing (mass transfer
  between two coordinates) is applied to escape possible plateaus.
* The whole budget (≈1000 s) is spent on a single large problem size
  (n≈8000) with several random restarts; the best sequence found is
  returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full linear self‑convolution p∗p using one FFT of size N.
    Returns the real vector of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Correlation rev_i = Σ_k w_k·p_{k−i} (i=0..n−1) using FFT.
    """
    n = p.shape[0]
    L = w.shape[0]               # = 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]
    rev = conv_rev[L - n : L][::-1]
    return rev


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    (standard O(n log n) algorithm).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _optimize_exponentiated_gradient(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the simplex to minimise
    max(p∗p).  Returns the best probability vector discovered and its
    objective value (max convolution after scaling).
    """
    # FFT size large enough for both convolution (2n‑1) and the
    # correlation with the weight vector (3n‑2).
    N = _next_pow2(3 * n - 2)

    # Parameters for the soft‑max smoothing of the max‑operator.
    beta_start = 0.1
    beta_end = 200.0               # very aggressive – focuses on the true max
    beta_ramp_iters = 100_000      # linear ramp for β

    # EG step‑size schedule (η decays slowly)
    eta0 = 1.0 / (4.0 * n)         # initial step size

    # Jitter to shake the iterate occasionally
    jitter_interval = 5000
    jitter_scale = 5e-4

    # Initialise log‑probabilities safely
    eps = 1e-30
    logp = np.log(p_start + eps)

    best_local_score = float("inf")
    best_local_p = p_start.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution (p∗p) -----
        conv = _full_convolution_fft(np.exp(logp) / np.exp(logp).sum(), N)
        M = conv.max()

        # current objective (remember evaluate_sequence == max_b once scaled)
        cur_score = M * 2.0 * n / (np.sum(np.exp(logp) / np.exp(logp).sum() * math.sqrt(2.0 * n)) ** 2)
        # the above simplifies to M, but we keep it for clarity.
        cur_score = M

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = np.exp(logp) / np.exp(logp).sum()
            print(f"[EG n={n} iter={it:6d}] new local best = {best_local_score:.6f}")

        # ----- soft‑max weights -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)
        w = np.exp(beta * (conv - M))        # w ∈ (0,1], w_max = 1

        # ----- weighted reverse correlation (sub‑gradient) -----
        rev = _weighted_rev_fft(np.exp(logp) / np.exp(logp).sum(), w, N)

        # sub‑gradient of the smoothed objective (2·n·M_β) w.r.t. p
        grad = 4.0 * n * rev

        # ----- step size (η decays) -----
        eta = eta0 / (1.0 + it * 1e-5)

        # ----- EG update in log‑space -----
        logp -= eta * grad

        # occasional jitter
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto simplex (via soft‑max) -----
        # we stay in log‑space, only need to normalise the probabilities.
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)          # avoid overflow
        p = exp_shift / exp_shift.sum()

        # keep logp consistent with projected p
        logp = np.log(p + eps)

    return best_local_p, best_local_score


def _hill_climb(
    p: np.ndarray,
    n: int,
    deadline: float,
    steps: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the probability simplex.
    Moves a small amount of mass between two coordinates and keeps the
    best solution seen.
    """
    best_p = p.copy()
    best_score = float("inf")

    # scaling factor to convert probabilities to the required step heights
    S = math.sqrt(2.0 * n)

    for s in range(steps):
        if time.time() >= deadline:
            break

        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        if best_p[i] <= 1e-12:
            continue

        # transfer a random fraction of the mass at i to j
        delta = best_p[i] * np.random.rand() * 0.05   # ≤5 % of the source mass
        new_p = best_p.copy()
        new_p[i] -= delta
        new_p[j] += delta

        # maintain feasibility (non‑negative, sum = 1)
        new_p = _project_onto_simplex(new_p)

        # evaluate
        a = new_p * S
        cur_score = evaluate_sequence(a.tolist())

        if cur_score < best_score:
            best_score = cur_score
            best_p = new_p
            # occasionally report progress
            if s % 500 == 0:
                print(f"[HC step={s:4d}] new best = {best_score:.6f}")

    return best_p, best_score

def propose_candidate(budget_s: float = 1000, seed: int = 123) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # problem size – we devote the whole budget to a large instance
    n = 8000                     # length of the step function

    # pre‑compute the normalisation factor S = √(2·n)
    S = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # generate a handful of diverse starting points
    starts = []

    # 1) uniform
    starts.append(np.full(n, 1.0 / n))

    # 2) geometric decreasing (r ≈ 0.995)
    r = 0.995
    geom = r ** np.arange(n, dtype=float)
    geom /= geom.sum()
    starts.append(geom)

    # 3) previous construction if available
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        if base.shape[0] == n:
            starts.append(base / base.sum())
        else:
            # up‑sample / down‑sample to length n
            old = np.linspace(0, 1, base.shape[0])
            new = np.linspace(0, 1, n)
            interp = np.interp(new, old, base)
            interp = np.maximum(interp, 0.0)
            if interp.sum() == 0:
                interp = np.full(n, 1.0 / n)
            else:
                interp /= interp.sum()
            starts.append(interp)
    except Exception:
        pass

    # 4) a few random Dirichlet draws
    for _ in range(3):
        starts.append(np.random.dirichlet(np.ones(n)))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # main optimisation loop – one EG run per start, then a brief HC
    for idx, p0 in enumerate(starts):
        if time.time() >= deadline:
            break

        # allocate a generous slice of the remaining time to each start
        remaining = deadline - time.time()
        # we ask for at most 200 k iterations; the routine will stop earlier
        max_iters = 200_000

        print(f"[start {idx+1}/{len(starts)}] EG optimisation, n={n}")

        p_opt, eg_score = _optimize_exponentiated_gradient(
            n=n,
            p_start=p0,
            deadline=deadline,
            max_iters=max_iters,
        )

        # convert to original scale and evaluate
        a_opt = p_opt * S
        cur_score = evaluate_sequence(a_opt.tolist())
        print(f"[start {idx+1}] EG result score = {cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"--> new global best after EG: {best_score:.6f}")

        # ------------------------------------------------------------------
        # short hill‑climbing refinement (if time permits)
        if time.time() < deadline:
            print(f"[start {idx+1}] hill‑climbing refinement")
            p_refined, hc_score = _hill_climb(
                p_opt,
                n,
                deadline=deadline,
                steps=3000,
            )
            a_refined = p_refined * S
            final_score = evaluate_sequence(a_refined.tolist())
            print(f"[start {idx+1}] after HC score = {final_score:.6f}")

            if final_score < best_score:
                best_score = final_score
                best_seq = a_refined.copy()
                print(f"*** new global best after HC: {best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback – uniform sequence (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
