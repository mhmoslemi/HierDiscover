Implement projection onto simplex (nonnegative) algorithm:

def proj_simplex(v, s):
    # returns projection of v onto simplex {x >= 0, sum x = s}
    # from: https://github.com/indradhanush/ProjectionOntoSimplex
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Thus after each gradient step we can enforce exact sum.

def _proj_simplex(v, s):
    # Project onto nonnegative simplex of sum s.
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All zero?
        tau = 0

        rho = ind[cond][-1]
        tau = cssv[cond][-1] / rho
    w = np.maximum(v - tau, 0)
    return w

    best_local_seq = a.copy()
    best_local_score = float('inf')
    step_alpha = step0 * target_sum / n   # scale to average magnitude
    gamma = gamma0
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # compute convolution max and conv array
        M, _, conv = _conv_max_fft(a)
        if M < best_local_score:
            best_local_score = M

Propose a coefficient vector that minimises the evaluate_sequence score.

Algorithm
---------

1. Initialise with the supplied `height_sequence_1` (if present) and evaluate it.
2. Try a handful of different lengths (from ~300 up to ~8000).  For each
   length we run several independent restarts:
     * First restart: linear interpolation of the current best vector onto
       the new grid (preserves the overall shape).
     * Other restarts: random positive Dirichlet‑like vectors.
3. For each restart a custom projected gradient descent is performed:

     * The full autocorrelation `c = a * a` is obtained with an FFT.
     * The current maximal value `M = max(c)` is identified.
     * A sub‑gradient of the max‑convolution is approximated by a
       *weighted* average of the reversed vectors belonging to the
       top‑K largest convolution entries.  The weighting uses a soft‑max
       (exp(γ·(c_k‑M))) with a slowly increasing temperature γ.
     * A back‑tracking line‑search finds a step size `α` that strictly
       decreases `M`.  After the step we project back onto the feasible
       set: clip to non‑negative and renormalise to the fixed L₁‑norm
       `sqrt(2·n)`.  If no improvement is found, a small Gaussian jitter
       is added to escape flat regions.
4. After the gradient phase a cheap coordinate‑swap hill‑climb moves a
   small amount of mass between two indices whenever it reduces the
   maximal convolution value.
5. The best vector seen over all lengths and restarts is returned.
   The code respects the 1000‑second budget and prints progress
   information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Full autocorrelation a*a using FFT; returns (max, argmax, full array)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    max_val = float(conv.max())
    idx = int(conv.argmax())
    return max_val, idx, conv


# ----------------------------------------------------------------------
def _proj_onto_sum_positive(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to ≥0 and scale so that the sum equals target_sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        # degenerate case – give a uniform feasible vector
        v = np.full_like(v, target_sum / v.shape[0])
    return v


# ----------------------------------------------------------------------
def _weighted_rev(a: np.ndarray, conv: np.ndarray, top_k: int, gamma: float) -> np.ndarray:
    """
    Approximate a sub‑gradient of max_k (a*a)_k.
    Takes the `top_k` largest convolution entries, weights them by
    exp(gamma * (c_k - M)), and returns the weighted average of the
    corresponding reversed vectors.
    """
    n = a.shape[0]
    M = conv.max()
    # indices of the `top_k` largest entries (unsorted)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    # soft‑max weights; noticeable only for entries close to the maximum
    w = np.exp(gamma * (conv[idxs] - M))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()

    rev = np.zeros_like(a)
    idx_arr = np.arange(n)
    for k, weight in zip(idxs, w):
        j = k - idx_arr
        mask = (j >= 0) & (j < n)
        rev[mask] += weight * a[j[mask]]
    rev /= w_sum
    return rev


# ----------------------------------------------------------------------
def _gd_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 200_000,
    step_factor: float = 0.2,
    top_k: int = 30,
    gamma_start: float = 0.5,
    gamma_max: float = 5.0,
    jitter_prob: float = 0.2,
    jitter_scale: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with back‑tracking line search.
    Returns possibly updated (best_seq, best_score).
    """
    a = a_init.copy()
    a = _proj_onto_sum_positive(a, target_sum)

    best_local_seq = a.copy()
    best_local_score = float("inf")

    # initial step size is a fraction of the average entry
    avg = target_sum / n
    step_alpha = step_factor * avg
    gamma = gamma_start
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        # current convolution statistics
        M, _, conv = _conv_max_fft(a)

        # keep the best ever seen (global)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new global best = {best_score:.6f}")

        # keep the best for this length as well
        if M < best_local_score:
            best_local_score = M
            best_local_seq = a.copy()

        # sub‑gradient (weighted average of reversed vectors)
        rev = _weighted_rev(a, conv, top_k, gamma)

        # slowly increase the temperature (more focus on the true max)
        gamma = min(gamma_max, gamma + 0.0005)

        # ----- back‑tracking line search -----
        accepted = False
        alpha = step_alpha
        for _ in range(5):          # at most 5 back‑track attempts
            a_new = a - alpha * rev
            a_new = _proj_onto_sum_positive(a_new, target_sum)
            M_new, _, _ = _conv_max_fft(a_new)
            if M_new < M:          # strict improvement
                a = a_new
                step_alpha = alpha * 1.1          # a touch more aggressive next time
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            # occasional random jitter to escape plateaus
            if np.random.rand() < jitter_prob:
                jitter = np.random.randn(n) * jitter_scale * target_sum
                a = _proj_onto_sum_positive(a + jitter, target_sum)
            else:
                # reduce step size gently
                step_alpha = max(step_alpha * 0.7, 1e-9)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: move a small fraction of mass from a random
    donor index to a random receiver index if it reduces the maximal
    convolution value.
    """
    a = a_start.copy()
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        # pick donor with some mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # pick a different receiver
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to move (up to 5 % of donor)
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0:
            continue

        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps
        new_vec = _proj_onto_sum_positive(new_vec, target_sum)

        M_new, _, _ = _conv_max_fft(new_vec)
        if M_new < best_score - 1e-12:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (prop {proposal})")
    return a, best_score

    Search for a non‑negative step‑function that minimises the given

    best_score = init_score
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose a set of lengths (300 … 8000) to explore
    length_factors = [0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [l for l in lengths if 300 <= l <= 8000]

    # ------------------------------------------------------------------
    # Main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        # number of restarts per length
        restarts = 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise -------------------------------------------------
            if r == 0 and best_seq.shape[0] == n0:
                # linear interpolation from the current best vector onto the new grid
                old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, best_seq)
                init_a = np.maximum(init_a, 0.0)
            else:
                # random positive start
                init_a = np.random.rand(n)
                init_a = np.maximum(init_a, 0.0)

            # normalise to the required sum
            init_a = _proj_onto_sum_positive(init_a, target_sum)

            # ----- gradient descent phase ------------------------------------
            best_seq, best_score = _gd_optimize(
                a_init=init_a,
                n=n,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iter=200_000,
                step_factor=0.2,

                gamma_start=0.5,
                gamma_max=5.0,

                jitter_scale=0.001,

                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,

        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
