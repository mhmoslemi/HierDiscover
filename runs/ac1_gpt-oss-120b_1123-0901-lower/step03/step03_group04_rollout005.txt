"""
Propose a non‑negative step‑function that minimizes ``evaluate_sequence``.

The algorithm mixes three complementary optimisation ideas:

1. **Fast sub‑gradient Adam** – identical to the previous baseline: minimise
   the peak of the autocorrelation (max‑convolution) using the exact
   sub‑gradient obtained from the active shifts.  This gives a quick,
   coarse improvement.

2. **Smooth‑max Adam** – replace the non‑smooth max by a
   log‑sum‑exp (soft‑max) with a temperature that is increased during the
   run.  The gradient of the soft‑max is a convex combination of all
   convolution gradients and can be computed efficiently with a single
   FFT convolution:
       grad = 2 * (p ⊛ a_rev)  ,
   where ``p`` are the soft‑max weights and ``a_rev`` is the reversed
   sequence.  Starting with a low temperature gives a smooth landscape,
   and raising it drives the solution towards a true max‑minimiser.

3. **Cheap local search** – after the continuous optimisation a greedy
   pairwise mass‑transfer is performed (move a tiny fraction of the
   height from one index to another) and kept if it lowers the maximal
   autocorrelation.  This often removes small remaining spikes.

The search is performed on several problem sizes (the original length and
a few scaled versions).  For each size a few random restarts are run.
The whole pipeline respects the 1000 s wall‑clock budget and returns the
best sequence found.
"""

def _conv_max_fft(a: np.ndarray):

    Compute the maximal value of the full convolution a*a and also return
    the full convolution array.  Uses an FFT of length equal to the next
    power of two for speed.

    N = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    C = np.fft.irfft(A * A, n=N)                 # convolution in frequency domain
    conv = C[: 2 * n - 1]                         # keep only the valid part
    max_val = float(conv.max())
    return max_val, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Exact sub‑gradient of ``max(conv)`` at the current point.
    For every active shift ``k`` the gradient contribution is the vector
    (a_{k‑i})_i, i.e. a shifted copy of ``a``.
    The returned gradient is the average over all active shifts.
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx                     # partner index in the convolution
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size > 0:
        grad /= float(active_idxs.size)

    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 10_000_000):
    """
    Adam optimiser on the exact sub‑gradient of the max‑convolution.
    Returns the best vector seen and the associated maximal convolution.
    """
    a = init_a.copy()
    # normalise sum = target_sum (≈ √(2n))
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.04
    iteration = 0
    jitter_interval = 4000

    best_vec = a.copy()
    _, conv = _conv_max_fft(a)
    best_val = conv.max()

    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # 1. convolution and active set
        M, conv = _conv_max_fft(a)
        active = np.where(conv >= M - 1e-12)[0]

        # --------------------------------------------------------------
        # 2. sub‑gradient of the max term
        rev = _subgradient(a, active)

        # derivative of the whole objective (scale invariant, we drop constants)
        grad = rev

        # --------------------------------------------------------------
        # 3. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 4. projection onto feasible set (non‑negative, correct L1‑norm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # --------------------------------------------------------------
        # 5. occasional jitter to escape flat plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # --------------------------------------------------------------
        # 6. keep the best solution seen so far
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[adam] len {n:5d} iter {iteration:7d}  new best  M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _smooth_max_optimize(init_a: np.ndarray,
                         n: int,
                         target_sum: float,
                         deadline: float):
    """
    Adam optimiser on a smooth approximation of the max‑convolution.
    The smooth max is ``(1/α)·log Σ exp(α·conv_i)``.
    The temperature α is increased linearly from ``α_start`` to
    ``α_end`` during the run.
    Returns the best vector seen and the corresponding (non‑smooth) max.
    """
    a = init_a.copy()
    # normalise sum
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.05

    iteration = 0
    best_vec = a.copy()
    _, conv = _conv_max_fft(a)
    best_val = conv.max()

    # temperature schedule
    α_start, α_end = 0.5, 30.0
    # estimate a rough number of iterations (will be limited by time)
    est_iters = 5000

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. current convolution and max (for tracking)
        M, conv = _conv_max_fft(a)
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[smooth] len {n:5d} iter {iteration:7d}  new best  M = {best_val:.6f}")

        # --------------------------------------------------------------
        # 2. temperature (linear increase, capped at α_end)
        α = min(α_end, α_start + (α_end - α_start) * (iteration / est_iters))

        # --------------------------------------------------------------
        # 3. soft‑max weights
        shifted = conv - M                     # ≤ 0, avoids overflow
        w = np.exp(α * shifted)                # max weight = 1
        sumw = w.sum()
        if sumw == 0.0:                       # numerical safety
            p = np.zeros_like(w)
            p[np.argmax(conv)] = 1.0
        else:
            p = w / sumw                       # probabilities (sum = 1)

        # --------------------------------------------------------------
        # 4. gradient of the smooth max:  grad = 2 * (p ⊛ a_rev)
        a_rev = a[::-1]
        N = 1 << ((len(p) + n - 1).bit_length())   # power‑of‑two for FFT
        Fp = np.fft.rfft(p, n=N)
        Fa_rev = np.fft.rfft(a_rev, n=N)
        conv_full = np.fft.irfft(Fp * Fa_rev, n=N)  # length = len(p)+n‑1
        start = n - 1
        grad = 2.0 * conv_full[start:start + n]

        # --------------------------------------------------------------
        # 5. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-5 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 6. projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # --------------------------------------------------------------
        # 7. occasional jitter (helps escape plateaus)
        if iteration % 4000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap stochastic local search.
    Repeatedly move a small fraction of the mass from one index to another.
    Accept the move only if it reduces the maximal convolution.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # Transfer a tiny fraction (2 %) of the source entry
        delta = best[i] * 0.02
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum unchanged, non‑negativity already guaranteed
        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M

    return best, best_val

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the given time budget
    (≈ 1000 s) and returns the best sequence found so far.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a tiny safety margin for the surrounding harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Load the previously discovered construction if available
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        print("[info] loaded previous construction of length", base.shape[0])
    except Exception:
        # fallback: uniform vector of moderate size
        base = np.full(2100, 1.0, dtype=float)
        print("[info] no previous construction – using uniform start")

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    if base.sum() > 0:
        base = base * (target_sum0 / base.sum())
    else:
        base = np.full_like(base, target_sum0 / n0)

    # --------------------------------------------------------------
    # 2. Global best tracker
    best_seq = base.copy()
    best_val, _ = _conv_max_fft(best_seq)
    print(f"[init] length={n0}, initial M={best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Choose a set of problem sizes to explore
    length_factors = [0.75, 1.0, 1.25, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted(
        {max(300, min(8000, int(n0 * f))) for f in length_factors}
    )
    print("[info] exploring lengths:", lengths)

    # --------------------------------------------------------------
    # 4. Main optimisation loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # number of random restarts (more restarts for smaller sizes)
        restarts = 3 if n <= 4000 else 2

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise -------------------------------------------------
            if n == n0 and r == 0:
                # first run on the original size – start from the base vector
                init = base.copy()
            else:
                # random Dirichlet‑like start (positive entries)
                rand = np.random.exponential(scale=1.0, size=n)
                init = rand * (target_sum / rand.sum())

            # blend with the current global best when lengths match
            if best_seq.shape[0] == n:
                mix = 0.2
                init = mix * best_seq + (1 - mix) * init
                init = np.maximum(init, 0.0)
                init *= (target_sum / init.sum())

            print(f"[run] length={n}, restart={r + 1}/{restarts}")

            # ----- time budgeting --------------------------------------------
            time_left = deadline - time.time()
            # allocate roughly half of the remaining time to each phase
            phase_time = min(150.0, time_left * 0.45)

            # ----- phase 1 – fast Adam on the exact sub‑gradient ------------
            sub_deadline = time.time() + phase_time
            seq_adam, val_adam = _adam_optimize(
                init, n, target_sum, sub_deadline, max_iters=5_000_000
            )

            # ----- phase 2 – smooth‑max Adam refinement ----------------------
            if time.time() < deadline:
                phase_time = min(150.0, deadline - time.time())
                sub_deadline = time.time() + phase_time
                seq_smooth, val_smooth = _smooth_max_optimize(
                    seq_adam, n, target_sum, sub_deadline
                )
            else:
                seq_smooth, val_smooth = seq_adam, val_adam

            # ----- phase 3 – cheap local search -----------------------------
            if time.time() < deadline:
                seq_local, val_local = _local_search(
                    seq_smooth,
                    n,
                    target_sum,
                    steps=30_000,
                    deadline=deadline,
                )
                # keep the best among the three variants
                if val_local < val_smooth:
                    cur_seq, cur_val = seq_local, val_local
                else:
                    cur_seq, cur_val = seq_smooth, val_smooth
            else:
                cur_seq, cur_val = seq_smooth, val_smooth

            # ----- update global best ---------------------------------------
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[global] new best M={best_val:.6f} (len={n})")

    # --------------------------------------------------------------
    # 5. Final sanity check – evaluate through the official API
    final_score = evaluate_sequence(best_seq.tolist())
    print(
        f"[done] best reported M={best_val:.6f}, official score={final_score:.6f}, "
        f"length={best_seq.shape[0]}"
    )
    return best_seq.tolist()
