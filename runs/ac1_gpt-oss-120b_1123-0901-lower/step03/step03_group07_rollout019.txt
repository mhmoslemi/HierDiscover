def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, math, random, scipy.optimize

    start = time.time()
    deadline = start + budget_s - 0.5

    np.random.seed(seed)

    # Choose length based on budget (simple heuristic)
    n = 3000
    # optionally adjust n if budget small
    if budget_s < 30:
        n = 500

    # create initial candidates
    candidates = []
    # uniform
    a_uni = np.full(n, 1.0/n)
    candidates.append(a_uni)

    # half sine
    i = np.arange(1, n+1)
    a_sine = np.sin(np.pi * i / (n+1))
    a_sine = np.maximum(a_sine, 0)
    a_sine = a_sine / a_sine.sum()
    candidates.append(a_sine)

    # random expo
    for _ in range(5):
        r = np.random.uniform(0.2, 0.95)
        powers = np.power(r, np.arange(n, dtype=np.float64))
        a_expo = powers / powers.sum()
        candidates.append(a_expo)

    # Dirichlet
    for _ in range(5):
        a_dir = np.random.dirichlet(np.ones(n))
        candidates.append(a_dir)

    # optionally global variable
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, list) and len(height_sequence_1) > 0:
            a_prev = np.array(height_sequence_1, dtype=np.float64)
            if a_prev.shape[0] != n:
                # Interpolate to length n
                x_old = np.arange(a_prev.shape[0])
                x_new = np.linspace(0, a_prev.shape[0]-1, n)
                a_prev = np.interp(x_new, x_old, a_prev)
            a_prev = np.maximum(a_prev, 0)
            a_prev = a_prev / a_prev.sum()
            candidates.append(a_prev)
    except NameError:
        pass

    best_val = float('inf')
    best_seq = None

    for idx, seq in enumerate(candidates):
        if time.time() > deadline:
            break
        val = evaluate_sequence(seq.tolist())
        print(f"Initial candidate {idx} val={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print("New best from init")

    # Now optimization loop
    # Use function adam_optimize(seq, total_time)
    def adam_optimize(a_init):
        a = a_init.copy()
        # Parameterization: logits
        u = np.log(a + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        base_lr = 0.1
        lam = 0.5  # start smoothing lower
        lam_factor = 1.005
        best_local_val = float('inf')
        best_local_seq = a.copy()
        it = 0
        while time.time() < deadline and it < 20000:
            it += 1
            # compute probabilities
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a = a / a.sum()
            # convolution via FFT
            n = a.shape[0]
            size_fft = 1 << int(np.ceil(np.log2(2*n-1)))
            fa = np.fft.rfft(a, size_fft)
            conv = np.fft.irfft(fa * fa, size_fft)[: 2*n - 1]

            # softmax weights
            m_conv = conv.max()
            w = np.exp(lam * (conv - m_conv))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # gradient wrt a: 2 * cross-correlation w * a
            # compute correlation via FFT
            rev_a = a[::-1]
            size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
            fw = np.fft.rfft(w, size_corr)
            frev = np.fft.rfft(rev_a, size_corr)
            corr = np.fft.irfft(fw * frev, size_corr)
            grad_a = 2.0 * corr[n-1 : n-1 + n]

            # transform to logits gradient
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_u
            v = beta2 * v + (1 - beta2) * (grad_u ** 2)
            it_float = float(it)
            m_hat = m / (1 - beta1 ** it_float)
            v_hat = v / (1 - beta2 ** it_float)
            lr = base_lr * (0.9999 ** it_float)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # increase lambda
            lam *= lam_factor

            if it % 200 == 0:
                # evaluate exact objective
                val = evaluate_sequence(a.tolist())
                if val < best_local_val:
                    best_local_val = val
                    best_local_seq = a.copy()
                    print(f"Adam iter {it} val={val:.6f}")
                    # update global best
                    nonlocal best_val, best_seq
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        print(f"Global best improved to {best_val:.6f}")

            # random shake if stuck (no improvement after some time)
            # optional, omitted for simplicity

        return best_local_seq

    # Run Adam for best_seq
    if best_seq is not None:
        print(f"Starting Adam refinement from best init val = {best_val:.6f}")
        best_seq = adam_optimize(best_seq)

    # LP refinement on downsampled version if time left
    if time.time() < deadline - 5:
        try:
            from scipy.optimize import linprog
            # downsample to n_lp
            n_lp = min(800, best_seq.shape[0])
            if n_lp < best_seq.shape[0]:
                x_old = np.arange(best_seq.shape[0])
                x_new = np.linspace(0, best_seq.shape[0] - 1, n_lp)
                a_lp = np.interp(x_new, x_old, best_seq)
            else:
                a_lp = best_seq.copy()
            a_lp = np.maximum(a_lp, 0)
            a_lp = a_lp / a_lp.sum()

    m = np.zeros_like(u)

    beta2 = 0.999

    base_lr = 0.1
    lam = 0.5
    lam_factor = 1.005

    best_seq = a.copy()
    best_val = init_best_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # compute probabilities
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a = a / a.sum()

        # conv
        n = a.shape[0]
        conv = _full_convolution(a)

        w = _softmax_weights(conv, lam)

        grad_u = a * (grad_a - dot)

"""
Propose a low‑value step‑function.

The algorithm works in three stages:

1. **Generate a diverse set of seed sequences** (uniform,
   half‑sine, geometric decay, random Dirichlet, and the historic
   `height_sequence_1` if available).  The best of these seeds is kept.

2. **Refine the best seed with a long Adam run** on a smooth‑max
   surrogate of the convolution infinity‑norm.  The variables are kept
   in log‑space (soft‑max) so the simplex constraint is satisfied
   automatically.  The smoothing parameter λ is increased slowly and
   capped (λ_max=20) to keep the gradient well‑behaved.

3. **Optional linear‑programming improvement** on a down‑sampled
   version of the current best sequence.  We solve
   maximize Σb  subject to  a⋆b ≤ M  and  b≥0  (where M is the current
   maximal self‑convolution).  The LP solution is mixed back into the
   sequence and up‑sampled.  This step often reduces the max‑convolution
   further with only a few linear solves.

All heavy work (FFT‑based convolution, soft‑max, gradients, Adam
updates, LP construction) is performed by top‑level helper functions,
so the main `propose_candidate` function is a thin orchestrator that
obeys the 1000 s time budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·(conv_k‑max(conv)))/Σ_j."""
    m = conv.max()
    # exponent is ≤0 → no overflow, only possible underflow
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety fallback
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇_a  φ_λ(a)  = 2·(w ★ a)   (cross‑correlation)."""
    n = a.shape[0]
    rev_a = a[::-1]
    size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    deadline: float,
    init_best_val: float,
    init_best_seq: np.ndarray,
    max_iters: int = 30000,
) -> Tuple[np.ndarray, float]:
    """
    Run Adam on the smooth‑max surrogate starting from ``a_init``.
    Returns the best sequence (as a NumPy array) and its exact
    evaluation value.
    """
    a = a_init.copy()
    u = np.log(a + 1e-30)            # logits for the soft‑max parametrisation
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.1

    # smoothing schedule
    lam = 0.5
    lam_factor = 1.005
    lam_max = 20.0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convert logits -> probabilities (simplex) -----
        u_shift = u - np.max(u)          # numerical stability
        a = np.exp(u_shift)
        a /= a.sum()

        # ----- convolution and soft‑max weighting -----
        conv = _full_convolution(a)
        w = _softmax_weights(conv, lam)

        # ----- gradient w.r.t. a -----
        grad_a = _softmax_gradient(a, w)

        # ----- transform gradient to log‑space (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        it_f = float(it)
        m_hat = m / (1.0 - beta1 ** it_f)
        v_hat = v / (1.0 - beta2 ** it_f)

        lr = base_lr * (0.9999 ** it_f)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- increase smoothing (capped) -----
        lam = min(lam * lam_factor, lam_max)

        # ----- periodic exact evaluation -----
        if it % 200 == 0:
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] iter {it:05d}  new best = {val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (small problem size)
# ----------------------------------------------------------------------
def _lp_improve(
    a: np.ndarray,
    deadline: float,
    current_best_val: float,
    current_best_seq: np.ndarray,
) -> Tuple[np.ndarray, float]:
    """
    Down‑sample ``a`` to a tractable size, solve a few LPs that
    maximise Σb under the convolution bound, mix the solution back,
    up‑sample and keep any improvement.
    """
    from scipy.optimize import linprog

    n_full = a.shape[0]
    n_lp = min(800, n_full)  # keep the LP modestly sized

    # ----- down‑sample to n_lp points -----
    if n_lp < n_full:
        x_old = np.arange(n_full)
        x_new = np.linspace(0, n_full - 1, n_lp)
        a_lp = np.interp(x_new, x_old, a)
    else:
        a_lp = a.copy()

    a_lp = np.maximum(a_lp, 0.0)
    a_lp /= a_lp.sum()

    best_seq = current_best_seq
    best_val = current_best_val

    for lp_iter in range(3):
        if time.time() >= deadline:
            break

        n = a_lp.shape[0]
        conv = np.convolve(a_lp, a_lp)
        M = conv.max()

        rows = 2 * n - 1
        # Build constraint matrix A_ub where each row encodes (a⋆b)[k] ≤ M
        A_ub = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A_ub[i : i + n, i] = a_lp[i]

        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σb ⇒ minimise –Σb
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            print(f"[LP] iteration {lp_iter} failed – {res.message}")
            break

        b_opt = res.x
        if b_opt.sum() == 0.0:
            break
        g = b_opt / b_opt.sum()

        # ----- convex mix with the current a_lp -----
        t_mix = 0.5
        a_lp = (1.0 - t_mix) * a_lp + t_mix * g
        a_lp = np.maximum(a_lp, 0.0)
        a_lp /= a_lp.sum()

        # ----- up‑sample back to full size for evaluation -----
        if n_lp < n_full:
            up_seq = np.interp(
                np.arange(n_full), np.linspace(0, n_full - 1, n_lp), a_lp
            )
        else:
            up_seq = a_lp.copy()
        up_seq = np.maximum(up_seq, 0.0)
        up_seq /= up_seq.sum()

        val = evaluate_sequence(up_seq.tolist())
        print(f"[LP] iteration {lp_iter}  val = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = up_seq.copy()
            print(f"[LP]   global best improved to {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Return a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine respects the given time budget,
    generates several seed shapes, refines the best one with Adam and
    (if time permits) an LP‑based improvement, and finally returns the
    best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Choose a problem size that balances expressiveness and speed.
    # --------------------------------------------------------------
    if budget_s > 300:
        n = 3000
    elif budget_s > 120:
        n = 2000
    else:
        n = 800

    # --------------------------------------------------------------
    # 1️⃣  Build a pool of diverse initial candidates.
    # --------------------------------------------------------------
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    # Half‑sine (strictly positive interior)
    i = np.arange(1, n + 1, dtype=np.float64)
    sine = np.sin(np.pi * i / (n + 1))
    sine = np.maximum(sine, 0.0)
    candidates.append(sine / sine.sum())

    # Geometric decays with random ratios
    for _ in range(4):
        r = np.random.uniform(0.2, 0.95)
        geom = np.power(r, np.arange(n, dtype=np.float64))
        candidates.append(geom / geom.sum())

    # Random Dirichlet draws
    for _ in range(4):
        cand = np.random.dirichlet(np.ones(n))
        candidates.append(cand)

    # Historic construction if it exists
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, list) and len(height_sequence_1) > 0:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.shape[0] != n:
                x_old = np.arange(prev.shape[0])
                x_new = np.linspace(0, prev.shape[0] - 1, n)
                prev = np.interp(x_new, x_old, prev)
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = np.full(n, 1.0 / n)
            else:
                prev = prev / prev.sum()
            candidates.append(prev)
    except NameError:
        pass

    # --------------------------------------------------------------
    # 2️⃣  Evaluate all seeds, keep the best.
    # --------------------------------------------------------------
    best_val = float("inf")
    best_seq = None

    for idx, cand in enumerate(candidates):
        if time.time() > deadline:
            break
        val = evaluate_sequence(cand.tolist())
        print(f"[Init] candidate {idx:02d}  val = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print("[Init]   ** new best **")

    # --------------------------------------------------------------
    # 3️⃣  Long Adam run starting from the best seed.
    # --------------------------------------------------------------
    if best_seq is not None:
        print(f"[Main] Starting Adam refinement (current best = {best_val:.6f})")
        best_seq, best_val = _adam_optimize(
            best_seq, deadline, best_val, best_seq, max_iters=30000
        )

    # --------------------------------------------------------------
    # 4️⃣  Optional LP‑based refinement (only if time remains).
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        try:
            print("[Main] Starting LP refinement")
            best_seq, best_val = _lp_improve(best_seq, deadline, best_val, best_seq)
        except Exception as exc:                     # scipy may be missing
            print("[Main] LP refinement skipped:", exc)

    # --------------------------------------------------------------
    # 5️⃣  Final sanity checks and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s == 0.0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])
    else:
        best_seq = best_seq / s

    print(f"[Result] best value = {best_val:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
