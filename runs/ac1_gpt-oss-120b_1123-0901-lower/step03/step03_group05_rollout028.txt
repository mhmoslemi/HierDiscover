max‑convolution objective, followed by simulated‑annealing‑style

"""
Softmax‑weighted Adam optimiser for the max‑auto‑convolution problem.

The objective (with the L1‑norm fixed to sqrt(2·n)) is to minimise
    M(a) = max_k (a * a)[k] .
A smooth surrogate of the max is the log‑sum‑exp (soft‑max):
    F(a) = (1/β)·log Σ_k exp(β·(a*a)[k]) .
Its gradient is
    ∂F/∂a_i = 2· Σ_k w_k·a_{k-i},
where w_k = exp(β·(a*a)[k]) / Σ_j exp(β·(a*a)[j]) .
The implementation evaluates all convolutions by FFT,
computes the soft‑max weights, obtains the gradient via an FFT‑based
cross‑correlation, and performs an Adam update.  After the Adam phase
a cheap coordinate‑swap hill‑climb (and a tiny simulated‑annealing
acceptance rule) refines the solution.  Several lengths, initialisations
and random restarts are tried within the allotted time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, N_conv: int):
    """
    Full auto‑convolution a*a using a pre‑allocated FFT length N_conv.
    Returns the full convolution (length 2·n‑1) and its maximum.
    """
    A = np.fft.rfft(a, n=N_conv)
    conv = np.fft.irfft(A * A, n=N_conv)[: 2 * a.shape[0] - 1]
    return conv, conv.max()


def _correlate_fft(a: np.ndarray, w: np.ndarray, N_corr: int):
    """
    Correlation rev[i] = Σ_j a[j]·w[i+j]   (0 ≤ i < len(a)).
    Both vectors are zero‑padded to length N_corr before the FFT.
    The offset for lag 0 is len(w)‑1.
    """
    A = np.fft.rfft(a, n=N_corr)
    W = np.fft.rfft(w, n=N_corr)
    corr = np.fft.irfft(A * np.conj(W), n=N_corr)
    start = w.shape[0] - 1            # = 2·n‑2
    return corr[start:start + a.shape[0]]


def _softmax_adam_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    *,
    max_iter: int = 200_000,
    jitter_interval: int = 4000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
):
    """
    Adam optimisation that uses a soft‑max weighted gradient
    (instead of a plain active‑set average).
    Returns possibly updated (best_seq, best_score).
    """
    # normalise start vector
    a = a_init.copy()
    a *= target_sum / a.sum()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05

    # FFT sizes – they stay constant for a given length
    N_conv = 1 << ((2 * n - 1).bit_length())          # ≥ 2·n‑1
    N_corr = 1 << ((3 * n - 2).bit_length())          # ≥ 3·n‑2

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------
        # 1) full auto‑convolution and soft‑max weights
        conv_full, cur_max = _conv_full_fft(a, N_conv)

        # update β linearly (helps to first move smoothly, later focus on the true max)
        beta = beta_start + (beta_end - beta_start) * it / max_iter
        # soft‑max of the convolution values (shifted for numerical stability)
        diff = conv_full - cur_max                # ≤ 0
        w = np.exp(beta * diff)
        w_sum = w.sum()
        if w_sum == 0.0:               # should never happen, but be safe
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------------------------------------------------
        # 2) gradient = 2·rev where rev is correlation a * w
        rev = _correlate_fft(a, w, N_corr)       # length n
        grad = 2.0 * rev

        # ------------------------------------------------------------
        # 3) Adam update (projected onto the simplex afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------
        # 4) projection onto feasible set (non‑negative + fixed L1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s
        else:                               # extremely unlikely
            a[:] = target_sum / n

        # ------------------------------------------------------------
        # 5) occasional jitter to break plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # ------------------------------------------------------------
        # 6) bookkeeping
        if cur_max < best_score:
            best_score = float(cur_max)
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:7d}] new best M = {best_score:.6f}")

    return best_seq, best_score

def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 30_000,
):
    """
    Very cheap hill‑climb (with a tiny simulated‑annealing flavour).
    Randomly moves a small amount of mass from a donor index to a receiver
    and accepts the move if it improves the maximal convolution, or
    with a tiny probability depending on a cooling temperature.
    """
    a = a_start.copy()
    temperature = 0.02                     # initial temperature for SA moves
    decay = 0.9995                         # geometric cooling

    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1

        i = np.random.randint(n)           # donor
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)       # receiver ≠ donor
        if j >= i:
            j += 1

        eps = a[i] * np.random.rand() * 0.05   # up to 5 % of donor mass
        if eps <= 0:
            continue

        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps

        M_new, _ = _conv_full_fft(new_vec, 1 << ((2 * n - 1).bit_length()))
        delta = M_new - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # simulated‑annealing acceptance
            if np.random.rand() < math.exp(-delta / max(temperature, 1e-9)):
                accept = True

        if accept:
            a = new_vec
            best_score = M_new
            print(f"[SA local] improvement to {best_score:.6f} (prop {proposal})")
            temperature *= decay                       # cool down slowly

    return a, best_score

                r_val = 1.0 - eps
                shift = np.random.randint(0, n // 10)
                init_a = (r_val ** np.arange(n)).astype(float)
                init_a = np.roll(init_a, shift)
            else:
                # completely random Dirichlet‑like start
                init_a = np.random.rand(n)

            init_a = np.maximum(init_a, 0.0)
            # normalise to the required L1‑norm
            tot = init_a.sum()
            if tot > 0.0:
                init_a = init_a * (target_sum / tot)
            else:
                init_a = np.full(n, target_sum / n)

            # ----- run the soft‑max Adam optimiser -------------------------
            best_seq, best_score = _softmax_adam_optimize(
                a_init=init_a,
                n=n,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iter=200_000,
                jitter_interval=3000,
                beta_start=5.0,
                beta_end=200.0,
            )

            # ----- cheap local refinement (with tiny SA element) ----------
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=20_000,
                )

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
