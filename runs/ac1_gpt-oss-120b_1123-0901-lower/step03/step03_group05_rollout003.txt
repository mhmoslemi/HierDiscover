"""
Propose a low‑autocorrelation step function.

The algorithm explores several sequence lengths.  For each length it
runs multiple random restarts of an Adam optimiser that minimises a
smooth surrogate of the max‑autoconvolution:
    f_β(a) = (1/β)·log∑_k exp(β·(a*a)[k]).

For small β the surrogate is smooth and the gradient is the
soft‑max‑weighted sum of the reversed vectors; β is increased
gradually so that the optimisation finally behaves like a sub‑gradient
method on the true max‑value.  After the Adam phase a cheap
coordinate‑swap hill‑climb (local search) further reduces the peak.

All iterates are projected onto the feasible set (non‑negative,
∑a_i = √(2·n)).  The best vector found before the time budget expires
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_autoconv_fft(a: np.ndarray) -> np.ndarray:
    """Full (zero‑padded) autocorrelation a*a using FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the β‑soft‑max surrogate  (1/β)·log Σ exp(β·conv)
    together with the current maximal convolution value.
    """
    n = a.shape[0]

    # (1) full autocorrelation
    conv = _full_autoconv_fft(a)
    max_conv = float(conv.max())

    # (2) soft‑max weights (normalised)
    #   w_k ∝ exp(β·(conv_k – max_conv))   – subtraction avoids overflow
    exp_vals = np.exp(beta * (conv - max_conv))
    w = exp_vals / exp_vals.sum()          # shape (2·n‑1,)

    # (3) weighted gradient = 2·∑_j a_j·w_{i+j}
    #     This is a convolution of a with the reversed weight vector.
    w_rev = w[::-1]                        # length 2·n‑1
    L = len(w_rev) - 1                     # = 2·n‑2
    N2 = 1 << ((n + len(w_rev) - 1).bit_length())
    A = np.fft.rfft(a, n=N2)
    B = np.fft.rfft(w_rev, n=N2)
    conv_rev = np.fft.irfft(A * B, n=N2)[: n + len(w_rev) - 1]

    #   grad_i = 2·conv_rev[L‑i]  for i = 0..n‑1
    grad = 2.0 * conv_rev[L - np.arange(n)]

    return grad, max_conv


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max surrogate.
    Returns possibly updated (best_seq, best_score).
    """
    # ----- initialise -------------------------------------------------
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                 # base learning rate
    beta = 1.0                 # start temperature
    beta_max = 200.0

    iteration = 0
    n = a.shape[0]

    while time.time() < deadline and iteration < max_iter:
        # ---- gradient of the smooth surrogate -----------------------
        grad, cur_max = _gradient_softmax(a, beta)

        # ---- Adam update --------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / np.sqrt(iteration)               # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto feasible set ----------------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---- keep best ------------------------------------------------
        # Because we enforce ∑a = √(2·n), the evaluate_sequence score
        # equals the maximal convolution value.
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}")

        # ---- temperature schedule ------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta * 1.015)   # slowly sharpen the max

        # ---- occasional tiny jitter to escape plateaus -------------
        if iteration % 2500 == 0:
            jitter = np.random.randn(n) * (target_sum * 0.0008 / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that moves a tiny amount of mass from one
    index to another only if the maximal autocorrelation drops.
    """
    a = a_start.copy()
    n = a.shape[0]
    L = 2 * n - 1

    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1

        # pick donor with positive mass
        i = np.random.randint(n)
        if a[i] <= 1e-15:
            continue

        # pick different receiver
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a small fraction (up to 4 % of donor)
        eps = a[i] * np.random.rand() * 0.04
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # normalise (projection)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)

        # evaluate new max convolution
        conv_new = _full_autoconv_fft(a_new)
        new_max = float(conv_new.max())

        if new_max < best_score - 1e-12:
            a = a_new
            best_score = new_max
            print(f"[local] improvement → {best_score:.6f} (prop {proposal})")
    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine tries several lengths, initialises
    from the supplied `height_sequence_1` (if present) and from random
    Dirichlet draws, runs the Adam optimiser described above and finishes
    with a short local‑search hill‑climb.  The best vector found before the
    time budget expires is returned as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Load the previously discovered construction (if any)
    try:
        base = np.array(height_sequence_1, dtype=float)    # type: ignore[name-defined]
    except Exception:
        base = np.ones(2100, dtype=float)

    # ------------------------------------------------------------------
    # Global best bookkeeping
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Normalise the base to its own length
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = np.maximum(base, 0.0)
    if base.sum() > 0:
        base = base * (target_sum0 / base.sum())
    else:
        base = np.full_like(base, target_sum0 / n0)

    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore
    length_factors = [1.0, 1.2, 1.4, 1.6, 2.0]      # modest expansions
    lengths = sorted({int(n0 * f) for f in length_factors})
    max_len = 5000
    lengths = [l for l in lengths if 300 <= l <= max_len]

    # ------------------------------------------------------------------
    # Main optimisation loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        restarts = 3                               # 1 interpolation + 2 random

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise -------------------------------------------------
            if r == 0:
                # linear interpolation of the existing base onto the new grid
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, base)
                init_a = np.maximum(init_a, 0.0)
            else:
                # random Dirichlet‑like start (positive entries, normalised)
                init_a = np.random.rand(n)
                init_a = np.maximum(init_a, 0.0)

            # project onto feasible set (non‑negative, correct L1 norm)
            s = init_a.sum()
            if s > 0:
                init_a = init_a * (target_sum / s)
            else:
                init_a = np.full(n, target_sum / n)

            # ----- optimisation ----------------------------------------------
            best_seq, best_score = _adam_optimize(
                a_init=init_a,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iter=300_000,
            )

            # ----- cheap local refinement ------------------------------------
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=20_000,
                )

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but keep a safe uniform vector
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    # ------------------------------------------------------------------
    # Final sanity check (optional – keep the score for logging)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best reported score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
