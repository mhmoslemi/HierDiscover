            fw = np.fft.rfft(w, size_corr)
            frev = np.fft.rfft(rev_a, size_corr)
            corr = np.fft.irfft(fw * frev, size_corr)
            grad_a = 2.0 * corr[n-1 : n-1 + n]
            # convert to log-space
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)
            
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u**2)
            m_hat = m / (1.0 - beta1**it)
            v_hat = v / (1.0 - beta2**it)
            lr = base_lr * (0.9995**it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # shaking if stagnation
            if time.time() - last_improve > 2.0:
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0); v.fill(0)
                last_improve = time.time()
            
            # stop criteria for inner loop
            if it >= 6000:
                break
        
        # after inner loop, we might apply LP refinement if time left
        if time.time() < deadline - 10:
            # compute current max M
            conv = _full_convolution(a, size=conv_fft_size)
            M = conv.max()
            # solve LP
            # Build constraint matrix with diags
            from scipy.sparse import diags
            data = [np.full(n, a[i]) for i in range(n)]
            offsets = np.arange(n)
            A = diags(data, offsets, shape=(2*n-1, n), format='csc')
            # constraints: A @ b <= M
            b_ub = np.full(2*n-1, M)
            c = -np.ones(n)  # maximize sum(b)
            # bounds: b >= 0
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
            if res.success:
                b = res.x
                # scaling to maintain sum=1 (optional)
                # blend
                t = 0.5
                a = (1 - t) * a + t * b / b.sum()
                # run a few more Adam steps to refine
                # reset Adam state
                m = np.zeros_like(u); v = np.zeros_like(u)
                for inner in range(2000):
                    # compute a from u (softmax)
                    u_shift = u - np.max(u)
                    a = np.exp(u_shift)
                    a /= a.sum()
                    conv = _full_convolution(a, size=conv_fft_size)
                    max_val = conv.max()
                    w = (conv >= max_val - 1e-12).astype(np.float64)
                    rev_a = a[::-1]
                    size_corr = 1 << int(np.ceil(np.log2(len(w) + n -1)))
                    fw = np.fft.rfft(w, size_corr)
                    frev = np.fft.rfft(rev_a, size_corr)
                    corr = np.fft.irfft(fw * frev, size_corr)
                    grad_a = 2.0 * corr[n-1:n-1+n]
                    dot = np.dot(a, grad_a)
                    grad_u = a * (grad_a - dot)
                    m = beta1*m + (1-beta1)*grad_u
                    v = beta2*v + (1-beta2)*(grad_u**2)
                    m_hat = m/(1-beta1**(inner+1))
                    v_hat = v/(1-beta2**(inner+1))
                    lr = base_lr * (0.9995**(it+inner+1))
                    u -= lr * m_hat/(np.sqrt(v_hat)+eps)
                # evaluate after LP refinement
                conv = _full_convolution(a, size=conv_fft_size)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Restart {restart:03d}] after LP refine new best = {val:.6f}")
        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")
    
    # fallback and return
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
    print(f"Final best objective = {best_val:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a step‑function that minimizes

    F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)²

The algorithm combines three ideas:

1. **Simplex parametrisation** – the coefficients are stored in log‑space
   (soft‑max) so that they are always non‑negative and sum to 1.
2. **Sub‑gradient descent on the exact L∞‑objective** – instead of a
   smooth‑max surrogate we differentiate the true max‑convolution using a
   fast FFT based convolution with an indicator of the active constraints.
   Adam is used to stabilise the updates.
3. **Linear‑programming refinement** – after a block of gradient steps we
   solve a linear program

        max   ∑ b_j
        s.t.  (a * b)[k] ≤ M    ∀k  (M = max(a*a))
              b_j ≥ 0

   The solution tends to increase the total mass while keeping the
   convolution peak ≤ M.  Mixing the current iterate with the LP solution
   (convex combination) usually reduces the peak further.  The LP is
   built as a sparse Toeplitz matrix and solved with SciPy's HiGHS
   optimiser.

Multiple random restarts with diverse initial shapes (Dirichlet,
geometric decay, half‑sine, Gaussian, sparse spikes) are performed until
the 1000 s budget expires.  The best sequence found is returned as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of appropriate length."""
    n = a.shape[0]
    if fft_size is None:
        fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
    return conv


def _grad_maxconv(a: np.ndarray, conv: np.ndarray, max_val: float,
                  fft_size: int = None) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution w.r.t. a.
    conv = a*a (full conv), max_val = conv.max().
    Returns 2·∑_{j∈argmax} a_{j‑i}.
    """
    n = a.shape[0]
    # indicator of active constraints (allow a tiny tolerance)
    w = (conv >= max_val - 1e-12).astype(np.float64)

    # convolution w ★ reverse(a)  (size = len(w)+n-1)
    rev_a = a[::-1]
    size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice corresponding to indices i = 0 … n‑1
    grad = 2.0 * corr[n - 1: n - 1 + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact evaluation used by the benchmark (scale‑invariant)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _build_lp_matrix(f: np.ndarray) -> diags:
    """
    Build the (2n‑1)×n Toeplitz matrix A with entries
        A[k, j] = f_{k‑j}   (0 if index out of bounds)
    such that (A @ b)[k] = (f * b)[k].
    """
    n = f.shape[0]
    # each diagonal offset d contains the constant f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Length – a compromise between expressive power and speed.
    n = 2000

    # Pre‑computed FFT size for the convolution (used many times)
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.05

    best_val = float("inf")
    best_seq = None
    restart = 0

    while time.time() < deadline:
        restart += 1

        # -------------------- diverse initialisations --------------------
        mode = restart % 5
        if mode == 0:
            a = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            r = np.random.uniform(0.2, 0.95)
            a = r ** np.arange(n, dtype=np.float64)
            a = a / a.sum()
        elif mode == 2:
            i = np.arange(1, n + 1, dtype=np.float64)
            a = np.sin(np.pi * i / (n + 1))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        elif mode == 3:
            sigma = np.random.uniform(0.05 * n, 0.2 * n)
            i = np.arange(n, dtype=np.float64)
            a = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)
            a = a / a.sum()
        else:
            # Sparse random spikes
            k = max(5, int(np.clip(np.random.normal(0.07 * n, 0.03 * n), 5, n // 3)))
            idx = np.random.choice(n, size=k, replace=False)
            vals = np.random.rand(k) + 0.1
            a = np.zeros(n, dtype=np.float64)
            a[idx] = vals
            a = a / a.sum()

        # soft‑max parametrisation (log‑space)
        u = np.log(a + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        it = 0
        last_improve = time.time()

        # -------------------- Adam on the true max‑convolution --------------------
        while time.time() < deadline:
            it += 1

            # recover a from logits
            u_shift = u - np.max(u)          # numeric stability
            a = np.exp(u_shift)
            a /= a.sum()

            # exact convolution & objective
            conv = _full_convolution(a, size=conv_fft_size)
            max_val = conv.max()
            cur_val = 2.0 * n * max_val          # sum(a)=1 ⇒ denominator = 1
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Restart {restart:03d}] iter {it:05d}  new best = {cur_val:.6f}")

            # sub‑gradient of the max‑convolution
            grad_a = _grad_maxconv(a, conv, max_val, fft_size=conv_fft_size)

            # convert to log‑space gradient (soft‑max Jacobian)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = base_lr * (0.9995 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # shake if stuck for a while
            if time.time() - last_improve > 2.0:
                u += (np.random.rand(n) - 0.5) * 5e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # limit inner loop length
            if it >= 6000:
                break

        # -------------------- Linear‑programming refinement --------------------
        # Use the LP only if enough time remains (≈10 s for the solve + a few
        # extra gradient steps).
        if time.time() < deadline - 12.0:
            # current peak M
            conv = _full_convolution(a, size=conv_fft_size)
            M = conv.max()

            # Build constraint matrix A (sparse Toeplitz)
            A = _build_lp_matrix(a)

            # Linear programme: maximise sum(b)  ⇔ minimise -1·b
            c = -np.ones(n, dtype=np.float64)               # objective
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # (f*b)[k] ≤ M
            bounds = [(0, None)] * n

            # Solve with HiGHS (fast interior‑point)
            lp_res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True, 'time_limit': 8.0})

            if lp_res.success:
                b = lp_res.x
                # blend current a with the LP solution (t = 0.5 works well)
                t = 0.5
                a = (1.0 - t) * a + t * (b / b.sum())

                # short Adam polishing after the mix
                u = np.log(a + 1e-30)
                m.fill(0.0)
                v.fill(0.0)
                for sub_it in range(1500):
                    u_shift = u - np.max(u)
                    a = np.exp(u_shift)
                    a /= a.sum()
                    conv = _full_convolution(a, size=conv_fft_size)
                    max_val = conv.max()
                    grad_a = _grad_maxconv(a, conv, max_val, fft_size=conv_fft_size)
                    dot = np.dot(a, grad_a)
                    grad_u = a * (grad_a - dot)
                    m = beta1 * m + (1.0 - beta1) * grad_u
                    v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
                    m_hat = m / (1.0 - beta1 ** (sub_it + 1))
                    v_hat = v / (1.0 - beta2 ** (sub_it + 1))
                    lr = base_lr * (0.9995 ** (it + sub_it + 1))
                    u -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # final evaluation of the refined candidate
                conv = _full_convolution(a, size=conv_fft_size)
                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[Restart {restart:03d}] after LP refine new best = {cur_val:.6f}")

        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:                       # should never happen
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # Return a plain Python list – the evaluator will handle scaling etc.
    return best_seq.tolist()
