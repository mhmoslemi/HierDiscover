"""
Hybrid optimizer for the step‑function problem.

The routine runs many independent restarts within the allotted time.
Each restart picks a length `n` and an initial shape (random power‑law,
optional exponential tail or a stretched version of the current global
best).  Two complementary optimisation schemes are used:

* **Soft‑max gradient descent** – a smooth approximation of the Ⓛ∞‑
  objective with a decaying temperature.  The gradient of the
  temperature‑weighted convolution is obtained with a single FFT‑based
  cross‑correlation.

* **Exponentiated‑gradient descent** – a mirror‑descent step on the
  simplex that follows the sub‑gradient formed by the (nearly) active
  constraints (indices where the convolution is close to its maximum).

Both methods keep the vector on the simplex
  Σ aᵢ = √(2 n)  (so that the evaluation reduces to max(conv)).
After the main optimisation a fast local hill‑climbing phase shuffled a
few random mass transfers is applied to squeeze out a little more
improvement.

The best sequence found over the whole budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Σ x_i = s, x_i ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helpers for the optimisers
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Smooth‑max (log‑sum‑exp) weights."""
    max_val = conv.max()
    scaled = (conv - max_val) / max_val / tau   # normalise by max for stability
    w = np.exp(scaled - np.max(scaled))        # shift again to avoid overflow
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    w /= w_sum
    return w


def _active_weights(conv: np.ndarray, rel_tol: float = 1e-3) -> np.ndarray:
    """Uniform weights on the (approximately) active set of constraints."""
    max_val = conv.max()
    eps = max_val * rel_tol
    active = np.where(conv >= max_val - eps)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_k w_k (a*a)_k with respect to a."""
    n = a.shape[0]
    w_rev = w[::-1]
    # cross‑correlation a * w_rev (full length 3n‑2)
    conv2 = _full_convolution_fft(a, w_rev)
    # indices that correspond to Σ_i a_i w_{i+j}
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * conv2[idx]


def _objective_from_conv(conv: np.ndarray, target_sum: float) -> float:
    """Return the evaluation value (max‑conv) for a vector with given sum."""
    return 2.0 * conv.max() / (target_sum ** 2)   # n cancels after normalisation


def _stretch_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Upsample `seq` to length `new_n` using linear interpolation (preserves mass)."""
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    # after interpolation the total mass may have changed a little; we will re‑project later
    return new_seq


# ----------------------------------------------------------------------
# Soft‑max gradient descent optimiser
# ----------------------------------------------------------------------
def _optimise_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9995,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max of the self‑convolution.
    Returns (best_sequence, best_score) where the score is the true
    evaluation (max conv after the normalisation Σ a = √(2n)).
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = _objective_from_conv(conv, target_sum)

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        # --- convolution and smooth‑max weights ---------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        w = _softmax_weights(conv, tau)

        # --- gradient -----------------------------------------------------------
        grad = _gradient_from_weights(a, w)

        # --- momentum -----------------------------------------------------------
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        # --- adaptive step ------------------------------------------------------
        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --- evaluation ---------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = _objective_from_conv(conv_new, target_sum)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n} softmax new best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 5e-4 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    # final evaluation (identical rounding to the provided evaluator)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _optimise_expgd(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_eta: float = 0.2,
    eta_decay: float = 0.9995,
    active_tol: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using exponentiated updates.
    The sub‑gradient is built from the (approximately) active constraints.
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = _objective_from_conv(conv, target_sum)

    eta = init_eta
    iteration = 0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # build uniform weight on the almost‑active set
        w = _active_weights(conv, rel_tol=active_tol)

        # gradient of weighted sum (sub‑gradient of max)
        grad = _gradient_from_weights(a, w)

        # exponentiated (mirror) step – keep positivity automatically
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        a = _project_onto_simplex(a, target_sum)

        # evaluate
        conv_new = _full_convolution_fft(a, a)
        score_new = _objective_from_conv(conv_new, target_sum)

        if score_new < best_score:
            best_score = score_new
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} expgd new best = {score_new:.6f}")

        eta *= eta_decay
        iteration += 1

        # tiny random shake‑up occasionally
        if np.random.rand() < 0.008:
            noise = np.random.randn(n) * (target_sum * 4e-4 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Lightweight local hill‑climbing (random mass swaps)
# ----------------------------------------------------------------------
def _local_hill_climb(
    seq: np.ndarray,
    target_sum: float,
    max_iter: int,
    overall_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic improvement: move a tiny amount of mass from a random
    index i to a random index j while keeping the sum fixed.  Accept only
    improving moves.
    """
    n = seq.shape[0]
    a = seq.copy()
    # compute current score once
    conv = _full_convolution_fft(a, a)
    best_score = _objective_from_conv(conv, target_sum)

    for _ in range(max_iter):
        if time.time() > overall_deadline:
            break

        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue
        # move at most a small fraction of a[i] (max 0.5% of total sum)
        max_move = min(a[i], target_sum * 5e-4)
        eps = np.random.rand() * max_move
        if eps == 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps

        conv_cand = _full_convolution_fft(a_candidate, a_candidate)
        score_cand = _objective_from_conv(conv_cand, target_sum)

        if score_cand < best_score:
            best_score = score_cand
            a = a_candidate
            # report progress
            print(f"[{time.time():.2f}] local hill‑climb improved = {score_cand:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The search consists of many short restarts using either a smooth‑max
    gradient descent or an exponentiated‑gradient (mirror) descent.
    The current global best is occasionally stretched to a larger length
    and re‑optimised.  A final cheap hill‑climbing phase attempts to squeeze
    out a further improvement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # warm‑start from a globally supplied construction if available
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] warm‑start from global seed = {score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # choose a problem size – we bias towards larger n later on
        if remaining > 300:
            n = np.random.randint(800, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(600, 3000 + 1)
        else:
            n = np.random.randint(400, 2500 + 1)

        # ---------- build an initial sequence --------------------------------
        use_warm = best_seq is not None and np.random.rand() < 0.45
        if use_warm:
            # interpolate / stretch the current best to the new length
            init = _stretch_sequence(best_seq, n)
        else:
            # random power‑law (optionally tapered) initialisation
            p = np.random.uniform(0.3, 0.7)               # exponent
            idx = np.arange(1, n + 1, dtype=float)
            init = idx ** (-p)
            if np.random.rand() < 0.35:
                tau_exp = np.random.uniform(0.4, 2.0) * n
                init *= np.exp(-idx / tau_exp)
            # tiny jitter to break symmetry
            init += np.random.rand(n) * 1e-4

        # ---------- decide which optimiser to use ---------------------------
        if np.random.rand() < 0.55:
            optimiser = _optimise_softmax
            opt_kwargs = dict(
                init_tau_factor=0.12,
                min_tau_factor=0.001,
                step_factor=0.65,
                step_decay=0.9996,
                momentum_beta=0.9,
            )
        else:
            optimiser = _optimise_expgd
            opt_kwargs = dict(
                init_eta=0.22,
                eta_decay=0.9994,
                active_tol=1e-3,
            )

        # allocate a time slice for this restart
        slice_time = min(remaining * 0.45, 12.0)
        run_deadline = time.time() + slice_time

        seq_local, score_local = optimiser(n, init, run_deadline, **opt_kwargs)

        # ---------- cheap hill‑climbing on the obtained sequence -------------
        target_sum = np.sqrt(2.0 * n)
        hc_deadline = deadline - 0.05
        seq_local, score_local = _local_hill_climb(
            seq_local,
            target_sum,
            max_iter=250,
            overall_deadline=hc_deadline,
        )
        # final evaluation again (ensures identical rounding)
        score_local = evaluate_sequence(seq_local.tolist())

        # ---------- update global best ---------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # early stop if we have comfortably beaten the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
