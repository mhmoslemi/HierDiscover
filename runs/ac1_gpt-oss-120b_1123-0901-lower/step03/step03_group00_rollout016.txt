def proj_simplex(v):
    # ensure nonneg, sum=1
    n = len(v)
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

using a two‑stage Adam optimiser on the softmax parametrisation.
Stage 1 uses a smooth‑max surrogate (exponential weights) to bring the
distribution into a low‑peak region, while Stage 2 focuses on the
largest few convolution entries (top‑K weighting) to fine‑tune the
maximum.  The optimiser runs several restarts on a range of lengths,
warm‑starts from the best known construction (`height_sequence_1`
if present) and adds occasional random shakes when progress stalls.
The best sequence found before the time budget expires is returned as a
plain Python list (the external `evaluate_sequence` will be called on it
outside this code).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative sequence with low evaluation score.
    The routine respects the given time budget, prints progress and
    finally returns the best sequence (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: convert a probability vector to logits (add epsilon for stability)
    def _logits_from_probs(p: np.ndarray) -> np.ndarray:
        return np.log(p + 1e-15)

    # ------------------------------------------------------------------
    # Problem sizes to explore (larger n gives more freedom)
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (powers of two) for the two convolutions we need.
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ len(w)+len(a)-1

        # Number of restarts per length – we allocate a bit more for smaller n.
        restarts = 5 if n <= 4096 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------- initialise logits -----------------------
            if r == 0 and best_seq is not None and len(best_seq) == n:
                # Warm‑start from the globally best sequence (if length matches)
                init_a = np.array(best_seq, dtype=np.float64)
                init_a /= init_a.sum()
                x = _logits_from_probs(init_a)
                print("[restart 0] warm start from previous best")
            elif 'height_sequence_1' in globals():
                # Optional warm‑start from a supplied construction
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand = cand / s
                x = _logits_from_probs(cand)
                print("[restart 0] start from height_sequence_1")
            else:
                # Pure random start
                x = np.random.randn(n) * 0.1

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05

            # Smooth‑max temperature (stage 1)
            lam = 0.5
            lam_factor = 1.03

            t = 0
            last_improve = time.time()
            # iteration budget – generous for small n, tighter for large n
            max_iters = 25000 if n <= 4096 else 18000
            # stage division (40 % of iterations use smooth‑max)
            stage1_iters = int(0.4 * max_iters)

            while t < max_iters and time.time() < deadline:
                # ----- logits → simplex (softmax) -----
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)               # stabilise
                a = exp_x / exp_x.sum()                # a_i ≥ 0, Σ a_i = 1

                # ----- self‑convolution -----
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()

                # ----- choose weighting for the surrogate -----
                if t < stage1_iters:
                    # smooth‑max weighting (exponential)
                    w = np.exp(lam * (conv - max_c))
                    w_sum = w.sum()
                    if w_sum == 0:
                        w = np.ones_like(conv) / conv.size
                    else:
                        w = w / w_sum
                    lam = min(lam * lam_factor, 1e6)   # increase temperature
                else:
                    # top‑K uniform weighting (hard max focus)
                    top_k = 5
                    # np.argpartition is O(n) – fast enough
                    top_idx = np.argpartition(-conv, top_k - 1)[:top_k]
                    w = np.zeros_like(conv)
                    w[top_idx] = 1.0 / top_k

                # ----- gradient of the surrogate w.r.t. a -----
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # ----- back‑prop through softmax (gradient w.r.t. logits) -----
                avg = np.dot(a, grad_a)                 # ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)             # ∇_x φ

                # ----- Adam update -----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- exact objective (using the true max) -----
                # Σ a_i = 1 by construction → denominator = 1
                val = float(2.0 * n * max_c)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

                # ----- stagnation handling -----
                if time.time() - last_improve > 5.0:
                    # small random perturbation of the logits
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

    # ------------------------------------------------------------------
    # Fallback – return a simple uniform vector if everything failed.
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Convert to plain Python list (evaluate_sequence expects this).
    return best_seq.tolist()
