def _optimise_one_length_adam(n, init, run_deadline):
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)
    best_a = a.copy()
    # compute initial score
    conv = _full_convolution_fft(a, a)
    best_score = 2.0*n*conv.max()/(target_sum**2)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    init_tau_factor = 0.05
    min_tau_factor = 0.001
    step_scale = 0.8
    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # compute smoothed weights
        tau = max(init_tau_factor * max_val * (0.90 ** t), min_tau_factor * max_val)
        # stable softmax
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient via cross-correlation
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step_dir = m_hat / (np.sqrt(v_hat) + eps)
        # step size based on max_val (like earlier)
        max_abs = np.max(np.abs(step_dir)) + 1e-12
        eta = step_scale * max_val / max_abs

"""
Propose a non‑negative height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search consists of many short restarts, each one using a projected
Adam optimiser on a soft‑max approximation of the max‑convolution.  A
handful of classic window shapes (power‑law, exponential, Hann,
Hamming, Black‑man, triangular, Gaussian) are used to initialise the
optimiser, and promising solutions are refined with a small random
local‑search stage.  The best sequence found before the time budget
expires is returned (as a plain Python list).  The code respects the
1000 s time limit, prints progress information and never uses
closures or filesystem I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (sum = s, non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # remove tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper – Generate a random initial shape
# ----------------------------------------------------------------------
def _generate_initial_sequence(n: int) -> np.ndarray:
    """Return a random non‑negative vector of length n from a collection of
    common window‑type families."""
    idx = np.arange(n, dtype=float)

    # pick a family uniformly at random
    family = np.random.choice(
        ["power", "exp", "hann", "hamming", "blackman", "triangular", "gaussian"]
    )

    if family == "power":
        p = np.random.uniform(0.2, 1.2)
        seq = (idx + 1.0) ** (-p)

    elif family == "exp":
        tau = np.random.uniform(0.3, 2.0) * n
        seq = np.exp(-idx / tau)

    elif family == "hann":
        seq = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))

    elif family == "hamming":
        seq = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))

    elif family == "blackman":
        seq = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))
        )

    elif family == "triangular":
        # symmetric triangle that touches zero at the ends
        seq = 1.0 - np.abs(idx - (n - 1) / 2.0) / ((n - 1) / 2.0)
        seq = np.clip(seq, 0.0, None)

    elif family == "gaussian":
        sigma = np.random.uniform(0.2, 0.6) * n
        centre = (n - 1) / 2.0
        seq = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)

    else:  # fallback – uniform
        seq = np.ones(n, dtype=float)

    # tiny jitter to break symmetry
    seq += np.random.rand(n) * 1e-5
    return seq


# ----------------------------------------------------------------------
# Optimiser – projected Adam on a soft‑max surrogate of the max convolution
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` starting from ``init``.
    Returns the best sequence (as a NumPy array) and its *exact* score
    obtained from ``evaluate_sequence``.
    """
    target_sum = np.sqrt(2.0 * n)                 # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # current best (internal objective = max(conv) )
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)
    best_a = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # hyper‑parameters for the soft‑max smoothing
    init_tau_factor = 0.05           # start with a fairly sharp approximation
    min_tau_factor = 0.001
    step_scale = 0.8                # controls absolute step length

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) self‑convolution and its maximum
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) soft‑max weights (smooth approximation of the max)
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (0.90 ** t), min_tau_factor * max_val)
        # shift for numerical stability
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of Σ w_k (a*a)_k.  It is a cross‑correlation.
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)           # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                  # positions for Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 4) Adam update
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step_dir = m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) step size (scaled so that a typical update is a fraction of max_val)
        # --------------------------------------------------------------
        max_abs = np.max(np.abs(step_dir)) + 1e-12
        eta = step_scale * max_val / max_abs

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 6) evaluate new candidate (internal score)
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score_new:.6f}")

        a = a_new

        # occasional random shake‑up to escape plateaux
        if np.random.rand() < 0.008:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    # ------------------------------------------------------------------
    # Simple random local‑search refinement (few hundred cheap perturbations)
    # ------------------------------------------------------------------
    for _ in range(200):
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.randn() * target_sum * 0.0005 / n
        cand = best_a.copy()
        cand[i] = max(cand[i] + delta, 0.0)
        cand[j] = max(cand[j] - delta, 0.0)
        cand = _project_onto_simplex(cand, target_sum)

        conv_c = _full_convolution_fft(cand, cand)
        s_c = 2.0 * n * conv_c.max() / (target_sum ** 2)
        if s_c < best_score:
            best_score = s_c
            best_a = cand.copy()

    # ------------------------------------------------------------------
    # Final evaluation through the provided evaluator (guarantees identical
    # rounding/clipping handling)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length ``n`` (typically a few thousand) and initialise
      a shape from a diverse set of classic window functions or power‑law
      profiles.
    * Optimise each candidate with a projected Adam optimiser that
      minimises a soft‑max surrogate of the maximum self‑convolution.
    * After each run a tiny random local search is performed.
    * The overall best across many independent restarts is kept.
    * Whenever the bound drops convincingly below the target (≈1.503) the
      search terminates early.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # spare margin for final return

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a globally supplied sequence, if present.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        arr = np.array(hs, dtype=float)
        arr = np.clip(arr, 0.0, 1000.0)
        score = evaluate_sequence(arr.tolist())
        best_seq = arr.copy()
        best_score = score
        print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size.  Early on we explore larger n,
        # later we focus on moderately sized problems.
        if remaining > 400:
            n = np.random.randint(2000, 8000 + 1)
        elif remaining > 200:
            n = np.random.randint(1500, 5000 + 1)
        else:
            n = np.random.randint(800, 3000 + 1)

        # ------------------- initialise a candidate --------------------
        init = _generate_initial_sequence(n)

        # sometimes bootstrap from the current global best (truncating / padding)
        if best_seq is not None and np.random.rand() < 0.4:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])

        # Allocate a time slice for this optimisation run
        slice_time = min(12.0, remaining * 0.6)      # at most ~12 s per restart
        run_deadline = time.time() + slice_time

        # Run the optimiser for the chosen length
        seq_local, score_local = _optimise_one_length_adam(
            n,
            init,
            run_deadline,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # Early exit when we have comfortably beaten the target bound.
            if best_score < 1.48:
                break

    # ------------------------------------------------------------------
    # Fallback – generate a trivial uniform sequence if nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
