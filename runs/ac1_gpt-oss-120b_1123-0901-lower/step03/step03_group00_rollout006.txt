def propose_candidate(...):
    seed random
    deadline = ...
    best_seq = None
    best_val = inf

    # define lambda schedule and training hyperparameters.
    lengths = [8192, 4096, 2048, 1024] # smaller for fallback

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"Optimizing n={n}")
        fft_len_conv = next_pow2(2*n-1)
        fft_len_corr = next_pow2(3*n)

        for restart in range(5):
            if time.time() > deadline: break

            # init logits
            if best_seq is not None and len(best_seq) == n:
                # warm start from previous best at this length
                init = np.array(best_seq)
                init = np.maximum(init, 0)
                init_sum = init.sum()
                if init_sum < 0.01:
                    init = np.random.dirichlet(np.ones(n))
                else:
                    init /= init_sum
                x = np.log(init + 1e-15)
                print(f"restart {restart}: warm start")
            else:
                # random
                x = np.random.randn(n) * 0.1

            # set Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            # lambda schedule
            lam = 0.5
            lam_factor = 1.5  # maybe more aggressive
            t = 0
            last_improve = time.time()
            max_iters = 50000

            while time.time() < deadline and t < max_iters:
                # compute a
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()
                # compute conv, objective
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c / (a.sum()**2)
                # check best
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"n={n} restart={restart} iter={t} new best {val:.6f}")
                    last_improve = time.time()
                # compute gradient via smooth max
                w = np.exp(lam * (conv - max_c))
                w /= w.sum()
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)
                # backprop to logits
                avg = np.dot(a, grad_a)
                grad_x = a * (grad_a - avg)
                # Adam update
                t += 1
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x**2)
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # lambda schedule: increase every few iterations
                if t % 200 == 0:
                    lam = min(lam * lam_factor, 1e7)

                # stagnation detection
                if time.time() - last_improve > 8.0:
                    # small random shake
                    x += (np.random.rand(n) - 0.5) * 1e-1
                    print(f"n={n} restart={restart} iter={t} shake")
                    last_improve = time.time()

            # After finishing this restart, run local refinement
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                # precompute conv for a_local
                conv_local = _full_convolution(a_local, fft_len=fft_len_conv)
                val_local = 2.0 * n * conv_local.max() / (a_local.sum()**2)
                # Local search
                steps = 0
                max_local_steps = 20000
                epsilon = 1e-3  # max delta mass per move
                while time.time() < deadline and steps < max_local_steps:
                    # select indices
                    i = np.random.randint(0, n)
                    if a_local[i] < 1e-12:
                        # skip if zero
                        steps += 1
                        continue
                    j = np.random.randint(0, n)
                    if i == j:
                        steps += 1
                        continue
                    delta = min(a_local[i], epsilon) * np.random.rand()
                    a_new = a_local.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    conv_new = _full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max() / (a_new.sum()**2)
                    if val_new < val_local:
                        a_local = a_new
                        val_local = val_new

                            best_seq = a_local.copy()

    # fallback uniform
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0/n_fallback)
    # convert to list
    return best_seq.tolist()

def _cross_correlation(w, a, fft_len):
    # linear cross-correlation of w (len 2n-1) with a (len n), same as (w ★ a) described earlier.
   # want vector length n: (w ★ a)[i] = sum_k w[k] * a[i + k - (n-1)]
    # Equivalent to conv(w, reversed(a)), then extract slice.
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = len(a) - 1
    return corr[start:start+len(a)]

"""
Hybrid optimizer for minimizing max convolution ‖a*a‖∞ under a≥0, Σa=1.

The algorithm works in three stages:
1. Adam updates on the log‑softmax parametrisation with a smooth‑max
   surrogate of the ℓ∞‑norm.  The temperature λ is increased gradually,
   turning the surrogate into the true max.
2. If progress stalls, a small random perturbation (“shake”) is added to
   the logits to escape local plateaus.
3. After the gradient phase a cheap hill‑climbing local search moves a
   small amount of mass between two coordinates; any improving move is
   kept.  This often reduces the objective a few percent further.

Multiple problem sizes (n=8192, 4096, …) and a few random restarts are
tried within the supplied time budget.  The best sequence found is
returned as a plain Python list, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT.  Returns length 2·len(a)-1 array."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i + k - (n-1)] for i=0…n-1.
    w length = 2·n-1, a length = n, fft_len ≥ 3·n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²).  a must be 1‑D."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    return 2.0 * n * max_c / (s * s)

    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence().  Returns the best sequence found as a plain
    Python list.  The routine respects the provided time budget.

    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_val = float("inf")

    def _logits_from_probs(p):
        return np.log(p + 1e-15)

    # Problem sizes to explore
    lengths = [8192, 4096, 2048, 1024]

        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        fft_len_corr = _next_pow2(3 * n)

        for r in range(restarts):

                x = _logits_from_probs(a0)

                if s < 0.01:

                else:
                    cand = cand / s
                x = _logits_from_probs(cand)
                print("[restart 0] start from height_sequence_1")
            else:

                if np.random.rand() < 0.5:
                    x = np.random.randn(n) * 0.2
                    print(f"[restart {r}] random Gaussian logits")
                else:

                    a0 = np.random.dirichlet(np.full(n, alpha))
                    x = _logits_from_probs(a0)
                    print(f"[restart {r}] random Dirichlet (α={alpha:.3f})")

            m = np.zeros(n, dtype=np.float64)

            beta1, beta2, eps = 0.9, 0.999, 1e-8

                a = exp_x / exp_x.sum()  # Σ a = 1, a ≥ 0

                # ----- exact objective (used for early stopping & logging) -----
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c / (a.sum() ** 2)

                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()

                # on those where conv is close to its maximum.
                diff = lam * (conv - max_c)          # max entry -> 0, others ≤ 0
                w = np.exp(diff)
                w_sum = w.sum()

                    mask = conv == max_c
                    w = mask.astype(float)
                    w_sum = w.sum()
                w /= w_sum

                # gradient w.r.t. a (not logits)
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # back‑propagate through soft‑max
                avg = np.dot(a, grad_a)                     # ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)                # ∇_x φ

                # ----- Adam update -------------------------------------------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule ------------------------------------------------
                if t % 200 == 0:
                    lam = min(lam * lam_factor, lam_max)

                # ----- stagnation handling ------------------------------------------------
                if time.time() - last_improve > 7.0:
                    # small random perturbation of logits
                    shake = (np.random.rand(n) - 0.5) * 1e-1
                    x += shake
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

            # ------------------------------------------------------------------
            # Optional cheap local refinement after each restart
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                val_local = _objective(a_local)
                # number of local moves – limited by remaining time
                max_local_steps = 20000
                step = 0
                eps = 1e-3  # max mass moved in one step
                while time.time() < deadline and step < max_local_steps:
                    i = np.random.randint(0, n)
                    if a_local[i] < 1e-12:
                        step += 1
                        continue
                    j = np.random.randint(0, n)
                    if i == j:
                        step += 1
                        continue
                    delta = min(a_local[i], eps) * np.random.rand()
                    a_new = a_local.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    val_new = _objective(a_new)
                    if val_new + 1e-12 < val_local:  # strict improvement
                        a_local = a_new
                        val_local = val_new
                        if val_local < best_val:
                            best_val = val_local
                            best_seq = a_local.copy()
                            print(f"[local n={n} step={step}] improved to {val_local:.6f}")
                    step += 1

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return as a plain Python list
    return best_seq.tolist()

                print("[restart 0] start from height_sequence_1")
            else:
                # Pure random start – either Gaussian logits or Dirichlet probabilities
                if np.random.rand() < 0.5:
                    x = np.random.randn(n) * 0.2
                    print(f"[restart {r}] random Gaussian logits")
                else:
                    alpha = np.random.uniform(0.05, 2.0)
                    a0 = np.random.dirichlet(np.full(n, alpha))
                    x = _logits_from_probs(a0)
                    print(f"[restart {r}] random Dirichlet (α={alpha:.3f})")

            # -------------------- Adam state --------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07

            # Smooth‑max temperature (λ)
            lam = 0.3
            lam_factor = 1.2
            lam_max = 1e5

            t = 0
            last_improve = time.time()
            max_iters = 25000   # safety cap per restart

            # -------------------- Main Adam loop --------------------
            while time.time() < deadline and t < max_iters:
                # Soft‑max → probability vector on the simplex
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()      # Σa = 1, a_i ≥ 0

                # Exact objective (used for logging / early exit)
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best {val:.6f}")

                    if best_val < 1.503:   # reached the target threshold
                        print("Target reached – exiting early.")
                        deadline = time.time()   # break outer loops soon

                # ------- Smooth‑max surrogate gradient -------
                # w_k ∝ exp(λ·(conv_k – max_c))   (max entry gets weight 1)
                diff = lam * (conv - max_c)      # max entry → 0, others ≤ 0
                w = np.exp(diff)
                w_sum = w.sum()
                if w_sum == 0.0:                # extreme under‑flow protection
                    mask = conv == max_c
                    w = mask.astype(float)
                    w_sum = w.sum()
                w /= w_sum

                # Gradient w.r.t. a (not logits)
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # Back‑propagation through the soft‑max
                avg = np.dot(a, grad_a)          # ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)      # ∇_x φ

                # ----- Adam update -----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- Temperature schedule -----
                if t % 200 == 0:
                    lam = min(lam * lam_factor, lam_max)

                # ----- Stagnation handling -----
                if time.time() - last_improve > 7.0:
                    # Small random shake of the logits to escape plateaus
                    x += (np.random.rand(n) - 0.5) * 1e-1
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

            # -------------------- Local refinement (hill‑climbing) --------------------
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                val_local = _objective(a_local)
                max_local_steps = 20000
                step = 0
                eps_move = 1e-3   # max mass shifted per move

                while time.time() < deadline and step < max_local_steps:
                    i = np.random.randint(0, n)
                    if a_local[i] < 1e-12:
                        step += 1
                        continue
                    j = np.random.randint(0, n)
                    if i == j:
                        step += 1
                        continue
                    delta = min(a_local[i], eps_move) * np.random.rand()
                    a_new = a_local.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    val_new = _objective(a_new)
                    if val_new + 1e-12 < val_local:      # accept only strict improvements
                        a_local = a_new
                        val_local = val_new
                        if val_local < best_val:
                            best_val = val_local
                            best_seq = a_local.copy()
                            print(f"[local n={n} step={step}] improved to {val_local:.6f}")
                    step += 1

    # ------------------------------------------------------------------
    # Fallback: uniform distribution (should never be needed)
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
