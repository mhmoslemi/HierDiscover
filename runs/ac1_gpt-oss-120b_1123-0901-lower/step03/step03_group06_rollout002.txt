def simulated_annealing(p, best_score, deadline, max_iters=200000):

    n = len(p)

    current_p = p.copy()
    current_score = best_score

    best_local_p = p.copy()
    best_local_score = best_score

    it = 0

    while time.time() < deadline and it < max_iters:

        it += 1

        # propose move:
        i = np.random.randint(n)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)
        # epsilon random fraction of p[i]
        max_move = current_p[i]
        if max_move == 0:
            continue
        epsilon = max_move * np.random.rand() * 0.1 # up to 10% of mass at i
        # move epsilon from i to j
        new_p = current_p.copy()
        new_p[i] -= epsilon
        new_p[j] += epsilon

        # compute new score
        conv = _full_convolution_fft(new_p, _next_pow2(2*n-1))
        M = conv.max()
        new_score = 2 * n * M

        # acceptance
        if new_score < current_score or np.random.rand() < np.exp((current_score - new_score) / (0.01 + 0.001 * it)):
            current_p = new_p
            current_score = new_score

            if new_score < best_local_score:
                best_local_score = new_score
                best_local_p = new_p.copy()
                # print improvement
                if it % 1000 == 0:
                    print(f"SA iter {it}: new best {best_local_score}")

        # optional temperature schedule: we may vary acceptance probability factor.

    return best_local_p, best_local_score

"""
Candidate generator for the step‑function minimisation task.

Algorithm
---------
1. Initialise a probability vector `p` (sum = 1) from the known
   construction ``height_sequence_1`` if available, otherwise use a
   uniform distribution of modest length.
2. Optimise `p` with *Exponentiated Gradient* (mirror descent on the
   simplex) for a sequence of lengths.  For each length `n` we:
   * Upsample the current best probability vector to the new length
     by linear interpolation.
   * Run many EG iterations, using a soft‑max smoothing of the
     self‑convolution to obtain a differentiable surrogate of the
     max‑peak.  The smoothing parameter `β` is increased from a tiny
     value to a larger one, while the step size decays.
   * Keep the best vector seen during the run.
   * Perform a few random restarts (small Gaussian perturbations of
     the start) to reduce the chance of getting stuck.
3. After the multiscale phase, run a short *hill‑climbing* refinement
   (a deterministic simulated‑annealing style walk):
   * Randomly move a small amount of mass from one coordinate to
     another and accept the move only if it reduces the peak.
   * Repeat while time permits.
4. Convert the final probability vector `p` back to the required
   step heights `a = sqrt(2 n)·p` (the evaluation is scale invariant)
   and return the list.

All heavy work is done with FFTs (O(n log n)) and the search stops
before the global deadline (default 1000 s).  The routine prints
progress and the best score found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Compute the full self‑convolution p∗p using a single FFT of size N.
    Returns a real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k-i}   (i = 0..n−1).
    This is a correlation of p with w, obtained via FFT.
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)   # reversed weights
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]    # keep linear part
    # extract the segment corresponding to i = 0..n‑1 and reverse
    rev = conv_rev[L - n : L][::-1]
    return rev


def _score_from_prob(p: np.ndarray) -> float:
    """Return the evaluation score 2·n·max(p∗p) for a probability vector."""
    n = p.shape[0]
    N = _next_pow2(2 * n - 1)
    conv = _full_convolution_fft(p, N)
    return 2.0 * n * conv.max()


def _interpolate_prob(p_old: np.ndarray, n_new: int) -> np.ndarray:
    """
    Linear interpolation of an old probability vector onto a new length.
    The result is re‑normalised to sum to 1.
    """
    if p_old.shape[0] == n_new:
        return p_old.copy()
    old_x = np.linspace(0.0, 1.0, p_old.shape[0])
    new_x = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(new_x, old_x, p_old)
    p_new = np.maximum(p_new, 0.0)
    if p_new.sum() == 0.0:                # extremely unlikely
        p_new = np.full(n_new, 1.0 / n_new)
    else:
        p_new /= p_new.sum()
    return p_new


def _eg_optimize(
    p_start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 0.01,
    beta_end: float = 30.0,
    jitter_interval: int = 2000,
    jitter_scale: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation of a probability vector of size `n`.
    Returns the best vector encountered and its score.
    """
    logp = np.log(p_start + 1e-30)               # stay away from log(0)
    best_score = float("inf")
    best_p = p_start.copy()

    # FFT size that accommodates both convolution (2n‑1) and correlation (3n‑2)
    N = _next_pow2(3 * n - 2)

    beta_len = int(0.6 * max_iters)  # iterations over which beta grows

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ---- convolution and current objective ----
        p = np.exp(logp - np.max(logp))          # stable exponentiation
        p /= p.sum()
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n:5d} it {it:7d}] new local best = {cur_score:.6f}")

        # ---- soft‑max smoothing ----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))            # w ≤ 1, w_max = 1

        # ---- correlation needed for the gradient ----
        rev = _weighted_rev_fft(p, w, N)

        # gradient of the smoothed max (see paper)
        grad = 4.0 * n * rev

        # ---- step size (decreasing) ----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # ---- EG update in log‑space ----
        logp -= eta * grad

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

    # final projection before returning
    p = np.exp(logp - np.max(logp))
    p /= p.sum()
    final_score = _score_from_prob(p)
    if final_score < best_score:
        best_score = final_score
        best_p = p.copy()
    return best_p, best_score


def _hill_climb(
    p_start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simple deterministic hill‑climbing (move a bit of mass from i to j).
    Accept only improving moves.  Returns the best vector found.
    """
    p = p_start.copy()
    best_score = _score_from_prob(p)
    best_p = p.copy()
    rng = np.random.default_rng()

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # propose a random transfer of mass
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or p[i] == 0.0:
            continue
        # up to 5 % of the mass at i is moved
        eps = p[i] * rng.random() * 0.05
        if eps <= 0.0:
            continue
        p_new = p.copy()
        p_new[i] -= eps
        p_new[j] += eps

        new_score = _score_from_prob(p_new)
        if new_score < best_score:
            best_score = new_score
            best_p = p_new.copy()
            p = p_new
            if it % 1000 == 0:
                print(f"[HC len {n:5d} it {it:7d}] new best = {best_score:.6f}")
        else:
            # keep the old p (no uphill moves)
            pass
    return best_p, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise from the previously discovered construction when possible.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        print("[init] loaded external construction")
    except Exception:
        # fall back to a simple uniform start
        base = np.ones(500, dtype=float)
        print("[init] using uniform fallback")

    # Normalise to a probability vector (sum = 1)
    p_current = base / base.sum()

    # Global best tracking (probability vector + its score)
    best_p = p_current.copy()
    best_score = _score_from_prob(best_p)
    best_len = p_current.shape[0]
    print(f"[init] length={best_len}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Multiscale schedule – we allocate more iterations for larger n.
    length_schedule = [500, 1000, 1500, 2100, 3000, 5000, 8000, 12000, 16000]
    # keep only lengths that we can actually afford (avoid absurdly large ones)
    length_schedule = [l for l in length_schedule if l >= 300 and l <= 20000]
    length_schedule = sorted(set(length_schedule))

    for n in length_schedule:
        if time.time() >= deadline:
            break

        print(f"[phase] up‑sampling to length {n}")

        # Interpolate current best probability vector to the new length
        p_start = _interpolate_prob(p_current, n)

        # Run a few random restarts (2‑3) and keep the best outcome
        n_restarts = 3 if n >= 5000 else 2
        local_best_p = None
        local_best_score = float("inf")

        for rs in range(n_restarts):
            if time.time() >= deadline:
                break

            # a tiny Gaussian jitter on log‑probabilities for diversity
            jitter = np.random.randn(n) * 0.02
            logp_start = np.log(p_start + 1e-30) + jitter
            logp_start = np.exp(logp_start - np.max(logp_start))
            p_jit = logp_start / logp_start.sum()

            # allocate a slice of the remaining time to this run
            remaining = deadline - time.time()
            # rough budget: a fixed proportion per restart, never > 150 k iters
            max_it = min(150_000, int(remaining * 200))  # heuristic
            p_opt, score_opt = _eg_optimize(
                p_start=p_jit,
                n=n,
                deadline=deadline,
                max_iters=max_it,
                beta_start=0.01,
                beta_end=30.0,
                jitter_interval=2000,
                jitter_scale=0.015,
            )
            if score_opt < local_best_score:
                local_best_score = score_opt
                local_best_p = p_opt.copy()
                print(f"[len {n}] restart {rs+1}/{n_restarts} new local best = {score_opt:.6f}")

        # after the restarts, possibly improve further with hill‑climbing
        if time.time() < deadline and local_best_p is not None:
            hc_iters = min(80_000, int((deadline - time.time()) * 250))
            p_refined, refined_score = _hill_climb(
                p_start=local_best_p,
                n=n,
                deadline=deadline,
                max_iters=hc_iters,
            )
            if refined_score < local_best_score:
                local_best_score = refined_score
                local_best_p = p_refined.copy()
                print(f"[len {n}] hill‑climb improved to {refined_score:.6f}")

        # Update global best if improvement found
        if local_best_p is not None and local_best_score < best_score:
            best_score = local_best_score
            best_p = local_best_p.copy()
            best_len = n
            print(f"[global] new best at length {n}: score = {best_score:.6f}")

        # carry the best found vector forward (as a probability distribution)
        p_current = best_p.copy()

    # ------------------------------------------------------------------
    # Final conversion back to step heights (scale invariant)
    S = math.sqrt(2.0 * best_len)
    a_final = best_p * S

    # Final verification (the harness will also call evaluate_sequence)
    final_score = evaluate_sequence(a_final.tolist())
    print(f"[done] final score = {final_score:.6f}, length = {best_len}")

    return a_final.tolist()
