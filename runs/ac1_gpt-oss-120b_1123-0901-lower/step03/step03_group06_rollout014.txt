def optimize(p_start, n, deadline):
    p = p_start.copy()
    logp = np.log(p + 1e-12)
    N = next_pow2(3*n)
    beta_start=0.1
    beta_end=40.0
    beta_len = 100000
    # Adam
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    best_score = np.inf
    best_p = p.copy()
    while time.time() < deadline and t < 500_000:   // maybe
        t+=1
        conv = fft_self_conv(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
        # compute w
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / beta_len)
        exp_arg = beta * (conv - M)
        # Clip for numeric stability: any exp_arg < -50 treat as 0.
        w = np.exp(np.clip(exp_arg, -50, 0))
        # weighted rev:
        rev = weighted_rev_fft(p, w, N)  # length n
        grad = 4.0 * n * rev
        # Adam update in log-space
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 * m_hat / (np.sqrt(v_hat) + epsilon)  # learning rate adjustable
        logp = logp - step
        # jitter occasionally
        if t % 5000 == 0:
            logp += np.random.randn(n) * 0.001
        # project
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)

Algorithm

    (w_k = 1 on maximal entries, 0 otherwise) to approach the true sub‑gradient.
  - The gradient of 2·n·M is   ∇ = 4·n·rev, where
        rev_i = Σ_k w_k·p_{k‑i}
    and rev is computed as a correlation via an FFT.
  - Update the log‑probabilities:  logp ← logp – η·∇,
    with a slowly decaying step‑size η = (1/(4n))/(1+it·1e‑5).
  - Project back onto the simplex by exponentiation and normalisation.
  - Small Gaussian jitter is added every few thousand steps to escape plateaus.
* Multi‑scale schedule: optimise first on a moderate size (≈2 k),
  then up‑sample the best probability vector to larger lengths
  (5 k, 10 k, 20 k).  The up‑sampling is linear interpolation on the
  probability vector.
* The whole search is time‑driven (≤ budget seconds).  For each length a
  time budget proportional to its size is allocated.  The best
  sequence (a = √(2n)·p) seen over all lengths is returned.

The routine complies with the required signature `propose_candidate`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Full self‑convolution (p∗p) using an FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i}  (correlation) via FFT.
    p has length n, w has length 2·n‑1.
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)   # reversed weights
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]    # keep linear part
    rev = conv_rev[L - n : L][::-1]     # extract the required slice and reverse
    return rev


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√(x_i (1−x_i)),  x_i = (i+½)/n."""
    i = np.arange(n) + 0.5
    x = i / n
    # avoid exact 0/1 – they are never reached because of the 0.5 shift
    p = 1.0 / np.sqrt(x * (1.0 - x))
    p /= p.sum()
    return p


def _interpolate_p(p_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_x = np.linspace(0.0, 1.0, p_old.shape[0])
    new_x = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(new_x, old_x, p_old)
    p_new = np.maximum(p_new, 0.0)
    p_new /= p_new.sum()
    return p_new


def _optimize_eg_length(
    n: int,
    p_start: np.ndarray,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector (sum = 1) and its score
    ``2·n·max(p∗p)``.
    """
    deadline = time.time() + time_limit
    # Parameters (tuned for the 1000 s total budget)
    max_iters = 600_000                    # safety cap
    jitter_interval = 5000
    jitter_scale = 0.001
    beta_start = 0.1
    beta_end = 40.0
    beta_len = 200_000                     # iterations over which beta grows
    # step‑size η = (1/(4n)) / (1 + it·1e‑5)
    p = np.maximum(p_start, 1e-30)        # avoid exact zeros
    logp = np.log(p)

    # FFT size that works for both the self‑convolution (2n‑1) and
    # the correlation with the weight vector (3n‑2)
    N = _next_pow2(3 * n - 2)

    best_local_score = float("inf")
    best_local_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution (p∗p) -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()

        cur_score = 2.0 * n * M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- weight vector for sub‑gradient -----
        if it < beta_len:
            beta = beta_start + (beta_end - beta_start) * (it / beta_len)
            # soft‑max weights, numerically stable
            exp_arg = beta * (conv - M)      # ≤ 0
            # clip to avoid underflow (exp(->50)≈0)
            exp_arg = np.clip(exp_arg, -50, 0)
            w = np.exp(exp_arg)              # w_max = 1
        else:
            # hard mask – exact sub‑gradient of the true max
            w = (conv >= M - 1e-12).astype(float)

        # ----- correlation rev = Σ_k w_k·p_{k‑i} -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the (smoothed) max -----
        grad = 4.0 * n * rev                # ∇_p (2·n·M)

        # ----- step size (η decays slowly) -----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-5)

        # ----- EG update in log‑space -----
        logp -= eta * grad

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        max_log = np.max(logp)               # for numerical stability
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

    return best_local_p, best_local_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The function respects a time budget (default 1000 s) and returns the best
    sequence found.  It employs a multi‑scale exponentiated‑gradient
    optimisation starting from a discrete arcsine law.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore
        # Normalise to canonical sum √(2·n)
        n0 = base_seq.shape[0]
        S0 = math.sqrt(2.0 * n0)
        base_seq = base_seq * (S0 / base_seq.sum())
        best_seq = base_seq.copy()
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[init] loaded previous sequence length={n0}, score={best_score:.6f}")
    except Exception:
        # No previous data – start from a fresh arcsine law of length 2000
        n0 = 2000
        p0 = _arcsine_distribution(n0)
        S0 = math.sqrt(2.0 * n0)
        best_seq = p0 * S0
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[init] start from arcsine length={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Multi‑scale schedule (larger lengths later)
    length_schedule = [2000, 5000, 10000, 20000]
    length_schedule = [l for l in length_schedule if l >= 300]

    # Main loop over the chosen lengths
    for idx, n in enumerate(length_schedule):
        if time.time() >= deadline:
            break

        # ----- initialise probability vector for this length -----
        if best_seq.shape[0] == n:
            p_init = best_seq / math.sqrt(2.0 * n)   # because Σa = √(2n)
        else:
            # Interpolate the current best probability vector
            p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])
            p_init = _interpolate_p(p_old, n)

        # ----- allocate time for this length (proportional to n) -----
        remaining = deadline - time.time()
        # total size of the not‑yet‑processed lengths
        remaining_lengths = length_schedule[idx:]
        total_n = sum(remaining_lengths)
        # give each length a share proportional to its n
        time_budget = max(30.0, remaining * (n / total_n))
        # leave a small safety margin
        time_budget = min(time_budget, remaining - 5.0)
        if time_budget <= 0:
            break

        print(f"[run] length={n}, allocated_time={time_budget:.1f}s")

        # ----- run the EG optimiser for this length -----
        p_opt, local_score = _optimize_eg_length(
            n=n,
            p_start=p_init,
            time_limit=time_budget,
        )

        # Convert back to the original scale (a = √(2·n)·p)
        S = math.sqrt(2.0 * n)
        a_opt = p_opt * S
        cur_score = evaluate_sequence(a_opt.tolist())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[len {n}] improved global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety check – if something went wrong fall back to uniform
    if best_seq is None or best_seq.size == 0:
        n = 2000
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # The harness expects a plain Python list
    return best_seq.tolist()
