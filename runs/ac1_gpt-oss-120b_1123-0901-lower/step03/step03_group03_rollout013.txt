def _smooth_max_optimize(a_init, n, target_sum, deadline, max_iters,
                         tau_factor, decay_rate, lr0,
                         jitter_interval, jitter_scale):
    a = a_init.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    coeff = 4.0 * n / (target_sum * target_sum)
    iteration = 0
    Nfft = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_score = np.inf
    while time.time() < deadline and iteration < max_iters:
        # compute convolution A_fft * A_fft etc.
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)
        rev = _weighted_rev(a, w, Nfft)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const
        iteration += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0 / (1 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection to nonnegative simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * jitter_scale * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        # evaluate actual objective
        cur_score = M  # after normalization, equals exact max conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[len {n} iter {iteration}] new best = {best_score:.6f}")
    return best_a, best_score

      Keep the best seed for this length.
2.  Starting from the best seed, run a *soft‑max Adam* optimiser that
    minimises a smooth approximation of the maximal self‑convolution.
    The optimiser differs from the reference implementation by
    – randomising all hyper‑parameters per run (initial temperature
      factor, decay rate, learning‑rate, jitter schedule);
    – allowing many short independent runs with different random
      initialisations (a few Dirichlet samples) to escape local minima;
    – more aggressive jitter early on.
    The iteration loop is fully time‑bounded; the exact objective
    (max‑convolution) is monitored and the global best vector is kept.
3.  After the time budget (≈1000 s) is exhausted, the best vector found
    over *all* lengths is returned as a plain Python list.

The implementation uses only ``numpy`` (FFT for speed) and respects the
required time budget.  No external files or network access are performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.

    Returns
    -------
    w : np.ndarray
        Weights that sum to one.
    Ltau : float
        Smooth approximation τ·log∑exp(conv/τ).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute the vector
        rev_i = Σ_k w_k * a_{k-i}
    (indices outside 0…len(a)-1 are ignored) using FFTs.

    Parameters
    ----------
    a : np.ndarray
        Current step‑function (length n).
    w : np.ndarray
        Weight vector of length 2·n‑1 (soft‑max weights).
    Nfft : int
        Power‑of‑two length used for the zero‑padded FFTs.

    Returns
    -------
    rev : np.ndarray, shape (n,)
        The weighted reverse convolution required for the gradient.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the part that contains the needed indices
    conv_rev = conv_rev[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _max_self_convolution(a: np.ndarray) -> float:
    """
    Exact maximal self‑convolution of a (full convolution).
    Assumes ``a`` is already normalised to the required ℓ₁‑norm.
    """
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _smooth_max_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    tau_factor: float,
    decay_rate: float,
    lr0: float,
    jitter_interval: int,
    jitter_scale: float,
):
    """
    Adam optimiser that minimises a smooth approximation of the maximal
    self‑convolution.

    Parameters
    ----------
    a_init : np.ndarray
        Starting point (will be normalised to ``target_sum``).
    n : int
        Length of the vector.
    target_sum : float
        Desired ℓ₁‑norm (√(2 n)).
    deadline : float
        Unix timestamp – stop once ``time.time()`` exceeds it.
    max_iters : int
        Hard iteration limit (also stopped early by the deadline).
    tau_factor, decay_rate, lr0, jitter_interval, jitter_scale :
        Hyper‑parameters controlling the smooth‑max temperature, Adam
        learning‑rate schedule and jitter.

    Returns
    -------
    best_a : np.ndarray
        The best vector discovered during the run.
    best_score : float
        Its exact objective value (max self‑convolution).
    """
    # enforce the ℓ₁‑norm constraint
    a = a_init.astype(float).copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    coeff = 4.0 * n / (target_sum * target_sum)   # factor 4 n / S²
    iteration = 0

    Nfft = 1 << ((2 * n - 1).bit_length())

    best_a = a.copy()
    best_score = _max_self_convolution(best_a)

    while time.time() < deadline and iteration < max_iters:
        # ----- 1) full convolution a∗a (FFT) -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- 2) smooth max (soft‑max) -----
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # ----- 3) weighted reverse convolution (sub‑gradient of Lτ) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- 4) Adam gradient step (objective with ℓ₁‑constraint) -----
        const = 4.0 * n * Ltau / (target_sum ** 3)    # derivative of the normalisation term
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 5) projection onto the feasible set (a ≥ 0, Σa = target_sum) -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ----- 6) jitter to help escape plateaus -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (jitter_scale * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- 7) exact score (after normalisation it equals max conv) -----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Retrieve the previously known construction if available
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)   # fallback uniform base
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    best_seq = base.copy()
    best_score = _max_self_convolution(best_seq)
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – include original and a few larger ones
    length_factors = [1.0, 1.25, 1.5, 2.0]   # as in the reference run
    lengths = sorted({int(n0 * f) for f in length_factors})
    # also add a couple of hand‑picked larger lengths (still fast enough)
    lengths.extend([5000, 8000])
    lengths = [l for l in lengths if l >= 300]
    lengths = [min(l, 12000) for l in lengths]   # hard cap for safety
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Search over each length
    for n in lengths:
        # check remaining time
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1) generate a portfolio of seed vectors for this length
        seeds = []

        # uniform
        seeds.append(np.ones(n, dtype=float))

        # triangular (peak in the middle)
        idx = np.arange(n, dtype=float)
        tri = 1.0 - np.abs(idx - (n - 1) / 2.0) / ((n - 1) / 2.0)
        seeds.append(np.maximum(tri, 0.0))

        # half‑sine (always positive)
        half_sine = np.sin(np.pi * (idx + 0.5) / n)
        seeds.append(half_sine)

        # cosine‑squared (aka sine‑squared shape)
        cos_sq = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
        seeds.append(cos_sq)

        # Gaussian bump centred in the middle
        sigma = n / 5.0
        gau = np.exp(-0.5 * ((idx - (n - 1) / 2.0) / sigma) ** 2)
        seeds.append(gau)

        # exponential decay from the left
        for alpha in (0.0005, 0.001, 0.002, 0.005):
            exp_decay = np.exp(-alpha * idx)
            seeds.append(exp_decay)

        # power‑law decay
        for p in (0.5, 1.0, 2.0):
            pw = (idx + 1.0) ** (-p)
            seeds.append(pw)

        # a few random Dirichlet‑like samples
        for _ in range(8):
            rand = rng.random(n) + 1e-6   # avoid exact zeros
            seeds.append(rand)

        # --------------------------------------------------------------
        # 2) evaluate all seeds (fast FFT based exact objective)
        best_seed = None
        best_seed_score = float('inf')
        for vec in seeds:
            a = vec.astype(float)
            a = a * (target_sum / a.sum())
            sc = _max_self_convolution(a)
            if sc < best_seed_score:
                best_seed_score = sc
                best_seed = a
        if best_seed is None:  # should never happen
            best_seed = np.full(n, target_sum / n)
            best_seed_score = _max_self_convolution(best_seed)

        print(f"[len {n}] best seed score = {best_seed_score:.6f}")

        # --------------------------------------------------------------
        # 3) allocate an iteration budget for this length
        time_left = deadline - time.time()
        if time_left < 5.0:
            break

        # Rough scaling: more iterations for smaller n (they are cheaper)
        max_iters = int(5_000_000 * n0 / n)
        max_iters = max(2_000, max_iters)   # safeguard a minimum

        # --------------------------------------------------------------
        # 4) run several short Adam optimisations with randomised hyper‑params
        runs_for_this_n = max(1, int(time_left // 20))   # aim ~20 s per run
        runs_for_this_n = min(runs_for_this_n, 6)        # cap to keep overhead low

        for run_id in range(runs_for_this_n):
            if time.time() >= deadline:
                break

            # possibly perturb the seed a little before each run
            a0 = best_seed.copy()
            if run_id > 0:
                perturb = rng.normal(scale=0.02, size=n) * (target_sum / n)
                a0 = np.maximum(a0 + perturb, 0.0)
                a0 = a0 * (target_sum / a0.sum())

            # random hyper‑parameters
            tau_factor = rng.uniform(0.2, 0.8)
            decay_rate = rng.uniform(15000.0, 50000.0)
            lr0 = rng.uniform(0.03, 0.07)
            jitter_interval = int(rng.uniform(2000, 8000))
            jitter_scale = rng.uniform(5e-4, 2e-3)

            a_opt, score_opt = _smooth_max_optimize(
                a_init=a0,
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                max_iters=max_iters,
                tau_factor=tau_factor,
                decay_rate=decay_rate,
                lr0=lr0,
                jitter_interval=jitter_interval,
                jitter_scale=jitter_scale,
            )

            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[global] new best from len {n}, run {run_id} -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _max_self_convolution(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain list as expected by the evaluation harness
    return best_seq.tolist()
