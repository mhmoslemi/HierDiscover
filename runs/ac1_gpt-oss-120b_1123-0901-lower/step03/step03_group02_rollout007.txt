"""
Proposes a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a Frank‑Wolfe style scheme:

* Fix a length `n` and normalise the sequence to Σ a_i = √(2 n).
* For the current sequence `a` compute M = max(a * a).
* Solve a linear programme:
        maximise   Σ b_j
        subject to  (a * b)_k ≤ M      for every convolution index k
                    b_j ≥ 0
  (the convolution constraints are linear in `b`).
* The optimal `b` has a sum at least √(2 n).  Scale it back to the
  simplex, then move a little towards it:
        a ← (1‑t)·a + t·b_scaled .
  The maximum convolution never increases.
* Repeat the LP‑step a few times, then restart with a new random length
  and a new initial shape (power‑law, exponential tail or a warm‑start
  from the current global best).  The best sequence seen before the
  budget expires is returned.

If SciPy is unavailable the implementation falls back to a simple
random‑initialisation gradient descent (the previous baseline).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – convolution utilities
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Return max of the full linear convolution a * a."""
    return np.convolve(a, a).max()


# ----------------------------------------------------------------------
# LP‑based Frank‑Wolfe step (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _solve_lp_for_b(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise Σ b_j   subject to   (a * b)_k ≤ M  (∀k) ,  b ≥ 0.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = a.shape[0]
    # Build the Toeplitz matrix defining the convolution constraints:
    #   A[i, j] = a[i - j] if 0 ≤ i-j < n else 0
    col = np.concatenate([a, np.zeros(n - 1, dtype=float)])          # length 2n‑1
    row = np.concatenate([np.array([a[0]], dtype=float), np.zeros(n - 1, dtype=float)])
    A = toeplitz(col, row)                                          # (2n‑1) × n
    # Right‑hand side: all entries must be ≤ M
    b_ub = np.full(2 * n - 1, M, dtype=float)
    # Objective: maximise Σb ⇒ minimise −Σb
    c = -np.ones(n, dtype=float)
    # Variable bounds: b_j ≥ 0
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast interior‑point/simplex hybrid)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


def _optimise_one_length_fw(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    t0: float = 0.5,
    t_decay: float = 0.97,
    max_iters: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style optimiser for a fixed length `n`.
    Returns the best sequence (numpy array) found within the time slice
    and its evaluation value (via `evaluate_sequence`).
    """
    target_sum = math.sqrt(2.0 * n)
    # initialise on the simplex
    a = _project_onto_simplex(init.astype(float), target_sum)

    # current max convolution
    M = _conv_max(a)
    best_a = a.copy()
    best_M = M

    iteration = 0
    t = t0

    while time.time() < run_deadline and iteration < max_iters:
        # ------------------------------------------------------------------
        # 1) Solve LP to obtain a direction b that maximises Σb under
        #    the current convolution bound M.
        # ------------------------------------------------------------------
        b_opt = _solve_lp_for_b(a, M)
        if b_opt is None:
            # LP failure – abort this length
            break

        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break

        # ------------------------------------------------------------------
        # 2) Scale b back to the simplex (sum = target_sum).  Because sum_b
        #    ≥ target_sum (a itself is feasible) the scaling factor ≤ 1.
        # ------------------------------------------------------------------
        scale = target_sum / sum_b
        b = b_opt * scale

        # ------------------------------------------------------------------
        # 3) Move a little towards b.  The mixing factor `t` decays over
        #    iterations to stabilise the process.
        # ------------------------------------------------------------------
        a_new = (1.0 - t) * a + t * b
        # Numerical drift may break the simplex – re‑project just in case.
        a_new = _project_onto_simplex(a_new, target_sum)

        # ------------------------------------------------------------------
        # 4) New max convolution and bookkeeping.
        # ------------------------------------------------------------------
        M_new = _conv_max(a_new)

        if M_new < best_M - 1e-12:                     # strict improvement
            best_M = M_new
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] n={n} iter={iteration} new best M={best_M:.6f}")

        # Prepare for next iteration
        a = a_new
        M = M_new
        iteration += 1
        t = max(0.02, t * t_decay)                    # never let t vanish

    # Final evaluation with the provided evaluator (ensures identical clipping)
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, final_score

# Gradient‑descent fallback (the previous baseline)
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9995,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Soft‑max (smooth‑max) sub‑gradient descent for a fixed length `n`.
    This is kept as a safe fallback when SciPy is unavailable.
    """
    target_sum = math.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        conv = np.convolve(a, a)
        max_val = conv.max()

        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        conv2 = np.convolve(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        conv_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n} new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

# Main entry point required by the evaluator

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function within `budget_s` seconds.  The method repeatedly
    picks a random length, builds a power‑law (or warm‑started) initial
    shape and runs the Frank‑Wolfe LP optimiser.  The best sequence seen
    is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a globally provided construction if available.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm start

    # ------------------------------------------------------------------
    # Helper for initial shape generation
    # ------------------------------------------------------------------
    def _make_initial(n, use_global=False):
        if use_global and best_seq is not None:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])
            return init
        # power‑law + optional exponential tail
        p = np.random.uniform(0.3, 0.7)
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.4, 2.0) * n
            init *= np.exp(-idx / tau_exp)
        init += np.random.rand(n) * 1e-4
        return init

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 3.0:
            break

        # pick a length – early budget prefers a wider range
        if remaining > 300:
            n = np.random.randint(800, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(600, 3000 + 1)
        else:
            n = np.random.randint(400, 2200 + 1)

        # initialise a candidate shape
        use_global = (best_seq is not None) and (np.random.rand() < 0.45)
        init_arr = _make_initial(n, use_global=use_global)

        # allocate a slice of the remaining time for this run
        slice_time = min(8.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # ------------------------------------------------------------------
        # Choose optimiser based on SciPy availability
        # ------------------------------------------------------------------
        if _SCIPY_AVAILABLE:
            seq_local, score_local = _optimise_one_length_fw(
                n,
                init_arr,
                run_deadline,
                t0=0.5,
                t_decay=0.96,
                max_iters=25,
            )
        else:
            seq_local, score_local = _optimise_one_length_softmax(
                n,
                init_arr,
                run_deadline,
                init_tau_factor=0.12,
                min_tau_factor=0.001,
                step_factor=0.6,
                step_decay=0.9994,
                momentum_beta=0.9,
            )

        # update global best
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            if best_score < 1.503:     # target reached – stop early
                break

    # ------------------------------------------------------------------
    # Fallback: uniform sequence if nothing was generated (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
