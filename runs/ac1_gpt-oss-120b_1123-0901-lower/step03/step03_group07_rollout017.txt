"""
Propose a non‑negative step function with a low evaluation score.

The routine consists of three stages:

1. **Coarse parametric search** – a handful of
   symmetric shape families (beta‑polynomial, exponential,
   Gaussian, power‑law and geometric) are generated for a few
   different lengths (n≈500…3000).  For each candidate the exact
   objective 2 n·max(a*a)/(∑a)² is evaluated with an FFT based
   convolution.  The best sequence seen so far is kept.

2. **Local refinement** – the current best sequence is refined
   with a projected‑gradient (Adam) optimizer that works on the
   smooth‑max surrogate of the max‑convolution.  The optimizer
   runs as long as time permits (up to ~15000 iterations).

3. **Fallback** – if nothing better than a uniform vector is found
   the uniform sequence of the chosen length is returned.

All operations respect the time budget (default 1000 s).  The final
sequence is returned as a plain Python list; it can be fed to
``evaluate_sequence`` directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for fast convolution and objective evaluation
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_conv(seq: np.ndarray, fftsize: int = None) -> float:
    """Return max_k (seq * seq)[k] using a single FFT."""
    n = seq.shape[0]
    if fftsize is None:
        fftsize = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, fftsize)
    conv = np.fft.irfft(f * f, fftsize)[: 2 * n - 1]
    return float(conv.max())


def _objective(seq: np.ndarray, fftsize: int = None) -> float:
    """Exact objective 2·n·max(conv) / (sum(seq))²."""
    n = seq.shape[0]
    max_b = _max_conv(seq, fftsize)
    s = seq.sum()
    # sum is guaranteed > 0 for all candidates we generate
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Shape families – all produce a non‑negative vector that sums to 1
# ----------------------------------------------------------------------
def _shape_beta(n: int, beta: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))**beta  (symmetric, beta may be negative)."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 1) * (n - i)
    a = np.power(base, beta, out=np.empty_like(base))
    # avoid possible overflow for extreme beta – clamp to reasonable range
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        a.fill(1.0 / n)
    else:
        a /= total
    return a


def _shape_exp(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-alpha·|i - (n‑1)/2|)"""
    d = np.abs(np.arange(n, dtype=np.float64) - (n - 1) / 2.0)
    a = np.exp(-alpha * d)
    a /= a.sum()
    return a


def _shape_gauss(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-0.5·(dist/σ)²)  (σ>0)"""
    d = np.abs(np.arange(n, dtype=np.float64) - (n - 1) / 2.0)
    a = np.exp(-0.5 * (d / sigma) ** 2)
    a /= a.sum()
    return a


def _shape_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1+|i‑mid|)^(-p)  (p≥0)"""
    d = np.abs(np.arange(n, dtype=np.float64) - (n - 1) / 2.0)
    a = (1.0 + d) ** (-p)
    a /= a.sum()
    return a


def _shape_geom(n: int, r: float) -> np.ndarray:
    """Geometric (non‑symmetric) a_i ∝ r^i."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient utilities for the Adam refinement stage
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """Gradient of the smooth‑max surrogate w.r.t. a (formula 2·(w ★ a))."""
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _adam_refine(a0: np.ndarray,
                 max_iters: int = 15000,
                 deadline: float = 1e9,
                 lam_start: float = 1.0,
                 lam_factor: float = 1.02) -> np.ndarray:
    """
    Adam optimisation of the smooth‑max surrogate starting from a0.
    Returns the best point encountered (projected onto the simplex).
    """
    n = a0.shape[0]
    # FFT sizes reused throughout the refinement
    conv_fft = _next_pow2(2 * n - 1)
    corr_fft = _next_pow2(3 * n)

    # log‑softmax parametrisation
    u = np.log(a0 + 1e-30)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.03
    lam = lam_start

    best_val = float("inf")
    best_a = a0.copy()
    last_improve = time.time()
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # probability vector on simplex
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # smooth‑max of the convolution
        conv = _full_convolution(a, size=conv_fft)
        w = _softmax_weights(conv, lam)

        # gradient in original space
        grad_a = _softmax_gradient(a, w, size_corr=corr_fft)

        # transform to log‑space (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr * (0.9995 ** it)

        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase smoothness slowly
        lam *= lam_factor

        # occasional exact evaluation
        if it % 30 == 0:
            val = _objective(a, fftsize=conv_fft)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                last_improve = time.time()
                # cheap progress report
                print(f"[Refine] iter {it:05d}  new best = {val:.6f}")

        # simple shake if stagnating
        if time.time() - last_improve > 1.2:
            u += (np.random.rand(n) - 0.5) * 5e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    return best_a

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a minimal evaluation score.

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

            best_n = seq_prev.shape[0]

        except Exception:
            best_val = float("inf")
            best_seq = None
            best_n = None
    else:
        best_val = float("inf")
        best_seq = None
        best_n = None

    # Main exploration loop
    for n in n_candidates:

            if val_ref < best_val:
                best_val, best_seq = val_ref, refined.copy()
                print(f"  [Shake] refined best = {val_ref:.6f}")

    # ------------------------------------------------------------------
