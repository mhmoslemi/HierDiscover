            jitter = np.random.randn(n) * (jitter_scale * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # update best (exact objective = M after normalisation)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[adam n={n} iter={iteration} M={M:.6f}] new best")

    return best_seq, best_score


def _hardmax_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int,
    lr0: float = 0.03,
    decay: float = 0.6,
):
    """Projected sub‑gradient descent using the exact max entry of the
    convolution (hard‑max).  This is a final polishing step."""
    n = a0.shape[0]
    a = a0.copy()
    a = a * (target_sum / a.sum())
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_score = float("inf")
    best_seq = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        p = int(conv.argmax())
        M = float(conv[p])

        # gradient of the hard‑max entry
        w = np.zeros_like(conv)
        w[p] = 1.0
        rev = _weighted_rev(a, w, Nfft)
        grad = 2.0 * rev   # derivative of Σ_i a_i a_{p-i}

        lr = lr0 / (1.0 + decay * it)
        a = a - lr * grad

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[hardmax n={n} iter={it} M={M:.6f}] improvement")

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Load any previously discovered construction
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Phase 1 – discover a good shape on a modest size
    n_small = 800
    target_sum_small = math.sqrt(2.0 * n_small)

    best_small_seq = None
    best_small_score = float("inf")

    # allocate roughly half of the total budget to this phase
    phase1_deadline = start_time + 0.55 * (deadline - start_time)
    restart = 0
    while time.time() < phase1_deadline:
        restart += 1
        if base is not None:
            # interpolate previous solution onto the small grid
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n_small)
            init = np.interp(new_x, old_x, base)
            init = np.maximum(init, 0.0)
        else:
            # random Dirichlet‑like start
            init = np.random.rand(n_small)

        init = init * (target_sum_small / init.sum())

        # number of iterations – we give each restart a generous budget
        max_iters = 1_500_000
        seq, score = _adam_optimize(
            a0=init,
            target_sum=target_sum_small,
            deadline=phase1_deadline,
            max_iters=max_iters,
        )
        if score < best_small_score:
            best_small_score = score
            best_small_seq = seq.copy()
            print(f"[phase1 restart {restart}] new best small score = {score:.6f}")

    if best_small_seq is None:
        # fallback to uniform if everything failed
        best_small_seq = np.full(n_small, target_sum_small / n_small)

    # ------------------------------------------------------------------
    # Phase 2 – scale to larger lengths and refine
    lengths = []

    # original length (if known) will be tried first
    if base is not None:
        orig_n = base.shape[0]
        lengths.append(orig_n)
    else:
        orig_n = n_small
        lengths.append(orig_n)

    # additional larger candidates (capped by 8000)
    for factor in (1.25, 1.5, 2.0, 3.0):
        lengths.append(min(int(orig_n * factor), 8000))

    lengths = sorted(set(lengths))

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # interpolate the best small shape onto the new grid
        old_x = np.linspace(0.0, 1.0, n_small)
        new_x = np.linspace(0.0, 1.0, n)
        init = np.interp(new_x, old_x, best_small_seq)
        init = np.maximum(init, 0.0)

        # tiny random perturbation to avoid pathological interpolation artefacts
        init += 1e-6 * np.random.randn(n)
        init = np.maximum(init, 0.0)
        init = init * (target_sum / init.sum())

        # iteration budget scaled inversely with length (more iterations for smaller n)
        remaining = deadline - time.time()
        est_iters = int(2_500_000 * n_small / n)
        # crude safety cap based on remaining time (≈0.5 ms per iteration)
        max_iters = max(200_000, min(est_iters, int(remaining / 0.0005)))

        print(f"[phase2] length={n}, max_iters≈{max_iters}")

        seq, score = _adam_optimize(
            a0=init,
            target_sum=target_sum,
            deadline=deadline,
            max_iters=max_iters,
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[phase2] new overall best = {score:.6f} at length {n}")

    # ------------------------------------------------------------------
    # Phase 3 – final hard‑max polishing
    if best_seq is not None:
        n_final = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n_final)
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining / 0.0004))
        if max_iters > 0:
            seq, score = _hardmax_refine(
                a0=best_seq,
                target_sum=target_sum,
                deadline=deadline,
                max_iters=max_iters,
            )
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[final polish] improved to {score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    # ------------------------------------------------------------------
    # Return the result as a plain list (the harness expects this)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] final score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
