def propose_candidate(seed=42, budget_s=1000.0):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_obj = np.inf
    # we will store best as numpy array

    while time.time() < deadline:
        # choose n.
        if best_seq is not None and np.random.rand() < 0.3:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3001)

        target_sum = np.sqrt(2.0 * n)
        # initialize a
        mode = np.random.rand()
        if mode < 0.5:
            # use previous best if exists
            if best_seq is not None:
                src = best_seq
                if src.size >= n:
                    a0 = src[:n].copy()
                else:
                    pad = np.random.rand(n - src.size) * 1e-3
                    a0 = np.concatenate([src, pad])
            else:
                # random smaller around U-shaped
                idx = np.arange(n, dtype=float)
                weight = 1.0/(idx+1) + 1.0/(n-idx)
                a0 = np.random.rand(n) * weight
        else:
            # Random small positive numbers
            a0 = np.random.rand(n)
        a = _project_onto_simplex(a0, target_sum)

        # Evaluate initial best.
        cur_obj = _max_conv(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init n={n} best={best_obj:.6f}")

        # Run optimizer
        # allocate time per restart: min(30 sec maybe)
        time_left = deadline - time.time()
        if time_left <= 0: break
        # run gradient descent with schedule
        a_opt, obj_opt = optimizer_adam(a, target_sum, deadline - time.time())
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Optimized n={n} best={best_obj:.6f}")

        # optionally run simulated annealing after gradient
        time_left = deadline - time.time()
        if time_left > 5.0:
            # run SA for small time
            a_sa, sa_obj = simulated_annealing(a_opt, target_sum, min(5.0, time_left), time.time())
            if sa_obj < best_obj:
                best_obj = sa_obj

Define optimizer_adam:

def optimizer_adam(a0, target_sum, time_budget):
    a = a0.copy()
    n = a.shape[0]
    L = 1 << ((3*n-2).bit_length())
    # Adam params
    beta1=0.9; beta2=0.999; eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    # schedule for tau: start tau=1.0, reduce gradually towards 0.01
    tau0 = 1.0; tau_min = 0.01
    iter=0
    best_a = a.copy()
    best_val = _max_conv(a)
    start_time = time.time()
    while time.time() < start_time + time_budget:
        # compute A fft
        A = np.fft.rfft(a, n=L)
        # compute conv full
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]   # length 2n-1
        max_conv = conv.max()
        # compute smooth weights
        # compute tau schedule: exponential decay based on progress
        # based on iteration relative to budget
        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        tau = max(tau_min, tau0 * (1 - frac))
        # softmax weights
        shifted = conv - max_conv
        # The shift ensures max weight = 1 (exp 0)
        exp_shift = np.exp(shifted / tau)
        sum_exp = exp_shift.sum()
        w = exp_shift / sum_exp
        # compute gradient correlation via FFT
        w_fft = np.fft.rfft(w, n=L)
        grad = 2 * np.fft.irfft(A * np.conj(w_fft), n=L)[:n]
        # Adam update
        iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        # learning rate scale: could be proportional to target_sum? Choose lr=0.1*target_sum/n perhaps.
        lr = 0.2 * target_sum / n  # base step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        a = a_new

        cur_val = _max_conv(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter {iter} best={best_val:.6f}")

    return best_a, best_val

"""
Hybrid optimizer for step‑function heights.

*   Projects the current vector onto the simplex Σa = √(2n), a_i ≥ 0.
*   Uses an Adam style gradient descent on a smooth approximation of
    the objective 𝑓(a)=max(conv(a,a)).  The smooth approximation is
    τ·log‑sum‑exp(conv/τ) with τ decreasing from 1 → 0.01.
    The gradient is obtained via FFT‑based correlation, which is
    O(n log n).
*   After the smooth optimisation the algorithm runs a short simulated
    annealing phase to escape possible local minima.
*   The whole procedure is restarted many times with varying lengths n
    (≈1500‑3000) and different seeds; the best candidate found before the
    time budget expires is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Max of the full convolution – the true objective (since Σa = √(2n)).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    """Returns max(conv(a,a))."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass moves.
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Simple mass‑transfer SA; returns best vector and its max‑conv."""
    a = a_init.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    T0 = max(0.01, cur_score * 0.05)
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        frac = np.random.rand() * 0.5
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)
        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Adam optimiser on a smooth approximation of max(conv).
# --------------------------------------------------------------------------- #
def adam_optimize(a_init: np.ndarray,
                  target_sum: float,
                  max_time: float,
                  start_time: float):
    """Adam‑style descent on τ·log‑sum‑exp(conv/τ)."""
    a = a_init.copy()
    n = a.shape[0]
    # FFT length sufficient for correlation (n + (2n‑1) – 1 = 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_vec = a.copy()
    best_score = _max_conv(a)

    iter_cnt = 0
    tau0 = 1.0   # start temperature for the smooth max
    tau_min = 0.01

    while time.time() < start_time + max_time:
        iter_cnt += 1

        # FFT of the current vector
        A = np.fft.rfft(a, n=L)

        # Full convolution a*a (length 2*n‑1)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        max_conv = conv.max()

        # Linear schedule for τ (smoothness parameter)
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        tau = tau_min + (tau0 - tau_min) * (1.0 - frac)

        # Soft‑max weights w_i ∝ exp((conv_i‑max)/τ)
        shifted = conv - max_conv          # ≤ 0, max at 0
        exp_shift = np.exp(shifted / tau)  # no overflow (shift ≤0)
        sum_exp = exp_shift.sum()
        w = exp_shift / sum_exp            # length 2n‑1, sums to 1

        # Gradient = 2·correlation(a, w)
        w_fft = np.fft.rfft(w, n=L)
        grad = 2.0 * np.fft.irfft(A * np.conj(w_fft), n=L)[:n]

        # Adam update (minimise, so move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # Base learning rate – scales with the mandatory sum.
        lr = 0.5 * target_sum / n
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new

        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] Adam iter {iter_cnt} new best = {best_score:.6f} (n={n})")

    return best_vec, best_score

# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,

    """

    start = time.time()

    best_score = float("inf")  # objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Helper to evaluate a candidate (fast, no extra clipping)
    # ------------------------------------------------------------------- #
    def _evaluate(a):
        return _max_conv(a)

    # ------------------------------------------------------------------- #
    # Outer restart loop
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.
        if best_seq is not None and np.random.rand() < 0.3:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3001)

        target_sum = np.sqrt(2.0 * n)   # Σ a_i = √(2n)

        # --------------------------------------------------------------- #
        # Initialise a candidate.
        # --------------------------------------------------------------- #
        init = None
        # Try to reuse the global best construction if it exists.
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:

                pad = np.random.rand(n - src.size) * 1e-3

            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init n={n} best={best_score:.6f}")

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        frac = np.random.rand() * 0.5
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)
        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Adam optimiser on a smooth approximation of max(conv).
# --------------------------------------------------------------------------- #
def adam_optimize(a_init: np.ndarray,
                  target_sum: float,
                  max_time: float,
                  start_time: float):
    """Adam‑style descent on τ·log‑sum‑exp(conv/τ)."""
    a = a_init.copy()
    n = a.shape[0]
    # FFT length for correlation (n + (2n‑1) – 1 = 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_vec = a.copy()
    best_score = _max_conv(a)

    iter_cnt = 0
    tau0 = 1.0
    tau_min = 0.01

    while time.time() < start_time + max_time:
        iter_cnt += 1

        # FFT of the current vector
        A = np.fft.rfft(a, n=L)

        # Full convolution a*a (length 2*n‑1)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        max_conv = conv.max()

        # Linear schedule for τ (smoothness parameter)
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        tau = tau_min + (tau0 - tau_min) * (1.0 - frac)

        # Soft‑max weights w_i ∝ exp((conv_i‑max)/τ)
        shifted = conv - max_conv          # ≤ 0, max at 0
        exp_shift = np.exp(shifted / tau)  # safe: exp ≤ 1
        sum_exp = exp_shift.sum()
        w = exp_shift / sum_exp            # shape (2n‑1,)

        # Gradient = 2·correlation(a, w)
        w_fft = np.fft.rfft(w, n=L)
        grad = 2.0 * np.fft.irfft(A * np.conj(w_fft), n=L)[:n]

        # Adam update (minimise → move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # Base learning rate – scales with the mandatory sum.
        lr = 0.5 * target_sum / n
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new

        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] Adam iter {iter_cnt} new best = {best_score:.6f} (n={n})")
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq = None            # numpy array of the best candidate
    best_score = float("inf")  # objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Helper for fast evaluation (no extra clipping)
    # ------------------------------------------------------------------- #
    def _evaluate(a):
        return _max_conv(a)

    # ------------------------------------------------------------------- #
    # Outer restart loop
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.
        if best_seq is not None and np.random.rand() < 0.3:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3001)

        target_sum = np.sqrt(2.0 * n)   # Σ a_i = √(2n)

        # --------------------------------------------------------------- #
        # Initialise a candidate.
        # --------------------------------------------------------------- #
        init = None
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])

        if init is None:
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
            init = np.random.rand(n) * weight

        a0 = _project_onto_simplex(init.astype(float), target_sum)

        # Quick check – update global best with the raw initial point.
        cur_score = _evaluate(a0)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init n={n} best={best_score:.6f}")

        # --------------------------------------------------------------- #
        # Allocate time for this restart.
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break

        adam_time = max(5.0, min(20.0, time_left * 0.7))
        sa_time = min(5.0, time_left - adam_time)

        # Adam optimisation (smooth max → true max)
        a_adam, score_adam = adam_optimize(a0, target_sum,
                                           adam_time,
                                           time.time())
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.2f}] After Adam n={n} best={best_score:.6f}")

        # Optional SA polish
        if sa_time > 1.0:
            a_sa, score_sa = simulated_annealing(a_adam, target_sum,
                                                 sa_time,
                                                 time.time())
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"[{time.time():.2f}] After SA   n={n} best={best_score:.6f}")

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
