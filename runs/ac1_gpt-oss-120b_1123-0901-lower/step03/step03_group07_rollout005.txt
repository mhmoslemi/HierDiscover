"""Propose candidate step function.
   Phase 1: Randomly sample a few parametric families (exponential,
   sine‑power, Gaussian, polynomial, mixed sinusoid) to obtain a good
   starting distribution.
   Phase 2: Refine that distribution with Adam on the exact objective
   (hard max of the self‑convolution) – we work in log‑space to stay
   on the probability simplex.
   Phase 3: Simple hill‑climbing local search that transfers a tiny
   amount of mass between two positions; each move is accepted only
   if it reduces the exact objective.
   The routine respects a global time budget (default 1000 s) and
   returns the best sequence found as a plain list of floats.
"""
import time
import numpy as np
import math

def _conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length ``size``."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv

def _objective_from_max(n: int, max_conv: float) -> float:
    """Given length n and max of a*a, return the value of the
       evaluation function (sum_a is 1 because we work with probability
       vectors)."""
    return 2.0 * n * max_conv

def _hard_max_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the gradient of max(conv) with respect to a (hard max)."""
    n = a.shape[0]
    k = int(np.argmax(conv))
    # gradient component g_i = a_{k-i} if 0<=k-i<n else 0
    idx = k - np.arange(n)
    grad = np.where((idx >= 0) & (idx < n), a[idx], 0.0)
    return grad, k

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation from Duchi et al. (2008)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _adam_step(u, grad_u, m, v, it, lr, beta1, beta2, eps):
    """Perform one Adam update on the log‑it vector u."""
    m = beta1 * m + (1.0 - beta1) * grad_u
    v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)

    u -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return u, m, v

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimizes the
    evaluation function.

    The routine works in three stages:
      1. Random sampling of a few low‑dimensional families (sine‑power,
         exponential, Gaussian, polynomial, mixed sinusoid) to obtain a
         decent initial vector.
      2. Adam optimisation in log‑space using the exact sub‑gradient of
         the hard max of the self‑convolution.
      3. Fast hill‑climbing that moves a tiny amount of mass between two
         positions and keeps the change only if the objective improves.

    The whole process respects the supplied time budget; the best
    sequence encountered is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.25   # small safety margin

    # length of the step function – 2000 is a good trade‑off
    n = 2000
    fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – random parametric families
    # ------------------------------------------------------------------
    families = ["sine_power", "exp_decay", "gaussian", "poly", "combo"]
    while time.time() < deadline * 0.4:
        family = np.random.choice(families)
        if family == "sine_power":
            p = np.random.uniform(0.2, 4.0)
            x = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64)) / (n + 1))
            a = np.power(x, p)
        elif family == "exp_decay":
            r = np.random.uniform(0.2, 0.98)
            a = np.power(r, np.arange(n, dtype=np.float64))
        elif family == "gaussian":
            sigma = np.random.uniform(0.1 * n, 0.5 * n)
            i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
            a = np.exp(-0.5 * (i / sigma) ** 2)
        elif family == "poly":
            p = np.random.uniform(0.2, 3.0)
            i = np.arange(n, dtype=np.float64)
            a = (i * (n - 1 - i) + 1e-12) ** p
        else:  # combo
            p = np.random.uniform(0.5, 2.5)
            alpha = np.random.uniform(-0.5, 0.5)
            sine = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64)) / (n + 1))
            cosine = np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / (n - 1))
            a = (sine ** p) * (1.0 + alpha * cosine)
            a = np.maximum(a, 0.0)

        # Normalise to a probability vector (sum = 1)
        a_sum = a.sum()
        if a_sum <= 0:
            continue
        a = a / a_sum

        conv = _conv_fft(a, fft_size)
        max_conv = conv.max()
        val = _objective_from_max(n, max_conv)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] family={family} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation using the exact hard‑max gradient
    # ------------------------------------------------------------------
    # If nothing was found (unlikely) initialise with a Dirichlet draw
    if best_seq is None:
        a0 = np.random.dirichlet(np.ones(n))
        best_seq = a0.copy()
        conv0 = _conv_fft(a0, fft_size)
        best_val = _objective_from_max(n, conv0.max())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.07

    # start from the best sequence discovered so far
    u = np.log(best_seq + 1e-30)        # log‑it vector
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    it = 0
    last_improve = time.time()
    while time.time() < deadline * 0.9 and it < 20000:
        it += 1
        # Soft‑max to get a probability vector (exact sum = 1)
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a = a / a.sum()

        # Linear self‑convolution via FFT
        conv = _conv_fft(a, fft_size)
        max_conv = conv.max()
        gradM, k_max = _hard_max_gradient(a, conv)

        # Gradient of the full objective (sum a = 1)
        grad_f = 2.0 * n * (gradM - 2.0 * max_conv * np.ones(n))

        # Convert to gradient w.r.t. log‑its (mirror Jacobian of soft‑max)
        dot = np.dot(a, grad_f)
        grad_u = a * (grad_f - dot)

        # Adam step
        lr = base_lr * (0.9997 ** it)     # gentle decay
        u, m, v = _adam_step(u, grad_u, m, v, it, lr, beta1, beta2, eps)

        # Periodic exact evaluation (no extra FFT needed – we already have max_conv)
        if it % 100 == 0:
            val = _objective_from_max(n, max_conv)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam] iter {it:05d} new best = {val:.6f}")

        # Simple shake if we have been stagnant for a while
        if time.time() - last_improve > 2.0:
            u += (np.random.rand(n) - 0.5) * 1e-3
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    # ------------------------------------------------------------------
    # Phase 3 – tiny hill‑climbing on the best candidate
    # ------------------------------------------------------------------
    a = best_seq.copy()
    # pre‑compute convolution size once more (cached)
    while time.time() < deadline:
        # pick two distinct positions
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] == 0:
            continue
        # move a small amount of mass from i to j
        delta = np.random.uniform(0.0, min(a[i], 1e-3))
        if delta == 0:
            continue
        a[i] -= delta
        a[j] += delta

        conv = _conv_fft(a, fft_size)
        max_conv = conv.max()
        val = _objective_from_max(n, max_conv)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Hill] improvement to {val:.6f}")
            last_improve = time.time()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

        # occasional larger random shake to escape plateaus
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 5e-4
            a = a + jitter
            a = np.clip(a, 0, None)
            a /= a.sum()
            last_improve = time.time()

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()

"""
Propose candidate step function.

The search proceeds in three stages:

1. **Random parametric families** – a few low‑dimensional families
   (sine‑power, exponential decay, Gaussian, polynomial, mixed sinusoid)
   are sampled extensively. Each candidate is normalised to a probability
   vector (∑a_i=1) and its exact objective value
   `F = 2·n·max_k (a*a)[k]` is computed by an FFT‑based convolution.
   The best among these cheap guesses is kept as a starting point.

2. **Adam refinement with the exact hard‑max gradient** – we work in
   log‑it space `u` (soft‑max) so the simplex constraint is satisfied
   automatically.  For a current probability vector `a` we compute the
   self‑convolution, locate the index `k*` of the maximal entry and
   obtain the sub‑gradient of `max(a*a)` (simply a shifted copy of `a`).
   Using the chain rule we obtain the gradient of the full objective
   (the denominator is constant because `∑a_i=1`).  An Adam optimiser
   updates `u`; the learning‑rate decays gently and a small “shake’’
   is injected when progress stalls.

3. **Local hill‑climbing** – with the best vector from stage 2 we perform
   quick mass‑transfer moves: a tiny amount `δ` is moved from one index
   to another, the objective is recomputed, and the move is kept only
   if it reduces the objective.  Occasional larger random perturbations
   help escape plateaus.

The whole routine respects a global time budget (default 1000 s) and
returns the best sequence found as a plain Python list of non‑negative
floats.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _objective_from_max(n: int, max_conv: float) -> float:
    """Exact objective when the vector is normalised (∑a_i = 1)."""
    return 2.0 * n * max_conv


def _hard_max_gradient(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Gradient of max(conv) w.r.t. a (hard max).  Returns the gradient vector
    and the index k* where the maximum is attained.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))
    idx = k - np.arange(n)                     # partner index for each i
    grad = np.where((idx >= 0) & (idx < n), a[idx], 0.0)
    return grad, k


def _adam_step(u: np.ndarray,
               grad_u: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               it: int,
               lr: float,
               beta1: float,
               beta2: float,
               eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam update on the log‑it vector ``u``."""
    m = beta1 * m + (1.0 - beta1) * grad_u
    v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)

    u -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return u, m, v

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The function respects the supplied time budget and returns
    the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.25   # safety margin

    # ------------------------------------------------------------------
    # Choose a length that gives a good trade‑off between expressiveness
    # and speed.  Empirically n≈2000 works well for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 2000
    fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – random exploration of a few low‑dimensional families.
    # ------------------------------------------------------------------
    families = ["sine_power", "exp_decay", "gaussian", "poly", "combo"]
    while time.time() < deadline * 0.4:
        family = np.random.choice(families)

        if family == "sine_power":
            p = np.random.uniform(0.2, 4.0)
            x = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64)) / (n + 1))
            a = np.power(x, p)

        elif family == "exp_decay":
            r = np.random.uniform(0.2, 0.98)
            a = np.power(r, np.arange(n, dtype=np.float64))

        elif family == "gaussian":
            sigma = np.random.uniform(0.1 * n, 0.5 * n)
            i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
            a = np.exp(-0.5 * (i / sigma) ** 2)

        elif family == "poly":
            p = np.random.uniform(0.2, 3.0)
            i = np.arange(n, dtype=np.float64)
            a = (i * (n - 1 - i) + 1e-12) ** p

        else:   # combo
            p = np.random.uniform(0.5, 2.5)
            alpha = np.random.uniform(-0.5, 0.5)
            sine = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64)) / (n + 1))
            cosine = np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / (n - 1))
            a = (sine ** p) * (1.0 + alpha * cosine)
            a = np.maximum(a, 0.0)

        a_sum = a.sum()
        if a_sum <= 0:
            continue
        a = a / a_sum

        conv = _conv_fft(a, fft_size)
        val = _objective_from_max(n, conv.max())

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] family={family:<10} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation with the exact (hard‑max) gradient.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform vector
        best_seq = np.full(n, 1.0 / n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.07

    # Initialise log‑it vector (soft‑max parametrisation)
    u = np.log(best_seq + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    it = 0
    last_improve = time.time()

    while time.time() < deadline * 0.9 and it < 20000:
        it += 1

        # soft‑max to obtain the probability vector (exact ℓ₁‑norm = 1)
        u_shift = u - np.max(u)          # improve numerical stability
        a = np.exp(u_shift)
        a = a / a.sum()

        # self‑convolution and its maximum
        conv = _conv_fft(a, fft_size)
        max_conv = conv.max()

        # sub‑gradient of the hard max
        gradM, _ = _hard_max_gradient(a, conv)

        # gradient of the full objective (∑a_i = 1 ⇒ denominator constant)
        grad_f = 2.0 * n * (gradM - 2.0 * max_conv * np.ones(n))

        # map to gradient w.r.t. the log‑its (soft‑max Jacobian)
        dot = np.dot(a, grad_f)
        grad_u = a * (grad_f - dot)

        # Adam update with a mild learning‑rate decay
        lr = base_lr * (0.9997 ** it)
        u, m, v = _adam_step(u, grad_u, m, v, it, lr, beta1, beta2, eps)

        # Periodic exact evaluation (no extra convolution – we already have max_conv)
        if it % 100 == 0:
            val = _objective_from_max(n, max_conv)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam] iter {it:05d} new best = {val:.6f}")

        # small random “shake’’ if we have not improved for a while
        if time.time() - last_improve > 2.0:
            u += (np.random.rand(n) - 0.5) * 1e-3
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    # ------------------------------------------------------------------
    # Phase 3 – fast hill‑climbing (tiny mass transfers) on the best vector.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] == 0.0:
            continue
        delta = np.random.uniform(0.0, min(a[i], 1e-3))
        if delta == 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        conv = _conv_fft(a, fft_size)
        val = _objective_from_max(n, conv.max())

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Hill] improvement to {val:.6f}")
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

        # occasional larger perturbation to jump out of plateaus
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 5e-4
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            last_improve = time.time()

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
