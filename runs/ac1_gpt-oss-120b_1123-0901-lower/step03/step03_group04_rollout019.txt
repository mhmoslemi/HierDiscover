"""
Propose a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
* Several problem sizes are examined (original size and a few scaled versions).
* For each size we run a two‑phase optimisation:
    1. A standard Adam optimiser on a sub‑gradient of the hard‐max 
       (as in the reference code).  This quickly reduces the largest
       autocorrelation peak.
    2. A “soft‑max” Adam optimiser that minimises a smooth approximation
       of the max‑convolution (μ·log∑exp(conv/μ)).  The smoothing parameter
       μ is slowly annealed, giving a dense gradient that escapes plateaus
       and drives the peak lower.
* After the Adam phases a cheap stochastic local‑search moves a small
  amount of mass between two entries whenever it lowers the true peak.
* The best vector found (over all lengths and random restarts) is returned.
* The implementation is fully vectorised, uses FFT‑based convolutions,
  and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)[k] using the active set."""
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
    if active.size > 0:
        grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, mu: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed max:   f(a)=μ·log∑exp(conv/μ)   with
    conv = a*a (full convolution).  Returns (grad_f, max_conv).
    """
    n = a.shape[0]
    # 1. full convolution (FFT)
    conv = _conv_full_fft(a)
    max_conv = conv.max()

    # 2. soft‑max weights (stable computation)
    shifted = (conv - max_conv) / mu            # ≤ 0
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()              # probabilities, sum to 1

    # 3. gradient = 2 * (p ⊛ a)   (cross‑correlation)
    #    implement as convolution of p with reversed a
    a_rev = a[::-1]
    N = 1 << ((len(p) + n - 1).bit_length())   # size for p ★ a_rev
    P = np.fft.rfft(p, n=N)
    A_rev = np.fft.rfft(a_rev, n=N)
    conv_grad = np.fft.irfft(P * A_rev, n=N)   # length = 3n‑2
    grad = 2.0 * conv_grad[n - 1 : 2 * n - 1]   # slice of length n
    return grad, max_conv


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Adam on the exact sub‑gradient of the hard max.
    Returns the best vector (according to the raw max‑convolution) and its M.
    """
    a = init_a.copy()
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a[:] = target_sum / n

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.04
    best_vec = a.copy()
    best_M = _conv_full_fft(a).max()
    it = 0
    jitter_interval = 4000

    while time.time() < deadline and it < max_iters:
        conv = _conv_full_fft(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        grad = _subgradient(a, active)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (1.0 + 1e-4 * it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto non‑negative simplex (simple scaling)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a[:] = target_sum / n

        # occasional jitter
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        if M < best_M:
            best_M = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {it:7d}] subgrad new best M = {best_M:.6f}")

    return best_vec, best_M


# ----------------------------------------------------------------------
def _adam_softmax_optimize(init_a: np.ndarray,
                           n: int,
                           target_sum: float,
                           deadline: float,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth approximation μ·log∑exp(conv/μ).
    μ is annealed (multiplied by 0.9995 each iteration).
    Returns the best vector w.r.t. the hard max‑convolution and its M.
    """
    a = init_a.copy()
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.05
    it = 0
    jitter_interval = 4000

    # initial μ – a modest fraction of the current peak
    _, M0 = _softmax_gradient(a, mu=1.0)   # dummy call just to get conv
    mu = max(M0 * 0.3, 1e-8)

    best_vec = a.copy()
    best_M = _conv_full_fft(a).max()

    while time.time() < deadline and it < max_iters:
        # gradient of smoothed max and current hard peak
        grad, M = _softmax_gradient(a, mu)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (1.0 + 1e-4 * it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a[:] = target_sum / n

        # anneal μ
        mu *= 0.9995
        mu = max(mu, 1e-12)

        # jitter to avoid stagnation
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0006 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        if M < best_M:
            best_M = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {it:7d}] softmax new best M = {best_M:.6f}")

    return best_vec, best_M


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic local search: move a tiny fraction of mass between
    two random positions if it reduces the true max‑convolution.
    """
    best = a.copy()
    best_M, _ = _conv_full_fft(best), None
    best_M = best_M.max()
    for step in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * 0.02
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        M = _conv_full_fft(cand).max()
        if M < best_M:
            best = cand
            best_M = M
    return best, best_M


# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n``."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, seq)

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Uses a mixture of Adam sub‑gradient, soft‑max Adam and cheap local search.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Base vector – use the provided construction if available
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        print("[info] loaded previous construction of length", base.shape[0])
    except Exception:
        base = np.full(2100, 1.0, dtype=float)
        print("[info] no previous construction – using uniform start")

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    if base.sum() > 0:
        base *= target_sum0 / base.sum()
    else:
        base[:] = target_sum0 / n0

    # ------------------------------------------------------------------
    best_seq = base.copy()
    best_M = _conv_full_fft(best_seq).max()
    print(f"[init] length={n0}, M={best_M:.6f}")

    # ------------------------------------------------------------------
    # Length schedule
    factors = [1.0, 1.25, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({max(300, min(12000, int(n0 * f))) for f in factors})
    print("[info] exploring lengths:", lengths)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length
        restarts = 3 if n <= 4000 else 2
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise -------------------------------------------------
            if n == n0 and r == 0:
                init = base.copy()
            else:
                # random Dirichlet‑like start
                rand = np.random.exponential(scale=1.0, size=n)
                init = rand * (target_sum / rand.sum())

            # blend with current global best if same size
            if best_seq.shape[0] == n:
                init = 0.2 * best_seq + 0.8 * init
                init = np.maximum(init, 0.0)
                init *= target_sum / init.sum()

            print(f"[run] length={n}, restart={r+1}/{restarts}")

            # ----- Phase 1 : sub‑gradient Adam --------------------------------
            time_left = deadline - time.time()
            phase1_deadline = time.time() + min(120.0, 0.4 * time_left)
            seq1, M1 = _adam_optimize(init, n, target_sum, phase1_deadline,
                                      max_iters=2_000_000)

            # ----- Phase 2 : soft‑max Adam (fine tuning) --------------------
            if time.time() < deadline:
                phase2_deadline = time.time() + min(200.0,
                                                   deadline - time.time())
                seq2, M2 = _adam_softmax_optimize(seq1, n, target_sum,
                                                  phase2_deadline,
                                                  max_iters=5_000_000)
            else:
                seq2, M2 = seq1, M1

            # ----- Cheap local search ----------------------------------------
            if time.time() < deadline:
                ls_seq, ls_M = _local_search(seq2, n, target_sum,
                                             steps=30_000,
                                             deadline=deadline)
                if ls_M < M2:
                    seq2, M2 = ls_seq, ls_M

            # ----- Update global best -----------------------------------------
            if M2 < best_M:
                best_M = M2
                best_seq = seq2.copy()
                print(f"[global] new best M = {best_M:.6f} (len={n})")

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best reported M = {best_M:.6f}, "
          f"official score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
