def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3
    n = 3000
    conv_fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    size_corr = 1 << int(np.ceil(np.log2(3*n)))  # for crossover

    best_val = float('inf')
    best_seq = None
    restart = 0
    while time.time() < deadline:
        restart += 1
        # initial a: Dirichlet with random concentration
        alpha = np.random.uniform(0.1, 5.0)  # if >1 uniform-ish, <1 sparse
        a = np.random.dirichlet(np.full(n, alpha))
        u = np.log(a + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # Adam hyper
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        base_lr = 0.05
        it = 0
        last_improve = time.time()
        while time.time() < deadline:
            it += 1
            # compute a from logits
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # convolution
            fa = np.fft.rfft(a, conv_fft_len)
            conv = np.fft.irfft(fa*fa, conv_fft_len)[:2*n-1]

            mu = 1.0/(2*n-1)
            diff = conv - mu
            w = 2.0 * diff  # gradient wrt conv
            # gradient wrt a
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # Convert to logits gradient
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1-beta1) * grad_u
            v = beta2 * v + (1-beta2) * (grad_u**2)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            lr = base_lr * (0.9995**it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                # evaluate objective
                val = _objective(a)    # can also call evaluate_sequence directly
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # Shake if stagnated
            if time.time() - last_improve > 1.5:
                u += (np.random.rand(n)-0.5) * 5e-4
                m.fill(0); v.fill(0)
                last_improve = time.time()

            if it >= 15000:  # more steps
                break

        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0/n)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()

"""
search for a non‑negative step function that minimises

        F(a) = 2·n·max_k (a*a)[k] / (∑a_i)²

The algorithm is deliberately different from the previous Adam‑softmax
approach:

* a is parametrised in log‑space (soft‑max) so the simplex constraint
  (non‑negative, sum = 1) holds automatically.
* The optimisation target is a *flattening* loss:
        J(a) = Σ_k ( (a*a)[k] – 1/(2n‑1) )²
  (i.e. make the self‑convolution as close to a uniform distribution as
  possible).  The gradient of J is cheap to obtain via FFT
  (gradient = 4·((a*a)−µ)★a).
* In parallel we also reduce the true max‑convolution using a smooth‑max
  surrogate (soft‑max over the convolution values).  The two gradients are
  mixed, giving a strong drive both to flatten the whole spectrum and to
  push down the peak.
* Adam with a modest learning‑rate schedule updates the logits.  Random
  “shakes’’ escape plateaus and a large number of restarts explore diverse
  initial shapes (Dirichlet with random concentration, geometric,
  half‑sine, etc.).
* The routine respects the 1000 s budget, constantly records the best
  objective seen (using the exact `evaluate_sequence` formula) and finally
  returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Weights proportional to exp(lam·conv) (numerically stable)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:            # safety – should never happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of  Σ w_k·(a*a)[k] with respect to a.
    Closed‑form:  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact evaluation used by the scoring script."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.
    The optimisation mixes a flattening loss (L2 distance of the
    self‑convolution to the uniform distribution) with a smooth‑max
    surrogate for the true objective.  Multiple Adam‑based restarts are
    performed until the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # length – a compromise between expressive power and speed
    n = 3000

    # FFT sizes reused in every iteration (powers of two are fastest)
    conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # enough for w★a

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05

    best_val = float("inf")
    best_seq = None

    restart = 0
    # mixing coefficient for the max‑surrogate gradient; tuned empirically
    gamma_max = 0.6

    while time.time() < deadline:
        restart += 1

        # --------------- initialise -------------------------------------------------
        if restart == 1 and 'height_sequence_1' in globals():
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.shape[0] != n:
                # truncate / pad to the required length
                if init.shape[0] > n:
                    init = init[:n]
                else:
                    pad = np.zeros(n, dtype=np.float64)
                    pad[:init.shape[0]] = init
                    init = pad
            a = init / init.sum()
        else:
            mode = restart % 4
            if mode == 0:
                # Dirichlet with random concentration (allows both dense and sparse)
                alpha = np.random.uniform(0.2, 5.0)
                a = np.random.dirichlet(np.full(n, alpha))
            elif mode == 1:
                # geometric decay with random ratio
                r = np.random.uniform(0.2, 0.95)
                a = np.power(r, np.arange(n, dtype=np.float64))
                a /= a.sum()
            elif mode == 2:
                # half‑sine (peaked in the middle)
                i = np.arange(1, n + 1, dtype=np.float64)
                a = np.sin(np.pi * i / (n + 1))
                a = np.maximum(a, 0.0)
                a /= a.sum()
            else:
                # “U‑shape’’ – inverse‑sqrt (large at the edges, small in the centre)
                i = np.arange(1, n + 1, dtype=np.float64)
                a = 1.0 / np.sqrt(i * (n + 1 - i))
                a /= a.sum()

        # work in log‑space (soft‑max parametrisation)
        u = np.log(a + 1e-30)          # avoid log(0)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # smoothing parameter for the max‑surrogate
        lam = 1.0
        lam_factor = 1.03               # grow a bit faster than the previous code

        it = 0
        last_improve = time.time()

        # -------------------- inner optimisation loop ----------------------------
        while time.time() < deadline:
            it += 1

            # ----- probabilities from logits -----
            u_shift = u - np.max(u)          # improve numerical stability
            a = np.exp(u_shift)
            a /= a.sum()

            # ----- convolution a*a -----
            fa = np.fft.rfft(a, conv_fft_len)
            conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

            # ----- flattening gradient (L2 loss) -----
            mu = 1.0 / (2 * n - 1)                # target uniform value for each entry
            diff = conv - mu                      # (a*a) - µ
            w_flat = 2.0 * diff                   # ∂J/∂conv = 2·diff
            grad_flat = _softmax_gradient(a, w_flat, size_corr=size_corr)

            # ----- smooth‑max gradient (focus on the current peak) -----
            w_max = _softmax_weights(conv, lam)    # soft‑max weights over conv values
            grad_max = _softmax_gradient(a, w_max, size_corr=size_corr)

            # ----- combine the two directions -----
            grad_a = grad_flat + gamma_max * grad_max

            # ----- convert to log‑space (soft‑max Jacobian) -----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = base_lr * (0.9995 ** it)

            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- increase smoothness (focus more on the true max) -----
            lam *= lam_factor

            # ----- evaluate exact objective every few iterations -----
            if it % 20 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # ----- shake if no improvement for a while -----
            if time.time() - last_improve > 1.5:
                u += (np.random.rand(n) - 0.5) * 5e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # ----- stop inner loop after a reasonable number of steps -----
            if it >= 15000:
                break

        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (guaranteed to be feasible)
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
