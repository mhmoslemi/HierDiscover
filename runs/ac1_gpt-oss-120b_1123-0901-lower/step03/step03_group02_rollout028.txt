def _lp_step(f, target_sum):
    M = _full_convolution_fft(f, f).max()
    n = f.shape[0]
    # build A matrix
    rows = 2*n-1
    A = np.zeros((rows, n))
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i range inclusive
        i = np.arange(i_start, i_end + 1)
        j = k - i
        A[k, j] = f[i]   # (i corresponds to row index)
    # constraints: A @ g <= M
    # but there is symmetric constraint also maybe? Not needed else.
    bounds = [(0, None)] * n
    c = -np.ones(n)  # maximize sum => minimize -sum
    # solve
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if not res.success:
        return None, None
    g = res.x
    sum_g = g.sum()
    if sum_g < 1e-12:
        return None, None
    scale = target_sum / sum_g
    g_scaled = g * scale
    return g_scaled, M

"""
Propose a low‑value step‑function sequence.

The evaluation function is
    eval(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

and is invariant under scaling of the sequence, therefore it is
sufficient to keep the sum of the heights on the simplex
    Σ a_i = √(2·n) .
The code below performs many short, independent restarts with
different lengths and random initial shapes.  Each restart runs a
projected (soft‑max / max‑subgradient) gradient descent:

* The current vector a is projected onto the simplex after every
  update (non‑negative, fixed sum).
* The self‑convolution c = a∗a is obtained by a real FFT (O(n log n)).
* A smooth‑max weight vector w approximates the peak of c; the
  temperature τ is exponentially decayed until it reaches a very
  small value where w becomes the exact indicator of the maximal
  entry.  This yields a true sub‑gradient of the max‑function.
* The gradient of the weighted sum Σ w_k c_k is a cross‑correlation,
  also computed via FFT.
* Momentum (β≈0.9) and an adaptive step‐size proportional to the
  current peak value keep the updates stable.
* Random “shake‑ups’’ (tiny noise added to a) are injected occasionally
  to escape plateaus.
* The best sequence seen over all restarts (measured with the provided
  evaluate_sequence) is returned.

The routine respects a global time budget (default 1000 s) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (sum = s, non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient descent for a single length (soft‑max → exact max)
# ----------------------------------------------------------------------
def _optimise_one_length_subgrad(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.08,
    min_tau_factor: float = 1e-5,
    tau_decay: float = 0.95,
    step_factor: float = 0.8,
    step_decay: float = 0.9998,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that initially smooths the max‑operator
    (soft‑max) and gradually switches to a true sub‑gradient of the
    max (τ → 0).  Returns the best sequence found and its exact
    evaluation (via evaluate_sequence).
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best candidate seen so far
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = float(2 * n * conv.max() / (target_sum ** 2))
    best_score = evaluate_sequence(best_seq.tolist())

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) convolution and current peak
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) weight vector – soft‑max with annealing temperature τ
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (tau_decay ** iteration),
                  min_tau_factor * max_val)

        if tau < 1e-9:                     # switch to exact max (sub‑gradient)
            # indices where the maximum is attained (allow tiny tolerance)
            tol = max_val * 1e-9
            max_inds = np.where(conv >= max_val - tol)[0]
            w = np.zeros_like(conv)
            if max_inds.size == 0:
                w[np.argmax(conv)] = 1.0
            else:
                w[max_inds] = 1.0 / max_inds.size
        else:
            # smooth‑max weights
            shifted = (conv - max_val) / tau
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of Σ w_k·c_k  (cross‑correlation a ⁎ w_rev)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3·n‑2
        idx = (2 * n - 2) - np.arange(n)                 # positions for Σ a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 4) momentum / adaptive step
        # --------------------------------------------------------------
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        max_abs = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_abs   # roughly move a fraction of the peak

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 5) evaluate improvement (using the cheap max‑conv estimate)
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_est = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_est < best_score:
            # verify with the official evaluator (identical rounding)
            true_score = evaluate_sequence(a_new.tolist())
            if true_score < best_score:
                best_score = true_score
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] n={n}  new best = {best_score:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.012:
            noise = np.random.randn(n) * (target_sum * 1e-4 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.85

    # final evaluation with the official function (ensures identical behaviour)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

            n = np.random.randint(600, 3000 + 1)
        else:
            n = np.random.randint(400, 2200 + 1)

        # ----------------------------------------------------------------
        # Initialise a diverse shape.
        # ----------------------------------------------------------------
        shape_kind = np.random.choice(["power", "exp", "gauss", "uniform", "rand"])
        idx = np.arange(1, n + 1, dtype=float)

        if shape_kind == "power":
            p = np.random.uniform(0.25, 0.75)
            init = idx ** (-p)
        elif shape_kind == "exp":
            tau = np.random.uniform(0.4, 2.2) * n
            init = np.exp(-idx / tau)
        elif shape_kind == "gauss":
            sigma = np.random.uniform(0.1, 0.4) * n
            mu = n / 2.0
            ii = np.arange(n, dtype=float)
            init = np.exp(-0.5 * ((ii - mu) / sigma) ** 2)
        elif shape_kind == "uniform":
            init = np.ones(n, dtype=float)
        else:   # random
            init = np.random.rand(n)

        # tiny jitter to avoid perfect symmetry
        init += np.random.randn(n) * 1e-4

        # Occasionally warm‑start from the current global best
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                init = np.concatenate([bs, pad])

        # Allocate a slice of the remaining budget to this run
        slice_time = min(8.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # --------------------------------------------------------------
        # Run the optimiser for this length
        # --------------------------------------------------------------
        seq_local, score_local = _optimise_one_length_subgrad(
            n,
            init,
            run_deadline,
            init_tau_factor=0.08,
            min_tau_factor=1e-5,
            tau_decay=0.94,
            step_factor=0.8,
            step_decay=0.9998,
            momentum_beta=0.9,
        )

        # --------------------------------------------------------------
        # Update the global best
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})")
            # stop early if we are already comfortably below the target
            if best_score < 1.49:
                break

    # ------------------------------------------------------------------
    # Fallback: uniform sequence if nothing was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain python list as required by the evaluator
    return best_seq.tolist()
```

The script follows the plan described above: many short restarts with diverse initial shapes, a projected gradient descent that anneals from a smooth‑max approximation to an exact max‑subgradient, momentum, adaptive step‑sizes and occasional random perturbations.  It keeps the best sequence found across all restarts and returns it as a Python list.  The implementation stays within the 1000 s budget and uses only NumPy (and the provided `evaluate_sequence`).
