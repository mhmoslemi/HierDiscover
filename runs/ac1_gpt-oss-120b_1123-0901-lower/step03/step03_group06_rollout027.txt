import time, math, numpy as np

def _next_pow2(x):
   return 1 << (x-1).bit_length()

def _full_convolution_fft(p, N):
   P = np.fft.rfft(p, n=N)
   conv = np.fft.irfft(P * P, n=N)
   return conv[:2*p.shape[0]-1]

def _weighted_rev_fft(p, w, N):
   n = p.shape[0]
   L = w.shape[0]   # = 2*n-1
   P = np.fft.rfft(p, n=N)
   W_rev = np.fft.rfft(w[::-1], n=N)   # reversed w
   rev_full = np.fft.irfft(P * W_rev, n=N)
   rev_full = rev_full[:n+L-1]
   rev = rev_full[L - n : L][::-1]
   return rev

def _arcsine_prob(n):
   i = np.arange(n)
   x = (i + 0.5) / n
   p = 1.0/np.sqrt(x * (1.0 - x))
   p = p / p.sum()
   return p

def _interpolate(p_old, n_new):
   old_n = p_old.shape[0]
   old_x = np.linspace(0.0, 1.0, old_n)
   new_x = np.linspace(0.0, 1.0, n_new)
   p_new = np.interp(new_x, old_x, p_old)
   p_new = np.maximum(p_new, 0.0)
   p_new /= p_new.sum()
   return p_new

def _optimize_adam(p_start, n, deadline, max_iters=200000):

    p = p_start.copy()
    eps = 1e-30
    logp = np.log(p + eps)

"""
Search for a low‑peak self‑convolution step function.

Algorithm
---------
* Initialise a probability vector p (∑p = 1) with the continuous
  arcsine density – the known optimiser of the *continuous* problem.
* Run a multiscale schedule: the current best p is interpolated to a
  larger length, mixed with a fresh arcsine‑shape and then optimised.
* Optimisation uses mirror‑descent on the simplex (exponentiated‑gradient
  style) with an Adam‑style adaptive step on the *log‑probabilities*.
  The objective is a smooth approximation of the max‑convolution:
      softmax_β(conv) = Σ_k exp(β·(conv_k−M))·conv_k,
  where M = max(conv).  Its gradient is a correlation that is obtained
  efficiently with FFTs.
* β grows from a small to a moderate value so that the optimisation
  first follows a smooth landscape and then focuses on the true peak.
* Small Gaussian jitter is added periodically to escape plateaus.
* The best sequence (scaled back to the required ℓ₂‑norm
  √(2·n)) is kept globally and returned when the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full (linear) self‑convolution of p using a single FFT of size N.
    Returns the real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Computes rev_i = Σ_k w_k · p_{k−i}   (i = 0 … n‑1)
    using a correlation realised by FFTs.
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    # FFT of p (size N already chosen ≥ n+L−1)
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weights
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep linear part
    rev = conv_rev[L - n : L][::-1]          # extract needed slice
    return rev


def _arcsine_prob(n: int) -> np.ndarray:
    """Discrete arcsine probability vector of length n (∑p = 1)."""
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n          # points in (0,1)
    p = 1.0 / np.sqrt(x * (1.0 - x))
    p /= p.sum()
    return p


def _interpolate(p_old: np.ndarray, n_new: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a new length.
    The result is re‑normalised to sum = 1.
    """
    old_n = p_old.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(new_x, old_x, p_old)
    p_new = np.maximum(p_new, 0.0)
    p_new /= p_new.sum()
    return p_new


def _optimize_adam(p_start: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent/Adam optimisation of the soft‑max approximation
    of the max‑convolution for a fixed length ``n``.
    Returns the best probability vector (∑p = 1) found and its
    objective value 2·n·max(p∗p).
    """
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Soft‑max schedule
    beta_start, beta_end, beta_len = 0.1, 12.0, 80_000

    # Jitter
    jitter_interval, jitter_scale = 2000, 0.002

    # Initialise log‑probabilities (mirror descent on the simplex)
    eps = 1e-30
    logp = np.log(p_start + eps)
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    # FFT size sufficient for both convolution (2n‑1) and correlation (3n‑2)
    N = _next_pow2(3 * n - 2)

    best_score = float("inf")
    best_p = p_start.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution (p∗p) -----
        conv = _full_convolution_fft(p_start, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        # keep the best seen during this run
        if cur_score < best_score:
            best_score = cur_score
            best_p = p_start.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- soft‑max weighting -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))        # w ∈ (0, 1], max = 1

        # ----- gradient (correlation) -----
        rev = _weighted_rev_fft(p_start, w, N)
        grad = 4.0 * n * rev                 # ∇_p (2·n·softmax)

        # ----- Adam update on log‑probabilities -----
        g = -grad                            # descent direction
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = (0.5 / (np.sqrt(v_hat) + eps_adam)) * m_hat
        logp += step

        # ----- occasional jitter -----
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex (exponentiate) -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)       # avoid overflow
        p_start = exp_shift / exp_shift.sum()

    return best_p, best_score
