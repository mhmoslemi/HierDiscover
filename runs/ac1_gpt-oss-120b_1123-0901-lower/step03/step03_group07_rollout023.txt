def full_convolution(a):
    n = a.shape[0]
    size = 1 << int(np.ceil(np.log2(2*n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n - 1]
    return conv

def objective(a):
    n = a.shape[0]
    conv = full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def softmax_weights(conv, lam):
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= w_sum
    return w

def softmax_gradient(a, w):
    # compute w ★ a (cross-correlation), using FFT method as before.
    n = a.shape[0]
    rev_a = a[::-1]
    # size: at least len(w) + n -1
    size = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size)
    frev = np.fft.rfft(rev_a, size)
    corr = np.fft.irfft(fw * frev, size)
    grad = 2.0 * corr[n - 1 : n - 1 + n]  # slice length n
    return grad

def maxconv_and_index(a):
    conv = full_convolution(a)
    max_val = conv.max()
    idx = int(conv.argmax())
    return max_val, idx, conv

def gradient_max_onehot(a, idx):
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, idx - (n-1))
    hi = min(n-1, idx)
    # vectorized
    j = np.arange(lo, hi+1, dtype=int)
    grad[j] = 2.0 * a[idx - j]
    return grad

def adam_optimize(a0, deadline, max_iters=8000, base_lr=0.05):
    # using smoothmax surrogate
    n = a0.shape[0]
    size_fft = 1 << int(np.ceil(np.log2(2*n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3*n)))  # large enough
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # init logits u
    u = np.log(a0 + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    lam = 1.0
    lam_factor = 1.02
    best_val = objective(a0)
    best_seq = a0.copy()
    it = 0
    last_improve = time.time()
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute a
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()
        # conv and softmax weights
        conv = full_convolution(a, size=size_fft)
        w = softmax_weights(conv, lam)
        # gradient w.r.t a
        grad_a = softmax_gradient(a, w)
        # gradient w.r.t u
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr * (0.9995 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # increase lam
        lam *= lam_factor
        # evaluate periodic
        if it % 20 == 0:
            val = objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"adam iter {it:05d} new best = {val:.6f}")
        # shake
        if time.time() - last_improve > 1.5:
            u += (np.random.rand(n) - 0.5) * 5e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
    return best_seq, best_val

def subgradient_optimize(a0, deadline, max_iters=3000, base_lr=0.05):
    # uses exact max (one-hot)
    n = a0.shape[0]
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    u = np.log(a0 + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    best_val = objective(a0)
    best_seq = a0.copy()
    it = 0
    last_improve = time.time()
    while time.time() < deadline and it < max_iters:
        it += 1
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()
        # compute max conv and index
        conv = full_convolution(a)
        max_val = conv.max()
        idx = int(conv.argmax())
        # gradient
        grad_a = gradient_max_onehot(a, idx)
        # convert to log-space gradient
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u**2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr * (0.9995 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate occasional
        if it % 10 == 0:
            val = objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"subgrad iter {it:05d} new best = {val:.6f}")
        # shaking when stagnating
        if time.time() - last_improve > 1.0:
            u += (np.random.rand(n) - 0.5) * 1e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
    return best_seq, best_val

def hill_climb(a0, deadline, max_moves=5000):
    a = a0.copy()
    best_val = objective(a)
    best_seq = a.copy()
    n = a.shape[0]
    # temperature schedule
    temp_start = 0.02
    temp_end = 0.0001
    moves = 0
    while time.time() < deadline and moves < max_moves:
        moves += 1
        # linear temperature decay
        t_frac = moves / max_moves
        temp = temp_start * (1 - t_frac) + temp_end * t_frac
        # propose a move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # choose delta fraction of a[i]
        delta = min(a[i], np.random.rand() * 0.05)  # up to 5% of source mass
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute objective
        val_new = objective(a_new)
        # accept
        if val_new < best_val or np.random.rand() < np.exp((best_val - val_new) / temp):
            a = a_new
            if val_new < best_val:
                best_val = val_new
                best_seq = a.copy()
                print(f"Hill move {moves:05d} new best = {val_new:.6f}")
    return best_seq, best_val

"""
Propose a step‑function (non‑negative heights) that minimizes

    F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)² .

The algorithm is a hybrid of three ideas:

1. **Linear‑programming “flattening’’** – starting from a strongly
   asymmetric vector (a ramp) we solve, for the current f,
   the LP  max ∑ g_j  s.t.  f*g ≤ M (M = max (f*f)).  The solution g
   spreads the mass as much as possible without increasing the
   current worst autocorrelation.  We mix f←(1‑t)f+t g and repeat
   a few times.  This quickly lowers the peak of the convolution.

2. **Adam optimisation on a smooth‑max surrogate** – the max in F is
   approximated by a soft‑max, giving a smooth objective whose gradient
   can be computed by FFT‑based correlations.  Several Adam steps
   polish the LP‑output.

3. **Exact sub‑gradient + hill‑climbing** – finally we switch to the
   true (non‑smoothed) max, perform a few Adam steps with the exact
   sub‑gradient (the active convolution index becomes a one‑hot
   vector), and finish with a cheap stochastic hill‑climb that moves a
   tiny amount of probability between two indices.

All phases respect the time budget (default 1000 s) and keep the
sequence on the probability simplex (non‑negative, sum = 1).  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Utility: fast linear convolution via FFT (full length 2n‑1)
def full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact evaluation (the same formula used by ``evaluate_sequence``)
def objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Soft‑max (smooth‑max) utilities used by the Adam phase
def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """w_k = exp(lam·(conv_k‑max(conv))) (normalised)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  φ(a)=∑_k w_k·(a*a)[k]   w.r.t. a.
    Formula: ∇φ = 2·(w ★ a)   (cross‑correlation).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]          # slice length n
    return grad


# ----------------------------------------------------------------------
# Exact max‑convolution gradient (sub‑gradient, one‑hot weight)
def _grad_max_onehot(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the active index ``idx``.
    The i‑th component is 2·a_{idx‑i} if the index lies in [0,n) else 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, idx - (n - 1))
    hi = min(n - 1, idx)
    j = np.arange(lo, hi + 1, dtype=int)
    grad[j] = 2.0 * a[idx - j]
    return grad

# Adam optimiser on the smooth‑max surrogate
def _adam_optimize(start_a: np.ndarray, deadline: float,
                   max_iters: int = 8000,
                   base_lr: float = 0.05) -> tuple[np.ndarray, float]:
    n = start_a.shape[0]
    # FFT sizes (pre‑computed for speed)
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # >= len(w)+n‑1

    # Adam state
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # logits (soft‑max parametrisation)
    u = np.log(start_a + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    lam = 1.0
    lam_factor = 1.02
    best_val = objective(start_a)
    best_seq = start_a.copy()
    it = 0
    last_improve = time.time()
    while time.time() < deadline and it < max_iters:
        it += 1
        # probabilities from logits
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # smooth‑max weights
        conv = full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)

        # gradient w.r.t. a
        grad_a = _softmax_gradient(a, w)

        # convert to gradient w.r.t. logits (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr * (0.9995 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase smoothness
        lam *= lam_factor

        # occasional exact evaluation
        if it % 20 == 0:
            val = objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam] iter {it:05d}  new best = {val:.6f}")

        # shake when stuck
        if time.time() - last_improve > 1.5:
            u += (np.random.rand(n) - 0.5) * 5e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    return best_seq, best_val

# Very cheap stochastic hill‑climbing starting from a good candidate
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                max_moves: int = 20000) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    best_val = objective(a)
    best_seq = a.copy()
    n = a.shape[0]

    # linear temperature schedule (helps to escape tiny plateaus)
    T0, T1 = 0.02, 1e-4

    for move in range(1, max_moves + 1):
        if time.time() >= deadline:
            break
        frac = move / max_moves
        temp = T0 * (1 - frac) + T1 * frac

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a small amount of probability from i to j
        delta = min(a[i], np.random.rand() * 0.05)   # ≤5 % of source mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val_new = objective(a_new)
        # Metropolis acceptance
        if val_new < best_val or np.random.rand() < np.exp((best_val - val_new) / temp):
            a = a_new
            if val_new < best_val:
                best_val = val_new
                best_seq = a.copy()
                print(f"[HillClimb] move {move:05d} new best = {val_new:.6f}")

    return best_seq, best_val

    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  LP flattening phase
    #    Reserve a few hundred seconds for the later refinements.
    lp_deadline = deadline - 200.0
    t_mix = 0.4
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < 20:
        lp_iter += 1
        # current worst autocorrelation
        conv = full_convolution(best_seq)
        M = conv.max()

        # Toeplitz matrix representing (best_seq * g)
        col = np.concatenate([best_seq, np.zeros(n - 1)])
        row = np.concatenate([best_seq[:1], np.zeros(n - 1)])
        A = toeplitz(col, row)                 # shape (2n‑1, n)

        # Linear programme: maximise 1ᵀg   s.t.   A·g ≤ M,   g ≥ 0
        c = -np.ones(n)                        # minimise –∑g
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP {lp_iter}] linprog failed – {res.message}")
            break

        g0 = res.x
        if g0.sum() == 0.0:
            print(f"[LP {lp_iter}] zero solution – stop")
            break

        # Rescale to keep the same ℓ₁‑norm (here equal to 1)
        g = g0 / g0.sum()

        # Convex combination and re‑normalise
        best_seq = (1.0 - t_mix) * best_seq + t_mix * g
        best_seq = np.maximum(best_seq, 0.0)
        best_seq /= best_seq.sum()

        val = objective(best_seq)
        if val < best_val:
            best_val = val
            best_seq_best = best_seq.copy()
            print(f"[LP {lp_iter}] improved objective = {val:.6f}")

        # slowly decrease the mixing weight
        t_mix = max(0.05, t_mix * 0.9)

    # ------------------------------------------------------------------
    # 4️⃣  Adam optimisation on the smooth‑max surrogate (starting from LP result)
    phase2_deadline = deadline - 100.0
    seq_adam, val_adam = _adam_optimize(best_seq, phase2_deadline, max_iters=8000)
    if val_adam < best_val:
        best_val = val_adam
        best_seq = seq_adam.copy()
        print(f"[Adam] improved objective = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exact sub‑gradient (Adam) on the true max‑convolution
    phase3_deadline = deadline - 30.0
    if time.time() < phase3_deadline:
        seq_sub, val_sub = _subgradient_optimize(best_seq,
                                                 phase3_deadline,
                                                 max_iters=5000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"[SubGrad] improved objective = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final cheap stochastic hill‑climb
    if time.time() < deadline:
        seq_hc, val_hc = _hill_climb(best_seq, deadline, max_moves=20000)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = seq_hc.copy()
            print(f"[HillClimb] improved objective = {val_hc:.6f}")

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # Return a plain Python list so the external evaluator can consume it.
    return best_seq.tolist()
