  Worse moves are occasionally accepted according to a temperature schedule,

def _conv_max_fft(a: np.ndarray):
    """Return maximal convolution value and the full convolution array."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv

def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """Sub‑gradient of max‑convolution at a (average over active shifts)."""
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad

    # 1) Adam optimisation – spend ~75 % of the budget
    adam_deadline = deadline - (budget_s * 0.25)
    if adam_deadline <= start_time:
        adam_deadline = deadline

    print("[stage] Adam optimisation (time left ≈",
          f"{max(0.0, adam_deadline - time.time()):.1f}s)")
    best_vec, best_val = _adam_optimize(base, n, target_sum,
                                        adam_deadline, max_iters=10_000_000)

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing refinement – use the remaining time
    if time.time() < deadline:
        print("[stage] Simulated‑annealing refinement")
        best_vec, best_val = _simulated_annealing(best_vec, n,
                                                  target_sum, deadline)
    else:
        print("[stage] no time left for annealing")

    # ------------------------------------------------------------------
    # Final sanity check through the official evaluator
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"[done] best reported M = {best_val:.6f}, "
          f"official score = {final_score:.6f}, length = {n}")

    return best_vec.tolist()
