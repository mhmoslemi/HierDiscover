"""
Propose a step‐function that (hopefully) beats the previous 1.5195 bound.

Algorithm
---------
1.  Choose a moderate length (≈3500).  The objective is scale‑invariant, so we
    always normalise the sequence to have sum = 1.
2.  Perform a cheap global search over a few simple parametric families
    (exponential decay, symmetric exponential, power‑sine, Beta‑shaped,
    triangular).  For each generated candidate we evaluate the exact
    objective using an FFT‑based convolution (O(n log n)).  The best
    candidate found is kept.
3.  Starting from that best candidate we run a short Adam optimisation on the
    smooth‑max surrogate (the same surrogate that was used in the reference
    code).  The optimisation runs only for the remaining time budget and
    occasionally adds tiny random “shakes’’ to escape plateaus.
4.  The best sequence discovered during the whole run is returned as a plain
    Python list.  The code respects the 1000 s wall‑time limit and never
    returns an empty or invalid list.
"""
import time
import math
import random
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution (linear, full) via FFT – reused many times.
def _full_convolution(a: np.ndarray, fft_size: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length) using an FFT of given size."""
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[: 2 * a.shape[0] - 1]
    return conv


# Helper: exact objective F(a) = 2·n·max(conv)/(sum(a))²
def _objective(a: np.ndarray, n: int, fft_size: int) -> float:
    """
    Compute the evaluation function exactly (the same as evaluate_sequence)
    for a non‑negative vector a.
    """
    s = a.sum()
    if s < 1e-12:               # avoid division by zero – will be rejected anyway
        return float("inf")
    conv = _full_convolution(a, fft_size)
    max_c = conv.max()
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Families used in the cheap global search.
def _make_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i (decay from the left)."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _make_symmetric_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**|i‑center|, symmetric peak in the middle."""
    idx = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.power(r, np.abs(idx))
    a /= a.sum()
    return a


def _make_power_sine(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))**p, 0 < i ≤ n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑pdf shape on {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # Guard against α or β < 1 (which would blow up at the ends) by adding a tiny epsilon.
    eps = 1e-12
    a = np.power(np.maximum(x, eps), alpha - 1.0) * np.power(
        np.maximum(1.0 - x, eps), beta - 1.0
    )
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # fall back to uniform if the parameters were pathological
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _make_triangular(n: int, frac: float) -> np.ndarray:
    """
    A simple symmetric triangle.
    frac = fraction of n that is occupied by the rising side (including the peak).
    """
    m = max(1, int(frac * n))
    # rising part 1…m, descending part m‑1…1
    rising = np.arange(1, m + 1, dtype=np.float64)
    descending = np.arange(m - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([rising, descending])
    if a.shape[0] < n:
        a = np.concatenate([a, np.zeros(n - a.shape[0], dtype=np.float64)])
    else:
        a = a[:n]
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low value
    of evaluate_sequence().  The routine mixes a fast parametric global search
    with a short Adam optimisation on a smooth‑max surrogate.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Length – a compromise between expressive power and speed.
    n = 3500
    # Size for the FFT that will be reused throughout the run.
    fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))

    deadline = time.time() + budget_s - 0.3   # small safety margin

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 0️⃣  If the test harness already supplied a previous construction,
    #     use it as a first incumbent.
    if "height_sequence_1" in globals():
        cand = np.array(height_sequence_1, dtype=np.float64)
        if cand.ndim == 1 and cand.size == n:
            val = _objective(cand, n, fft_size)
            best_val, best_seq = val, cand.copy()
            print(f"[Init] using supplied height_sequence_1, value {val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Fast global search over a handful of parametric families.
    #     Roughly half of the time budget is spent here.
    print("[Search] starting coarse parametric search")
    while time.time() < deadline:
        # pick a family uniformly at random
        family = np.random.randint(5)

        if family == 0:          # exponential decay
            r = np.random.uniform(0.30, 0.99)
            cand = _make_exponential(n, r)

        elif family == 1:        # symmetric exponential
            r = np.random.uniform(0.55, 0.99)
            cand = _make_symmetric_exponential(n, r)

        elif family == 2:        # power‑sine
            p = np.random.uniform(0.2, 2.0)
            cand = _make_power_sine(n, p)

        elif family == 3:        # Beta pdf
            alpha = np.random.uniform(0.1, 5.0)
            beta = np.random.uniform(0.1, 5.0)
            cand = _make_beta(n, alpha, beta)

        else:                    # triangular
            frac = np.random.uniform(0.25, 0.75)
            cand = _make_triangular(n, frac)

        # Occasionally try a random Dirichlet (adds diversity)
        if np.random.rand() < 0.025:
            conc = np.random.uniform(0.2, 2.0)
            cand = np.random.dirichlet(np.full(n, conc))

        # Evaluate (exact objective)
        val = _objective(cand, n, fft_size)

        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Search] new best (family {family}) = {val:.6f}")

        # A very lightweight termination check – keep a few seconds margin.
        if time.time() > deadline - 4.0:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Local refinement with Adam on the smooth‑max surrogate.
    #     We reuse the well‑tested code from the reference solution but
    #     initialise it with the best candidate found so far.
    if best_seq is None:
        # Fallback – uniform vector (should never happen)
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq, n, fft_size)

    print("[Refine] starting Adam‑based refinement")
    # Soft‑max (log‑space) parametrisation – guarantees non‑negativity & sum = 1.
    u = np.log(best_seq + 1e-30)          # avoid log(0)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05

    # λ‑continuation (smooth‑max temperature)
    lam = 1.0
    lam_factor = 1.025   # a bit more aggressive than the reference

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1

        # -------- obtain probabilities from logits --------
        u_shift = u - np.max(u)               # numeric stabilisation
        a = np.exp(u_shift)
        a /= a.sum()

        # -------- convolution and smooth‑max weights --------
        conv = _full_convolution(a, fft_size)
        # soft‑max over the convolution values
        conv_max = conv.max()
        w = np.exp(lam * (conv - conv_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.shape[0])
        else:
            w /= w_sum

        # -------- gradient of the smooth‑max surrogate --------
        # grad = 2·(w ★ a)   (★ = correlation)
        rev_a = a[::-1]
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
        fw = np.fft.rfft(w, size_corr)
        frev = np.fft.rfft(rev_a, size_corr)
        corr = np.fft.irfft(fw * frev, size_corr)
        grad_a = 2.0 * corr[n - 1 : n - 1 + n]   # slice length n

        # -------- convert to log‑space gradient --------
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # -------- Adam update --------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr * (0.9995 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # -------- increase λ (sharpen soft‑max) --------
        lam *= lam_factor

        # -------- occasional exact evaluation --------
        if it % 15 == 0:
            val = _objective(a, n, fft_size)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                last_improve = time.time()
                print(f"[Refine] iter {it:06d} new best = {val:.6f}")

        # -------- shake if we are stuck for >1 s --------
        if time.time() - last_improve > 1.0:
            u += (np.random.rand(n) - 0.5) * 5e-4   # tiny perturbation
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

        # -------- safety stop (prevent runaway) --------
        if it >= 25000:       # plenty of updates for the remaining budget
            break

    # ------------------------------------------------------------------
    # Final sanity check – ensure we return a plain Python list.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
    else:
        # make sure the vector complies with the evaluation function's clipping:
        best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[Result] final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
