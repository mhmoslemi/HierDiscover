def solve_lp_best_g(f, M, target_sum):
    n = len(f)
    import cvxpy as cp
    from scipy.linalg import toeplitz

    # Build conv matrix
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([[f[0]], np.zeros(n-1)])
    A = toeplitz(col, row)[:2*n-1, :]  # shape (2n-1, n)
    # define variable
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]

    # maximize sum(g)
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    # solve
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)

    if g.value is None:
        return None, None
    g_opt = np.array(g.value).flatten()
    # After solving, may have small violation due to tolerance, we can enforce nonnegativity.
    g_opt = np.maximum(g_opt, 0.0)
    # Scale to target sum if needed
    sum_g = g_opt.sum()
    if sum_g > 0:
        g_scaled = g_opt * (target_sum / sum_g)
    else:
        g_scaled = np.full_like(g_opt, target_sum / n)
    return g_scaled, prob.value

"""
Hybrid optimizer for the step‑function problem.

The goal is to minimise  max(conv(a,a))  (the objective returned by
``evaluate_sequence`` after the mandatory L1‑normalisation).  The programme
mixes three techniques:

1. **Linear‑programming step** – following the scheme described in the
   statement: for the current vector f compute M = max(f*f) and solve

        maximise   sum(g)
        subject to g ≥ 0 ,  f * g ≤ M   (convolution inequality)

   The solution g is then rescaled to the required L1‑norm and mixed into f.
   Each LP reduces M (or leaves it unchanged) and quickly drives the
   sequence away from the uniform local minimum.

2. **Projected Adam optimiser** – after a handful of LP updates a short
   Adam run on the exact sub‑gradient of the max‑convolution refines the
   shape.  The Adam routine keeps the vector non‑negative and on the L1‑
   sphere (sum = √(2 n)).

3. **Cheap stochastic local search** – a final greedy mass‑transfer
   neighbourhood search removes any remaining small peaks.

The whole search stops after the allotted time (default 1000 s) and the
best vector found is returned as a plain list of floats.
"""

import time
import math
import random
import numpy as np
import cvxpy as cp
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Compute the maximal value of the full convolution a*a using an FFT.
    Returns (max_val, full_convolution_array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution at point a.
    The gradient for each component i is the average of a_{k‑i} over all
    active shifts k (those attaining the current maximum).
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx                     # partner indices
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 500_000):
    """
    Minimise the peak of a*a using Adam on the exact sub‑gradient.
    Returns the best vector seen and its peak value.
    """
    a = init_a.copy()
    # normalise
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.04
    best_vec = a.copy()
    best_val = _conv_max_fft(a)[0]

    it = 0
    jitter_every = 4000
    while time.time() < deadline and it < max_iters:
        M, conv = _conv_max_fft(a)

        # active set (tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max term
        rev = _subgradient(a, active)

        # gradient of the whole objective (scaling factor 2*n/(sum^2) = 1)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = base_lr / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set (non‑negative, correct L1‑norm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a[:] = target_sum / n

        # occasional jitter to escape plateaus
        if it % jitter_every == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # keep best
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[adam] iter {it:7d}  new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Simple greedy hill‑climbing: repeatedly transfer a tiny fraction of mass
    from one index to another if the peak of the convolution improves.
    """
    n = a.shape[0]
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break
        i, j = random.randint(0, n - 1), random.randint(0, n - 1)
        if i == j:
            continue
        delta = best[i] * 0.02
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep L1‑norm
        cand = np.maximum(cand, 0.0)
        cand *= target_sum / cand.sum()
        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M
    return best, best_val


# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve the linear programme

        maximise   sum(g)
        subject to g ≥ 0 ,  f * g ≤ M

    Returns the optimal g (already on the L1‑sphere) or None on failure.
    """
    n = f.shape[0]
    # Build the convolution matrix T such that T @ g = f * g
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([[f[0]], np.zeros(n - 1)])
    T = toeplitz(col, row)                     # shape (2n‑1, n)

    g = cp.Variable(n)
    constraints = [g >= 0, T @ g <= M + 1e-9]   # tiny slack for numerical safety
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-7, reltol=1e-7, feastol=1e-7)
    except Exception as e:
        print("[lp] solve exception:", e)
        return None

    if g.value is None:
        print("[lp] solver returned no solution")
        return None
    return np.maximum(np.array(g.value).flatten(), 0.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine mixes a LP‑based improvement, a short
    Adam optimiser and a final greedy local search, always respecting the
    required L1‑normalisation (sum = √(2 n)).  It returns the best vector found
    before the time budget expires.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0               # 5 s safety margin

    # ------------------------------------------------------------------
    # initialise with the previously known construction if present
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        print("[info] loaded previous sequence, length =", base.shape[0])
    except Exception:
        # fallback: uniform vector of moderate length
        base = np.full(2100, 1.0, dtype=float)
        print("[info] no previous construction – using uniform start")

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # normalise to the required L1‑norm
    s = base.sum()
    if s > 0:
        base = base * (target_sum / s)
    else:
        base = np.full_like(base, target_sum / n)

    best_seq = base.copy()
    best_M, _ = _conv_max_fft(best_seq)
    print(f"[init] length = {n},  initial max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # 1) Linear‑programming phase
    t = 0.25                     # mixing factor (how much of g we take each round)
    lp_max_iters = 30
    lp_iter = 0

    while time.time() < deadline and lp_iter < lp_max_iters:
        M = _conv_max_fft(best_seq)[0]

        # solve LP for the current f = best_seq
        g_raw = _solve_lp_for_g(best_seq, M)
        if g_raw is None:
            print("[lp] aborting LP phase")
            break

        sum_g = g_raw.sum()
        if sum_g <= 0:
            print("[lp] got zero‑mass g, break")
            break

        # scale g to the required L1‑norm
        g = g_raw * (target_sum / sum_g)

        # convex combination
        new_seq = (1 - t) * best_seq + t * g

        # tiny numeric correction to stay exactly on the L1‑sphere
        new_seq = np.maximum(new_seq, 0.0)
        new_seq *= target_sum / new_seq.sum()

        new_M = _conv_max_fft(new_seq)[0]
        print(f"[lp] iter {lp_iter+1:2d}  M = {new_M:.6f}")

        # accept if we improve (or at least do not worsen)
        if new_M < best_M - 1e-7:
            best_seq, best_M = new_seq, new_M
            # slowly shrink step size to fine‑tune later
            t = max(0.05, t * 0.9)
        else:
            # no improvement – damp the step size more aggressively
            t = max(0.02, t * 0.7)

        lp_iter += 1

    # ------------------------------------------------------------------
    # 2) Short Adam optimisation starting from the best LP result
    if time.time() < deadline:
        time_left = deadline - time.time()
        adam_deadline = time.time() + min(200.0, time_left * 0.5)
        print("[adam] starting Adam optimisation")
        opt_seq, opt_val = _adam_optimize(best_seq, n, target_sum,
                                          adam_deadline, max_iters=5_000_000)
        if opt_val < best_M:
            best_seq, best_M = opt_seq, opt_val
            print(f"[adam] improved to M = {best_M:.6f}")

    # ------------------------------------------------------------------
    # 3) Final cheap local search (greedy mass transfers)
    if time.time() < deadline:
        steps = int(min(200_000, (deadline - time.time()) * 3000))
        print(f"[local] performing {steps} greedy steps")
        best_seq, best_M = _local_search(best_seq, target_sum,
                                         steps=steps, deadline=deadline)

    # ------------------------------------------------------------------
    # final sanity check via the official evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best M = {best_M:.6f}, official score = {final_score:.6f}, "
          f"length = {len(best_seq)}")
    return best_seq.tolist()
