def project_simplex(v, z):
    """Projects vector v onto simplex sum = z, v>=0."""
    if v.sum() == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1 # index last true
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

```python
def project_simplex(v, target):
    """Project vector v onto the simplex {x >= 0, sum(x) = target}."""
    # Implements the algorithm from Duchi et al. (2008)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

"""
Propose a step‑function minimising the evaluate_sequence objective.
Algorithm
---------
* Use a set of candidate lengths (original, 1.5×, 2×, 3×, up‑to a
  configurable cap). For each length we run several independent
  Adam‑style optimisations of a *smooth* max‐convolution.
* The smooth max is the log‑sum‑exp of the convolution, with a temperature
  that is annealed exponentially.
* After each gradient step the iterate is projected onto the simplex
  {a≥0, Σa = √(2 n)} using the exact Euclidean projection (Duchi et al.).
* A tiny Gaussian jitter is injected every few thousand iterations and a
  fresh random restart is performed when time permits.
* The best sequence (according to the exact objective) across all lengths
  and restarts is returned.
"""
import time
import math
import numpy as np

def _softmax_weights(conv: np.ndarray, tau: float):
    """Return soft‑max weights w (sum = 1) and the smoothed max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau

def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """Compute rev_i = Σ_k w_k a_{k‑i} using FFTs."""
    # convolution of a with reversed w
    A = np.fft.rfft(a, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(A * W_rev, n=Nfft)
    # keep valid part
    conv_rev = conv_rev[:len(w) + len(a) - 1]
    # we need rev_i for i = 0…len(a)-1
    start = len(a) - 1
    rev = conv_rev[start:start + len(a)]
    return rev

def _project_simplex(v: np.ndarray, target_sum: float):
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    # Duchi et al. 2008
    n = v.shape[0]
    # if already feasible, return scaled version (avoid division by zero)
    if v.min() >= 0 and abs(v.sum() - target_sum) < 1e-12:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_optimize(
    n: int,
    a0: np.ndarray,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
):
    """Run Adam on the smoothed max‑convolution for a fixed length."""
    target_sum = math.sqrt(2.0 * n)

    # ---- initialise -------------------------------------------------
    a = a0.copy()
    a = _project_simplex(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 4000

    # temperature schedule
    tau_factor = 0.20          # τ₀ = τ_factor·M₀
    decay_rate = 25000.0      # controls exponential decay

    Nfft = 1 << ((2 * n - 1).bit_length())

    # ---- main loop -------------------------------------------------
    while time.time() < deadline and iteration < max_iters:
        # 1) convolution a*a
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # 2) soft‑max smoothing
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # 3) weighted reverse convolution = gradient of Lτ (up to constant 2)
        rev = _weighted_rev(a, w, Nfft)

        # 4) Adam update (the constant factor is absorbed in the learning‑rate)
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * rev
        v = beta2 * v + (1.0 - beta2) * (rev * rev)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) project onto the simplex Σa = √(2 n), a≥0
        a = _project_simplex(a, target_sum)

        # 6) occasional jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = _project_simplex(a + jitter, target_sum)

        # 7) exact objective (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score

    sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # -----------------------------------------------------------------
    # Load a known construction if present, otherwise start from uniform.
    try:

    for n in lengths:

        # Build an initial shape for the new length.
        if n == n0:

        else:

            old_x = np.linspace(0.0, 1.0, n0)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best score = {final_score:.6f}, length = {len(best_seq)}")

"""
Propose a step‑function minimising the ``evaluate_sequence`` objective.

Algorithm
---------
* A handful of sequence lengths (original and several enlargements) are
  examined.  For each length we run a few independent Adam‑type
  optimisations that minimise a *smooth* approximation of the maximal
  self‑convolution.
* The smooth max is the log‑sum‑exp of the convolution; its temperature
  is annealed exponentially (τ₀ = 0.2·M₀).  The gradient of this
  smoothed objective is obtained by a weighted reverse convolution,
  computed efficiently with a single FFT.
* After every gradient step the iterate is projected **exactly** onto the
  feasible simplex `{a ≥ 0, Σa = √(2 n)}` using the Duchi et al. Euclidean
  projection.  This replaces the simple rescaling used previously and
  yields more accurate updates.
* A tiny Gaussian jitter is added every few thousand iterations and a
  fresh random restart is performed when time permits.
* The best sequence (according to the exact ``evaluate_sequence``
  value) across all lengths and restarts is returned.
"""

import time
import math
import numpy as np


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.

    Returns
    -------
    w : np.ndarray
        Weights that sum to 1.
    Ltau : float
        The smoothed maximum τ·log∑exp(conv/τ).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """Compute rev_i = Σ_k w_k·a_{k‑i} for i = 0 … n‑1 using FFTs."""
    # convolution of a with the reversed weight vector
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(A_fft * W_rev_fft, n=Nfft)
    # keep the valid part (length = n + len(w) – 1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    L = len(w)                         # = 2·n‑1
    rev = conv_rev[L - 1 - np.arange(a.shape[0])]
    return rev


def _project_simplex(v: np.ndarray, target_sum: float):
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum} (Duchi et al.)."""
    n = v.shape[0]
    # quick exit if already feasible
    if v.min() >= 0 and abs(v.sum() - target_sum) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _adam_optimize(
    n: int,
    a0: np.ndarray,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
):
    """
    Adam optimisation of the smoothed max‑convolution for a fixed length.

    Parameters
    ----------
    n : int
        Length of the vector.
    a0 : np.ndarray
        Starting point (already feasible).
    deadline : float
        Wall‑clock time at which the optimisation must stop.
    max_iters : int
        Upper bound on the number of iterations.
    best_seq, best_score : current global best (updated in‑place).

    Returns
    -------
    best_seq, best_score : the updated global best after this run.
    """
    target_sum = math.sqrt(2.0 * n)

    a = _project_simplex(a0.copy(), target_sum)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    it = 0
    jitter_interval = 4000

    # temperature schedule
    tau_factor = 0.20
    decay_rate = 25000.0

    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and it < max_iters:
        # ---- (1) exact self‑convolution ---------------------------------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- (2) soft‑max smoothing ------------------------------------
        tau = max(1e-12, tau_factor * M * math.exp(-it / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # ---- (3) gradient of the smooth max (up to a constant factor) --
        rev = _weighted_rev(a, w, Nfft)

        # ---- (4) Adam update -------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * rev
        v = beta2 * v + (1.0 - beta2) * (rev * rev)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- (5) projection onto the simplex ----------------------------
        a = _project_simplex(a, target_sum)

        # ---- (6) occasional jitter --------------------------------------
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = _project_simplex(a + jitter, target_sum)

        # ---- (7) exact objective (after normalisation it equals M) -----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score
