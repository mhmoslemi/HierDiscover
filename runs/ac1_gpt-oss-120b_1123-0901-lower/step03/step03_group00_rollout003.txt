"""
Propose a step‑function (non‑negative vector) that approximately minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a two‑phase Adam optimisation on the probability simplex.
The method works for any length n; we try several lengths (up to 65536)
and several restarts.  A softmax parametrisation guarantees a_i≥0 and Σa_i=1.
The objective is approximated by a smoothed max (log‑sum‑exp) whose
temperature λ is increased during optimisation.  When λ becomes large we
switch to a hard‑max (top‑k) weighting to focus the gradient on the
currently worst convolution entries.  Adam updates are projected back to
the simplex via the softmax, and occasional random “shakes’’ avoid stagnation.
The best sequence encountered (according to the exact objective) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT of given length."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Result length = len(a) (the “valid’’ part of the correlation).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]


def _objective_from_conv(conv: np.ndarray, n: int, s: float) -> float:
    """
    Exact objective used by evaluate_sequence for a vector a with
    sum(a)=s (normally s≈1).  conv is a*a.
    """
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _topk_weights(conv: np.ndarray, top_frac: float = 0.001) -> np.ndarray:
    """
    Return a probability vector w of length len(conv) that puts equal mass on
    the top‑k entries of conv, where k = max(1, floor(top_frac * len(conv))).
    """
    L = conv.shape[0]
    k = max(1, int(top_frac * L))
    # argpartition is O(L), faster than full sort.
    idx = np.argpartition(-conv, k - 1)[:k]
    w = np.zeros_like(conv)
    w[idx] = 1.0 / k
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 1000 s).
    **kwargs : dict
        Unused – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found (as plain Python floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes to explore – larger n gives more flexibility.
    # The list is ordered from largest to smallest so that we may find a
    # very good construction early (and reuse it as a warm‑start later).
    lengths = [65536, 32768, 16384, 8192, 4096, 2048, 1024]

    # Main optimisation loop over lengths
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT sizes (powers of two) – pre‑computed once per length.
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)   # (2n‑1) + n – 1 ≤ 3n

        # Number of independent restarts per length.
        restarts = 5

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------- initial logits ---------------------------
            if r == 0 and best_seq is not None and len(best_seq) == n:
                # Warm‑start from the current global best (if same length)
                a0 = np.array(best_seq, dtype=np.float64)
                a0 = np.maximum(a0, 0.0)
                a0_sum = a0.sum()
                if a0_sum < 0.01:
                    a0 = np.random.dirichlet(np.ones(n))
                else:
                    a0 /= a0_sum
                x = np.log(a0 + 1e-15)
                print("[restart 0] warm start from previous best")
            elif 'height_sequence_1' in globals():
                # Use the historic construction, trimmed / padded to length n
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= s
                x = np.log(cand + 1e-15)
                print("[restart 0] start from height_sequence_1")
            else:
                # Pure random Dirichlet (alpha=0.5 encourages sparsity)
                a0 = np.random.dirichlet(np.full(n, 0.5))
                x = np.log(a0 + 1e-15)
                print("[restart 0] random Dirichlet start")

            # --------------------------- Adam state ---------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04                     # a bit smaller than the baseline
            lam = 0.5                     # initial smooth‑max temperature
            lam_factor = 1.04             # geometric growth
            lam_switch = 2_000            # after this we use hard‑max (top‑k)

            t = 0
            max_iters = 60_000           # generous cap per restart
            last_improve = time.time()
            shake_intensity = 1e-2

            while time.time() < deadline and t < max_iters:
                # ------- logits → probability vector (softmax) -------
                a = _softmax(x)                     # guarantees a_i≥0, Σa_i=1

                # ------- self‑convolution (FFT) -------
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()

                # ------- smooth‑max weights -------
                if lam < lam_switch:
                    # exponential weighting (stable because conv‑max_c ≤ 0)
                    w = np.exp(lam * (conv - max_c))
                    w_sum = w.sum()
                    if w_sum == 0.0:          # numerical safety
                        w = np.full_like(conv, 1.0 / conv.size)
                    else:
                        w /= w_sum
                else:
                    # hard‑max: put uniform weight on the top‑k entries
                    w = _topk_weights(conv, top_frac=0.001)

                # ------- gradient of the surrogate w.r.t. a -------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ------- back‑prop through softmax (gradient w.r.t. logits) -------
                avg = np.dot(a, grad_a)               # scalar ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)           # ∇_x φ

                # ------- Adam update -------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ------- temperature schedule -------
                lam = min(lam * lam_factor, 1e6)

                # ------- exact objective evaluation (fast) -------
                # sum(a) is ≈1, but we keep the exact denominator for safety.
                cur_val = _objective_from_conv(conv, n, a.sum())

                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {cur_val:.6f}")

                # ------- stagnation handling -------
                if time.time() - last_improve > 7.0:
                    # random shake on logits to escape a flat region
                    x += (np.random.rand(n) - 0.5) * shake_intensity
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

                # optional early exit if we already hit the target bound
                if best_val < 1.503:
                    # we have already surpassed the required threshold – stop early.
                    print("Target bound reached, exiting early.")
                    break

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (very unlikely) return a uniform vector.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        print("Fallback: uniform vector used.")

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
