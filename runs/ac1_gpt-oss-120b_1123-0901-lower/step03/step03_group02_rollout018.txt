def _optimise_one_length_adam(n, init, run_deadline, ...):
   target_sum = sqrt(2n)
   a = project(init, target_sum)
   best_seq = a.copy()
   conv = conv_fft(a,a)
   best_score = evaluate via conv: 2n*conv.max()/target_sum**2 (or evaluate_sequence)
   # parameters:
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1=0.9, beta2=0.999, eps=1e-8
   lr0 = 0.5 (or step_factor)
   t=0
   while time.time() < run_deadline:
        conv = conv_fft(a,a)
        max_val = conv.max()
        # compute tau schedule
        tau = max(init_tau_factor * max_val * (0.95**t), min_tau_factor*max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)  # maybe shift to avoid overflow
        w_sum = w.sum()
        w = w / w_sum
        # gradient
        w_rev = w[::-1]
        grad_conv = _full_convolution_fft(a, w_rev)   # length = 3n-2
        idx = (2*n -2) - np.arange(n)
        grad = 2.0 * grad_conv[idx]
        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        # step size scaling may be too large; we can scale to maintain appropriate magnitude.
        # Actually we also want to ensure a stays non-negative, so after update we project onto simplex.
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        # Evaluate new score
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0*n*conv_new.max()/(target_sum**2)
        # track best
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
        # accept new a
        a = a_new
        # optional step size decay: lr0 *= step_decay per iteration
        lr0 *= step_decay
        # occasional random shake
        if np.random.rand() < 0.005:
            noise = np.random.randn(n) * (target_sum * 0.0001 / n)
            a = _project_onto_simplex(np.maximum(a+noise,0), target_sum)
   # final evaluate with evaluate_sequence
   final_score = evaluate_sequence(best_seq.tolist())
   return best_seq, final_score

"""
Beam‑search + Adam optimizer for the “max self‑convolution” objective.

The routine runs many independent restarts with random lengths and
random power‑law / exponential initialisations.  For each restart an
Adam‑style gradient descent on a smooth‑max approximation of the
objective is performed.  After the Adam phase a cheap random‑local
refinement (mass transfer between two coordinates) is applied.  The
best sequence seen across the whole budget is returned.

The implementation follows the imposed API:
    propose_candidate(seed=…, budget_s=…) → list[float]
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project the vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors ``x`` and ``y``."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())            # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv

# Adam‑style optimisation of a single length
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.08,
    min_tau_factor: float = 0.0005,
    lr_initial: float = 0.6,
    lr_decay: float = 0.9995,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam gradient descent on a smooth‑max surrogate of the max‑convolution.
    Returns the best sequence (as a NumPy array) and the exact
    evaluate_sequence() score.
    """
    target_sum = np.sqrt(2.0 * n)      # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # bookkeeping
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-12
    iteration = 0
    lr = lr_initial

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) self‑convolution and max
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) smooth‑max weights
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau        # shift for stability
        w = np.exp(scaled)                     # all entries ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of Σ w_k (a*a)_k  =  2·a ⊛ w   (cross‑correlation)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        corr = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)            # pick the needed slice
        grad = 2.0 * corr[idx]

        # --------------------------------------------------------------
        # 4) Adam update
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # keep feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 5) evaluate (cheap version) and possibly update best
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] n={n}  Adam local best = {best_score:.6f}"
            )

        a = a_new
        lr *= lr_decay

        # --------------------------------------------------------------
        # 6) occasional random shake‑up to escape plateaus
        # --------------------------------------------------------------
        if np.random.rand() < 0.008:
            noise = np.random.randn(n) * (target_sum * 1e-4 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            lr *= 0.9

        # --------------------------------------------------------------
        # 7) optional quick local refinement (mass transfer)
        # --------------------------------------------------------------
        if np.random.rand() < 0.004:
            # pick two distinct indices and move a tiny amount of mass
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.rand() * 0.01 * a[i]   # ≤1 % of a[i]

            a_tmp[i] -= delta
            a_tmp[j] += delta
            # projection not necessary – total sum unchanged, non‑negativity kept

            score_tmp = 2.0 * n * conv_tmp.max() / (target_sum ** 2)
            if score_tmp < best_score:
                a = a_tmp
                best_score = score_tmp
                best_seq = a_tmp.copy()
                print(
                    f"[{time.time():.2f}] n={n}  shake‑refine best = {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Final exact evaluation (uses the provided evaluator)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Optional cheap LP refinement (sidestepping via SciPy)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, M: float) -> np.ndarray:
    """
    Given a current sequence ``f`` and its max‑convolution value ``M``,
    solve the LP that maximises Σ g_j under the constraint
    ``f * g ≤ M`` (element‑wise).  The result is rescaled to the
    canonical simplex Σ g = √(2 n).  If SciPy is unavailable, the input
    is returned unchanged.
    """
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        return f

    n = f.shape[0]
    L = 2 * n - 1                     # length of the convolution

    # Build the sparse matrix A (shape L × n) representing the convolution f * g
    rows = []
    cols = []
    data = []
    for col in range(n):
        # the non‑zero entries for column ``col`` lie on rows ``col … col+n‑1``
        row_start = col
        row_end = col + n                # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([col] * n)
        data.extend([f[col]] * n)
    A = csr_matrix((data, (rows, cols)), shape=(L, n))

    # Linear program: maximise cᵀ g  <=> minimise –cᵀ g
    c = -np.ones(n)                     # minimise –Σ g_j
    bounds = [(0, None)] * n            # g_j ≥ 0
    res = linprog(c, A_ub=A, b_ub=np.full(L, M),
                  bounds=bounds, method="highs", options={"presolve": True})

    if not res.success:
        return f  # fallback

    g = res.x
    # normalise to the simplex Σ g = √(2 n)
    target = np.sqrt(2.0 * n)
    g = _project_onto_simplex(g, target)
    return g
