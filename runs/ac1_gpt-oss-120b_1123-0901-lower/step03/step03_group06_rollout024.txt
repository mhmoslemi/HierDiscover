def _optimize_adam_length(n, p_start, deadline, max_iters=400000):

    N = _next_pow2(3*n - 2)
    p = p_start.copy()
    logp = np.log(p + 1e-30)
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5   # can adjust.
    jitter_interval = 5000
    jitter_scale = 0.001
    beta_start = 0.1
    beta_end = 30.0   # bigger.
    beta_len = 200000  # number of it to reach beta_end.

    best_score = float('inf')
    best_p = p.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # convolution
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new best = {cur_score:.6f}")

        # beta schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)

        # weights
        w = np.exp(beta * (conv - M))

        # rev
        rev = _weighted_rev_fft(p, w, N)   # shape n

        # gradient (use rev directly)
        grad = rev  # convenient

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        # decaying learning rate
        lr = base_lr / (1 + it * 1e-5)

        # update logp
        logp = logp - lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter
        if it % jitter_interval == 0:
            logp = logp + np.random.randn(n) * jitter_scale

        # project onto simplex
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

    return best_p, best_score

"""
Propose a low‑peak step function via Adam‑accelerated exponentiated‑gradient.

Algorithm
---------
1. Load the previously discovered construction (global `height_sequence_1`);
   if unavailable fall back to a uniform distribution.
2. Upsample the starting probability vector to a target length
   (default 8000).  The vector `p` lives on the simplex (Σp = 1).
3. Optimise `p` with a mirror‑descent step on the log‑probabilities
   (exponentiated gradient).  The gradient is the sub‑gradient of the
   smoothed maximum of the self‑convolution `p∗p`.  Smoothing is performed
   by a soft‑max weight `w_k = exp(β·(conv_k−M))` where β grows from a
   small to a large value, thus gradually focusing on the true peak.
4. Adam’s adaptive moments are applied to the log‑gradient to obtain a
   more aggressive yet stable descent.  A small jitter is added every few
   thousand iterations to escape plateaus.
5. The best probability vector seen during the run is kept.  At the end it
   is rescaled to the required L1‑norm `sqrt(2·n)` and returned as a list
   of heights.
6. The whole optimisation respects the wall‑clock deadline (`budget_s`);
   if the time runs out the best vector found so far is returned.

The objective evaluated by `evaluate_sequence` reduces (under the normalisation
Σa = √(2n)) to `max_k (p∗p)_k`.  Minimising this peak is exactly what the
gradient steps target.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Linear self‑convolution (p∗p) using a single FFT of size N.
    Result length = 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i}   for i = 0 … n−1.
    This is a correlation of p with the (reversed) weight vector w.
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # linear part
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _optimize_adam_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector (Σp = 1) and its (scaled) score
    ``2·n·max(p∗p)``.
    """
    # ---------- hyper‑parameters ----------
    beta1, beta2 = 0.9, 0.999            # Adam moments
    eps = 1e-8
    base_lr = 0.5                       # initial learning‑rate on log‑space
    jitter_interval = 5_000
    jitter_scale = 0.001
    beta_start = 0.1
    beta_end = 100.0
    beta_ramp_iters = 200_000           # where β reaches beta_end

    # ---------- initialise ----------
    p = p_start.copy()
    logp = np.log(p + 1e-30)            # stay in log‑space
    m = np.zeros_like(p)                # first Adam moment
    v = np.zeros_like(p)                # second Adam moment

    N = _next_pow2(3 * n - 2)           # size sufficient for all convolutions

    best_score = float("inf")
    best_p = p.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution (p∗p) -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new best = {cur_score:.6f}")

        # ----- soft‑max weights (focus on the peak) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)
        w = np.exp(beta * (conv - M))               # w_k ≤ 1, w_max = 1

        # ----- gradient of the smoothed max (rev) -----
        rev = _weighted_rev_fft(p, w, N)            # shape (n,)
        grad = rev                                   # effective gradient on log‑p

        # ----- Adam moments -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- decaying step‑size -----
        lr = base_lr / (1.0 + it * 1e-5)

        # ----- EG‑style log‑update with Adam scaling -----
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat regions
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)          # avoid overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_score

        print("[init] no existing construction found")

    # ----- target length (larger usually gives a better bound) -----
    target_n = 8000

    # ----- initialise probability vector p (Σp = 1) -----
    if base is not None and base.size > 0:
        n0 = base.shape[0]
        S0 = math.sqrt(2.0 * n0)
        p0 = base / S0                     # convert to probability vector
        p0 = np.maximum(p0, 0.0)
        p0 /= p0.sum()

        # linear interpolation to the target length
        x_old = np.linspace(0.0, 1.0, n0)
        x_new = np.linspace(0.0, 1.0, target_n)
        p_init = np.interp(x_new, x_old, p0)
        p_init = np.maximum(p_init, 0.0)
        p_init /= p_init.sum()
        print(f"[init] up‑sampled from {n0} → {target_n}")
    else:
        # simple uniform start
        p_init = np.full(target_n, 1.0 / target_n)
        print(f"[init] using uniform start, length = {target_n}")

    # ----- run the main optimisation -----
    best_p, best_local_score = _optimize_adam_length(
        n=target_n,
        p_start=p_init,
        deadline=deadline,
    )

    # ----- convert back to the original scale required by evaluate_sequence -----
    S = math.sqrt(2.0 * target_n)
    best_seq = (best_p * S).tolist()

    # ----- final validation (optional, cheap) -----
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] length = {target_n}, score = {final_score:.6f}")
    except Exception as exc:
        print(f"[final] evaluation failed: {exc}")

    return best_seq
