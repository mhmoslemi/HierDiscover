        F(a) = 2·n·max(a*a) / (sum(a)²)

using a projected‐gradient descent that explicitly enforces a
non‑increasing shape (the intuition behind the best known constructions

reversed copy almost flat).  The algorithm works for arbitrary lengths,
starts from several random restarts (and from the previous global best
if available), continually refines the surrogate smooth‑max objective
with an Adam‑style update, and after each step projects the iterate onto

    – the monotone decreasing cone,
    – the probability simplex (non‑negative, sum = 1).

A simple “shake’’ (random perturbation) is applied when the search
stagnates.  The routine respects the time budget, prints progress and
returns the best sequence found as a plain Python list.

# FFT based convolution utilities (same as in the reference implementation)
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n

# Simple isotonic regression (non‑increasing) – O(n)
def _isotonic_decreasing(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the cone of non‑increasing sequences
    (PAVA algorithm).  Returns a new ndarray with the same shape.
    """
    # Work with the negated vector to reuse the classic increasing PAVA.
    y = -v
    n = y.shape[0]
    # Each block is represented by (sum, count)
    sums = []
    counts = []
    for i in range(n):
        sums.append(y[i])
        counts.append(1)
        # Merge while the previous block's average exceeds the current one.
        while len(sums) >= 2 and (sums[-2] / counts[-2]) > (sums[-1] / counts[-1]):
            last_sum = sums.pop()
            last_cnt = counts.pop()
            sums[-1] += last_sum
            counts[-1] += last_cnt
    # Expand the blocks back to a full vector.
    out = np.empty_like(y)
    idx = 0
    for block_sum, block_cnt in zip(sums, counts):
        avg = block_sum / block_cnt
        out[idx: idx + block_cnt] = avg
        idx += block_cnt
    return -out


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The function respects the given time budget and returns a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: exact objective (sum of a is 1) – cheaper than calling
    # evaluate_sequence repeatedly.
    def _exact_obj(a: np.ndarray, n: int, conv_max: float) -> float:
        """Exact return value of evaluate_sequence when sum(a)=1."""
        return 2.0 * n * conv_max / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # Problem sizes – start small, finish with a larger one.
    lengths = [512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # safe upper bound

        # Number of random restarts for this length – more restarts for
        # smaller n (fast) and fewer for the large n.
        restarts = 6 if n <= 2048 else 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # -------------------------------------------------------------
            # Initialisation ------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                # Warm start from the current global best of the same length.
                a = np.array(best_seq, dtype=np.float64)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 0:
                    a = a / s
                else:
                    a = np.random.dirichlet(np.ones(n))
                print("[restart 0] warm start from global best")
            elif best_seq is not None and len(best_seq) < n:
                # Upsample the previous best (linear interpolation) to new n.
                old = np.array(best_seq, dtype=np.float64)
                old = np.maximum(old, 0.0)
                x_old = np.linspace(0, 1, old.shape[0])
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, old)
                a = np.maximum(a, 0.0)
                s = a.sum()
                a = a / s if s > 0 else np.random.dirichlet(np.ones(n))
                a = _isotonic_decreasing(a)           # enforce monotone shape
                a = a / a.sum()
                print("[restart 0] up‑sample from previous best")
            elif 'height_sequence_1' in globals():
                # Use the provided construction (trim / pad) as a seed.
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                a = _isotonic_decreasing(a)
                a = a / a.sum()
                print("[restart 0] start from height_sequence_1")
            else:
                # Pure random Dirichlet start.
                a = np.random.dirichlet(np.ones(n))
                a = _isotonic_decreasing(a)
                a = a / a.sum()
                print("[restart 0] random start")

            # -------------------------------------------------------------
            # Adam‑style state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05

            # Smooth‑max temperature (λ) – start small and grow.
            lam = 0.5
            lam_factor = 1.04

            t = 0
            max_iters = 20000
            last_improve_time = time.time()

            while time.time() < deadline and t < max_iters:
                # ---------- (softmax‑free) variables – a already lies on simplex ----------
                # Compute full self‑convolution.
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()

                # ---------- smooth‑max weights ----------
                # w ∝ exp(λ·(conv - max_c))
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.ones_like(w) / w.size
                else:
                    w = w / w_sum

                # ---------- gradient of the surrogate w.r.t. a ----------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # ---------- Adam update ----------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

                # Enforce feasibility: non‑negative, monotone decreasing, simplex.
                a = np.maximum(a, 0.0)
                a = _isotonic_decreasing(a)
                s = a.sum()
                if s > 0:
                    a = a / s
                else:
                    a = np.random.dirichlet(np.ones(n))

                # ---------- temperature schedule ----------
                lam = min(lam * lam_factor, 1e6)

                # ---------- exact objective (fast) ----------
                val = _exact_obj(a, n, max_c)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve_time = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

                # ---------- stagnation handling ----------
                if time.time() - last_improve_time > 6.0:
                    # Small random shake of the current point.
                    shake = (np.random.rand(n) - 0.5) * 5e-3
                    a = a + shake
                    a = np.maximum(a, 0.0)
                    a = _isotonic_decreasing(a)
                    a = a / a.sum()
                    last_improve_time = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

                # optional: slowly decay learning rate
                lr *= 0.9999

    # ------------------------------------------------------------------
    # Fallback: uniform vector if nothing else succeeded.
    if best_seq is None:
        fallback_n = 2000
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
