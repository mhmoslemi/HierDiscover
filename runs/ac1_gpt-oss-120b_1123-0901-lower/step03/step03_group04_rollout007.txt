def _smooth_max_gradient(a, alpha):
    n = a.shape[0]
    # convolution of a with a
    max_val, conv = _conv_max_fft(a)
    # compute weights
    exp_weights = np.exp(alpha * (conv - max_val))
    w = exp_weights / exp_weights.sum()
    # compute gradient (2 * convolution of w and reversed a)
    a_rev = a[::-1]
    # compute convolution via FFT
    L = conv.shape[0] # =2n-1
    N = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, n=N)
    A_rev = np.fft.rfft(a_rev, n=N)
    grad_conv = np.fft.irfft(W * A_rev, n=N)
    # grad_conv shape length L + n - 1 = 3n-2
    start = n - 1
    grad = 2 * grad_conv[start:start + n]
    return max_val, grad
```

def _adam_smooth_optimize(init_a, n, target_sum, deadline, max_iters=200000):
    a = init_a.copy()
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.03
    iteration = 0
    best_val, _ = _conv_max_fft(a)
    best_vec = a.copy()
    # schedule for alpha
    alpha_start = 0.5
    alpha_increment = 0.005   # per iteration
    alpha = alpha_start

    while time.time() < deadline and iteration < max_iters:
        # compute smooth gradient
        M, grad = _smooth_max_gradient(a, alpha)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = base_lr / (1 + 1e-4 * iteration)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto non-negative
        a = np.maximum(a, 0.0)
        # renormalize sum
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # update best
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[smooth] n={n} iter={iteration} M={best_val:.6f}")

        # increase alpha
        alpha = min(alpha + alpha_increment, 50.0)  # upper bound

        # occasional jitter
        if iteration % 3000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

    return best_vec, best_val
```

"""
Hybrid optimiser for the max‑convolution objective.

The score returned by ``evaluate_sequence`` is
    2·n·max(conv(a,a)) / (sum(a))²

and is invariant under scaling of the whole vector.  Hence we can keep the
ℓ₁‑norm fixed ( = √(2·n) ) and simply minimise the peak of the (linear)
self‑convolution.

The search proceeds as follows for a given length n:

1.  Random initial vector (Dirichlet‑like) satisfying the ℓ₁‑norm.
2.  *Smooth‑max* phase – minimise a differentiable approximation of the
    maximum using Adam.  The smooth objective is
            M_α(a) = (1/α)·log( Σ_k exp(α·conv_k) )
    whose gradient can be written as a convolution of the softmax weights
    with the reversed vector; both convolutions are performed via FFT.
    α is increased gradually so that the optimisation starts very smooth
    and ends close to the true max.

3.  *Active‑set* phase – switch to the exact sub‑gradient of the true
    max (average of the partner entries over all active shifts) and continue
    with Adam.  This refines the solution once it is already close to a
    plateau.

4.  A cheap stochastic local‑search that moves a small amount of mass
    between two random indices if it improves the peak.

The routine explores several problem sizes (multiples of the length of the
previous best construction) and performs a few random restarts for each.
Time is limited to the supplied budget (≈1000 s).  The best sequence found
before the deadline is returned.
"""
import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return max(conv(a,a)) and the full convolution array using FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active: np.ndarray):
    """Sub‑gradient of the max‑convolution (average over the active set)."""
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
    if active.size > 0:
        grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth approximation of the max (log‑sum‑exp) of the
    self‑convolution.  Returns the current max value and the gradient vector.
    """
    n = a.shape[0]
    # full convolution
    max_val, conv = _conv_max_fft(a)

    # softmax weights (shifted by the max to avoid overflow)
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (w * a_rev) convolution, where a_rev[i] = a[n-1-i]
    a_rev = a[::-1]
    L = conv.shape[0]                     # 2*n‑1
    N = 1 << ((L + n - 1).bit_length())   # for w (len L) and a_rev (len n)
    W = np.fft.rfft(w, n=N)
    A_rev = np.fft.rfft(a_rev, n=N)
    grad_full = np.fft.irfft(W * A_rev, n=N)[: L + n - 1]   # length 3n‑2

    start = n - 1
    grad = 2.0 * grad_full[start:start + n]   # size n
    return max_val, grad


# ----------------------------------------------------------------------
def _adam_optimize_smooth(init_a: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 10_000):
    """Adam optimiser using the smooth‑max gradient."""
    a = init_a.copy()
    # normalise to the prescribed ℓ₁‑norm
    s = a.sum()
    a = a * (target_sum / s) if s > 0 else np.full_like(a, target_sum / a.size)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.04
    iteration = 0
    best_vec = a.copy()
    best_val, _ = _conv_max_fft(a)

    alpha = 0.5          # start with a very smooth approximation
    alpha_inc = 0.005   # increase each iteration, capped later

    while time.time() < deadline and iteration < max_iters:
        M, grad = _smooth_max_gradient(a, alpha)

        iteration += 1
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate decay
        lr = base_lr / (1.0 + 1e-4 * iteration)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce feasibility
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a * (target_sum / s) if s > 0 else np.full_like(a, target_sum / a.size)

        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[smooth] iter {iteration:6d} n={len(a)} M={best_val:.6f}")

        # increase smoothing strength
        alpha = min(alpha + alpha_inc, 50.0)

        # occasional jitter to escape plateaus
        if iteration % 4000 == 0:
            jitter = np.random.randn(len(a)) * 0.001 * target_sum / len(a)
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

    return best_vec, best_val


# ----------------------------------------------------------------------
def _adam_optimize_max(init_a: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       max_iters: int = 10_000):
    """Adam optimiser on the exact max‑convolution sub‑gradient."""
    a = init_a.copy()
    s = a.sum()
    a = a * (target_sum / s) if s > 0 else np.full_like(a, target_sum / a.size)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.04
    iteration = 0
    best_vec = a.copy()
    best_val, _ = _conv_max_fft(a)

    while time.time() < deadline and iteration < max_iters:
        M, conv = _conv_max_fft(a)

        # active shifts: within a tiny tolerance of the current max
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # exact sub‑gradient of the max term
        grad_rev = _subgradient(a, active)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_rev
        v = beta2 * v + (1.0 - beta2) * (grad_rev * grad_rev)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a * (target_sum / s) if s > 0 else np.full_like(a, target_sum / a.size)

        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[max]    iter {iteration:6d} n={len(a)} M={best_val:.6f}")

        if iteration % 3000 == 0:
            jitter = np.random.randn(len(a)) * 0.001 * target_sum / len(a)
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """Simple greedy local search: transfer a tiny fraction of mass."""
    n = a.shape[0]
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = 0.02 * best[i]          # move 2 % of source
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # non‑negativity is already satisfied
        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M
    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a safety margin for the outer harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # baseline: try to reuse a previously discovered construction
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        print("[info] loaded previous construction of length", base.shape[0])
    except Exception:
        base = None
        print("[info] no previous construction – will start from scratch")

    # --------------------------------------------------------------
    # global best tracker
    best_seq = None
    best_obj = float('inf')

    if base is not None:
        val, _ = _conv_max_fft(base)
        best_seq = base.copy()
        best_obj = val
        print(f"[init] loaded sequence length={len(base)}  M={best_obj:.6f}")

    # --------------------------------------------------------------
    # choose a set of problem sizes to explore
    n0 = len(best_seq) if best_seq is not None else 2100
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({
        max(300, min(12000, int(n0 * f))) for f in length_factors
    })
    print("[info] exploring lengths:", lengths)

    # --------------------------------------------------------------
    # Main optimisation loops
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length – fewer for the larger sizes
        restarts = 3 if n <= 4000 else 2

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise -------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n and r == 0:
                init = best_seq.copy()
                print(f"[restart] length={n} using global best as start")
            else:
                rand = np.random.exponential(scale=1.0, size=n)
                init = rand * (target_sum / rand.sum())

                # blend a little with the current global best if compatible
                if best_seq is not None and best_seq.shape[0] == n:
                    mix = 0.2
                    init = mix * best_seq + (1 - mix) * init
                    init = np.maximum(init, 0.0)
                    init *= (target_sum / init.sum())

            # ----- smooth‑max optimisation ------------------------------------
            smooth_deadline = deadline
            seq_smooth, val_smooth = _adam_optimize_smooth(
                init, target_sum, smooth_deadline, max_iters=5000)

            # ----- exact‑max optimisation ------------------------------------
            seq_max, val_max = _adam_optimize_max(
                seq_smooth, target_sum, smooth_deadline, max_iters=5000)

            # ----- cheap stochastic local search -------------------------------
            if time.time() < deadline:
                seq_local, val_local = _local_search(
                    seq_max, target_sum, steps=30000, deadline=deadline)
                # pick the best among the three phases
                candidates = [(seq_smooth, val_smooth),
                              (seq_max, val_max),
                              (seq_local, val_local)]
                seq_best, val_best = min(candidates, key=lambda kv: kv[1])
            else:
                seq_best, val_best = seq_max, val_max

            # ----- update global best -----------------------------------------
            if val_best < best_obj:
                best_obj = val_best
                best_seq = seq_best.copy()
                print(f"[global] new best M={best_obj:.6f} (len={n})")

    # --------------------------------------------------------------
    # final sanity check through the official evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] reported M={best_obj:.6f}, official score={final_score:.6f}, "
          f"length={len(best_seq)}")
    return best_seq.tolist()
