def project_to_simplex(v, s):
    # v: array
    # returns projected onto {x>=0, sum=x = s}
    # Duchi algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def gradient_descent(a, target_sum, deadline, current_best, max_iter=300000):
    n = a.shape[0]
    best_local_seq = a.copy()
    best_local_score = float('inf')
    # compute initial score
    M, _, _ = _conv_max_fft(a)
    if M < best_local_score:
        best_local_score = M
        best_local_seq = a.copy()

    eta0 = 0.5
    tol = 1e-12
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # compute conv and max
        M, _, conv = _conv_max_fft(a)
        # record best
        if M < best_local_score:
            best_local_score = M
            best_local_seq = a.copy()
        # also update global if needed (passed via caller)
        # compute active shifts
        active = np.where(conv >= M - tol)[0]
        # build subgradient (average rev)
        rev = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        if active.size > 0:
            rev = rev / active.size

        # step size
        eta = eta0 / math.sqrt(iteration)

        # descent step
        a_new = a - eta * rev

        # project onto simplex
        a_new = project_to_simplex(a_new, target_sum)

        a = a_new
    return best_local_seq, best_local_score

def _gradient_descent(a, target_sum, deadline, current_best_score, max_iter=200000):
    n = a.shape[0]
    best_seq = a.copy()
    best_score = current_best_score

    eta0 = 0.5
    tol = 1e-12
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        M, _, conv_full = _conv_max_fft(a)

        # update best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[{n}] iter {iteration} new best local score {best_score:.6f}")

        # active shifts
        active = np.where(conv_full >= M - tol)[0]

        # compute subgradient
        if active.size > 0 and np.random.rand() < 0.2:
            k = np.random.choice(active)
            rev = _rev_for_shift(a, k)
        else:
            rev = np.zeros_like(a)
            idx = np.arange(n)
            for k in active:
                j = k - idx
                mask = (j >= 0) & (j < n)
                rev[mask] += a[j[mask]]
            if active.size:
                rev /= float(active.size)

        # step size
        eta = eta0 / math.sqrt(iteration)

        # gradient step (descending)
        a = a - eta * rev

        # projection onto feasible set (non-neg then renorm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (target_sum / n) * 0.05
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_seq, best_score

"""
Propose a step‑function that (approximately) minimizes the
evaluate_sequence objective.

The algorithm proceeds in several stages:

1.  Choose a sequence of lengths (starting from a modest size and
    increasing up to ~12 k).  For each length we run a few independent
    restarts.
2.  Initialise each restart with a “Beta‑shaped’’ distribution
    (heavy at the ends, light in the middle).  The shape is generated
    by a simple 1/√(i·(n‑i)) law – an analogue of the continuous
    arcsine law whose self‑convolution is relatively flat.
3.  Perform a projected sub‑gradient descent on the infinity‑norm of
    the autocorrelation.  The sub‑gradient is obtained from a randomly
    chosen active shift (the shift attaining the maximal convolution
    value).  After each step the vector is projected back onto the
    simplex {x ≥ 0, Σx = √(2 n)} by simple clipping + renormalisation.
    A decreasing step‑size  ηₜ = η₀/√t  and occasional small Gaussian
    jitter help to escape plateaus.
4.  Refine the result with a cheap coordinate‑swap hill‑climb that
    moves a small amount of mass from one index to another only
    when the maximal convolution value decreases.
5.  Keep the best sequence found over all lengths and restarts and
    return it as a plain Python list before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Compute the full autocorrelation a*a via FFT and return the maximal
    value, its index and the whole convolution array.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, return the vector rev where
    rev[i] = a[k‑i] if the index is inside the support, otherwise 0.
    This is the sub‑gradient of the convolution at shift k.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _init_arcsine(n: int, alpha: float = 0.6) -> np.ndarray:
    """
    Initialise a vector with a Beta(α,α)‑like shape.
    The density ∝ (x(1‑x))^{α‑1} is heavy near the endpoints,
    approximating an arcsine distribution while staying finite.
    """
    i = np.arange(n)
    # avoid division by zero at the exact boundaries
    eps = 1e-12
    x = (i + 0.5) / n
    pdf = np.where((x <= eps) | (x >= 1 - eps),
                   0.0,
                   (x * (1.0 - x)) ** (alpha - 1.0))
    if pdf.sum() == 0.0:
        pdf = np.ones_like(x)
    return pdf.astype(float)


# ----------------------------------------------------------------------
def _gradient_descent(a: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      global_best: float,
                      max_iter: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on max‑convolution.
    Returns the best vector found and its corresponding max‑convolution.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = float('inf')
    eta0 = 0.5                 # initial step size (tuned empirically)
    tol = 1e-12
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        M, _, conv_full = _conv_max_fft(a)

        # keep track of the best local vector
        if M < best_score:
            best_score = M
            best_vec = a.copy()
            # occasional report
            if it % 5000 == 0:
                print(f"[{n}] iter {it:7d}  local best = {best_score:.6f}")

        # stop early if we cannot beat the current global best any more
        if best_score >= global_best:
            # continue a few more iterations in case we escape a plateau
            if it % 2000 != 0:
                continue

        # active shifts (indices achieving the maximal value)
        active = np.where(conv_full >= M - tol)[0]

        # random active shift -> cheap sub‑gradient
        if active.size:
            k = np.random.choice(active)
            rev = _rev_for_shift(a, k)
        else:
            rev = np.zeros_like(a)

        # decreasing step size
        eta = eta0 / math.sqrt(it)

        # gradient step (we move opposite to the sub‑gradient)
        a = a - eta * rev

        # projection onto the feasible simplex (non‑negative, fixed L1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional tiny Gaussian jitter to leave flat regions
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (target_sum / n) * 0.05
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_vec, best_score


# ----------------------------------------------------------------------
def _local_search(a_start: np.ndarray,
                  n: int,
                  target_sum: float,
                  global_best: float,
                  deadline: float,
                  max_proposals: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: repeatedly move a tiny amount of mass from one
    index to another, accepting the move only if the maximal convolution
    value decreases.
    """
    a = a_start.copy()
    best_score = global_best
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        eps = a[i] * np.random.rand() * 0.05  # up to 5 % of donor mass
        if eps <= 0:
            continue
        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps
        M_new, _, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (proposal {proposal})")
    return a, best_score
