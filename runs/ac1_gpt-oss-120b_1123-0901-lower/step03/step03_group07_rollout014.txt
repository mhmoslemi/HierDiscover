def project_to_simplex(v, S):
    # ensures non-negative
    # algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def refine_lp(a, n, deadline, max_iter=15, t=0.5):
    S = np.sqrt(2*n)
    for i in range(max_iter):
        if time.time() >= deadline - 0.5:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        # Build A matrix for constraints
        # we can use scipy.linalg.toeplitz for speed
        from scipy.linalg import toeplitz
        col = np.concatenate([a, np.zeros(n-1)])
        row = np.concatenate([[a[0]], np.zeros(n-1)])
        A = toeplitz(col, row)
        # solve LP: maximize sum(b) subject to A @ b <= M, b>=0
        c = -np.ones(n)   # minimize -sum(b)
        b_ub = np.full(2*n-1, M)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            break
        scale = S / sum_b
        g = b * scale
        # compute mixture and project to simplex (sum remains S)
        a_new = (1 - t) * a + t * g
        # (since both sum to S, mixture sum = S)
        # compute objective improvement
        new_val = 2.0 * n * np.convolve(a_new, a_new).max() / (S**2)
        old_val = 2.0 * n * np.convolve(a, a).max() / (S**2)
        if new_val < old_val - 1e-8:
            a = a_new
            # maybe reduce t (increase movement?) but we can keep .
            # maybe continue.
            # Print improvement
            print(f"LP iter {i}: improved val {new_val:.6f}")
        else:
            # no improvement, break
            break
    return a
```

def _adam_optimize(a0, n, deadline, max_iter=12000):
    a = a0.copy()
    u = np.log(a + 1e-30)       # log space
    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1=0.9; beta2=0.999; eps=1e-8; base_lr=0.05; lam=1.0; lam_factor=1.02
    best_val = float('inf')
    best_seq = a.copy()
    last_improve = time.time()
    it = 0
    conv_fft_size = 1 << int(np.ceil(np.log2(2*n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3*n))) 
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute a
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()
        # convolution
        conv = _full_convolution(a, size=conv_fft_size)
        # smoothmax weights
        w = _softmax_weights(conv, lam)
        # grad w.r.t a
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)
        # log-space gradient
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr * (0.9995 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # Lambda update
        lam *= lam_factor
        # periodic evaluation
        if it % 20 == 0:
            val = _objective(a)  # compute exact objective
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                # print progress
                print(f"ADAM iter {it} best = {val:.6f}")
        # shake if stagnated
        if time.time() - last_improve > 2.0:
            u += (np.random.rand(n) - 0.5) * 5e-4
            m.fill(0)
            v.fill(0)
            last_improve = time.time()
    return best_seq

        # mixture

        if val_new < best_val - 1e-8:

            best_a = a_new
            print(f"LP refine step improved val = {val_new:.6f}")
            # optionally decrease t for smaller steps if improvement continues
        else:
            # no improvement, break
            break
    return best_a

"""
Propose a step‐function (non‑negative heights) that minimizes

        F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)²

The routine mixes three ideas that proved useful for this problem:

1. **Adam on a smooth‑max surrogate** – the sequence is kept in soft‑max
   (log‑space) so the simplex constraint Σ a_i = 1 is fulfilled automatically.
   A smooth approximation of the max (soft‑max with a temperature λ) yields
   a differentiable surrogate that we minimise with Adam.  The temperature is
   increased slowly (continuation) so the surrogate becomes tighter over time.

2. **Linear‑programming (LP) refinement** – after a (possibly long) Adam run we
   solve the LP

        maximise   Σ b_i
        subject to  a ∗ b  ≤  M   (component‑wise)
                    b_i ≥ 0

   where M = max(a∗a).  The LP finds a “partner” sequence b with the largest
   possible total mass that still respects the current maximal convolution.
   Rescaling b to unit sum and mixing it with a (convex combination) is a
   provably non‑increasing step for the exact objective.  A few LP steps are
   cheap and often push the value noticeably lower than pure gradient descent.

3. **Multiple random restarts & families** – to avoid getting trapped in a
   local minimum we start from several simple families (Dirichlet,
   geometric, half‑sine) and keep the best sequence seen before the time budget
   expires.

The whole search respects the 1000 s budget, prints progress, and finally
returns the best sequence as a plain Python list ready to be fed to
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy import optimize, linalg

# ----------------------------------------------------------------------
# FFT‑based full (linear) self‑convolution
def _full_convolution(a: np.ndarray, size: int | None = None) -> np.ndarray:
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max weights  w_k = exp(lam·(conv_k - max(conv))) / Σ_j …
def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    m = conv.max()
    # conv - m ≤ 0 ⇒ exponent never blows up
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                         # extreme under‑flow protection
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate w.r.t. the probabilities a
def _softmax_gradient(a: np.ndarray, w: np.ndarray,
                      size_corr: int | None = None) -> np.ndarray:
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # pick the slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Random initialisers (three simple families)
def _init_sequence(n: int, mode: int) -> np.ndarray:
    if mode == 0:          # Dirichlet (uniformly random on the simplex)
        a = np.random.dirichlet(np.ones(n))
    elif mode == 1:        # geometric decay with a random ratio
        r = np.random.uniform(0.2, 0.95)
        powers = np.power(r, np.arange(n, dtype=np.float64))
        a = powers / powers.sum()
    else:                  # half‑sine shape (strictly positive)
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.sin(np.pi * i / (n + 1))
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
def _adam_optimize(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    # parameters – a little more aggressive than the reference code
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.07          # slightly larger initial step
    lam = 1.0
    lam_factor = 1.015      # slower continuation (more accurate gradients)

    # FFT sizes that stay constant for the whole run
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))

    # log‑space parametrisation (soft‑max) – guarantees Σ a_i = 1
    u = np.log(a0 + 1e-30)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_val = float("inf")
    best_seq = None
    it = 0
    last_improve = time.time()

    # run until the global deadline or until a sensible iteration limit is reached
    max_iter = 15000
    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- probabilities from logits -----
        u_shift = u - np.max(u)          # for numerical stability
        a = np.exp(u_shift)
        a /= a.sum()

        # ----- surrogate quantities -----
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)

        # ----- gradient -----
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)      # Jacobian of soft‑max

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr * (0.9995 ** it)    # gentle decay
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- increase temperature -----
        lam *= lam_factor

        # ----- exact evaluation every few steps -----
        if it % 20 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam] it {it:05d}  new local best = {val:.6f}")

        # ----- shake if we have stalled -----
        if time.time() - last_improve > 1.8:
            u += (np.random.rand(n) - 0.5) * 5e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    # fall back to the last feasible iterate if the inner loop never updated best_seq
    if best_seq is None:
        best_seq = a.copy()
    return best_seq


# ----------------------------------------------------------------------
# One or a few LP refinement steps (see the description above)
def _lp_refine(a: np.ndarray, n: int, deadline: float,
               max_steps: int = 8, mix_factor: float = 0.5) -> np.ndarray:
    """
    Given a unit‑sum vector a, perform up to max_steps of the LP update described
    in the statement.  mix_factor controls how far we move towards the LP
    partner (default 0.5 i.e. a simple averaging).
    """
    best_a = a.copy()
    best_val = _objective(best_a)

    # build the Toeplitz matrix once per LP call (it depends on the current a)
    # The matrix encodes the linear convolution a * b.
    def _make_A(vec):
        col = np.concatenate([vec, np.zeros(n - 1)])
        row = np.concatenate([[vec[0]], np.zeros(n - 1)])
        return linalg.toeplitz(col, row)

    steps = 0
    while time.time() < deadline and steps < max_steps:
        steps += 1
        conv = np.convolve(best_a, best_a)
        M = conv.max()

        A = _make_A(best_a)                     # shape (2n‑1) × n
        c = -np.ones(n)                         # minimise –Σ b_i  ⇔ maximise Σ b_i
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = optimize.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=bounds, method='highs')
        if not res.success:
            # Numerical issues – abort the refinement for this restart
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Scale partner to unit sum
        g = b / sum_b

        # Convex combination; unit‑sum is preserved automatically
        a_new = (1.0 - mix_factor) * best_a + mix_factor * g
        val_new = _objective(a_new)

        if val_new < best_val - 1e-8:
            best_val = val_new
            best_a = a_new
            print(f"[LP] step {steps}  improved value = {val_new:.6f}")
        else:
            # no improvement – stop early
            break

    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a low‑evaluation step function within the allotted time budget.
    The function returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin for the caller

    # --------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed.
    n = 2000                                    # can be tuned up/down

    best_val = float("inf")
    best_seq = None
    restart = 0

    while time.time() < deadline:
        restart += 1
        mode = restart % 3

        # ----- random start -----
        a0 = _init_sequence(n, mode)

        # ----- Adam optimisation -----
        a_adam = _adam_optimize(a0, n, deadline)

        # ----- LP refinement (often yields a noticeable drop) -----
        a_refined = _lp_refine(a_adam, n, deadline)

        # ----- final evaluation -----
        val = _objective(a_refined)
        if val < best_val:
            best_val = val
            best_seq = a_refined.copy()
            print(f"[Restart {restart:03d}] NEW GLOBAL BEST = {val:.6f}")

        # early exit if we have already hit the target
        if best_val <= 1.503:
            break

    # --------------------------------------------------------------
    # In the extremely unlikely event that nothing was found,
    # fall back to a uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # Scale for safety (evaluate_sequence caps values to [0,1000] anyway)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"FINAL BEST OBJECTIVE = {best_val:.6f} (length = {n})")
    return best_seq.tolist()
