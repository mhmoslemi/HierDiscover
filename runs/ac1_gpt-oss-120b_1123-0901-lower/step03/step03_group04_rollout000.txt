"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Approach
--------
1. **Normalise** every candidate to Σa = √(2·n).  With this normalisation the
   score returned by ``evaluate_sequence`` equals the maximum entry of the
   full convolution a∗a, so we simply minimise that peak.

2. **Conditional‑gradient (Frank‑Wolfe) step** – for the current vector `a`
   we solve the linear programme  

        maximise   Σ b_i
        subject to a∗b ≤ M          (M = max(a∗a))
                   b_i ≥ 0

   This finds a direction `b` that puts as much mass as possible into
   locations where `a` is small without violating the current peak `M`.

3. **Mix** the new direction with the current vector,
   `a ← (1‑t)·a + t·b` (t≈0.2) and re‑normalise.  The peak of the
   convolution usually drops.  The process is repeated while it yields
   improvement.

4. **Local refinement** – after the LP loop a short Adam optimiser
   (the one used in the reference code) is run for a few thousand
   iterations to polish the result.

5. **Multi‑size search** – the whole procedure is executed for several
   lengths (≈1200 … 2400).  The best sequence over all lengths is returned.

The algorithm stays well below the 1000 s budget, uses only ``numpy``,
``scipy`` (HiGHS LP solver) and the supplied ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (max convolution value, full convolution) using FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    return float(conv.max()), conv

    best_vec = a.copy()
    best_val, _ = _conv_max_fft(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        M, conv = _conv_max_fft(a)
        # active set (values within a tiny tolerance of the maximum)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of the max‑convolution (average over active shifts)
        n = a.size
        idx = np.arange(n)
        grad = np.zeros_like(a)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += a[j[mask]]
        if active.size:
            grad /= float(active.size)

        # full gradient of the objective 2·n·max/(sum²)  (after normalisation the factor is 1)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * grad - const

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + 1e-4 * it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto non‑negative simplex of the required L1‑norm
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a.fill(target_sum / a.size)

        if M < best_val:
            best_val = M
            best_vec = a.copy()
    return best_vec, best_val


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float, target_sum: float):
    """
    Solve   maximise Σ b_i   s.t.   a∗b ≤ M ,   b ≥ 0.
    Returns the optimiser vector b (already normalised to target_sum).
    """
    n = a.size
    # Build the (2n‑1)×n convolution matrix A where (A·b)[k] = (a∗b)[k]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]                     # column i contributes to rows i … i+n‑1

    c = -np.ones(n)                              # maximise Σb → minimise –Σb
    ub = np.full(rows, M, dtype=np.float64)      # a∗b ≤ M componentwise
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        raise RuntimeError("LP failed: " + (res.message or "unknown"))
    b = res.x
    # normalise to the required L1‑norm
    s = b.sum()
    if s == 0:
        raise RuntimeError("LP produced all‑zero solution")
    b *= (target_sum / s)
    return b

def propose_candidate(seed: int = 42, budget_s: float = 1000, **_):
    """
    Search for a step‑function with minimal ``evaluate_sequence`` value.
    Returns the best list of heights found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # Keep a small safety margin for the surrounding harness.
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a previous construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        print("[info] loaded previous construction of length", base.shape[0])
    except Exception:
        base = None
        print("[info] no previous construction available")

    # ------------------------------------------------------------------
    # Lengths that we will explore (kept moderate so that the LP stays cheap)
    lengths = [1200, 1500, 1800, 2100, 2400]
    best_seq = None
    best_val = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- initialise candidate ------------------------------------------------
        if base is not None and base.shape[0] == n:
            a = base.copy()
        else:
            # start from a uniform distribution
            a = np.full(n, target_sum / n, dtype=float)

        # small random perturbation to avoid pathological symmetry
        a += np.random.randn(n) * 0.001 * target_sum / n
        a = np.maximum(a, 0.0)
        a *= (target_sum / a.sum())

        # ---- main conditional‑gradient loop ------------------------------------
        M, _ = _conv_max_fft(a)                     # current peak
        no_improve = 0
        iter_lp = 0
        max_lp_iters = 8                           # keep the total time low

        while time.time() < deadline and iter_lp < max_lp_iters:
            iter_lp += 1
            try:
                b = _lp_step(a, M, target_sum)
            except Exception as e:
                print(f"[lp] length {n} iteration {iter_lp} failed:", e)
                break

            # mix a little towards the direction found by the LP
            t = 0.2                                   # mixing weight
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            a_new *= (target_sum / a_new.sum())

            M_new, _ = _conv_max_fft(a_new)

            if M_new < M - 1e-6:                     # strict improvement
                a = a_new
                M = M_new
                no_improve = 0
                print(f"[len {n:5d}] LP iter {iter_lp:2d} → M = {M:.6f}")
            else:
                no_improve += 1
                if no_improve >= 3:
                    break

        # ---- short Adam polishing ------------------------------------------------
        # give the Adam optimiser ~5 % of the remaining budget (or 150 s max)
        time_left = deadline - time.time()
        if time_left > 5.0:
            adam_deadline = time.time() + min(150.0, 0.05 * time_left)
            a, M = _adam_optimize(a, target_sum, adam_deadline, max_iters=2_000_000)

        # ---- check if this length produced a new global best --------------------
        if M < best_val:
            best_val = M
            best_seq = a.copy()
            print(f"[global] new best M = {best_val:.6f} (length {n})")

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluation routine
    if best_seq is None:
        # as a fallback, return a simple uniform vector of length 300
        best_seq = np.full(300, math.sqrt(600) / 300.0, dtype=float)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best recorded peak = {best_val:.6f}, official score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
