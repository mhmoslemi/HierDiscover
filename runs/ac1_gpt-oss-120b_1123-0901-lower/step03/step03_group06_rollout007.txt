def _optimize_adam_length(n, p_start, deadline, max_iters=500_000):
    logp = np.log(p_start + 1e-30)
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    # Compute needed FFT size
    N = _next_pow2(3 * n - 2)
    best_score = float('inf')
    best_p = p_start.copy()
    it=0
    while time.time() < deadline and it < max_iters:
        it += 1
        p = np.exp(logp - np.max(logp))
        p /= p.sum()
        # Compute convolution
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n} iter {it}] new local best = {cur_score:.6f}")
        # softmax weights
        beta = 0.1 + (20.0) * min(1.0, it / 50000) # e.g., up to 20
        w = np.exp(beta * (conv - M))
        # compute rev (correlation)
        rev = _weighted_rev_fft(p, w, N)  # shape n
        grad = 4.0 * n * rev
        # Adam step on logp
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = 0.5 / (4.0 * n)  # base step small
        delta = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        logp -= delta

        # jitter
        if it % 5000 == 0:
            logp += np.random.randn(n) * 0.001
    return best_p, best_score

"""
Propose a low‑peak self‑convolution step function.

Algorithm Overview
------------------
* **Scale invariance** – work with a probability vector `p` (sum = 1) and
  recover the actual heights `a = sqrt(2·n)·p`.  The evaluation value then
  equals `2·n·max(p∗p)`.
* **Exponentiated‑gradient / Adam** – we optimise `log(p)` on the simplex
  using an Adam‑style update.  The (soft) gradient of the smoothed maximum
  is obtained by a weighted correlation, which is computed with a single
  FFT per iteration.
* **Multi‑scale schedule** – start from a modest length, repeatedly
  up‑sample the current best probability vector and continue optimisation.
  Lengths up to 12 000 are explored (larger lengths give a finer
  asymptotic bound but are slower per iteration).
* **Annealed soft‑max** – the soft‑max temperature `β` grows from a tiny
  value to a larger one, slowly turning the smooth objective into the true
  `max`.
* **Jitter & diverse seeds** – periodic random jitter on `log(p)` and a
  few deterministic seeds (uniform, edge‑heavy, centre‑heavy) help to
  escape local minima.
* **Time‑aware iteration limits** – the number of inner iterations is
  adapted to the current length, guaranteeing that the whole search fits
  into the 1000 s budget.

The routine returns the best height sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Linear self‑convolution p∗p (real) using a single FFT of size N.
    Returns the first 2·len(p)‑1 entries.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i} for i=0..n‑1.
    This is a correlation of p with the weight vector w (length 2·n‑1).
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    # keep linear part
    conv_rev = conv_rev[: n + L - 1]
    # extract the segment corresponding to rev and reverse it
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int,
    beta_max: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of log‑probabilities for a fixed length ``n``.
    Returns the best probability vector (sum = 1) and its raw score
    ``2·n·max(p∗p)``.
    """
    # FFT size sufficient for both the self‑convolution and the weighted
    # correlation (needs length 3·n‑2).
    N = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(p_start)
    v = np.zeros_like(p_start)

    # initialise log‑probability (add tiny epsilon to avoid log(0))
    logp = np.log(p_start + 1e-30)

    best_score = float("inf")
    best_p = p_start.copy()
    last_improve = 0

    # soft‑max schedule
    beta_start = 0.05
    beta_ramp_len = max(1, int(0.4 * max_iters))

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ------- project logp back onto the simplex -------
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)      # stable exponentiation
        p = exp_shift / exp_shift.sum()

        # ------- convolution and current score -------
        conv = _full_convolution_fft(p, N)      # length 2·n‑1
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            last_improve = it
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ------- soft‑max weights (smooth max) -------
        beta = beta_start + (beta_max - beta_start) * min(1.0, it / beta_ramp_len)
        w = np.exp(beta * (conv - M))          # max(w)=1, all ≤1

        # ------- weighted correlation (sub‑gradient of smooth max) -------
        rev = _weighted_rev_fft(p, w, N)       # shape (n,)

        # gradient of the smoothed objective 2·n·M_β
        grad = 4.0 * n * rev

        # ------- Adam update on log‑probabilities -------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate – small enough for stability
        base_lr = 0.6 / (4.0 * n)
        delta = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        logp -= delta

        # ------- occasional jitter to avoid plateaus -------
        if it % 4000 == 0:
            logp += np.random.randn(n) * 0.001

        # early exit if no improvement for a long time
        if it - last_improve > 20000:
            break

    return best_p, best_score

def _candidate_initial_ps(n: int, best_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a small diversified set of initial probability vectors for length ``n``.
    """
    cand = []

    # 1. Upsampled previous best (or uniform if not available)
    if best_seq is not None:
        if best_seq.shape[0] == n:
            p = best_seq / math.sqrt(2.0 * n)
        else:
            # linear interpolation of the old probability vector
            p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])
            old_x = np.linspace(0.0, 1.0, p_old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            p = np.interp(new_x, old_x, p_old)
            p = np.maximum(p, 0.0)
        p /= p.sum()
        cand.append(p)

    # 2. Uniform
    cand.append(np.full(n, 1.0 / n))

    # 3. Edge‑heavy (more mass near the boundaries)
    inc = np.arange(1, n + 1, dtype=float)
    dec = np.arange(n, 0, -1, dtype=float)
    base = inc * dec
    p_edge = base ** (-0.5)          # exponent <0 concentrates mass at the ends
    p_edge /= p_edge.sum()
    cand.append(p_edge)

    # 4. Centre‑heavy (more mass near the centre)
    p_center = base ** (0.5)         # exponent >0 concentrates mass in the middle
    p_center /= p_center.sum()
    cand.append(p_center)

    return cand
