def _simulated_annealing(init_a, target_sum, deadline, max_iters=200_000):
    a = init_a.copy()
    n = a.shape[0]
    best = a.copy()
    best_M = _conv_max_fft(best)[0]
    cur_M = best_M

    # compute temperature schedule
    T_start = cur_M * 0.15 + 0.001  # small offset
    T_end = 1e-6

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1
        # linear schedule or exponential
        T = T_start * (T_end/T_start)**(iteration / max_iters)

        # random source index with positive mass
        i_src = np.random.randint(0, n)
        if a[i_src] <= 1e-12:
            continue
        i_tgt = np.random.randint(0, n-1)
        if i_tgt >= i_src:
            i_tgt += 1

        # delta magnitude
        delta = a[i_src] * np.random.rand() * 0.3  # up to 30%
        if delta <= 0:
            continue

        # propose
        cand = a.copy()
        cand[i_src] -= delta
        cand[i_tgt] += delta

        # ensure non-negative (source might become tiny)
        if cand[i_src] < 0:
            continue

        # maintain sum automatically
        M_cand,_ = _conv_max_fft(cand)
        diff = M_cand - cur_M
        if diff <= 0 or np.random.rand() < math.exp(-diff / T):
            a = cand
            cur_M = M_cand
            if M_cand < best_M:
                best_M = M_cand
                best = cand.copy()

        # occasional jitter: every 5000 iters
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            cand_jit = a + jitter
            cand_jit = np.maximum(cand_jit, 0.0)
            cand_jit = cand_jit * (target_sum / cand_jit.sum())
            M_j,_ = _conv_max_fft(cand_jit)
            if M_j < cur_M:
                a = cand_jit
                cur_M = M_j
                if M_j < best_M:
                    best_M = M_j
                    best = cand_jit.copy()
    return best, best_M

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* The objective after normalising the L1‑norm to √(2 n) is simply the
  maximum value of the full auto‑convolution (call it M).  The score
  returned by ``evaluate_sequence`` equals M, so we only have to minimise
  that peak.
* For a given length n we run a two‑phase optimiser:
    1. **Adam** – an exact sub‑gradient method for the piecewise‑smooth
       function M (the same sub‑gradient used by the reference solution,
       but with a few tuned hyper‑parameters and a tighter stopping
       condition).
    2. **Simulated annealing** – a cheap stochastic local‑search that
       repeatedly moves a small amount of mass from one index to another.
       Moves that lower M are always accepted; uphill moves are accepted
       with a temperature‑controlled probability, allowing the search to
       escape shallow local minima.
* Several problem sizes are explored (the original length and a few
  larger multiples).  For the original size we perform many more random
  restarts because it is the most promising size.
* A global best sequence is kept and mixed into new random starts
  (20 % of the mass) to guide the search.
* The whole routine stops as soon as the 1000 s budget (minus a small
  safety margin) is exhausted and returns the best sequence found.
"""

def _conv_max_fft(a: np.ndarray):
    """
    Compute the maximal value of the linear convolution a*a using FFT.
    Returns (max_value, full_convolution_array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    C = A * A                                     # element‑wise product
    conv = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return float(conv.max()), conv

def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of the max‑convolution at a.  ``active_idxs`` are the
    shift indices k where the convolution value is within a tiny tolerance
    of the current maximum.  The gradient is the average over those
    active shifts of the reversed vector a.
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx                         # partner index for each i
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5_000_000):
    """
    Adam optimiser on the exact sub‑gradient of M = max(conv(a,a)).
    The vector is projected onto the feasible set (a_i ≥ 0, Σa_i = target_sum)
    after every update.  The function returns the best vector seen and its
    corresponding max‑convolution value.
    """
    a = init_a.copy()
    # normalise to the required L1‑norm
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.04
    iteration = 0
    jitter_interval = 4000

    # keep the best seen so far
    best_vec = a.copy()
    best_val, _ = _conv_max_fft(best_vec)

    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # 1. convolution and active set
        M, conv = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # 2. sub‑gradient of the max term
        rev = _subgradient(a, active)

        # gradient of the whole objective  (scaling factor 2 n/(sum²) is 1)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # 3. Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4. projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # 5. occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # 6. keep the best solution
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[adam] len {n:5d} iter {iteration:7d} new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(init_a: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iters: int = 200_000):
    """
    Cheap stochastic local search with a temperature schedule.
    Randomly moves a fraction of the mass from one index to another.
    Accepts all improving moves and uphill moves with probability
    exp(-Δ/temperature).  Returns the best vector found and its M.
    """
    a = init_a.copy()
    n = a.shape[0]
    best = a.copy()
    best_M, _ = _conv_max_fft(best)
    cur_M = best_M

    # temperature schedule (geometric)
    T_start = max(1e-3, cur_M * 0.15)
    T_end = 1e-6

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1
        # geometric cooling
        T = T_start * (T_end / T_start) ** (iteration / max_iters)

        # pick source with some mass
        i_src = np.random.randint(0, n)
        src_mass = a[i_src]
        if src_mass <= 1e-12:
            continue
        # pick destination (different from source)
        i_tgt = np.random.randint(0, n - 1)
        if i_tgt >= i_src:
            i_tgt += 1

        # amount to transfer – up to 30 % of the source entry
        delta = src_mass * np.random.rand() * 0.30
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i_src] -= delta
        cand[i_tgt] += delta

        # evaluate candidate
        M_cand, _ = _conv_max_fft(cand)
        diff = M_cand - cur_M
        if diff <= 0 or np.random.rand() < math.exp(-diff / T):
            a = cand
            cur_M = M_cand
            if M_cand < best_M:
                best_M = M_cand
                best = cand.copy()

        # occasional gentle jitter to diversify the search
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            jittered = np.maximum(a + jitter, 0.0)
            jittered *= (target_sum / jittered.sum())
            M_j, _ = _conv_max_fft(jittered)
            if M_j < cur_M:
                a = jittered
                cur_M = M_j
                if M_j < best_M:
                    best_M = M_j
                    best = jittered.copy()

    return best, best_M

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget (seconds) and returns the best
    sequence discovered within that budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the outer harness
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a previously found construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        print("[info] loaded previous construction, length", base.shape[0])
    except Exception:
        # fall back to a simple uniform vector
        base = np.full(2100, 1.0, dtype=float)
        print("[info] no previous construction – using uniform start")

    n0 = base.shape[0]
    target0 = math.sqrt(2.0 * n0)
    if base.sum() > 0:
        base = base * (target0 / base.sum())
    else:
        base = np.full_like(base, target0 / n0)

    # ------------------------------------------------------------------
    # Global best tracker
    best_seq = base.copy()
    best_M, _ = _conv_max_fft(best_seq)
    print(f"[init] length={n0} initial M={best_M:.6f}")

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (original and a few larger multiples)
    length_factors = [1.0, 1.2, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({
        max(300, min(15000, int(n0 * f))) for f in length_factors
    })
    if n0 not in lengths:
        lengths.insert(0, n0)

    print("[info] exploring lengths:", lengths)

    # ------------------------------------------------------------------
    # Main optimisation loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # allocate more restarts to the original size (it is usually the best)
        if n == n0:
            restarts = 10
        elif n <= 4000:
            restarts = 6
        else:
            restarts = 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ------------------- initialise --------------------------------
            if n == n0 and r == 0:
                init = best_seq.copy()                       # start from global best
            else:
                # symmetric random start (helps because convolution is symmetric)
                half = (n + 1) // 2
                half_vec = np.random.exponential(scale=1.0, size=half)
                if n % 2 == 0:
                    init = np.concatenate([half_vec, half_vec[::-1]])
                else:
                    init = np.concatenate([half_vec, half_vec[-2::-1]])
                init = init * (target_sum / init.sum())

                # blend a bit of the current global best if sizes match
                if best_seq.shape[0] == n:
                    mix = 0.15
                    init = mix * best_seq + (1.0 - mix) * init
                    init = np.maximum(init, 0.0)
                    init = init * (target_sum / init.sum())

            print(f"[run] length={n} restart={r + 1}/{restarts}")

            # ------------------- Adam optimisation -------------------------
            # give Adam a modest slice of the remaining budget (max 30 s)
            time_left = deadline - time.time()
            adam_budget = min(30.0, time_left * 0.4)
            adam_deadline = time.time() + adam_budget
            opt_seq, opt_M = _adam_optimize(init, n, target_sum,
                                            adam_deadline, max_iters=4_000_000)

            # ------------------- Simulated annealing ----------------------
            if time.time() < deadline:
                # use whatever time is still left for the SA phase
                sa_deadline = deadline
                opt_seq, opt_M = _simulated_annealing(opt_seq, target_sum,
                                                      sa_deadline, max_iters=200_000)

            # ------------------- update global best -----------------------
            if opt_M < best_M:
                best_M = opt_M
                best_seq = opt_seq.copy()
                print(f"[global] new best M={best_M:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Final evaluation through the official function (sanity check)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best reported M={best_M:.6f}, official score={final_score:.6f}, "
          f"length={best_seq.shape[0]}")
    return best_seq.tolist()
