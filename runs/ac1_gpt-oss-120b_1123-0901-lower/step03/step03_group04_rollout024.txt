"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The algorithm is a hybrid of a smoothed Adam optimiser and a cheap
stochastic local search.
It differs from the previous submission in three ways:

1. **Exact sub‑gradient** – the derivative of a convolution entry
   ``c[k] = Σ a[i]·a[k‑i]`` is ``2·a[k‑i]`` (or ``a[i]`` on the
   diagonal).  Using the exact factor gives a more faithful descent
   direction than the average that was used before.

2. **Annealed smoothing** – early iterations use a soft‑max of the
   convolution (temperature ``α``) which yields a smooth approximation
   of the ``max``.  ``α`` is increased slowly so that the optimiser
   first sees a gentle landscape and later focuses on the true peak.

3. **More aggressive exploration** – each optimisation run is followed
   by a longer stochastic “mass‑swap” local search, and a small amount
   of Gaussian jitter is injected periodically.

The routine explores several problem sizes (the original length and a
few larger multiples) and keeps the best sequence found before the
time budget (≈1000 s) is exhausted.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Return the *full* linear convolution a*a using an FFT."""
    n = a.shape[0]
    # Next power of two that can hold the full convolution (2*n-1 entries)
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Maximum value of the convolution and the convolution itself."""
    conv = _conv_full(a)
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgrad_exact(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k Σ_i a_i a_{k‑i}
    (ignoring the normalisation factor).  For each active shift k we add
    the contribution ``2·a_{k‑i}`` to the gradient component i whenever
    ``k‑i`` lies inside the vector.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    # loop over the (few) active indices – normally a tiny set
    for k in active:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i = np.arange(i_low, i_high + 1)
        j = k - i                      # partner index
        grad[i] += 2.0 * a[j]          # derivative of a_i * a_j (both orders)

    # When the active set is empty the gradient is zero – this never happens
    # because the global maximum is always active.
    return grad


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Smoothed “max” weights: w_k ∝ exp(alpha·conv_k).  Larger ``alpha``
    makes the distribution concentrate on the true maximum.
    """
    # shift to avoid overflow
    m = conv.max()
    shifted = conv - m
    exp_vals = np.exp(alpha * shifted)
    total = exp_vals.sum()
    if total == 0.0:
        # numerical catastrophe – fall back to hard max
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        return w
    return exp_vals / total


def _subgrad_softmax(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of the peak.
    It is a weighted sum of the exact per‑shift gradients.
    """
    n = a.shape[0]
    w = _softmax_weights(conv, alpha)
    # compute weighted sum of contributions efficiently via convolution:
    # grad_i = 2 * Σ_k w_k * a_{k‑i}
    # This is exactly 2 * (a ⋆ w)[i] where ⋆ denotes cross‑correlation.
    # Using the identity  a ⋆ w = correlate(a, w, mode='full')
    # the value needed for i is the slice centred at offset len(w)-1.
    corr = np.correlate(a, w, mode='full')
    offset = len(w) - 1
    grad = 2.0 * corr[offset: offset + n]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5_000_000,
                   start_alpha: float = 0.02,
                   final_alpha: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the autocorrelation using Adam on a
    smoothed‑max objective.  ``start_alpha`` is the temperature at the
    beginning of the run; it is increased linearly to ``final_alpha``
    (annealing).  Returns the best vector seen and its raw maximum
    convolution value (i.e. the value that ``evaluate_sequence`` would
    return after normalisation).
    """
    a = init_a.copy()
    n = a.shape[0]

    # Normalise the L1‑norm to the requested value
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a[:] = target_sum / n

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08

    best = a.copy()
    best_val, _ = _conv_max(a)

    iteration = 0
    # linear schedule for the smoothing parameter α
    alpha_inc = (final_alpha - start_alpha) / max_iters if max_iters > 1 else 0.0

    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # 1. convolution and (soft) max
        M, conv = _conv_max(a)

        # Linear increase of the temperature
        alpha = start_alpha + iteration * alpha_inc

        # 2. (soft) sub‑gradient of the max term
        if alpha < 0.5:                              # cheap path for very soft regime
            grad_max = _subgrad_exact(a, np.where(conv == M)[0])
        else:
            grad_max = _subgrad_softmax(a, conv, alpha)

        # 3. complete gradient of the objective f = 2*n*max / (Σa)^2
        #    df = (2*n/Σa^2)·grad_max - (4*n*max/Σa^3)·1
        sum_a = a.sum()
        coeff = (2.0 * n) / (sum_a * sum_a)
        const = (4.0 * n * M) / (sum_a * sum_a * sum_a)
        grad = coeff * grad_max - const

        # --------------------------------------------------------------
        # 4. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5. Projection onto the feasible set (non‑negative, correct L1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a[:] = target_sum / n

        # --------------------------------------------------------------
        # 6. occasional jitter to push off plateaus
        if iteration % 3_000 == 0:
            jitter = np.random.randn(n) * 1e-4 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # --------------------------------------------------------------
        # 7. keep the best solution seen so far
        if M < best_val:
            best_val = M
            best = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best  M = {best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  target_sum: float,
                  steps: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search.
    Repeatedly move a tiny amount of mass from one random index to another
    (preserving the L1 norm).  The move is kept only if it improves the
    maximal convolution.
    """
    n = a.shape[0]
    best = a.copy()
    best_val, _ = _conv_max(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of the source mass
        delta = best[i] * 0.015
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is not needed – the move preserves non‑negativity
        val, _ = _conv_max(cand)
        if val < best_val:
            best = cand
            best_val = val

    return best, best_val
