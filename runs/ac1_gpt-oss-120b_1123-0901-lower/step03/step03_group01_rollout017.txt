def _full_convolve_fft(x, y):
    # returns linear convolution via FFT, shape len(x)+len(y)-1
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n-1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X*Y, n=n_fft)[:n]
    return conv

def adam_optimize(a_init, target_sum, max_seconds, start_time):
    a = a_init.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.2  # base learning rate
    t = 0
    alpha = 1.0
    best_a = a.copy()
    best_score = _max_conv(a)
    # loop
    while time.time() < start_time + max_seconds:
        t += 1
        # compute conv
        conv = _conv_full_fft(a)
        max_conv = conv.max()
        # evaluate occasionally
        if t % 50 == 0:
            if max_conv < best_score:
                best_score = max_conv
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam inner best = {best_score:.6f} (n={n})")

        # update alpha schedule: slowly increase to focus on max
        if t % 100 == 0:
            alpha = min(alpha * 1.2, 80.0)  # multiply gradually up to 80

        # compute weights
        # shift for numerical stability
        exp_scaled = np.exp(alpha * (conv - max_conv))
        w = exp_scaled / exp_scaled.sum()

        # gradient via convolution with w
        a_rev = a[::-1]
        conv_grad_full = _full_convolve_fft(w, a_rev)
        grad = 2.0 * conv_grad_full[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a - step
        # projection
        a = _project_onto_simplex(a, target_sum)

    # final evaluation after loop
    final_score = _max_conv(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

import time

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution of a."""
    return float(_conv_full_fft(a).max())


def _adam_optimize(a_init: np.ndarray,
                   target_sum: float,
                   max_seconds: float,
                   start_time: float):
    """Adam optimisation of the smoothed max‑convolution objective."""
    a = a_init.copy()
    n = a.shape[0]

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing parameter α (starts small, grows)
    alpha = 1.0

    best_a = a.copy()
    best_score = _max_conv(a)

    t = 0
    while time.time() < start_time + max_seconds:
        t += 1

        # convolution of the current iterate
        conv = _conv_full_fft(a)
        cur_max = conv.max()

        # occasional exact evaluation
        if t % 80 == 0:
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam inner best = {best_score:.6f} (n={n})")

        # increase α slowly – this is the “annealing” of the smooth surrogate
        if t % 200 == 0:
            alpha = min(alpha * 1.25, 80.0)

        # soft‑max weights for the smoothed max
        # subtract cur_max for numerical stability (weights sum to 1)
        w_raw = np.exp(alpha * (conv - cur_max))
        w = w_raw / w_raw.sum()

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        grad_full = _full_convolve_fft(w, a_rev)            # length 3·n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]            # slice of size n

        # Adam update (descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, target_sum)

        # small random perturbation to escape shallow plateaus
        if t % 500 == 0:
            noise = np.random.randn(n) * (target_sum * 1e-3 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    final_score = _max_conv(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

Adam‑based smoothed‑max optimisation for the step‑function problem.

The algorithm keeps the ℓ₁‑norm fixed (Σ a_i = √(2n)) by Euclidean
projection onto the simplex.  It minimises the smooth surrogate

    f_α(a) = (1/α)·log Σ_i exp(α·(a∗a)_i)

using the Adam optimiser.  As α grows the surrogate approaches the
original objective 2·n·max(a∗a)/(Σa)², so α is increased slowly during
the run (annealing).  Several restarts with different lengths and
