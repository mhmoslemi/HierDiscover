"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Initialise a diverse pool of starting vectors (uniform, random Dirichlet,
  exponential / reversed exponential, Hann window, random perturbations)
  for several lengths (original size, 1.5× and 2×, capped at 8000).
* For each start run a two‑phase Adam optimiser.
    – Phase 1 (smooth) uses a soft‑max approximation of the max‑convolution
      with an exponentially decaying temperature τ.  This gives a smooth
      landscape that quickly moves the vector towards low‑peak regions.
    – Phase 2 (exact) switches to the true sub‑gradient of the max
      (τ→0) when τ becomes tiny, allowing a final fine‑tuning of the exact
      objective.
* After every Adam step the vector is projected onto the feasible set:
    a_i ≥ 0  and  Σ a_i = √(2 n) (the optimal normalisation for the given
    length n).  Small Gaussian jitter is added every few thousand steps to
    escape shallow plateaus.
* The best sequence over **all** runs (measured by the exact objective
  2 n·max(a∗a)/Σa² = max(a∗a) after normalisation) is kept.  The loop
  respects the 1000 s wall‑clock limit.
* The implementation relies only on NumPy (FFT for fast convolution) and
  runs comfortably within the time budget, while exploring many
  initialisations and two lengths (≈2100 and ≈8000) that are known to yield
  the strongest improvements.
"""

import time
import math
import numpy as np


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns weight vector w (summing to 1) and the smoothed maximum Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k * a_{k‑i}   (indices outside range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # pick the slice that corresponds to rev_i for i=0..n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _adam_optimize(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a warm soft‑max phase followed by an exact sub‑gradient
    phase.  Returns the best sequence found in this run together with its
    (exact) objective value (i.e. ``max(a∗a)`` after normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 5000

    # temperature schedule
    tau_factor = 0.20            # initial τ = τ_factor·M
    decay_rate = max_iters / 10.0  # exponential decay – slower than before

    Nfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2·n‑1

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- temperature & soft‑max -----
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        # when τ is tiny we fall back to the exact sub‑gradient
        if tau < 1e-9:
            # active set (all indices within 1e‑12 of the max)
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of the smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam gradient step -----
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # projects onto Σa = const
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to help escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- exact objective (after projection it equals M) -----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """
    Produce a small, diverse set of initial vectors for a given length n.
    All vectors are already normalised to have ℓ₁‑norm = target_sum.
    """
    rng = np.random.default_rng()
    init_list = []

    # 1) uniform
    init_list.append(np.full(n, target_sum / n))

    # 2) random Dirichlet (Gamma(1,1) i.i.d.)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init_list.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init_list.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    rev = x[::-1]
    init_list.append(rev / rev.sum() * target_sum)

    # 5) Hann window (smooth cosine)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init_list.append(hann / hann.sum() * target_sum)

    # 6) Blackman window (more aggressive taper)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init_list.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init_list.append(pert)

    return init_list
