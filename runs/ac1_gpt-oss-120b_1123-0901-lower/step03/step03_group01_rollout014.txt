def adam_optimize(a_init, target_sum, max_time, start_time):

    a = a_init.copy()

    m = np.zeros_like(a)

    v = np.zeros_like(a)

    beta1=0.9; beta2=0.999; eps=1e-8

    t = 0

    best_vec = a.copy()

    best_score = _max_conv(a)

    while time.time() < start_time + max_time:

        conv = _conv_full_fft(a)

        max_val = conv.max()

        if max_val < best_score:

            best_score = max_val

            best_vec = a.copy()

            print(...)

        # compute subgradient of max conv

        eps_t = 1e-12

        active = np.where(conv >= max_val - eps_t)[0]

        grad = np.zeros_like(a)

        for k in active:

            lo = max(0, k - (n - 1))

            hi = min(n - 1, k)

            idx = np.arange(lo, hi+1, dtype=int)

            partner = k - idx

            grad[idx] += 2.0 * a[partner]

        # ADAM update

        t += 1

        m = beta1*m + (1-beta1)*grad

        v = beta2*v + (1-beta2)*(grad**2)

        m_hat = m/(1-beta1**t)

        v_hat = v/(1-beta2**t)

        step = -0.02 * m_hat / (np.sqrt(v_hat) + eps)   # base step param

        a_new = a + step

        a_new = np.maximum(a_new, 0.0)

        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random noise
        if np.random.rand() < 0.01:
            a_new = a_new + np.random.randn(len(a)) * (target_sum / len(a) * 0.01)
            a_new = np.maximum(a_new,0.0)
            a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new

    return best_vec, best_score

"""
Optimisation of a step function minimizing 2*n*max(conv)/sum^2.

The algorithm performs many restarts with different lengths.
For each length a parametric U‑shaped family a_i = (i+1)^{-p}+ (n-i)^{-p}
(or exponential) is scanned, the best member is projected onto the simplex
Σa = √(2n) and then refined by a projected Adam‑style sub‑gradient method
that follows the active convolution entries.  Random Gaussian shakes are
added from time to time to escape shallow local minima.  The global best
sequence found before the time budget expires is returned.
"""
import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    # find the last i where rho_arr[i] > 0
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    # choose power‑of‑2 length for speed
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv

def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_conv_full_fft(a).max())

def _make_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped power law a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    idx = np.arange(n, dtype=float)
    w = (idx + 1.0) ** (-p) + (n - idx) ** (-p)
    return w

def _make_exp_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped exponential a_i ∝ exp(-p*i) + exp(-p*(n-1-i))."""
    idx = np.arange(n, dtype=float)
    w = np.exp(-p * idx) + np.exp(-p * (n - 1 - idx))
    return w

def _initial_candidate(n: int) -> np.ndarray:
    """Return a good starting vector for length n."""
    # try a handful of parametric shapes and keep the best.
    target_sum = np.sqrt(2.0 * n)
    best_vec = None
    best_val = np.inf

    # power‑law parameters
    power_params = [0.05, 0.1, 0.2, 0.4, 0.8, 1.2, 2.0, 3.0, 5.0, 8.0]
    for p in power_params:
        w = _make_power_shape(n, p)
        a = _project_onto_simplex(w, target_sum)
        val = _max_conv(a)
        if val < best_val:
            best_val = val
            best_vec = a

    # exponential parameters
    exp_params = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0, 4.0]
    for p in exp_params:
        w = _make_exp_shape(n, p)
        a = _project_onto_simplex(w, target_sum)
        val = _max_conv(a)
        if val < best_val:
            best_val = val
            best_vec = a

    # fallback to a tiny random perturbation around the best
    # (helps later optimisation to escape exact ties)
    if best_vec is not None:
        noise = np.random.randn(n) * (target_sum / n) * 0.001
        a = _project_onto_simplex(np.maximum(best_vec + noise, 0.0), target_sum)
        return a

    # very unlikely – uniform vector as ultimate fallback
    return np.full(n, target_sum / n, dtype=float)

def _adam_optimize(a_init: np.ndarray, target_sum: float,
                   max_time: float, start_time: float):
    """Projected Adam‑style sub‑gradient optimisation of max convolution."""
    a = a_init.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.025

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_vec = a.copy()
    best_score = _max_conv(a)

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_vec = a.copy()
            print(f"[{time.time():.2f}] Adam new best = {best_score:.6f} (n={n})")

        # Active set (indices attaining the maximum up to numerical tolerance)
        eps_act = 1e-12
        active = np.where(conv >= max_val - eps_act)[0]

        # Sub‑gradient of the sum of active convolution entries
        grad = np.zeros_like(a)
        for k in active:
            # i runs over all indices that contribute to conv[k]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = -lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a + step
        # enforce non‑negativity and the simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional stochastic shake to avoid stagnation
        if np.random.rand() < 0.015:
            shake = np.random.randn(n) * (target_sum / n) * 0.02
            a_new = _project_onto_simplex(np.maximum(a_new + shake, 0.0), target_sum)
            # reduce learning rate temporarily after a shake
            lr0 *= 0.9

        a = a_new
        # slowly decay learning rate as optimisation proceeds
        lr0 *= 0.99997

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    2 * n * max(conv(a, a)) / (sum(a) ** 2).

    The function runs many restarts with different lengths, builds an
    initial guess from a parametric U‑shaped family, refines the guess by
    a projected Adam optimiser and keeps the globally best sequence.
    The returned list will be evaluated by the external ``evaluate_sequence``
    routine.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # keep a small safety margin

    best_seq = None
    best_score = np.inf  # this stores the raw max‑convolution value

    # pre‑selected lengths that have historically yielded good results
    length_choices = [1200, 1500, 1800, 2200, 2600, 3000, 3500, 4200, 5000, 6000]

    while time.time() < deadline:
        # choose a length – occasionally reuse the current best length
        if best_seq is not None and np.random.rand() < 0.15:
            n = len(best_seq)
        else:
            n = int(np.random.choice(length_choices))

        target_sum = np.sqrt(2.0 * n)

        # ---- initialise candidate -------------------------------------------------
        a0 = _initial_candidate(n)

        # quick evaluation of the raw initial point
        cur_val = _max_conv(a0)
        if cur_val < best_score:
            best_score = cur_val
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # ---- allocate time for this restart --------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # give up to 9 s for the refinement; if we are close to the deadline we take less.
        phase_time = min(9.0, remaining * 0.6)

        # ---- Adam optimisation ----------------------------------------------------
        a_opt, val_opt = _adam_optimize(a0, target_sum, phase_time, time.time())
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam best = {best_score:.6f} (n={n})")

        # ---- early exit if we have already crossed the target --------------------
        if best_score < 1.503:
            print("[INFO] Target reached, exiting early.")
            break

    # fallback – uniform distribution (should never be used)
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to plain Python list as required by the grader
    return best_seq.tolist()

"""
Optimisation of a step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine performs many restarts with different lengths.
For each length a fast parametric “U‑shaped’’ or monotone family
(a_i ∝ (i+1)^{-p} or a_i ∝ exp(-p·i)) is scanned, the best member
is projected onto the simplex Σa = √(2n) and then refined by a
projected Adam‑style sub‑gradient method that follows the active
(convolution‑maximal) entries.  Random Gaussian shakes are added
occasionally to escape shallow minima.  The globally best sequence
found before the time budget expires is returned as a plain Python
list – the grader will evaluate it with the supplied ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0, Σx_i = s}.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast convolution via real FFT (full linear convolution, length 2*n‑1).
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Return the maximum entry of the auto‑convolution of a.
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Parametric families for the initial guess.
# --------------------------------------------------------------------------- #
def _make_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped power law: (i+1)^{-p} + (n-i)^{-p}."""
    idx = np.arange(n, dtype=float)
    return (idx + 1.0) ** (-p) + (n - idx) ** (-p)


def _make_monotone_power_shape(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power law: (i+1)^{-p}."""
    idx = np.arange(n, dtype=float)
    return (idx + 1.0) ** (-p)


def _make_exp_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped exponential decay."""
    idx = np.arange(n, dtype=float)
    return np.exp(-p * idx) + np.exp(-p * (n - 1 - idx))


def _make_monotone_exp_shape(n: int, p: float) -> np.ndarray:
    """Monotone decreasing exponential decay."""
    idx = np.arange(n, dtype=float)
    return np.exp(-p * idx)


# --------------------------------------------------------------------------- #
# Build a good starting vector for a given length.
# --------------------------------------------------------------------------- #
def _initial_candidate(n: int) -> np.ndarray:
    """Return a promising initial vector of length n."""
    target_sum = np.sqrt(2.0 * n)

    best_vec = None
    best_val = np.inf

    # --- try a few parametric families ------------------------------------ #
    power_ps = [0.05, 0.1, 0.2, 0.4, 0.8, 1.2, 2.0, 3.0, 5.0, 8.0]
    for p in power_ps:
        w = _make_power_shape(n, p)
        a = _project_onto_simplex(w, target_sum)
        v = _max_conv(a)
        if v < best_val:
            best_val, best_vec = v, a

        w = _make_monotone_power_shape(n, p)
        a = _project_onto_simplex(w, target_sum)
        v = _max_conv(a)
        if v < best_val:
            best_val, best_vec = v, a

    exp_ps = [0.01, 0.03, 0.07, 0.12, 0.2, 0.5, 1.0, 2.0, 4.0]
    for p in exp_ps:
        w = _make_exp_shape(n, p)
        a = _project_onto_simplex(w, target_sum)
        v = _max_conv(a)
        if v < best_val:
            best_val, best_vec = v, a

        w = _make_monotone_exp_shape(n, p)
        a = _project_onto_simplex(w, target_sum)
        v = _max_conv(a)
        if v < best_val:
            best_val, best_vec = v, a

    # --- optionally restart from the globally known construction -----------
    src = globals().get("height_sequence_1")
    if src is not None:
        src = np.asarray(src, dtype=float)
        if src.size >= n:
            a = src[:n].copy()
        else:
            pad = np.random.rand(n - src.size) * 1e-3
            a = np.concatenate([src, pad])
        a = _project_onto_simplex(a, target_sum)
        v = _max_conv(a)
        if v < best_val:
            best_val, best_vec = v, a

    # --- add a tiny random jitter to break exact ties ----------------------
    if best_vec is not None:
        jitter = np.random.randn(n) * (target_sum / n) * 0.001
        best_vec = _project_onto_simplex(np.maximum(best_vec + jitter, 0.0), target_sum)
    else:
        # very unlikely – fall back to a uniform vector
        best_vec = np.full(n, target_sum / n, dtype=float)

    return best_vec


# --------------------------------------------------------------------------- #
# Projected Adam optimiser that follows the active (max‑valued) convolution
# entries.  A small random shake is added from time to time.
# --------------------------------------------------------------------------- #
def _adam_optimize(a_init: np.ndarray, target_sum: float,
                   max_time: float, start_time: float):
    a = a_init.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.025

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_vec = a.copy()
    best_score = _max_conv(a)

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_vec = a.copy()
            print(f"[{time.time():.2f}] Adam new best = {best_score:.6f} (n={n})")

        # active indices where convolution attains the maximum
        eps_act = 1e-12
        active = np.where(conv >= max_val - eps_act)[0]

        # sub‑gradient of the sum of the active entries
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = -lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a + step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake to escape plateaus
        if np.random.rand() < 0.015:
            shake = np.random.randn(n) * (target_sum / n) * 0.02
            a_new = _project_onto_simplex(np.maximum(a_new + shake, 0.0), target_sum)
            lr0 *= 0.9   # a brief learning‑rate drop after a shake

        a = a_new
        lr0 *= 0.99997   # very slow decay

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point required by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function defined in ``evaluate_sequence``.
    The function runs many restarts with varying lengths, builds an
    initial guess from a parametric family, refines it with a projected
    Adam optimiser and finally returns the best sequence found
    (as a plain Python list) before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the grader

    best_seq = None
    best_score = np.inf   # stores the raw max‑convolution value after normalisation

    # lengths that have historically shown good behaviour;
    # we also allow a few larger values if time permits.
    length_pool = [1200, 1500, 1800, 2200, 2600, 3000,
                   3500, 4200, 5000, 6000, 8000]

    while time.time() < deadline:
        # choose a length – occasionally reuse the length of the current best
        if best_seq is not None and np.random.rand() < 0.15:
            n = len(best_seq)
        else:
            n = int(np.random.choice(length_pool))

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------- #
        # Build an initial candidate for this length.
        # ------------------------------------------------------------------- #
        a0 = _initial_candidate(n)

        cur_val = _max_conv(a0)
        if cur_val < best_score:
            best_score = cur_val
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------- #
        # Time allocation for this restart.
        # ------------------------------------------------------------------- #
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate up to about 9 s for the local optimiser (or the fraction left)
        phase_time = min(9.0, remaining * 0.6)

        # ------------------------------------------------------------------- #
        # Refine the candidate with the Adam optimiser.
        # ------------------------------------------------------------------- #
        a_opt, val_opt = _adam_optimize(a0, target_sum, phase_time, time.time())
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam best = {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------- #
        # Early exit if the target bound has already been reached.
        # ------------------------------------------------------------------- #
        if best_score < 1.503:
            print("[INFO] Desired bound reached – terminating early.")
            break

    # ------------------------------------------------------------------- #
    # Fallback – a uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grading script.
    return best_seq.tolist()
