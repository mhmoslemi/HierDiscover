def _conv_two_fft(x, y):
    m = len(x); n = len(y)
    L = m + n - 1
    n_fft = 1 << ((L-1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:L]
    return conv

"""
Proposes a step‑function using a two‑phase optimisation:

1. Smooth‑gradient descent on a soft‑max approximation of the maximal
   convolution value.  The soft‑max temperature (parameter ``alpha``)
   is increased during the run, so the method starts with a smooth
   landscape and gradually focuses on the true L∞ objective.
   The update is a projected gradient step onto the simplex
   ``{a_i≥0, Σa_i=√(2n)}``.

2. A short projected sub‑gradient refinement (the same routine that
   was used in the reference implementation) to polish the final
   candidate.

Multiple restarts with different lengths ``n`` (≈1500–8000) and
different random seeds are performed until the time budget expires.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – used throughout the optimiser.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex {x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast convolution using real FFT (full length = len(a)+len(b)-1).
# --------------------------------------------------------------------------- #
def _conv_two_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution of ``x`` and ``y`` using a real FFT."""
    m = x.shape[0]
    n = y.shape[0]
    size = m + n - 1
    # next power of two for zero‑padding
    n_fft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:size]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution of ``a`` with itself (used for evaluation)."""
    return _conv_two_fft(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the self‑convolution of ``a``."""
    return float(_conv_self_fft(a).max())


# --------------------------------------------------------------------------- #
# Sub‑gradient optimiser (exact max‐convolution, same as the reference).
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Refine a vector using a projected sub‑gradient descent on the active set."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2          # learning rate factor
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad

        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Smooth‑gradient descent on a soft‑max approximation of the L∞ norm.
# --------------------------------------------------------------------------- #
def smooth_gradient_optimize(a_init: np.ndarray,
                             target_sum: float,
                             max_time: float,
                             start_time: float):
    """Projected gradient descent on a smoothed max‑convolution objective."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    # Parameters that evolve with time
    max_lr = 0.5 * target_sum / n          # aggressive start
    min_lr = 0.001 * target_sum / n        # final fine‑tune
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.time() < start_time + max_time:
        # Full self‑convolution and true (unsmoothed) max – used for book‑keeping.
        conv = _conv_self_fft(a)
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_vec = a.copy()
            print(f"[{time.time():.2f}] Smooth GD new best = {best_score:.6f} (n={n})")

        # ---- soft‑max weights ------------------------------------------------
        # linear interpolation of the smoothing factor α ∈ [5, 200]
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / max_time)
        alpha = 5.0 + 195.0 * frac   # start 5, finish 200

        # shift by the current maximum for numerical stability
        shifted = conv - cur_max
        w = np.exp(alpha * shifted)          # max‑entry becomes 1.0
        p = w / w.sum()                      # soft‑max probabilities

        # ---- gradient of the smoothed objective -----------------------------
        # grad_i = 2 * Σ_k p_k * a_{k-i}
        a_rev = a[::-1]
        full_grad = _conv_two_fft(p, a_rev)   # length = 3n‑2
        grad = 2.0 * full_grad[n - 1 : n - 1 + n]   # slice → 0…n‑1

        # ---- step size -------------------------------------------------------
        lr = max_lr * (1.0 - frac) + min_lr * frac

        # heavy‑ball momentum
        momentum = beta * momentum - lr * grad
        a_new = a + momentum

        # enforce feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake (helps escape shallow plateaus)
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            # slightly reduce the learning‑rate after a shake
            max_lr *= 0.95
            min_lr *= 0.95

        a = a_new

    return best_vec, best_score

    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    max_lr = 0.5 * target_sum / n          # aggressive start
    min_lr = 0.001 * target_sum / n        # fine‑tune end
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.time() < start_time + max_time:
        conv = _conv_self_fft(a)
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_vec = a.copy()
            print(f"[{time.time():.2f}] Smooth GD new best = {best_score:.6f} (n={n})")

        # ----- soft‑max (smooth maximum) ----------------------------------
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / max_time)
        alpha = 5.0 + 195.0 * frac          # α grows from 5 to 200

        shifted = conv - cur_max            # stability shift (max entry → 0)
        w = np.exp(alpha * shifted)         # w_max = 1
        p = w / w.sum()                     # soft‑max distribution

        # ----- gradient of the smoothed objective -------------------------
        a_rev = a[::-1]
        full_grad = _conv_two_fft(p, a_rev)           # length = 3n‑2
        grad = 2.0 * full_grad[n - 1 : n - 1 + n]      # slice for indices 0…n‑1

        # ----- step size --------------------------------------------------
        lr = max_lr * (1.0 - frac) + min_lr * frac

        # ----- heavy‑ball momentum update ---------------------------------
        momentum = beta * momentum - lr * grad
        a_new = a + momentum

        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake to jump out of shallow plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            max_lr *= 0.95
            min_lr *= 0.95

        a = a_new

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point required by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the grader

    best_seq = None
    best_score = float("inf")

    # Helper for quick evaluation (max convolution = true objective after normalisation)
    def _eval(a):
        return _max_conv(a)

    # ------------------------------------------------------------------- #
    # Outer loop – many restarts with varying sequence lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Pick a length; sometimes reuse the current best length.
        if best_seq is not None and np.random.rand() < 0.15:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 8001)   # 1.5 k … 8 k entries

        target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2n)

        # --------------------------------------------------------------- #
        # Initialise a candidate.
        # --------------------------------------------------------------- #
        # (a) U‑shaped deterministic seed.
        idx = np.arange(n, dtype=float)
        weight = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        init = np.random.rand(n) * weight

        # (b) Optionally seed from the previously discovered construction.
        if "height_sequence_1" in globals() and np.random.rand() < 0.4:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])

        a0 = _project_onto_simplex(init.astype(float), target_sum)

        cur_score = _eval(a0)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Phase 1 – smooth gradient descent (≈ half of the remaining time).
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase1 = min(12.0, time_left * 0.5)

        a_gd, score_gd = smooth_gradient_optimize(a0, target_sum, phase1, time.time())
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After smooth GD best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Phase 2 – short sub‑gradient polishing.
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase2 = min(6.0, time_left)

        a_sg, score_sg = subgradient_optimize(a_gd, target_sum, phase2, time.time())
        if score_sg < best_score:
            best_score = score_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f} (n={n})")

        # early exit if the target bound has already been beaten
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return as a plain Python list, as required by the grader.
    return best_seq.tolist()
