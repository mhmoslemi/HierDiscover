"""
Multi‑stage optimizer for the max‑convolution bound.

The algorithm repeatedly restarts with different lengths (1500–4000) and
initialisations.  Each restart consists of three phases:

1. Simulated annealing with random mass transfers (global exploration).
2. Projected sub‑gradient descent on the currently active convolution
   entries (fast local improvement).
3. Smoothed‑gradient descent: we minimise a soft‑max surrogate
   ϕ_α(a)=∑_k w_k·conv_k with w_k∝exp(α·conv_k).  As α grows the
   surrogate approaches the true maximum, allowing the use of a smooth
   gradient (computed via FFT) together with Adam‑style momentum.

A final greedy hill‑climb cleans up the best solution found.
All iterates are projected onto the simplex {a_i≥0, Σa_i=√(2n)} so the
objective equals max(conv(a,a)).  The best sequence discovered before the
budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution via real FFT (linear convolution).
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary real vectors via FFT."""
    n = len(a) + len(b) - 1
    n_fft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Return max entry of a*a convolution."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass transfers between two positions.
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    a = a_init.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    T0 = max(0.01, cur_score * 0.05)
    T_min = 1e-7

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        # move up to 40 % of the mass at i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the active convolution entries.
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.15
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # gradient of sum of active entries
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.001 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99996
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Smoothed‑gradient descent using a soft‑max surrogate and Adam.
# --------------------------------------------------------------------------- #
def smoothed_gradient_optimize(a_init: np.ndarray,
                               target_sum: float,
                               max_time: float,
                               start_time: float,
                               alpha0: float = 1.0,
                               alpha_max: float = 200.0,
                               decay: float = 1.01,
                               base_lr: float = 0.25):
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    alpha = alpha0

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_vec = a.copy()
            print(f"[{time.time():.2f}] SMGD new best = {best_score:.6f} (n={n})")

        # soft‑max weights (shifted by max for numerical stability)
        exp_term = np.exp(alpha * (conv - cur_max))
        w = exp_term / exp_term.sum()          # shape (2n‑1,)

        # gradient of Σ w_k·conv_k
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)        # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # slice of length n

        # Adam update (gradient descent)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new

        # slowly increase the smoothing parameter
        alpha = min(alpha_max, alpha * decay)

        # decay learning rate a little
        base_lr *= 0.99995

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Simple greedy local hill‑climbing (zero‑temperature annealing).
# --------------------------------------------------------------------------- #
def hill_climb_local(a_init: np.ndarray,
                     target_sum: float,
                     max_time: float,
                     start_time: float):
    a = a_init.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.02  # tiny transfer
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum unchanged → projection not needed except for rounding errors
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
    return a, cur_score


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------- #
    # Helper to evaluate a candidate (fast, using max(conv) directly).
    # ------------------------------------------------------------------- #
    def _eval_vec(v):
        return _max_conv(v)

    # ------------------------------------------------------------------- #
    # Outer restart loop
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # problem size selection
        if best_seq is not None and np.random.rand() < 0.25:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 4001)   # 1500‑4000

        target_sum = np.sqrt(2.0 * n)   # simplex radius

        # --------------------------------------------------------------- #
        # Initialise a candidate (try to reuse known good constructions)
        # --------------------------------------------------------------- #
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # U‑shaped random seed
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)
            init = np.random.rand(n) * weight

        a = _project_onto_simplex(init.astype(float), target_sum)

        cur_score = _eval_vec(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Allocate time for this restart (≈12 s split 3 s/3 s/5 s)
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break

        # Phase 1 – Simulated annealing
        phase = min(3.0, time_left * 0.3)
        a_sa, s_sa = simulated_annealing(a, target_sum, phase, time.time())
        if s_sa < best_score:
            best_score = s_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After SA best = {best_score:.6f} (n={n})")

        # Phase 2 – Sub‑gradient descent
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase = min(3.0, time_left * 0.3)
        a_gd, s_gd = subgradient_optimize(a_sa, target_sum, phase, time.time())
        if s_gd < best_score:
            best_score = s_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD best = {best_score:.6f} (n={n})")

        # Phase 3 – Smoothed gradient descent (Adam)
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase = min(5.0, time_left)
        a_sg, s_sg = smoothed_gradient_optimize(a_gd, target_sum,
                                                phase, time.time())
        if s_sg < best_score:
            best_score = s_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.2f}] After SMGD best = {best_score:.6f} (n={n})")

        # Optional greedy polishing if a bit of time remains
        time_left = deadline - time.time()
        if time_left > 1.0:
            a_hc, s_hc = hill_climb_local(best_seq, target_sum,
                                          min(2.0, time_left), time.time())
            if s_hc < best_score:
                best_score = s_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.2f}] After HC best = {best_score:.6f} (n={n})")

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
