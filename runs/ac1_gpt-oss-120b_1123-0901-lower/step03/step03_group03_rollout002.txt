```python
def softmax_weights(conv, tau):
    if tau <= 0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, conv.max()
    maxc = conv.max()
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau
```

"""
Propose a step‑function (non‑negative vector) that minimises the
evaluate_sequence objective.

Algorithm
---------
* Work with the probability vector p = a / Σa (so Σp = 1).  The
  objective after normalisation is proportional to max_k (p∗p)_k.
* Use a smooth approximation of the maximum:
        Lτ(p) = τ·log Σ exp((p∗p)_k / τ)
  whose gradient can be written in closed form using a weighted reverse
  convolution.
* Optimise Lτ with a limited‑memory BFGS (L‑BFGS‑B) on the unconstrained
  variables y with p_i = exp(y_i) / Σ exp(y).  This automatically enforces
  non‑negativity and Σp = 1.
* A continuation schedule gradually reduces the temperature τ (starting
  with a relatively large τ and multiplying by a factor <1 after each
  L‑BFGS run).  Small random jitter is added to y between stages to
  escape shallow basins.
* Several restarts (base solution, random Dirichlet, uniform) are tried
  within the 1000 s budget.  The best sequence found (according to the
  exact evaluate_sequence call) is returned.
"""

import time
import math
import numpy as np
from typing import List

# SciPy is allowed; use it if present, otherwise fall back to a simple Adam.
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns weight vector w (summing to 1) and smooth max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract max for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i}
    using FFTs (indices outside the support are ignored).
    """
    # FFT of a (zero‑padded to Nfft)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution of w_rev with a
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]          # keep the valid part
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int):
    """Full convolution p∗p (real FFT based). Returns array of length 2·n‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Returns (Lτ, ∇_y Lτ) for the given y (log‑probability parameters).
    """
    # p = softmax(y)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # stable
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient of Lτ w.r.t. p: 2 * rev
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # transform gradient to y (softmax Jacobian)
    dot = np.dot(p, grad_p)                 # scalar = Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float, best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """
    Repeatedly minimise the smooth maximum with decreasing temperature.
    Returns the best (y, score) found (score = exact evaluate_sequence value).
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.65          # shrink factor after each L‑BFGS stage
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ---- L‑BFGS on current temperature ---------------------------------
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # limit the inner optimisation to a modest number of function calls
        options = {'maxfun': 250, 'maxiter': 250, 'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)

        y = res.x

        # ---- evaluate exact objective ---------------------------------------
        # compute p, a and the exact max‑convolution
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # equivalent to evaluate_sequence

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ---- schedule updates ------------------------------------------------
        tau *= tau_decay

        # occasional jitter to leave plateaus
        if np.random.rand() < 0.15:
            y += np.random.randn(y.shape[0]) * 0.02

        # break early if time is exhausted
        if time.time() >= deadline:
            break

    return best_seq, best_score

    """

    deadline = start + budget_s - 5.0       # safety margin

    # ----------------------------------------------------------------------

    n = base.shape[0]

    base = base * (target_sum / base.sum())

    p0 = base / target_sum
    # avoid exact zeros – add a tiny epsilon before taking log
    eps_log = 1e-12

    # initialise best record
    best_seq = base.copy()

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ---- generate several starting points ---------------------------------
    starts = []

    # Convert to plain Python list as required by the evaluator.
