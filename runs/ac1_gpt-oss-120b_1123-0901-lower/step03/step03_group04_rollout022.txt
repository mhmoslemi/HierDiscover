"""
Search for a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* For several problem sizes (original length and a few scaled versions) run
  a short Adam optimiser that directly minimises the maximal value of the
  self‑convolution (the quantity returned by ``evaluate_sequence`` after
  normalising the L1‑norm to √(2 n)).
* After each Adam run perform a cheap greedy local‑search that transfers a
  small amount of mass between two random positions.
* Refine the result with a simulated‑annealing (SA) phase that works with
  an *incremental* update of the convolution, so each move costs O(n) only.
  The SA accepts uphill moves according to a linear cooling schedule,
  allowing it to escape local minima.
* Keep the best sequence seen over all lengths and restarts and return it
  before the time budget (≈1000 s) expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Return the maximal value of the full convolution a*a and the full
    convolution array (real).  Uses an FFT of length equal to the next
    power of two for speed.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A                                     # a * a in the frequency domain
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]      # keep only the valid part
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Compute a sub‑gradient of the max‑convolution at the current point.
    The gradient component for index i equals the average of a_{k-i}
    over all active shift positions k (out‑of‑range indices are ignored).
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx               # partner index
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5_000_000):
    """
    Minimise the peak of a*a using Adam on the exact sub‑gradient.
    Returns the best vector seen and the corresponding max‑convolution value.
    """
    a = init_a.copy()
    # normalise to the required L1‑norm
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.06                     # a slightly larger learning rate
    iteration = 0
    jitter_interval = 3000
    best_vec = a.copy()
    best_val, _ = _conv_max_fft(a)

    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # 1. convolution and active set
        M, conv = _conv_max_fft(a)

        # indices whose value is within a tiny tolerance of the current max
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # --------------------------------------------------------------
        # 2. sub‑gradient of the max term (including the sum‑constraint term)
        rev = _subgradient(a, active)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # --------------------------------------------------------------
        # 3. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 4. projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # --------------------------------------------------------------
        # 5. occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0015 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # --------------------------------------------------------------
        # 6. keep the best solution seen so far
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap stochastic local search: repeatedly move a tiny amount of mass
    from one random index to another if the move reduces the max‑convolution.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)
    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = best[i] * 0.02
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M
    return best, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         target_sum: float,
                         deadline: float):
    """
    Simulated‑annealing refinement that works with an incremental update of the
    convolution.  At each step a random amount of mass is transferred from
    index i to index j; the convolution is updated in O(n) time and the move
    is accepted according to a linear cooling schedule.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = np.convolve(cur_a, cur_a)          # length 2n‑1
    cur_val = cur_conv.max()
    best_a = cur_a.copy()
    best_val = cur_val

    start = time.time()
    # initial temperature proportional to the current excess over the lower bound 1
    init_temp = max(0.02, 0.2 * max(0.0, cur_val - 1.0))

    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if cur_a[i] <= 0.0:
            continue

        # move up to 30 % of the source mass
        frac = np.random.rand() * 0.3
        delta = cur_a[i] * frac
        if delta <= 0.0:
            continue

        # incremental convolution update (copy because we may reject)
        new_conv = cur_conv.copy()
        # contribution of the moved mass with the *old* vector
        new_conv[i:i + n] -= 2.0 * delta * cur_a
        new_conv[j:j + n] += 2.0 * delta * cur_a
        new_conv[2 * i] += delta * delta
        new_conv[2 * j] += delta * delta
        new_conv[i + j] -= 2.0 * delta * delta

        new_val = new_conv.max()

        # linear cooling
        elapsed = time.time() - start
        total = deadline - start
        temp = init_temp * max(0.0, 1.0 - elapsed / total)

        # Metropolis acceptance
        if (new_val < cur_val) or (np.random.rand() < np.exp(-(new_val - cur_val) / max(temp, 1e-12))):
            # accept the move
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_conv = new_conv
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()

        it += 1

        # occasional tiny jitter to keep the search fluid
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 0.0003 * target_sum / n
            cand = cur_a + jitter
            cand = np.maximum(cand, 0.0)
            cand *= (target_sum / cand.sum())
            cand_conv = np.convolve(cand, cand)
            cand_val = cand_conv.max()
            if cand_val < cur_val:
                cur_a = cand
                cur_conv = cand_conv
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best_a = cand.copy()

    return best_a, best_val
