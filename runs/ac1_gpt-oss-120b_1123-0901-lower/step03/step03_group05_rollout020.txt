def _annealed_grad_optimize(a_init, n, target_sum, deadline):
    a = a_init.copy()
    best_score = float('inf')
    best_a = a.copy()
    a_rev = a[::-1]
    # We'll pre-allocate FFT size for conv and gradient
    N_conv = 1 << ((2*n - 1).bit_length())
    N_grad = 1 << ((3*n - 2).bit_length())
    # hyperparameters
    beta = 0.5
    beta_max = 30.0
    eta = 1.0   # step size
    # schedule for beta increase
    beta_inc = 1.01
    # schedule for eta adapt maybe.
    iteration = 0
    while time.time() < deadline and iteration < 200000:
        iteration += 1
        # conv
        A = np.fft.rfft(a, n=N_conv)
        conv = np.fft.irfft(A * A, n=N_conv)[:2*n-1]
        M = conv.max()
        if M < best_score:
            best_score = M
            best_a = a.copy()
        # compute softmax weights
        # subtract M to stabilize
        w = np.exp(beta * (conv - M))
        w_sum = w.sum()
        p = w / w_sum
        # gradient via convolution of p with reverse a
        B = np.fft.rfft(p, n=N_grad)
        C = np.fft.rfft(a[::-1], n=N_grad)
        grad_full = np.fft.irfft(B * C, n=N_grad)
        # slice gradient: indices n-1 .. n-1 + n -1
        grad = 2.0 * grad_full[n-1:n-1+n]  # length n
        # update via exponentiated gradient for positivity
        # Option: additive update with projection.
        # Use adaptive learning rate
        step = eta / (np.linalg.norm(grad) + 1e-12)
        a_new = a - step * grad
        # clip and renorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)
        else:
            a_new = np.full_like(a_new, target_sum / n)
        a = a_new
        # increase beta gradually
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)
        # optionally decay eta
        if iteration % 5000 == 0:
            eta *= 0.9
        # occasional random jitter to escape plateaus
        if iteration % 10000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
    # after loop compute final score
    final_score = _conv_max_fft(a)[0]  # returns max
    # choose better of best_a and current a
    if final_score < best_score:
        best_a = a
        best_score = final_score

"""
Annealed soft‑max gradient descent for the max‑autoconvolution problem.

The objective (with the usual normalisation Σa = √(2n)) is simply the
maximum entry of the full convolution a*a.  This maximum is a piecewise‑
linear convex function of the vector a, and its sub‑gradient can be written
as a weighted sum of the reversed vectors that realise each shift.

The algorithm proceeds as follows:

1.  Choose a few candidate lengths (the original length and a few larger
    ones).  For each length initialise several vectors:
    •  a linear interpolation of the best known construction;
    •  a few random Dirichlet‑like starts.

2.  For each start run an *annealed* optimisation:
    *   Compute the full convolution by FFT.
    *   Form a soft‑max over the convolution entries with a temperature β.
        The weights w_k = exp(β·(c_k−max(c))) give a smooth approximation
        of the active‑set.  The gradient of the smoothed objective is
        g = 2·(w * a_rev), which can also be obtained by a single FFT
        convolution.
    *   Take a projected gradient step
            a ← a − η·g /‖g‖,
        enforce non‑negativity and renormalise to the required L1‑norm.
    *   Increase β (sharpen the max) and slowly decay η.
        Occasionally a tiny random jitter is added to escape plateaus.

    The loop stops when the global time budget is exhausted or a fixed
    iteration limit is reached.  The best vector (lowest max‑convolution)
    seen during the run is retained.

3.  A cheap coordinate‑swap hill‑climb refines the best candidate:
    repeatedly move a small amount of mass from a donor index to a receiver
    index and keep the move only if the maximal convolution value decreases.

The whole search respects the 1000 s wall‑clock limit; the best sequence
found (as a plain Python list) is returned by ``propose_candidate``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Full autoconvolution a*a using FFT and return the maximal value."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # power‑of‑2 ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _annealed_optimize(
    a_init: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 60_000,
    eta0: float = 0.6,
    beta0: float = 0.5,
    beta_max: float = 30.0,
):
    """
    Annealed soft‑max gradient descent.
    Returns the best vector found and its maximal convolution value.
    """
    a = a_init.copy()
    n = a.shape[0]

    # FFT sizes that will be reused
    N_conv = 1 << ((2 * n - 1).bit_length())
    N_grad = 1 << ((3 * n - 2).bit_length())   # length of p (2n‑1) + a_rev (n) –1

    # state for annealing
    beta = beta0
    eta = eta0
    best_a = a.copy()
    best_val = float("inf")
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        # ---- convolution -------------------------------------------------
        A = np.fft.rfft(a, n=N_conv)
        conv = np.fft.irfft(A * A, n=N_conv)[: 2 * n - 1]
        cur_max = float(conv.max())
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[anneal n={n:5d} it={iteration:6d}] new best max = {best_val:.6f}")

        # ---- soft‑max weights --------------------------------------------
        # work in the shifted domain to avoid overflow
        w = np.exp(beta * (conv - cur_max))          # max entry → weight 1
        w_sum = w.sum()
        if w_sum == 0.0:   # numerical safety
            w = np.ones_like(w) / w.shape[0]
            w_sum = 1.0
        p = w / w_sum                               # normalized weights

        # ---- gradient (2 * p * a_rev) ------------------------------------
        # convolution of p (length 2n‑1) with a_rev (length n)
        B = np.fft.rfft(p, n=N_grad)
        C = np.fft.rfft(a[::-1], n=N_grad)
        grad_full = np.fft.irfft(B * C, n=N_grad)
        # slice gives the vector of length n:
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- projected gradient step --------------------------------------
        gn = np.linalg.norm(grad) + 1e-14
        step = eta / gn
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)

        s = a_new.sum()
        if s > 0.0:
            a = a_new * (target_sum / s)
        else:
            a = np.full_like(a_new, target_sum / n)

        # ---- annealing schedule -------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta * 1.012)   # slowly sharpen the max
        if iteration % 5_000 == 0:
            eta *= 0.9                           # modest decay

        # occasional jitter to break plateaus
        if iteration % 12_000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    # final evaluation (may be slightly better than best_val)
    final_max, _ = _conv_max_fft(a)
    if final_max < best_val:
        best_val = final_max
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    target_sum: float,
    best_val: float,
    deadline: float,
    max_proposals: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: move a small amount of mass from a donor index
    to a receiver index and keep the change only if the maximal convolution
    value decreases.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_val = best_val
    proposal = 0

    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # keep feasibility
        new_max, _ = _conv_max_fft(new_a)
        if new_max < cur_val:
            a = new_a
            cur_val = new_max
            print(f"[local] improvement to {cur_val:.6f} (proposal {proposal})")
    return a, cur_val

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Uses several lengths, random restarts and an annealed soft‑max gradient
    descent followed by a cheap local search.  The best sequence found before
    the time budget expires is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # load the previously discovered construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore

        n0 = base.shape[0]
        target_sum0 = math.sqrt(2.0 * n0)
        base = base * (target_sum0 / base.sum())
        init_score = evaluate_sequence(base.tolist())
        best_seq = base.copy()
        best_score = init_score
        print(f"[init] length={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # set of lengths to explore – original plus enlargements
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100
    length_factors = [1.0, 1.3, 1.6, 2.0, 3.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    max_len = 12000
    lengths = [l for l in lengths if 300 <= l <= max_len]

    # ------------------------------------------------------------------
    # main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # number of restarts per length (interpolated + several random)
        restarts = 4
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise -------------------------------------------------
            if r == 0 and base is not None:
                # linear interpolation of the existing best onto the new grid
                old_x = np.linspace(0.0, 1.0, base.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, base)
                a0 = np.maximum(a0, 0.0)
            else:
                # random Dirichlet‑like start
                a0 = np.random.rand(n)
                a0 = np.maximum(a0, 0.0)

            a0 = a0 * (target_sum / a0.sum())

            # ----- annealed soft‑max gradient descent -----------------------
            # iteration budget adapts to the length (longer vectors get fewer iters)
            max_iter = 70_000 if n <= 3000 else 30_000
            a_opt, cur_val = _annealed_optimize(
                a_init=a0,
                target_sum=target_sum,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.8,
                beta0=0.4,
                beta_max=35.0,
            )

            # ----- cheap local refinement -----------------------------------
            if time.time() < deadline:
                a_opt, cur_val = _local_search(
                    a_start=a_opt,
                    target_sum=target_sum,
                    best_val=cur_val,
                    deadline=deadline,
                    max_proposals=20_000,
                )

            # ----- record best ------------------------------------------------
            if cur_val < best_score:
                best_score = cur_val
                best_seq = a_opt.copy()
                print(f"[global] new best: length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback – use a uniform vector if nothing was found (should never happen)
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
