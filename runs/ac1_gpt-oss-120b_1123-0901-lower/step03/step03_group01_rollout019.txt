"""
Minimum‑max‑convolution step‑function search.

The core idea
----------------
* Keep the ℓ₁‑norm Σ aᵢ = √(2 n) – with this normalisation the
  evaluation function reduces to max(conv(a, a)).
* Start from a cheap “U‑shaped” seed (large values at the ends,
  small in the middle) and optionally improve it by a short simulated‑annealing
  phase.
* **LP‑mix** – given the current vector f, solve the linear program

        maximise   Σ gⱼ
        subject to  gⱼ ≥ 0
                    (f * g)[k] ≤ M   for all k,
                    (where M = max(f * f))

  The solution g pushes mass towards locations that are “safe” w.r.t.
  the current maximal convolution value.  A convex combination
  `f←(1‑t) f + t g` (projected back onto the simplex) strictly reduces
  the maximal convolution in practice.  Repeating this step a few
  times yields a large drop in the objective.
* Finally a short projected sub‑gradient descent (the routine from the
  previous state‑of‑the‑art) polishes the result.

The whole loop restarts many times with different lengths (≈1200‑2500)
and keeps the best sequence found before the time budget expires.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean)
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution via FFT (real‑valued)
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Max of the full convolution (the value we actually minimise)
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Simulated annealing – a very quick global‑search phase (optional)
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    a = a_init.copy()
    n = a.shape[0]

    cur = _max_conv(a)
    best = a.copy()
    best_score = cur

    # 5 % of the current value as initial temperature, never below 1e‑2
    T0 = max(0.01, cur * 0.05)
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        # move a random fraction (≤50 %) from i to j
        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)

        new = _max_conv(a_new)
        d = new - cur

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        accept = False
        if d <= 0.0:
            accept = True
        else:
            if np.exp(-d / (T + 1e-12)) > np.random.rand():
                accept = True

        if accept:
            a = a_new
            cur = new
            if new < best_score:
                best_score = new
                best = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")

    return best, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the active convolution indices
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    a = a_init.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        mx = conv.max()
        if mx < best_score:
            best_score = mx
            best = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # active set – indices where conv is (almost) maximal
        eps = 1e-12
        act = np.where(conv >= mx - eps)[0]

        grad = np.zeros_like(a)
        for k in act:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        maxg = grad.max()
        if maxg <= 0.0:
            break

        eta = base_lr * mx / maxg
        momentum = beta * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best, best_score


# --------------------------------------------------------------------------- #
# LP‑mix phase – linear‑programming based improvement
# --------------------------------------------------------------------------- #
def _build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A where
        A[k, i] = f[i]   if 0 ≤ k‑i < n
                 = 0    otherwise.
    This is exactly the matrix that encodes (f * g)[k] = Σ_i A[k,i]·g[i].
    """
    n = f.shape[0]
    # each diagonal offset i contains the constant value f[i] repeated n times
    diag_vals = np.tile(f[:, np.newaxis], (1, n))
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = scipy.sparse.diags(diag_vals, offsets,
                           shape=(2 * n - 1, n),
                           format="csr")
    return A


def lp_mix(a_init: np.ndarray,
           target_sum: float,
           max_time: float,
           start_time: float):
    """
    Repeatedly solve the LP
          maximise Σ g_j
          s.t.   g_j ≥ 0
                 (a * g)[k] ≤ M   for all k
          where M = max(a * a).
    The obtained g is mixed back into a (convex combination) and the
    whole process is repeated until time expires.
    """
    a = a_init.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_conv(a)

    # start with a fairly aggressive mixing weight
    mix = 0.5

    while time.time() < start_time + max_time:
        M = _max_conv(a)

        # build the constraint matrix for the current a
        A = _build_conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # LP: minimise -Σ g_j   → maximise Σ g_j
        c = -np.ones(n, dtype=np.float64)

        # solve
        lp_res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "verbosity": 0}
        )

        if not lp_res.success:
            # fall back to the current best – LP failed (rare)
            print(f"[{time.time():.2f}] LP failure: {lp_res.message}")
            break

        g = lp_res.x
        # enforce the simplex (sum = target_sum)
        g = _project_onto_simplex(g, target_sum)

        # convex combination and re‑project to the simplex
        a_new = (1.0 - mix) * a + mix * g
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)

        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            print(f"[{time.time():.2f}] LP‑mix new best = {best_score:.6f} (n={n})")

        # adapt mixing weight – if we made progress, be a bit bolder,
        # otherwise become more conservative.
        if new_score < M:
            a = a_new
            mix = min(mix * 1.07, 0.9)
        else:
            mix = max(mix * 0.6, 0.05)

        # safety check for the global deadline
        if time.time() >= start_time + max_time:
            break

    return best, best_score
