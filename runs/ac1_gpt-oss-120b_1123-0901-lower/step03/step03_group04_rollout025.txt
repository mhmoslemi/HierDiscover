def _conv_full_fft(a, b):
    n = a.shape[0]
    m = b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)
    return conv[:n+m-1]

def propose_candidate(seed=42, budget_s=1000, **kwargs):

        else:

        return vec

        target_sum = math.sqrt(2.0 * n)

                break

            if n == n0 and r == 0:

            else:

                init = np.clip(init, 0.0, None)
                init = normalize(init, target_sum)
            print(f"[run] len={n} restart={r+1}/{restarts}")

            # allocate time for optimizer
            time_left = deadline - time.time()
            sub_deadline = time.time() + min(200.0, time_left * 0.5)

            # softmax optimizer
            opt_seq, opt_val = _adam_optimize_softmax(init, n, target_sum, sub_deadline)

            if time.time() < deadline:
                # run a few thousand iterations of subgradient to polish

                if opt_val2 < opt_val:

The objective of `evaluate_sequence` is  

\[
\text{score}= \frac{2\,n\;\max_k (a*a)_k}{(\sum_i a_i)^2},
\]

and the function automatically rescales every candidate to the norm  

\[
\sum_i a_i = \sqrt{2n}.
\]

After this normalisation the score is simply  

\[
\text{score}= \max_k (a*a)_k .
\]

Hence we have to **minimise the peak of the aperiodic autocorrelation** of a
non‑negative vector with a fixed \(\ell_1\)‑norm.

The code below combines three complementary optimisation ideas

1. **Soft‑max smoothed gradient descent** – the non‑smooth `max` is replaced
   by a smooth log‑sum‑exp approximation.  The gradient can be computed
   efficiently with two FFT‑convolutions.
2. **Projected Adam** – the smoothed gradient is fed to an Adam optimiser
   together with a projection onto the feasible set
   (`a_i ≥ 0` and `∑a_i = √(2n)`).
3. **Optional linear‑programming refinement** – a linear programme
   (solved with `cvxpy`) maximises the total mass of a partner vector `g`
   under the same peak‑constraint.  A convex combination of the current
   vector and the LP‑solution provably does not increase the peak.
4. **Lightweight local search** – after the continuous optimisation a
   greedy swap of a tiny amount of mass between two positions is performed
   to escape tiny plateaus.

The search is performed on several problem sizes (the original length and a
few larger multiples).  Time is split dynamically among the lengths; the
best vector seen so far is kept globally and returned at the end of the
budget (≈ 1000 s).  All operations are `numpy`‑based, the FFT‑based convolutions
run in \(\mathcal{O}(n\log n)\) time and easily fit the time limit.


```python
"""
Propose a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The routine combines a smoothed soft‑max gradient
descent (Adam optimiser), an optional CVXPY linear‑programming refinement,
and a cheap stochastic local search.  Several lengths are explored and the
best vector found before the time budget expires is returned.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# cvxpy is optional – if unavailable the algorithm works without it.
try:
    import cvxpy as cp
except Exception:                     # pragma: no cover
    cp = None

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full discrete convolution of two real vectors using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n, m = a.shape[0], b.shape[0]
    N = 1 << ((n + m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)
    return conv[: n + m - 1]


def _conv_max_fft(a: np.ndarray):
    """
    Convolution of a with itself and the maximal entry.
    Returns (max_value, full_convolution_array).
    """
    conv = _conv_full_fft(a, a)
    return float(conv.max()), conv


# ----------------------------------------------------------------------
# Normalisation utilities
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projects a non‑negative vector onto the simplex {x≥0, Σx = target_sum}.
    """
    s = v.sum()
    if s > 0.0:
        v *= (target_sum / s)
    else:
        v[:] = target_sum / v.shape[0]
    return v


# ----------------------------------------------------------------------
# Soft‑max based Adam optimiser
# ----------------------------------------------------------------------
def _adam_optimize_softmax(init_a: np.ndarray,
                           n: int,
                           target_sum: float,
                           deadline: float,
                           alpha_start: float = 0.5,
                           alpha_end: float = 12.0,
                           max_iters: int = 10_000_000):
    """
    Adam optimiser for the smoothed max objective.
    Returns the best vector seen and its true peak (max convolution).
    """
    a = _normalise(init_a.astype(float).copy(), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.04

    best_vec = a.copy()
    best_val = float('inf')
    iteration = 0

    while time.time() < deadline and iteration < max_iters:
        # ------------------------------------------------------------------
        # current objective and convolution
        M, conv = _conv_max_fft(a)

        # ------------------------------------------------------------------
        # soft‑max weights (stable: shift by the maximum)
        if iteration < 2000:
            # linear warm‑up of the temperature
            alpha = alpha_start + (alpha_end - alpha_start) * (iteration / 2000.0)
        else:
            alpha = alpha_end
        exp_vals = np.exp(alpha * (conv - M))       # max = 0 → exp ≤ 1
        w = exp_vals / exp_vals.sum()               # shape (2n‑1,)

        # ------------------------------------------------------------------
        # gradient of the smoothed max (analytical, see write‑up)
        a_rev = a[::-1]                             # reversed a
        grad_full = _conv_full_fft(w, a_rev)        # length 3n‑2
        # slice corresponds to ∂/∂a_i
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ------------------------------------------------------------------
        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # projection onto feasible set
        a = np.maximum(a, 0.0)
        a = _normalise(a, target_sum)

        # occasional jitter to shake off plateaus
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = _normalise(a, target_sum)

        # ------------------------------------------------------------------
        # keep best (according to the *true* maximum)
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[softmax] len {n:5d} it {iteration:7d}  new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
# Original sub‑gradient Adam optimiser (kept as a fine‑tuner)
# ----------------------------------------------------------------------
def _adam_optimize_subgrad(init_a: np.ndarray,
                           n: int,
                           target_sum: float,
                           deadline: float,
                           max_iters: int = 10_000_000):
    """
    Adam optimiser that uses the exact sub‑gradient of the max‑convolution.
    Returns the best vector seen and its peak.
    """
    a = _normalise(init_a.astype(float).copy(), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.04

    best_vec = a.copy()
    best_val, _ = _conv_max_fft(best_vec)
    iteration = 0
    jitter_interval = 4000

    while time.time() < deadline and iteration < max_iters:
        # convolution and active set
        M, conv = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max term (average over active shifts)
        idx = np.arange(n)
        rev = np.zeros_like(a)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        if active.size > 0:
            rev /= float(active.size)

        # gradient of the full objective (the scaling factor 2·n/(sum²) is 1)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        a = _normalise(a, target_sum)

        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = _normalise(a, target_sum)

        # best‑track
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[subgrad] len {n:5d} it {iteration:7d} new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
# Very cheap stochastic local search (greedy)
# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Repeatedly move a small fraction of mass from one index to another
    if the maximal convolution value decreases.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = 0.02 * best[i]               # move 2 % of the source mass
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is preserved, only non‑negativity may be violated
        cand[i] = max(cand[i], 0.0)

        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M

    return best, best_val


# ----------------------------------------------------------------------
# Optional linear‑programming refinement (cvxpy)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               n: int,
               target_sum: float,
               M: float,
               max_val: float = 1000.0):
    """
    Solve   max Σ g_j   s.t.   a * g ≤ M (element‑wise) , 0 ≤ g_j ≤ max_val.
    The solution g is returned after rescaling to the required ℓ₁‑norm.
    Returns None if cvxpy is unavailable or the problem is infeasible.
    """
    if cp is None:                         # pragma: no cover
        return None

    g = cp.Variable(n)
    f_param = cp.Parameter(n, value=a)     # constant first factor
    conv = cp.conv(f_param, g)             # length 2n‑1

    constraints = [
        conv <= M,
        g >= 0,
        g <= max_val
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    except Exception:                     # pragma: no cover
        return None

    if g.value is None:
        return None

    g_val = np.asarray(g.value, dtype=float)
    # rescale to the required ℓ₁‑norm
    s = g_val.sum()
    if s > 0.0:
        g_resc = g_val * (target_sum / s)
    else:
        g_resc = np.full_like(g_val, target_sum / n)

    return g_resc

            # --------------------------------------------------------------
            # 1) soft‑max based Adam optimisation
            opt_seq, opt_val = _adam_optimize_softmax(init, n, target_sum,
                                                      sub_deadline)

            # --------------------------------------------------------------
            # 2) optional LP refinement (few cheap steps)
            if time.time() < deadline:
                # compute current peak
                cur_M, _ = _conv_max_fft(opt_seq)
                for lp_step in range(12):
                    if time.time() >= deadline:
                        break
                    g = _lp_refine(opt_seq, n, target_sum, cur_M)
                    if g is None:
                        break
                    # convex combination – small step size guarantees non‑increase
                    t = 0.05
                    opt_seq = (1.0 - t) * opt_seq + t * g
                    opt_seq = _normalise(opt_seq, target_sum)
                    cur_M, _ = _conv_max_fft(opt_seq)
                opt_val = cur_M

            # --------------------------------------------------------------
            # 3) fine‑tune with the original sub‑gradient Adam (quick)
            if time.time() < deadline:
                opt_seq, opt_val2 = _adam_optimize_subgrad(opt_seq, n, target_sum,
                                                          deadline, max_iters=200_000)
                if opt_val2 < opt_val:
                    opt_seq, opt_val = opt_seq, opt_val2

            # --------------------------------------------------------------
            # 4) cheap greedy local search
            if time.time() < deadline:
                ls_seq, ls_val = _local_search(opt_seq, n, target_sum,
                                               steps=30_000,
                                               deadline=deadline)
                if ls_val < opt_val:
                    opt_seq, opt_val = ls_seq, ls_val

            # --------------------------------------------------------------
            # 5) update global best
            if opt_val < best_val:
                best_val = opt_val
                best_seq = opt_seq.copy()
                print(f"[global] new best peak = {best_val:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Final official evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best peak = {best_val:.6f}, official score = {final_score:.6f}, "
          f"length = {best_seq.shape[0]}")
    return best_seq.tolist()
```
