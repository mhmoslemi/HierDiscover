```python
"""
Gradient‑based search for a non‑negative step function that
minimises  F(a)=2·n·max(a*a)/(∑a)² .
We work on the simplex by exponentiating logits (soft‑max
parametrisation) and use Adam on a smooth‑max surrogate.
The smoothing parameter λ is increased gradually; after it
exceeds a threshold we switch to a sub‑gradient that only
weights the maximal convolution indices.
Multiple random restarts (and occasional use of the known
best sequence) are performed until the time budget expires.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w

def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] w.r.t. a.
    The formula is  ∇ = 2·(w ★ a)  (cross‑correlation).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a*a)/(sum a)²."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

    surrogate (λ grows geometrically) and after λ exceeds a threshold
    switches to a true sub‑gradient that only uses the maximal convolution
    indices.  The best vector seen is kept and finally returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # Length of the step function – a compromise between expressive power and speed.
    n = 3000

    # FFT sizes – pre‑computed for speed.
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))   # comfortably large

    # Hyper‑parameters for Adam.
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.08

    # Smoothing schedule.
    lam_start = 0.5
    lam_factor = 1.015
    lam_stop = 30.0          # after this we switch to a hard sub‑gradient

    best_val = float("inf")
    best_seq = None

    # If a previous good construction exists we keep a resized copy for occasional restarts.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        prev = prev / prev.sum()
    except Exception:
        prev = None

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ----- initialise a random start (or occasionally reuse the known one) -----
        if prev is not None and restart % 15 == 0:
            a = prev.copy()
        else:
            mode = restart % 4
            if mode == 0:

            elif mode == 1:

            else:

        # Transform to log‑space (soft‑max parametrisation)
        u = np.log(a + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        last_improve = time.time()

            dot = np.dot(a, grad_a)

            lr = base_lr / (1.0 + 1e-4 * it)

                if val < best_val:

                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # Return plain Python list as required by the evaluator.
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

The function is scale‑invariant, therefore we work on the probability simplex  
\(a_i\ge0,\;\sum a_i=1\).  
A convenient parametrisation is the **soft‑max** of a log‑vector `u`:

```
a_i = exp(u_i) / Σ_j exp(u_j)
```

so the constraints are automatically satisfied.

The optimisation proceeds in two phases inside each *restart*:

1. **Smooth‑max phase** – a soft‑max over the convolution values with a
   temperature λ that starts small (`λ≈0.5`) and is multiplied by a factor
   (`1.015`) each iteration.  
   The gradient of the surrogate `Σ_k w_k·conv_k` w.r.t. `a` is  
   `∇_a = 2·(w ★ a)` (cross‑correlation), which we compute efficiently with
   FFTs.

2. **Hard‑subgradient phase** – once λ exceeds a threshold (`≈30`) the weight
   vector `w` collapses to the (almost) maximal convolution indices.
   This gives the true sub‑gradient of the original max‑operator.

The gradient on the simplex is transferred to the log‑variables `u`
using the soft‑max Jacobian  

\[
∇_u = a\odot\bigl(∇_a - (a^\top∇_a)\mathbf{1}\bigr).
\]

We update `u` with **Adam** (β₁ = 0.9, β₂ = 0.999) and a gently decaying
learning‑rate.  
If no improvement is observed for a couple of seconds a tiny random shake
is applied to escape shallow plateaux.

Many restarts are executed until the 1000 s budget is exhausted.
Occasionally a previously found good sequence (`height_sequence_1`,
if present) is used as a starting point to diversify the search.
The best vector seen over the whole run is returned as a plain Python list.

The implementation stays fully vectorised, uses only `numpy` and a few
hand‑crafted helper functions, and respects the required time budget.

```python
"""
Gradient‑based search for a non‑negative step function that minimises

    F(a)=2·n·max_k (a*a)[k] / (∑ a_i)² .

The algorithm works on the probability simplex via a soft‑max
(log‑space) parametrisation, applies Adam optimisation on a smooth‑max
surrogate, and after a λ‑continuation switches to a hard sub‑gradient that
only weights the maximal convolution indices.  Multiple random restarts
(and occasional use of the known construction `height_sequence_1`) are
performed until the time budget expires.  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety, should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] w.r.t. a.
    Formula: ∇ = 2·(w ★ a) (cross‑correlation).  Computed with FFTs.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)²."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """Return a list of non‑negative heights with a low evaluation score.

    The routine respects the given time budget (seconds).  It performs
    many Adam‑optimised restarts; each restart first follows a smooth‑max
    surrogate (λ grows geometrically) and after λ exceeds a threshold
    switches to a true sub‑gradient that only uses the maximal convolution
    indices.  The best vector seen is kept and finally returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Length of the step function – a trade‑off between expressive power
    # and per‑iteration cost.
    n = 3000

    # Pre‑compute FFT sizes used repeatedly.
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))   # comfortably large

    # Adam hyper‑parameters.
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.08

    # Smoothing schedule (soft‑max → hard sub‑gradient).
    lam_start = 0.5
    lam_factor = 1.015
    lam_stop = 30.0          # after this we use a hard sub‑gradient

    best_val = float("inf")
    best_seq = None

    # Optional: reuse a previously known good construction (if it exists).
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # Linear interpolation to required length.
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        prev = prev / prev.sum()
    except Exception:
        prev = None

    restart = 0
    while time.time() < deadline:
        restart += 1

        # -------------------------- initialisation --------------------------
        if prev is not None and restart % 15 == 0:
            a = prev.copy()
        else:
            mode = restart % 4
            if mode == 0:
                a = np.random.dirichlet(np.ones(n))
            elif mode == 1:
                r = np.random.uniform(0.2, 0.96)
                a = np.power(r, np.arange(n, dtype=np.float64))
                a /= a.sum()
            elif mode == 2:
                i = np.arange(1, n + 1, dtype=np.float64)
                a = np.sin(np.pi * i / (n + 1))
                a = np.maximum(a, 0.0)
                a /= a.sum()
            else:
                # beta‑distributed start – slightly asymmetric
                alpha = np.random.uniform(0.5, 3.0)
                beta = np.random.uniform(0.5, 3.0)
                a = np.random.beta(alpha, beta, size=n)
                a /= a.sum()

        # Log‑space (soft‑max) parametrisation.
        u = np.log(a + 1e-30)

        # Adam state.
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = lam_start
        it = 0
        last_improve = time.time()

        # --------------------------- optimisation loop ---------------------------
        while time.time() < deadline:
            it += 1

            # Soft‑max to obtain a probability vector.
            u_shift = u - np.max(u)                # numerical stability
            a = np.exp(u_shift)
            a /= a.sum()

            # Linear self‑convolution.
            conv = _full_convolution(a, size=conv_fft_size)

            # Choose weight vector w.
            if lam < lam_stop:
                w = _softmax_weights(conv, lam)
                lam *= lam_factor
            else:
                # Hard sub‑gradient: equal weight on all (almost) maximal indices.
                max_val = conv.max()
                idx = np.where(conv >= max_val - 1e-12)[0]
                w = np.zeros_like(conv)
                w[idx] = 1.0 / len(idx)

            # Gradient w.r.t. a.
            grad_a = _softmax_gradient(a, w, size_corr=corr_fft_size)

            # Convert to gradient w.r.t. logits u (soft‑max Jacobian).
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update.
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # Learning‑rate schedule (gentle decay).
            lr = base_lr / (1.0 + 1e-4 * it)

            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- occasional exact evaluation -----
            if it % 20 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # ----- stagnation handling -----
            if time.time() - last_improve > 2.0:
                # Tiny random shake in log‑space.
                u += (np.random.rand(n) - 0.5) * 5e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # Upper bound on iterations per restart – prevents overruns.
            if it >= 12000:
                break

        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # Convert to a plain Python list for the external evaluator.
    return best_seq.tolist()
```
