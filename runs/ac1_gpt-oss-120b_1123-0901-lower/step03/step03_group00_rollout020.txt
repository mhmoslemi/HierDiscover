    best_val = float('inf')

    # define helper functions: softmax, conv_fft, project_simplex (maybe not needed), gradient_max.

        print(f"\n=== Length {n} ===")
        # Precompute FFT size
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        # Determine number of restarts based on remaining time and n
        remaining = deadline - time.time()
        # allocate maybe 5 restarts with each up to 30000 iterations or until time
        # Based on rough iteration time (maybe 0.001s)
        restarts = 5
        for r in range(restarts):
            if time.time()>deadline:
                break
            # initialize logits
            if best_seq is not None and len(best_seq)==n:
                a0 = np.array(best_seq)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print("[warm start]")
            elif 'height_sequence_1' in globals():

                else:

                cand = np.maximum(cand, 0.0)
                cand_sum = cand.sum()
                if cand_sum < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand = cand / cand_sum
                x = np.log(cand + 1e-15)
                print("[init from height_seq]")
            else:
                # random init from Dirichlet
                a = np.random.dirichlet(np.ones(n))
                x = np.log(a + 1e-15)

            # Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.02

            # scheduling variables
            t = 0
            last_improve = time.time()
            max_iters = 50000

            while time.time() < deadline and t < max_iters:
                # compute softmax probabilities a
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # linear convolution via FFT
                fa = np.fft.rfft(a, fft_len)
                conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

                max_c = conv.max()

                max_idxs = np.flatnonzero(conv == max_c)
                # compute gradient wrt a
                grad_a = np.zeros_like(a)
                for k in max_idxs:
                    if k < n:
                        grad_a[:n - k] += a[k:]
                        grad_a[k:] += a[:n - k]

                avg = np.dot(a, grad_a)
                grad_x = a * (grad_a - avg)

                # Adam update

                m = beta1 * m + (1 - beta1) * grad_x

                m_hat = m / (1 - beta1**t)

                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # Evaluate objective

                if val < best_val:

                    best_seq = a.copy()
                    last_improve = time.time()

        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback)
    return best_seq.tolist()

    # v is a vector, returns projection onto {x >=0, sum=1}
    # standard algorithm:
    u = np.sort(v)[::-1]

    rho = ind[cond].max()

"""
Optimise a non‑negative step‑function (probability vector) to minimise

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a projected Adam optimiser on the probability simplex.
The exact sub‑gradient of the hard max (instead of a smooth‑max surrogate)
is employed, which usually yields a tighter descent direction.
Multiple random restarts, occasional random “shakes”, and a modest schedule
of learning‑rate decay are used.  The routine respects a total time budget
and returns the best sequence found (as a plain Python list) before the
deadline.
"""

import time

# Helper functions – all defined at top level (no closures)
# ------------------------------------------------------------

def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – guarantees non‑negative entries summing to 1."""
    max_x = np.max(x)
    e = np.exp(x - max_x)
    return e / e.sum()


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using the FFT.
    `fft_len` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def max_conv_gradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact gradient of the convolution entry (or average of several entries)
    that attains the current maximum.
    The gradient w.r.t. a is:
        ∂/∂a_j Σ_i a_i a_{i+k} = a_{j+k} + a_{j-k}
    where out‑of‑range accesses are ignored.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        if k < n:                     # first half of the symmetric convolution
            grad[: n - k] += a[k:]    # term a_{j+k}
            grad[k: ]      += a[: n - k]  # term a_{j-k}
        else:                         # use symmetry: conv[k] = conv[2n-2-k]
            k_rev = 2 * n - 2 - k
            # work on the reversed vector and flip back at the end
            a_rev = a[::-1]
            g_rev = np.zeros_like(a_rev)
            g_rev[: n - k_rev] += a_rev[k_rev:]
            g_rev[k_rev: ]      += a_rev[: n - k_rev]
            grad += g_rev[::-1]
    if max_idx.size > 1:
        grad /= max_idx.size
    return grad


def adam_update(x: np.ndarray,
                grad_x: np.ndarray,
                m: np.ndarray,
                v: np.ndarray,
                t: int,
                lr: float,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform a single Adam update on the logits `x`.
    Returns the updated (x, m, v).
    """
    m = beta1 * m + (1.0 - beta1) * grad_x
    v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x, m, v

                # Purely random Dirichlet start
                a = np.random.dirichlet(np.ones(n))
                x = np.log(a + 1e-15)

            # ----------------------------------------------------
            # Adam state
            # ----------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)

            # iteration counters
            t = 0
            last_improve = time.time()
            lr = init_lr

            while time.time() < deadline and t < max_iters:
                # ---------- logits → probabilities (softmax) ----------
                a = softmax(x)          # a_i ≥ 0, Σ a_i = 1

                # ---------- self‑convolution (FFT) ----------
                conv = full_convolution(a, fft_len)
                max_c = conv.max()
                max_idx = np.flatnonzero(conv == max_c)

                # ---------- exact gradient of the hard max ----------
                grad_a = max_conv_gradient(a, max_idx)   # ∇_a max(conv)

                # ---------- back‑prop through softmax ----------
                avg = np.dot(a, grad_a)                 # ⟨a, ∇φ⟩
                grad_x = a * (grad_a - avg)             # ∇_x φ

                # ---------- Adam update ----------
                t += 1
                x, m, v = adam_update(x, grad_x, m, v, t, lr)
                # simple exponential decay of the learning rate
                lr *= lr_decay_factor

                # ---------- exact objective ----------
                # (Σ a_i ≈ 1, so denominator ≈ 1)
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t:05d}] new best: {val:.6f}")

                # ---------- stagnation handling ----------
                if time.time() - last_improve > 5.0:
                    # small random perturbation of the logits to escape plateaus
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t}] shake (stagnation)")

    # -----------------------------------------------------------------
    # Fallback – return a simple uniform vector if the optimiser never improved
    # -----------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2000
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert NumPy array to plain Python list as required by the external evaluator
    return best_seq.tolist()
