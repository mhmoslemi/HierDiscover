Implementation of isotonic regression (non-increasing):

Given vector v, we need to find w that is monotone non-increasing (w_i >= w_{i+1}) minimizing sum (w_i - v_i)^2. This can be done via PAVA: treat decreasing. It's simple: start with each point as a block, each block's weight = 1, value = v_i. Then while any block value less than next block value (i.e., decreasing condition violated), merge blocks: compute weighted average of two adjacent blocks, replace them. Continue.

def isotonic_decreasing(v):
    n = len(v)
    v = v.astype(float)
    # initial blocks values:
    avg = v.copy()
    w = np.ones(n, dtype=int)  # weights per block
    i = 0
    while i < len(avg)-1:
        if avg[i] < avg[i+1] - 1e-12: # violation: not non-increasing
            # merge blocks i and i+1
            total_weight = w[i] + w[i+1]
            avg[i] = (avg[i]*w[i] + avg[i+1]*w[i+1]) / total_weight
            w[i] = total_weight
            # delete block i+1
            avg = np.delete(avg, i+1)
            w = np.delete(w, i+1)
            i = max(i-1, 0)
        else:
            i += 1

def isotonic_regression_increasing(y):
    n = len(y)
    # each block will be stored as (total sum, weight, start index, end index)
    sums = []
    weights = []
    start = []
    end = []
    for i, val in enumerate(y):
        sums.append(val)
        weights.append(1.0)
        start.append(i)
        end.append(i)
        # while there are at least two blocks and average of previous > average of last
        while len(sums) >= 2 and sums[-2] / weights[-2] > sums[-1] / weights[-1]:
            # merge last two blocks
            sums[-2] += sums[-1]
            weights[-2] += weights[-1]
            # new end index is end of last block
            end[-2] = end[-1]
            # remove last block
            sums.pop()
            weights.pop()
            start.pop()
            end.pop()
    # now construct the solution
    result = np.empty(n, dtype=float)
    for s, w, a, b in zip(sums, weights, start, end):
        avg = s / w
        result[a:b+1] = avg
    return result
```

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation runs a time‑limited multi‑restart search.
Each restart optimises a sequence of a chosen length using a smooth‑max
gradient descent, but *after each gradient step* the vector is projected
onto the set of monotone non‑increasing vectors and then scaled onto the
simplex Σa_i = √(2 n).  This extra monotonic projection is a cheap
heuristic that dramatically lowers the maximal self‑convolution for
many lengths.  Both a monotone optimiser and the original soft‑max
optimiser are employed (the former ~70 % of the restarts).  Random
initialisations are power‑law shapes (a_i ∝ (i+1)^{‑p}) possibly with an
exponential tail; warm‑starts from the current global best are used
periodically.  The best sequence found before the time budget expires is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (sum = s, non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)  # protect against tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper – Isotonic regression (non‑decreasing) via PAVA
# ----------------------------------------------------------------------
def _isotonic_regression_increasing(y: np.ndarray) -> np.ndarray:
    """Return the isotonic regression of y (non‑decreasing) w.r.t. L2."""
    n = y.shape[0]
    # each block is represented by (sum, weight, start_index)
    sums = [float(val) for val in y]
    weights = [1] * n
    starts = list(range(n))

    i = 0
    while i < len(sums) - 1:
        # violation of non‑decreasing order?
        if sums[i] / weights[i] > sums[i + 1] / weights[i + 1] + 1e-12:
            # merge block i with i+1
            sums[i] += sums[i + 1]
            weights[i] += weights[i + 1]
            # start index stays the same
            del sums[i + 1]
            del weights[i + 1]
            del starts[i + 1]
            if i > 0:
                i -= 1
        else:
            i += 1

    result = np.empty(n, dtype=float)
    for block_sum, block_weight, start in zip(sums, weights, starts):
        avg = block_sum / block_weight
        result[start:start + block_weight] = avg
    return result


def _isotonic_decreasing(v: np.ndarray) -> np.ndarray:
    """Project v onto the cone of non‑increasing vectors (preserves L2)."""
    return -_isotonic_regression_increasing(-v)


# ----------------------------------------------------------------------
# Optimisation – smooth‑max gradient descent with monotonic projection
# ----------------------------------------------------------------------
def _optimise_one_length_monotonic(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9995,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Optimise a sequence of length n using soft‑max gradient descent
    and a projection onto the monotone decreasing simplex.
    Returns the best sequence (as a NumPy array) and its official score."""
    target_sum = np.sqrt(2.0 * n)

    # initialise – ensure non‑negativity, monotonicity and correct sum
    a = np.maximum(init, 0.0)
    a = _isotonic_decreasing(a)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full(n, target_sum / n)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) self‑convolution & smooth‑max weights
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        shifted = (conv - max_val) / tau
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 2) gradient of the smooth‑max (cross‑correlation)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # pick the needed slice
        grad = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 3) momentum + adaptive step size
        # --------------------------------------------------------------
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)                # clip negatives
        a_new = _isotonic_decreasing(a_new)           # enforce monotone shape
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new * (target_sum / s)
        else:
            a_new = np.full(n, target_sum / n)

        # --------------------------------------------------------------
        # 4) evaluate new candidate
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  monotonic new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _isotonic_decreasing(a)
            s = a.sum()
            if s > 0.0:
                a = a * (target_sum / s)
            step_factor *= 0.9

    # final evaluation with the provided evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Optimisation – original soft‑max optimiser (kept for diversity)
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9995,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Soft‑max gradient descent without monotone projection."""
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  softmax new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the required
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length ``n`` (≈ 800 – 5000) and initialise a
      power‑law decreasing shape a_i ∝ (i+1)^{‑p} (p drawn from [0.3, 0.7]).
      Occasionally append a mild exponential taper.
    * With probability ≈0.7 run the monotone optimiser;
      otherwise fall back to the original soft‑max optimiser.
    * Keep the overall best across many independent restarts,
      occasionally warm‑starting from the current global best
      (truncating or padding as needed).
    * Stop early if the bound drops below the target (1.503).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction if available.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size.  Early on we explore a wide range,
        # later we focus on medium lengths.
        if remaining > 300:
            n = np.random.randint(800, 5000 + 1)
        elif remaining > 120:
            n = np.random.randint(600, 4000 + 1)
        else:
            n = np.random.randint(400, 2500 + 1)

        # ------------------- initial shape -----------------------------
        p = np.random.uniform(0.3, 0.7)               # power‑law exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)

        # optional exponential taper (adds extra flexibility)
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.4, 2.0) * n
            init *= np.exp(-idx / tau_exp)

        # tiny jitter to break perfect symmetry
        init += np.random.rand(n) * 1e-4

        # occasional warm‑start from the current global best
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                init = np.concatenate([bs, pad])

        # allocate a time slice (≈ 8 s, never exceed the global deadline)
        slice_seconds = min(8.0, remaining * 0.45)
        run_deadline = min(deadline, time.time() + slice_seconds)

        # decide which optimiser to use
        if np.random.rand() < 0.70:
            seq_local, score_local = _optimise_one_length_monotonic(
                n,
                init,
                run_deadline,
                init_tau_factor=0.12,
                min_tau_factor=0.001,
                step_factor=0.6,
                step_decay=0.9995,
                momentum_beta=0.9,
            )
        else:
            seq_local, score_local = _optimise_one_length_softmax(
                n,
                init,
                run_deadline,
                init_tau_factor=0.10,
                min_tau_factor=0.001,
                step_factor=0.6,
                step_decay=0.9995,
                momentum_beta=0.9,
            )

        # update the global best if improvement was found
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # early exit if we comfortably beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
