def _subgradient_optimize(n, init_a, best_seq, best_score, deadline, max_iters=None):
    target_sum = math.sqrt(2*n)
    a = init_a.copy()
    a = a * (target_sum / a.sum())
    # hyperparams
    lr0 = 0.15   # initial learning rate
    beta = 0.9
    momentum = np.zeros_like(a)
    # FFT size
    Nfft = 1 << ((2*n - 1).bit_length())
    iter_count = 0
    # Determine max_iters based on time left: maybe 500000 * (2100/n)
    # But allow arbitrary max when not provided.
    if max_iters is None:
        # compute based on remaining time
        remaining = deadline - time.time()
        # estimate iteration cost ~ 0.00002 sec for n=2100? Not sure
        # We'll set max iters to 200000 for n<=4000, less for bigger n
        est = int(200000 * 2100 / n)
        max_iters = max(2000, est)
    # while loop
    while time.time() < deadline and iter_count < max_iters:
        # compute convolution a*a via FFT (full)
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2*n - 1]
        M = conv.max()
        # Identify active indices
        # tolerance: consider values >= M - 1e-12
        # also ensure at least 1 index
        eps = max(1e-12, M * 1e-12)
        active = np.where(conv >= M - eps)[0]
        # ensure non-empty
        if active.size == 0:
            active = np.array([np.argmax(conv)])
        # weight vector
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        # compute rev = correlation (weighted_rev)
        rev = _weighted_rev(a, w, Nfft)  # from earlier code
        # gradient = 2 * rev
        grad = 2.0 * rev
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        # learning rate schedule: lr = lr0 / np.sqrt(iter_count + 1)
        lr = lr0 / np.sqrt(iter_count + 1)
        # take step
        a -= lr * momentum
        # projection onto feasible set (nonneg + sum constraint)
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a > 0:
            a = a * (target_sum / sum_a)
        else:
            a = np.full_like(a, target_sum / n)
        # evaluate objective (M)
        cur_score = M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iter_count:8d}] new best {best_score:.6f}")
            # optional: reduce lr after improvement? maybe not.
        # jitter occasionally
        if iter_count % 10000 == 0 and iter_count > 0:
            # small noise
            jitter = np.random.randn(n) * (0.001 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        iter_count += 1
    return best_seq, best_score

def _subgradient_optimize(n, init_a, init_best_seq, init_best_score, deadline, max_iters):
    a = init_a.copy()
    target_sum = math.sqrt(2 * n)
    a = a * (target_sum / a.sum())
    # hyperparameters
    lr0 = 0.2  # maybe 0.2
    beta = 0.9
    momentum = np.zeros_like(a)
    Nfft = 1 << ((2 * n - 1).bit_length())
    iter = 0
    best_seq = init_best_seq
    best_score = init_best_score
    # for loop
    while time.time() < deadline and iter < max_iters:
        # conv
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2*n -1]
        M = conv.max()
        # active set
        eps = 1e-12 * max(1.0, M)
        # choose indices where conv >= M - eps
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        # compute rev
        rev = _weighted_rev(a, w, Nfft)
        grad = 2.0 * rev
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        lr = lr0 / np.sqrt(iter + 1)
        a = a - lr * momentum
        # projection
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a > 0:
            a = a * (target_sum / sum_a)
        else:
            a = np.full_like(a, target_sum / n)
        # update best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iter:8d}] new best = {best_score:.6f}")
            # maybe reduce lr0 after improvement?
        # jitter occasionally
        if iter % 5000 == 0 and iter != 0:
            jitter = np.random.randn(n) * (0.001 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        iter += 1
    # final evaluation via evaluate_sequence for good measure:
    try:
        exact = evaluate_sequence(best_seq.tolist())
        if exact < best_score: # due to rounding
            best_score = exact
    except Exception:
        pass
    return best_seq, best_score

* For each length we run a projected sub‑gradient method on the exact
  objective  M(a)=max_k (a∗a)_k .
    – In every iteration the full convolution a∗a is obtained with a single
      FFT.
    – All indices attaining the current maximum (within a tiny tolerance)
      form the active set.  The sub‑gradient of M is
          g = 2·rev ,   rev_i = Σ_{k∈active} a_{k‑i} / |active|
      which is computed by one additional FFT (correlation of a with the
      active‑set weights).
    – A momentum term (β=0.9) gives an Adam‑like update but without the
      adaptive per‑coordinate scaling (the problem is already well‑scaled).
    – After the step we project onto the feasible set
          a_i ≥ 0 ,   Σ a_i = √(2 n)
      by simple clipping and rescaling.
    – Occasionally a tiny Gaussian jitter is added to help escape plateaus.
* Each run keeps the best sequence found (lowest exact objective).  The
  global best over all lengths and restarts is returned.
* The routine respects the supplied wall‑clock budget (default 1000 s).

The method searches directly on the true max‑convolution (no soft‑max
approximation) and therefore avoids the bias introduced by temperature
annealing, while still being fast thanks to FFT‑based convolutions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: correlation needed for the sub‑gradient
def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k‑i} for i=0…n‑1 (indices outside the
    range of a are ignored).  Implemented via FFT for speed.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]             # keep the valid part
    # The needed slice starts at index 2*n‑2 and runs backwards.
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


# ----------------------------------------------------------------------
# Main optimisation routine for a single length
def _subgradient_optimize(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on M(a)=max(a∗a) for a fixed length n.
    Returns the best sequence (and its score) found during the run.
    """
    target_sum = math.sqrt(2.0 * n)

    # start from supplied initial vector; enforce exact L1‑norm
    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam‑style momentum (no per‑coordinate adaptation)
    beta = 0.9
    momentum = np.zeros_like(a)

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # basic learning‑rate schedule
    lr0 = 0.18

    # tolerance for the active set (floating point safety)
    eps_rel = 1e-12

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- 1) full self‑convolution ---------------------------------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- 2) active set (indices attaining the maximum) ------------
        eps = max(1e-12, eps_rel * M)
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:                     # should never happen
            active = np.array([np.argmax(conv)])

        # ----- 3) sub‑gradient of the exact max -------------------------
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size               # uniform weights over active set
        rev = _weighted_rev(a, w, Nfft)             # rev_i = Σ_k w_k a_{k‑i}
        grad = 2.0 * rev                             # ∂M/∂a = 2·rev

        # ----- 4) momentum update ---------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- 5) step size (decays like 1/√t) -------------------------
        lr = lr0 / math.sqrt(it + 1.0)
        a -= lr * momentum

        # ----- 6) projection onto a ≥ 0 , Σa = √(2 n) -------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:                                      # all entries became zero (unlikely)
            a = np.full_like(a, target_sum / n)

        # ----- 7) record improvement ------------------------------------
        if M < best_score - 1e-12:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:9d}] new best = {best_score:.6f}")

        # ----- 8) occasional jitter to escape shallow plateaus ----------
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * (0.001 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    # final safety evaluation (in case rounding changed the exact value)
    try:
        exact = float(np.max(np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]))
        if exact < best_score:
            best_score = exact
            best_seq = a.copy()
    except Exception:
        pass

    return best_seq, best_score

            init_candidates.append(cand)
