"""
Propose a candidate step‑function for the objective

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

* Each sequence is represented by an unconstrained real vector `v`.
  The actual heights are obtained by a soft‑max mapping

        a_i = S * exp(v_i) / Σ_j exp(v_j) ,   S = √(2 n)

  which guarantees `a_i ≥ 0` and Σ a_i = S without any explicit
  projection.

* For a fixed length `n` we minimise a *smooth* approximation of the
  maximal convolution value

        Φ(v; τ) = τ * log Σ_k exp( (a*a)_k / τ )

  (τ is a temperature).  When τ → 0 this coincides with the true
  maximum.  The gradient of Φ w.r.t. `v` can be written in closed form
  using a cross‑correlation, which we evaluate with a real FFT (O(n log n)).

* The smooth maximum is reduced gradually:
  τ is initialised as a fraction of the initial maximal convolution
  and multiplied by a decay factor after each “stage”.

* Within a stage we run a few iterations of L‑BFGS‑B (SciPy) on the
  smooth objective.  The method supplies a line‑search and momentum,
  giving a fast convergence compared with hand‑crafted step‑size
  schedules.

* After each stage we evaluate the *exact* objective via the supplied
  `evaluate_sequence` function and keep the best sequence found.
  Occasional random perturbations are added to escape plateaux.

* The whole procedure is wrapped in a time‑budgeted outer loop that
  performs many independent restarts with random lengths and random
  initialisations (or warm‑starts from the current global best).  The
  best sequence over all restarts is returned.

The implementation respects the 1000 s budget, prints progress, and
only uses NumPy, SciPy and the provided `evaluate_sequence`.
"""

import time

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex (used only for the
# fallback optimiser).
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – full linear convolution using a real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # cut tiny negatives
    return conv


# ----------------------------------------------------------------------
# Soft‑max objective wrapped for SciPy optimisation
# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """
    Callable object returning (value, gradient) for the smooth maximum
    of the self‑convolution of the sequence defined by the soft‑max
    parameters `v`.

    Parameters
    ----------
    n : int
        Length of the sequence.
    target_sum : float
        Desired Σ a_i = √(2 n).
    tau : float
        Current temperature (smoothing parameter).
    """
    def __init__(self, n: int, target_sum: float, tau: float):
        self.n = n
        self.target_sum = target_sum
        self.tau = tau

    def __call__(self, v: np.ndarray):
        # ---- map v -> a (soft‑max, strict Σ a_i = target_sum) ----
        vmax = np.max(v)
        ev = np.exp(v - vmax)                # shift for numerical stability
        sum_ev = np.sum(ev)
        u = ev / sum_ev                       # lies on the probability simplex
        a = self.target_sum * u

        # ---- full self‑convolution ----
        conv = _full_convolution_fft(a, a)

        # ---- smooth maximum (τ · log‑sum‑exp) ----
        if self.tau < 1e-12:                  # effectively exact max
            max_c = np.max(conv)
            smooth_val = float(max_c)
            mask = (conv >= max_c - 1e-12)
            w = mask.astype(float)
            w /= w.sum()
        else:
            max_c = np.max(conv)
            shifted = conv - max_c
            # log‑sum‑exp in a stable way
            sum_exp = np.sum(np.exp(shifted / self.tau))
            smooth_val = max_c + self.tau * np.log(sum_exp)
            w = np.exp(shifted / self.tau)
            w /= np.sum(w)                     # normalised smooth‑max weights

        # ---- gradient w.r.t. a (cross‑correlation with reversed weights) ----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)       # length = 3n‑2
        idx = (2 * self.n - 2) - np.arange(self.n)    # positions giving Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- chain rule for the soft‑max parametrisation ----
        dot = np.dot(grad_a, u)                       # Σ_i ∂Φ/∂a_i * u_i
        grad_v = self.target_sum * (grad_a - dot * u)

        return float(smooth_val), grad_v.astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation of a single length using L‑BFGS‑B on the smooth maximum
# ----------------------------------------------------------------------
def _optimise_one_length_lbfgs(
    n: int,
    init: np.ndarray,
    *,
    init_tau_factor: float = 0.20,
    min_tau: float = 1e-5,
    tau_decay: float = 0.5,
    stage_iters: int = 30,
    stages: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n` starting from `init` using a
    multi‑stage L‑BFGS‑B optimiser.  Returns the best sequence (as a
    NumPy array) and its exact evaluation score.
    """
    target_sum = np.sqrt(2.0 * n)

    # ---- initialise the soft‑max parameters v ----
    a0 = init.astype(np.float64)
    a0 = a0 / np.sum(a0) * target_sum           # enforce the sum exactly
    eps = 1e-14
    v = np.log(a0 + eps)

    # ---- initialise temperature τ ----
    conv0 = _full_convolution_fft(a0, a0)
    max0 = conv0.max()
    tau = max(init_tau_factor * max0, min_tau)

    # ---- keep track of the best exact score ----
    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ---- main multi‑stage optimisation loop ----
    for stage in range(stages):
        # reduce the temperature
        tau = max(min_tau, tau * tau_decay)
        obj = _SmoothMaxObjective(n, target_sum, tau)

        # Run L‑BFGS‑B for a limited number of iterations
        res = minimize(
            obj,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": stage_iters, "disp": False},
        )
        v = res.x

        # ----- reconstruct the current sequence -----
        vmax = np.max(v)
        ev = np.exp(v - vmax)
        u = ev / np.sum(ev)
        a_curr = target_sum * u

        # ----- exact evaluation -----
        sc = evaluate_sequence(a_curr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_curr.copy()
            print(f"[{time.time():.2f}] n={n} stage={stage}  new best = {best_score:.6f}")

        # ----- occasional random jitter to escape shallow plateaux -----
        if np.random.rand() < 0.15:
            jitter = np.random.randn(n) * (target_sum * 0.001 / n)
            v += jitter

    return best_seq, best_score


# ----------------------------------------------------------------------
# Fallback optimiser (soft‑max gradient descent) – used when SciPy is
# unavailable.  The implementation is essentially the one shown in the
# problem statement.
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9995,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Older soft‑max sub‑gradient descent (kept for robustness)."""
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction if present.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size.  Early on we explore broadly,
        # later we focus on a tighter band to keep iterations cheap.
        if remaining > 400:
            n = np.random.randint(1800, 5000 + 1)
        elif remaining > 150:
            n = np.random.randint(1200, 3500 + 1)
        else:
            n = np.random.randint(800, 2000 + 1)

        # ----- initial shape ------------------------------------------------
        init = np.random.rand(n)                      # Dirichlet‑like start
        init += np.random.rand(n) * 1e-6               # tiny jitter

        # Occasionally warm‑start from the current global best (truncated/padded)
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-6
                init = np.concatenate([best_seq, pad])

        # ----- allocate a time slice for this optimisation -------------------
        run_deadline = min(deadline, time.time() + min(12.0, remaining * 0.45))

        # ----- select optimiser ------------------------------------------------
        if _SCIPY:
            seq_local, score_local = _optimise_one_length_lbfgs(
                n,
                init,
                init_tau_factor=0.20,
                min_tau=1e-5,
                tau_decay=0.5,
                stage_iters=30,
                stages=5,
            )
        else:
            seq_local, score_local = _optimise_one_length_softmax(
                n,
                init,
                run_deadline,
                init_tau_factor=0.12,
                min_tau_factor=0.001,
                step_factor=0.6,
                step_decay=0.9994,
                momentum_beta=0.9,
            )

        # ----- update global best ------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            if best_score < 1.49:          # early exit if we are already very good
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
