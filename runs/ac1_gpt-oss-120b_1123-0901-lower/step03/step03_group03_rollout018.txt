def run_cccp(a0, deadline, max_iter=20, keep_top=500):
    n = len(a0)
    target_sum = math.sqrt(2.0*n)
    a = a0.copy()
    best_a = a.copy()
    best_M = max_self_convolution(a)
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        conv = max_self_convolution(a, full=True) # return full conv array
        M = conv.max()
        # Determine active constraints indices
        # Choose top_k indices
        K = min(2*n, max(200, int(0.05*n)))  # say 5% of n or min 200
        idx = np.argpartition(-conv, K-1)[:K]   # fastest top K
        # Build sparse A_ub matrix
        rows = []
        cols = []
        data = []
        b_ub = []
        for r_idx, k in enumerate(idx):
            # compute j range
            j_start = max(0, k-(n-1))
            j_end = min(n-1, k)
            # for j in range(j_start, j_end+1):
            # coefficient = a[j] + a[k-j]
            # store in row r_idx
            for j in range(j_start, j_end+1):
                coeff = a[j] + a[k-j]
                rows.append(r_idx)
                cols.append(j)
                data.append(coeff)
            # coefficient for M variable (index n)
            rows.append(r_idx)
            cols.append(n)
            data.append(-1.0)
            # RHS
            b_ub.append(conv[k])  # as per inequality -M + sum coeff * a <= conv[k]
        # Convert to csr matrix
        A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(len(idx), n+1))
        b_ub = np.array(b_ub, dtype=float)
        # Equality sum a = target_sum
        A_eq = sparse.csr_matrix(np.hstack([np.ones((1,n)), np.zeros((1,1))]))
        b_eq = np.array([target_sum])
        # Bounds: a_i >=0, M >=0
        bounds = [(0, None)]*n + [(0, None)]
        # Solve LP
        res = linprog(c=np.hstack([np.zeros(n), [1.0]]), A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0, deadline-time.time())})
        if not res.success:
            print(f"[cccp] LP not solved: {res.message}")
            break
        a_new = res.x[:n]
        M_new = res.x[n]
        # Update best
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            print(f"[cccp n={n} iter={iteration}] new best M={best_M:.6f}")
        # Stopping condition: improvement small or M_new ~ M
        if M_new >= M - 1e-9:
            # no improvement
            break
        a = a_new
        iteration += 1
    return best_a, best_M

def adam_fine_tune(a, max_iters, deadline):
    n = len(a)
    target_sum = sqrt(2*n)
    # Adam state: m,v = zeros
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1,beta2,eps = 0.9,0.999,1e-8
    lr0 = 0.02
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute conv
        Nfft = 1 << ((2 * n - 1).bit_length())
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2 * n - 1]
        M = conv.max()
        # find index of max
        k_max = int(conv.argmax())
        # compute weighted_rev: w vector one-hot
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        rev = _weighted_rev(a, w, Nfft)  # uses FFT inside
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1 - beta1**t)
        v_hat = v/(1 - beta2**t)
        lr = lr0/(1+1e-4*t)
        a = a - lr*m_hat/(np.sqrt(v_hat)+eps)

        # projection onto feasible region
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

    return a

def solve_linearized_lp(a, conv, idx_top):
    n = len(a)
    target_sum = math.sqrt(2.0*n)
    rows, cols, data = [], [], []
    b = []
    for r,k in enumerate(idx_top):
        # compute j_start,j_end
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # j_range
        j_range = np.arange(j_start, j_end+1)
        coeff = a[j_range] + a[k - j_range]
        # add entries
        rows.extend([r]*len(j_range))
        cols.extend(j_range.tolist())
        data.extend(coeff.tolist())
        # M var column
        rows.append(r)
        cols.append(n)
        data.append(-1.0)
        b.append(conv[k])
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(len(idx_top), n+1))
    b = np.array(b, dtype=float)
    # equality
    A_eq = sparse.csr_matrix(np.append(np.ones((1,n)), np.zeros((1,1)), axis=1))
    b_eq = np.array([target_sum])
    bounds = [(0, None)]*n + [(0, None)]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        return None, None
    a_new = res.x[:n]
    M_new = res.x[n]
    return a_new, M_new

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm (two‑phase composite):
* Phase‑1 (optional, uses a linear program):
  – Linearise the quadratic constraints   (a*a)_k ≤ M   at the current
    vector a and keep only the K most critical constraints (those with
    the largest convolution values).
  – Solve the resulting LP (variables a_0…a_{n‑1}, M) which directly
    minimises M under the linearised constraints and the fixed L¹‑norm.
  – The LP solution is projected back onto the feasible simplex.
* Phase‑2 (smooth Adam optimisation):
  – Starting from the LP solution (or directly from the initial guess
    if SciPy is unavailable) perform a long projected Adam run that
    minimises the exact max‑convolution.
  – The gradient is the sub‑gradient of the max‑convolution (a single
    one‑hot weight in the soft‑max), computed efficiently by a single
    FFT.
  – Random jitter is injected periodically to escape plateaus.
* The routine tries several sequence lengths (original, 1.25×, 1.5×,
  2×, and up to 8000 entries) and keeps the best sequence found
  before the 1000 s wall‑clock limit expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP phase)
try:
    from scipy.optimize import linprog
    from scipy import sparse
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k a_{k‑i}
    (indices outside 0…len(a)-1 are ignored) using FFTs.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a∗a (length 2·n‑1)."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Phase‑1: linearised LP (only if SciPy is present)
def _solve_linearised_lp(a: np.ndarray,
                         conv: np.ndarray,
                         idx_top: np.ndarray) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    One LP step:
    minimise M  subject to
        Σ_j (a_j^{old}+a_{k-j}^{old}) a_j - conv_k ≤ M   for k∈idx_top,
        a_j ≥ 0,
        Σ a_j = sqrt(2·n).

    Returns the new vector a and the scalar M (or (None,None) on failure).
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    rows, cols, data = [], [], []
    b = []

    for r, k in enumerate(idx_top):
        # valid j such that 0≤j<n and 0≤k‑j<n
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        j_range = np.arange(j_start, j_end + 1, dtype=int)
        coeff = a[j_range] + a[k - j_range]          # a_j + a_{k-j}
        rows.extend([r] * len(j_range))
        cols.extend(j_range.tolist())
        data.extend(coeff.tolist())

        # coefficient for the variable M (last column)
        rows.append(r)
        cols.append(n)                               # column index of M
        data.append(-1.0)

        b.append(conv[k])

    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(len(idx_top), n + 1))
    b = np.array(b, dtype=float)

    # equality Σ a_j = target_sum
    A_eq_dense = np.zeros((1, n + 1), dtype=float)
    A_eq_dense[0, :n] = 1.0
    A_eq = sparse.csr_matrix(A_eq_dense)
    b_eq = np.array([target_sum], dtype=float)

    # bounds (non‑negative, M also non‑negative)
    bounds = [(0, None)] * n + [(0, None)]

    # minimise M
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:          # pragma: no cover
        return None, None

    if not res.success:
        return None, None

    a_new = res.x[:n]
    M_new = res.x[n]
    return a_new, M_new


def _lp_phase(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform a single linear‑programming refinement (if SciPy is available).
    Returns a possibly improved vector a.
    """
    if not _SCIPY_AVAILABLE:
        return a

    n = a.shape[0]
    conv = _full_convolution(a)
    # keep the most critical constraints (top K values)
    K = min(800, max(200, n // 3))
    idx_top = np.argpartition(-conv, K - 1)[:K]

    a_new, M_new = _solve_linearised_lp(a, conv, idx_top)
    if a_new is None:
        return a

    # projection back onto the feasible simplex
    a_new = np.maximum(a_new, 0.0)
    target_sum = math.sqrt(2.0 * n)
    s = a_new.sum()
    if s > 0:
        a_new = a_new * (target_sum / s)
    else:                       # extremely unlikely, but guard against division by zero
        a_new = np.full_like(a_new, target_sum / n)

    return a_new


# ----------------------------------------------------------------------
# Phase‑2: long Adam optimisation (smooth max -> exact max)
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser that directly minimises the exact max
    self‑convolution (sub‑gradient obtained from a one‑hot weight).
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 2000

    # pre‑compute FFT size (same throughout the run)
    Nfft = 1 << ((2 * n - 1).bit_length())

    iteration = 0
    while iteration < max_iters and time.time() < deadline:
        iteration += 1

        # full convolution and current max
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())
        k_max = int(conv.argmax())

        # one‑hot weight for the active index
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        rev = _weighted_rev(a, w, Nfft)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto {a≥0, Σa = target_sum}
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:                       # defensive fallback
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape shallow plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return a
