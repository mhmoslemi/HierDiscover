def project_to_sum_and_nonneg(a, target_sum):
    a = np.maximum(a,0)
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a = np.full_like(a, target_sum / len(a))
    return a

def gradient_descent_maxconv(a_init, target_sum, time_deadline):
    a = a_init.copy()
    a = project_to_sum_and_nonneg(a, target_sum)
    best_seq = a.copy()
    # compute initial max conv
    best_val = _conv_max_fft(a)[0]  # max conv
    iteration = 0
    step0 = 0.2  # initial step size; adapt later
    while time.time()<time_deadline:
        # compute conv and max
        M, k_max, conv = _conv_max_fft(a)
        # compute gradient: derivative of M wrt a
        rev = _rev_for_shift(a, k_max)
        grad = 2 * rev  # derivative of conv entry
        # optionally normalize gradient to have unit norm for stable step
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break
        # step size schedule: step = step0 / sqrt(iter+1)
        step = step0 / np.sqrt(iteration+1)
        a = a - step * grad
        # project
        a = project_to_sum_and_nonneg(a, target_sum)
        # evaluate current max conv
        M_new = _conv_max_fft(a)[0]
        if M_new < best_val - 1e-12:

            best_seq = a.copy()

        if iteration % 5000 ==0:
            # we can optionally random jitter
            jitter = np.random.randn(len(a)) * step * 0.01
            a = project_to_sum_and_nonneg(a + jitter, target_sum)
    return best_seq, best_val

"""
Improved optimizer for the step‑function attack.

The strategy builds on the Adam optimiser used previously but
*always* follows the sub‑gradient of the *worst* (maximal) convolution
entry – i.e. we move mass away from indices that contribute most to the
current peak.
* The optimiser is run on several lengths (interpolated from the known
construction and from random / exponential starts) and each run is
followed by a cheap coordinate‑swap hill‑climb.
* A final refinement is performed on the overall best sequence.
All updates keep the vector non‑negative and renormalised to the required
ℓ₁‑norm ( √(2 n) ), which makes the score returned by `evaluate_sequence`
equal to the maximal convolution value.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Full autocorrelation a*a via FFT; returns (max, argmax, full array)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    max_val = float(conv.max())
    argmax = int(conv.argmax())
    return max_val, argmax, conv


# ----------------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """For shift k return the reversed vector:
       rev[i] = a[k‑i] if the index lies inside the support, else 0."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _project_to_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to non‑negative and renormalise so that sum(a) == target_sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:                                   # degenerate – fill uniformly
        a = np.full_like(a, target_sum / a.shape[0])
    return a


# ----------------------------------------------------------------------
def _adam_optimize(a_init: np.ndarray, n: int, target_sum: float,
                   best_seq: np.ndarray, best_score: float,
                   deadline: float, max_iter: int = 20000,
                   jitter_interval: int = 2000, lr0: float = 0.10):
    """
    Adam optimiser that always follows the sub‑gradient of a currently
    maximal convolution entry (worst shift).  Returns possibly improved
    (best_seq, best_score).
    """
    a = _project_to_sum(a_init.copy(), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    coeff = 2.0                         # derivative factor of a_i a_{k-i}
    sqrt2 = math.sqrt(2.0)

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        # ------------------------------------------------------------
        # 1) full convolution, locate maximal entry
        M, _, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        # ------------------------------------------------------------
        # 2) pick a worst shift (always pick one from the active set)
        k = np.random.choice(active) if active.size > 1 else active[0]
        rev = _rev_for_shift(a, k)

        # ------------------------------------------------------------
        # 3) gradient of the objective (M is the quantity we minimise)
        const = sqrt2 * M / math.sqrt(n)          # Lagrange multiplier proxy
        grad = coeff * rev - const                # (n,)

        # ------------------------------------------------------------
        # 4) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------
        # 5) projection onto feasible set
        a = _project_to_sum(a, target_sum)

        # ------------------------------------------------------------
        # 6) occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = _project_to_sum(a + jitter, target_sum)

        # ------------------------------------------------------------
        # 7) current score (exactly evaluate_sequence’s value)
        sum_a = a.sum()
        cur_score = 2.0 * n * M / (sum_a * sum_a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best score = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a_start: np.ndarray, n: int, target_sum: float,
                  best_score: float, deadline: float,
                  max_proposals: int = 30000):
    """
    Very cheap hill‑climb: repeatedly transfer a tiny amount of mass from a
    donor index to a receiver index if the move decreases the maximal
    convolution value.
    """
    a = a_start.copy()
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        eps_transfer = a[i] * np.random.rand() * 0.05
        if eps_transfer <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps_transfer
        new_a[j] += eps_transfer
        new_a = _project_to_sum(new_a, target_sum)

        M_new, _, _ = _conv_max_fft(new_a)
        cur_score = 2.0 * n * M_new / (new_a.sum() ** 2)
        if cur_score < best_score - 1e-12:
            a = new_a
            best_score = cur_score
            print(f"[local] improvement to {best_score:.6f} (prop {proposal})")
    return a, best_score


# ----------------------------------------------------------------------
def _init_exponential(n: int, target_sum: float, symmetric: bool = False):
    """
    Initialise a vector whose entries decay exponentially.
    If `symmetric` is True we add a reflected copy to create peaks at both
    ends of the interval.
    """
    lam = np.random.uniform(0.0005, 0.0100)          # decay rate
    idx = np.arange(n, dtype=float)
    if symmetric:
        a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    else:
        a = np.exp(-lam * idx)
    a = _project_to_sum(a, target_sum)
    return a
