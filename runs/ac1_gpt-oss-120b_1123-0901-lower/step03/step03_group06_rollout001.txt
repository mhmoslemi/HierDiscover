"""
search for a low‑peak self‑convolution step function.

Algorithm
---------
* Work with a probability vector p (∑p = 1).  The true heights are
  a = sqrt(2·n)·p, so the evaluation score equals max(p∗p).
* Multi‑scale optimisation:
     – start with a modest length (≈400), optimise, up‑sample to a
       larger length and continue.
* For a fixed length we run an exponentiated‑gradient (mirror descent)
  on the simplex with the following improvements over the reference
  implementation:
     – a momentum term (simple exponential smoothing of the gradient);
     – a β‑schedule that grows up to 100, so the soft‑max approaches a
       hard max as optimisation proceeds;
     – an adaptive step‑size that slowly decays;
     – occasional Gaussian jitter to escape plateaus.
* After the EG phase we perform a short “hill‑climbing” phase that moves
  a tiny amount of mass from one coordinate to another if this reduces
  the current max‑convolution.
* Random restarts (short runs from a Dirichlet start) are tried when
  time permits.
* The best sequence seen over all lengths and restarts is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full self‑convolution p∗p using FFT of size N.
    Returns the real part of length 2·len(p)‑1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i}  (i = 0..n‑1) via FFT.
    This is the correlation of p with the weight vector w.
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]    # keep linear part
    rev = conv_rev[L - n : L][::-1]    # extract and reverse
    return rev


# ----------------------------------------------------------------------
def _optimize_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    beta_end: float = 100.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns best probability vector (∑p = 1) and its max‑convolution.
    """
    # --- hyper‑parameters (tuned heuristically) ---
    max_iters = 300_000                 # safety cap
    jitter_interval = 5_000
    jitter_scale = 5e-4
    beta_start = 0.05
    beta_len = 120_000                  # iterations over which β grows
    momentum = 0.9

    p = np.clip(p_start, 1e-30, None)   # avoid exact zero for log
    logp = np.log(p)

    # FFT size sufficient for both convolution and weighted correlation
    N = _next_pow2(3 * n - 2)

    # momentum accumulator
    m_grad = np.zeros_like(p)

    best_score = float("inf")
    best_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution and current max -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = M                     # because evaluate_sequence == max(conv)

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n:5d} it {it:7d}] new local best = {cur_score:.6f}")

        # ----- soft‑max weights -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))    # values ≤ 1, w_max = 1

        # ----- weighted rev (gradient of the smoothed max) -----
        rev = _weighted_rev_fft(p, w, N)
        grad = 4.0 * n * rev              # ∇_p (2·n·M_β)

        # ----- momentum & step‑size -----
        m_grad = momentum * m_grad + (1.0 - momentum) * grad
        # Simple decay schedule for η
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 5e-5)

        # ----- EG update in log‑space -----
        logp -= eta * m_grad

        # occasional jitter
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- projection onto the simplex (KL‑projection) -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)          # safe exponentiation
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


# ----------------------------------------------------------------------
def _hill_climb(p: np.ndarray, n_iters: int, step_frac: float, deadline: float) -> np.ndarray:
    """
    Simple local search that moves a tiny amount of mass from one
    coordinate to another if the max‑convolution improves.
    """
    n = p.shape[0]
    rng = np.random.default_rng()
    # pre‑compute FFT size for speed
    N = _next_pow2(3 * n - 2)

    best_p = p.copy()
    best_conv = _full_convolution_fft(best_p, N)
    best_M = best_conv.max()

    for _ in range(n_iters):
        if time.time() >= deadline:
            break

        # pick a source and a destination
        i = rng.integers(0, n)
        if best_p[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # propose moving a small fraction of p[i] to p[j]
        delta = best_p[i] * step_frac * rng.random()
        if delta <= 0.0:
            continue

        cand = best_p.copy()
        cand[i] -= delta
        cand[j] += delta

        conv = _full_convolution_fft(cand, N)
        M = conv.max()

        if M + 1e-12 < best_M:      # accept only strict improvements
            best_p = cand
            best_M = M

    return best_p
