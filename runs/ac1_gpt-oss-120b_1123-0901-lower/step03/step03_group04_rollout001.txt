Search for a non‑negative step function (list of heights) that minimizes
the black‑box ``evaluate_sequence``.  The objective after normalising the
ℓ₁‑norm to √(2 n) is simply the peak of the full autocorrelation
convolution ``a*a``.  The algorithm works as follows:

1.  Choose a few candidate lengths (the original length and a few
    multiples).  For each length start from either the previously known
    construction (``height_sequence_1``) or a random Dirichlet‑like vector.

2.  Iteratively improve the vector *f*:
    * Compute the full convolution ``c = f*f`` and its maximum ``M``.
    * Select the “active’’ shift positions where ``c`` is within a tiny
      tolerance of ``M`` (at most ``max_active`` indices).
    * Solve a linear programme that **maximises** the ℓ₁‑norm of a new
      vector ``g`` under the linear constraints ``f⋆g ≤ M`` at the active
      shifts.  Variables are bounded by ``[0, target_sum]`` to keep the
      LP bounded.
    * Scale ``g`` so that its ℓ₁‑norm equals the required value
      ``target_sum = √(2 n)``.
    * Find the optimal mixing coefficient ``t∈[0,1]`` for the convex
      combination ``h = (1‑t)·f + t·g`` that minimises the new peak
      autocorrelation.  A cheap ternary search is used; each evaluation
      of the peak uses an FFT (``O(n log n)``).
    * Replace ``f`` by the best mixture and renormalise.

3.  Occasionally perform a cheap stochastic local search that moves a
    tiny amount of mass between two random positions if it reduces the
    peak.

4.  Keep the best vector seen over all lengths, restarts and iterations.
    The routine stops as soon as the 1000 s budget (minus a safety margin)
    is exhausted and returns the best sequence.

The whole search is deterministic apart from the random seeds and uses
only ``numpy`` and ``scipy.optimize.linprog`` (the latter is available in
the execution environment).  No closures or lambdas are used – all helper
functions are top‑level.
"""

def _conv_max_fft(a: np.ndarray):
    """
    Return the maximal value of the full convolution a*a and the full
    convolution array (real).  Uses an FFT of length equal to the next
    power of two for speed.
    """
    n = a.shape[0]
    # size for zero‑padding: next power of two ≥ 2*n‑1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A                     # elementwise product → FFT of convolution
    conv = np.fft.irfft(B, n=N)
    conv = conv[: 2 * n - 1]      # keep only the valid part
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _mixed_max(a: np.ndarray, g: np.ndarray, t: float) -> float:
    """
    Return the peak of the autocorrelation of the mixture (1‑t)·a + t·g.
    """
    h = (1.0 - t) * a + t * g
    M, _ = _conv_max_fft(h)
    return M


# ----------------------------------------------------------------------
def _optimal_mix_t(a: np.ndarray, g: np.ndarray, deadline: float,
                   max_iter: int = 12) -> float:
    """
    Ternary search on t∈[0,1] to minimise the peak autocorrelation of
    the mixture (1‑t)·a + t·g.  Stops early if the time budget is exhausted.
    """
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        M1 = _mixed_max(a, g, t1)
        M2 = _mixed_max(a, g, t2)
        if M1 < M2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0


# ----------------------------------------------------------------------
def _solve_lp_for_g(a: np.ndarray, M: float, active: np.ndarray,
                    target_sum: float):
    """
    Solve the linear programme

        maximise   Σ g_i
        subject to (a ⋆ g)[k] ≤ M   for k in active,
                   0 ≤ g_i ≤ target_sum.

    Returns the scaled vector g (ℓ₁‑norm = target_sum) or ``None`` if the
    LP fails or yields a zero solution.
    """
    n = a.shape[0]
    num_cons = active.shape[0]

    # Build the constraint matrix (each row corresponds to a shift k)
    A = np.zeros((num_cons, n), dtype=np.float64)
    for row, k in enumerate(active):
        # indices j where 0 ≤ j < n and i = k‑j satisfies 0 ≤ i < n
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        i_vals = k - np.arange(j_start, j_end + 1)          # i = k‑j
        A[row, j_start:j_end + 1] = a[i_vals]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  → minimise -Σ g_i
    b_ub = np.full(num_cons, M, dtype=np.float64)
    bounds = [(0.0, target_sum) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None

    if not res.success:
        return None

    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return None
    g = g0 * (target_sum / s)               # renormalise to the required ℓ₁‑norm
    return g


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray, target_sum: float,
                  steps: int, deadline: float):
    """
    Very cheap stochastic local search: repeatedly move a tiny amount of
    mass from one random index to another if the move reduces the peak
    autocorrelation.
    Returns the best vector found and its peak value.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    n = a.shape[0]
    for _ in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] <= 0.0:
            continue
        delta = best[i] * 0.02                     # move 2 % of the source mass
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is preserved, non‑negativity holds automatically
        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M
    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best list of heights found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a previously discovered construction if available
    base_seq = None
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore
        print("[info] loaded previous construction, length =", base_seq.shape[0])
    except Exception:
        print("[info] no previous construction found.")

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None               # type: np.ndarray | None
    best_obj = float("inf")       # this is the peak of the autocorrelation (M)

    # ------------------------------------------------------------------
    # Lengths to explore (keep them modest to stay within the time budget)
    # Original length is 2100 in the reference run.
    length_factors = [1.0, 1.2, 1.5, 2.0]          # multiplicative factors
    lengths = sorted(
        {max(300, min(12000, int(2100 * f))) for f in length_factors}
    )
    print("[info] exploring lengths:", lengths)

    # ------------------------------------------------------------------
    # Main optimisation loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)           # required ℓ₁‑norm
        max_restarts = 3 if n <= 4000 else 2

        for restart in range(max_restarts):
            if time.time() >= deadline:
                break

            # ---- initialise -------------------------------------------------
            if base_seq is not None and n == base_seq.shape[0] and restart == 0:
                a = base_seq.astype(float).copy()
                a *= (target_sum / a.sum())
                print("[init] using loaded construction for n =", n)
            else:
                # random exponential start (positive entries)
                a = np.random.exponential(scale=1.0, size=n)
                a *= (target_sum / a.sum())

            # optional: blend with current global best if same length
            if best_seq is not None and best_seq.shape[0] == n:
                a = 0.3 * best_seq + 0.7 * a
                a = np.maximum(a, 0.0)
                a *= (target_sum / a.sum())

            print(f"[run] length={n}, restart={restart + 1}/{max_restarts}")

            # ---- iterative improvement ---------------------------------------
            # we cap the number of inner iterations loosely – the loop also
            # stops when the wall‑clock budget runs out.
            iter_cnt = 0
            while time.time() < deadline and iter_cnt < 5000:
                iter_cnt += 1

                # 1. current autocorrelation peak
                M, conv = _conv_max_fft(a)

                # update global best if improved
                if M < best_obj:
                    best_obj = M
                    best_seq = a.copy()
                    print(f"[global] new best M={M:.6f} (len={n}, iter={iter_cnt})")

                # early exit if we have already reached the target bound
                if best_obj <= 1.503:
                    print("[info] target bound reached, stopping early.")
                    deadline = time.time() + 1.0   # force outer loops to finish soon
                    break

                # 2. choose active constraints (shifts where conv is close to M)
                # we take at most max_active indices to keep the LP small
                max_active = min(200, n)
                # tolerance relative to M
                tol = max(1e-12, M * 1e-6)
                active = np.where(conv >= M - tol)[0]
                if active.shape[0] > max_active:
                    # keep only the top‑max_active values
                    idx = np.argsort(conv)[-max_active:]
                    active = idx
                # ensure at least one constraint (the max itself)
                if active.size == 0:
                    active = np.array([np.argmax(conv)], dtype=int)

                # 3. solve LP to obtain a candidate g
                g = _solve_lp_for_g(a, M, active, target_sum)

                if g is None:
                    # LP failed – perform a tiny random jitter and continue
                    jitter = np.random.randn(n) * 0.0005 * target_sum / n
                    a = np.maximum(a + jitter, 0.0)
                    a *= (target_sum / a.sum())
                    continue

                # 4. find the optimal mixing coefficient t
                t_opt = _optimal_mix_t(a, g, deadline, max_iter=12)

                # 5. update the current vector
                a = (1.0 - t_opt) * a + t_opt * g
                # the convex combination stays non‑negative; renormalise to target_sum
                a *= (target_sum / a.sum())

                # 6. occasional jitter to escape plateaus
                if iter_cnt % 400 == 0:
                    jitter = np.random.randn(n) * 0.001 * target_sum / n
                    a = np.maximum(a + jitter, 0.0)
                    a *= (target_sum / a.sum())

            # ---- cheap local refinement ------------------------------------
            if time.time() < deadline and best_seq is not None:
                # start local search from the final vector of this restart
                local_seq, local_val = _local_search(a, target_sum,
                                                     steps=20000,
                                                     deadline=deadline)
                if local_val < best_obj:
                    best_obj = local_val
                    best_seq = local_seq.copy()
                    print(f"[local] improved best M={local_val:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_obj = 2.0      # uniform sequence peak

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the official function
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best peak M = {best_obj:.6f}, official score = {final_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
