def _solve_lp_for_direction(a, M):
    n = a.shape[0]
    # Build sparse matrix A of shape (2n-1, n)
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        rows = np.arange(i, i+n)
        row_ind.extend(rows)
        col_ind.extend([i]*n)
        data.extend(a.tolist())
    A = scipy.sparse.csr_matrix((data, (row_ind, col_ind)), shape=(2*n-1, n))
    # define upper bound vector b_ub = np.full(2*n-1, M)
    res = scipy.optimize.linprog(c = -np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if not res.success:
        return None
    b = res.x
    return b

"""
Propose a non‑negative step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines two complementary ideas:

1. **Linear‑programming improvement (old idea)**  
   For the current sequence `a` we compute M = max(a*a).  
   Keeping `a` fixed, the constraint `a * b <= M` is *linear* in `b`.  
   We therefore maximise Σ b_i under `b ≥ 0` and `a * b ≤ M` with a single
   LP (SciPy HiGHS).  The optimal `b` tends to increase the total mass while
   respecting the current worst‑case convolution value.  After scaling `b`
   to the required simplex (Σ b = √(2n)) we mix it with `a` (convex
   combination) and repeat a few iterations.  This quickly reduces the
   L∞‑norm of the self‑convolution.

2. **Smooth‑max gradient fine‑tuning**  
   After the LP stage we run a short soft‑max (smooth‑max) sub‑gradient
   descent (the routine from the previous solution) to polish the result.

Multiple independent restarts with random lengths, power‑law initialisation,
and occasional warm‑starts from the global best are performed until the
time budget (≈ 1000 s) is exhausted.  The best sequence found is returned.

All helper functions are top‑level; no closures, no file I/O, and only
NumPy/SciPy are used.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Linear‑programming step: given a and current max M, find a direction b
# ----------------------------------------------------------------------
def _solve_lp_direction(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise Σ b_i
        subject to  b ≥ 0      and  (a * b)[k] ≤ M   for all k
    where '*' denotes linear convolution.
    Returns the optimal b (or None on failure).
    """
    n = a.shape[0]
    # Build sparse convolution matrix A such that A @ b = a * b (full conv)
    # For column i, a[i] contributes to rows i … i+n‑1.
    # rows = i + j, cols = i, data = a[i]   (j = 0..n‑1)
    # Use vectorised construction.
    cols = np.repeat(np.arange(n), n)                     # length n*n
    offsets = np.tile(np.arange(n), n)                    # 0..n‑1 repeated n times
    rows = cols + offsets                                 # row indices
    data = np.repeat(a, n)                               # a[i] repeated n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Upper bounds vector for the constraints
    ub = np.full(2 * n - 1, M, dtype=float)

    # Linear program (HiGHS) – maximise Σ b ⇒ minimise -Σ b
    res = opt.linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=ub,
        bounds=(0, None),
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# LP‑based improvement of a single sequence
# ----------------------------------------------------------------------
def _lp_improve(
    a: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
) -> np.ndarray:
    """
    Starting from a, repeatedly solve the LP described above,
    scale the solution to the simplex, and mix it back into a.
    Stops after `max_iters` or when the time budget expires.
    """
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution_fft(a, a)
        M = conv.max()

        b = _solve_lp_direction(a, M)
        if b is None:
            # LP failed – abort the LP phase
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Scale b to meet the simplex constraint Σ a = target_sum
        b = b * (target_sum / sum_b)

        # Mixing coefficient – larger early, smaller later
        t = 0.5 if it < 10 else 0.2

        a_new = (1.0 - t) * a + t * b
        a_new = _project_onto_simplex(a_new, target_sum)

        # Check actual improvement (using the true max of the convolution)
        conv_new = _full_convolution_fft(a_new, a_new)
        if conv_new.max() >= M - 1e-12:
            # No real improvement – shrink step size and try again
            t *= 0.5
            if t < 1e-4:
                break
            a = (1.0 - t) * a + t * b
            a = _project_onto_simplex(a, target_sum)
        else:
            a = a_new

    return a


# ----------------------------------------------------------------------
# Soft‑max gradient descent (the previous approach) – retained for polishing
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9995,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Perform soft‑max sub‑gradient descent for a fixed length ``n``.
    Returns the best sequence found and its evaluation score.
    """
    target_sum = np.sqrt(2.0 * n)  # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # smooth‑max weights
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of Σ w_k conv_k  (cross‑correlation)
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # positions for gradient
        grad = 2.0 * conv2[idx]

        # momentum
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        # Adaptive step size
        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate locally (faster than full evaluate_sequence)
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional random shake‑up
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score
