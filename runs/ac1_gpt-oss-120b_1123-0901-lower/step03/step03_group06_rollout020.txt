def hill_climb(p, n, time_deadline, max_steps=20000):
    best_score = compute_score(p, n)
    p_cur = p.copy()
    for step in range(max_steps):
        if time.time() >= time_deadline: break
        # random pick two indices i != j
        i, j = np.random.choice(n, 2, replace=False)
        # transfer amount delta
        # max we can take from i is p_cur[i]
        # choose delta = p_cur[i] * np.random.rand() * factor, maybe factor = 0.1
        if p_cur[i] == 0: continue
        delta = p_cur[i] * 0.1 * np.random.rand()
        # apply move
        p_new = p_cur.copy()
        p_new[i] -= delta
        p_new[j] += delta
        # compute new score
        score = compute_score(p_new, n)
        if score < best_score:
            best_score = score
            p_cur = p_new
    return p_cur, best_score

"""
Propose a low‑peak self‑convolution step function.

Algorithm
---------
* Work with the *probability* vector `p` (sum = 1) and the objective
      f(p) = 2·n·max(p∗p).
  The actual heights are a = √(2·n)·p, which makes the evaluate_sequence()
  call return exactly f(p).

* For a fixed length `n` we run Exponentiated‑Gradient (mirror descent on the
  simplex) on a *smooth* version of the max:
      M_β(p) = (1/β)·log Σ_k exp(β·(conv_k(p)−M(p)))  (M = max conv)
  The gradient of M_β is a correlation of `p` with the soft‑max weights.
  We compute it by FFT (O(n log n)) and update `log(p)`.

* The smoothing parameter β is increased linearly from a tiny value to a
  large one (≈ 150) so that the algorithm first explores and then focuses
  on the true maximum.

* After the EG phase we perform a cheap stochastic hill‑climbing phase:
  repeatedly move a small amount of probability mass from a random index
  to another and keep the move only if the objective improves.

* A multiscale schedule is used.  Starting from the provided construction
  (global variable `height_sequence_1`) we repeatedly up‑sample the current
  best probability vector to a larger length and re‑optimise.  The
  schedule explores lengths up to 8000; the time budget is shared
  automatically between stages.

* All intermediate scores are computed directly from the convolution
  (no calls to evaluate_sequence) for speed; the final sequence is
  returned in the original scale (`a = √(2·n)·p`).

The whole search runs within the given time budget (≈ 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Full linear self‑convolution (p∗p) using a single FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k·p_{k‑i}   for i=0…n‑1.
    It is a correlation of `p` with the reversed weight vector `w`.
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]   # keep linear part
    rev = conv_rev[L - n : L][::-1]    # extract the needed slice
    return rev


def _score_from_p(p: np.ndarray, n: int) -> float:
    """Objective 2·n·max(p∗p) for a probability vector p."""
    N = _next_pow2(2 * n - 1)
    conv = _full_convolution_fft(p, N)
    return 2.0 * n * conv.max()


def _eg_optimize(
    p_start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector and its score.
    """
    # hyper‑parameters
    beta_start = 0.02
    beta_end = 150.0
    beta_len = max_iters // 2          # linear schedule length
    jitter_interval = 2000
    jitter_scale = 0.001

    p = p_start.copy()
    logp = np.log(p + 1e-30)           # stay in log‑space

    N = _next_pow2(3 * n - 2)          # big enough for conv and correlation
    best_score = _score_from_p(p, n)
    best_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution and current score -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[EG n={n:5d} iter={it:7d}] new local best = {cur_score:.6f}")

        # ----- smooth max weights (normalized) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w_raw = np.exp(beta * (conv - M))
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum > 0 else w_raw

        # ----- correlation (rev) -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smooth objective -----
        grad = 4.0 * n * rev   # ∇_p (2·n·M_β)

        # ----- adaptive step size (simple decay) -----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # ----- EG update in log‑space -----
        logp -= eta * grad

        # ----- jitter to escape plateaus -----
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)       # avoid overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


def _hill_climb(
    p_start: np.ndarray,
    n: int,
    deadline: float,
    max_steps: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill climbing: move a small amount of mass between
    two random entries if the move decreases the objective.
    """
    p = p_start.copy()
    best_score = _score_from_p(p, n)

    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        # pick two different indices
        i, j = np.random.choice(n, 2, replace=False)
        if p[i] <= 1e-12:
            continue
        # propose moving a fraction of p[i] to p[j]
        delta = p[i] * 0.1 * np.random.rand()
        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta
        # no need to renormalise – the move preserves the total mass
        new_score = _score_from_p(p_new, n)
        if new_score < best_score:
            best_score = new_score
            p = p_new
            # occasional log
            if step % 1000 == 0:
                print(f"[HC n={n:5d} step={step:5d}] improved score = {new_score:.6f}")

    return p, best_score


def _upsample(p_old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_x = np.linspace(0.0, 1.0, len(p_old))
    new_x = np.linspace(0.0, 1.0, new_n)
    p_new = np.interp(new_x, old_x, p_old)
    p_new = np.maximum(p_new, 0.0)
    p_new /= p_new.sum()
    return p_new
