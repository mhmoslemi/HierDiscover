"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.  The code is a two‑phase optimiser:

* Phase‑1: Adam on the logits of a soft‑max (ensuring Σa = 1) with a
  smooth‑max surrogate that slowly turns into the true max.
* Phase‑2: a cheap hill‑climber that moves a small amount of mass
  between random positions (a Simulated‑Annealing style refinement).

Both phases use FFT‑based (full) convolution and cross‑correlation,
so a single iteration is fast even for n = 8192.  The optimiser is
restarted many times with random Dirichlet starts and, when possible,
warm‑starts from the best sequence found so far (or from the global
variable ``height_sequence_1``).  The best sequence encountered before
the time budget expires is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    for i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective_from_conv(max_c: float, n: int) -> float:
    """
    Exact objective for a sequence whose Σa = 1.
    (The denominator of evaluate_sequence is then 1.)
    """
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    The routine respects the given time budget, prints intermediate
    progress and finally returns the best sequence (as a plain Python
    list) that it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Helper: initialise logits from a probability vector
    def _logits_from_probs(p: np.ndarray) -> np.ndarray:
        return np.log(p + 1e-15)

    # --------------------------------------------------------------
    # Problem sizes to explore – the objective does not depend on n
    # except through the scaling factor, so a moderate size (1024) is
    # a good trade‑off between expressiveness and speed.
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (powers of two) – one for the self‑convolution,
        # another for the cross‑correlation needed for the gradient.
        fft_len_conv = 1 << int(math.ceil(math.log2(2 * n - 1)))
        fft_len_corr = 1 << int(math.ceil(math.log2(3 * n)))  # w length + a length - 1

        # Number of independent restarts – we keep this modest for the
        # larger lengths, but for n=1024 we spend many iterations.
        restarts = 5 if n > 1024 else 12

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---------- initialise logits ---------------------------------
            if r == 0 and best_seq is not None and len(best_seq) == n:
                init_a = np.array(best_seq, dtype=np.float64)
                init_a /= init_a.sum()
                x = _logits_from_probs(init_a)
                print("[restart 0] warm start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= s
                x = _logits_from_probs(cand)
                print("[restart 0] start from height_sequence_1")
            else:
                # pure random Dirichlet start
                a0 = np.random.dirichlet(np.ones(n))
                x = _logits_from_probs(a0)

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04

            # Smooth‑max temperature (λ).  It starts small and grows
            # geometrically; once it exceeds a threshold we switch to a
            # hard indicator (exact sub‑gradient of the max).
            lam = 0.5
            lam_factor = 1.02
            lam_hard_threshold = 1e5  # after this we use the exact max

            t = 0
            last_improve = time.time()

            # Upper bound on iterations – the loop will also stop when
            # the budget runs out.
            max_iters = 300000

            while time.time() < deadline and t < max_iters:
                # ----- logits → probabilities (softmax) -----
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()          # a_i ≥ 0, Σ a_i = 1

                # ----- self‑convolution -------------------------------------------------
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()

                # ----- smooth‑max weights -------------------------------------------------
                if lam < lam_hard_threshold:
                    # smooth surrogate (soft‑max) – numerically stable
                    w = np.exp(lam * (conv - max_c))
                    w /= w.sum()
                else:
                    # exact sub‑gradient: uniform weight on all maximisers
                    mask = conv >= max_c - 1e-12
                    w = np.zeros_like(conv)
                    cnt = np.count_nonzero(mask)
                    if cnt == 0:                 # should never happen
                        w[conv.argmax()] = 1.0
                    else:
                        w[mask] = 1.0 / cnt

                # ----- gradient of the surrogate w.r.t. a ---------------------------------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # ----- back‑propagation through the soft‑max (gradient w.r.t. logits) -----
                avg = np.dot(a, grad_a)            # scalar ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)        # ∇_x φ

                # ----- Adam update -------------------------------------------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule ------------------------------------------------
                lam = min(lam * lam_factor, 1e9)

                # ----- exact objective ----------------------------------------------------
                val = _objective_from_conv(max_c, n)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best: {val:.6f}")

                # ----- stagnation handling -------------------------------------------------
                if time.time() - last_improve > 5.0:
                    # small random jitter to escape possible plateaus
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] jitter due to stagnation")

    # ----------------------------------------------------------------------
    # Phase‑2 : simple Monte‑Carlo hill‑climber / simulated annealing
    # ----------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
        fft_len_conv = 1 << int(math.ceil(math.log2(2 * n - 1)))
        # compute the current convolution once
        a = best_seq.copy()
        conv = _full_convolution(a, fft_len=fft_len_conv)
        cur_val = _objective_from_conv(conv.max(), n)

        #   tweak parameters
        max_refine_steps = 200000
        T_start = 1e-2
        T_end = 1e-5
        t_ref = 0

        while time.time() < deadline and t_ref < max_refine_steps:
            # linear cooling schedule
            T = T_start + (T_end - T_start) * (t_ref / max_refine_steps)

            # propose a tiny mass transfer between two random positions
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = (np.random.rand() * 0.02) * a[i]   # up to 2 % of a[i]
            if delta == 0.0:
                continue
            new_a = a.copy()
            new_a[i] = max(0.0, new_a[i] - delta)
            new_a[j] += delta
            # renormalise – tiny numerical drift may occur
            new_a /= new_a.sum()

            new_conv = _full_convolution(new_a, fft_len=fft_len_conv)
            new_val = _objective_from_conv(new_conv.max(), n)

            diff = new_val - cur_val
            if diff < 0 or np.exp(-diff / max(T, 1e-12)) > np.random.rand():
                a = new_a
                conv = new_conv
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[refine step {t_ref:06d}] improved to {best_val:.6f}")

            t_ref += 1

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (unlikely)
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    return best_seq.tolist()
