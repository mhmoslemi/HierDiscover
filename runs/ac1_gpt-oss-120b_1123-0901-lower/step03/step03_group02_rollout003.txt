    deadline = start + budget_s - 0.1

def _optimise_adam_softmax(n, x0, target_sum, run_deadline,
                           init_tau_factor=0.12, min_tau_factor=0.001,
                           alpha=0.05, beta1=0.9, beta2=0.999,
                           eps=1e-8):
    a = None
    best_seq = None
    best_score = float('inf')
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    iteration = 0
    while time.time() < run_deadline:
        # compute softmax p
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)   # subtract max for numeric stability
        sum_exp = np.sum(exp_x)
        p = exp_x / sum_exp
        a = target_sum * p

        # conv
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        score = max_val   # because target_sum ensures evaluation = max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # optional print
            # print(f"[{time.time():.2f}] n={n} local best {score:.6f}")

        # compute smoothed weights
        tau = max(init_tau_factor * max_val * (0.95**iteration), min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a:
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length 3n-2
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]    # shape n

        # convert to gradient w.r.t x
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam updates
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        # step size might be scaled by decreasing schedule:
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        x = x - step

        # optional: add small random perturbation for exploration
        if iteration % 200 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # optionally decay alpha:
        alpha *= 0.9995

    # final evaluation using evaluate_sequence
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation simplifies to `max(conv(a, a))` if we keep the sum
fixed to √(2 n).  The optimisation therefore seeks a distribution
with a flat self‑convolution.

Algorithm
---------
* **Parameterisation** – a is written as a soft‑max of an unconstrained
  vector `x`:  a_i = S·exp(x_i)/∑exp(x)  with  S = √(2 n).  This enforces
  non‑negativity and the simplex constraint without a projection step.
* **Objective surrogate** – we smooth the max using a temperature τ and
  optimise the weighted sum Σ w_k·c_k where c = a*a and
  w_k ∝ exp((c_k – max(c))/τ).  As τ → 0 the surrogate approaches the
  true max.
* **Gradient** – the gradient w.r.t. a of the weighted sum is a
  cross‑correlation `grad_a = 2·(a ⊛ w_rev)`.  Using the chain rule we
  obtain the gradient w.r.t. the log‑it variables x.
* **Adam optimisation** – an Adam optimiser on `x` (with a modest
  learning‑rate schedule) provides adaptive step sizes and fast
  convergence.
* **Multi‑restart search** – many independent runs with random lengths
  (≈ 500 – 3500) and random power‑law initialisations are performed.
  The best sequence found across all runs is returned.  The current
  global best (if any) is occasionally injected as a warm‑start.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed‑max surrogate on the log‑it space
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it variables.
    Returns the best sequence encountered (as a NumPy array) and its
    evaluation score obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ---- 1. map logits to the simplex (a = S·softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # improve numerical stability
        p = exp_shift / np.sum(exp_shift)      # probabilities, sum = 1
        a = target_sum * p                     # a lives on the simplex

        # ---- 2. convolution and current (smooth) max ----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # because sum(a)=√(2n), the true evaluation equals max_val
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ---- 3. smooth‑max weights ---------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4. gradient w.r.t. a ---------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # positions that give Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- 5. gradient w.r.t. logits x (chain rule) -----------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- 6. Adam update ---------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate
        alpha *= 0.9995

    # Final evaluation using the provided evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly choose a length ``n`` (≈ 500 – 3500) and initialise a
      power‑law shape a_i ∝ (i+1)^{-p} (p ∈ [0.25,0.75]), optionally
      tapered by an exponential factor.
    * Optimise the shape with the Adam‑based soft‑max descent defined above.
    * Keep the overall best across many independent restarts.
    * Occasionally warm‑start from the currently best known sequence.
    * Stop early if we beat the target bound (≈ 1.503).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:

        if score < best_score:

            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size. Early on we explore a broad range,
        # later we concentrate to keep iterations cheap.
        if remaining > 300:
            n = np.random.randint(1200, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(800, 2500 + 1)
        else:
            n = np.random.randint(500, 2000 + 1)

        # ------------------- initial shape ----------------------------

        idx = np.arange(1, n + 1, dtype=float)

                init = bs[:n].copy()
            else:

                init = np.concatenate([bs, pad])

        # ensure strictly positive values for the log‑it conversion
        init = np.maximum(init, 1e-12)

        # ------------------------------------------------------------------

        init = init / init.sum() * target_sum        # scale to the correct sum
        x0 = np.log(init + 1e-12)                    # logits (avoid log(0))

        # Allocate a time slice for this run (≈ 8‑12 s, never exceed deadline)
        slice_time = min(12.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            alpha=0.05,
            beta1=0.9,

            best_seq = seq_local.copy()

        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
