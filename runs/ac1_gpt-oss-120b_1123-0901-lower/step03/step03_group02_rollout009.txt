def _optimise_one_length_subgradient(
    n,
    init,
    run_deadline,
    *,
    init_step_factor=0.3,
    step_decay=0.9999,
    momentum_beta=0.8,
    active_tol=1e-9,
    min_step_factor=1e-6,
):
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)
    best_seq = a.copy()
    conv = _full_convolution_fft(a,a)
    best_val = conv.max()  # equals objective
    best_score = float(2 * n * best_val / (target_sum**2))  # should equal best_val

    # initial step factor scale
    step_factor = init_step_factor
    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # active indices: treat near max
        tol = max(active_tol * max_val, 1e-12)
        active_idx = np.where(conv >= max_val - tol)[0]
        if active_idx.size == 0:
            # fallback: use all indices (unlikely)
            active_idx = np.arange(conv.size)
        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        # compute gradient
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)  # length = 3n - 2
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        # momentum
        if momentum_beta > 0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            direction = velocity
        else:
            direction = grad

        # step size based on current max and gradient magnitude
        max_grad = np.max(np.abs(direction)) + 1e-15
        eta = step_factor * max_val / max_grad

        # line search (simple backtracking)
        improved = False
        step = eta
        while step > min_step_factor * target_sum:
            a_new = a - step * direction
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_onto_simplex(a_new, target_sum)

            conv_new = _full_convolution_fft(a_new, a_new)
            new_max = conv_new.max()

            if new_max < max_val - 1e-12 * max_val:
                # improvement
                a = a_new
                max_val = new_max
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            # small step not improving, maybe stuck; reduce step_factor
            step_factor *= 0.7

        # update best
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_seq = a.copy()
            best_score = float(2 * n * max_val / (target_sum**2))
            print(f"[{time.time():.2f}] n={n} new local best = {best_score:.6f}")

        # decay step factor gradually
        step_factor *= step_decay
        iteration += 1

        # random shake up occasionally
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

"""
Sub‑gradient optimisation for the self‑convolution maximum.

We minimise
        2·n·max(conv(a,a)) / (sum(a)**2)

by keeping Σ a_i = √(2 n) (so the objective equals max(conv(a,a))).
For a fixed length *n* we run a projected sub‑gradient method:
 * The active set of convolution indices attaining (or almost attaining) the
   maximum defines a piecewise‑linear sub‑gradient.
 * The gradient is obtained efficiently by one FFT‑based convolution.
 * A simple back‑tracking line‑search guarantees a monotone decrease.
 * After each update we project back onto the simplex {a≥0, Σa = √(2 n)}.
 * Momentum, occasional random “shake‑ups”, and a variety of random
   initialisations (power‑law, exponential taper, sparse spikes, uniform)
   improve robustness.

The optimiser is called repeatedly with different lengths and random
starts; the best sequence found before the global deadline (≈1000 s) is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # discard tiny negatives
    return conv


# ----------------------------------------------------------------------
# Optimisation of a single length using a projected sub‑gradient method
# ----------------------------------------------------------------------
def _optimise_one_length_subgradient(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_step_factor: float = 0.30,
    step_decay: float = 0.9998,
    momentum_beta: float = 0.80,
    active_tol: float = 1e-9,
    min_step_factor: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv(a,a)) for a fixed length `n`.
    Returns the best sequence (as a NumPy array) and its score obtained via
    `evaluate_sequence`.
    """
    target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # best found locally
    conv = _full_convolution_fft(a, a)
    best_val = conv.max()
    best_seq = a.copy()
    best_score = float(2.0 * n * best_val / (target_sum ** 2))

    velocity = np.zeros_like(a)                 # momentum term
    step_factor = init_step_factor
    iteration = 0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # ---- active set (indices where conv is (almost) maximal) ----
        tol = max(active_tol * max_val, 1e-12)
        active_idx = np.where(conv >= max_val - tol)[0]
        if active_idx.size == 0:                 # safety fallback
            active_idx = np.arange(conv.size)
        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size     # uniform weight on the active set

        # ---- gradient of Σ w_k conv_k (sub‑gradient of max) ----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)  # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)         # positions giving Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # ---- momentum ----
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            direction = velocity
        else:
            direction = grad

        # ---- step size (scaled to current max value) ----
        max_grad = np.max(np.abs(direction)) + 1e-15
        eta = step_factor * max_val / max_grad

        # ---- back‑tracking line search ensuring descent ----
        improved = False
        step = eta
        while step > min_step_factor * target_sum:
            a_new = a - step * direction
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_onto_simplex(a_new, target_sum)

            conv_new = _full_convolution_fft(a_new, a_new)
            new_max = conv_new.max()

            if new_max < max_val - 1e-12 * max_val:
                a = a_new
                max_val = new_max
                improved = True
                break
            step *= 0.5

        if not improved:
            # no improvement – shrink the global step factor a little
            step_factor *= 0.7

        # ---- record local best ----
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_seq = a.copy()
            best_score = float(2.0 * n * max_val / (target_sum ** 2))
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")

        # ---- decay the step factor ----
        step_factor *= step_decay
        iteration += 1

        # ---- occasional random shake‑up to escape plateaus ----
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    # final evaluation through the provided evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score
