"""
Hybrid gradient‑descent + simulated‑annealing search for a step‑function
that minimises ``evaluate_sequence`` (i.e. max(a∗a) after normalising
∑a = √(2 n)).  The algorithm works on several sequence lengths, runs a
soft‑max Adam optimiser (as in the previous baseline) and then refines
the result with a cheap stochastic mass‑transfer annealer.  Multiple
restarts and aggressive jitter are used to escape local minima.
The whole search respects a 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable softmax of convolution values (weights sum to 1)."""
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, conv.max()
    maxc = conv.max()
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i} (indices outside range are ignored)
    using a single FFT convolution.
    """
    # FFT of a (already zero‑padded)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # Keep only the valid part (length n + len(w) - 1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # Extract the slice that corresponds to rev_i for i=0…n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _full_conv_max(a: np.ndarray, Nfft: int):
    """Return max(a∗a) using an FFT (full convolution, length 2n‑1)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_global_seq: np.ndarray,
    best_global_score: float,
    max_iters: int,
):
    """
    Adam optimiser that minimises the smooth approximation of max(a∗a).
    Returns (best_seq, best_score) found during this run.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # enforce exact L1‑norm
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.07
    iteration = 0
    jitter_interval = 4000

    # soft‑max temperature schedule
    tau_factor = 0.25       # initial τ = τ_factor·M
    decay_rate = 15000.0   # exponential decay constant

    Nfft = 1 << ((2 * n - 1).bit_length())

    best_seq = best_global_seq.copy()
    best_score = best_global_score

    while time.time() < deadline and iteration < max_iters:
        # ---- exact convolution and current max -------------------------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- smooth max (log‑sum‑exp) ---------------------------------
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # ---- weighted reverse convolution (gradient of smooth max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ---- Adam update -----------------------------------------------
        coeff = 4.0 * n / (target_sum * target_sum)      # d/d a of (2n*conv/target_sum²)
        const = 4.0 * n * Ltau / (target_sum ** 3)      # normalisation term’s grad
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto feasible set (a≥0, ∑a = √(2n)) ------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---- occasional jitter -----------------------------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---- update global best (exact objective equals M) -------------
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n} iter={iteration}] new best = {best_score:.6f}")

    return best_seq, best_score


def _simulated_annealing(
    a0: np.ndarray,
    init_score: float,
    deadline: float,
    best_global_seq: np.ndarray,
    best_global_score: float,
    max_steps: int = 20000,
):
    """
    Very cheap SA that repeatedly transfers a random amount of mass between
    two coordinates while keeping the L1‑norm fixed.
    """
    a = a0.copy()
    cur_score = init_score
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # pre‑compute FFT size for fast convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_seq = best_global_seq.copy()
    best_score = best_global_score

    # annealing schedule – start relatively hot, cool geometrically
    T0 = max(0.05, (cur_score - 1.0) * 0.5)   # heuristic
    T = T0
    decay = 0.9995

    rng = np.random.default_rng()

    step = 0
    while time.time() < deadline and step < max_steps:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a fraction of a[i] to a[j]
        max_move = a[i]
        if max_move == 0.0:
            continue
        delta = rng.random() * max_move * 0.5   # up to 50 % of a[i]

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate new max convolution
        new_score = _full_conv_max(a_new, Nfft)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_new
            cur_score = new_score
            # update best if improved
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[SA n={n} step={step}] new best = {best_score:.6f}")

        # cooling
        T *= decay
        step += 1

    return best_seq, best_score

    # Lengths to try – we cap growth to keep FFT size reasonable.
    # ------------------------------------------------------------------
    length_factors = [1.0, 1.5, 2.0, 3.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    max_len = 7000
    lengths = [min(l, max_len) for l in lengths if l >= 300]
    lengths = sorted(set(lengths))
    print(f"[plan] exploring lengths: {lengths}")

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # Prepare an initial shape for the current length.
        if n == n0:
            shape = base.copy()
        else:
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            shape = np.interp(new_x, old_x, base)
            shape = np.maximum(shape, 0.0)

        # Normalise to the required L1‑norm.
        target_sum = math.sqrt(2.0 * n)
        shape = shape * (target_sum / shape.sum())

        # Number of Adam iterations – more for smaller n.
        max_adam_iters = int(3_000_000 * 2100 / n)
        max_adam_iters = max(400_000, max_adam_iters)   # at least a few hundred k

        # Number of SA steps (scale with n, but bounded).
        max_sa_steps = int(30_000 * 2100 / n)
        max_sa_steps = max(5_000, min(max_sa_steps, 30_000))

        # ------------------------------------------------------------------
        # Several restarts per length (one deterministic, rest random).
        # ------------------------------------------------------------------
        restarts = 2
        for r in range(restarts):
            if time.time() >= deadline:
                break

            if r == 0:
                a0 = shape.copy()
            else:
                # random start: Dirichlet distributed (positive, sum constraint)
                rand = np.random.gamma(1.0, 1.0, size=n)
                a0 = rand * (target_sum / rand.sum())

            # ---- Adam optimisation ---------------------------------------
            a_opt, sc_opt = _adam_optimize(
                a0,
                deadline,
                best_seq,
                best_score,
                max_iters=max_adam_iters,
            )

            # ---- Simulated annealing refinement ---------------------------
            # Evaluate exact score for the Adam result (max conv).
            Nfft = 1 << ((2 * n - 1).bit_length())
            exact_score = _full_conv_max(a_opt, Nfft)

            a_sa, sc_sa = _simulated_annealing(
                a_opt,
                exact_score,
                deadline,
                best_seq,
                best_score,
                max_steps=max_sa_steps,
            )

            # Update global best if needed
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.copy()
                print(f"[global] new overall best = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None or len(best_seq) == 0:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    # Return as plain Python list for the harness.
    return best_seq.tolist()
