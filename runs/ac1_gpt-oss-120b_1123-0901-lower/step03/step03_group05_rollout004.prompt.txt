    sequence = [max(0, x) for x in sequence]

def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation a*a via FFT and returns the maximal value,
    its index and the whole convolution array.
    """
    n = a.shape[0]
    # n_fft is the next power of two >= 2*n-1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv

def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, returns the vector rev where rev[i] = a[k-i]
    if the index is inside the support, otherwise 0.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev
