    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = np.inf

    # hyperparams
    lengths = [1024, 2048, 4096, 8192]
    # functions to generate init proposals
    init_funcs = [init_uniform, init_exponential_left, init_exponential_right, init_symmetric_piecewise, init_random_dirichlet]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== n={n} ===")
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))
        # Determine approx number of iterations given time left:
        remaining = deadline - time.time()
        # allocate ~remaining * proportion.
        # We'll aim to run at most 50000 iters per restart, but limited by time.
        for init_fn in init_funcs:
            if time.time() > deadline:
                break
            # create initial probabilities
            a0 = init_fn(n)
            # warm start from best if same length
            if best_seq is not None and len(best_seq) == n:
                a0 = np.array(best_seq)
            # convert to logits
            x = np.log(a0 + 1e-15)
            # Adam states
            m = np.zeros(n)
            v = np.zeros(n)
            beta1=0.9; beta2=0.999; eps=1e-8
            lr = 0.1
            lam = 0.5
            lam_factor = 1.05
            iter = 0
            last_improve = time.time()
            max_iters = 60000  # safety cap
            while time.time() < deadline and iter < max_iters:
                # softmax probability
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()
                # conj and conv
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                # compute smooth or sub gradient
                if lam < 1e4:
                    # softmax weights
                    w = np.exp(lam * (conv - max_c))
                    w /= w.sum()
                else:
                    # subgradient: uniform over active set (within tiny tol)
                    tol = 1e-12
                    mask = conv >= max_c - tol
                    w = mask.astype(float)
                    w /= w.sum()
                # grad a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)
                # backprop through softmax to logits
                avg = np.dot(a, grad_a)
                grad_x = a * (grad_a - avg)
                # Adam update
                iter += 1
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
                m_hat = m / (1 - beta1 ** iter)
                v_hat = v / (1 - beta2 ** iter)
                # maybe decay lr with lam
                adj_lr = lr / (1 + lam*0.01)
                x = x - adj_lr * m_hat / (np.sqrt(v_hat) + eps)
                # schedule lambda

                # exact objective

                    best_seq = a.copy()

                    last_improve = time.time()
                    print(f"[n={n} init={init_fn.__name__} iter={iter}] shake")

        # hill climbing steps
        a = np.array(best_seq)
        val_best = 2.0 * n * _full_convolution(a, fft_len=fft_len_conv).max()
        start_time = time.time()
        while time.time() < deadline and time.time() - start_time < max(10, (deadline - time.time())/2):

            x[idx] += delta
            # recompute a

            conv = _full_convolution(a_new, fft_len=fft_len_conv)
            val = 2.0 * n * conv.max()
            if val < val_best:

                val_best = val
                best_seq = a.copy()

"""
Hybrid optimization for the step‑function objective

The function `propose_candidate` searches for a non‑negative vector ``a``
(minimising  F(a)=2·n·max(a*a)/(sum(a)^2) ).
It combines several ideas:

* many diverse initialisations (uniform, exponential, inverse‑triangle,
  Dirichlet, and the previously found global best if available);
* an Adam optimiser on the soft‑max surrogate of the max‑convolution;
  the temperature λ is increased slowly (continuation);
* the optimiser works in the log‑probability (log‑softmax) space,
  guaranteeing a_i≥0 and Σa_i=1 exactly;
* when progress stalls a small random “shake’’ of the logits is applied;
* after the Adam phase a cheap hill‑climbing step with random perturbations
  refines the best candidate found;
* the search respects a total time budget (default 1000 s) and always
  returns a valid probability vector (fallback is uniform).

The code uses only NumPy (FFT for fast convolution) and no external files.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution helpers -------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where

        (w ★ a)[i] = Σ_k w[k]·a[i + k – (n‑1)]

    for i = 0 … n‑1.
    """
    # reverse a for correlation via convolution
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Initialisation helpers -------------------------------------------------
def _init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _init_random_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


def _init_exponential_random(n: int) -> np.ndarray:
    """sample an exponential decay from either side with random rate."""
    beta = 10.0 ** np.random.uniform(-4.0, -1.0)  # 1e‑4 … 1e‑1
    direction = np.random.choice([0, 1])
    i = np.arange(n, dtype=np.float64)
    if direction == 0:               # decay from the left
        a = np.exp(-beta * i)
    else:                            # decay from the right
        a = np.exp(-beta * (n - 1 - i))
    a /= a.sum()
    return a


def _init_inverse_triangle(n: int) -> np.ndarray:
    """large values near the edges, small in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 1) * (n - i) + 1.0)
    a /= a.sum()
    return a


def _init_from_existing(n: int) -> np.ndarray:

    cand = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):

    Search for a non‑negative sequence that minimises
    F(a)=2·n·max(a*a)/(sum(a)**2).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds (default 1000 s).

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # which problem sizes to tackle (larger n → more degrees of freedom)
    lengths = [1024, 2048, 4096, 8192]

    # a few generic initialisers
    generic_inits = [_init_random_dirichlet, _init_exponential_random,
                     _init_inverse_triangle, _init_uniform]

    # add the previous global solution if it exists
    if 'height_sequence_1' in globals():
        generic_inits.append(_init_from_existing)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT sizes (powers of two) – reuse across iterations
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # safe upper bound

        if n >= 4096:
            inits = generic_inits[:2]          # random Dirichlet + exponential
        else:

            inits = [_init_from_existing] + inits

            if time.time() > deadline:

            x = np.log(a0 + 1e-15)

            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            # optimiser hyper‑parameters
            lr = 0.08                         # base learning rate
            lam = 0.5                         # temperature for smooth‑max
            lam_factor = 1.05                 # geometric increase per iteration

            iter_cnt = 0
            last_improve = time.time()
            # time‑dependent iteration cap – larger n needs fewer steps
            max_iters = int(20000 * 8192 / n)

            while time.time() < deadline and iter_cnt < max_iters:
                # ---------- logits → probabilities (exact simplex) ----------
                mx = np.max(x)
                exp_x = np.exp(x - mx)
                a = exp_x / exp_x.sum()          # Σa_i = 1, a_i≥0

                # ---------- self‑convolution (full) ----------
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()

                # ---------- smooth‑max (or sub‑gradient) weights ----------
                if lam < 1e4:
                    w = np.exp(lam * (conv - max_c))
                    w /= w.sum()
                else:
                    # after λ is large we use a true sub‑gradient:
                    # uniform weight over the (tiny) active set
                    tol = 1e-12
                    mask = conv >= max_c - tol
                    w = mask.astype(np.float64)
                    w /= w.sum()

                # ---------- gradient of the surrogate w.r.t. a ----------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # ---------- back‑prop through the softmax (grad wrt logits) ----------
                avg = np.dot(a, grad_a)
                grad_x = a * (grad_a - avg)

                # ---------- Adam update ----------
                iter_cnt += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** iter_cnt)
                v_hat = v / (1.0 - beta2 ** iter_cnt)

                # slowly decay the step‑size when λ grows (helps stability)
                adj_lr = lr / (1.0 + 0.1 * lam)

                x -= adj_lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---------- temperature schedule ----------
                lam = min(lam * lam_factor, 1e6)

                # ---------- exact objective (Σa_i == 1) ----------
                val = 2.0 * n * max_c   # denominator = 1

                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} init={init_fn.__name__} iter={iter_cnt:05d}]"
                          f" new best {val:.6f}")

                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} init={init_fn.__name__} iter={iter_cnt}] shake")

        if best_seq is not None and len(best_seq) == n:
            a = best_seq.copy()
            conv = _full_convolution(a, fft_len=fft_len_conv)
            val_best = 2.0 * n * conv.max()
            hc_start = time.time()
            # try a few thousand tiny random moves
            while time.time() < deadline and time.time() - hc_start < 5.0:
                # perturb a handful of logits
                logits = np.log(a + 1e-12)
                idx = np.random.choice(n, size=5, replace=False)
                logits[idx] += (np.random.rand(5) - 0.5) * 0.04
                mx = np.max(logits)
                exp_x = np.exp(logits - mx)
                a_new = exp_x / exp_x.sum()
                conv_new = _full_convolution(a_new, fft_len=fft_len_conv)
                val_new = 2.0 * n * conv_new.max()
                if val_new < val_best:
                    a = a_new
                    val_best = val_new
                    best_seq = a.copy()
                    best_val = val_new
                    print(f"[n={n} hill‑climb] improved to {val_new:.6f}")
            # keep the refined vector
            best_seq = a.copy()

    # ----------------------------------------------------------------------
    # Fallback – if nothing succeeded return a simple uniform vector.
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)

    print(f"\nBest value found: {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
