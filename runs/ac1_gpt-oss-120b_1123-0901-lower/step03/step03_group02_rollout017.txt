def generate_initial(n):
    shapes = []
    # power law
    p = np.random.uniform(0.2, 1.0)
    shapes.append(np.arange(1, n+1) ** (-p))
    # exponential
    lam = np.random.uniform(0.001, 0.01) * n
    shapes.append(np.exp(-np.arange(n)/lam))
    # sin shape
    x = np.arange(1, n+1)
    shapes.append(np.sin(np.pi * x/(n+1)))
    # Gaussian bump near start
    sigma = np.random.uniform(0.05, 0.2) * n
    mu = np.random.uniform(0.2, 0.8) * n
    shapes.append(np.exp(-0.5 * ((x - mu)/sigma)**2))
    # Mix of two exponentials
    lam1 = np.random.uniform(0.001,0.01)*n
    lam2 = np.random.uniform(0.001,0.01)*n
    mix = np.exp(-np.arange(n)/lam1) + np.exp(-np.arange(n)[::-1]/lam2)
    shapes.append(mix)
    # Random jitter to break symmetry
    for s in shapes:
        s = s + np.random.rand(n) * 1e-4

"""
Hybrid stochastic‑gradient / Adam optimizer for the non‑negative step‑function
problem.

The algorithm explores many random lengths and random initial shapes (power‑law,
exponential, Gaussian, sine, mixed exponentials).  For each length it runs a
soft‑max (smooth‑max) gradient descent on the simplex Σ a_i = √(2 n) using an
Adam‑style adaptive step.  The temperature of the soft‑max decays slowly so
that the optimisation first smooths the objective and finally concentrates on
the true maximum of the self‑convolution.  Good solutions are kept globally;
the current best sequence is reused as a warm‑start for later runs.  After the
main exploration phase the remaining time is spent refining the global best
with a longer, lower‑temperature Adam run.

The code respects the 1000 s budget, prints useful progress information and
returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=np.float64)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives from rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Random initial shape generator
# ----------------------------------------------------------------------
def _random_initial(n: int) -> np.ndarray:
    """Generate a random non‑negative vector of length n (not yet projected)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape_type = np.random.choice(["power", "exp", "sin", "gauss", "mix"])
    if shape_type == "power":
        p = np.random.uniform(0.2, 1.0)
        init = idx ** (-p)
    elif shape_type == "exp":
        tau = np.random.uniform(0.2, 0.8) * n
        init = np.exp(-idx / tau)
    elif shape_type == "sin":
        init = np.sin(np.pi * idx / (n + 1))
    elif shape_type == "gauss":
        mu = np.random.uniform(0.3, 0.7) * n
        sigma = np.random.uniform(0.05, 0.2) * n
        init = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    else:  # mix of two exponentials, one from each side
        lam1 = np.random.uniform(0.3, 0.9) * n
        lam2 = np.random.uniform(0.3, 0.9) * n
        init = np.exp(-idx / lam1) + np.exp(-(n + 1 - idx) / lam2)

    # tiny jitter to break symmetry
    init += np.random.rand(n) * 1e-4
    init = np.maximum(init, 0.0)
    # final simplex projection will be done inside the optimiser
    return init


# ----------------------------------------------------------------------
# Adam‑powered soft‑max gradient descent for a *single* length
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    lr: float = 0.10,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑`n` sequence starting from ``init``.
    Returns the best sequence discovered and its score as reported by
    ``evaluate_sequence``.
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = float(conv.max())          # with the chosen normalisation this equals the objective
    m = np.zeros_like(a)                    # Adam first moment
    v = np.zeros_like(a)                    # Adam second moment
    it = 0

    while time.time() < run_deadline:
        # ----- self‑convolution and current max -----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # ----- soft‑max weights (temperature τ) -----
        tau = max(init_tau_factor * max_val * (0.99 ** it),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)                 # all values ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of Σ w_k (a*a)_k -----
        w_rev = w[::-1]
        tmp = _full_convolution_fft(a, w_rev)          # length = 2n‑1
        idx = (2 * n - 2) - np.arange(n)               # positions that give Σ_j a_j w_{i+j}
        grad = 2.0 * tmp[idx]

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # ----- evaluate candidate -----
        conv_new = _full_convolution_fft(a_new, a_new)
        cur_score = float(conv_new.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}   new local best = {best_score:.6f}")

        a = a_new

        # ----- occasional random shake‑up -----
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # reset Adam moments after a big perturbation
            m.fill(0.0)
            v.fill(0.0)

    # final evaluation through the provided wrapper (guarantees same rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    under the 1000 s budget.  The routine performs many random restarts,
    each time running an Adam‑based soft‑max gradient descent, keeps the best
    candidate globally and finally refines it with a longer low‑temperature run.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] warm‑start from global seed = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Exploration phase – many independent restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 20.0:
            break  # leave some time for final refinement

        # choose a length; larger lengths tend to give smaller peaks
        if remaining > 300:
            n = np.random.randint(1500, 5000 + 1)
        elif remaining > 120:
            n = np.random.randint(1000, 4000 + 1)
        else:
            n = np.random.randint(600, 3000 + 1)

        # initialise the vector
        if best_seq is not None and np.random.rand() < 0.45:
            # warm‑start from the current global best (truncated or padded)
            if best_seq.size >= n:
                init_vec = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init_vec = np.concatenate([best_seq, pad])
        else:
            init_vec = _random_initial(n)

        # give each restart a generous time slice (up to ~15 s)
        run_deadline = min(deadline, time.time() + min(15.0, remaining * 0.45))

        # optimise this length
        seq_loc, score_loc = _optimise_one_length(
            n,
            init_vec,
            run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=5e-4,
            lr=0.12,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )

        if score_loc < best_score:
            best_score = score_loc
            best_seq = seq_loc.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.49:
                # already very good – stop early
                break

    # ------------------------------------------------------------------
    # Refinement phase – polish the global best for the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.size
        # a slightly longer low‑temperature Adam run
        refin_deadline = deadline
        print(f"[{time.time():.2f}] Refinement started (n={n})")
        best_seq, best_score = _optimise_one_length(
            n,
            best_seq,
            refin_deadline,
            init_tau_factor=0.10,
            min_tau_factor=1e-4,
            lr=0.08,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )
        print(f"[{time.time():.2f}] Refinement finished – final score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (very unlikely) – uniform distribution of modest length
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
