"""
Propose a step‑function (non‑negative vector) that minimizes the
`evaluate_sequence` score.

Algorithm
---------
1. Choose a few problem sizes (lengths).  For each length generate a
   portfolio of simple parametric shapes (linear, power, exponential,
   double‑exponential, sine, triangular, the previously‑found
   construction, and a few random Dirichlet vectors).  Each shape is
   normalised to the required L1‑norm sqrt(2·n) and its maximal
   autocorrelation `max(a*a)` is evaluated by a fast FFT based
   convolution.  The best shape over all lengths is kept.

2. Starting from that best shape, run a *simulated‑annealing* local
   search that repeatedly moves a small amount of mass from one index
   to another.  The convolution is updated incrementally (only three
   entries of the full autocorrelation change).  Moves that lower the
   current maximum are always accepted; worsening moves are accepted
   with a temperature‑controlled probability.  The temperature is
   linearly cooled.  Every few thousand steps a full convolution is
   recomputed to avoid drift.

3. After the annealing phase a short Adam optimiser (the same routine
   used in the previous baseline) is applied for a few thousand
   iterations to polish the solution.

The whole search runs for the allotted time (≈ 1000 s) and returns the
best sequence found.  All numbers are non‑negative; the routine never
produces values > 1000, so the security checks in `evaluate_sequence`
are harmless.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_autoconv(a: np.ndarray):
    """Return full autocorrelation (a*a) and its maximum using FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv, float(conv.max())


# ----------------------------------------------------------------------
def _shape_linear(n: int, inc: bool = True) -> np.ndarray:
    """Linear ramp, increasing if inc else decreasing."""
    if inc:
        return np.arange(1, n + 1, dtype=float)
    else:
        return np.arange(n, 0, -1, dtype=float)


# ----------------------------------------------------------------------
def _shape_power(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^p (p may be negative)."""
    return (np.arange(1, n + 1, dtype=float) ** p)


# ----------------------------------------------------------------------
def _shape_exponential(n: int, alpha: float) -> np.ndarray:
    """a_i = exp(-alpha * i)."""
    return np.exp(-alpha * np.arange(n, dtype=float))


# ----------------------------------------------------------------------
def _shape_double_exp(n: int, alpha: float) -> np.ndarray:
    """Two exponentials from both ends: e^{-α i}+e^{-α (n-1-i)}."""
    idx = np.arange(n, dtype=float)
    return np.exp(-alpha * idx) + np.exp(-alpha * (n - 1 - idx))


# ----------------------------------------------------------------------
def _shape_sine(n: int) -> np.ndarray:
    """Shifted sine to stay strictly positive."""
    x = np.linspace(0, math.pi, n, dtype=float)
    return np.sin(x) + 0.1


# ----------------------------------------------------------------------
def _shape_triangular(n: int) -> np.ndarray:
    """Triangular (tent) shape."""
    return np.minimum(np.arange(1, n + 1, dtype=float),
                      np.arange(n, 0, -1, dtype=float))

def _normalise(a: np.ndarray, target_sum: float) -> np.ndarray:

    total = a.sum()
    if total > 0.0:
        return a * (target_sum / total)
    else:
        # fallback uniform vector
        return np.full_like(a, target_sum / a.shape[0])

def _anneal_optimize(a_init: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    max_moves: int = 300_000) -> np.ndarray:
    """Simulated‑annealing local search that moves mass between indices."""
    a = a_init.copy()
    n = a.shape[0]

    # exact convolution (full) – used for accurate max evaluation
    conv, cur_max = _full_autoconv(a)

    best_a = a.copy()
    best_max = cur_max

    # temperature schedule (linear cooling)
    T0 = 0.1 * cur_max          # start temperature
    Tf = 1e-4 * cur_max         # final temperature

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_moves:
        it += 1
        # --------------------------------------------------------------
        #  choose donor (i) with positive mass and receiver (j)
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount to move – up to 30 % of donor's current mass
        eps = a[i] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        # indices in the full convolution that change
        k1 = 2 * i
        k2 = 2 * j
        k3 = i + j

        ai = a[i]
        aj = a[j]

        # incremental update (including the ε² terms for exactness)
        conv_new = conv.copy()
        conv_new[k1] = conv_new[k1] - 2.0 * eps * ai + eps * eps
        conv_new[k2] = conv_new[k2] + 2.0 * eps * aj + eps * eps
        conv_new[k3] = conv_new[k3] + 2.0 * eps * (ai - aj) - 2.0 * eps * eps

        new_max = float(conv_new.max())

        # temperature for this iteration
        T = T0 * (1.0 - it / max_moves) + Tf * (it / max_moves)

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < math.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            # commit move
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[anneal] new best max={best_max:.6f} iter={it}")

        # occasionally recompute the convolution from scratch to avoid drift
        if it % 4000 == 0:
            conv, cur_max = _full_autoconv(a)

    return best_a

        if iteration % 5000 == 0:
            # tiny jitter to escape plateaus
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

    return a

# Helpers duplicated from the original baseline (required for Adam)
def _conv_max_fft(a: np.ndarray):
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


def _subgradient_average(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= float(active.size)
    return rev

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Returns a Python list representing the step‑function.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore (original length + a few scalings)
    try:
        base_vec = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base_vec.shape[0]
    except Exception:
        n0 = 2100
        base_vec = np.full(n0, 1.0, dtype=float)

    length_factors = [1.0, 1.2, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [l for l in lengths if 300 <= l <= 8000]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric search
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for idx, cand in enumerate(_candidate_vectors(n, target_sum, seed=seed)):
            if time.time() >= deadline:
                break
            # fast max‑convolution
            _, max_b = _full_autoconv(cand)
            if max_b < best_score:
                best_score = max_b
                best_seq = cand.copy()
                print(f"[init] len={n} cand={idx} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing refinement
    if best_seq is not None:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        best_seq = _anneal_optimize(best_seq, target_sum, deadline,
                                    max_moves=250_000)

        # recompute exact score after annealing
        _, best_score = _full_autoconv(best_seq)
        print(f"[anneal] finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short Adam polishing (optional, cheap)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        best_seq = _adam_refine(best_seq, target_sum, deadline,
                                max_iter=1_000_000)
        # final exact evaluation via the provided function
        try:
            final_score = evaluate_sequence(best_seq.tolist())
        except Exception:
            # fallback to our own max‑conv computation
            _, final_score = _full_autoconv(best_seq)
        print(f"[final] score after Adam = {final_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – uniform vector if everything failed
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        final_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence, score =", final_score)

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
