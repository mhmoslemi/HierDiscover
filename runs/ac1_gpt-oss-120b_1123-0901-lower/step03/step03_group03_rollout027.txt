def lp_minimize_max(f, target_sum, deadline, max_outer_iter=20):
    n = len(f)
    # Build matrix A once per iteration since f changes after solving (new f)
    best_t = float('inf')
    for it in range(max_outer_iter):
        if time.time() >= deadline:
            break
        # Build A matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f
        # Constraints: A @ g <= t  ==> A @ g - t <= 0
        # So create A_ub = np.hstack([A, -np.ones((2*n-1,1))])
        A_ub = np.hstack([A, -np.ones((2*n-1, 1), dtype=np.float64)])
        b_ub = np.zeros(2*n-1, dtype=np.float64)
        # Equality: sum(g) = target_sum, plus t not in equality.
        A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
        b_eq = np.array([target_sum], dtype=np.float64)
        # bounds for g variables: (0, None); for t: (0, None)
        bounds = [(0, None)] * n + [(0, None)]
        # Objective: minimize t => c = [0,...,0, 1]
        c = np.zeros(n+1, dtype=np.float64)
        c[-1] = 1.0
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it}: linprog failed: {res.message}")
            break

"""
A two‑stage optimizer for the step‑function problem.

Stage 1 – fast Adam optimisation (soft‑max smoothed max of the
self‑convolution) works from several initial shapes (the known best,
a power‑law, random Dirichlet vectors).  The Adam loop uses FFTs for
speed, projects onto the simplex (a≥0, Σa=√(2 n)) and anneals a
soft‑max temperature.

Stage 2 – a few linear‑programming refinements.  For a fixed vector
``f`` we solve

    minimise   t
    s.t.       f * g  ≤  t          (every entry of the convolution)
               Σ g   =  √(2 n)
               g ≥ 0

which is a linear programme in the variables ``g`` and ``t``.
The optimum gives a new feasible vector with a (weakly) smaller
max‑convolution.  Repeating a handful of times drives the bound
further down.

Both stages respect the 1000 s wall‑clock limit and keep the best
sequence seen.  The final sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute the reverse‑weighted convolution:
        rev_i = Σ_k w_k a_{k‑i}
    using a single FFT.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int = 5_000_000,
):
    """
    Adam optimiser that minimises a smoothed version of the maximal
    self‑convolution.  The exact objective (the max after normalisation)
    is used to update the global best.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.08
    iteration = 0
    jitter_interval = 3000

    # temperature control
    tau_factor = 0.25
    decay_rate = 15000.0

    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and iteration < max_iters:
        # exact convolution a*a
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # soft‑max smoothing of the max
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # weighted reverse convolution (sub‑gradient of the smooth max)
        rev = _weighted_rev(a, w, Nfft)

        # Adam gradient step (including normalisation term)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set (a≥0, Σa = √(2 n))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # exact objective (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _lp_refine_sequence(a: np.ndarray, deadline: float, max_outer: int = 10):
    """
    Linear‑programming refinement.

    For the current vector ``a`` we repeatedly solve
        minimize   t
        s.t.       a * g ≤ t
                   Σ g = √(2 n)
                   g ≥ 0
    and replace ``a`` with the optimal ``g``.
    The process stops when no further improvement is observed
    or the time budget expires.
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # compute current objective
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2 * n - 1]
    best_score = float(conv.max())
    best_seq = a.copy()
    print(f"[LP] start score = {best_score:.6f}")

    for outer in range(max_outer):
        if time.time() >= deadline:
            break

        # build the convolution matrix A (size (2n‑1) × n)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = best_seq

        # inequality: A @ g - t ≤ 0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1), dtype=np.float64)])
        b_ub = np.zeros(2 * n - 1, dtype=np.float64)

        # equality: sum(g) = target_sum
        A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
        b_eq = np.array([target_sum], dtype=np.float64)

        # objective: minimise t
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0

        bounds = [(0.0, None)] * n + [(0.0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"[LP] iteration {outer}: linprog failed – {res.message}")
            break

        t_opt = res.x[-1]
        g_opt = res.x[:n]

        # evaluate the true max‑convolution of the candidate
        G_fft = np.fft.rfft(g_opt, n=Nfft)
        conv_g = np.fft.irfft(G_fft * G_fft, n=Nfft)[:2 * n - 1]
        max_g = float(conv_g.max())

        print(f"[LP] iter {outer}  t_opt={t_opt:.6f}  max(g)={max_g:.6f}")

        if max_g + 1e-9 < best_score:
            best_seq = g_opt
            best_score = max_g
        else:
            # no improvement – stop
            break

    return best_seq, best_score

    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [min(l, 8000) for l in lengths]
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break

        # Prepare a few different initial vectors for this length
        inits = []

        # 1) interpolation of the best known shape
        if n == n0:
            inits.append(base.copy())
        else:
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            inits.append(np.maximum(interp, 0.0))

        # 2) power‑law shape a_i ∝ 1/√(i+1)
        i_arr = np.arange(n, dtype=float)
        powerlaw = 1.0 / np.sqrt(i_arr + 1.0)
        inits.append(powerlaw)

        # 3) a couple of random Dirichlet‑like starts
        for rnd_seed in (seed + 100 + n, seed + 200 + n):
            rng = np.random.default_rng(rnd_seed)
            rand_vec = rng.random(n)
            inits.append(rand_vec)

        # normalise all initial vectors and run the Adam optimiser
        for init_idx, vec in enumerate(inits):
            vec = np.maximum(vec, 0.0)
            target_sum = math.sqrt(2.0 * n)
            if vec.sum() == 0.0:
                vec = np.full_like(vec, target_sum / n)
            else:
                vec = vec * (target_sum / vec.sum())

            # iteration budget (scaled roughly with problem size)
            max_iters = int(2_000_000 * 2100 / n)
            max_iters = max(200_000, max_iters)

            print(f"[run] length={n}, init {init_idx + 1}/{len(inits)}, max_iters≈{max_iters}")

            best_seq, best_score = _optimize_length(
                n=n,
                init_a=vec,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iters=max_iters,
            )

        # after all inits for this length we move on
        # (early exit if time is almost exhausted)
        if time.time() + 30.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # Linear‑programming refinement (only on the final best length)
    if time.time() < deadline:
        print("[LP] Starting refinement phase")
        best_seq, best_score = _lp_refine_sequence(best_seq, deadline, max_outer=12)

    # ------------------------------------------------------------------
    # Final sanity check (evaluate_sequence is cheap compared to the loops)
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score - 1e-12:
        best_score = final_score

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
