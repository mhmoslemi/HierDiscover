"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The algorithm is a multi‑restart Adam optimiser
with a *soft* active set: at each iteration we take the K largest
convolution values (instead of only the exact maximiser) and use the
average reversed vector of those shifts as a sub‑gradient.  The size K
grows slowly during optimisation, providing a smoother descent and
helping to escape plateaus.  After the Adam phase a cheap hill‑climb
(local search) moves small amounts of mass between indices whenever
the maximal autocorrelation value improves.  Several sequence lengths
are tried (interpolating the previous best construction and random
Dirichlet starts).  The best sequence found before the time budget
expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Returns the full (linear) autocorrelation a * a using FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, returns the vector rev where
    rev[i] = a[k-i] if the index is inside the support, otherwise 0.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _active_top_k(conv: np.ndarray, K: int) -> np.ndarray:
    """
    Return the indices of the K largest entries of conv.
    If K >= len(conv) all indices are returned.
    """
    if K >= conv.size:
        return np.arange(conv.size, dtype=int)
    # argpartition is O(n); we then take the K largest
    idx = np.argpartition(conv, -K)[-K:]
    return idx.astype(int)


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.04,
    K0: int = 1,
    K_max: int = 15,
    inc_iter: int = 4_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the maximal autocorrelation using a soft
    active set of the K largest convolution values.
    """
    # normalise start vector
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    sqrt2 = math.sqrt(2.0)
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        # ----- 1) full autocorrelation & max value --------------------
        conv = _conv_full_fft(a)
        M = float(conv.max())
        const = sqrt2 * M / math.sqrt(n)          # gradient term from denominator

        # ----- 2) soft active set ------------------------------------
        K = min(K_max, K0 + iteration // inc_iter)   # grow K slowly
        active = _active_top_k(conv, K)

        # ----- 3) averaged reversed vector ---------------------------
        rev = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        rev /= float(active.size)

        # ----- 4) gradient of the objective ---------------------------
        #  grad = 2 * rev - const (broadcast const)
        grad = 2.0 * rev - const

        # ----- 5) Adam update -----------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 6) projection onto feasible set (non‑negative + sum) ---
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:                                   # extremely unlikely
            a = np.full_like(a, target_sum / n)

        # ----- 7) occasional jitter to escape plateaus ---------------
        if iteration % 4_000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- 8) keep best (objective = M after normalisation) -------
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {iteration:7d} K {K:2d}] "
                f"new best score = {best_score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 50_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that repeatedly moves a (potentially larger)
    fraction of mass from a donor index to a receiver index, accepting
    the move only if the maximal convolution value decreases.
    """
    a = a_start.copy()
    n_range = np.arange(n)
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1

        # pick donor with positive mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue

        # pick a different recipient
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 15 % of donor's mass (larger moves than the original)
        eps = a[i] * np.random.rand() * 0.15
        if eps <= 0.0:
            continue

        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps

        # evaluate new maximal convolution (no need to call evaluate_sequence)
        M_new = float(_conv_full_fft(new_vec).max())
        if M_new < best_score - 1e-12:          # strict improvement
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (prop {proposal})")
    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising evaluate_sequence().
    The function respects the 1000 s time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Load the previously discovered construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # ------------------------------------------------------------------
    # Book‑keeping of the global best
    best_seq = None
    best_score = float("inf")

    # evaluate the initial construction (keep its normalisation)
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    init_score = float(_conv_full_fft(base).max())     # same as evaluate_sequence after normalisation
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (original + a few enlargements)
    length_factors = [1.0, 1.3, 1.6, 2.0, 2.5]
    lengths = sorted({int(n0 * f) for f in length_factors})
    max_len = 8000
    lengths = [l for l in lengths if 300 <= l <= max_len]

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        restarts = 4                       # 1 interpolation + 3 random

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise -----------------------------------------
            if r == 0:
                # linear interpolation of the current best onto the new grid
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, best_seq)
                init_a = np.maximum(init_a, 0.0)
            else:
                # random Dirichlet‑like start (positive numbers, renormalised)
                init_a = np.random.rand(n)
                init_a = np.maximum(init_a, 0.0)

            # normalise to the required total sum
            s = init_a.sum()
            if s > 0.0:
                init_a = init_a * (target_sum / s)
            else:
                init_a = np.full(n, target_sum / n)

            # ----- Adam optimisation ---------------------------------
            best_seq, best_score = _adam_optimize(
                a_init=init_a,
                n=n,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iter=200_000,
                lr0=0.04,
                K0=1,
                K_max=15,
                inc_iter=4_000,
            )

            # ----- cheap local refinement -------------------------------
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=30_000,
                )

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence (may differ by
    # rounding / clipping but ensures validity)
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
        if final_score < best_score:
            best_score = final_score

    # ------------------------------------------------------------------
    # Fallback: in the unlikely case that nothing was found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
