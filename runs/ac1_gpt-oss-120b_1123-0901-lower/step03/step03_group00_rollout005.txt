"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)          (a_i ≥ 0)

The algorithm combines a few ideas that proved useful in earlier
experiments:

* **Soft‑max parametrisation** – we optimise over logits `x` and obtain a
  probability vector `a = softmax(x)`.  This guarantees the simplex
  constraints `a_i ≥ 0, Σ a_i = 1`.

* **Smooth‑max surrogate** – the hard maximum in the objective is
  replaced by a log‑sum‑exp with a temperature `λ`.  Its gradient can be
  written in closed form using a single cross‑correlation, which we
  evaluate with FFTs.

* **Adam updates** – an Adam optimiser on the logits provides fast
  convergent updates.

* **Multi‑scale warm‑starts** – we solve the problem for a series of
  increasing lengths `[512, 1024, 2048, 4096, 8192]`.  The solution of a
  coarser problem is up‑sampled (linear interpolation) and used as the
  initial point for the next, larger problem.

* **Stagnation handling** – if no improvement is seen for a few seconds
  we add a small random “shake” to the logits.

* **Light‑weight local refinement** – after the Adam phase a short
  simulated‑annealing style refinement moves a little mass between two
  coordinates, accepting only improvements (or occasional uphill moves).

The routine respects the supplied time budget (default 1000 s) and always
returns the best sequence found so far (as a plain Python list).  All
numbers are non‑negative, and the sequence can be of any length – the
best among the examined sizes is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation of F(a) = 2·n·max(a*a) / (sum(a)²).
    The vector a is assumed to be non‑negative.
    """
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_len (both ≥ 1)."""
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    if old_len == 1:
        return np.full(new_len, seq[0], dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence (as a plain Python list) it discovers.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes explored (increasing resolution)
    lengths = [512, 1024, 2048, 4096, 8192]

    # Sequence from the previous length (for warm‑starts)
    prev_seq = None

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (powers of two) for the two convolutions we need.
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # w (2n‑1) + a (n) - 1

        # Number of independent restarts – we keep adding restarts while we have time.
        max_restarts = 8 if n <= 2048 else 5 if n <= 4096 else 3
        restart_id = 0

        while restart_id < max_restarts and time.time() < deadline:
            # --------------------------------------------------------------
            # Initialise logits
            if restart_id == 0 and prev_seq is not None and len(prev_seq) == n:
                a0 = prev_seq / prev_seq.sum()
                x = np.log(a0 + 1e-15)
                print("[restart 0] warm start from previous size")
            elif restart_id == 0 and prev_seq is not None and len(prev_seq) < n:
                a0 = _upsample(prev_seq, n)
                a0 = np.maximum(a0, 0.0)
                s = a0.sum()
                a0 = a0 / s if s > 0 else np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print("[restart 0] up‑sampled warm start")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                cand = cand / s if s > 0 else np.random.dirichlet(np.ones(n))
                x = np.log(cand + 1e-15)
                print("[restart 0] start from height_sequence_1")
            else:
                # pure random Dirichlet start
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print("[restart] random start")
            # --------------------------------------------------------------

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            # Learning‑rate schedule (slightly larger for small n)
            lr = 0.07 if n <= 1024 else 0.05 if n <= 4096 else 0.04

            # Smooth‑max temperature λ
            lam = 0.5
            lam_factor = 1.02   # geometric increase, slower than 1.03

            t = 0
            max_iters = 25000 if n <= 2048 else 20000 if n <= 4096 else 15000
            last_improve = time.time()

            while time.time() < deadline and t < max_iters:
                # ---- logits → probabilities (softmax) ----
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()          # Σ a = 1, a ≥ 0

                # ---- self‑convolution -------------------
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()

                # ---- smooth‑max weights -----------------
                w = np.exp(lam * (conv - max_c))
                w /= w.sum()                     # Σ w = 1

                # ---- gradient of the surrogate ---------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # ---- back‑prop through softmax ----------
                avg = np.dot(a, grad_a)          # ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)      # ∇_x φ

                # ---- Adam update -----------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---- temperature schedule ---------------
                lam = min(lam * lam_factor, 1e4)

                # ---- exact objective evaluation ----------
                # (sum a is 1 up to floating error)
                cur_val = 2.0 * n * max_c / (a.sum() ** 2)

                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={restart_id} iter={t:05d}] new best: {cur_val:.6f}")

                # ---- stagnation handling ----------------
                if time.time() - last_improve > 5.0:
                    # small random perturbation of logits
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={restart_id} iter={t}] shake (stagnation)")

            # finished one restart
            restart_id += 1

        # ------------------------------------------------------------------
        # Light‑weight local refinement for the current best of this size
        if best_seq is not None and len(best_seq) == n:
            # allocate a modest amount of remaining time
            refine_deadline = deadline - 0.5
            if time.time() < refine_deadline:
                print(f"Refining sequence of length {n}")
                a = best_seq.copy()
                cur_val = best_val
                # temperature schedule for simulated annealing
                temp = max(0.01 * cur_val, 0.001)
                decay = 0.9995
                # max number of refinement steps inversely proportional to n
                max_refine = 2000 if n <= 2048 else 800
                it_ref = 0
                while time.time() < refine_deadline and it_ref < max_refine:
                    it_ref += 1
                    i, j = np.random.randint(0, n, size=2)
                    if i == j:
                        continue
                    # move a small fraction of mass from i to j
                    eps_move = (np.random.rand() * 0.02) * a[i]   # ≤2 % of a[i]
                    if eps_move <= 0:
                        continue
                    a_new = a.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    if a_new[i] < 0:
                        continue
                    # quickly evaluate new objective
                    conv_new = _full_convolution(a_new, fft_len=fft_len_conv)
                    new_val = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)
                    # accept if better or with Metropolis probability
                    if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / temp):
                        a = a_new
                        cur_val = new_val
                        if new_val < best_val:
                            best_val = new_val
                            best_seq = a.copy()
                            print(f"[refine n={n} it={it_ref}] improvement: {new_val:.6f}")
                    temp *= decay
                # store for the next size
                prev_seq = best_seq.copy()
        else:
            # if we did not obtain a sequence of this length, keep the previous one for up‑sampling
            if best_seq is not None and len(best_seq) == n:
                prev_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – return a simple uniform vector if nothing was found
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print("Fallback uniform sequence used.")

    print(f"\nBest objective found: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
