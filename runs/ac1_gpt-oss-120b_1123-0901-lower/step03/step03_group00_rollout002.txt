def _project_onto_simplex(v):
    # Input array v (any shape 1d)
    # output array x (same shape) where x>=0, sum=1, minimize Euclidean distance.
    # algorithm: sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

1. **Smooth‑max Adam** – the hard max is replaced by a soft‑max
   (log‑sum‑exp) with a temperature λ that grows geometrically.
   The vector is parametrised by logits `x` (softmax → simplex), and
   Adam updates are applied to the logits.

2. **Projected sub‑gradient** – once λ is large (the surrogate is
   close to the true max) we switch to a classic sub‑gradient step:
   the gradient of the exact objective is 2·(w ★ a) where the weight
   vector `w` puts equal mass on all convolution entries attaining the
   current maximum.  After a gradient step the iterate is projected back
   onto the probability simplex (non‑negative, sum = 1).

The routine runs several restarts, tries a few problem sizes
(1024 … 8192), keeps the best sequence seen so far and returns it as a
plain Python list.  Simple `print` statements report every improvement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]                     # reversed a
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]        # length n


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, Σ_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Ensure numerical sum=1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function stops a bit before the
        deadline to avoid a hard timeout).

    Returns
    -------
    list[float]
        The best sequence found, as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes to explore (larger n gives more freedom but costs time)
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # FFT lengths (next power of two)
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n)))   # for w★a

        # Number of independent restarts for this length.
        restarts = 5

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Initialise a (probability vector, i.e. sum = 1)
            if best_seq is not None and len(best_seq) == n:
                a = np.asarray(best_seq, dtype=np.float64)
                a = np.maximum(a, 0.0)
                if a.sum() < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = a / a.sum()
                print("[restart] warm start from global best")
            elif 'height_sequence_1' in globals():
                # Use the provided historic construction if available.
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                print("[restart] start from height_sequence_1")
            else:
                # Pure random start (Dirichlet yields a point on the simplex)
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random start")

            # --------------------------------------------------------------
            # Phase 1 – Adam on a smooth‑max surrogate
            x = np.log(a + 1e-15)          # logits → softmax will give a
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.08                      # learning rate for logits
            lam = 0.5                      # initial smooth‑max temperature
            lam_factor = 1.025             # geometric increase
            max_iters_phase1 = 6000

            last_improve = time.time()
            for t in range(max_iters_phase1):
                if time.time() > deadline:
                    break

                # logits → probabilities (softmax)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()          # now a lies on the simplex

                # self‑convolution
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()

                # smooth‑max weights (numerically stable)
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:                 # extreme case – fallback to max index
                    w = np.zeros_like(conv)
                    w[conv.argmax()] = 1.0
                    w_sum = 1.0
                w /= w_sum

                # gradient w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # back‑prop through softmax (gradient w.r.t. logits)
                avg = np.dot(a, grad_a)          # ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)      # ∇_x φ

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** (t + 1))
                v_hat = v / (1.0 - beta2 ** (t + 1))
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a) is 1 up to floating error)
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} phase1 iter={t:05d}] new best: {val:.6f}")

                # stagnation handling – occasional random shake
                if time.time() - last_improve > 6.0:
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} phase1 iter={t}] shake due to stagnation")

            # --------------------------------------------------------------
            # Phase 2 – Projected sub‑gradient on the true max objective
            # Ensure a is on the simplex (possible tiny drift)
            a = _project_onto_simplex(a)

            step_size = 0.12                # initial step size for sub‑gradient
            decay = 0.996                   # geometric decay
            max_iters_phase2 = 18000
            last_improve = time.time()

            for t in range(max_iters_phase2):
                if time.time() > deadline:
                    break

                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c            # denominator = 1

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} phase2 iter={t:05d}] new best: {val:.6f}")

                # Identify all indices that are (almost) maximal
                tol = 1e-12 * max_c
                mask = conv >= max_c - tol
                num_max = int(mask.sum())
                if num_max == 0:      # safety fallback
                    mask = np.zeros_like(conv, dtype=bool)
                    mask[conv.argmax()] = True
                    num_max = 1
                w = np.zeros_like(conv)
                w[mask] = 1.0 / num_max

                # Sub‑gradient of the exact objective
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # Gradient step + projection onto simplex
                a_new = a - step_size * grad_a
                a = _project_onto_simplex(a_new)

                # step‑size schedule
                step_size *= decay
                if step_size < 1e-9:
                    break

                # occasional random shake if no improvement for a while
                if time.time() - last_improve > 8.0:
                    a += (np.random.rand(n) - 0.5) * 1e-3
                    a = _project_onto_simplex(a)
                    last_improve = time.time()
                    print(f"[n={n} restart={r} phase2 iter={t}] shake due to stagnation")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (highly unlikely) return a uniform vector.
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        print("[fallback] uniform vector returned")

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
