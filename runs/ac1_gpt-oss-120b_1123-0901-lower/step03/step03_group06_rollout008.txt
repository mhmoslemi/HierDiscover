def _optimize_adam_length(...):
    p = p_start.copy()
    u = np.log(p+1e-30)   # (or use logit? We'll treat as logits)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1=0.9, beta2=0.999, eps=1e-8
    for it in range(max_iters):
        # compute conv, M
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        # compute softmax weights w : exp(beta*(conv-M)), then lse normalization maybe not needed.
        beta = beta_start + (beta_end-beta_start) * min(1.0, it / beta_iters)
        # weights w_k = exp(beta*(conv_k - M))
        w = np.exp(beta*(conv-M))   # <=1
        # compute rev using weighted correlation: rev_i = Σ_k w_k * p_{k-i}
        rev = _weighted_rev_fft(p, w, N)
        grad_p = 4.0 * n * rev   # gradient of smoothed objective w.r.t p
        # compute gradient w.r.t logits u (softmax)
        dot = np.dot(p, grad_p)
        grad_u = p * (grad_p - dot)

        # Adam update
        m = beta1*m + (1-beta1)*grad_u
        v = beta2*v + (1-beta2)*(grad_u**2)
        m_hat = m/(1-beta1**(it+1))
        v_hat = v/(1-beta2**(it+1))
        # step size: lr/(sqrt(v_hat)+eps)
        u -= learning_rate * m_hat/(np.sqrt(v_hat)+eps)

        # optionally jitter: add small noise to u periodically
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # update p via softmax
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        p = exp_u / exp_u.sum()

        # evaluate current objective: cur_score = 2*n*max(conv)
        cur_score = 2.0 * n * M   # (since M is max conv)
        if cur_score < best_local_score:
            best_local_score=cur_score
            best_local_p = p.copy()
            # print progress

        # maybe break if conv highly flat? Not necessary.

   return best_local_p, best_local_score

def _optimize_adam_length(n, p_start, deadline, max_iters=300000):
    p = p_start.copy()
    u = np.log(p + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyperparameters
    lr0 = 1.0   # base learning rate
    beta_start = 0.01
    beta_end = 50.0
    beta_len = 100000  # iterations to reach beta_end

    jitter_interval = 2000
    jitter_scale = 0.005

    best_local_score = float("inf")
    best_local_p = p.copy()

    N = _next_pow2(3 * n - 2)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # conv
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        # update best
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print("[len {} iter {}] local best {:.6f}".format(n, it, cur_score))

        # beta schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)

        # softmax weight
        w = np.exp(beta * (conv - M))

        # weighted rev (correlation)
        rev = _weighted_rev_fft(p, w, N)

        # gradient w.r.t p
        grad_p = 4.0 * n * rev

        # gradient w.r.t logits u
        dot = np.dot(p, grad_p)
        grad_u = p * (grad_p - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning rate decay
        lr = lr0 / (1 + it * 1e-4)

        # update logits
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter
        if jitter_interval and it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # project back to simplex via softmax
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        p = exp_u / exp_u.sum()

    return best_local_p, best_local_score

"""
Multi‑scale Adam optimizer for the autocorrelation L∞‑norm.

The objective (after normalising Σa = √(2n)) is
    F(p) = 2·n·max(p∗p) ,   p_i ≥ 0 , Σ p_i = 1 .
We treat p as a probability vector and parameterise it by logits
    p_i = softmax(u_i).
A smooth approximation of the max is the soft‑max
    M_β(p) = (1/β)·log Σ_k exp(β·(p∗p)_k) .
Its gradient is a correlation of p with the weight vector
    w_k = exp(β·((p∗p)_k − max_k(p∗p)_k)).
We minimise M_β via Adam on the logits u, gradually raising β
from a small to a moderate value.
A multiscale schedule starts from a modest length and successively
upsamples the best solution to larger lengths (up to 8192).  The
best sequence found (converted back to the original scale) is
returned.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full linear self‑convolution p∗p using a single FFT of size N.
    Returns the real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k · p_{k‑i}   for i = 0 … n‑1.
    This is a correlation (p ⋆ w) and is obtained via FFT.
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    # FFT of p
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weight vector
    W_rev = np.fft.rfft(w[::-1], n=N)
    # Convolution of p with reversed w
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]    # keep linear part
    # Extract rev = conv_rev[L‑n : L] reversed → rev_i = conv_rev[L‑1‑i]
    rev = conv_rev[L - n : L][::-1]
    return rev

def _optimize_adam_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution for a fixed length ``n``.
    Returns the best probability vector (sum = 1) and its objective value
    ``2·n·max(p∗p)`` encountered during the run.
    """
    # Hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5                       # initial learning‑rate (decays)
    beta_start, beta_end = 0.01, 50.0
    beta_len = 100_000              # iterations over which β is increased
    jitter_interval = 2_000
    jitter_scale = 0.005

    # Initialise logits (softmax parametrisation)
    p = p_start.copy()
    u = np.log(p + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # FFT size sufficient for both the convolution (2·n‑1) and the
    # correlation with w (n+2·n‑2 = 3·n‑2)
    N = _next_pow2(3 * n - 2)

    best_local_score = float("inf")
    best_local_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- full self‑convolution and current objective -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- smooth‑max temperature schedule -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)

        # ----- soft‑max weights -----
        w = np.exp(beta * (conv - M))          # 0 ≤ w ≤ 1, w_max = 1

        # ----- weighted correlation (gradient w.r.t. p) -----
        rev = _weighted_rev_fft(p, w, N)

        # Gradient of 2·n·M_β w.r.t. p
        grad_p = 4.0 * n * rev

        # ----- transform gradient to logits (softmax Jacobian) -----
        dot = np.dot(p, grad_p)
        grad_u = p * (grad_p - dot)

        # ----- Adam update in log‑space -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + it * 1e-4)          # simple decay

        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter to escape plateaus -----
        if jitter_interval and (it % jitter_interval == 0):
            u += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex via softmax -----
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)            # stability
        p = exp_u / exp_u.sum()

    return best_local_p, best_local_score

    # Keep only lengths that fit within the time budget (rough heuristic)
    length_schedule = [l for l in length_schedule if l >= 200]
    length_schedule = sorted(set(length_schedule))

    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ----- initialise probability vector for this length -----
        if best_seq.shape[0] == n:
            p_init = best_seq / math.sqrt(2.0 * n)   # sum‑to‑1 version
        else:
            # Upsample the previous best (linear interpolation)
            p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])
            old_x = np.linspace(0.0, 1.0, p_old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            p_init = np.interp(new_x, old_x, p_old)
            p_init = np.maximum(p_init, 0.0)
            if p_init.sum() == 0.0:
                p_init = np.full(n, 1.0 / n)
            else:
                p_init /= p_init.sum()

        print(f"[run] start optimisation for length = {n}")

        # ----- run the Adam optimiser for this length -----
        p_opt, local_score = _optimize_adam_length(
            n=n,
            p_start=p_init,
            deadline=deadline,
        )

        # Convert the probability vector back to the original scale
        a_opt = p_opt * math.sqrt(2.0 * n)
        cur_score = evaluate_sequence(a_opt.tolist())
        print(
            f"[run] length={n}, raw score={local_score:.6f}, evaluates to {cur_score:.6f}"
        )

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[global] new best score = {best_score:.6f} at length {n}")

    # --------------------------------------------------------------
    # Fallback: uniform distribution (should never be reached)
    if best_seq is None:
        n = 500
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
