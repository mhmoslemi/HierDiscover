        if j >= i:
            j += 1
        # fraction based on temperature
        max_frac = min(0.5, T / (cur_max + 1e-12))
        delta = a[i] * max_frac * np.random.rand()

        conv_new[i:i+n] -= 2.0 * delta * a
        conv_new[j:j+n] += 2.0 * delta * a
        conv_new[2*i] += delta * delta
        conv_new[2*j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        if new_max < cur_max or np.random.rand() < math.exp(-(new_max - cur_max) / (T + 1e-12)):

            a[j] += delta

"""
Hybrid Simulated‑Annealing + Sub‑gradient polishing for minimizing the peak of the
self‑convolution of a non‑negative step function.

The routine works with a fixed ℓ₁‑norm ( √(2 n) ) and a non‑negative
constraint.  For each candidate length it performs

1. A fast simulated‑annealing phase that updates the convolution *incrementally*
   (O(n) per move).  The move consists of transferring a small amount of mass
   from one index to another while keeping the total sum constant.

2. A short sub‑gradient refinement (Adam‑style) that fine‑tunes the best
   candidate from the annealing phase.

The algorithm respects the 1000 s wall‑clock budget, prints progress,
and finally returns the best sequence found (as a plain ``list[float]``)
ready to be evaluated by ``evaluate_sequence``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a*a using an FFT (fast for
    n up to a few ten‑thousands).  Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A                                     # element‑wise product
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    return conv


def _conv_max_and_array(a: np.ndarray):
    """
    Returns (max value of a*a, the full convolution array).
    """
    conv = _full_convolution(a)
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at the current point.
    active_idxs are the indices k where convolution value ≈ max.
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx                       # partner indices for this shift
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _apply_move_convolution(a: np.ndarray,
                           conv: np.ndarray,
                           i: int,
                           j: int,
                           delta: float) -> None:
    """
    Apply the move “transfer ``delta`` from index i to index j’’.
    ``a`` and ``conv`` are modified *in place*.
    The update uses the exact linear formula:
        conv' = conv - 2·δ·(a shifted by i) + 2·δ·(a shifted by j)
                + δ²·e_{2i}+δ²·e_{2j}−2·δ²·e_{i+j}
    where e_k denotes a unit impulse at index k.
    ``a`` is updated after the convolution change.
    """
    n = a.shape[0]

    # update convolution (use the *old* a values)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta

    # update the vector itself
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def simulated_annealing(a: np.ndarray,
                        target_sum: float,
                        deadline: float):
    """
    Fast SA that works directly on the convolution array.
    Returns the best vector seen and its peak‑convolution value.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    cur_max = float(conv.max())
    best_seq = a.copy()
    best_val = cur_max

    # temperature initialised as a fraction of the current objective
    T = max(cur_max * 0.3, 1e-12)
    cooling = 0.9998
    iteration = 0

    while time.time() < deadline and T > 1e-8:
        # pick a source index with some mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # size of the transferred mass scales with temperature
        max_frac = min(0.5, T / (cur_max + 1e-12))
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # tentative update (work on a copy of the convolution)
        conv_new = conv.copy()
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta
        new_max = float(conv_new.max())

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
                # occasional progress notice
                # print(f"[SA] iter {iteration:7d}  new best M = {best_val:.6f}")

        # temperature schedule
        T *= cooling
        iteration += 1

        # every few thousand steps re‑normalise to avoid drift
        if iteration % 5000 == 0:
            s = a.sum()
            if s > 0.0:
                a *= (target_sum / s)
            else:
                a[:] = target_sum / n
            conv = _full_convolution(a)
            cur_max = float(conv.max())
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def gradient_refine(a: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    max_iters: int = 2000):
    """
    A few Adam‑style sub‑gradient steps that polish the result of SA.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_val, _ = _conv_max_and_array(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        M, conv = _conv_max_and_array(a)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of the max‑convolution term
        subgrad = _subgradient(a, active)

        # Adam update (the outer scaling factor of the objective is constant)
        m = beta1 * m + (1.0 - beta1) * subgrad
        v = beta2 * v + (1.0 - beta2) * (subgrad * subgrad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (non‑negative, fixed ℓ₁‑norm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # keep the best observed point
        cur_val, _ = _conv_max_and_array(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **_kwargs):
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  The function respects the 1000 s wall‑clock budget,
    prints intermediate progress, and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Load the previously discovered construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)     # type: ignore
        print("[info] loaded previous construction, length =", base.shape[0])
    except Exception:
        base = None
        print("[info] no previous construction available – will start from scratch")

    # ------------------------------------------------------------------
    # Choose a handful of problem sizes (original and a few larger multiples).
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100
    factors = [1.0, 1.25, 1.5, 2.0, 3.0]   # keep within allowed limits
    lengths = sorted({max(300, min(12000, int(n0 * f))) for f in factors})

    print("[info] will explore lengths:", lengths)

    global_best_seq = None
    global_best_val = float('inf')

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Allocate a slice of the remaining budget for this length
        time_left = deadline - time.time()
        # keep a small buffer for the next lengths
        slice_deadline = time.time() + min(0.45 * time_left, 250.0)

        restarts = 3 if n <= 4000 else 2
        for r in range(restarts):
            if time.time() >= slice_deadline:
                break

            # ----- initial vector -------------------------------------------------
            if base is not None and n == n0 and r == 0:
                a = base.copy()
            else:
                rnd = np.random.exponential(scale=1.0, size=n)
                a = rnd * (target_sum / rnd.sum())

            # blend a little with the current global best if lengths match
            if global_best_seq is not None and global_best_seq.shape[0] == n:
                mix = 0.2
                a = mix * global_best_seq + (1 - mix) * a
                a = np.maximum(a, 0.0)
                a *= target_sum / a.sum()

            # ----- simulated annealing -------------------------------------------
            sa_deadline = time.time() + 0.55 * (slice_deadline - time.time())
            seq_sa, val_sa = simulated_annealing(a.copy(), target_sum, sa_deadline)

            # ----- gradient polishing (Adam‑style) -------------------------------
            if time.time() < slice_deadline:
                refine_deadline = slice_deadline
                seq_ref, val_ref = gradient_refine(seq_sa.copy(),
                                                   target_sum,
                                                   refine_deadline,
                                                   max_iters=3000)
                # keep the better of SA and refined version
                if val_ref < val_sa:
                    seq_best, val_best = seq_ref, val_ref
                else:
                    seq_best, val_best = seq_sa, val_sa
            else:
                seq_best, val_best = seq_sa, val_sa

            # ----- update global best ---------------------------------------------
            if val_best < global_best_val:
                global_best_val = val_best
                global_best_seq = seq_best.copy()
                print(f"[global] new best M = {global_best_val:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Final sanity check via the official evaluator
    if global_best_seq is None:
        # fallback – return a uniform vector of moderate size
        n_fallback = 2100
        target_sum_fallback = math.sqrt(2.0 * n_fallback)
        global_best_seq = np.full(n_fallback, target_sum_fallback / n_fallback)
        print("[fallback] no candidate generated – using uniform vector")

    final_score = evaluate_sequence(global_best_seq.tolist())
    print(f"[done] best reported M = {global_best_val:.6f}, official score = {final_score:.6f}, length = {len(global_best_seq)}")
    return global_best_seq.tolist()
