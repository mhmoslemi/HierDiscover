def _frank_wolfe_step(a, M, target_sum, deadline):
    # Solve LP to find g
    g = _lp_max_sum(a, M)
    if g is None:
        return a, M
    sum_g = g.sum()
    if sum_g <= target_sum:
        return a, M
    # Scale g to have same L1 as a
    g_scaled = g * (target_sum / sum_g)
    # Try mixing
    for t in [0.5, 0.3, 0.1]:
        a_new = (1 - t) * a + t * g_scaled
        new_M, _ = _conv_max_fft(a_new)
        if new_M < M:
            a = a_new
            M = new_M
            break
    return a, M
```

def _conv_max_fft(a: np.ndarray):
    """Return (max_convolution, full_convolution) using an FFT of suitable size."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution term.
    For each active shift k we add the partner entries a_{k-i}.
    The result is averaged over the active set.
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad

    base_lr = 0.04
    jitter_interval = 4000
    best_vec = a.copy()
    best_M, _ = _conv_max_fft(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        M, conv = _conv_max_fft(a)

        # active shifts (within a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑convolution term
        rev = _subgradient(a, active)

        # derivative of the whole objective
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # keep the best observed
        if M < best_M:
            best_M = M
            best_vec = a.copy()
            print(f"[Adam] n={n:5d} it={it:7d}  new best M={best_M:.6f}")

    return best_vec, best_M


# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A (size (2n‑1) × n) such that
    (A @ g)[k] = (a * g)[k]  for all k.
    """
    n = a.shape[0]
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over indices of a that contribute to shift k
        for i in range(i_start, i_end + 1):
            j = k - i                     # the index of g
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to a * g ≤ M (pointwise) , g ≥ 0
    Returns the optimal vector g (or None if the LP failed).
    """
    if not _HAS_SCIPY:
        return None                     # LP not available

    n = a.shape[0]
    A = _build_conv_matrix(a)           # (2n‑1) × n
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                     # minimise -Σ g_j  ⇔ maximise Σ g_j
    bounds = [(0, None)] * n

    # Using HiGHS (the default for recent SciPy) – robust and fast
    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def _frank_wolfe_step(a: np.ndarray, M: float,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a single Frank‑Wolfe style improvement:
    * solve the LP that pushes as much mass as possible without exceeding M
    * mix the current vector with a scaled LP solution
    """
    if not _HAS_SCIPY:
        return a, M

    g = _lp_max_sum(a, M)
    if g is None:
        return a, M

    sum_g = g.sum()
    if sum_g <= target_sum + 1e-12:
        # No improvement possible – the LP could not increase ℓ₁‑norm
        return a, M

    # Scale g back to the original ℓ₁‑norm
    g_scaled = g * (target_sum / sum_g)

    # Try a few mixing coefficients, keep the first that reduces the peak
    for t in (0.5, 0.3, 0.1):
        a_new = (1.0 - t) * a + t * g_scaled
        new_M, _ = _conv_max_fft(a_new)
        if new_M < M - 1e-12:
            print(f"[FW] reduction {M:.6f} → {new_M:.6f}  (t={t})")
            return a_new, new_M
        # time guard
        if time.time() >= deadline:
            break

    return a, M


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  target_sum: float,
                  steps: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic improvement:
    repeatedly move a tiny amount of mass from one index to another if it
    reduces the peak autocorrelation.  A simple simulated‑annealing schedule
    (occasional uphill acceptance) is added when time permits.
    """
    n = a.shape[0]
    best = a.copy()
    best_M, _ = _conv_max_fft(best)

    # temperature schedule for occasional uphill moves
    start_temp = best_M * 0.05
    end_temp = 1e-4
    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] <= 0:
            continue

        # transfer a small fraction of the source entry
        delta = best[i] * 0.02
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # ℓ₁‑norm unchanged, non‑negativity preserved

        M, _ = _conv_max_fft(cand)
        if M < best_M:
            best, best_M = cand, M
        else:
            # accept with a temperature‑dependent probability
            if step % 50 == 0:   # cheap temperature update every few steps
                frac = step / steps
                T = start_temp * (1 - frac) + end_temp * frac
            else:
                T = start_temp
            if np.random.rand() < math.exp(-(M - best_M) / max(T, 1e-12)):
                best, best_M = cand, M

    return best, best_M


# ----------------------------------------------------------------------
def _random_initial_vector(n: int, target_sum: float) -> np.ndarray:
    """
    Generate a random non‑negative vector of length n with ℓ₁‑norm = target_sum.
    The distribution is randomly chosen among a few families to diversify
    the search space.
    """
    mode = np.random.choice(['exp', 'power', 'uniform'])
    if mode == 'exp':
        vec = np.random.exponential(scale=1.0, size=n)
    elif mode == 'power':
        p = np.random.uniform(0.2, 4.0)          # shape exponent
        idx = np.arange(n)[::-1] + 1
        vec = idx.astype(float) ** p
    else:   # uniform
        vec = np.ones(n, dtype=float)

    vec *= target_sum / vec.sum()
    return vec

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimizes ``evaluate_sequence``.
    The function respects a wall‑clock budget (≈budget_s seconds) and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    # ------------------------------------------------------------------
    # Try to load a previously discovered construction (if available)
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        print("[info] loaded existing construction, length =", len(base))
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Search parameters
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0, 4.0, 5.0]
    base_len = len(base) if base is not None else 2000
    lengths = sorted({max(300, min(12000, int(base_len * f))) for f in length_factors})
    print("[info] exploring lengths:", lengths)

    global_best_vec = None
    global_best_M = float('inf')

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of random restarts per length (more for smaller problems)
        restarts = 4 if n <= 3000 else 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # initialise -------------------------------------------------
            if base is not None and n == len(base) and r == 0:
                init = base.copy()
                # ensure correct ℓ₁‑norm
                init *= target_sum / init.sum()
                print(f"[init] using loaded base (len={n})")
            else:
                init = _random_initial_vector(n, target_sum)

            # optional slight mix with current global best (if same size)
            if global_best_vec is not None and global_best_vec.shape[0] == n:
                mix = 0.15
                init = mix * global_best_vec + (1 - mix) * init
                init = np.maximum(init, 0.0)
                init *= target_sum / init.sum()

            print(f"[run] length={n}, restart={r + 1}/{restarts}")

            # --------------------------------------------------------------
            # Adam optimisation (sub‑gradient descent)
            remaining = deadline - time.time()
            max_time = min(180.0, remaining * 0.5)           # generous time slice
            sub_deadline = time.time() + max_time
            opt_vec, opt_M = _adam_optimize(init, target_sum, sub_deadline,
                                            max_iters=5_000_000)

            # --------------------------------------------------------------
            # Frank‑Wolfe / LP refinement (up to 3 mixing steps)
            for _ in range(3):
                if time.time() >= deadline:
                    break
                opt_vec, opt_M = _frank_wolfe_step(opt_vec, opt_M,
                                                  target_sum, deadline)

            # --------------------------------------------------------------
            # Local stochastic search (cheap greedy + occasional uphill)
            if time.time() < deadline:
                ls_steps = int(2e4 * (n / 4000))            # more steps for larger n
                opt_vec, opt_M = _local_search(opt_vec, target_sum,
                                               steps=ls_steps,
                                               deadline=deadline)

            # --------------------------------------------------------------
            # update global best
            if opt_M < global_best_M:
                global_best_M = opt_M
                global_best_vec = opt_vec.copy()
                print(f"[global] new best M={global_best_M:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Final evaluation through the official interface (mandatory)
    if global_best_vec is None:
        # Fallback – extremely unlikely but keep the harness happy
        global_best_vec = np.full(500, 1.0, dtype=float)
        global_best_vec *= math.sqrt(2.0 * len(global_best_vec)) / global_best_vec.sum()

    final_score = evaluate_sequence(global_best_vec.tolist())
    print(f"[done] reported M={global_best_M:.6f}, official score={final_score:.6f}, "
          f"length={global_best_vec.shape[0]}")
    return global_best_vec.tolist()
