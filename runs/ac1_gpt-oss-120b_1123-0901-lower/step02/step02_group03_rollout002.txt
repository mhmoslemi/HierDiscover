def conv_full_fft(a):
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv
```

```
def gradient_of_active(a, active_idxs):
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad
```

def generate_initial(n, shape_type, rng):
    if shape_type == 'random':
        return rng.random(n)
    elif shape_type == 'uniform':
        return np.full(n, 1.0)
    elif shape_type == 'triangular':
        # increasing then decreasing
        half = n // 2
        inc = np.arange(1, half+1)
        dec = np.arange(half, 0, -1) if n%2==0 else np.arange(half,0,-1)
        tri = np.concatenate([inc, dec])
        return tri.astype(float)
    elif shape_type == 'exponential':
        # decaying from start
        lam = rng.uniform(0.001, 0.01)
        x = np.arange(n)
        exp = np.exp(-lam * x)
        return exp
    elif shape_type == 'sinusoid':
        x = np.arange(n)
        sin = np.sin(np.pi * x / n) + 0.1
        return sin
    else:
        return rng.random(n)

def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto the simplex {x >= 0, sum(x) = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv

    """Sub‑gradient of max‑entry of a*a with respect to a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad

        # Either reuse a piece of the current best or start from scratch.
        if best_seq is not None and rng.random() < 0.25:
            src = best_seq
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = rng.random(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            kind = rng.choice(init_kinds)
            init = generate_initial(n, kind, rng)

        best_seq, best_score = run_optimisation_instance(
            n, init, best_seq, best_score, deadline, rng
        )

        # Stop early if we already beat the target bound.
        if best_score < 1.503:
            print(f"Target achieved: best_score={best_score:.6f}")
            break

    # Fallback – uniform distribution (should never be needed).
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
