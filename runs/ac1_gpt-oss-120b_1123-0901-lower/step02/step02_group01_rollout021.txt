"""
Propose a step‑function (non‑negative sequence) with a low
evaluation score.  

The algorithm works in three stages:

1. **Initial candidates** – geometric decays, Gaussian‑like,
   arcsine‑like, random Dirichlet vectors and (if provided) the
   previous best sequence.  Short candidates are up‑sampled when
   we move to a larger length.

2. **Projected Adam optimisation** – we minimise the maximum entry
   of the self‑convolution (equivalently the original objective)
   using the Adam optimiser.  The gradient of the pointwise maximum
   is obtained by a sub‑gradient: for the set of indices where the
   convolution attains its maximum we form a uniform weight vector
   `w` and compute `∇ = 2·(w ★ a)` via an FFT‑based correlation.
   After each Adam step we project back onto the probability simplex
   (`a_i ≥ 0, ∑a_i = 1`).  Stagnation triggers a tiny random shake
   and a reset of the Adam moments.

3. **Multi‑scale refinement** – after a phase with a given length
   the best solution is linearly interpolated to a larger length,
   providing a richer search space.  The process repeats for several
   lengths until the time budget (≈ 1000 s) expires or the target
   objective (≈ 1.503) is reached.

The routine respects the 1000 s budget, prints progress, and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all numbers non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence` (sum(a) is assumed > 0)."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    up /= up.sum()
    return up
