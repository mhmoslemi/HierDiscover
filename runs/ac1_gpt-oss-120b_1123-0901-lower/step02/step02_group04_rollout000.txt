    conv = _conv_self(a)  # shape 2n-1
    maxc = conv.max()
    # stable softmax
    shifted = conv - maxc

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(\mathbf a)=\frac{2\,n\,\max_k (a*a)[k]}{(\sum_i a_i)^2}\;,
\]

where `a*a` is the full convolution of `a` with itself.  
Because the objective is invariant under a common scaling of the
coefficients, we can fix the ‑ℓ₁‑norm

\[
\sum_i a_i = \sqrt{2n}\qquad\Longrightarrow\qquad  F(\mathbf a)=
\max_k (a*a)[k]\; .
\]

Hence we have to **push the peak of the autocorrelation down** while
keeping the vector non‑negative and on the simplex.

The previous implementation used a sub‑gradient on the single maximal
lag.  The new optimiser replaces that by a *smooth* approximation of the
maximum (`soft‑max`) and an **Adam**‑style projected optimisation.  The
soft‑max spreads the gradient over the lags whose autocorrelation values
are close to the maximum, yielding a much richer descent direction.
A schedule that slowly raises the soft‑max temperature focuses the
optimiser onto the true maximum in later stages.

The optimiser works as follows

1. **Initial candidates** – low‑dimensional families (exponential,
   triangular, …) are evaluated quickly.  The best of those seeds the
   long‑run optimiser.
2. **Adam optimisation** – for each seed we run Adam on the smooth‑max
   objective, projecting after every step onto the simplex
   `{x≥0, Σx=√(2n)}`.  The soft‑max temperature is annealed from `β=2`
   to `β=30`.
3. **Random restarts** – many different lengths and random
   exponential‑type starts are tried until the time budget expires.
4. **LP refinement** – if SciPy is available we perform one (or a few)
   linear‑programming refinements exactly as in the state‑of‑the‑art
   description.
5. **Fallback** – if nothing succeeded we return a uniform sequence.

All heavy work (convolutions, correlations) is performed with real FFTs
(`O(n log n)`).  The implementation contains no closures or lambdas and
conforms to the required function signature.

```python
"""
propose_candidate

A projected Adam optimiser that minimises the peak of the autocorrelation
of a non‑negative step function.  The loss is a smooth approximation of
the max‑convolution, allowing gradients that take all “almost‑max’’ lags
into account.  The algorithm:

* scans a few parametric families (exponential, triangular, …) for a
  quick baseline,
* runs Adam with a soft‑max temperature that is increased during the
  optimisation,
* performs many random restarts with lengths up to a few thousand,
* optionally refines the best result with a linear‑programming step
  (the “LP‑step’’ from the published SOTA).

All vectors are projected onto the simplex
{ x ≥ 0 , Σ x = √(2 n) } after every update, guaranteeing feasibility.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution a * a using real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth‑max of the autocorrelation
# ----------------------------------------------------------------------
def _smoothmax_grad(a: np.ndarray, beta: float):
    """
    Compute the gradient of  g(a)= (1/beta) * log( Σ exp(beta·(a*a)[k]) )
    together with the current autocorrelation.
    Returns (grad, conv) where conv = a*a .
    """
    n = a.shape[0]
    conv = _conv_self(a)                # length 2n‑1
    maxc = conv.max()
    # stable softmax weights
    shifted = conv - maxc                # max element becomes 0
    exps = np.exp(beta * shifted)       # exp(β·(c_i‑max))
    w = exps / exps.sum()                # softmax, ∑w = 1

    # gradient = 2 * correlate(w, a)
    # correlation can be obtained as convolution with the reversed vector
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    corr_full = np.fft.irfft(fw * fa_rev, n=size)
    start = n - 1                         # index of first “valid’’ entry
    grad = 2.0 * corr_full[start:start + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max objective
# ----------------------------------------------------------------------
def _run_adam(start_vec: np.ndarray,
              target_sum: float,
              deadline: float,
              cur_best_score: float,
              max_iters: int = 5000):
    """
    Projected Adam on the smooth‑max loss.
    Returns the best sequence found and its (max‑convolution) score.
    """
    a = start_vec.astype(float).copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    beta_start = 2.0
    beta_end = 30.0

    best_local_seq = a.copy()
    best_local_score = float("inf")

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal the smooth‑max temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / max_iters)

        grad, conv = _smoothmax_grad(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / (1.0 + 1e-3 * t)               # slowly decreasing step size

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex {x≥0, Σx = target_sum}
        a = _project_onto_simplex(a, target_sum)

        # occasional evaluation (every 10 iterations)
        if t % 10 == 0:
            score = float(conv.max())          # because Σa = target_sum
            if score < best_local_score:
                best_local_score = score
                best_local_seq = a.copy()
                print(f"[{time.time():.2f}] Adam n={n} iter={t} score={score:.6f}")

            # early exit if we already beat the global best
            if score < cur_best_score:
                # returning early is safe – the caller will keep the sequence
                best_local_score = score
                best_local_seq = a.copy()
                break

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# Shape factories (non‑normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)

            a = _shape_quad_desc(n, beta)

            a = a / a.sum() * target
            _maybe_update(a)
```
