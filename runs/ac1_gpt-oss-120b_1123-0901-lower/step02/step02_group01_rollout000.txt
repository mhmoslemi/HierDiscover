"""
propose_candidate
=================

Search for a step‑function (non‑negative sequence) with a small
``evaluate_sequence`` value.

The algorithm is a **projected Adam optimiser** applied to the
smooth‑max surrogate of the objective

    F(a) = 2·n·max(a*a) / (sum(a)^2) ,   a_i ≥ 0 ,  sum(a)=1.

* ``λ`` (the log‑sum‑exp temperature) is increased gradually (continuation)
  so that early iterations see a very smooth landscape, whilst later
  iterations focus on the true maximum.
* After each Adam step we **project onto the probability simplex**
  (Euclidean projection) to keep the constraints satisfied.
* When progress stalls we add a tiny random shake to escape shallow
  minima.
* A short final polishing phase with a very large ``λ`` refines the
  solution toward the true max.
* The routine works for any length; we use ``n≈3000`` which gives a
  good trade‑off between expressive power and speed.
* If the global variable ``height_sequence_1`` (produced by a previous
  run) is available we up‑sample it and use it as an additional start
  point.

The whole search respects the 1000 s time budget; the best sequence
found is returned as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
def _full_convolution(a, size=None):
    """Return the linear self‑convolution a*a (length 2·len(a)-1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (the one used by evaluate_sequence)
def _objective(a):
    """2·n·max(conv) / (sum(a)^2) .  a must be non‑negative, sum>0."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) surrogate
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Gradient of φ_λ(a) = (1/λ)·log Σ_k exp(λ·c_k)  where c = a*a.
    Returns ∇φ_λ(a) = 2·(w ★ a)  (linear cross‑correlation).
    """
    n = len(a)
    # numerically stable soft‑max weights
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    # cross‑correlation w ★ a  (linear)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice corresponding to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _initial_candidates(n, rng):
    """
    Produce a list of promising start vectors of length n.
    Returns a list of numpy arrays (already normalised to sum‑=1).
    """
    cand = []

    # 1) geometric sequences a_i ∝ r^i  (r in (0,1))
    for exponent in np.linspace(0.8, 3.0, 30):
        r = 1.0 - exponent / n
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        cand.append(seq)

    # 2) random Dirichlet draws
    for _ in range(15):
        seq = rng.dirichlet(np.ones(n))
        cand.append(seq)

    # 3) if a previous solution exists, up‑sample it
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.ndim == 1:
            # linear interpolation to new length n
            old_x = np.linspace(0.0, 1.0, len(prev_arr))
            new_x = np.linspace(0.0, 1.0, n)
            up = np.interp(new_x, old_x, prev_arr)
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up = up / up.sum()
                cand.append(up)

    # Keep only unique (tiny duplicate removal)
    uniq = []
    seen = set()
    for v in cand:
        h = hash(v.tobytes())
        if h not in seen:
            seen.add(h)
            uniq.append(v)
    return uniq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 1000).
    **kwargs
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a balance between quality and speed.
    n = 3000

    # Pre‑compute FFT size used for all cross‑correlations (linear)
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))

    # ------------------------------------------------------------------
    # Generate initial candidates and pick the best one.
    rng = np.random.default_rng(seed)
    candidates = _initial_candidates(n, rng)

    best_seq = None
    best_val = float("inf")
    for seq in candidates:
        if time.time() > deadline:
            break
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Init] objective={val:.6f}")

    # If, for any reason, we have no candidate (should not happen), use uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Adam optimiser with continuation on λ.
    seq = best_seq.copy()
    lam = 5.0          # start smoothing parameter
    lam_factor = 1.03  # λ growth per iteration
    lam_max = 2000.0

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.08       # base learning rate for Adam
    alpha_decay = 0.99995

    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    it = 0
    last_improve = time.time()

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # 1) self‑convolution
        conv = _full_convolution(seq)

        # 2) smooth‑max gradient
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_corr)

        # 3) Adam update (before projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        new_seq = seq - step
        # 4) enforce simplex constraints
        new_seq = _project_onto_simplex(new_seq)

        # 5) evaluate exact objective
        val = _objective(new_seq)

        if val < best_val:
            best_val = val
            best_seq = new_seq.copy()
            last_improve = time.time()
            print(f"[Iter {it:05d}] new best = {val:.6f}")

        # 6) stagnation handling – tiny random shake
        if time.time() - last_improve > 2.0:
            noise = (rng.random(n) - 0.5) * 1e-3
            new_seq = _project_onto_simplex(new_seq + noise)
            last_improve = time.time()

        # 7) prepare next iteration
        seq = new_seq
        lam = min(lam_max, lam * lam_factor)
        alpha *= alpha_decay
        it += 1

    # ------------------------------------------------------------------
    # Final polishing with a very large λ (near‑max surface)
    polish_iters = 2000
    lam_polish = 5000.0
    for i in range(polish_iters):
        if time.time() > deadline:
            break
        conv = _full_convolution(seq)
        grad = _softmax_gradient(seq, conv, lam_polish, size_corr=size_corr)

        # simple SGD step (no Adam) – small learning rate
        step = 0.02 * grad
        seq = _project_onto_simplex(seq - step)

        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Polish {i:04d}] improved to {val:.6f}")

    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
