def _adam_optimize(a0, deadline, max_iter=10_000_000):
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)
    # Ensure normalization
    a = a * (target_sum / a.sum())
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    best_local_seq = a.copy()
    best_local_score = float('inf')
    while time.time() < deadline and iteration < max_iter:
        conv = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(a), n=2*n-1)[:2*n-1]
        M = conv.max()
        # compute active set
        # tolerance use relative? use absolute 1e-12
        active = np.where(conv >= M - 1e-12)[0]
        # subgradient
        rev = _subgradient(a, active)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const
        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = lr0 / (1 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # jitter every some iterations
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())
        # evaluate
        cur_score = M  # because normalized
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
        # optional early break if improvement tiny
    return best_local_seq, best_local_score

objective.  The algorithm combines:
  * Adam optimisation on the exact max‑convolution sub‑gradient for several
    candidate lengths (original length and larger interpolated versions).
  * Multiple random restarts (including uniform and random initial vectors).
  * A cheap hill‑climbing phase that moves a small amount of mass between

import time
import math
import numpy as np

def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _subgradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Compute the average reversed vector needed for the sub‑gradient.
    The i‑th component is (1/|active|) * sum_{k in active} a_{k-i},
    where out‑of‑range indices are ignored.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= float(active.size)
    return rev


def _adam_optimize(a0: np.ndarray, deadline: float, max_iters: int = 5_000_000):
    """
    Adam optimiser working on the sub‑gradient of the max‑convolution.
    Returns the best vector encountered and its (normalised) score.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # enforce the required normalisation
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

        # active set (indices attaining the max, within tiny tolerance)
        tol = 1e-12

        # sub‑gradient of the max‑convolution objective (normalised)
        rev = _subgradient(a, active)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple learning‑rate decay
        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set (non‑negative + fixed sum)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional tiny Gaussian jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # keep the best vector seen so far
        cur_score = M  # because the normalisation makes the objective equal to M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()

    return best_local_seq, best_local_score


def _hill_climb(a0: np.ndarray, score0: float, deadline: float,
                max_moves: int = 200_000):

    Very cheap hill‑climbing: repeatedly move a random fraction of mass from
    one position to another and keep the change only if it strictly improves
    the max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = score0

        new_score = float(conv.max())
        if new_score < best_score - 1e-12:
            a = a_new
            best_score = new_score

        lengths.append(np.random.randint(300, max_len + 1))
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        # Build a collection of starting vectors for this length
        inits = []

        if n == n0:
            inits.append(base.copy())
        else:
            # linear interpolation of the base shape onto the new grid
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            inits.append(np.maximum(interp, 0.0))

        # uniform vector
        uniform = np.full(n, math.sqrt(2.0 * n) / n)
        inits.append(uniform)

        # two random non‑negative vectors
        for _ in range(2):
            rnd = np.random.rand(n)
            rnd = np.maximum(rnd, 0.0)
            rnd = rnd * (math.sqrt(2.0 * n) / rnd.sum())
            inits.append(rnd)

        print(f"[run] optimisation for length={n}  ({len(inits)} starts)")

        for init_a in inits:
            if time.time() >= deadline:
                break

            # Adam optimisation from this start
            seq, score = _adam_optimize(init_a, deadline, max_iters=2_000_000)
            if score < best_score:
                best_seq = seq.copy()
                best_score = score
                print(f"[global] new best (length {n}) -> {best_score:.6f}")

            # Light hill‑climbing to polish the result
            seq_hc, score_hc = _hill_climb(seq, score, deadline,
                                           max_moves=100_000)
            if score_hc < best_score:
                best_seq = seq_hc.copy()
                best_score = score_hc
                print(f"[global] new best after hill‑climb (len {n}) -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (just in case) and sanity check
    n_best = best_seq.shape[0]
    target_best = math.sqrt(2.0 * n_best)
    best_seq = best_seq * (target_best / best_seq.sum())
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] final score = {final_score:.6f}, length = {n_best}")

    # Return as plain Python list for the harness
    return best_seq.tolist()
