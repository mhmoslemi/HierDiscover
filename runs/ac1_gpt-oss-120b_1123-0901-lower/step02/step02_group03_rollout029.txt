def correlation_fft(a, w):
    n = len(a)
    m = len(w)
    full_len = n + m - 1
    n_fft = 1 << ((full_len - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    W = np.fft.rfft(w, n_fft)
    corr = np.fft.irfft(A * np.conj(W), n_fft)[:full_len]
    return corr

def project_onto_simplex(v, s):
    v = np.asarray(v, dtype=float).copy()
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_full_fft(a):
    n = a.shape[0]
    m = 2*n - 1
    n_fft = 1 << ((m-1).bit_length())
    A = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(A * A, n_fft)[:m]
    return conv

assistantfinal**Solution Overview**

The objective  

\[
F(a)=\frac{2n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is *scale invariant*.  By projecting each candidate onto the simplex  

\[
\{a_i\ge 0,\;\sum_i a_i = \sqrt{2n}\}
\]

the denominator becomes constant and the objective reduces to  

\[
F(a)=\max_k (a*a)_k .
\]

The non‑negative, normalized vector \(a\) is therefore searched for a **small maximal
self‑convolution coefficient**.

The previous implementation used a sub‑gradient on the *active* (max‑tight) entries.
Below we employ a **smooth approximation of the max** (log‑sum‑exp) which yields a
dense gradient that steers the mass of the step function away from the positions
that contribute most to the large convolution values.  
A projected gradient descent with an annealed temperature and a light momentum term
explores the landscape efficiently.  Several independent restarts with different
lengths (up to 8000) are performed within the 1000 s budget.  Whenever a new
record is found the best sequence is stored and finally returned as a plain
Python list.

All helper functions are defined at top level, no closures are used, and the code
conforms to the required API.

```python
"""
Propose a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a smooth “log‑sum‑exp” approximation of the max and projected
gradient descent on the simplex Σ a_i = √(2n).  Multiple random restarts
with lengths 1500‑8000 are executed until the time budget expires.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}.
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    v = np.asarray(v, dtype=float).copy()
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Helper 2 – Linear convolution via FFT (full length).
# --------------------------------------------------------------------------- #
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of ``a`` with itself."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(A * A, n_fft)[:m]
    return conv


# --------------------------------------------------------------------------- #
# Helper 3 – Cross‑correlation (a ⋆ w) by FFT.
# --------------------------------------------------------------------------- #
def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the (full) cross‑correlation of ``a`` and ``w``:
    r[j] = Σ_i a[i] * w[i + j] .
    """
    n = a.shape[0]
    m = w.shape[0]
    full_len = n + m - 1
    n_fft = 1 << ((full_len - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    W = np.fft.rfft(w, n_fft)
    corr = np.fft.irfft(A * np.conj(W), n_fft)[:full_len]
    return corr


# --------------------------------------------------------------------------- #
# Helper 4 – Soft‑max (log‑sum‑exp) weights for a vector of values.
# --------------------------------------------------------------------------- #
def softmax_weights(x: np.ndarray, mu: float) -> np.ndarray:
    """
    Return a probability vector proportional to ``exp(x/mu)``.
    ``mu`` must be > 0.
    """
    max_x = np.max(x)
    exps = np.exp((x - max_x) / mu)
    w = exps / np.sum(exps)
    return w


# --------------------------------------------------------------------------- #
# Core optimisation routine for a single length ``n``.
# --------------------------------------------------------------------------- #
def run_one_instance(n: int,
                     init_vec: np.ndarray,
                     target_sum: float,
                     deadline: float):
    """
    Perform a smoothed gradient descent on the simplex for a fixed length ``n``.
    Returns the best sequence encountered and its (max‑convolution) score.
    """
    a = project_onto_simplex(init_vec.astype(float), target_sum)

    # hyper‑parameters
    mu = 0.20                # initial temperature for the soft‑max
    base_lr = 0.50           # base learning‑rate
    beta = 0.90              # heavy‑ball momentum coefficient
    momentum = np.zeros_like(a)

    best_local_seq = a.copy()
    best_local_score = float("inf")
    it = 0

    while time.time() < deadline:
        # ----- convolution and current objective (max entry) -----
        conv = conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)          # with Σa = √(2n) this equals evaluate_sequence()
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()

        # ----- soft‑max weights (smooth approximation of the max) -----
        w = softmax_weights(conv, mu)

        # ----- gradient of the smoothed objective -----
        # g_j = 2 * Σ_i a_i * w_{i+j}
        corr = correlation_fft(a, w)
        offset = len(w) - 1           # lag = -(len(w)-1) corresponds to index 0
        grad = 2.0 * corr[offset:offset + n]

        # ----- step size -----
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-12:
            break
        eta = base_lr * max_val / (max_grad + 1e-12)

        # ----- momentum update and projection onto the simplex -----
        momentum = beta * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)

        # ----- occasional random shake (simulated‑annealing style) -----
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum / (1000.0 * n))
            a_new = project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.95

        a = a_new

        # ----- anneal temperature and learning‑rate -----
        mu = max(mu * 0.99, 0.001)
        base_lr *= 0.9999
        it += 1

    return best_local_seq, best_local_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    global_best_seq = None
    global_best_score = float("inf")

    # --------------------------------------------------------------- #
    # Helper to obtain an initial vector (random or based on a known good).
    # --------------------------------------------------------------- #
    def make_initial(n, src_seq):
        if src_seq is not None and src_seq.size >= n:
            return src_seq[:n].copy()
        if src_seq is not None and src_seq.size < n:
            pad = np.random.rand(n - src_seq.size) * 1e-4
            return np.concatenate([src_seq, pad])
        return np.random.rand(n)

    # --------------------------------------------------------------- #
    # Main outer loop – many independent restarts of varying length.
    # --------------------------------------------------------------- #
    # length range – larger n gives a lower theoretical lower bound
    min_n, max_n = 1500, 8000

    # If a previous good construction is present, keep a reference.
    src_seq = None
    if "height_sequence_1" in globals():
        try:
            src_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            src_seq = None

    while time.time() < deadline:
        n = np.random.randint(min_n, max_n + 1)
        target_sum = np.sqrt(2.0 * n)           # Σ a_i = √(2n)

        init = make_initial(n, src_seq if np.random.rand() < 0.4 else None)

        # allocate a slice of the remaining budget for this run
        run_deadline = min(deadline, time.time() + 12.0)   # ≈12 s per run
        seq, score = run_one_instance(n, init, target_sum, run_deadline)

        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f"[{time.time():.2f}] New global best = {global_best_score:.6f} (n={n})")

        # early exit if we already beat the target bound
        if global_best_score < 1.503:
            break

    # --------------------------------------------------------------- #
    # Fallback – return a uniform construction if nothing was found.
    # --------------------------------------------------------------- #
    if global_best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        global_best_seq = np.full(n, target_sum / n, dtype=float)

    return global_best_seq.tolist()
```
