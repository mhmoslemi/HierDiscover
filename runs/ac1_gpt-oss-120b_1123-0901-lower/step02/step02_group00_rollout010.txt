def optimize_with_smoothing(n, a, best_seq, best_score, deadline, max_iter=None):
    target_sum = math.sqrt(2.0*n)
    # Adam parameters
    m=np.zeros_like(a); v=np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8; lr0=0.05
    iteration=0
    alpha = 1.0  # start smoothing
    alpha_max = 30.0
    # schedule: increase alpha slowly: maybe alpha *= (1+1e-4) each iteration.
    while time.time() < deadline:
        iteration+=1
        # compute conv and M
        conv = conv_full_fft(a)
        M = conv.max()
        # smoothing weights
        # compute softmax weights p
        # shift conv for numerical stability
        max_conv = M
        diff = conv - max_conv
        # using current alpha
        exp_vals = np.exp(alpha * diff)
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp
        # gradient via convolution with p_rev
        p_rev = p[::-1]
        # convolution a * p_rev
        # Using FFT
        L = len(p_rev)   # =2n-1
        N2 = 1 << ((n + L - 1).bit_length())
        A_fft = np.fft.rfft(a, n=N2)
        P_fft = np.fft.rfft(p_rev, n=N2)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=N2)
        # slice needed part
        start_idx = n - 1
        end_idx = 2*n - 1
        grad = 2.0 * conv_rev[start_idx:end_idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection non-negative and normalize
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a > 0:
            a = a * (target_sum / sum_a)
        else:
            a = np.full_like(a, target_sum/n)
        # increase alpha slowly
        if alpha < alpha_max:
            alpha *= 1.0005
        # jitter optionally
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.002 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        # evaluate current score
        cur_score = 2.0 * n * M / (sum_a * sum_a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:6d} alpha {alpha:.2f}] new best={best_score:.6f}")
        # early stop condition for stage: if iteration > max_iter break maybe
        # can also break if no improvement for many steps?
        # Continue until deadline reached.
        # Optionally restrict max iterations for smoothing stage.
    return best_seq, best_score

def optimize_with_subgradient(n, a, best_seq, best_score, deadline):
    target_sum = math.sqrt(2.0 * n)
    # enforce normalization
    a = a * (target_sum / a.sum())
    m=np.zeros_like(a); v=np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8; lr0=0.05
    iteration=0
    jitter_interval=5000
    while time.time() < deadline:
        iteration+=1
        # conv max and active set
        conv = conv_full_fft(a)
        M = conv.max()
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        rev = _subgradient(a, active)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum**3)
        grad = coeff * rev - const
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a > 0:
            a = a * (target_sum / sum_a)
        else:
            a = np.full_like(a, target_sum / n)
        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        # evaluate
        cur_score = 2.0 * n * M / (sum_a*sum_a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} subgrad iter {iteration:6d}] new best={best_score:.6f}")
        # break if no time left
    return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1. Load the best known construction (``height_sequence_1``) if available.
   Normalise it so that Σ a = √(2·n).

2. For a few selected lengths (original, 1.5× and 2× larger, capped at 8000)
   initialise a shape by linear interpolation of the base vector.

3. **Stage A – smoothed Adam.**
   * minimise a smooth approximation of the max‑convolution:
         M_α(a) = (1/α)·log( Σ_k exp(α·(a∗a)[k]) )
   * the gradient of M_α can be written as a convolution of ``a`` with the
     reversed soft‑max weights; it is computed with FFTs in O(n log n).
   * Adam updates are performed, after each step we project onto the
     feasible set (a_i≥0, Σa_i = √(2n)).
   * The smoothing parameter ``α`` starts small (0.5) and is slowly increased
     (up to ~30) so the optimiser first sees a smooth landscape and then
     increasingly focuses on the true maximum.

4. **Stage B – exact max‑subgradient.**
   * Starting from the best vector found in Stage A we run a few thousand
     Adam steps using the true sub‑gradient (active‑set of indices attaining
     the current maximum).  This refines the solution once we are already
     close to a local optimum.

5. Keep the globally best sequence (across all lengths) and return it.
   The whole search stops when the time budget (≈ 1000 s) is exhausted.

The code uses only ``numpy`` and the already‑imported ``evaluate_sequence``.
All helper functions are defined at module level (no closures or lambdas).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution of a real vector with itself (via FFT)
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a as a 1‑D np.ndarray (length 2n‑1)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    C = A * A                                    # elementwise product
    conv = np.fft.irfft(C, n=N)
    return conv[: 2 * n - 1]                    # keep the valid part


# ----------------------------------------------------------------------
# Sub‑gradient for the *exact* max‑convolution (active‑set version)
def _subgradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Compute the average reversed vector a_{k‑i} summed over all active shift
    indices k.  Returned array has the same shape as ``a``.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active_idxs.size)  # average when several k are active
    return rev


# ----------------------------------------------------------------------
# Adam optimiser that minimises the *smooth* max‑convolution for a given length
def _optimize_smoothing(
    n: int,
    a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on the smoothed max‑convolution (log‑sum‑exp) for vector length n.
    The function updates and returns the global best sequence / score.
    """
    target_sum = math.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0

    # smoothing schedule
    alpha = 0.5           # start with a very smooth landscape
    alpha_max = 30.0
    alpha_mul = 1.001    # geometric increase per iteration

    jitter_interval = 4000

    while time.time() < deadline:
        iteration += 1

        # ----- 1. convolution and soft‑max weights -----
        conv = _conv_full_fft(a)               # length 2n‑1
        M = conv.max()
        # soft‑max (log‑sum‑exp) weights, shifted by the current max for stability
        diff = conv - M
        exp_vals = np.exp(alpha * diff)
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp                  # length 2n‑1, sums to 1

        # ----- 2. gradient of the smoothed objective -----
        p_rev = p[::-1]                         # reversed weights
        L = p_rev.shape[0]                      # = 2n‑1
        N2 = 1 << ((n + L - 1).bit_length())    # FFT size for a * p_rev
        A_fft = np.fft.rfft(a, n=N2)
        P_fft = np.fft.rfft(p_rev, n=N2)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=N2)
        conv_rev = conv_rev[: n + L - 1]        # keep the valid part

        # slice that corresponds to ∑_i a_i p_{i+j}
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1]   # length n

        # ----- 3. Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 4. projection onto the feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:                                   # practically never occurs
            a = np.full_like(a, target_sum / n)

        # ----- 5. occasional jitter to escape plateaus -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.002 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- 6. possibly increase smoothing -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # ----- 7. evaluate current (exact) objective -----
        cur_score = 2.0 * n * M / (s * s)       # exact formula, s≈target_sum
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} it {iteration:7d} α {alpha:5.2f}] "
                f"new best score = {best_score:.6f}"
            )

        # let the outer loop enforce the time budget
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser that minimises the *exact* max‑convolution (active‑set)
def _optimize_exact_max(
    n: int,
    a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on the true max‑convolution (active‑set sub‑gradient) for a few
    thousand steps.  Starts from the vector ``a`` and updates the global best.
    """
    target_sum = math.sqrt(2.0 * n)

    # make sure we start with correct normalisation
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 3000

    while time.time() < deadline:
        iteration += 1

        # ----- 1. convolution and active set -----
        conv = _conv_full_fft(a)
        M = conv.max()
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]      # indices attaining the max

        # ----- 2. sub‑gradient for the exact max -----
        rev = _subgradient(a, active)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # ----- 3. Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 4. projection -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ----- 5. jitter -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- 6. evaluate -----
        cur_score = 2.0 * n * M / (s * s)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} exact it {iteration:7d}] "
                f"new best score = {best_score:.6f}"
            )

        # limit the exact‑max phase to about a few ten‑thousands of steps
        # (the outer deadline will break us earlier if needed)
        if iteration >= 200_000:
            break

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Main entry point required by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) minimises the
    ``evaluate_sequence`` objective within the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the surrounding harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Load a good starting construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        print("[info] loaded height_sequence_1 as base.")
    except Exception:
        # fallback: a modest‑size uniform vector
        base = np.ones(2100, dtype=float)
        print("[info] height_sequence_1 not available – using uniform fallback.")

    n0 = base.shape[0]
    target0 = math.sqrt(2.0 * n0)
    base = base * (target0 / base.sum())   # normalise to the required sum

    best_seq = base.copy()
    best_score = float("inf")
    # evaluate the initially supplied construction
    try:
        init_score = evaluate_sequence(best_seq.tolist())
        best_score = init_score
        print(f"[init] length={n0}, initial score={best_score:.6f}")
    except Exception as e:
        print("[warning] initial evaluate_sequence failed:", e)

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths (including larger interpolated ones)
    length_factors = [1.0, 1.5, 2.0]          # can be tuned
    lengths = sorted({int(max(300, n0 * f)) for f in length_factors})
    lengths = [min(l, 8000) for l in lengths]   # hard cap to keep FFT cheap

    # ------------------------------------------------------------------
    # 2. Optimise each length in turn
    for n in lengths:
        if time.time() >= deadline:
            break

        # ------ initialise a vector of the required length ------
        if n == n0:
            a0 = base.copy()
        else:
            # linear interpolation of the base shape onto a coarser grid
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, base)
            a0 = np.maximum(a0, 0.0)

        target_sum = math.sqrt(2.0 * n)
        a0 = a0 * (target_sum / a0.sum())

        print(f"[run] starting optimisation for length = {n}")

        # ------ Stage A : smoothed Adam ------
        best_seq, best_score = _optimize_smoothing(
            n=n,
            a=a0,
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
        )
        if time.time() >= deadline:
            break

        # ------ Stage B : exact max‑subgradient refinement ------
        best_seq, best_score = _optimize_exact_max(
            n=n,
            a=best_seq,          # start from the best found for this length
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
        )
        # loop continues with next length (if time permits)

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case nothing was found
    if best_seq is None or best_score == float("inf"):
        n = 2100
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
