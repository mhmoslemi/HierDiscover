"""
Propose a non‑negative step function that minimises the evaluate_sequence
objective.

Algorithm
---------
* Use several candidate lengths (the original length and a few larger
  interpolated versions up to 8000 entries).
* For each length run a few random restarts.
* Each restart is optimised with Adam on a *smoothed* version of the
  max‑convolution: we replace the hard max by a soft‑max with a
  temperature β that is increased during optimisation (β → ∞ recovers
  the true max).  The gradient of the smoothed objective can be written
  in closed form and is computed efficiently with FFT‑based convolutions.
* After every Adam step we project onto the feasible set
  (a_i ≥ 0 and Σ a_i = √(2 n)).
* Occasionally we add a tiny Gaussian jitter to escape plateaus and we
  keep the best sequence seen (according to the exact evaluate_sequence
  value) across all lengths and restarts.
* The whole search respects a global 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, N: int) -> np.ndarray:
    """
    Full convolution of a with itself using FFT of size N (>= 2*len(a)-1).
    Returns the real-valued convolution (length 2*len(a)-1).
    """
    a_fft = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(a_fft * a_fft, n=N)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Compute soft‑max weights w_i = exp(beta*conv_i)/∑exp(beta*conv) in a stable way."""
    if beta == 0:
        # uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    # subtract max for numeric stability
    m = np.max(conv)
    exp_term = np.exp(beta * (conv - m))
    sum_exp = np.sum(exp_term)
    # guard against divide‑by‑zero (should not happen)
    if sum_exp == 0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / sum_exp


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, w: np.ndarray, N_rev: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. a.
    Uses the relation ∂F/∂a_i = 2 * Σ_k w_k a_{k-i},
    which is a convolution of w with the reversed a.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=N_rev)
    a_rev_fft = np.fft.rfft(a_rev, n=N_rev)
    rev_full = np.fft.irfft(w_fft * a_rev_fft, n=N_rev)
    # slice the centre part corresponding to indices 0 .. n-1
    start = n - 1
    rev = rev_full[start : start + n]
    return 2.0 * rev


# ----------------------------------------------------------------------
def _adam_optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_restarts: int = 2,
):
    """
    Run several Adam optimisations for a fixed length `n`.
    Returns the globally best sequence/score found for this length.
    """
    target_sum = math.sqrt(2.0 * n)

    # Pre‑compute FFT sizes
    N_conv = _next_pow2(2 * n - 1)
    N_rev = _next_pow2(3 * n - 2)   # length of w (2n‑1) + a_rev (n)

    # Beta schedule (iteration threshold, beta value)
    beta_sched = [
        (0, 0.5),
        (20000, 2.0),
        (50000, 5.0),
        (100000, 10.0),
        (200000, 20.0),
        (400000, 40.0),
    ]

    def _beta_at(iter_idx: int) -> float:
        """Return current β according to the schedule."""
        for thr, b in reversed(beta_sched):
            if iter_idx >= thr:
                return b
        return beta_sched[0][1]

    # Use a few random initialisations for robustness
    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        # Initialise a (copy and optionally perturb)
        a = init_a.copy()
        if restart > 0:
            # add a small random perturbation to escape identical runs
            jitter = np.random.randn(n) * 0.01 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
        # Normalise to the required total sum
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr0 = 0.05
        iteration = 0
        jitter_interval = 6000

        while time.time() < deadline:
            # ---------- convolution and max ----------
            conv = _conv_full_fft(a, N_conv)
            M = float(conv.max())

            # ---------- soft‑max weights ----------
            beta = _beta_at(iteration)
            w = _softmax_weights(conv, beta)

            # ---------- gradient ----------
            grad = _gradient_softmax(a, w, N_rev)

            # ---------- Adam update ----------
            iteration += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            lr = lr0 / (1.0 + 1e-4 * iteration)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---------- projection ----------
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a * (target_sum / s)
            else:
                a = np.full_like(a, target_sum / n)

            # ---------- occasional jitter ----------
            if iteration % jitter_interval == 0:
                jitter = np.random.randn(n) * 0.001 * target_sum / n
                a = np.maximum(a + jitter, 0.0)
                a = a * (target_sum / a.sum())

            # ---------- evaluate exact score ----------
            cur_score = 2.0 * n * M / (target_sum ** 2)   # exact evaluate_sequence value
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[len {n:5d} restart {restart} iter {iteration:7d}] "
                    f"new best score = {best_score:.6f}"
                )

            # ---------- early exit check ----------
            if time.time() >= deadline:
                break

    return best_seq, best_score
