      `fβ(a)= (1/β)·log Σ exp(β·c_k)` is
          ∇fβ(a) = 2·(w ⋆ a)        (cross‑correlation)
      which we evaluate with ``np.correlate``.
    * Perform a back‑tracking line‑search along the direction
      ``-∇fβ``; each trial is projected onto the probability simplex
      (non‑negative, sum = 1).  Accept the first trial that strictly
      improves the *exact* objective (computed with a fast FFT‑based
      convolution).  If none succeeds, try a tiny random mass‑transfer
      move.
    * After a successful step increase β (focuses more on the true max);
      after a failure keep β unchanged.
    * If no improvement has happened for a long stretch, restart from a
      fresh random simplex point.

3.  Return the best vector found before the deadline.  The returned list
    can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. Simplex projection
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries are zero – return uniform
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2. Fast FFT‑based self‑convolution (full length 2·n‑1)
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (real FFT version)."""
    n = a.shape[0]
    L = 2 * n - 1
    nfft = _next_pow2(L)
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:L]
    return conv


# ----------------------------------------------------------------------
# 3. Exact objective (fast, used for acceptance tests)
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv)/sum(a)² for a non‑negative vector."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _self_convolve(a)
    return float(2.0 * a.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# 4. Parametric families for initial seeds
def _geom_seq(n: int, r: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r**i, 0<r<1."""
    if not (0.0 < r < 1.0):
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    return powers / powers.sum()


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a-1}(n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# 5. Smoothed gradient (soft‑max of the convolution)
def _smooth_grad_and_max(a: np.ndarray, beta: float):
    """
    Return (max_conv, gradient) where gradient is that of the
    smooth surrogate (log‑sum‑exp) of the max‑convolution.
    """
    conv = _self_convolve(a)                     # length L = 2n‑1
    max_c = conv.max()
    # soft‑max weights – subtraction for numerical stability
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()                # shape (L,)

    # gradient = 2·(w ⋆ a)   (cross‑correlation)
    # np.correlate returns correlation with lags –(n‑1)…+(L‑1)
    corr = np.correlate(w, a, mode='full')
    n = a.size
    # lag = 0 corresponds to index n‑1
    grad = 2.0 * corr[n - 1: n - 1 + n]
    return max_c, grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     length: int = 2000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will stop a little before the limit).
    length : int
        Length of the sequence to optimise (default 2000).

    Returns
    -------
    list[float]
        The best non‑negative sequence found (already normalised to sum = 1).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # tiny safety margin

    n = length
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional global seed
    if "height_sequence_1" in globals():
        try:
            gl = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            gl = np.clip(gl, 0.0, 1000.0)
            if gl.sum() > 0.0:
                gl = gl / gl.sum()
                sc = _objective(gl)
                if sc < best_score:
                    best_score = sc
                    best_seq = gl.copy()
                    print(f"[Init] global seed  score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. systematic sweep of simple families
    #    (stops early if time is exhausted)
    # geometric (parameter r)
    for r in np.linspace(0.55, 0.995, 20):
        if time.time() > deadline:
            break
        seq = _geom_seq(n, r)
        if seq is None:
            continue
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] geometric r={r:.4f}  score={sc:.6f}")

    # power‑law
    for p in np.linspace(0.2, 2.5, 20):
        if time.time() > deadline:
            break
        seq = _powerlaw_seq(n, p)
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] power‑law p={p:.3f}  score={sc:.6f}")

    # beta‑like (two‑parameter grid)
    for a in np.linspace(0.5, 3.0, 6):
        for b in np.linspace(0.5, 3.0, 6):
            if time.time() > deadline:
                break
            seq = _beta_seq(n, a, b)
            sc = _objective(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[Init] beta a={a:.2f} b={b:.2f}  score={sc:.6f}")

    # pure random Dirichlet points
    for _ in range(40):
        if time.time() > deadline:
            break
        seq = _random_simplex(n)
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] random  score={sc:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:                       # safety net
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2. projected smoothed gradient descent with restarts
    cur_seq = best_seq.copy()
    cur_score = best_score
    beta = 0.5                               # start with gentle smoothing
    base_step = 1.0                          # initial line‑search step
    min_step = 1e-8
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # compute smooth gradient
        _, grad = _smooth_grad_and_max(cur_seq, beta)

        # line‑search along -grad
        step = base_step
        improved = False
        while step > min_step:
            cand = _project_onto_simplex(cur_seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score - 1e-12:       # strict improvement
                improved = True
                break
            step *= 0.5

        if improved:
            cur_seq = cand
            cur_score = cand_score
            base_step = max(step * 1.2, min_step)   # a bit larger next time
            beta = min(beta * 1.002, 50.0)           # slowly focus more
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[Iter {it:06d}] new best = {best_score:.6f}")
        else:
            # ----------------------------------------------------------
            # try a tiny random mass transfer (still respect simplex)
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(cur_seq[i], 0.1)       # move ≤1 % of total mass
            if delta > 0.0:
                cand = cur_seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score - 1e-12:
                    cur_seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        print(f"[Iter {it:06d}] rand‑move improved to {best_score:.6f}")
                    # keep step size unchanged
                    no_improve = 0
                    continue

            # ----------------------------------------------------------
            # no progress – increase counter, maybe restart
            no_improve += 1
            if no_improve >= 2000:
                # fresh random start (keeps the global best untouched)
                cur_seq = _random_simplex(n)
                cur_score = _objective(cur_seq)
                base_step = 1.0
                beta = 0.5
                no_improve = 0
                print(f"[Iter {it:06d}] restart after stagnation")

        it += 1

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length={n})")
    return best_seq.tolist()
