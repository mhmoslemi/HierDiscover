def _adam_optimize(a_init, target_sum, best_seq, best_score, deadline, max_iters=200000):
    a = a_init.copy()
    # Ensure sum normalization
    s = a.sum()
    if s > 0: a = a * (target_sum / s)
    else: a = np.full_like(a, target_sum / len(a))

    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    decay = 1e-4
    jitter_interval = 5000
    iteration = 0
    while iteration < max_iters and time.time() < deadline:
        # convolution
        conv = _conv_full(a) # returns array
        M = float(conv.max())
        # active set
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        rev = _subgradient(a, active)

        # gradient
        coeff = 2.0  # Since target_sum = sqrt(2n) => coeff = 2
        const = M * math.sqrt(2.0 / n)  # sqrt(2/n) * M
        grad = coeff * rev - const

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + decay * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # evaluate
        cur_score = M  # because normalized sum -> objective equals max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}")

    return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Several candidate lengths are examined – the original length and a few
  larger ones (up to 8000).  Longer vectors give more freedom to flatten
  the self‑convolution, which can lower the objective.
* For each length several random restarts are performed.  The initial
  vectors use different shapes (uniform, linear, arcsine‑like,
  exponential) to avoid being trapped in the same local minimum.
* Each restart runs a projected Adam optimiser that minimises the
  maximal entry of the convolution ``a*a`` (with the ℓ1‑norm of ``a``
  fixed to ``sqrt(2·n)``).  The gradient is the exact sub‑gradient of the
  max‑convolution:
      grad = 2·rev – M·sqrt(2/n)
  where ``rev_i = average_k a_{k‑i}`` over the active shifts that attain
  the current maximum ``M``.
* After every Adam step the vector is projected onto the feasible set:
  non‑negative entries and the required ℓ1‑norm.
* Small Gaussian jitter is added every few thousand iterations to help
  escape shallow plateaus.
* Throughout the run the best sequence (lowest maximal convolution
  value) is kept.  The function returns this sequence as a plain Python
  list before the 1000 s time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT (real-valued)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    B = A * A                                 # element‑wise product = FFT(conv)
    conv = np.fft.irfft(B, n=N)
    return conv[: 2 * n - 1]                # keep the linear part only


def _max_convolution(a: np.ndarray):
    """Return maximal convolution value and the full convolution array."""
    conv = _conv_full(a)
    M = float(conv.max())
    return M, conv


# ----------------------------------------------------------------------
# Sub‑gradient (average reversed vector over the active set) ----------
def _subgradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Average of the reversed vectors a_{k‑i} for all active shift indices k."""
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= float(active.size)
    return rev

# Adam optimiser for a single length ------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:

    Run Adam on a fixed length vector `a_init`.  The objective after
    normalisation equals the maximal convolution value `M`.  The function
    updates `best_seq`/`best_score` whenever a better vector is found.
    """
    a = a_init.copy()
    # Normalise to the prescribed ℓ1‑norm
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])

    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    decay = 1e-4
    jitter_interval = 5000

        it += 1

        M, conv = _max_convolution(a)
        tol = 1e-12

        # 2. sub‑gradient of the maximal entry
        rev = _subgradient(a, active)             # shape (n,)
        coeff = 2.0                                 # = 4n / (sqrt(2n))**2
        const = M * math.sqrt(2.0 / n)            # = 4nM / (sqrt(2n))**3
        grad = coeff * rev - const                 # sub‑gradient

        # 3. Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + decay * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4. projection onto the feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # 5. occasional jitter to shake off plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # 6. keep the best vector seen so far (objective == M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:7d}] new best score = {best_score:.6f}")

    return best_seq, best_score

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function respects the wall‑clock limit (≈1000 s) and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # small safety margin

    # ------------------------------------------------------------------
    # Base sequence (if previously discovered) – otherwise fall back.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    target0 = math.sqrt(2.0 * n0)
    base = base * (target0 / base.sum())

    # Global best
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length = {n0}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Candidate lengths – original plus a few larger ones (max 8000)
    factors = [1.0, 0.75, 1.25, 1.5, 2.0]      # allow a slightly smaller one as well
    lengths = sorted({int(n0 * f) for f in factors if int(n0 * f) >= 300})
    lengths = [min(l, 8000) for l in lengths]

    # Different initial shapes for the random restarts
    shape_names = ["uniform", "linear", "arcsine", "exp"]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        print(f"[run] starting length {n}")

        # Number of restarts depends on the size (more for small vectors)
        restarts = 5 if n <= 4000 else 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            shape = shape_names[r % len(shape_names)]

            # ----------- build initial vector ---------------------------------
            if n == n0 and shape == "uniform":
                a0 = base.copy()
            else:
                if shape == "uniform":
                    a0 = np.ones(n, dtype=float)
                elif shape == "linear":
                    a0 = np.linspace(1.0, 0.0, n, dtype=float)   # descending line
                elif shape == "arcsine":
                    i = np.arange(n, dtype=float)
                    # Discrete analogue of the continuous arcsine density
                    a0 = 1.0 / np.sqrt((i + 1.0) * (n - i))
                elif shape == "exp":
                    a0 = np.random.exponential(scale=1.0, size=n)
                else:
                    a0 = np.random.rand(n)

                a0 = np.maximum(a0, 0.0)
                a0 = a0 * (target_sum / a0.sum())

            # -------------------- optimise ------------------------------------
            best_seq, best_score = _adam_optimize(
                a_init=a0,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iters=300_000 if n <= 4000 else 150_000,
            )

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
