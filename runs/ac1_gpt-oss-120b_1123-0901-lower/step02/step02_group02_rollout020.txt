def _optimise_one_length_adam(n, init, run_deadline, *, target_sum=None, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, alpha=5.0):
    if target_sum None: target_sum = np.sqrt(2*n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_local = a.copy()
    best_score = 2.0 * n * np.max(_conv_fft(a)) / (a.sum()**2)
    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        score = 2.0 * n * max_val / (a.sum()**2)    # same as max_val after scaling
        if score < best_score:
            best_score = score; best_local = a.copy()
            print(f"[{time.time():.2f}] n={n} local best {score:.6f}")
        # compute softmax weights
        # shift by max to avoid overflow
        shifted = conv - max_val
        exp_alpha = np.exp(alpha * shifted)
        w = exp_alpha / np.sum(exp_alpha)
        # gradient
        grad = 2.0 * np.correlate(w, a, mode='full')[:n][::-1]  # shape (n,)
        # optional smoothing regularization: lambda * (2*a - shift)
        # we can add lam=1e-4
        lam = 1e-5
        grad += lam * (2 * a - np.concatenate(([0], a[:-1])) - np.concatenate((a[1:], [0])))
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Project onto simplex
        a = _project_onto_simplex(a_new, target_sum)
        # Possibly decay lr: lr *= 0.9995
        lr *= 0.9995
    return best_local, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------
1.  Choose a length `n` (larger n → finer approximation of the
    continuous optimum).  Lengths are sampled throughout the run,
    favouring larger values early on and smaller values when the
    remaining time is low.

2.  Initialise the vector with a simple power‑law shape
    a_i ∝ (i+1)^{-p} (p∈[0.4,0.8]) – the classical “1/√x’’ start.
    Occasionally warm‑start from the current global best and add a tiny
    jitter.

3.  Optimise the shape with an Adam‑style projected gradient descent.
    The non‑smooth objective *max(conv)* is replaced by the smooth
    soft‑max approximation

        F(a) = (1/α)·log Σ_k exp(α·conv_k)

    whose gradient is   ∇F = 2·corr( w , a )
    where  w_k ∝ exp(α·conv_k)  (weights sum to one) and
    corr denotes the (full) cross‑correlation.
    After each gradient step we project back onto the simplex
    { x ≥ 0 , Σ x = √(2n) }.

4.  When the allocated time slice for a restart expires,
    a short stochastic “hill‑climb’’ (mass transfer between two
    coordinates) is performed for a few seconds.  The move is kept
    if it reduces the true objective (the exact max‑convolution).

5.  The best sequence seen over all restarts is returned.
    The whole routine stops before the global time‑budget (≈1000 s).

The implementation uses only NumPy (FFT based convolutions and
correlations) and runs comfortably on a single core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (s ≥ 0)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full (linear) convolution a * a via FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (real FFT, returns length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # eliminate spurious tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: smooth soft‑max weights for the convolution vector
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return w_k ∝ exp(alpha·conv_k) with Σ w = 1 (numerically stable)."""
    max_v = conv.max()
    shifted = conv - max_v                     # avoid overflow
    exp_alpha = np.exp(alpha * shifted)
    w = exp_alpha / exp_alpha.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser for one length (soft‑max surrogate of max‑conv)
# ----------------------------------------------------------------------
def _adam_optimize_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    target_sum: float,
    lr: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha: float = 5.0,
    smooth_lambda: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max approximation of max(conv).
    Returns the best vector (as a NumPy array) found within the time slice
    and its **exact** evaluation score.
    """
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    # best local bookkeeping
    conv = _conv_fft(a)
    max_val = conv.max()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)   # = max_val when summed to √(2n)
    best_vec = a.copy()

    while time.time() < run_deadline:
        t += 1

        # --------------------------------------------------------------
        # 1) current convolution and true objective
        # --------------------------------------------------------------
        conv = _conv_fft(a)
        max_val = conv.max()
        score = 2.0 * n * max_val / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")

        # --------------------------------------------------------------
        # 2) soft‑max weights (smooth max)
        # --------------------------------------------------------------
        w = _softmax_weights(conv, alpha)          # shape (2·n‑1,)

        # --------------------------------------------------------------
        # 3) gradient of the smooth objective:
        #    ∇F = 2 * corr(w, a)   (first n entries, reversed)
        # --------------------------------------------------------------
        # full cross‑correlation, then keep the first n entries reversed
        corr = np.correlate(w, a, mode="full")
        grad = 2.0 * corr[:n][::-1]

        # optional smoothing regulariser (encourage mild smoothness)
        if smooth_lambda > 0:
            left = np.concatenate(([0.0], a[:-1]))
            right = np.concatenate((a[1:], [0.0]))
            grad += smooth_lambda * (2.0 * a - left - right)

        # --------------------------------------------------------------
        # 4) Adam update, followed by projection onto the simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_onto_simplex(a_new, target_sum)

        # slowly decay learning‑rate – stabilises the final phase
        lr *= 0.9997

    return best_vec, best_score


# ----------------------------------------------------------------------
# Small stochastic hill‑climb using exact objective (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, end_time: float, target_sum: float) -> tuple[np.ndarray, float]:
    """
    Perform cheap random mass‑transfer moves for a few seconds.
    Keeps the best exact objective encountered.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_vec = a.copy()
    cur_vec = a.copy()
    cur_conv = conv
    cur_score = best_score

    # step size as a fraction of total mass per move
    step_frac = 0.001
    step = max(step_frac * target_sum / n, 1e-8)

    # simple simulated‑annealing schedule
    temp0 = 1.0
    while time.time() < end_time:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if cur_vec[i] < step:
            continue

        # propose move: transfer `step` from i → j
        proposal = cur_vec.copy()
        proposal[i] -= step
        proposal[j] += step

        # evaluate (full convolution, cheap for our n)
        prop_conv = _conv_fft(proposal)
        prop_score = 2.0 * n * prop_conv.max() / (proposal.sum() ** 2)

        # acceptance rule
        accept = False
        if prop_score < cur_score:
            accept = True
        else:
            # annealing probability
            t_elapsed = time.time() - (end_time - 10)  # rough elapsed in this phase
            temp = max(temp0 * (1 - t_elapsed / 10.0), 1e-4)
            if np.random.rand() < np.exp(-(prop_score - cur_score) / temp):
                accept = True

        if accept:
            cur_vec = proposal
            cur_conv = prop_conv
            cur_score = prop_score
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur_vec.copy()

    return best_vec, best_score

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation value.
    The routine runs for at most `budget_s` seconds and returns the
    best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    best_seq = None
    best_score = float("inf")

    # Length search bounds – larger n gives a finer approximation.
    min_n, max_n = 2000, 12000

    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a length; keep it modest when we are close to the deadline
        if remaining < 30.0:
            n = np.random.randint(800, 2000)
        else:
            n = np.random.randint(min_n, max_n + 1)

        # -----------------------------------------------------------------
        # 0) initial shape (power‑law) – exponent sampled broadly
        # -----------------------------------------------------------------
        p_exp = np.random.uniform(0.4, 0.8)
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p_exp)

        # occasional warm‑start from the current global best
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-6
                init = np.concatenate([best_seq, pad])

        # tiny random jitter to break symmetry
        init += np.random.rand(n) * 1e-6

        # -----------------------------------------------------------------
        # 1) project onto the simplex of the right total mass
        # -----------------------------------------------------------------
        target_sum = np.sqrt(2.0 * n)

        # -----------------------------------------------------------------
        # 2) allocate a time slice for the Adam optimisation
        # -----------------------------------------------------------------
        # roughly 5‑12 s per restart (adapt to the remaining budget)
        max_slice = min(12.0, remaining * 0.5)
        slice_len = np.random.uniform(5.0, max_slice)
        run_deadline = time.time() + slice_len

        seq_local, score_local = _adam_optimize_one_length(
            n,
            init,
            run_deadline,
            target_sum=target_sum,
            lr=0.09,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha=6.0,               # slightly higher gives sharper max‑approx
            smooth_lambda=1e-5,
        )

        # -----------------------------------------------------------------
        # 3) short stochastic hill‑climb to escape possible plateaus
        # -----------------------------------------------------------------
        hill_deadline = min(deadline, time.time() + min(4.0, remaining * 0.2))
        seq_local, score_local = _hill_climb(seq_local, hill_deadline, target_sum)

        # -----------------------------------------------------------------
        # 4) update global best
        # -----------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )

        # early exit if we are already comfortably below the target
        if best_score < 1.45:   # margin below the 1.503 goal
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid return.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
