def gradient_descent(w, n, deadline, base_step=0.5):
    target = math.sqrt(2.0 * n)
    w = w.copy()
    best_local_w = w.copy()
    best_local_score = float('inf')
    it = 0
    gamma = 5.0
    gamma_max = 200.0
    gamma_inc_interval = 100   # adjust gamma every 100 iterations
    while time.time() < deadline:
        conv = _conv_self(w)  # length 2n-1
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        # update local best
        if score < best_local_score:
            best_local_score = score
            best_local_w = w.copy()
        # compute gradient via softmax
        maxc = max_conv
        # stable exp
        exp_vals = np.exp(gamma * (conv - maxc))
        p = exp_vals / exp_vals.sum()
        # compute correlation via FFT
        size = 1 << ((len(p) + len(w) - 1).bit_length())
        fp = np.fft.rfft(p, n=size)
        fw = np.fft.rfft(w, n=size)
        corr = np.fft.irfft(fp.conj() * fw, n=size)
        start = len(p) - 1
        grad = 2.0 * corr[start: start + len(w)]
        # descent direction
        direction = -grad
        # adapt step size
        it += 1
        step = base_step / math.sqrt(it)
        norm = np.linalg.norm(direction, ord=2)
        if norm > 0:
            eta = step / norm
        else:
            eta = 0.0
        w_new = w + eta * direction
        # project onto simplex sum=1
        w_new = _project_onto_simplex(w_new, 1.0)
        w = w_new
        # adjust gamma
        if it % gamma_inc_interval == 0 and gamma < gamma_max:
            gamma = min(gamma_max, gamma * 1.3)
        # early stop if score already lower than target bound global
        if score < 1.503:
            break
        # also break if step size becomes tiny?
    return best_local_w, best_local_score

def gradient_descent(start_w, n, deadline, best_state):
    """
    Performs projected gradient descent on probability vector w (sum=1, w>=0)
    to minimize 2*n*max(conv(w,w)). Updates ``best_state`` in-place.
    Returns the best w encountered.
    """
    w = start_w.astype(float)
    it = 0
    gamma = 5.0
    gamma_max = 200.0
    base_step = 0.5
    while time.time() < deadline:
        conv = _conv_self(w)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        # update best
        if score < best_state['score']:
            best_state['score'] = score
            best_state['seq'] = (w * math.sqrt(2.0 * n)).tolist()
            print(f"[{time.time():.2f}] New best (grad) = {score:.6f} (len={n})")
        # compute softmax gradient
        maxc = max_conv
        exp_vals = np.exp(gamma * (conv - maxc))
        p = exp_vals / exp_vals.sum()
        size = 1 << ((len(p) + len(w) - 1).bit_length())
        fp = np.fft.rfft(p, n=size)
        fw = np.fft.rfft(w, n=size)

        it += 1

            eta = 0.0

            break

"""
Optimiser for the step‑function evaluation problem.

The goal is to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

under the constraints a_i ≥ 0.  Because the objective is invariant
under scaling of a we work with a normalised probability vector
w with sum(w)=1 and minimise 2·n·max(conv(w,w)).  This allows us
to use fast FFT based convolutions and to stay in the simplex
by simple Euclidean projection.

Algorithm
---------

1.  Initialise the global best from the supplied *height_sequence_1*
    (if present) and from a cheap sweep over several parametric
    families (exponential, power‑law, linear, quadratic, triangular).

2.  Run many independent gradient‑descent runs on the simplex.
    The descent uses a smooth approximation of the “max” by a
    soft‑max with an increasing temperature (γ).  The gradient of the
    soft‑max is computed via FFT‑based correlation, the step is
    normalised and the iterate is projected back onto the simplex.

3.  Refine the current best with a classic sub‑gradient descent that
    directly tackles the true max‑convolution (momentum optimiser).

4.  If SciPy is available, perform a few LP‑refinements exactly as in
    the published SOTA method.

The routine respects the time budget (default 1000 s) and returns the
best sequence found as a plain ``list[float]`` ready for the
evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (top‑level – no closures)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution via real FFT (used both for evaluation and gradients)
# ----------------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution x*x (real, using FFT)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Safe wrapper around the external evaluator (top‑level)
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Return the score from evaluate_sequence or +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Shape factories – generate raw (non‑normalised) vectors
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


def _shape_power_law_desc(n: int, p: float) -> np.ndarray:
    """Decreasing power‑law: (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-p)


def _shape_mix_two_exp(n: int, a1: float, a2: float, w: float) -> np.ndarray:
    """Mixture a*exp(-a1*i) + (1-a)*exp(-a2*i).  w∈[0,1]."""
    i = np.arange(n, dtype=float)
    return w * np.exp(-a1 * i) + (1.0 - w) * np.exp(-a2 * i)


# ----------------------------------------------------------------------
# Gradient descent on the simplex (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _gradient_descent_on_simplex(start_w: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 best_state: dict):
    """
    Projected gradient descent on the probability vector w (Σw=1, w≥0)
    to minimise 2·n·max(conv(w,w)).  Uses a soft‑max with an increasing
    temperature γ to obtain a smooth gradient.
    Updates ``best_state`` in‑place.
    Returns the final w (not used further – best_state holds the best).
    """
    w = start_w.astype(float).copy()
    it = 0
    gamma = 5.0
    gamma_max = 200.0
    base_step = 0.5

    while time.time() < deadline:
        conv = _conv_self(w)                 # length 2n‑1
        max_conv = conv.max()
        score = 2.0 * n * max_conv           # exact objective value

        # ----- global best update ------------------------------------------------
        if score < best_state['score']:
            best_state['score'] = score
            best_state['seq'] = (w * math.sqrt(2.0 * n)).tolist()
            print(f"[{time.time():.2f}] New best (grad) = {score:.6f} (len={n})")

        # ----- soft‑max weights --------------------------------------------------
        maxc = max_conv
        # shift by max to avoid overflow/underflow
        exp_vals = np.exp(gamma * (conv - maxc))
        p = exp_vals / exp_vals.sum()       # p_i ≥0, Σp=1

        # ----- gradient of the smoothed objective (via FFT correlation) -------
        size = 1 << ((len(p) + len(w) - 1).bit_length())
        fp = np.fft.rfft(p, n=size)
        fw = np.fft.rfft(w, n=size)
        corr_full = np.fft.irfft(fp.conj() * fw, n=size)
        start = len(p) - 1                   # aligns with correlation definition
        grad = 2.0 * corr_full[start:start + n]

        # ----- descent step -------------------------------------------------------
        direction = -grad                    # we minimise
        it += 1
        step = base_step / math.sqrt(it)
        norm = np.linalg.norm(direction, ord=2)
        eta = step / norm if norm > 0 else 0.0
        w_new = w + eta * direction
        w_new = _project_onto_simplex(w_new, 1.0)
        w = w_new

        # ----- temperature schedule ------------------------------------------------
        if it % 100 == 0 and gamma < gamma_max:
            gamma = min(gamma_max, gamma * 1.2)

        # ----- trivial early stop -------------------------------------------------
        if score < 1.503:          # reached the target bound
            break

    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution, with momentum)
# ----------------------------------------------------------------------
def _subgradient_descent_on_simplex(start_w: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    best_state: dict):
    """
    Projected sub‑gradient descent directly on the max‑convolution.
    Uses a simple momentum scheme.
    """
    w = start_w.astype(float).copy()
    m = np.zeros_like(w)
    beta = 0.9
    base_step = 0.2
    it = 0

    while time.time() < deadline:
        conv = _conv_self(w)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_state['score']:
            best_state['score'] = score
            best_state['seq'] = (w * math.sqrt(2.0 * n)).tolist()
            print(f"[{time.time():.2f}] New best (sub) = {score:.6f} (len={n})")

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        # gradient of conv[kmax] w.r.t. w (see derivation in analysis)
        low = max(0, kmax - (n - 1))
        high = min(n - 1, kmax)
        idx = np.arange(low, high + 1, dtype=int)
        partner = kmax - idx
        grad = np.zeros_like(w)
        grad[idx] = 2.0 * w[partner]

        direction = -grad
        m = beta * m + (1.0 - beta) * direction
        it += 1
        step = base_step / math.sqrt(it)
        max_abs = np.max(np.abs(m))
        eta = step / max_abs if max_abs > 0 else 0.0
        w_new = w + eta * m
        w_new = _project_onto_simplex(w_new, 1.0)
        w = w_new

        if score < 1.503:
            break

    return w


# ----------------------------------------------------------------------
# Optional LP refinement (same as in the reference implementation)
# ----------------------------------------------------------------------
def _lp_refine(seq_w: np.ndarray,
               n: int,
               deadline: float,
               best_state: dict):
    """
    Solve the linear programme max Σ g_j  s.t.  (f * g)_k ≤ M,
    where f is the current candidate (scaled to the required sum).
    Perform a few iterations of convex combination.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return  # SciPy not available

    a = seq_w * math.sqrt(2.0 * n)           # back‑to the a‑space
    target = math.sqrt(2.0 * n)

    while time.time() < deadline:
        conv = np.convolve(a, a)
        M = conv.max()
        rows = 2 * n - 1

        # Constraint matrix A (size rows × n)
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(c=-np.ones(n),               # maximise Σ g_j
                      A_ub=A,
                      b_ub=np.full(rows, M),
                      bounds=(0, None),
                      method="highs",
                      options={"presolve": True})
        if not res.success:
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g * (target / g.sum())               # renormalise to the simplex

        # convex blend
        t = 0.2
        a = (1.0 - t) * a + t * g

        # update best state
        score = _evaluate(a.tolist())
        if score < best_state['score']:
            best_state['score'] = score
            best_state['seq'] = a.tolist()
            print(f"[{time.time():.2f}] New best (LP) = {score:.6f}")

        if best_state['score'] < 1.503 or time.time() + 0.5 > deadline:
            break

def _fast_score_from_w(w: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(w,w)) for a normalised probability vector w.

    n = w.shape[0]
    conv = _conv_self(w)

        n_best = len(_GLOBAL_BEST["seq"])
        _lp_refine(w_best, n_best, deadline, _GLOBAL_BEST)

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if everything else failed
    # ------------------------------------------------------------------
    if _GLOBAL_BEST["seq"] is None:
        n = 1000
        a = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)
        _GLOBAL_BEST["seq"] = a.tolist()
        _GLOBAL_BEST["score"] = _evaluate(_GLOBAL_BEST["seq"])

    # ------------------------------------------------------------------
    # Return the best sequence found (already scaled to the required sum)
    # ------------------------------------------------------------------
    return _GLOBAL_BEST["seq"]
