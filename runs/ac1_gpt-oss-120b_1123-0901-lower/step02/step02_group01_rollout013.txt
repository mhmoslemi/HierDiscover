The score to minimise is  

\[
F(a)=\frac{2\,n\,\max_k (a*a)[k]}{(\sum_i a_i)^2},
\]

which is *scale‑invariant*.  We therefore keep the vector on the probability simplex  
\(\{a\ge 0,\; \sum a = 1\}\) and minimise  

\[
G(a)=2n\,\max_k (a*a)[k].
\]

`max` is non‑smooth, so we replace it by the smooth “log‑sum‑exp’’ surrogate  

\[
\phi_\lambda(a)=\frac{1}{\lambda}\log\!\Bigl(\sum_k e^{\lambda\,c_k}\Bigr),\qquad
c = a*a .
\]

For a fixed \(\lambda\) the gradient of \(\phi_\lambda\) is  
\(\nabla\phi_\lambda(a)=2\,(w\star a)\) where  

\[
w_k=\frac{e^{\lambda c_k}}{\sum_j e^{\lambda c_j}}\;,
\]

and \(\star\) denotes the **linear** cross‑correlation.  
All convolutions are performed with an FFT ( \(O(n\log n)\) ).

The optimisation proceeds in three stages  

1. **Coarse optimisation** on a short vector ( \(n=500\) ) – cheap, many
   iterations, gives a good global shape.  
2. **Upsampling** the coarse optimum to a finer resolution
   ( \(n=2000\) ) by linear interpolation – preserves the shape while
   adding degrees of freedom.  
3. **Fine optimisation** on the finer grid (and, if time permits,
   a still finer grid \(n=4000\)).  

Gradient steps are taken with the **Adam** optimiser (first‑ and
second‑moment adaptation) and after each step the vector is projected
onto the simplex.  A continuation schedule slowly increases
\(\lambda\) (capped at 200) so that the surrogate starts very smooth
and ends close to the true \(\max\).  When the improvement stalls for a
few seconds a tiny random “shake’’ is added to escape shallow plateaus.
All work respects the 1000 s wall‑time limit.

"""
Gradient‑based multi‑scale optimiser for the step‑function problem.

* Uses a log‑sum‑exp surrogate for the max‑convolution term.
* Adam updates + projection onto the probability simplex.
* Continuation on the smoothing parameter λ (capped to avoid overflow).
* Coarse‑to‑fine strategy: optimise n=500 → upsample → optimise n=2000
  (and optionally n=4000 if time permits).
* Occasional random “shakes’’ to escape plateaus.

The function `propose_candidate` complies with the required signature
and returns the best sequence found within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Projection onto the probability simplex {x ≥ 0, sum(x) = 1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:          # all numbers non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# 2.  Linear (zero‑padded) self‑convolution via FFT
def _full_convolution(a, size=None):
    """Return a*a (full linear convolution, length 2·len(a)-1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# 3.  Exact objective used by `evaluate_sequence` (sum(a)=1 ⇒ factor 2n)
def _objective(a):
    """Exact evaluation 2·n·max(conv) (a is assumed non‑negative, sum=1)."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    return 2.0 * n * max_b

# ----------------------------------------------------------------------
# 4.  Gradient of the smooth‑max surrogate (log‑sum‑exp)
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Gradient of (1/lam)·log∑exp(lam·conv_k) w.r.t. a.
    Returned vector has length len(a).
    """
    n = len(a)
    # soft‑max weights (already numerically stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:               # safety – should not happen with capped λ
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # linear cross‑correlation w ★ a.  (rev_a is a reversed)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice which corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

# ----------------------------------------------------------------------
# 5.  Optimisation sub‑routine for a fixed length
def _optimize_fixed_n(
    n,
    init_seq=None,
    max_iters=20000,
    deadline=None,
    best_seq_holder=None,
    best_val_holder=None,

    Returns the final sequence (normalised to sum = 1).
    """

    else:

        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            seq = np.ones_like(seq) / n
        else:
            seq /= seq.sum()

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.15          # works well for the surrogate
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    # continuation on λ
    lam = 5.0
    max_lambda = 200.0
    lam_factor = 1.02

    # FFT size (next power of two for the convolution)
    size_fft = 1 << int(np.ceil(np.log2(2 * n - 1)))

    last_improve = time.time()
    it = 0
    while it < max_iters:
        if deadline is not None and time.time() >= deadline:
            break

        # 1) convolution of the current vector
        conv = _full_convolution(seq, size=size_fft)

        # 2) gradient of the smooth‑max surrogate
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_fft)

        # 3) Adam update (gradient step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # 4) evaluate the *exact* objective (max‑convolution)
        val = _objective(new_seq)

        # 5) keep global best
        if best_val_holder is not None and val < best_val_holder[0]:
            best_val_holder[0] = val
            best_seq_holder[0] = new_seq.copy()
            last_improve = time.time()
            print(f"[n={n:5d} it={it:6d}] new best obj={val:.6f}")

        # 6) occasional shake when stuck
        if time.time() - last_improve > 2.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            new_seq = _project_onto_simplex(new_seq + noise)
            last_improve = time.time()

        seq = new_seq
        lam = min(lam * lam_factor, max_lambda)
        it += 1

    return seq

# 6.  Main entry point required by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a small evaluation score.
    Uses a multi‑scale Adam optimiser with a log‑sum‑exp surrogate.
    Returns the best list found before the time budget expires.
    """
    np.random.seed(seed)
    # give a tiny safety margin for the interpreter
    deadline = time.time() + budget_s - 0.5

    # holders for the globally best sequence / value (shared across stages)
    best_seq_holder = [None]   # list so it can be mutated inside helpers
    best_val_holder = [float("inf")]

    # ------------------------------------------------------------------
    # Stage 1 – coarse optimisation (n = 500)
    n_coarse = 500
    print(f"--- Stage 1: coarse optimisation (n={n_coarse}) ---")
    _optimize_fixed_n(
        n=n_coarse,
        init_seq=None,
        max_iters=25000,
        deadline=deadline,
        best_seq_holder=best_seq_holder,
        best_val_holder=best_val_holder,
    )

    # ------------------------------------------------------------------
    # Stage 2 – upsample to a finer grid (n = 2000)
    n_fine = 2000
    print(f"--- Stage 2: up‑sample to n={n_fine} ---")
    # linear interpolation of the current best coarse vector
    coarse = best_seq_holder[0]
    if coarse is None:
        # fallback – random Dirichlet if something went wrong
        coarse = np.random.dirichlet(np.ones(n_coarse))
    x_coarse = np.linspace(0.0, 1.0, n_coarse)
    x_fine = np.linspace(0.0, 1.0, n_fine)
    fine = np.interp(x_fine, x_coarse, coarse)
    fine = np.maximum(fine, 0.0)
    fine_sum = fine.sum()
    if fine_sum == 0.0:
        fine = np.ones_like(fine) / n_fine
    else:
        fine /= fine_sum

    _optimize_fixed_n(
        n=n_fine,
        init_seq=fine,
        max_iters=50000,
        deadline=deadline,
        best_seq_holder=best_seq_holder,
        best_val_holder=best_val_holder,
    )

    # ------------------------------------------------------------------
    # Stage 3 – optional extra resolution (n = 4000) if time permits
    if time.time() < deadline - 5.0:
        n_big = 4000
        print(f"--- Stage 3: extra resolution n={n_big} ---")
        fine = best_seq_holder[0]
        x_fine = np.linspace(0.0, 1.0, n_fine)
        x_big = np.linspace(0.0, 1.0, n_big)
        big = np.interp(x_big, x_fine, fine)
        big = np.maximum(big, 0.0)
        big_sum = big.sum()
        if big_sum == 0.0:
            big = np.ones_like(big) / n_big
        else:
            big /= big_sum

        _optimize_fixed_n(
            n=n_big,
            init_seq=big,
            max_iters=30000,
            deadline=deadline,
            best_seq_holder=best_seq_holder,
            best_val_holder=best_val_holder,
        )

    # ------------------------------------------------------------------
    # Final report
    best_seq = best_seq_holder[0]
    best_val = best_val_holder[0]
    if best_seq is None:
        # as a safety net return a uniform distribution
        best_seq = np.ones(2000) / 2000.0
        best_val = _objective(best_seq)

    print(f"=== Search finished === best objective = {best_val:.6f} "
          f"length = {len(best_seq)}")
    # return a plain Python list – exactly what `evaluate_sequence` expects
    return best_seq.tolist()
