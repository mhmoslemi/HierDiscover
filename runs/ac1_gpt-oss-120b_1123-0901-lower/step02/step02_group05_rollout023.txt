def optimize_sequence(a, target_sum, time_allowed):
    size = a.size
    a = project_onto_simplex(a, target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_start = time.time()
    it = 0
    best = a.copy()
    best_score = eval(a)
    while time.time() < t_start + time_allowed:
        # compute max conv and gradient
        max_val, grad, _, _ = _max_conv_and_grad(a)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it+1))
        v_hat = v / (1 - beta2 ** (it+1))
        # step size adapt: maybe base_lr = 0.1 * max_val / np.linalg.norm(grad) but we can use scaling...
        step = 0.01 * max_val  # scaling? maybe bigger
        # compute direction to keep sum zero:
        direction = -grad + np.mean(grad)
        # update a:
        a_new = a + step * direction / (np.sqrt(v_hat) + eps)
        a_new = _project_onto_simplex(a_new, target_sum)
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        # accept if improvement else maybe accept with small prob
        if score < best_score:
            best_score = score
            best = a_new.copy()
            print(f"time {time.time():.2f}: new best {best_score}")
        a = a_new
        it += 1
    return best, best_score

def _adam_descent(init, target_sum, time_budget, base_lr=0.1):
    a = _project_onto_simplex(init, target_sum)
    # momentum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    t_start = time.time()
    it = 0
    while time.time() - t_start < time_budget:
        max_val, grad, _, _ = _max_conv_and_grad(a)
        # compute direction that preserves sum
        # gradient of max conv w.r.t a is grad. We want to move opposite direction.
        direction = -grad + np.mean(grad)  # zero sum
        # Adam update
        m = beta1 * m + (1 - beta1) * direction
        v = beta2 * v + (1 - beta2) * (direction ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        # adaptive step scaling based on current max_val
        lr = base_lr * max_val / (np.max(np.abs(direction)) + 1e-12)
        a_new = a + lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a_new = _project_onto_simplex(a_new, target_sum)
        # evaluate
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best = a_new.copy()
            print(f"[{time.time():.2f}] adam n={len(a)} new best={score:.6f}")
        # acceptance: simple greedy
        a = a_new
        it += 1
        # occasional random reset to avoid stall
        if it % 500 == 0:
            if np.random.rand() < 0.1:
                # add small noise
                a = _project_onto_simplex(a + np.random.randn(a.size) * 0.01 * target_sum / len(a), target_sum)
    return best, best_score

"""
Propose a non‑negative step function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The code is a simple global‑local optimiser:

* Global phase – scan a few classic families (uniform, linear, exponential,
  power‑law) for several lengths, keep the best candidate.
* Local phase – for the current best length repeatedly run an Adam‑style
  projected optimisation (sub‑gradient of the maximal convolution entry)
  with occasional random perturbations.  The optimisation respects the
  simplex constraint Σ a = √(2n) via Euclidean projection.
* Occasionally a tiny LP refinement (if cvxpy is present) is tried on the
  current point.
* The search stops after the allotted time (default 1000 s) and returns the
  best sequence found.

All helper functions are defined at module level – no closures – and the
routine works with any `evaluate_sequence` that follows the specification
in the statement.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution (FFT) – returns max value, its index and the whole conv
# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (max_val, max_idx, full_conv) using FFT (real input)."""
    n = a.size
    # length of linear convolution = 2*n-1
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]
    return max_val, max_idx, conv


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index, its gradient and the full convolution.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    max_val, max_idx, conv = _conv_max_fft(a)

    # The gradient of the entry (max_idx) w.r.t each a_i:
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Simple mutation (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Gaussian perturbation followed by projection onto the simplex."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Adam‑style projected optimisation of the max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(
    init: np.ndarray,
    target_sum: float,
    time_budget: float,
    base_lr: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on the sub‑gradient of the maximal entry.
    Returns (best_sequence, best_score) seen within ``time_budget`` seconds.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial evaluation
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.time()
    it = 0
    while time.time() - t0 < time_budget:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # direction respecting the simplex sum constraint (zero‑mean)
        d = -grad + np.mean(grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * d
        v = beta2 * v + (1.0 - beta2) * (d ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size scaled by the current maximal convolution value
        scale = base_lr * max_val / (np.max(np.abs(d)) + 1e-12)
        a_new = a + scale * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] Adam n={len(a)} new best = {score:.6f}")

        # accept move (greedy)
        a = a_new
        it += 1

        # occasional random shake‑up to escape plateaus
        if it % 400 == 0 and np.random.rand() < 0.07:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a, target_sum, sigma)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (same idea as the paper sketch)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j s.t. g≥0 and f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[idx], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_budget: float,
    mix: float = 0.25,
    max_iter: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP updates starting from ``init``.
    Returns (best_seq, best_score).
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_budget:
            break
        M = np.convolve(a, a).max()
        g0 = _solve_lp_for_g(a, float(M))
        if g0 is None or np.sum(g0) == 0.0:
            break
        g = g0 * (target_sum / np.sum(g0))
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Resizing helper (linear interpolation / down‑sampling)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to length ``new_n`` using linear interpolation."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Families of parametric initial sequences
# ----------------------------------------------------------------------
def _init_uniform(n: int) -> np.ndarray:
    s = math.sqrt(2.0 * n)
    return np.full(n, s / n, dtype=float)


def _init_linear_desc(n: int) -> np.ndarray:
    coeff = np.arange(n, 0, -1, dtype=float)  # n, n‑1, … ,1
    target = math.sqrt(2.0 * n)
    return coeff * (target / np.sum(coeff))


def _init_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i   (0<r<1)."""
    idx = np.arange(n, dtype=float)
    unnorm = r ** idx
    target = math.sqrt(2.0 * n)
    return unnorm * (target / unnorm.sum())


def _init_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)**p   (p ≤ 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    unnorm = idx ** p
    target = math.sqrt(2.0 * n)
    return unnorm * (target / unnorm.sum())

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the time budget and prints occasional
    progress messages.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.4  # safety margin

    # ------------------------------------------------------------------
    # Global best holder (may be pre‑populated by the environment)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Start from historic sequence, score = {sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # ------------------------------------------------------------------
    # 1) Scan a few parametric families for a few lengths
    # ------------------------------------------------------------------
    lengths_to_try = [400, 800, 1200, 1600, 2000]
    exp_rs = [0.99, 0.995, 0.998, 0.999, 0.9995]
    power_ps = [-0.3, -0.5, -0.8, -1.0, -1.3]

    for n in lengths_to_try:
        target = math.sqrt(2.0 * n)

        # uniform
        cand = _init_uniform(n)
        try:
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.2f}] Uniform n={n} new global best {sc:.6f}")
        except Exception:
            pass

        # linear decreasing
        cand = _init_linear_desc(n)
        try:
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.2f}] Linear n={n} new global best {sc:.6f}")
        except Exception:
            pass

        # exponential family
        for r in exp_rs:
            cand = _init_exponential(n, r)
            try:
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.2f}] Exp(r={r:.5f}) n={n} new best {sc:.6f}")
            except Exception:
                pass

        # power‑law family
        for p in power_ps:
            cand = _init_powerlaw(n, p)
            try:
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[{time.time():.2f}] Power(p={p:.2f}) n={n} new best {sc:.6f}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 2) Main optimisation loop – Adam + occasional LP
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # --------------------------------------------------------------
        # pick a length – we stay close to the best now but allow drift
        # --------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.7:
            n = best_seq.size
            if np.random.rand() < 0.25:  # stretch / shrink a bit
                n = int(n * np.random.uniform(0.85, 1.15))
                n = max(250, min(3000, n))
        else:
            n = np.random.randint(300, 2500)

        target_sum = math.sqrt(2.0 * n)

        # initialise candidate (resize the best known or random)
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        # allocate a slice of remaining time for this run
        time_left = deadline - time.time()
        if time_left < 8.0:
            break

        # ------------------------------------------------------------------
        # a) Adam optimisation (most of the budget for this run)
        # ------------------------------------------------------------------
        adam_time = min(0.65 * time_left, 22.0)
        cand_seq, cand_score = _adam_optimize(init, target_sum, adam_time)

        # ------------------------------------------------------------------
        # b) optional cheap LP refinement (only for modest n, cheap)
        # ------------------------------------------------------------------
        if _CVX_AVAILABLE and n <= 800 and time.time() < deadline - 2.0:
            lp_time = min(0.25 * (time_left - adam_time), 6.0)
            cand_seq, cand_score = _lp_refine(cand_seq, target_sum, lp_time)

        # ------------------------------------------------------------------
        # c) Global update
        # ------------------------------------------------------------------
        if cand_seq is not None and cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})")
            # early exit if we already beat the target
            if best_score < 1.503:
                break

        # occasional random restart to diversify the search space
        if np.random.rand() < 0.07:
            # simple random vector, projected onto the simplex of the chosen size
            rand_vec = np.random.rand(n)
            rand_vec = _project_onto_simplex(rand_vec, target_sum)
            best_seq = rand_vec.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
            print(f"[{time.time():.2f}] Random restart, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = _init_uniform(n)

    return best_seq.tolist()
