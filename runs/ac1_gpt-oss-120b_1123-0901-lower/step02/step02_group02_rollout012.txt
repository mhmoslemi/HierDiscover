def propose_candidate(seed=42, budget_s=1000.0):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    while time.time() < deadline:
        # select length
        n = np.random.randint(800, 3500)
        target_sum = np.sqrt(2.0 * n)

        # random parameters for initial shape
        p = np.random.uniform(0.35, 0.75)
        delta = np.random.uniform(0, 8)  # shift
        lam = np.random.uniform(0.0, 0.02) # expo rate
        # generate shape
        i = np.arange(n, dtype=float)
        a = ((i + 1 + delta) ** (-p)) * np.exp(-lam * i)
        # apply small jitter
        a += np.random.rand(n) * 1e-5
        # project onto simplex
        a = _project_onto_simplex(a, target_sum)

        # compute score and maybe store
        conv = _conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (target_sum ** 2)  # sum = target_sum

        # do gradient descent for t steps or time slice
        run_deadline = min(deadline, time.time() + 7.0)  # 7 sec per run
        step0 = 0.7
        iter_idx = 0
        while time.time() < run_deadline:
            # compute max conv and active indices
            conv = _conv_fft(a)
            max_val = conv.max()
            # use relative tolerance for active to handle multiple ties
            tol = max(1e-12, max_val * 1e-6)
            active = np.where(conv >= max_val - tol)[0]
            # compute gradient sum
            grad = np.zeros_like(a)
            for k in active:
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                idx = np.arange(low, high+1)
                partner = k - idx
                grad[idx] += 2.0 * a[partner]
            grad /= len(active)  # average

            # step size decaying
            eta = step0 / np.sqrt(iter_idx + 1)

            # gradient step and projection
            a = _project_onto_simplex(a - eta * grad, target_sum)

            iter_idx += 1

            # occasional random perturb
            if np.random.rand() < 0.01:
                noise = np.random.randn(n) * (target_sum * 1e-3 / n)
                a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
                step0 *= 0.9

            # evaluate & update best
            conv = _conv_fft(a)
            max_val = conv.max()
            score = 2.0 * n * max_val / (target_sum ** 2)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] new global best {best_score:.6f} n={n}")
                # maybe break early if below target
                if best_score < 1.503:
                    break

        # early exit if found good
        if best_score < 1.503:
            break
    # fallback
    if best_seq is None:
        n=1000; target_sum = np.sqrt(2.0*n)
        best_seq = np.full(n,target_sum/n)
    return best_seq.tolist()

            print(f"[{time.time():.2f}] Init new best from params p={p:.3f}, delta={delta:.2f}, lam={lam:.4f} -> {obj:.6f} n={n}")
            if best_obj < 1.503:

        # now local search on parameters

                sigma_lam *= 0.99

                    print(f"[{time.time():.2f}] Param search improved to {obj_new:.6f} n={n}")

"""
Hybrid random‑parameter search + subgradient descent for the
max‑convolution objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The code works in three stages:

1. **Randomised shape generation** – a_i = (i+1+δ)^‑p·exp(-λ·i) with
   parameters (p,δ,λ) sampled uniformly in generous intervals.
   The vector is projected onto the simplex Σa_i = √(2n) (the objective
   is scale‑invariant, the simplex merely prevents degenerate zero‑sum
   vectors).

2. **Low‑dimensional parameter optimisation** – a simple (1+1)‑ES
   (mutate parameters, keep improvements) refines (p,δ,λ) for the
   current length.  This quickly discovers good “global” shapes.

3. **High‑dimensional subgradient descent** – the active (maximal)
   convolution index is identified, its sub‑gradient
   ∂/∂a_i max(conv) = 2·a_{k‑i} is computed and a projected
   gradient step with a decaying stepsize is performed.  Occasional
   random jitter helps escape plateaus.

All heavy work (full convolution and its maximum) is performed with
FFT‑based convolution.  The routine repeatedly explores many lengths
between 800 and 3500 (larger lengths are tried when time permits) and
keeps the best vector found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – full linear convolution via real FFT (fast, small memory)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using a power‑of‑2 FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper – compute the objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv) / (sum(a)**2).  Handles the trivial reject case."""
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Helper – generate a power‑law / exponential shape
# ----------------------------------------------------------------------
def _generate_shape(n: int, p: float, delta: float, lam: float) -> np.ndarray:
    """a_i = (i+1+δ)^‑p · exp(-λ·i).  Returns an unnormalised vector."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0 + delta) ** (-p) * np.exp(-lam * i)
    return a


# ----------------------------------------------------------------------
# Helper – sub‑gradient of the max‑convolution entry (averaged over ties)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, rel_tol: float = 1e-9) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (average over all maximal indices)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # treat all entries within a tiny relative tolerance as active (to avoid empty set)
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # derivative of a_i a_{k-i} appears twice (i ↔ k-i) → 2·a_{partner}
        grad[idx] += 2.0 * a[partner]
    grad /= float(len(active))
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The routine mixes a cheap parametric random
    search with a projected sub‑gradient descent on the active convolution
    constraint and respects the 1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Lengths we are willing to try – larger lengths give finer discretisation.
    length_choices = np.arange(800, 3501, 400)   # 800,1200,…,3400

    # ------------------------------------------------------------------
    # Main outer loop – each iteration grabs a fresh problem size and
    # performs an inexpensive parametric search followed by a short
    # sub‑gradient refinement.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a length (adapt to remaining time)
        if deadline - time.time() < 30.0:
            n = int(np.random.choice(length_choices[:3]))   # stay modest
        else:
            n = int(np.random.choice(length_choices))

        target_sum = np.sqrt(2.0 * n)

        # --- 1. random initial parameters ---------------------------------
        p = np.random.uniform(0.35, 0.75)
        delta = np.random.uniform(0.0, 8.0)
        lam = np.random.uniform(0.0, 0.02)

        a = _generate_shape(n, p, delta, lam)
        a = _project_onto_simplex(a, target_sum)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init new best = {best_val:.6f}  n={n}")

        # --- 2. cheap (1+1)‑ES over the three shape parameters -------------
        # allocate a few seconds for this low‑dimensional optimisation
        param_deadline = min(deadline, time.time() + 5.0)
        sigma_p, sigma_delta, sigma_lam = 0.06, 1.5, 0.006

        while time.time() < param_deadline:
            p_new = p + np.random.randn() * sigma_p
            delta_new = delta + np.random.randn() * sigma_delta
            lam_new = lam + np.random.randn() * sigma_lam

            # enforce sensible bounds
            if not (0.20 <= p_new <= 1.20):
                continue
            if delta_new < 0.0:
                delta_new = 0.0
            if lam_new < 0.0:
                lam_new = 0.0

            a_new = _generate_shape(n, p_new, delta_new, lam_new)
            a_new = _project_onto_simplex(a_new, target_sum)
            val_new = _objective(a_new)

            # accept only improvements (simple ES)
            if val_new < cur_val:
                p, delta, lam = p_new, delta_new, lam_new
                a = a_new
                cur_val = val_new
                sigma_p *= 0.99
                sigma_delta *= 0.99
                sigma_lam *= 0.99
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] Param‑search improved = {best_val:.6f}  n={n}")

                if best_val < 1.503:
                    break

        if best_val < 1.503:
            break

        # --- 3. projected sub‑gradient descent on the max‑convolution -----
        gd_deadline = min(deadline, time.time() + 5.0)
        step0 = 0.72                                 # initial step‑size factor
        it = 0
        while time.time() < gd_deadline:
            grad = _subgradient_max(a, rel_tol=1e-9)
            eta = step0 / np.sqrt(it + 1.0)
            a = _project_onto_simplex(a - eta * grad, target_sum)
            it += 1

            # occasional random shake‑up (helps escape flat plateaus)
            if np.random.rand() < 0.01:
                noise = np.random.randn(n) * (target_sum * 0.001 / n)
                a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
                step0 *= 0.92

            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] GD improved = {best_val:.6f}  n={n}")
                if best_val < 1.503:
                    break

        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – in the unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _objective(best_seq)

    print(f"Final best objective = {best_val:.6f} (length = {len(best_seq)})")
    # convert to plain Python list as required by the evaluator
    return best_seq.tolist()
