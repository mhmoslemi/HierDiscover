def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_val = float('inf')
    best_seq = None

    n = 2500 # maybe adjust later

    # allocate size for FFT: for convolution and correlation
    size_conv = 1 << int(np.ceil(np.log2(2*n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3*n))) # enough for w and rev_a

    # define step scheduler
    step0 = 0.5
    lam = 5.0
    lam_inc = 1.001
    lam_max = 300.0

    # start with random Dirichlet or geometric curv
    seq = np.random.dirichlet(np.ones(n))
    # ensure sum exactly 1
    seq = seq / seq.sum()

    # optionally improve with smoothing steps. 
    iter_count = 0
    last_improve_time = time.time()
    while True:
        now = time.time()
        if now > deadline:
            break

        # compute convection
        conv = _full_convolution(seq)
        max_conv = conv.max()
        obj = 2 * n * max_conv # sum=1

        # update best
        if obj < best_val:
            best_val = obj
            best_seq = seq.copy()
            last_improve_time = now
            print(f"[Iter {iter_count:06d}] new best obj={obj:.6f}")

        # compute gradient of smooth surrogate
        # w weights via softmax
        m = max_conv
        w = np.exp(lam * (conv - m))
        w_sum = w.sum()
        if w_sum == 0:
            # numerical issue; fall back to uniform
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # compute gradient via correlation
        nlen = n
        rev_a = seq[::-1]
        size_corr_cur = size_corr  # predetermined
        fw = np.fft.rfft(w, size_corr_cur)
        frev = np.fft.rfft(rev_a, size_corr_cur)
        corr = np.fft.irfft(fw * frev, size_corr_cur)
        grad = 2.0 * corr[nlen - 1: nlen - 1 + nlen]  # shape n

        # step size
        eta = step0 / np.sqrt(iter_count + 1)
        # optionally scaling grad to unit norm
        #grad_norm = np.linalg.norm(grad) # maybe l2
        #if grad_norm > 0:
        #    grad = grad / grad_norm

        # candidate update
        seq_new = seq - eta * grad
        seq_new = _project_onto_simplex(seq_new)

        # check improvement; if not, reduce step size
        if _objective(seq_new) >= obj:
            # try smaller step sizes
            for factor in [0.5, 0.25, 0.125]:
                eta2 = eta * factor
                seq_tmp = seq - eta2 * grad
                seq_tmp = _project_onto_simplex(seq_tmp)
                obj_tmp = _objective(seq_tmp)
                if obj_tmp < obj:
                    seq_new = seq_tmp
                    obj = obj_tmp
                    break
            # else, keep original seq (no update)
        # assign new seq
        seq = seq_new

        # schedule update lambda
        if lam < lam_max:
            lam *= lam_inc
        # schedule progress
        iter_count += 1

        # random shake if stuck >1.5 sec
        if now - last_improve_time > 1.5:
            noise = (np.random.rand(n) - 0.5) * 1e-4
            seq = _project_onto_simplex(seq + noise)
            last_improve_time = now

    print(f"Finished: best objective = {best_val:.6f} (n={n})")
    return best_seq.tolist()

Optimise step‑function heights to minimise the evaluation bound.

Algorithm
---------
* Represent the heights as a probability vector (non‑negative, sum = 1).
* Optimise a smooth approximation of the max‑convolution with
  exponentiated‑gradient (mirror‑descent) updates.
* λ controls the smooth‑max: start small and increase slowly.
* Step‑size η decays ∝1/√t, keeping progress possible for the whole
  budget.
* If no improvement for a few seconds a tiny random shake is added.
  The best sequence found is returned.
"""

import time
import numpy as np

def _project_onto_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a):
    """Linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def _gradient_from_weights(w, a, size_corr=None):
    """
    Compute ∇φ where φ uses the weight vector w (sum w = 1):
        ∇φ = 2·(w ★ a)
    Returned gradient has length len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to the original vector indices
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

    """Exact objective used by evaluate_sequence (sum of a is assumed ≈1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()

            last_improve = time.time()

        it += 1

    # ----------------------------------------------------------------
    print(f"Finished optimisation – best objective = {best_val:.6f} (iterations {it})")
    return best_a.tolist()
