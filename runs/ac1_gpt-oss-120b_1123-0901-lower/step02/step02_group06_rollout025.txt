def project_to_simplex(v, target):
    # similar to original but returning sum=target.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        # all non-positive, return uniform distribution
        w = np.full_like(v, target / len(v))
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence value.

Algorithm
---------
1. Choose a modest length `n` (default 2000) and fix the L1‑norm to
   `S = sqrt(2*n)`, which makes the evaluation equal to
   `max_k (a*a)[k]`.
2. Build a small pool of diverse initial sequences (uniform,
   geometric families, arcsine, a few Dirichlet draws and, if
   available, the previously best global `height_sequence_1`).
   Keep the best as the start point.
3. Optimise the sequence with a **soft‑max approximation** of the
   max‑convolution:
      * `M(a) = log_sum_exp(alpha * (a*a)) / alpha`
      * `∇M = 2 * (w ★ a)` where `w_i = exp(alpha*(conv_i-M)) / Σ`
        and `★` denotes cross‑correlation.
   The gradient is computed efficiently with FFT‑based convolution.
4. Use the Adam optimiser (first‑ and second‑moment estimates),
   project after each step onto the simplex `{x≥0, Σx = S}` and
   slowly increase the temperature `alpha` (smooth → hard max).
5. If no improvement is observed for a while, inject a small Gaussian
   jitter to escape possible plateaus.
6. Continue until the time budget expires and return the best
   sequence found.

The use of the full soft‑max gradient (instead of a few top‑K
sub‑gradients) gives a much more accurate descent direction,
while the FFT makes each iteration cheap enough to explore many
updates within the 1000 s budget.

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project onto the probability simplex (sum = 1, all entries ≥ 0)."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # all non‑positive → uniform point
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_to_simplex_sum(v, target):
    """Project onto the simplex with prescribed sum `target`."""
    w = _project_onto_simplex(v)
    return w * target

def _initial_candidates(n, rng, target_sum):
    """Create a handful of promising starters (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, target_sum / n))

    # 2. geometric families
    for factor in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 1.8, 2.2]:

        seq = seq / seq.sum() * target_sum
        cand.append(seq)

    # 3. arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))

def _conv_self_fft(a, Nfft):
    """Self‑convolution a*a using FFT, result length 2*len(a)-1."""
    fa = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(fa * fa, n=Nfft)[: 2 * len(a) - 1]
    return conv


def _gradient_softmax(a, w, Nfft):
    """
    Gradient of the soft‑max approximation of the max convolution.

    grad_i = 2 * Σ_k w_k * a_{k-i}
    Implemented via cross‑correlation using FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=Nfft)
    fa_rev = np.fft.rfft(a_rev, n=Nfft)
    corr = np.fft.irfft(fw * fa_rev, n=Nfft)[: (len(w) + n - 1)]
    # slice centred at offset n-1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

                     **kwargs) -> list[float]:
    """Return a non‑negative height sequence that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    a = best_seq.copy()

    v = np.zeros_like(a)        # second moment
    beta1, beta2 = 0.9, 0.999

    lr = 0.6                    # base learning rate
    alpha = 1.0                 # temperature for soft‑max
    alpha_max = 30.0
    alpha_inc = 0.0005
    it = 1
    no_improve = 0

    # pre‑compute FFT lengths (next power of two)
    Nfft_conv = 1 << (int(np.ceil(np.log2(2 * n - 1))))
    Nfft_grad = 1 << (int(np.ceil(np.log2(3 * n - 2))))

    # bias‑correction powers
    beta1_pow = beta1
    beta2_pow = beta2

    while time.time() < deadline:
        # ----- convolution of current iterate -----
        conv = _conv_self_fft(a, Nfft_conv)

        # ----- soft‑max weights -----
        max_conv = conv.max()
        # stable softmax (shift by the maximum)
        exp_term = np.exp(alpha * (conv - max_conv))
        w = exp_term / exp_term.sum()   # shape (2n‑1,)

        # ----- gradient -----
        grad = _gradient_softmax(a, w, Nfft_grad)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # ----- projection onto the feasible simplex (sum = target_sum) -----
        a_new = _project_to_simplex_sum(a_new, target_sum)

        # ----- evaluate -----
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[Iter {it:06d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- schedule adjustments -----
        lr *= 0.9999                     # very slow decay
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # ----- occasional random perturbation to escape stagnation -----
        if no_improve > 5000:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a_new = a_new + jitter
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / a_new.sum() * target_sum

            v.fill(0.0)
            beta1_pow = beta1
            beta2_pow = beta2
            no_improve = 0
            print(f"[Iter {it:06d}] stagnation – random jitter applied")

        # ----- bookkeeping for next iteration -----
        a = a_new
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
