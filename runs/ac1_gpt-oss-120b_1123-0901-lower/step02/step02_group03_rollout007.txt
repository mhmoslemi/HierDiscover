"""
Smooth‑max gradient descent for minimizing
        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective is scale‑invariant, therefore we keep the vector on the
simplex Σ a_i = √(2n).  The max‑convolution is smoothed with a
log‑sum‑exp (soft‑max) with temperature μ, giving a differentiable
approximation.  An Adam optimizer moves the point on the simplex;
after each step we project back onto the simplex.  The temperature is
cooled geometrically so that the algorithm first explores a smooth
landscape and then focuses on the true max.  Several independent
restarts with different lengths (≈1200‑3000) are performed within the
time budget.  The best sequence found is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – Σ x_i = s,   x_i ≥ 0
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# FFT based full convolution (real inputs). Returns length len(x)+len(y)-1.
# --------------------------------------------------------------------------- #
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real FFT)."""
    return _conv_fft(a, a)


# --------------------------------------------------------------------------- #
# Smooth approximation of max(conv) and its gradient.
# --------------------------------------------------------------------------- #
def _smooth_max_and_grad(a: np.ndarray, mu: float):
    """
    Returns three values:
        f      – smooth approximation of max(conv) (log‑sum‑exp)
        grad   – ∇_a f  (same shape as a)
        max_c  – the actual max of the convolution (objective value)
    """
    n = a.shape[0]
    conv = _conv_full_fft(a)                # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (numerically stable)
    shifted = conv - max_c                  # ≤ 0
    w = np.exp(shifted / mu)                # exp( (c_i - max)/mu )
    w_sum = w.sum()
    w /= w_sum                              # normalized weights, sum = 1

    # gradient: 2 * (w * a) convolved with reversed a
    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)        # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    # smooth objective value: μ·log Σ exp(c_i/μ) = max_c + μ·log Σ exp((c_i-max_c)/μ)
    f = max_c + mu * np.log(w_sum)

    return f, grad, max_c


# --------------------------------------------------------------------------- #
# One optimisation instance on a given length.
# --------------------------------------------------------------------------- #
def _run_one_instance(
    n: int,
    init_vec: np.ndarray,
    target_sum: float,
    deadline: float,
    best_state: dict,
    rng: np.random.Generator,
):
    """
    Runs a single Adam‑based optimisation for up to ~12 s.
    Updates ``best_state`` (keys: 'seq', 'score') when a better
    candidate is discovered.
    """
    # start from a feasible point
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 1.0                      # base learning rate (works well on simplex)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    # will be set after the first convolution (needs a scale)
    mu = None
    mu_decay = 0.998
    mu_min = 1e-6

    iter_cnt = 0
    run_deadline = min(deadline, time.time() + 12.0)

    while time.time() < run_deadline:
        # first iteration: initialise temperature from scale of the problem
        if mu is None:
            conv0 = _conv_full_fft(a)
            max0 = conv0.max()
            mu = max(0.2 * max0, 1e-4)   # start relatively smooth

        # smooth max and gradient
        f, grad, max_conv = _smooth_max_and_grad(a, mu)

        # record best (the true objective is max_conv)
        if max_conv < best_state["score"]:
            best_state["score"] = float(max_conv)
            best_state["seq"] = a.copy()
            print(
                f"[{time.time():.2f}] New best = {best_state['score']:.6f} (n={n})"
            )
            # early exit if we beat the target
            if best_state["score"] < 1.503:
                return

        # Adam update (minimisation)
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # project back onto the feasible simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        a = a_new

        # temperature cooling
        mu = max(mu * mu_decay, mu_min)

        # occasional random shake to escape plateaus
        if rng.random() < 0.02:
            noise = rng.standard_normal(n) * (0.005 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # reset Adam statistics a little
            m *= 0.5
            v *= 0.5

    # end of this run (time out or finished iterations)
    return


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Searches for a step‑function (non‑negative heights) that minimises
    the evaluation function.  The method uses a smooth‑max Adam optimiser
    with simplex projection and many independent restarts of varying
    lengths.  The best sequence found before ``budget_s`` seconds expires
    is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # container for the current best candidate
    best_state = {"seq": None, "score": float("inf")}

    # ------------------------------------------------------------------- #
    # Outer loop: many independent restarts with random lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # pick a length (larger n gives more freedom but is slower)
        n = rng.integers(1200, 3001)

        # initialise a vector – either from the current best, from the
        # historic solution ``height_sequence_1``, or from a fresh random point.
        if best_state["seq"] is not None and rng.random() < 0.4:
            src = best_state["seq"]
            if src.size >= n:
                init = src[:n].copy()
            else:
                # upscale by interpolation to the new size
                x_old = np.linspace(0, 1, src.size)
                x_new = np.linspace(0, 1, n)
                init = np.interp(x_new, x_old, src)
        elif "height_sequence_1" in globals() and rng.random() < 0.35:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = rng.random(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # Dirichlet‑like random start (positive entries, then projected)
            random_expo = -np.log(rng.random(n))
            init = random_expo

        target_sum = np.sqrt(2.0 * n)

        _run_one_instance(
            n=n,
            init_vec=init,
            target_sum=target_sum,
            deadline=deadline,
            best_state=best_state,
            rng=rng,
        )

        # stop early if we already reached the target bound
        if best_state["score"] < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_state["seq"] is None:
        n = 1000
        t_sum = np.sqrt(2.0 * n)
        best_state["seq"] = np.full(n, t_sum / n, dtype=float)

    # Convert to plain Python list as required by the grader
    return best_state["seq"].tolist()
