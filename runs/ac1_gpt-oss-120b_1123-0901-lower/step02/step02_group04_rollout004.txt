def _grad_softmax(a, lam):
    conv = _conv_self(a)
    # shift for numeric stability
    maxc = conv.max()
    exp = np.exp(lam * (conv - maxc))
    w = exp / exp.sum()
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)

    a = initial.copy()
    n = a.shape[0]
    beta = 0.9
    base_lr = 0.5
    m = np.zeros_like(a)
    # schedule
    lam_start = 0.1
    lam_end = 10.0

    except Exception:
        return best_seq, best_score

        for k in range(rows):
            low = max(0, k - (n - 1))

            best_seq = a.copy()
            print(f"[{time.time():.2f}] New best after LP = {best_score:.6f} (len={n})")
        # possibly break if time low

            break

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function heights sequence that minimises
    2·n·max(conv(a,a)) / (sum a)².

    The algorithm proceeds in three stages:
    1. Generate a diverse set of candidate shapes (exponential,
       linear, triangular) for several lengths and keep the best.
    2. Refine the current best with a smooth‑gradient descent that
       minimises a soft‑max approximation of the ‖a*a‖∞ norm.
       Momentum (exponential moving average) and a decreasing step‑size
       are used; the iterate is projected onto the simplex
       Σa_i = √(2 n) after each step.
    3. Optionally perform one or more linear‑programming “LP‑step”
       (the SOTA refinement) if SciPy is available.
    The routine respects the supplied time budget and returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Helper: shape factories (non‑negative, not normalised)
    # ------------------------------------------------------------------
    def shape_exp(n, alpha):
        return np.exp(-alpha * np.arange(n, dtype=float))
    def shape_exp_rev(n, alpha):
        return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))
    def shape_linear_desc(n, beta):
        i = np.arange(n, dtype=float)
        return np.maximum(0.0, 1.0 - beta * i)
    def shape_quad_desc(n, beta):
        i = np.arange(n, dtype=float)
        return np.maximum(0.0, 1.0 - beta * i * i)
    def shape_triangular(n):
        i = np.arange(n, dtype=float)
        return np.minimum(i + 1.0, n - i)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric sweep
    # ------------------------------------------------------------------
    lengths = list(range(200, 1601, 200)) + [1326, 1500]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-4, -0.3, num=12):
            a = shape_exp(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _max_conv(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (sweep) = {best_score:.6f} (len={n})")
        # reversed exponential
        for alpha in np.logspace(-4, -0.3, num=12):
            a = shape_exp_rev(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _max_conv(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (sweep rev) = {best_score:.6f} (len={n})")
        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=10):
            a = shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _max_conv(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (linear) = {best_score:.6f} (len={n})")
        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=10):
            a = shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _max_conv(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (quad) = {best_score:.6f} (len={n})")
        # triangular shape
        a = shape_triangular(n)
        a = a / a.sum() * target
        score = _max_conv(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New best (triangular) = {best_score:.6f} (len={n})")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Include global height_sequence_1 if available (previous best)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is expected to be a list of floats
        hs1 = np.array(height_sequence_1, dtype=float)
        n = hs1.shape[0]
        target = math.sqrt(2.0 * n)
        hs1 = hs1 / hs1.sum() * target
        score = _max_conv(hs1)
        if score < best_score:
            best_score = score
            best_seq = hs1.copy()
            print(f"[{time.time():.2f}] New best (global init) = {best_score:.6f} (len={n})")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage 2 – smooth gradient descent (soft‑max) refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        target = math.sqrt(2.0 * best_seq.shape[0])
        best_seq, best_score = _gradient_descent(best_seq, target, deadline, best_score, best_seq)

    # Random restarts with gradient descent
    while time.time() < deadline:
        # choose length randomly in a reasonable range
        n = np.random.randint(400, 2001)
        target = math.sqrt(2.0 * n)
        # initialise with a random exponential shape
        alpha = np.random.rand() * 0.02 + 1e-5
        a0 = shape_exp(n, alpha)
        a0 = a0 / a0.sum() * target
        best_seq, best_score = _gradient_descent(a0, target, deadline, best_score, best_seq)
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – LP refinement (if SciPy present)
    # ------------------------------------------------------------------
    best_seq, best_score = _lp_refinement(best_seq, math.sqrt(2.0 * best_seq.shape[0]), deadline, best_score, best_seq)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # return as plain Python list
    return best_seq.tolist()

      differentiable.  Its gradient can be written as a convolution and
      is computed efficiently via FFT.
    * An exponential moving‑average momentum and a diminishing step size
      are employed; after each update the iterate is projected back onto
      the simplex.
    * The current best sequence from stage 1 (or the global variable
      ``height_sequence_1`` if present) is refined, and additional random
      restarts are performed while time permits.

3.  Optional LP refinement (the “SOTA” step) using SciPy’s ``linprog``.
    A linear programme maximises Σg_j under the constraint (a∗g) ≤ M,
    where M is the current maximal autocorrelation.  The resulting g is
    blended with a.

The routine respects a time budget (default 1000 s) and returns the
best sequence found as a plain Python list.
"""

import math
import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# 1.  Simplex projection
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# 2.  Fast convolution (full) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using real FFT (fast for n up to a few thousand)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# 3.  Cheap evaluation (max of the autocorrelation)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a,a)) – the quantity that equals the evaluation
    score when Σa = √(2 n)."""
    return float(_conv_self(a).max())

# ----------------------------------------------------------------------
# 4.  Shape factories (non‑normalised)
# ----------------------------------------------------------------------
def shape_exp(n: int, alpha: float) -> np.ndarray:
    """Monotonically decreasing exponential."""
    return np.exp(-alpha * np.arange(n, dtype=float))


def shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    """Reversed exponential (decreasing from the right)."""
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def shape_linear_desc(n: int, beta: float) -> np.ndarray:
    """Linear decay truncated at zero."""
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def shape_quad_desc(n: int, beta: float) -> np.ndarray:
    """Quadratic decay truncated at zero."""
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def shape_triangular(n: int) -> np.ndarray:
    """Symmetric triangular bump."""
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)

# ----------------------------------------------------------------------
# 5.  Gradient of the soft‑max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   L(a) = (1/lam) * log Σₖ exp(lam * cₖ)   with   c = a∗a .
    The gradient can be written as
        ∇L = 2 * conv_rev( a , w_rev )
    where wₖ ∝ exp(lam * cₖ) and w_rev is w reversed.
    """
    n = a.shape[0]

    # autocorrelation
    conv = _conv_self(a)

    # numerically stable softmax weights
    max_c = conv.max()
    exp_weights = np.exp(lam * (conv - max_c))
    w = exp_weights / exp_weights.sum()          # shape (2n‑1,)

    # reverse the weight vector
    w_rev = w[::-1]

    # convolution a * w_rev (full)
    conv_rev = np.convolve(a, w_rev)

    # Extract the n components needed for the gradient.
    # Index mapping: grad[i] = 2 * Σ_j w[i+j] * a[j]
    grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]

    return grad

# ----------------------------------------------------------------------
# 6.  Smooth gradient descent (soft‑max) refinement
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    start_vec: np.ndarray,
    target_sum: float,
    deadline: float,
    cur_best_vec: np.ndarray,
    cur_best_score: float,
) -> Tuple[np.ndarray, float]:
    """
    Perform many steps of gradient descent on the soft‑max approximation.
    Momentum (exponential moving average) and a diminishing step size are
    used.  The iterate is projected onto the simplex after each step.
    Whenever a better sequence (lower max‑convolution) is found the global
    best is updated.
    """
    a = start_vec.astype(float)
    n = a.shape[0]

    # hyper‑parameters
    beta = 0.9                     # momentum decay
    base_lr = 0.5                  # base learning rate
    lam_start = 0.1
    lam_end = 10.0
    max_iters_for_lambda = 5000   # iterations before λ reaches lam_end

    m = np.zeros_like(a)           # momentum accumulator
    it = 0

    best_vec = cur_best_vec
    best_score = cur_best_score

    while time.time() < deadline:
        # ------- schedule for λ (soft‑max temperature) -------
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iters_for_lambda)

        # ------- gradient of the smooth objective -------
        grad = _grad_softmax(a, lam)

        # ------- momentum update -------
        m = beta * m + (1.0 - beta) * grad

        # ------- step size (diminishing) -------
        step = base_lr / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        # ------- gradient step and projection -------
        a_new = a - eta * m
        a_new = _project_onto_simplex(a_new, target_sum)

        # ------- occasional evaluation -------
        if it % 10 == 0:
            cur_score = _max_conv(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a_new.copy()
                print(f"[{time.time():.2f}] New best (smooth) = {best_score:.6f} (len={n})")

        a = a_new
        it += 1

        # stop early if we are already below the target threshold
        if best_score < 1.503:
            break

    return best_vec, best_score

# ----------------------------------------------------------------------
# 7.  Sub‑gradient descent on the true max‑norm (fallback / extra refinement)
# ----------------------------------------------------------------------
def _subgradient_descent(
    start_vec: np.ndarray,
    target_sum: float,
    deadline: float,
    cur_best_vec: np.ndarray,
    cur_best_score: float,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method on the exact max‑norm (the routine used
    in the previous state‑of‑the‑art implementation).  Kept here as a
    cheap secondary refinement that can be invoked if desired.
    """
    a = start_vec.astype(float)
    n = a.shape[0]

    beta = 0.9
    base_step = 0.2
    m = np.zeros_like(a)
    it = 0

    best_vec = cur_best_vec
    best_score = cur_best_score

    while time.time() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        low = max(0, kmax - (n - 1))
        high = min(n - 1, kmax)
        idx = np.arange(low, high + 1, dtype=int)
        partner = kmax - idx

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # sub‑gradient step
        direction = -grad
        m = beta * m + (1.0 - beta) * direction
        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        a = a + eta * m
        a = _project_onto_simplex(a, target_sum)

        if it % 5 == 0:
            cur_score = _max_conv(a)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] New best (subgrad) = {best_score:.6f} (len={n})")

        it += 1
        if best_score < 1.503:
            break

    return best_vec, best_score

# ----------------------------------------------------------------------
# 8.  LP refinement (the SOTA “LP‑step”)
# ----------------------------------------------------------------------
def _lp_refinement(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    cur_best_vec: np.ndarray,
    cur_best_score: float,
) -> Tuple[np.ndarray, float]:
    """
    Solve the linear programme
        maximise Σ g_j
        s.t. g_j ≥ 0  and  (a∗g)_k ≤ M   for all k,
    where M = max(a∗a).  The solution g is rescaled to the simplex and
    blended with a.  The process is repeated while time permits.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return cur_best_vec, cur_best_score

    n = a.shape[0]
    best_vec = cur_best_vec
    best_score = cur_best_score

    while time.time() < deadline and best_score > 1.503:
        conv = _conv_self(a)
        M = conv.max()
        rows = 2 * n - 1

        # Build matrix A where (a∗g)_k = Σ_j a_{k-j} g_j .
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(
            c=-np.ones(n),                 # maximise Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0:
            break
        g = g * (target_sum / s)

        # convex combination with a small mixing coefficient
        t = 0.2
        a = (1 - t) * a + t * g

        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] New best (LP) = {best_score:.6f} (len={n})")

        # stop if we are close to the hard deadline
        if time.time() + 1.0 > deadline:
            break

    return best_vec, best_score

# ----------------------------------------------------------------------
# 9.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine respects ``budget_s`` seconds of
    CPU time and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None          # type: np.ndarray | None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric sweep
    # --------------------------------------------------------------
    lengths = list(range(200, 1601, 200)) + [1326, 1500]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-4, -0.3, num=12):
            a = shape_exp(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            sc = _max_conv(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best (exp) = {best_score:.6f} (len={n})")

        # reversed exponential
        for alpha in np.logspace(-4, -0.3, num=12):
            a = shape_exp_rev(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            sc = _max_conv(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best (exp_rev) = {best_score:.6f} (len={n})")

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=10):
            a = shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            sc = _max_conv(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best (linear) = {best_score:.6f} (len={n})")

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=10):
            a = shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            sc = _max_conv(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best (quad) = {best_score:.6f} (len={n})")

        # triangular bump
        a = shape_triangular(n)
        a = a / a.sum() * target
        sc = _max_conv(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.2f}] New best (triangular) = {best_score:.6f} (len={n})")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Stage 1b – seed from the previous run if available
    # --------------------------------------------------------------
    try:
        hs1 = np.array(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        n = hs1.shape[0]
        target = math.sqrt(2.0 * n)
        hs1 = hs1 / hs1.sum() * target
        sc = _max_conv(hs1)
        if sc < best_score:
            best_score, best_seq = sc, hs1.copy()
            print(f"[{time.time():.2f}] New best (global init) = {best_score:.6f} (len={n})")
    except Exception:
        pass  # no global variable present

    # --------------------------------------------------------------
    # Stage 2 – smooth gradient descent refinement
    # --------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        best_seq, best_score = _smooth_gradient_descent(
            best_seq, target, deadline, best_seq, best_score
        )

    # Additional random restarts using the same smooth descent
    while time.time() < deadline:
        # choose a moderate length for a quick restart
        n = np.random.randint(400, 2001)
        target = math.sqrt(2.0 * n)

        # initialise with a random exponential shape
        alpha = np.random.rand() * 0.02 + 1e-5
        a0 = shape_exp(n, alpha)
        a0 = a0 / a0.sum() * target

        best_seq, best_score = _smooth_gradient_descent(
            a0, target, deadline, best_seq, best_score
        )
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # Stage 3 – optional LP refinement (SOTA step)
    # --------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target = math.sqrt(2.0 * n)
        best_seq, best_score = _lp_refinement(
            best_seq, target, deadline, best_seq, best_score
        )

    # --------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Return as a plain Python list for the evaluator.
    return best_seq.tolist()
