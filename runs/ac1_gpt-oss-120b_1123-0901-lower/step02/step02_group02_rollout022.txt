def _optimise_one_length_smooth(n, init, run_deadline, gamma0=0.5, gamma_factor=1.02, gamma_max=30.0, base_step=0.5, momentum=0.9):
    a = project onto simplex (target sum S)
    best_local = a.copy(); best_score = compute_score(a)
    step_factor = base_step
    gamma = gamma0
    velocity = np.zeros_like(a)

    while time.time() < run_deadline:
        # compute convolution + weighted gradient
        smooth_max, grad, max_val = _smooth_max_grad(a, gamma)
        # compute velocity for momentum
        velocity = momentum * velocity + (1 - momentum) * grad
        # step size: adapt as before: step = step_factor * smooth_max / (max(|velocity|)+eps)
        max_vel = np.max(np.abs(velocity)) if np.max(np.abs(velocity)) > 0 else 1.0
        eta = step_factor * smooth_max / max_vel
        a_new = a - eta * velocity
        # project onto simplex
        a_new = _project_onto_simplex(a_new, S)

        # random jitter occasionally
        if np.random.rand() < 0.01:
            a_new += np.random.randn(n) * (S * 1e-4 / n)
            a_new = _project_onto_simplex(np.maximum(a_new, 0.0), S)

        # evaluate actual score
        conv_new = _conv_fft(a_new)  # maybe compute again via FFT (full linear conv)
        max_val_new = conv_new.max()
        score = 2 * n * max_val_new / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n} smooth_step best={score:.6f}")

        # update a
        a = a_new

        # anneal gamma
        gamma = min(gamma_max, gamma * gamma_factor)

        # decay step factor
        step_factor *= 0.9995

    return best_local, best_score

"""
Enhanced soft‑max gradient descent for minimizing 
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm:
- Initialise a non‑negative power‑law vector and project onto the simplex
  Σ a_i = √(2 n).
- At each iteration compute the full auto‑convolution via FFT.
- Form a soft‑max weight vector over the convolution entries; this yields a
  smooth approximation of the maximum and a differentiable objective.
- The weighted gradient is obtained by a second convolution (computed again
  with FFT), giving an O(n log n) update.
- A momentum term and an adaptive step size are used, together with occasional
  random jitter.
- Many independent restarts with varying length `n` explore the search space.
  The best vector found before the time budget expires is returned.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)      # clip tiny negatives
    return conv

def _smooth_max_and_grad(a: np.ndarray, gamma: float):
    """
    Returns (smooth_max, gradient, max_conv) where
    - smooth_max approximates max(conv) via a soft‑max with temperature 1/gamma,
    - gradient is the derivative of the smooth objective w.r.t. a,
    - max_conv is the true maximum of the convolution.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # FFT size large enough for both convolutions (a*a and a*w_rev)
    size = 1 << ((n + m - 1 - 1).bit_length())
    # FFT of a
    Fa = np.fft.rfft(a, n=size)
    # full auto‑convolution a*a
    conv = np.fft.irfft(Fa * Fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)
    max_val = conv.max()
    # soft‑max weights over the convolution entries
    exp_term = np.exp(gamma * (conv - max_val))
    w = exp_term / exp_term.sum()
    smooth_max = np.dot(w, conv)

    # weighted gradient: 2 * (a (*) w_rev) where (*) denotes convolution
    w_rev = w[::-1]
    Fw_rev = np.fft.rfft(w_rev, n=size)
    grad_full = np.fft.irfft(Fa * Fw_rev, n=size)[:n + m - 1]
    # extract the n components corresponding to the gradient
    idx = (m - 1) - np.arange(n)   # descending slice
    grad = 2.0 * grad_full[idx]
    return smooth_max, grad, max_val

def _optimise_one_length_smooth(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    base_step: float = 0.5,
    momentum_beta: float = 0.85,
    gamma0: float = 0.5,
    gamma_factor: float = 1.02,
    gamma_max: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed objective for a single length `n`.

    Returns the best vector encountered and its true evaluation score.
    """
    target_sum = np.sqrt(2.0 * n)    # Σ a_i = √(2 n) gives objective = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # best local solution seen so far
    conv = _conv_fft(a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_local = a.copy()

    velocity = np.zeros_like(a)
    step_factor = base_step
    gamma = gamma0

    while time.time() < run_deadline:
        # smooth max, gradient, true max
        smooth_max, grad, max_val = _smooth_max_and_grad(a, gamma)

        # momentum update
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad

        # adaptive step size – scale by smooth_max / max|velocity|
        max_abs_vel = np.max(np.abs(velocity))
        if max_abs_vel < 1e-12:
            max_abs_vel = 1.0
        eta = step_factor * smooth_max / max_abs_vel

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional stochastic perturbation to escape plateaus
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 5e-4 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.9   # become a bit more careful after shake‑up

        # evaluate true objective
        conv_new = _conv_fft(a_new)
        score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score:.6f}")

        a = a_new
        # anneal the soft‑max temperature
        gamma = min(gamma_max, gamma * gamma_factor)
        # slowly decay the global step scaling
        step_factor *= 0.9996

    return best_local, best_score

Improved soft‑max gradient descent for the step‑function objective

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works as follows:

1.  Initialise a power‑law vector `a_i ∝ (i+1)^{-p}` (p≈0.5) and project it onto

            n,
            init,
            run_deadline,
            base_step=0.5,
            momentum_beta=0.85,
            gamma0=0.5,
            gamma_factor=1.018,
            gamma_max=40.0,
        )

        # --------------------------------------------------------------
        # update global best
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # early exit if we are already well below the target
        if best_score < 1.45:
            break

    # --------------------------------------------------------------
    # fallback – extremely unlikely that we found nothing
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # return a plain Python list as required by the evaluator
    return best_seq.tolist()
