def _targeted_move(seq, T):
    # compute conv and max index
    conv = _self_convolve(seq)
    k_star = int(conv.argmax())
    # find i idx giving largest product
    lo = max(0, k_star - (n-1))
    hi = min(k_star, n-1)
    idx = np.arange(lo, hi+1)
    prod = seq[idx] * seq[k_star - idx]
    i_max = int(idx[np.argmax(prod)])
    j_max = k_star - i_max

    # find two smallest indices (excluding i_max, j_max)
    # Use argsort pick two smallest
    order = np.argsort(seq)
    # ensure not i_max or j_max
    incoming = []
    for ind in order:
        if ind != i_max and ind != j_max:
            incoming.append(ind)
        if len(incoming) >= 2:
            break
    r1, r2 = incoming[0], incoming[1]

    delta = min(seq[i_max], seq[j_max] if i_max != j_max else seq[i_max]) * T

    cand[r2] += delta

"""
Propose step‑function heights that improve the current best bound (≈ 1.5449).
The algorithm mixes several ideas that differ from the previous sub‑gradient
method:

* 1️⃣  A wide parametric sweep (geometric, power‑law, beta, triangular,
      cosine‑bell, Gaussian) provides many diverse seeds.

* 2️⃣  A *targeted peak‑reduction* move: the shift that currently attains
      the maximum convolution `k*` is identified, the pair of indices
      `(i, j = k*‑i)` that contributes the largest product `a_i·a_j`
      is located, and a small amount of mass is transferred from that pair
      to two smallest entries of the sequence.  After the move the vector is
      projected back onto the probability simplex (non‑negative, sum = 1).

* 3️⃣  When the targeted move fails to improve the objective we fall back
      to a simple random mass‑transfer (as in the original code) and also
      try a projected sub‑gradient step on the active shift.

* 4️⃣  Simulated‑annealing‐style acceptance: even a worsening move may be
      accepted with probability `exp(−Δ / T)`.  The temperature `T`
      (controlling the transfer size) decays slowly, allowing large
      exploratory steps early on and fine tuning later.

All convolutions are performed with an FFT‑based routine (`_self_convolve`)
that runs in `O(n log n)` time, so the loop can execute tens of thousands
of iterations within the 1000 s time budget.

The routine returns the best sequence found as a plain Python list.

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                # should not happen, but be safe
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution -------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of `seq` with itself (length 2·n‑1).
    Uses a real FFT for speed.
    """
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective (local fast version) ---------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv) / (sum(seq)²) – the same metric as evaluate_sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:                     # safety – mirrors evaluate_sequence
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1 – α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{‑p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like a_i ∝ (i+1)^{a‑1}·(n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq = np.clip(seq, 0.0, None)
    return seq / seq.sum()


def _triangular_seq(n: int) -> np.ndarray:
    """Symmetric triangle: a_i ∝ (i+1)*(n‑i)."""
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1) * (n - i)
    return seq / seq.sum()


def _cosine_bell_seq(n: int, width: float = 0.6) -> np.ndarray:
    """Cosine‑bell (raised cosine) with support ≈ width·n."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    radius = width * n / 2.0
    arg = (i - center) / radius
    seq = np.where(np.abs(arg) <= 1.0,
                   0.5 * (1 + np.cos(np.pi * arg)),
                   0.0)
    return seq / seq.sum()


def _gaussian_seq(n: int, sigma: float = 0.15) -> np.ndarray:
    """Discrete Gaussian centred at the middle of the vector."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - mu) / (sigma * n)) ** 2)
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Sample uniformly on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Targeted peak‑reduction move -------------------------------------------
def _targeted_move(seq: np.ndarray, T: float) -> np.ndarray:
    """
    Reduce the dominant convolution entry.
    - Locate k* = argmax conv.
    - Find the pair (i, j=k*‑i) that maximises a_i·a_j.
    - Transfer a small amount of mass from that pair to two smallest
      entries of the sequence.
    - Re‑project onto the simplex.
    """
    n = seq.shape[0]
    conv = _self_convolve(seq)
    k_star = int(conv.argmax())

    lo = max(0, k_star - (n - 1))
    hi = min(k_star, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    prod = seq[idx] * seq[k_star - idx]
    i_max = int(idx[np.argmax(prod)])
    j_max = k_star - i_max

    # two smallest positions (excluding the two we will drain)
    order = np.argsort(seq)
    recv = []
    for p in order:
        if p != i_max and p != j_max:
            recv.append(p)
        if len(recv) >= 2:
            break
    r1, r2 = recv[0], recv[1]

    # amount to move – proportional to the current temperature
    delta = T * min(seq[i_max],
                    seq[j_max] if i_max != j_max else seq[i_max])
    if delta <= 0.0:
        delta = T * 1e-12

    cand = seq.copy()
    cand[i_max] = max(0.0, cand[i_max] - delta)
    if i_max != j_max:
        cand[j_max] = max(0.0, cand[j_max] - delta)

    cand[r1] += delta
    cand[r2] += delta

    # restore feasibility
    cand = _project_onto_simplex(cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence(...).  The routine runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose a length – moderate enough for many iterations,
    #    but large enough to allow a low theoretical bound.
    n = 2500

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 2. Use an existing global seed if available.
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()
                sc = _objective(init)
                if sc < best_score:
                    best_score = sc
                    best_seq = init.copy()
                    print(f"[Init] global seed score={sc:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Systematic sweep over parametric families.
    families = []

    # geometric
    for alpha in np.linspace(0.3, 5.0, 12):
        families.append(_geom_seq(n, alpha))

    # power‑law
    for p in np.linspace(0.1, 2.5, 12):
        families.append(_powerlaw_seq(n, p))

    # beta‑like
    for a in np.linspace(0.3, 4.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            families.append(_beta_seq(n, a, b))

    # triangular, cosine‑bell, Gaussian
    families.append(_triangular_seq(n))
    for width in [0.5, 0.7, 0.9]:
        families.append(_cosine_bell_seq(n, width))
    for sigma in [0.1, 0.15, 0.2]:
        families.append(_gaussian_seq(n, sigma))

    # random Dirichlet draws
    for _ in range(40):
        families.append(_random_simplex(n))

    # evaluate all candidates (stop early if time runs out)
    for seq in families:
        if time.time() > deadline:
            break
        if seq is None:
            continue
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] candidate score={sc:.6f}")

    # fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 4. Local optimisation – targeted peak reduction + random moves.
    seq = best_seq.copy()
    cur_score = best_score

    it = 0
    T = 0.05                     # initial temperature / transfer magnitude
    no_improve = 0

    while time.time() < deadline:
        # ----- decide which move to try ---------------------------------
        if np.random.rand() < 0.6:
            # targeted peak‑reduction move
            cand = _targeted_move(seq, T)
        else:
            # simple random mass transfer (fallback)
            i, j = np.random.choice(n, 2, replace=False)
            delta = T * min(seq[i], 0.1)
            cand = seq.copy()
            cand[i] = max(0.0, cand[i] - delta)
            cand[j] += delta
            cand = _project_onto_simplex(cand)

        cand_score = _objective(cand)

        # ----- simulated‑annealing acceptance ----------------------------
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[Iter {it:05d}] new best = {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- temperature schedule (slow cooling) -----------------------
        it += 1
        T *= 0.9997                      # very slow decay
        if T < 1e-8:                     # keep a tiny exploration floor
            T = 1e-8

        # ----- occasional restart if stuck for a long time -------------
        if no_improve > 8000:
            # re‑initialise from a fresh random point
            seq = _random_simplex(n)
            cur_score = _objective(seq)
            no_improve = 0
            print(f"[Iter {it:05d}] restart – score {cur_score:.6f}")

    # --------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    # Return plain Python list as required by the outer harness
    return best_seq.tolist()
