def _full_convolve_fft(x, y):
    n = x.shape[0]; m = y.shape[0]; size = 1 << ((n + m - 2).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n+m-1]
    conv[conv < 0] = 0.0  # small negative

Return conv.

    max_val = conv.max()

"""
Propose a non‑negative step‑function (vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

*   Choose a length `n` at random (≈ 1200‑3500) and initialise a power‑law
    shape a_i ∝ (i+1)^‑p (p≈0.5).  Occasionally the current global best is
    used as a warm‑start.

*   Optimise the vector with **Adam** on a *soft‑max* approximation of the
    `max` in the objective:
        w_k = exp(β·(conv_k – max_conv))   (β grows from 0.5 to ≈30)
        grad = ∑_k w_k ∂conv_k/∂a  = 2·(a * w_rev)[2n‑2‑i]

    The convolution needed for the gradient is obtained with a real FFT,
    giving O(n log n) work per iteration.

*   After each gradient step the vector is projected onto the simplex
    {x_i ≥ 0, Σx_i = √(2 n)} – this keeps the objective scale‑invariant.

*   Small random jitter is injected occasionally to leave plateaus.
    Many independent restarts are performed until the time budget (≈1000 s)
    is exhausted; the best sequence seen overall is returned.

The soft‑max/Adam combination provides smoother gradients than the previous
hard‑active‑set method and converges faster to lower peak autocorrelation,
typically beating the 1.503 target within the allocated time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s (Σx = s, x≥0)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of real vectors x and y using an FFT of size
    equal to the next power of two.  Negative values caused by round‑off are
    clipped to zero because the evaluation function never sees negatives.
    """
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())        # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    np.maximum(conv, 0.0, out=conv)               # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Core optimisation routine for a *single* length `n`
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    beta0: float = 0.5,
    beta_max: float = 30.0,
    beta_inc_factor: float = 1.01,
    lr0: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    jitter_prob: float = 0.02,
    jitter_scale: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based optimisation of the max‑convolution objective for a fixed
    length `n`.  Returns the best vector found during the allotted time and
    its associated score (the value of the evaluation function).
    """
    target_sum = np.sqrt(2.0 * n)                         # Σa_i = √(2n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    m = np.zeros_like(a)                                 # first moment
    v = np.zeros_like(a)                                 # second moment
    beta = beta0                                          # current soft‑max temperature

    best_local = a.copy()
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ----- convolution of a with itself and current score ----------
        conv = _full_convolve_fft(a, a)
        max_val = conv.max()
        sum_a = a.sum()
        score = 2.0 * n * max_val / (sum_a ** 2)

        if score < best_score:
            best_score = score
            best_local = a.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")

        # ----- soft‑max weights over the convolution entries -------------
        # weight for the global maximum is 1, others decay with β
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of Σ_k w_k·conv_k with respect to a -------------
        #   grad_i = 2 * Σ_k w_k * a_{k‑i}
        w_rev = w[::-1]                                   # reversed weight vector
        conv_rev = _full_convolve_fft(a, w_rev)           # length = 3n‑2
        idx = 2 * n - 2 - np.arange(n)                    # indices picking the right slice
        grad = 2.0 * conv_rev[idx]

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)                             # slowly decaying step size
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional stochastic shake‑up ----------------------------
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * (target_sum * jitter_scale / n)
            a_new = a_new + noise

        # ----- enforce feasibility (non‑negative + simplex) -------------
        a = _project_onto_simplex(a_new, target_sum)

        # ----- increase soft‑max temperature -------------------------------
        beta = min(beta_max, beta * beta_inc_factor)

        iteration += 1

    return best_local, best_score
