"""
A hybrid optimiser for the step‑function problem.

Strategy
--------
1. **Initial shape** – a power law a_i ∝ (i+1)^‑p (p≈0.5) is already close to the
   optimal constant autocorrelation.  The vector is projected onto the simplex
   {x_i≥0, Σx_i = √(2n)} so that the evaluation function equals the
   maximal convolution value.

2. **LP refinement** – given the current vector f we solve the linear program

        maximise   Σ b_j
        subject to  b_j ≥ 0
                    (f * b)_k ≤ M   for all k
   where M = max(f*f).  The solution b has a larger ℓ₁‑norm than f while keeping
   the convolution bound unchanged.  After scaling b to the required sum we
   mix it with f (f←(1‑t)f + t·b).  Repeating this step (a few inner
   iterations) drives down the maximal convolution markedly.

3. **Fine‑tuning** – a short projected (sub‑)gradient descent on the active
   constraints (the indices whose convolution is close to the current maximum)
   is applied to polish the result.

4. **Restarts** – many independent runs with random lengths (800–3500) and
   random exponents are performed within the time budget.  The best sequence
   seen is kept globally and occasionally used as warm‑start for new runs.

The routine respects the 1000 s limit and returns a plain Python list of
non‑negative floats.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – full linear convolution using a power‑of‑2 FFT
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a using an FFT of size power‑of‑2."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)           # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper 3 – gradient of Σ_{k∈active} (f*f)_k w.r.t. f
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of the sum of the active convolution entries."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]          # ∂/∂a_i Σ a_i a_{k-i}
    return grad


# ----------------------------------------------------------------------
# Core: projected gradient descent for a fixed length
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    rel_tol: float = 5e-4,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent on the max‑convolution objective.
    Returns the best vector discovered in the allocated time slice and its
    (approximate) score.
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best vector seen in this run
    conv = _conv_fft(a)
    max_val = conv.max()
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    velocity = np.zeros_like(a)
    step_factor = base_step

    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        grad = _gradient(a, active)

        # momentum (classical)
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake‑up
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92

        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            # occasional progress report
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")

        a = a_new
        step_factor *= 0.9997

    return best_local, best_score


# ----------------------------------------------------------------------
# Helper – indices whose convolution value is close to the current max
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    """Return (max_value, indices) where conv ≥ max – rel_tol·max."""
    max_val = conv.max()
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# LP based refinement (inner loop)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, target_sum: float,
               max_iter: int = 4, mix: float = 0.2,
               time_limit: float = None) -> np.ndarray:
    """
    Repeatedly solves the linear programme

        maximise   Σ b_j
        subject to  b_j ≥ 0 ,
                    (a * b)_k ≤ M    (M = max(a*a))

    The obtained vector b is scaled to the required ℓ₁‑norm and mixed with a.
    Returns the refined a (still satisfying the simplex constraint).
    """
    n = a.shape[0]
    a_cur = a.copy()
    for it in range(max_iter):
        if time_limit is not None and time.time() > time_limit:
            break

        conv = _conv_fft(a_cur)
        M = conv.max()

        # –‑ build and solve the LP –‑
        f_param = cp.Parameter(n, nonneg=True)
        f_param.value = a_cur
        b = cp.Variable(n, nonneg=True)
        constraints = [cp.conv(f_param, b) <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       verbose=False, max_iters=2000)
        except Exception:                     # pragma: no cover
            break

        if b.value is None:
            break
        sum_b = np.sum(b.value)
        if sum_b <= 0.0:
            break

        # –‑ mix –‑
        g = (target_sum / sum_b) * b.value
        a_cur = (1.0 - mix) * a_cur + mix * g
        mix *= 0.85                           # decay mixing weight

    return a_cur


# ----------------------------------------------------------------------

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
