def _mirror_descent(a, target_sum, time_limit):
    # a is initial vector of length n.
    # We'll treat w = a / target_sum (so sum w = 1).
    w = a / target_sum
    best_w, best_score = w.copy(), float(evaluate_sequence((w*target_sum).tolist()))
    t0 = time.time()
    iteration = 0
    while time.time() < t0 + time_limit:
        conv = np.convolve(w, w)  # note conv of w (since sum of w =1) -> sum conv = 1
        max_val = conv.max()
        # threshold for near max
        eps = 1e-8
        idxs = np.where(conv >= max_val - eps)[0]
        # compute gradient for each idx and average
        grad = np.zeros_like(w)
        for idx in idxs:
            low = max(0, idx - (len(w)-1))
            high = min(len(w)-1, idx)
            idx_range = np.arange(low, high+1, dtype=int)
            partner = idx - idx_range
            grad[idx_range] += 2 * w[partner]  # gradient w.r.t w for that idx
        grad = grad / len(idxs) if len(idxs)>0 else grad
        # step size schedule
        eta = 0.2 / (1 + 0.01 * iteration)  # diminishing
        # exponentiated update: w = w * exp(-eta*grad)
        # but grad may be large; scale
        # use negative subgradient direction (increase to reduce conv)
        # Actually max conv increasing direction is grad; we want to move opposite direction.
        exp_factor = np.exp(-eta * grad)
        w = w * exp_factor
        # renormalize to sum 1
        w_sum = w.sum()
        if w_sum <= 0:
            # fallback to uniform
            w = np.full_like(w, 1.0 / len(w))
        else:
            w = w / w_sum
        # evaluate
        a_candidate = w * target_sum
        try:
            score = float(evaluate_sequence(a_candidate.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_w = w.copy()
            print(f"[{time.time():.2f}] Mirror descent new best {score:.6f}")
        # occasional random perturbation
        if np.random.rand() < 0.01:
            sigma = 0.01 * target_sum / len(w)
            w = _mutate((w*target_sum), target_sum, sigma) / target_sum
        iteration += 1
    return best_w*target_sum, best_score

"""
Hybrid optimizer for the step‑function evaluation problem.

Main ideas
----------
* Multi‑scale search – start from a “U‑shaped” heuristic, then repeatedly
  resize to a new length and refine.
* Phase 1 – Mirror‑descent (exponentiated gradient) on the simplex using a
  sub‑gradient of the maximal convolution entry.  The KL‑mirror keeps the
  iterates strictly non‑negative and automatically respects the sum constraint.
* Phase 2 – Projected sub‑gradient descent (the same routine used in the
  reference implementation) to polish the result.
* Phase 3 – An optional LP‑refinement (if cvxpy is available) exactly
  implements the update described in the paper sketch.
* Random mutations / occasional annealing steps give the algorithm a modest
  global‑exploration capability.

The routine runs for at most ``budget_s`` seconds (default 1000 s) and
returns the best sequence it has found.
"""

import time
import math
import numpy as np

# -----------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# -----------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -----------------------------------------------------------
# Gradient of a single convolution entry (used by both optimisers)
# -----------------------------------------------------------
def _grad_for_idx(a: np.ndarray, idx: int) -> np.ndarray:
    """Gradient of conv(a,a)[idx] w.r.t. a."""
    n = a.shape[0]
    low = max(0, idx - (n - 1))
    high = min(n - 1, idx)
    i = np.arange(low, high + 1, dtype=int)
    j = idx - i
    grad = np.zeros_like(a)
    grad[i] = 2.0 * a[j]
    return grad


# -----------------------------------------------------------
# Gaussian mutation + simplex projection
# -----------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Add Gaussian noise to a, clip to ≥0 and project onto the simplex."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# -----------------------------------------------------------
# Local projected gradient descent (sub‑gradient of the maximal entry)
# -----------------------------------------------------------
def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.05,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()

    while time.time() < t0 + time_limit:
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # sub‑gradient of the maximal entry
        grad = _grad_for_idx(a, max_idx)

        # make the direction sum‑zero (to stay on the simplex)
        mean_g = np.mean(grad)
        d = -grad + mean_g

        # adaptive step size
        max_abs = np.max(np.abs(d))
        if max_abs == 0:
            max_abs = 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})"
            )

        # occasional shake‑up
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
            step *= 0.9
        else:
            a = a_new

        # simple step decay
        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score


# -----------------------------------------------------------
# Mirror‑descent on the simplex using a KL‑divergence (exponentiated gradient)
# -----------------------------------------------------------
def _mirror_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Run exponentiated‑gradient descent on the simplex.
    Returns the best sequence seen and its evaluation score."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    w = a / target_sum                     # w lives on the probability simplex

    best_w = w.copy()
    try:
        best_score = float(evaluate_sequence((w * target_sum).tolist()))
    except Exception:
        best_score = math.inf

    t0 = time.time()
    it = 0
    while time.time() < t0 + time_limit:
        conv = np.convolve(w, w)
        max_val = conv.max()
        eps = max(1e-12, max_val * 1e-8)
        active = np.where(conv >= max_val - eps)[0]

        # average sub‑gradient over the active set
        grad = np.zeros_like(w)
        for idx in active:
            grad += _grad_for_idx(w, idx)
        if len(active) > 0:
            grad /= len(active)

        # decreasing learning rate
        eta = 0.5 / math.sqrt(it + 1)

        # multiplicative update (KL‑mirror)
        w = w * np.exp(-eta * grad)

        # renormalise – this automatically satisfies the sum‑constraint
        s = w.sum()
        if s == 0 or not np.isfinite(s):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / s

        # evaluate
        cand = (w * target_sum).tolist()
        try:
            score = float(evaluate_sequence(cand))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_w = w.copy()
            print(
                f"[{time.time():.2f}] mirror descent new best = {score:.6f} (n={len(w)})"
            )

        # tiny random perturbation for escaping shallow plateaus
        if np.random.rand() < 0.01:
            sigma = 0.01 * target_sum / len(w)
            a_noisy = _mutate((w * target_sum), target_sum, sigma)
            w = a_noisy / target_sum

        it += 1

    return (best_w * target_sum), best_score

# Optional LP refinement (mirrors the paper sketch)
# -----------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve  max Σ g_j   s.t.  g≥0 ,  f*g ≤ M (linear program)."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """Perform a few LP updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())

        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break

        scale = target_sum / np.sum(g0)
        g = g0 * scale

        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})"
            )
    return best_seq, best_score


# -----------------------------------------------------------
# Simple resizing (linear interpolation / down‑sampling)
# -----------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# -----------------------------------------------------------
# Heuristic initial shape – a U‑shaped distribution
# -----------------------------------------------------------
def _initial_u_shape(n: int) -> np.ndarray:
    """
    Return a non‑negative vector of length n with Σ = √(2 n).
    The shape behaves like 1/√(x·(1‑x)) (large at the ends, small in the centre).
    """
    target = math.sqrt(2.0 * n)
    # shift by 0.5 to avoid a zero at the extreme ends
    i = np.arange(n) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    # tiny safeguard against numerical issues
    w = np.maximum(w, 1e-12)
    w = w / w.sum() * target
    return w
