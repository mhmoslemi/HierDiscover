def _conv_fft(x, y):
    n = len(x); m = len(y)
    size = 1 << ((n+m-1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n+m-1]

"""
Multi‑start smoothed‑max gradient optimizer for the step‑function bound

The routine works on probability vectors (sum = 1).  The objective to
minimise is

    2·n·max_k (a * a)[k]               (scale‑invariant)

where ``*`` is the discrete convolution.  The algorithm

* starts from a variety of initial shapes (uniform, exponential,
  random Dirichlet, the previous best if available);
* runs a smoothed‑max descent (soft‑max with a slowly increasing
  temperature) using FFT‑based convolutions and a simple momentum scheme;
* repeatedly projects onto the probability simplex;
* optionally refines the best vector with a short linear‑programming
  step (the same “LP‑step’’ used by the SOTA baseline).

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex with total mass `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – fast for n up to several thousand)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (real FFT, result length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b (real FFT)."""
    n, m = a.shape[0], b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n + m - 1]

# Fast objective (scale‑invariant, sum = 1)
# ----------------------------------------------------------------------

    """Return 2·n·max(conv(a,a)) for a probability vector a."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()

    a = a0.copy()
    a = a / a.sum()                        # safety normalisation
    n = a.shape[0]

    # momentum
    m = np.zeros_like(a)
    beta = 0.9

    # step‑size schedule
    base_step = 1.0

    # soft‑max temperature schedule
    lam = 0.5
    lam_max = 12.0
    lam_inc = (lam_max - lam) / 5000.0     # very slow increase

    it = 0
    while time.time() < deadline:
        # convolution of the current iterate
        conv = _conv_self(a)
        maxb = conv.max()

        # update global best
        _maybe_update_best(a, best_info)

        # increase the temperature slowly (makes the soft‑max sharper)
        if lam < lam_max:
            lam = min(lam_max, lam + lam_inc)

        # -------- soft‑max weights ----------
        # shift to avoid overflow
        conv_shift = conv - conv.max()
        w = np.exp(lam * conv_shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum  # now Σ w = 1

        # -------- gradient of the smooth max ----------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        # which is 2 * (a * w_rev)[2·n‑2‑i] (see derivation in analysis)
        w_rev = w[::-1]
        grad_full = _conv(a, w_rev)               # length = 3·n‑2
        idx = 2 * n - 2 - np.arange(n)            # pick the needed slice
        grad = 2.0 * grad_full[idx]

        # -------- momentum update ----------
        m = beta * m + (1.0 - beta) * grad

        # -------- step ----------
        step = base_step / math.sqrt(it + 1)
        a_new = a - step * m
        a_new = _project_onto_simplex(a_new, 1.0)

        a = a_new
        it += 1

        # stop early if we have already hit the target bound
        if best_info[1] is not None and best_info[1] < 1.503:
            break

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The function respects the given time budget
    and returns the best list of floats found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # best_info = [best_sequence (np.ndarray), best_score (float)]
    best_info = [None, None]

    # --------------------------------------------------------------
    # 1️⃣  Seed with the previously found construction if available
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is injected by the evaluation harness
        prev = np.array(height_sequence_1, dtype=float)        # type: ignore
        prev = prev / prev.sum()
        _maybe_update_best(prev, best_info)
        _optimize_vector(prev, deadline, best_info)
    except Exception:
        # no previous sequence – simply continue
        pass

    # --------------------------------------------------------------
    # 2️⃣  Systematic exploration of several lengths and families
    # --------------------------------------------------------------
    lengths = list(range(300, 1701, 200)) + [1800, 2000, 2500, 3000]
    # shuffle a little to get varied start points
    np.random.shuffle(lengths)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- uniform -------------------------------------------------
        a_uni = np.full(n, 1.0 / n)
        _maybe_update_best(a_uni, best_info)
        _optimize_vector(a_uni, deadline, best_info)

        # ----- exponential decay (parameterised) --------------------
        for alpha in np.logspace(-4, -0.5, num=6):
            a_exp = np.exp(-alpha * np.arange(n, dtype=float))
            a_exp = a_exp / a_exp.sum()
            _maybe_update_best(a_exp, best_info)
            _optimize_vector(a_exp, deadline, best_info)

        # ----- random Dirichlet (flat) -------------------------------
        for _ in range(2):
            a_rand = np.random.rand(n)
            a_rand = a_rand / a_rand.sum()
            _maybe_update_best(a_rand, best_info)
            _optimize_vector(a_rand, deadline, best_info)

        # early exit if already below the target
        if best_info[1] is not None and best_info[1] < 1.503:
            break

    # --------------------------------------------------------------
    # 3️⃣  Random restarts until time runs out
    # --------------------------------------------------------------
    while time.time() < deadline:
        n = np.random.randint(400, 3500)           # moderately large
        # start from a random Dirichlet vector
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
        _maybe_update_best(a0, best_info)
        _optimize_vector(a0, deadline, best_info)

        if best_info[1] is not None and best_info[1] < 1.503:
            break

    # --------------------------------------------------------------
    # 4️⃣  Optional LP refinement (as in the SOTA description)
    # --------------------------------------------------------------
    if linprog is not None and best_info[0] is not None and time.time() < deadline:
        a = best_info[0].copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        # Perform at most a few LP refinements – each one costs O(n²)
        for _ in range(3):
            conv = _conv_self(a)
            M = conv.max()
            rows = 2 * n - 1

            # Build matrix A for (a * g)_k ≤ M
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                cols = np.arange(low, high + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),        # maximise Σ g_j  → minimise –Σ g_j
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                break

            g = res.x
            g = g / g.sum() * target   # keep the same L1–norm as the baseline

            # convex combination step (t chosen modestly)
            t = 0.2
            a = (1.0 - t) * a + t * g

            _maybe_update_best(a, best_info)

            if best_info[1] is not None and best_info[1] < 1.503:
                break
            if time.time() + 0.5 > deadline:
                break

    # --------------------------------------------------------------
    # 5️⃣  Fallback – uniform vector if nothing else succeeded
    # --------------------------------------------------------------
    if best_info[0] is None:
        n = 1000
        a = np.full(n, 1.0 / n)
        best_info[0] = a
        best_info[1] = _fast_score(a)

    # --------------------------------------------------------------
    # 6️⃣  Return the best sequence (scaled to sqrt(2·n) for familiarity)
    # --------------------------------------------------------------
    best_vec = best_info[0]
    n = best_vec.shape[0]
    scale = math.sqrt(2.0 * n)   # the classic normalisation used by the baseline
    best_vec = best_vec * scale
    return best_vec.tolist()
