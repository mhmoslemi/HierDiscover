"""
Propose a step‑function sequence with a low objective value.

Algorithm
---------
1. **Initial candidates** – geometric progressions (a_i ∝ r^i) with a
   range of decay rates, a few random Dirichlet vectors and, if
   available, the previous best construction `height_sequence_1`.
   All candidates are projected onto the probability simplex
   (∑a_i = 1, a_i ≥ 0) and the best of them is kept.

2. **Adam optimisation of a smooth‑max surrogate** – we minimise
   φ_λ(a) = (1/λ)·log ∑_k exp(λ·(a*a)[k]) whose gradient is
   2·(w ★ a) with w the soft‑max over the convolution entries.
   Adam (adaptive moment estimation) is used for the updates,
   together with projection onto the simplex after each step.
   The smoothing parameter λ is increased slowly (continuation) so
   that the optimisation gradually focuses on the true max.

3. **Local random perturbations** – after the Adam phase we spend the
   remaining time performing cheap hill‑climbing: a small Gaussian
   perturbation is added to the current best, the result is projected
   back onto the simplex, and the exact objective is evaluated.
   Improvements are kept; the process repeats until the time budget
   expires.

All heavy work (convolution and correlation) is performed with FFTs,
so each iteration is O(n log n).  The routine respects the 1000 s
budget, prints occasional progress, and finally returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution using FFT
def _full_convolution(a, size=None):
    """Return a*a (linear convolution, length 2·len(a)-1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective as used by evaluate_sequence (sum of a assumed >0)
def _objective(a):
    """2·n·max(convolution) / (sum(a)^2); a must be non‑negative."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a > 0 by construction
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate φ_λ
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Compute ∇φ_λ(a) = 2·(w ★ a) where w is the soft‑max over conv
    with temperature λ.
    """
    n = len(a)
    # soft‑max weights (stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w ★ a (linear)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice corresponding to indices 0 … n‑1 of the gradient
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the time budget, prints progress and returns
    the best sequence found as a Python list.
    """
    # ------------------------------------------------------------------
    # Setup and time bookkeeping
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a safety margin

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction is available, reuse its
    # length; otherwise use a moderate size (2000‑4000) that balances
    # expressive power and speed.
    prev = globals().get("height_sequence_1", None)
    if prev is not None and isinstance(prev, (list, tuple, np.ndarray)):
        n = len(prev)
        prev_seq = np.array(prev, dtype=np.float64)
        prev_seq = np.maximum(prev_seq, 0.0)
        prev_seq /= prev_seq.sum() if prev_seq.sum() > 0 else 1.0
    else:
        # No previous data – pick 2500 as a compromise.
        n = 2500
        prev_seq = None

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (exact objective)
    def eval_obj(seq):
        return _objective(seq)

    # ------------------------------------------------------------------
    # 1. Initial candidate generation
    best_seq = None
    best_val = float("inf")

    # a) geometric progressions
    factors = np.linspace(0.5, 4.0, 32)   # try a wide range of decay rates
    for f in factors:
        if time.time() > deadline:
            break
        # choose a decay r = 1 - eps; eps proportional to factor/n
        eps = f / n
        r = max(0.0, 1.0 - eps)
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            continue
        seq = seq / seq_sum
        val = eval_obj(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] factor={f:5.3f} eps={eps:.2e} obj={val:.6f}")

    # b) random Dirichlet vectors
    for _ in range(20):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = eval_obj(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init rand] obj={val:.6f}")

    # c) previous best if present
    if prev_seq is not None:
        val = eval_obj(prev_seq)
        if val < best_val:
            best_val, best_seq = val, prev_seq.copy()
            print(f"[Init prev] obj={val:.6f}")

    # Guard against the unlikely case where nothing was generated
    if best_seq is None:
        # fall back to uniform
        best_seq = np.full(n, 1.0 / n)
        best_val = eval_obj(best_seq)
        print(f"[Init fallback] uniform obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation of smooth‑max surrogate
    seq = best_seq.copy()               # current iterate
    lam = 1.0                           # initial smoothing parameter
    lam_factor = 1.015                  # increase λ slowly
    lam_max = 80.0                      # stop increasing after this
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    alpha = 0.03                       # base learning rate (adjustable)
    m = np.zeros_like(seq)              # first moment
    v = np.zeros_like(seq)              # second moment
    t = 0                               # iteration counter

    # FFT size for correlation (w ★ a)
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # >= len(w)+n-1
    # FFT size for self‑convolution (a*a)
    size_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))

    last_improve = time.time()
    it = 0
    print_interval = 2000   # print progress every this many iterations

    while time.time() < deadline:
        # --- compute convolution and gradient of smooth‑max ---
        conv = _full_convolution(seq, size=size_conv)
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_corr)

        # --- Adam update ---
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Project back onto the simplex
        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # Evaluate exact objective
        val = eval_obj(new_seq)

        # Keep the best ever seen
        if val < best_val:
            best_val, best_seq = val, new_seq.copy()
            last_improve = time.time()
            print(f"[Adam iter {it:06d}] new best obj={val:.6f}")

        # Periodic logging
        if it % print_interval == 0:
            print(f"[Adam iter {it:06d}] cur obj={val:.6f} best={best_val:.6f} λ={lam:.2f}")

        # Update λ (smooth‑max continuity)
        if lam < lam_max:
            lam = min(lam_max, lam * lam_factor)

        # Simple step‑size decay (optional): we keep α fixed but λ growth
        # effectively tightens the landscape.

        seq = new_seq
        it += 1

        # If stuck for a while, inject a tiny random shake to help escape
        if time.time() - last_improve > 3.0:
            noise = (np.random.rand(n) - 0.5) * 2e-4   # very small noise
            seq = _project_onto_simplex(seq + noise)
            # reset moments to avoid stale gradients
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[Adam iter {it:06d}] shake injected")

    # ------------------------------------------------------------------
    # 3. Local random perturbations (hill climbing) with remaining time
    #    Simple Gaussian perturbation + projection.
    remaining = deadline - time.time()
    if remaining > 5.0:   # only if we have a few seconds left
        print(f"[Perturb] switching to local search, time left {remaining:.1f}s")
    else:
        remaining = 0.0

    # Use the best sequence from Adam as the starting point
    seq = best_seq.copy()
    improvement = True
    last_improve = time.time()
    perturb_iter = 0

    while time.time() < deadline:
        # propose a small random move
        noise = np.random.normal(scale=0.0008, size=n)
        cand = _project_onto_simplex(seq + noise)
        val = eval_obj(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            seq = cand
            last_improve = time.time()
            print(f"[Perturb {perturb_iter:05d}] improved to {val:.6f}")
        else:
            # occasional acceptance of non‑improving moves (simulated annealing)
            # with a very low probability that decays over time.
            elapsed = time.time() - last_improve
            prob = np.exp(-elapsed) * 0.01
            if np.random.rand() < prob:
                seq = cand
        perturb_iter += 1

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
