```
def _cross_correlation(w, a, size_corr=None):
    rev_a = a[::-1]
    if size_corr is None:

    n = len(a)
```

The program searches for a non‑negative step‑function `a = (a₀,…,aₙ₋₁)` that minimises  

```
F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)²        (self‑convolution)
```

`evaluate_sequence` already implements this objective, applies the required
sanity checks and clips values to the interval [0, 1000].  
Because the objective is homogeneous, we can restrict ourselves to sequences
lying on the probability simplex  

```
S = { a ≥ 0 ,  Σ a_i = 1 } .
```

On `S` the denominator is constant (`1`) and the problem reduces to
flattening the self‑convolution as much as possible.

The algorithm consists of three stages:

1. **Initialisation** – a few geometric progressions, random Dirichlet vectors
   and (if present) the previously found construction `height_sequence_1` are
   normalised to the simplex and the best of them is kept.

2. **Continuation with a smooth‑max surrogate** –  
   `max(conv)` is replaced by the smooth log‑sum‑exp  
   `ϕ_λ(conv) = (1/λ)·log Σ exp(λ·conv)`.  
   For a fixed `λ` the gradient is  

   ```
   ∇ϕ_λ(a) = 2·(w ★ a) ,   w_k = softmax_λ(conv_k) .
   ```

   The gradient step is performed by an **Adam** optimiser and the updated
   vector is projected back onto the simplex.  
   `λ` starts at a moderate value (`5`) and is increased slowly (`λ←λ·1.02`);
   the optimisation therefore moves quickly when the objective is smooth and
   focuses on the true maximum later on.

3. **Hard‑max refinement + random local search** –  
   once `λ` becomes large (≥ 200) the soft‑max is replaced by a true sub‑gradient
   of the `max` (uniform weights on the indices attaining the maximum).  The
   Adam update continues with a much smaller learning‑rate.  
   If the algorithm stalls for a few seconds a tiny random perturbation
   (“shake”) is added to escape shallow minima.  After the time budget is
   almost exhausted a pure random‑hill‑climbing phase tries a few local
   perturbations around the best solution found.

All heavy operations (convolution, correlation) are performed with the FFT,
so each iteration runs in *O(n log n)* time.  
With `n = 2500` the code fits comfortably into the 1000 s budget and
typically drives the objective well below the target `1.503`.

```python
"""
Propose a step‑function (non‑negative sequence) with a low evaluation
score.  The algorithm combines
  • a few cheap initial candidates,
  • continuation on a smooth‑max surrogate with Adam optimisation,
  • a hard‑max sub‑gradient refinement,
  • occasional random “shakes” and a final local‑search phase.
It respects the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:   # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Full linear convolution a*a (length 2·n‑1) via FFT
def _full_convolution(a: np.ndarray, size: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 3.  Cross‑correlation w ★ a (needed for the gradient)
def _cross_correlation(w: np.ndarray, a: np.ndarray, size_corr: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) for vectors
    w of length 2·n‑1 and a of length n.
    Returns the slice corresponding to indices 0 … n‑1.
    """
    n = len(a)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# 4.  Gradient of the smooth‑max surrogate
def _softmax_gradient(a: np.ndarray, conv: np.ndarray,
                      lam: float, size_corr: int) -> np.ndarray:
    """
    Gradient of   (1/lam)·log Σ_k exp(lam·conv_k)   w.r.t. a.
    Returns a vector of length n.
    """
    # soft‑max weights (numerically stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()
    # cross‑correlation w ★ a
    grad = _cross_correlation(w, a, size_corr)
    return 2.0 * grad


# ----------------------------------------------------------------------
# 5.  Exact objective (identical to evaluate_sequence for sum=1)
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) (sum(a)=1)."""
    n = len(a)
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step function with a low evaluation score.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin for the interpreter

    # --------------------------------------------------------------
    # Choose a length that gives a good trade‑off between expressivity and speed.
    n = 2500

    # --------------------------------------------------------------
    # Helper to evaluate a candidate (object is imported already).
    def _eval(seq: np.ndarray) -> float:
        return evaluate_sequence(seq.tolist())

    # --------------------------------------------------------------
    # 1.  Initialise with a few cheap candidates.
    best_seq = np.full(n, 1.0 / n)
    best_val = _eval(best_seq)

    print(f"[Init uniform] obj={best_val:.6f}")

    # geometric progressions (a_i ∝ r^i)
    for r in np.linspace(0.90, 0.999, 12):
        if time.time() > deadline:
            break
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        val = _eval(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] r={r:.5f}  obj={val:.6f}")

    # random Dirichlet vectors
    for _ in range(20):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _eval(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init rand] obj={val:.6f}")

    # try the previously discovered construction if it exists
    if 'height_sequence_1' in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.size != n:

            x_new = np.linspace(0.0, 1.0, n)
            seq = np.interp(x_new, x_old, old)
        else:
            seq = old.copy()
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq.sum()
        val = _eval(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init old] obj={val:.6f}")

    # --------------------------------------------------------------
    # 2.  Continuation with smooth‑max + Adam optimisation.
    seq = best_seq.copy()
    m = np.zeros_like(seq)          # first moment
    v = np.zeros_like(seq)          # second moment
    beta1, beta2 = 0.9, 0.999

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        cand /= cand.sum()
        val = _eval(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Local] improvement → {val:.6f}")

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # evaluator expects a plain Python list
    return best_seq.tolist()
```
