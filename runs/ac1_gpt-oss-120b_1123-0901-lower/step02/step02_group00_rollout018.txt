"""
Propose a step function that minimizes evaluate_sequence using a
combined Adam optimisation (sub‑gradient) warm‑start and a
guided hill‑climbing / simulated‑annealing refinement.  The Adam phase
produces a good smooth local minimum, then we repeatedly move a small
amount of mass from indices that contribute most to the current maximal
convolution value towards low‑mass indices, accepting only improvements
(and occasionally adding jitter to escape plateaus).  The algorithm
operates directly on the max‑convolution via FFT, respects the sum
constraint Σa_i = √(2n) and non‑negativity, and runs until the time
budget expires.
"""

import time
import math
import numpy as np

def _conv_max_fft(a: np.ndarray):
    """Return (max convolution value, index of max, full convolution) for vector a."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)
    conv = conv[: 2 * n - 1]
    k_max = int(conv.argmax())
    return float(conv.max()), k_max, conv

def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """Average reversed vector a_{k-p} over all active shift indices."""
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active_idxs.size)
    return rev

def _optimize_on_length(n: int, init_a: np.ndarray,
                        best_seq: np.ndarray, best_score: float,
                        deadline: float):
    """Adam optimiser (sub‑gradient) for a fixed length."""
    target_sum = math.sqrt(2.0 * n)
    a = init_a.copy()
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    while time.time() < deadline:
        M, k_max, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_score = M  # because the normalisation makes the objective equal to max convolution
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best score = {best_score:.6f}")

    return best_seq, best_score

def _guided_move(a: np.ndarray, kmax: int, target_sum: float):
    """Perform a single guided mass‑transfer move based on the current maximal shift."""
    n = a.shape[0]
    # contributions to the current maximal convolution
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    contrib = np.zeros(n)
    contrib[mask] = a[idx[mask]] * a[j[mask]]
    # pick one of the top contributors (randomised)
    top = min(5, np.count_nonzero(contrib))
    if top > 0:
        # indices of the top values
        idx_top = np.argpartition(contrib, -top)[-top:]
        i = np.random.choice(idx_top)
    else:
        i = np.random.randint(n)

    # pick a target index with small current value (sample a few candidates)
    sample = np.random.choice(n, size=min(10, n), replace=False)
    # avoid picking i itself
    sample = sample[sample != i]
    if sample.size == 0:
        t = (i + 1) % n
    else:
        t = sample[np.argmin(a[sample])]
    # size of transferred mass – a few percent of the source entry
    eps = a[i] * (0.01 + 0.01 * np.random.rand())
    eps = min(eps, a[i])  # cannot remove more than we have
    a_new = a.copy()
    a_new[i] -= eps
    a_new[t] += eps
    return a_new

    best_seq = np.maximum(best_seq, 0.0)

        M_new, kmax_new, _ = _conv_max_fft(a_new)
        new_score = M_new

        # deterministic improvement, otherwise stochastic SA acceptance
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            delta = new_score - cur_score
            prob = np.exp(-delta / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            kmax = kmax_new
            no_improve = 0
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[refine] iter {iteration:7d} new best score = {best_score:.6f}")
        else:
            no_improve += 1

        # temperature decay
        T = max(min_T, T * decay)

        # occasional jitter to jump out of plateaus
        if no_improve >= jitter_thr:
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            M, kmax, _ = _conv_max_fft(a)
            cur_score = M
            no_improve = 0
            print(f"[jitter] iter {iteration:7d} re‑jittered")

        # periodic status report
        if iteration % 50000 == 0:
            print(f"[status] iter {iteration:7d} cur {cur_score:.6f} best {best_score:.6f} T {T:.3e}")

    # ------------------------------------------------------------------
    # Final normalisation & evaluation
    best_seq = best_seq * (target_sum / best_seq.sum())
    best_seq = np.maximum(best_seq, 0.0)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] final score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
