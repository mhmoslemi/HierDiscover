"""
Propose a height sequence that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

For a normalised sequence (sum = sqrt(2 n)) this is exactly the maximum entry
of the autocorrelation `conv(a, a)`.  The algorithm consists of three stages:

1.  A cheap sweep over a few parametric families (exponential, linear,
    quadratic, triangular) for several lengths.  The best candidate is kept.

2.  A smooth‑max gradient descent (Adam) on the current best candidate.
    The max is approximated by a log‑sum‑exp with a temperature β that is
    increased gradually; this yields a well‑behaved gradient and avoids getting
    stuck in poor local minima.  Each gradient step is projected onto the
    simplex {x ≥ 0, Σx = √(2 n)}.

3.  Random restarts: new lengths and random exponential starts are optimised
    with the same optimiser for a limited number of iterations.  If a better
    sequence is found it replaces the global best.

Optionally a tiny linear‑programming refinement (SciPy `linprog`) is applied
if the library is available and time permits.

The routine respects the given time budget (default 1000 s) and returns a
plain list of non‑negative floats ready for the external `evaluate_sequence`
call.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex Σx = s , x ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _conv_linear(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    nx = x.shape[0]
    ny = y.shape[0]
    size = 1 << ((nx + ny - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: nx + ny - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (full length)."""
    return _conv_linear(a, a)


# ----------------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) w.r.t. a
# ----------------------------------------------------------------------
def _compute_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a weight vector w (same length as a*a) that sums to one,
    return the gradient of Σ_k w_k (a*a)_k.
    The formula reduces to 2 * (a ⊛ w_rev) restricted to n entries.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse for the correct correlation
    conv = _conv_linear(a, w_rev)       # length = 3n‑2
    start = 2 * n - 2
    end = n - 2
    # slice with negative step gives the required n elements
    grad_part = conv[start:end:-1]
    return 2.0 * grad_part


# ----------------------------------------------------------------------
# Safe wrapper around the external evaluator
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    try:
        return evaluate_sequence(seq)   # provided by the harness
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Shape factories (non‑normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – quick parametric sweep
    # ------------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200, 1500, 1800]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-5, -0.2, num=12):
            a = _shape_exp(n, alpha)
            a = _project_onto_simplex(a, target)
            sc = _evaluate(a.tolist())
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")

        # reversed exponential
        for alpha in np.logspace(-5, -0.2, num=12):
            a = _shape_exp_rev(n, alpha)
            a = _project_onto_simplex(a, target)
            sc = _evaluate(a.tolist())
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=10):
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = _project_onto_simplex(a, target)
            sc = _evaluate(a.tolist())
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=10):
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = _project_onto_simplex(a, target)
            sc = _evaluate(a.tolist())
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")

        # symmetric triangular
        a = _shape_triangular(n)
        a = _project_onto_simplex(a, target)
        sc = _evaluate(a.tolist())
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # If a global construction is already available, use it as a seed
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        cand = np.array(height_sequence_1, dtype=float)
        # adjust length if needed (truncate / pad)
        n0 = cand.shape[0]
        target0 = math.sqrt(2.0 * n0)
        cand = _project_onto_simplex(cand, target0)
        sc = _evaluate(cand.tolist())
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.2f}] New best (global) = {best_score:.6f} (len={n0})")

    # ------------------------------------------------------------------
    # Optimiser parameters (Adam)
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    init_lr = 0.5
    beta_start = 0.5
    beta_max = 200.0
    beta_mul = 1.001

    # ------------------------------------------------------------------
    # Helper to run the Adam optimiser on a given start vector
    # ------------------------------------------------------------------
    def _adam_optimize(a0: np.ndarray):
        nonlocal best_seq, best_score
        a = a0.copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        beta = beta_start

        # run until time runs out or convergence is reached
        while time.time() < deadline:
            conv = _conv_self(a)
            max_c = conv.max()
            # update global best
            if max_c < best_score:
                best_score = max_c
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")
                if best_score < 1.503:
                    return

            # smoothed max weights (log‑sum‑exp)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w = w / w_sum

            # gradient of weighted sum
            grad = _compute_gradient(a, w)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_onto_simplex(a, target)

            # increase smoothing temperature slowly
            beta = min(beta * beta_mul, beta_max)

            # occasional exact evaluation (cheap because conv already computed)
            if t % 20 == 0:
                if max_c < best_score:
                    best_score = max_c
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] New best (periodic) = {best_score:.6f} (len={n})")
                    if best_score < 1.503:
                        return

    # ------------------------------------------------------------------
    # Phase 2 – optimise the current best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        _adam_optimize(best_seq)

    # ------------------------------------------------------------------
    # Phase 3 – random restarts with limited budget per run
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a random length (moderate to keep each run cheap)
        n = np.random.randint(500, 2001)
        target = math.sqrt(2.0 * n)

        # initialise with a random exponential distribution (broadly useful)
        a0 = np.random.exponential(scale=1.0, size=n)
        a0 = _project_onto_simplex(a0, target)

        # run a short Adam optimisation (max 800 iterations or until time out)
        max_iters = 800
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_local = 0
        beta = beta_start
        while time.time() < deadline and t_local < max_iters:
            conv = _conv_self(a)
            max_c = conv.max()
            if max_c < best_score:
                best_score = max_c
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (restart) = {best_score:.6f} (len={n})")
                if best_score < 1.503:
                    break

            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w = w / w_sum

            grad = _compute_gradient(a, w)

            t_local += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_local)
            v_hat = v / (1.0 - beta2 ** t_local)
            step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_onto_simplex(a, target)
            beta = min(beta * beta_mul, beta_max)

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 4 – optional LP refinement (if SciPy is available)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            rows = 2 * n - 1
            # Build constraint matrix A where (a * g)_k ≤ M
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                cols = np.arange(lo, hi + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),                # maximise Σ g_j
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                break

            g = res.x
            s = g.sum()
            if s <= 0:
                break
            g = g * (target / s)

            # convex blend
            t = 0.2
            a = (1 - t) * a + t * g

            # evaluate
            sc = _evaluate(a.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best after LP = {best_score:.6f} (len={n})")
                if best_score < 1.503:
                    break

            if time.time() + 0.5 >= deadline:
                break

    # ------------------------------------------------------------------
    # Fallback – uniform sequence if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
