"""
Hybrid non‑convex optimisation for the step‑function problem.

The goal is to minimise
    2 n · max(conv(a,a)) / (sum(a)**2)

which, after fixing Σa = √(2 n), is equivalent to minimising the
maximum entry of the auto‑convolution.  The algorithm combines:

* several simple parametric initialisations (uniform, triangular,
  sine, symmetric exponential, inverse‑sqrt);
* a projected sub‑gradient descent with heavy‑ball momentum;
* occasional random “shakes” for escaping shallow minima;
* a final hill‑climbing pair‑wise mass‑transfer phase.

Different lengths (1200–5000) are tried repeatedly within the time
budget, and the best sequence found is returned.  The code avoids
closures/lambdas as required and works with the provided
`evaluate_sequence` function (which is not called directly – the
objective is computed via FFT for speed).
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# 1) Simple utilities
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a using real FFT."""
    n = a.shape[0]
    # length for efficient FFT (next power of two)
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. `a` using only the active
    (max‑tying) convolution indices.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# 2) Parametric initialisers (all produce non‑negative vectors)
# --------------------------------------------------------------------------- #
def init_random(n: int) -> np.ndarray:
    return np.random.rand(n)


def init_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the centre."""
    if n % 2 == 0:
        half = np.arange(1, n // 2 + 1)
        a = np.concatenate([half, half[::-1]])
    else:
        half = np.arange(1, (n + 1) // 2 + 1)
        a = np.concatenate([half, half[-2::-1]])
    return a.astype(float)


def init_sine(n: int) -> np.ndarray:
    """Half‑sine curve (always positive)."""
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n)
    return a


def init_exponential(n: int) -> np.ndarray:
    """Symmetric exponential decay about the centre."""
    r = np.random.uniform(0.90, 0.99)
    centre = (n - 1) / 2.0
    idx = np.abs(np.arange(n) - centre)
    a = r ** idx
    return a


def init_inv_sqrt(n: int) -> np.ndarray:
    """Inverse‑sqrt shape, larger at the borders."""
    centre = (n - 1) / 2.0
    idx = np.abs(np.arange(n) - centre)
    a = 1.0 / np.sqrt(idx + 1.0)
    return a


_INITIALISERS = [
    init_random,
    init_triangular,
    init_sine,
    init_exponential,
    init_inv_sqrt,
]


# --------------------------------------------------------------------------- #
# 3) One optimisation run (sub‑gradient descent with momentum)
# --------------------------------------------------------------------------- #
def run_descent(
    init_vec: np.ndarray,
    n: int,
    run_deadline: float,
    best_seq_global: np.ndarray | None,
    best_score_global: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a short optimisation run (≈ a few seconds).
    Returns possibly updated best sequence and its max‑convolution value.
    """
    target_sum = np.sqrt(2.0 * n)
    a = project_onto_simplex(init_vec.astype(float), target_sum)

    base_lr = 0.25                # initial learning rate
    decay = 0.99995               # slow decay
    momentum = np.zeros_like(a)
    beta_mom = 0.9                # heavy‑ball momentum factor
    shake_prob = 0.025            # chance of a random perturbation each step
    shake_scale = 0.02            # relative size of the shake

    # Local copies of the global best (updated after the run)
    best_seq = best_seq_global
    best_score = best_score_global

    while time.time() < run_deadline:
        conv = conv_full_fft(a)
        cur_max = conv.max()
        # update global best if improved
        if cur_max < best_score:
            best_score = float(cur_max)
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] New best = {best_score:.6f} (n={n})"
            )

        # active (max‑tying) indices – a tiny tolerance only picks true ties
        eps = 1e-12
        active = np.where(conv >= cur_max - eps)[0]

        grad = gradient_of_active(a, active)
        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        # step size proportional to current objective / gradient magnitude
        eta = base_lr * cur_max / max_grad

        # heavy‑ball momentum update
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        # enforce feasibility
        a_new = np.clip(a_new, 0.0, None)
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional random shake (helps escape shallow basins)
        if np.random.rand() < shake_prob:
            noise = np.random.randn(n) * (shake_scale * target_sum / n)
            a_new = np.clip(a_new + noise, 0.0, None)
            a_new = project_onto_simplex(a_new, target_sum)
            base_lr *= 0.9  # reduce learning rate a bit after a shake

        a = a_new
        base_lr *= decay

    return best_seq, best_score


# --------------------------------------------------------------------------- #
# 4) Final fine‑tuning via pair‑wise mass transfers (simple hill‑climb)
# --------------------------------------------------------------------------- #
def fine_tune(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Small local search that repeatedly moves a tiny amount of mass from
    one index to another if it reduces the maximal convolution entry.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_max = conv_full_fft(a).max()
    # we keep the sum constant, so no projection is needed
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a fraction of a[i] (but never more than a[i])
        max_transfer = min(a[i], 0.01 * cur_max)  # 1 % of current max as a safe bound
        if max_transfer <= 0:
            continue
        delta = np.random.uniform(0, max_transfer)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # numerical safety – clip to non‑negative before evaluation
        cand = np.clip(cand, 0.0, None)
        cand = project_onto_simplex(cand, a.sum())  # re‑project to keep sum exact
        new_max = conv_full_fft(cand).max()
        if new_max < cur_max:
            a = cand
            cur_max = new_max
    return a

# 5) Main entry point required by the grader
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns the best sequence (list of floats) discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin to avoid overshooting the external timeout
    deadline = start_time + budget_s - 0.5

    best_seq = None          # numpy array of the current global best
    best_score = float("inf")

    # ------------------------------------------------------------------- #
    # Helper to decide on a problem size for the next run
    # ------------------------------------------------------------------- #
    def choose_length(remaining: float) -> int:
        """Pick a length; larger lengths become more likely as time passes."""
        # Early in the search we stay near 1500‑2500, later we allow up to 5000
        max_len = 5000 if remaining > 200 else 3000
        return np.random.randint(1200, max_len + 1)

    # ------------------------------------------------------------------- #
    # Main outer loop – many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        remaining = deadline - time.time()
        n = choose_length(remaining)

        # Choose an initialisation strategy.
        # With some probability we reuse the current best (truncated or padded).
        if best_seq is not None and np.random.rand() < 0.30:
            src = best_seq
            if src.size >= n:
                init_vec = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init_vec = np.concatenate([src, pad])
        else:
            init_func = np.random.choice(_INITIALISERS)
            init_vec = init_func(n)

        # Allocate a slice of the remaining budget for this run (≈ 6‑8 s).
        run_deadline = min(deadline, time.time() + np.random.uniform(5.5, 8.0))
        best_seq, best_score = run_descent(
            init_vec,
            n,
            run_deadline,
            best_seq,
            best_score,
        )

        # Stop early if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Final refinement phase (if any time left).
    # ------------------------------------------------------------------- #
    if best_seq is not None and time.time() < deadline:
        refined = fine_tune(best_seq, deadline)
        refined_score = conv_full_fft(refined).max()
        if refined_score < best_score:
            best_seq = refined
            best_score = refined_score

    # ------------------------------------------------------------------- #
    # Fallback – if nothing was discovered (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # expose the result for possible later runs
    global height_sequence_2
    height_sequence_2 = best_seq.tolist()

    return best_seq.tolist()
