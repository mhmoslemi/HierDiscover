```
def build_convolution_matrix(f):
    n = len(f)
    m = 2*n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # each column j corresponds to f[j] added to rows j..j+n-1
    for j in range(n):
        A[j:j+n, j] = f[j]
    return A
```

assistantfinal**Solution Overview**

The score returned by `evaluate_sequence` is  

\[
\text{score}= \frac{2\,n\;\max (f*f)}{(\sum f)^2}\;
\]

(the function caps the values, makes them non‑negative and rescales them to the interval
`[0,1000]`).  
Because the denominator is fixed (`∑f = 1` after we project onto the simplex) the
problem is *equivalent* to:

* minimise the largest entry of the self‑convolution `f*f`  
  subject to `f ≥ 0` and `∑f = 1`.

The objective is convex in the *convolution* but not convex in `f`.  
A classic way to deal with a minimisation of an ∞‑norm is the **conditional‑gradient
(Frank–Wolfe) method**:

* keep the current feasible `f`,
* solve a linear programme that finds the best feasible direction `g` **under the
  current maximal convolution value**,  
* move a little towards `g` and repeat.

The linear programme is tiny because, for a fixed `f`, the map `g ↦ f*g` is **linear**:

```
 maximise   Σ g_j
 subject to  (f * g)[t] ≤ M     for all t
             g_j ≥ 0
```

where `M = max(f*f)`.  
Only the constraints that are *active* (those producing the current maximum) are
necessary – we keep the top 1 % of the convolution entries (at least five rows).
This keeps the LP size modest (≈ 200 constraints for n ≈ 2000) and makes a single
iteration very cheap.

The algorithm therefore consists of

1. **initialisation** – a small pool of hand‑crafted candidates (uniform,
   geometric, arcsine, Dirichlet) and selection of the best one;
2. **outer loop** – until the time budget expires:  

   * compute `M = max(f*f)`;
   * build the *active* constraint matrix;
   * solve the LP with `scipy.optimize.linprog`;
   * obtain a feasible `g` (normalised to `∑g = 1`);
   * perform a back‑tracking line‑search on the step size `τ` and accept the
     move only if the score does not increase;
   * if no progress is made for a while, add a tiny random jitter to escape
     plateaus.

All vectors are projected onto the probability simplex after every modification,
so the denominator of the score stays constant (`∑f = 1`).  
The method is deterministic apart from the random draws which are seeded, thus
reproducible.

The routine below implements exactly this strategy and returns the best sequence
found within the given time budget.

```python
"""
Conditional‑gradient (Frank–Wolfe) optimisation of a non‑negative step function.

*   The objective is 2·n·max(f*f) under the simplex constraint Σf = 1.
*   In each iteration we solve a linear programme
        maximise   Σ g_j
        subject to (f * g)[t] ≤ M   (only the top 1 % of the convolution entries)
                    g_j ≥ 0 .
  The LP gives a feasible direction `g`.  We move a fraction τ towards `g`
  (projected back onto the simplex) and keep the move only if the score
  improves (or the maximal convolution does not increase).

The algorithm mixes large, LP‑driven steps with occasional random jitter
to escape flat plateaus.  It runs until the time budget (≈ 1000 s) expires and
returns the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x ≥ 0, Σ x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a small heterogeneous set of feasible starting sequences."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families (different decay rates)
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        eps = factor / n
        r = max(0.0, 1.0 - eps)
        seq = r ** np.arange(n, dtype=np.float64)
        cand.append(seq / seq.sum())

    # 3. arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def _build_active_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix A of shape (len(active_idx), n) such that
        A @ g = (f * g)[active_idx]
    with the usual convolution convention (zero outside the support).
    """
    n = len(f)
    m = len(active_idx)
    A = np.zeros((m, n), dtype=np.float64)

    for row, t in enumerate(active_idx):
        # all j for which 0 ≤ t-j < n
        j_low = max(0, t - (n - 1))
        j_high = min(n - 1, t)
        # for those columns the coefficient is f[j]
        A[row, j_low:j_high + 1] = f[j_low:j_high + 1]

    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that minimises `evaluate_sequence`
    within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # 1. problem size – a compromise between expressiveness and speed
    n = 2000                                           # keep the same size used previously

    # --------------------------------------------------------------
    # 2. choose a good starting point from a diverse pool
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best score after seeding = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. conditional‑gradient optimisation loop
    seq = best_seq.copy()                     # current iterate (always on the simplex)
    tau = 0.2                                 # initial step size for the convex combination
    tau_min = 1e-4
    tau_decay = 0.9998
    it = 0
    no_improve = 0
    max_no_improve = 20000                    # after this we inject a jitter

    while time.time() < deadline:
        # ----- (a) current convolution and maximal entry -----------------
        conv = np.convolve(seq, seq)          # length 2n‑1
        M = conv.max()
        # short‑circuit: if we already beat the target we can stop early
        current_score = 2 * n * M              # because Σseq = 1
        if current_score < 1.503:
            best_score = current_score
            best_seq = seq.copy()
            print(f"[Early stop] achieved target {best_score:.6f} at iter {it}")
            break

        # ----- (b) select the active constraints (top 1 % of entries) -----
        # we always keep at least five rows
        nb_active = max(5, int(0.01 * conv.size))
        top_idx = np.argpartition(conv, -nb_active)[-nb_active:]
        # sort for nicer matrix (optional)
        active_idx = np.sort(top_idx)

        # ----- (c) build the active constraint matrix --------------------
        A = _build_active_matrix(seq, active_idx)

        # ----- (d) linear programme: maximise Σ g_j under (f*g)[t] ≤ M -----
        c = -np.ones(n, dtype=np.float64)               # linprog minimises; -Σg maximises Σg
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        lp_res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})

        if not lp_res.success:
            # fallback – tiny random move so that the algorithm does not stall
            jitter = rng.normal(scale=0.005, size=n)
            seq = _project_onto_simplex(seq + jitter)
            continue

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0.0:                     # degenerate – random perturbation
            jitter = rng.normal(scale=0.005, size=n)
            seq = _project_onto_simplex(seq + jitter)
            continue

        # ----- (e) direction g (normalised to Σg = 1) --------------------
        g = g0 / sum_g0

        # ----- (f) line‑search on τ (keep the move only if it does not increase the score)
        accepted = False
        trial_tau = tau
        for _ in range(10):
            new_seq = (1.0 - trial_tau) * seq + trial_tau * g
            new_seq = _project_onto_simplex(new_seq)

            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < best_score:
                # improvement – keep it
                best_score = new_score
                best_seq = new_seq.copy()
                seq = new_seq
                accepted = True
                print(f"[Iter {it:05d}] new best = {best_score:.6f} (τ={trial_tau:.5f})")
                no_improve = 0
                break
            # if the score did not improve but the maximal convolution did not increase,
            # we also accept (helps to glide along flat regions)
            new_conv = np.convolve(new_seq, new_seq)
            if new_conv.max() <= M + 1e-12:
                seq = new_seq
                accepted = True
                break
            # otherwise try a smaller step
            trial_tau *= 0.5
            if trial_tau < tau_min:
                break

        if not accepted:
            # no improvement – count stagnation
            no_improve += 1
            # occasionally shake the current solution
            if no_improve > max_no_improve:
                jitter = rng.normal(scale=0.01, size=n)
                seq = _project_onto_simplex(seq + jitter)
                print(f"[Iter {it:05d}] stagnation – jitter applied")
                no_improve = 0
            # keep the old seq (no change) and continue
        else:
            # successful step – slowly decay τ
            tau = max(tau_min, tau * tau_decay)

        it += 1

    print(f"Finished after {it} iterations. Best objective = {best_score:.6f} (length {n})")
    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
```
