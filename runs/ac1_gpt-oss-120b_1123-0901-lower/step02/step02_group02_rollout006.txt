def _alternating_optimize_one_length(n, init, run_deadline, step=0.15, max_iters=200, min_improve=1e-6):
    target_sum = np.sqrt(2.0*n)
    a = init.astype(np.float64)
    # Normalize to target sum
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    conv = _conv_fft(a)
    max_val = conv.max()
    best_score = 2.0 * n * max_val / (a.sum()**2)
    # store t schedule that decays
    t = step
    it = 0
    while time.time() < run_deadline and it < max_iters:
        # Compute current max M
        conv = _conv_fft(a)
        M = conv.max()
        # Build convolution matrix A_ub for a*b constraint
        col = np.concatenate([a, np.zeros(n-1, dtype=a.dtype)])
        row = np.concatenate([np.array([a[0]], dtype=a.dtype), np.zeros(n-1, dtype=a.dtype)])
        A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
        # Use linprog
        c = -np.ones(n)   # maximize sum(b)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            break
        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            # no improvement possible
            break
        # scaling to target sum
        scale = target_sum / sum_b
        g = b * scale
        # mixing
        a = (1 - t) * a + t * g
        # optionally reproject onto simplex due to floating errors.
        # a = _project_onto_simplex(a, target_sum)   # adjust if needed.
        # Evaluate
        conv = _conv_fft(a)
        max_val = conv.max()
        score = 2.0 * n * max_val / (target_sum**2)   # a.sum() is target_sum.
        if score < best_score - 1e-8:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} iter={it} new best {score:.6f}")
        # adapt t: reduce if improvement small
        if score > best_score * (1 - 1e-4):
            t *= 0.9  # reduce step size
        # ensure t not too small
        if t < 1e-4:
            t = 1e-4
        it += 1

    return best_a, best_score

The evaluator wants a non‑negative vector `a` that minimises  

```
2 * n * max(conv(a,a)) / (sum(a)**2)
```

Because the objective is homogeneous, we may *fix* the ℓ₁‑norm  

```
sum(a) = √(2·n)
```

and then only try to **minimise the maximal convolution entry**  
`max(conv(a,a))`.

The algorithm below combines two complementary ideas:

1. **Fast projected gradient descent** –  
   works on the *active* constraints (the convolution entries that are
   close to the current maximum) and quickly drives the value down.
2. **Linear‑programming refinement** –  
   for a fixed `a` we solve  

```
maximise   Σ b_j
subject to (a * b)_k ≤ M      for all k
           b_j ≥ 0
```

   where `M = max(conv(a,a))`.  
   The LP finds a vector `b` that puts as much mass as possible under the
   same bound `M`.  After scaling `b` back to the required ℓ₁‑norm we
   mix it with the current `a`.  This step provably never increases the
   objective and often yields a substantial extra improvement.

The search is **restarted many times** with randomly chosen lengths
(`800 ≤ n ≤ 2500`) and random power‑law initialisations
`a_i ∝ (i+1)^‑p ,  p≈0.5`.  
Each restart receives a small time slice (≈ 8 s): a few seconds of
gradient descent followed by a few LP refinement steps.  The best vector
found over all restarts is returned.

The code respects all restrictions (no closures, no file I/O, only
standard‑library + `numpy` + `scipy`).  Progress is printed so that the
runner can see intermediate scores.

```python
"""
Hybrid optimisation for the step‑function challenge.

*   Initialise a power‑law vector a_i ∝ (i+1)^‑p (p≈0.5) and normalise
    Σ a_i = √(2n).
*   Run a fast projected (momentum) gradient descent that works only on
    the “active” convolution entries (those within a tiny relative
    tolerance of the current maximum).  This quickly reduces
    max(conv(a,a)).
*   Refine the result with a tiny linear‑programming loop:
        maximise   Σ b_j
        s.t.       (a * b)_k ≤ M   (M = current max conv)
                   b_j ≥ 0
    Scale the LP solution to the required ℓ₁‑norm and mix it with `a`
    (convex combination).  The mix factor decays during the run.
*   Perform many independent restarts (different length, different p,
    occasional warm‑starts from the current global best) within the
    allotted time budget.
*   Keep the overall best sequence and return it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the LP refinement step.
# ----------------------------------------------------------------------
try:
    import scipy.linalg
    import scipy.optimize
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # Clip tiny negatives that appear because of round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper 3 – active set (indices whose convolution value is close to max)
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    """
    Return (max_value, indices) where conv[k] is within rel_tol·max of the maximum.
    """
    max_val = conv.max()
    # Avoid an empty set when max_val is practically zero
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# Helper 4 – gradient of Σ_{k∈active} conv_k w.r.t. a
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of the sum of the active convolution entries."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        # the terms a_i * a_{k-i} where both indices lie in [0, n-1]
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]          # ∂/∂a_i Σ a_i a_{k-i} = 2 a_{k-i}
    return grad


# ----------------------------------------------------------------------
# Gradient‑descent optimisation for a fixed length
# ----------------------------------------------------------------------
def _gd_optimize_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    rel_tol: float = 5e-4,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Fast projected (momentum) gradient descent on the max‑convolution objective.
    Returns the best vector encountered and its (unscaled) max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # Best point seen in this run
    conv = _conv_fft(a)
    best_val, _ = _active_indices(conv, rel_tol)
    best_local = a.copy()
    best_score = best_val

    # Momentum term
    velocity = np.zeros_like(a)
    step_factor = base_step

    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        # Gradient of the sum of the active entries
        grad = _gradient(a, active)

        # Momentum update
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity))
        if max_grad == 0.0:
            max_grad = 1.0
        eta = step_factor * max_val / max_grad

        # Gradient step + projection onto simplex
        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # Occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92

        # Evaluate the new point
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        if max_val_new < best_score:
            best_score = max_val_new
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  GD new best = {best_score:.6f}")

        a = a_new
        step_factor *= 0.9997                                 # simulated‑annealing decay
    return best_local, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (active only when SciPy is available)
# ----------------------------------------------------------------------
def _lp_refine_one_length(
    a: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    max_iters: int = 12,
    t_initial: float = 0.35,
    t_decay: float = 0.93,
    min_t: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    LP‑based refinement: maximise Σ b_j under (a * b)_k ≤ M,
    then mix the scaled solution back into `a`.

    Returns the best vector obtained in this refinement phase and its
    max‑convolution value.
    """
    if not _SCIPY_OK:                     # pragma: no cover – fallback
        return a, _conv_fft(a).max()

    n = a.shape[0]
    best_a = a.copy()
    best_score = _conv_fft(a).max()
    t = t_initial
    it = 0

    while time.time() < run_deadline and it < max_iters:
        # current bound
        conv = _conv_fft(a)
        M = conv.max()

        # Build the Toeplitz matrix representing the convolution a * b
        # (a * b)_k = Σ_i a_i b_{k-i}
        col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
        row = np.concatenate([np.array([a[0]], dtype=a.dtype), np.zeros(n - 1, dtype=a.dtype)])
        A_ub = scipy.linalg.toeplitz(col, row)            # shape (2n‑1, n)

        # Linear programme: maximise Σ b_j   ⇔ minimise -Σ b_j
        c = -np.ones(n)                                   # objective
        bounds = [(0.0, None)] * n                         # b_j ≥ 0
        b_ub = np.full(2 * n - 1, M)

        res = scipy.optimize.linprog(
            c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs"
        )
        if not res.success:                               # pragma: no cover
            break

        b = res.x
        sum_b = b.sum()
        if sum_b == 0.0:
            break

        # Scale to the required ℓ₁‑norm
        g = b * (target_sum / sum_b)

        # Convex mixing
        a_candidate = (1.0 - t) * a + t * g

        # Evaluate candidate
        conv_cand = _conv_fft(a_candidate)
        cand_score = conv_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[{time.time():.2f}] n={n}  LP iter={it} new best = {cand_score:.6f}")

        # Prepare for next iteration
        a = a_candidate
        t = max(min_t, t * t_decay)
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # If the runner already supplied a previous construction, use it as a warm start.
    global_best = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and np.all(arr >= 0):
                global_best = arr
        except Exception:   # pragma: no cover
            pass

    best_seq = None
    best_score = float("inf")

    # Length range – we keep it moderate so that the LP step stays affordable.
    MIN_N, MAX_N = 800, 2500

    while time.time() < deadline:
        # Choose a length; shrink the upper bound when time runs low.
        remaining = deadline - time.time()
        if remaining < 30.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(MIN_N, MAX_N + 1)

        # Power‑law start a_i ∝ (i+1)^{-p}  with p ≈ 0.5
        p = np.random.uniform(0.45, 0.55)
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p)
        init += np.random.rand(n) * 1e-4      # tiny jitter

        # Occasionally warm‑start from the best vector we have seen so far.
        if best_seq is not None and np.random.rand() < 0.4:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        # Normalise to the required ℓ₁‑norm.
        target_sum = math.sqrt(2.0 * n)
        init = init * (target_sum / init.sum())

        # Allocate a small time slice for this restart.
        slice_end = min(deadline, time.time() + 8.0)

        # ----- Phase 1 – fast projected gradient descent -----
        gd_deadline = min(slice_end, time.time() + 5.0)
        seq_gd, score_gd = _gd_optimize_one_length(
            n, init, gd_deadline, rel_tol=5e-4, base_step=0.12, momentum_beta=0.9
        )

        # ----- Phase 2 – LP refinement (if we still have time) -----
        seq_refined, score_refined = seq_gd, score_gd
        if time.time() < slice_end:
            seq_refined, score_refined = _lp_refine_one_length(
                seq_gd,
                target_sum,
                slice_end,
                max_iters=12,
                t_initial=0.35,
                t_decay=0.93,
                min_t=1e-4,
            )

        # Final evaluation using the supplied evaluate_sequence (adds the clipping
        # and the exact formula).  This guards us against any tiny numerical drift.
        try:
            eval_score = evaluate_sequence(seq_refined.tolist())
        except Exception:                     # pragma: no cover
            eval_score = float("inf")

        if eval_score < best_score:
            best_score = eval_score
            best_seq = np.array(seq_refined, copy=True)
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # Early stopping when we beat the target bound.
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback – extremely unlikely, but guarantees a valid return.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
```
