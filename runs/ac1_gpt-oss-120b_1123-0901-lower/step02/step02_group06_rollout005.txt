def solve_lp(f, M):
    n = f.shape[0]
    # Build Toeplitz matrix A
    c = np.concatenate([f, np.zeros(n-1, dtype=np.float64)])
    r = np.zeros(n, dtype=np.float64)
    r[0] = f[0]
    A = la.toeplitz(c, r)  # shape (2n-1, n)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(2*n-1), g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, verbose=False, max_iters=5000)
    except cp.error.SolverError:
        # fallback to CP.SCS
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)
```

2.  Repeatedly improve the sequence by solving a small linear programme:

        maximise   Σ g_j
        subject to (f * g)_k ≤ M   for all k
                   g_j ≥ 0

    where `f` is the current sequence, `M = max_k (f*f)_k`
    (the current worst autocorrelation value).  The constraints are
    linear in the unknown `g` because `f` is fixed; the matrix of the
    convolution is a Toeplitz matrix built from `f`.

3.  The optimiser returns a feasible `g`.  Scale it to a probability
    vector `ĝ = g / Σ g`.  Update the current sequence by a convex
    combination

        f_new = (1‑t)·f + t·ĝ     (t ∈ (0,1))

    The sum stays equal to 1.  Several trial values of `t` are tested
    and the one giving the largest reduction of `max(f_new*f_new)` is
    kept.

4.  If no improvement occurs the step size `t` is reduced; after a
    prolonged stagnation a small random jitter is added to escape
    flat regions.

5.  The process runs until the time budget expires (≈ 1000 s).  The
    best sequence found is returned.

The method is a deterministic analog of the “LP‑based” refinement
described in the prompt, but it uses a much tighter linear programme
(only convolution constraints) and a cheap line‑search over the mixing
parameter.  It converges quickly to a sequence with a lower maximal
self‑convolution, thereby reducing the evaluation metric.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.linalg as la

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    """
    Return the maximal entry of the self‑convolution of `seq` (linear
    convolution, not circular).  Uses an FFT of length a power of two
    for speed.
    """
    n = seq.shape[0]
    size = 2 ** int(np.ceil(np.log2(2 * n - 1)))
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   Σ g_j
        subject to (f * g)_k ≤ M   for all k
                   g_j ≥ 0
    The convolution matrix (Toeplitz) is built from `f`.
    Returns the optimal `g` as a dense ndarray, or None if the solver
    fails.
    """
    n = f.shape[0]
    # Toeplitz matrix: first column = f followed by zeros,
    # first row = [f[0], 0, 0, …]
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    r = np.zeros(n, dtype=np.float64)
    r[0] = f[0]
    A = la.toeplitz(c, r)          # shape (2n‑1, n)

    g = cp.Variable(n, nonneg=True)
    constraints = [
        A @ g <= M * np.ones(2 * n - 1, dtype=np.float64),
        g >= 0
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=False, verbose=False, max_iters=5000)
    except cp.error.SolverError:
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        except cp.error.SolverError:
            return None

    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the
    `evaluate_sequence` objective within the allocated time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1. problem size
    n = 2000

    # --------------------------------------------------------------
    # 2. starting sequence (use previously found one if available)
    if "height_sequence_1" in globals():
        # use the global sequence, interpolating to the required length
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size != n:
            xs = np.linspace(0.0, 1.0, hs.size)
            xs_target = np.linspace(0.0, 1.0, n)
            hs = np.interp(xs_target, xs, hs)
        seq = hs.copy()
        seq = seq.clip(min=0.0)            # guard against negatives
        if seq.sum() == 0:
            seq = np.full(n, 1.0 / n)
        else:
            seq = seq / seq.sum()
        print("[Init] Using provided height_sequence_1 (interpolated).")
    else:
        # arcsine shape – a strong baseline
        i = np.arange(n, dtype=np.float64)
        arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        seq = arcs / arcs.sum()
        print("[Init] No external sequence – using arcsine start.")

    # --------------------------------------------------------------
    # 3. bookkeeping
    best_seq = seq.copy()
    best_max = _conv_max(seq)
    best_score = 2.0 * n * best_max
    print(f"[Init] objective = {best_score:.6f}  (max‑conv = {best_max:.6f})")

    # mixing step size (will be adapted)
    t = 0.20
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    while time.time() < deadline:
        # current worst autocorrelation
        M = _conv_max(seq)

        # ----------------------------------------------------------
        # linear‑programming step
        g0 = _solve_lp(seq, M)
        if g0 is None:
            print("[LP] Solver failed – aborting.")
            break

        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:               # pathological, re‑jitter
            jitter = rng.normal(scale=0.01, size=n)
            seq = (seq + jitter).clip(min=0.0)
            seq /= seq.sum()
            t = 0.20
            no_improve = 0
            continue

        # normalise the LP solution to a probability vector
        g_norm = g0 / sum_g0

        # ----------------------------------------------------------
        # try a few mixing coefficients and keep the best reduction
        candidate_ts = [t, t * 0.5, min(0.5, t * 1.5), 0.1, 0.05]
        best_local_seq = None
        best_local_max = M

        for cand_t in candidate_ts:
            if not (0.0 < cand_t < 1.0):
                continue
            new_seq = (1.0 - cand_t) * seq + cand_t * g_norm
            new_max = _conv_max(new_seq)
            if new_max < best_local_max:
                best_local_max = new_max
                best_local_seq = new_seq
                best_t = cand_t

        # ----------------------------------------------------------
        # accept the improvement if any
        if best_local_seq is not None and best_local_max < M - 1e-12:
            seq = best_local_seq
            M = best_local_max
            # adapt step size: if we succeeded with a relatively large t,
            # gently encourage larger steps; otherwise slowly shrink.
            t = min(0.5, max(0.01, best_t * (1.1 if best_t > t else 0.9)))
            no_improve = 0
        else:
            # no improvement – shrink the mixing weight
            t *= 0.7
            no_improve += 1

        # ----------------------------------------------------------
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_max = M
            best_seq = seq.copy()
            print(f"[Iter {iteration:05d}] new best = {best_score:.6f} "
                  f"(max‑conv = {best_max:.6f})  t={t:.4f}")
            no_improve = 0

        iteration += 1

        # ----------------------------------------------------------
        # occasional random perturbation to escape flat plateaus
        if no_improve > 250:
            jitter = rng.normal(scale=0.005, size=n)
            seq = (seq + jitter).clip(min=0.0)
            seq /= seq.sum()
            t = 0.20
            no_improve = 0
            print(f"[Iter {iteration}] stagnation – jitter applied")

    # --------------------------------------------------------------
    print(f"Finished. Best objective = {best_score:.6f} "
          f"(max‑conv = {best_max:.6f}, length = {n})")
    return best_seq.tolist()
