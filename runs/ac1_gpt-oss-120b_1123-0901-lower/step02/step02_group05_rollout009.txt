def conv_fft(a):
    n = len(a)
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    return conv

assistantfinal**Solution Overview**

We search for a non‑negative step‑function `a = (a₀,…,aₙ₋₁)` that minimises  

\[
E(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2}\; .
\]

The value is *scale invariant* – multiplying a by any constant does not change `E`.  
Hence we can keep the ℓ¹‑norm of the sequence at any convenient value (we use
\(\sqrt{2n}\) as in the reference code).

The optimisation combines three ideas:

1. **Fast *soft‑max* surrogate** – we replace the hard maximum of the
   convolution by a smooth log‑sum‑exp.  Its gradient can be computed with one
   FFT‑based convolution, which makes the objective cheap to evaluate many
   times.

2. **Simulated Annealing (SA)** – a simple Metropolis scheme that moves a
   random amount of mass from one index to another while preserving the total
   sum.  The surrogate objective is used for the accept/reject test; the real
   `evaluate_sequence` is called only for the current best candidate of a run.

3. **Optional LP refinement** – when `cvxpy` is available and the length is
   modest ( ≤ 800 ) we perform the “LP‑update” described in the statement.
   This often improves a SA‑solution locally.

The routine repeatedly launches short SA runs on different lengths
(≈ 300 … 3000) and keeps the globally best sequence found so far.
All helper functions are top‑level, no closures are used, and the total
running time never exceeds the provided budget.

```python
"""
Simulated‑annealing optimiser for the step‑function bound

*   uses a log‑sum‑exp surrogate of the maximal convolution entry,
    evaluated with a single FFT (O(n log n)) → fast enough for thousands
    of iterations;
*   moves mass between two indices while preserving the ℓ¹‑norm;
*   temperature is decreased geometrically and the step size shrinks;
*   after each SA run a short LP‑refinement (cvxpy, if available) is
    performed for extra descent.

The function `propose_candidate` returns the best sequence found within the
time budget (default 1000 s).  All intermediate progress is printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simple helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence preserving shape (linear interpolation / decimation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        # equispaced down‑sampling
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        # linear interpolation
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
#  Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _maxconv_fft(a: np.ndarray) -> float:
    """Maximum entry of the autocorrelation a*a (real FFT version)."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power‑of‑two
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


def _softmax_max(conv: np.ndarray, alpha: float) -> float:
    """Soft‑max approximation of max(conv).  Used only for the SA surrogate."""
    # stability via subtraction of max
    v = conv * alpha
    v_max = np.max(v)
    return (v_max + np.log(np.sum(np.exp(v - v_max)))) / alpha


def _approx_score(a: np.ndarray, alpha: float = 5.0) -> float:
    """Fast surrogate of evaluate_sequence using a soft‑max over the convolution."""
    n = a.shape[0]
    # compute the whole convolution once (FFT)
    nfft = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[: 2 * n - 1]
    max_approx = _softmax_max(conv, alpha)
    s = a.sum()
    return 2.0 * n * max_approx / (s * s)


# ----------------------------------------------------------------------
#  Simulated annealing core
# ----------------------------------------------------------------------
def _simulated_annealing(init: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Run SA on ``init`` (sum is kept constant).  Returns the best **real**
    score obtained (using ``evaluate_sequence``) together with the corresponding
    sequence.
    """
    n = init.shape[0]
    a = init.astype(float).copy()

    # we keep the ℓ¹‑norm exactly constant (by moving mass)
    total = a.sum()
    if total == 0:
        # fallback to a small uniform vector
        a[:] = 1.0 / n
        total = a.sum()
    # ensure the exact target sum (arbitrary, not required for the objective)
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / total

    cur_score = _approx_score(a)
    best_seq = a.copy()
    best_score = float("inf")

    # annealing schedule
    T = max(cur_score * 0.2, 1e-6)           # initial temperature
    step_factor = 0.25                       # relative amount moved each step
    alpha = 6.0                              # soft‑max sharpness

    deadline = time.time() + time_budget
    it = 0
    while time.time() < deadline:
        # ---- propose a move -------------------------------------------------
        i = np.random.randint(n)
        if a[i] <= 1e-14:          # source empty – try another index
            continue
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # move a random fraction of a[i] to j
        delta = a[i] * np.random.rand() * step_factor
        a[i] -= delta
        a[j] += delta

        # ---- evaluate surrogate --------------------------------------------
        new_score = _approx_score(a, alpha)

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            # keep the best *real* score we have seen
            if cur_score < best_score:
                # compute the true objective (expensive but rare)
                try:
                    real = float(evaluate_sequence(a.tolist()))
                except Exception:
                    real = math.inf
                if real < best_score:
                    best_score = real
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] SA n={n} new global best ≈{best_score:.6f}")
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

        # ---- schedule updates -----------------------------------------------
        it += 1
        if it % 500 == 0:
            T *= 0.94               # exponential cooling
            step_factor = max(step_factor * 0.99, 0.02)

        # safeguard against accidental negativity caused by rounding
        if a[i] < 0:
            a[i] = 0.0
        if a[j] < 0:
            a[j] = 0.0

    # final evaluation of the best sequence found in this run
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = math.inf
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Optional LP refinement (the same idea as in the statement)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Linear program: maximise Σ g_j subject to g≥0,  f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[lo : hi + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    gv = np.asarray(g.value).flatten()
    gv = np.maximum(gv, 0.0)
    return gv


def _lp_refine(init: np.ndarray, target_sum: float, time_budget: float,
               mix: float = 0.2, max_iter: int = 8) -> tuple[np.ndarray, float]:
    """Apply a few LP‑updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    deadline = time.time() + time_budget
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or g0.sum() == 0.0:
            break
        g = g0 * (target_sum / g0.sum())
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            sc = math.inf
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {best_score:.6f} (n={len(a)})")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    #  initialise global best (use historic solution if present)
    # --------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] start from historic seq, score={sc:.6f}")
        except Exception:
            pass

    if best_seq is None:
        n0 = 800
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] start from uniform ({n0}), score={best_score:.6f}")

    # --------------------------------------------------------------
    #  main search loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick a length – usually stay close to the current best, but allow
        # occasional jumps.
        if best_seq is not None and np.random.rand() < 0.7:
            n = best_seq.size
            if np.random.rand() < 0.3:
                n = int(n * np.random.uniform(0.85, 1.15))
        else:
            n = np.random.randint(300, 2500)

        n = max(200, min(5000, n))
        target_sum = math.sqrt(2.0 * n)

        # initialise starting point
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            # random Dirichlet‑like start
            raw = np.random.gamma(shape=0.7, scale=1.0, size=n)
            init = raw / raw.sum() * target_sum

        # allocate a slice of the remaining time for this SA run
        time_left = deadline - time.time()
        if time_left < 10.0:
            break
        run_time = min(time_left * 0.5, 25.0)   # at most ~25 s per run

        cand_seq, cand_score = _simulated_annealing(init, run_time)

        # update global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})")
            if best_score < 1.503:
                # we have already beaten the target – stop early
                break

        # optional LP refinement (cheap for modest sizes)
        if _CVX_AVAILABLE and n <= 800 and time_left - run_time > 4.0:
            lp_time = min(4.0, deadline - time.time() - 1.0)
            seq_lp, sc_lp = _lp_refine(best_seq, target_sum, lp_time)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = seq_lp.copy()
                print(f"[{time.time():.2f}] *** GLOBAL NEW BEST after LP = {best_score:.6f} (n={n})")
                if best_score < 1.503:
                    break

    # final safety net
    if best_seq is None:
        n = 800
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
```
