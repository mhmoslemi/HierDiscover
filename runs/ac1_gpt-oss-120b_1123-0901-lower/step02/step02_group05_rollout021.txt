"""
Simple simulated‑annealing optimizer for the step‑function problem.

The algorithm works directly on a sequence `a` with a fixed ℓ₁‑norm
(`sum(a) = sqrt(2·n)`).  Starting from a uniform (or historic) sequence
it repeatedly picks a random source index `i` and a distinct target
index `j`, transfers a small amount `δ` from `i` to `j` and updates the
full convolution `c = a * a` *incrementally* (O(n) work).  The new
maximum convolution value `max(c)` is accepted either if it improves the
current state or with a probability governed by a linear cooling schedule
(standard simulated annealing).  The best ever sequence seen is kept as
the global optimum.  After the annealing phase a short projected‑gradient
refinement (the same sub‑gradient routine used in the reference code) is
run with the remaining time budget.

All helper functions are top‑level; the only external routine used is
`evaluate_sequence`, which is called only for the final verification.
The implementation aims to explore many cheap moves, avoid getting stuck
in the uniform stationary point and drive the maximal convolution below
the target bound 1.503.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Convolution helper – compute max entry and its sub‑gradient
# --------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Returns (max_val, grad, max_idx, full_conv) where
    max_val = max(conv(a, a)),
    grad is a sub‑gradient of the maximal entry,
    max_idx is the index attaining the maximum,
    full_conv is the whole convolution.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient of the maximal entry (each overlapping term appears twice)
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# --------------------------------------------------------------
# Local projected‑gradient descent (sub‑gradient of current max)
# --------------------------------------------------------------
def _local_descent(init: np.ndarray,
                   target_sum: float,
                   time_limit: float,
                   base_step: float = 0.07,
                   print_tag: str = "") -> tuple[np.ndarray, float]:
    """
    Short gradient‑descent run starting from ``init``.
    Returns (best_sequence, best_score).
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()
    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # direction with zero‑sum correction
        mean_g = np.mean(grad)
        d = -grad + mean_g

        max_abs = np.max(np.abs(d))
        max_abs = max_abs if max_abs > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new local best = {score:.6f}")

        # simple random shake‑up
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
            step *= 0.9
        else:
            a = a_new

        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score


# --------------------------------------------------------------
# Tiny mutation helper (adds Gaussian noise and re‑projects)
# --------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# --------------------------------------------------------------
# Simulated‑annealing optimiser (pairwise mass transfers)
# --------------------------------------------------------------
def _sa_optimize(start_seq: np.ndarray,
                 target_sum: float,
                 time_limit: float) -> tuple[np.ndarray, float]:
    """
    Run a simulated‑annealing search that preserves the ℓ₁‑norm.
    Returns (best_sequence, best_max_conv).
    """
    a = start_seq.astype(float)          # current state
    n = a.shape[0]
    conv = np.convolve(a, a)            # current full convolution
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start_time = time.time()
    T0 = cur_max * 0.2 + 1e-6            # initial temperature
    while time.time() < start_time + time_limit:
        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        src_val = a[i]
        if src_val <= 0.0:
            continue

        # transfer a random fraction of the source mass
        delta = src_val * np.random.rand() * 0.5
        if delta <= 1e-12:
            continue

        # ----- propose move -----
        conv_new = conv.copy()
        # linear part of the update (uses the *old* a)
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        # quadratic correction
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        # apply the mass transfer
        a_i_old = a[i]
        a_j_old = a[j]
        a[i] -= delta
        a[j] += delta

        new_max = conv_new.max()

        # temperature decays linearly with elapsed time
        elapsed = time.time() - start_time
        progress = elapsed / time_limit
        T = T0 * (1.0 - progress) + 1e-12

        # Metropolis acceptance
        if new_max < cur_max or np.random.rand() < math.exp(-(new_max - cur_max) / T):
            # accept
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new global best max = {best_max:.6f} (n={n})")
                if best_max < 1.503:
                    break
        else:
            # reject – revert the tiny change
            a[i] = a_i_old
            a[j] = a_j_old

    return best_a, best_max


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a simulated‑annealing phase followed by a short projected‑gradient
    refinement.  The routine respects the 1000 s budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initialise from a historic solution if available, otherwise uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=float)
            # ensure non‑negative and normalised
            init = np.maximum(init, 0.0)
            n0 = init.size
            target0 = math.sqrt(2.0 * n0)
            init = _project_onto_simplex(init, target0)
            print(f"[{time.time():.2f}] Loaded historic sequence (n={n0})")
        except Exception:
            init = None
    else:
        init = None

    if init is None:
        # default length – a moderate size that balances speed / quality
        n0 = 1500
        target0 = math.sqrt(2.0 * n0)
        init = np.full(n0, target0 / n0, dtype=float)   # uniform start
        print(f"[{time.time():.2f}] Starting from uniform (n={n0})")

    n = init.size
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Simulated annealing phase (≈70 % of the budget)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0:
        sa_time = min(remaining * 0.7, 600.0)   # cap to 10 min for safety
        best_seq, best_max = _sa_optimize(init, target_sum, sa_time)
    else:
        best_seq = init.copy()
        best_max = np.max(np.convolve(best_seq, best_seq))

    # ------------------------------------------------------------------
    # Final local projected‑gradient refinement (remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 1.0:
        # give the gradient routine a bit of time – it can still improve
        refined_seq, refined_score = _local_descent(best_seq,
                                                    target_sum,
                                                    remaining,
                                                    base_step=0.05,
                                                    print_tag="final")
        # compare – keep whichever gives a lower evaluation
        try:
            cur_score = float(evaluate_sequence(best_seq.tolist()))
        except Exception:
            cur_score = math.inf
        try:
            ref_score = float(evaluate_sequence(refined_seq.tolist()))
        except Exception:
            ref_score = math.inf

        if ref_score < cur_score:
            best_seq = refined_seq
            final_score = ref_score
        else:
            final_score = cur_score
    else:
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"[{time.time():.2f}] Finished – best max conv = {best_max:.6f}, final score = {final_score:.6f}")

    # The evaluation routine is homogeneous, so returning the raw sequence is fine.
    return best_seq.tolist()
