"""
Propose a non‑negative step‑function (probability distribution) that
minimises  evaluate_sequence(...).

Algorithm
---------

1. **Parameter sweep** – generate many candidates from simple families:
   geometric, power‑law, beta‑like (including U‑shaped exponents < 1),
   sinusoidal and random Dirichlet points.  Keep the best candidate.

2. **Projected (sub‑)gradient descent** – at each iteration compute the full
   self‑convolution, select all shift indices whose value is within a small
   relative tolerance of the current maximum (active set), and form the
   sub‑gradient by adding the reversed fragments that create these entries.
   Perform a back‑tracking line‑search and project back onto the probability
   simplex.  Random tiny mass‑transfer moves are attempted when no improvement
   is found.

3. **Time‑bounded loop** – the search runs until the 1000 s budget is exhausted,
   constantly updating the globally best sequence.

The implementation uses FFT for O(n log n) convolutions, a fast simplex
projection, and prints progress information.  The returned list is a valid
candidate for the competition.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean, onto {x ≥ 0, sum(x)=1})
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of two real 1‑D arrays (full length)."""
    m = len(a) + len(b) - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[:m]
    return conv


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself (length 2n‑1)."""
    return _convolve(seq, seq)


# ----------------------------------------------------------------------
# Objective (fast local version – sum(seq) is 1 after projection)
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv) for seq already normalised to sum = 1."""
    conv = _self_convolve(seq)
    return float(2.0 * len(seq) * conv.max())


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1‑α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    return powers / powers.sum()


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like a_i ∝ (i+1)^{a‑1} (n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    # Numerical guard: replace possible NaNs / negatives
    seq = np.where(np.isfinite(seq) & (seq > 0), seq, 0.0)
    return seq / seq.sum()


def _sine_seq(n: int, power: float = 1.0) -> np.ndarray:
    """Sine‑shaped a_i ∝ sin(π(i+0.5)/n)^power."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** power
    return seq / seq.sum()


def _quadratic_seq(n: int, power: float = 1.0) -> np.ndarray:
    """Quadratic‑like a_i ∝ ((i+1)*(n-i))^power."""
    i = np.arange(n, dtype=np.float64)
    seq = ((i + 1) * (n - i)) ** power
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point uniformly from the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()

            # ----- random tiny mass‑transfer move (fallback) ---------------
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)   # move at most 1 % of total mass
            if delta > 0.0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score - 1e-12:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score - 1e-12:
                        best_score, best_seq = cand_score, seq.copy()
                        print(f"[Iter {it:06d}] random tweak improved to {best_score:.6f}")
            # optionally try a larger random perturbation every few hundred iters
            if it % 500 == 0 and np.random.rand() < 0.3:
                t = 0.2
                rand_seq = _random_simplex(n)
                cand = _project_onto_simplex((1 - t) * seq + t * rand_seq)
                cand_score = _objective(cand)
                if cand_score < cur_score - 1e-12:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score - 1e-12:
                        best_score, best_seq = cand_score, seq.copy()
                        print(f"[Iter {it:06d}] random restart improved to {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
