"""
Propose a step‑function (non‑negative vector) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The method combines three ideas that differ from the reference
implementation:

1. **Smooth‑max gradient** – instead of using only the active (max)
   entries we minimise a differentiable “soft‑max’’ of the convolution:
   softmax_T(conv) = max(conv) + T·log Σ exp((conv‑max)/T).  Its gradient
   is a weighted sum of the whole convolution, which yields a smoother
   descent direction and reduces the risk of getting stuck on a single
   peak.

2. **FFT‑based cross‑correlation** – the gradient of the smooth‑max is
   the (linear) cross‑correlation of the weight vector with ``a``.
   Both the convolution and the cross‑correlation are computed with
   padded FFTs (O(n log n)) and therefore remain cheap even for n≈4000.

3. **Randomised restarts + short hill‑climbing** – many independent
   runs with random lengths, exponents and occasional warm‑starts from
   the global best are performed.  After the gradient phase a tiny
   simulated‑annealing / hill‑climbing phase moves a small amount of
   mass between two coordinates which can still improve the final
   objective.

All updates are projected onto the simplex { x_i ≥ 0, Σ x_i = √(2 n) }
using the Euclidean projection, guaranteeing that the denominator of
the evaluated objective stays constant.  The routine respects the
1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – convolution via FFT (full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)            # remove tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper – cross‑correlation (gradient of smooth‑max)
# ----------------------------------------------------------------------
def _cross_corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i} for i=0..n‑1.
    This is a cross‑correlation; implemented via an FFT convolution
    of the reversed weight vector with ``a``.
    """
    # reverse the weight vector
    w_rev = w[::-1]
    size = 1 << ((len(w_rev) + len(a) - 1 - 1).bit_length())
    wf = np.fft.rfft(w_rev, n=size)
    af = np.fft.rfft(a, n=size)
    c = np.fft.irfft(wf * af, n=size)[: len(w_rev) + len(a) - 1]
    offset = len(w) - 1                      # 2n‑2
    grad = c[offset - np.arange(len(a))]
    # Numerical noise may generate tiny negatives – clip.
    np.maximum(grad, 0.0, out=grad)
    return grad


# ----------------------------------------------------------------------
# Core optimisation for a fixed length using smooth‑max gradient descent
# ----------------------------------------------------------------------
def _optimise_one_length_smooth(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
    temp_start: float = 0.20,
    temp_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of the max‑convolution.
    Returns the best vector found during the run and its exact score.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best sequence seen in this run
    conv = _conv_fft(a)
    best_local = a.copy()
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)
    step_factor = base_step
    temperature_factor = temp_start          # multiplies current max(conv)

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) current convolution and smooth‑max parameters
        # --------------------------------------------------------------
        conv = _conv_fft(a)
        max_val = conv.max()
        T = max(max_val * temperature_factor, 1e-12)

        # --------------------------------------------------------------
        # 2) soft‑max weights (the gradient of the smooth‑max)
        # --------------------------------------------------------------
        # w_i = exp((conv_i - max)/T) / Σ exp(...)
        w_raw = np.exp((conv - max_val) / T)
        w = w_raw / w_raw.sum()

        # --------------------------------------------------------------
        # 3) gradient of the smooth‑max (cross‑correlation)
        # --------------------------------------------------------------
        grad = 2.0 * _cross_corr_fft(w, a)

        # --------------------------------------------------------------
        # 4) momentum + adaptive step size
        # --------------------------------------------------------------
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(velocity)
        if max_grad <= 0:
            max_grad = 1.0
        eta = step_factor * max_val / max_grad

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 5) occasional random shake‑up (helps escape tiny plateaus)
        # --------------------------------------------------------------
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.93

        # --------------------------------------------------------------
        # 6) evaluate exact objective (no smooth approximation)
        # --------------------------------------------------------------
        conv_new = _conv_fft(a_new)
        score = 2.0 * n * conv_new.max() / (target_sum ** 2)

        # --------------------------------------------------------------
        # 7) keep global best of this run
        # --------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")

        # --------------------------------------------------------------
        # 8) update state and anneal
        # --------------------------------------------------------------
        a = a_new
        step_factor *= 0.9997
        temperature_factor = max(0.01, temperature_factor * temp_decay)

    return best_local, best_score


# ----------------------------------------------------------------------
# Local hill‑climbing after the gradient phase
# --------------------------------------------------------------
def _hill_climb(a: np.ndarray, best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small amount of mass between two
    coordinates and keep the move if the exact objective improves.
    """
    n = a.shape[0]
    target_sum = a.sum()
    cur = a.copy()
    cur_score = best_score

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        if cur[i] <= 0.0:
            continue
        # move a random fraction of the mass of i to j
        delta = np.random.rand() * cur[i] * 0.05   # ≤5 % of i
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # project only to fix possible tiny negatives (mass is already conserved)
        new = np.maximum(new, 0.0)
        new = _project_onto_simplex(new, target_sum)

        conv = _conv_fft(new)
        score = 2.0 * n * conv.max() / (target_sum ** 2)

        if score < cur_score:
            cur, cur_score = new, score
            print(f"[{time.time():.2f}] hill‑climb improvement -> {score:.6f}")

    return cur, cur_score

        # Choose a problem size.

        init += np.random.rand(n) * 1e-4

        if best_seq is not None and np.random.rand() < 0.45:

                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4

        # ----- run the optimisation --------------------------------------
        run_deadline = time.time() + run_time

            init,
            run_deadline,

            momentum_beta=0.9,

        )

        # ----- optional hill‑climb ---------------------------------------

                seq_local,
                score_local,
                deadline=run_deadline,
            )

        # ----- update global best ----------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # Early exit if we have already beaten the target comfortably.
        if best_score < 1.45:

    # ------------------------------------------------------------------
    # Fallback – in the extremely unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list (the evaluator expects a list, not a ndarray).
    return best_seq.tolist()
