"""
Optimiser for the step‑function attack.

The goal is to minimise the security‐evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

over non‑negative sequences a (the sum scaling does not affect the value,
so we normalise to sum = 1).  The optimiser combines three complementary
strategies:

1. **Broad parametric sweep** – a few hundred candidates from simple families
   (geometric, power‑law, beta‑like, linear‑decay, random Dirichlet).  The best
   of these seeds the later stages.

2. **Projected sub‑gradient descent** – the objective is 2 n · max conv,
   i.e. a convex piecewise‑linear function.  At a point where the maximum
   convolution is attained at several shifts we take the (averaged) sub‑gradient
   from *all* active shifts, then perform a back‑tracking line search while
   projecting back onto the probability simplex.

3. **LP‑based refinement** – given a current f we solve the linear programme

       maximise   Σ b_j
       subject to  (f * b)_k ≤ M   for all k,
                   b ≥ 0,

   where M = max_k (f * f)_k.  The solution b is a direction that cannot increase
   the maximal convolution.  After normalising b we mix it with f
   (convex combination) and accept the mixture if it improves the objective.
   This step is repeated a handful of times; each LP is cheap (≈ 4000 constraints).

The whole routine runs for at most *budget_s* seconds (default 1000 s) and
returns the best sequence found.  All helper functions are top‑level and
avoid closures, fulfilling the competition requirements.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast self‑convolution (real FFT) --------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself (length 2·n‑1)."""
    n = seq.size
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Local objective (matches evaluate_sequence) ---------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv) / (sum(seq)^2); seq is assumed non‑negative."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    return float(2.0 * seq.size * conv.max() / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric a_i ∝ r^i with r = 1 – alpha / n (alpha>0)."""
    r = 1.0 - alpha / n
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^(–p)."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like a_i ∝ (i+1)^{a-1} (n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _linear_decay_seq(n: int, exp: float) -> np.ndarray:
    """Monotonically decreasing a_i ∝ (1 – i/(n‑1))^{exp}."""
    if exp == 0:
        return np.full(n, 1.0 / n)
    t = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    seq = t ** exp
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Linear programme that, given f and M, finds a direction b
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """Return the optimal b (sum‑maximising) for given f, M.
    Returns None on failure."""
    n = f.size
    # Build the (2n‑1) × n convolution matrix A such that A@b = f*b
    # Column j contributes f on rows j … j+n‑1
    data = np.repeat(f, n)
    row_ind = np.concatenate([np.arange(j, j + n) for j in range(n)])
    col_ind = np.repeat(np.arange(n, dtype=np.int64), n)
    A = _sp.csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))

    c = -np.ones(n)                      # maximise Σ b_j  → minimise –Σ b_j
    ub = np.full(2 * n - 1, M)           # (f*b)[k] ≤ M
    bounds = [(0, None) for _ in range(n)]

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                           method='highs', options={'presolve': True})
    except Exception:                   # safety net – any unexpected error
        return None

    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)           # tiny negative values may appear numerically
    return b


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that minimises
    evaluate_sequence within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1. Problem size – 2000 gives a good balance between expressiveness
    #    and speed of the FFT / LP.
    n = 2000

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 2. Seed from a previously discovered sequence if available
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()
                sc = _objective(init)
                best_seq, best_score = init, sc
                print(f"[Init] global seed score={sc:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Broad parametric sweep – keep the best as start point
    families = []

    # geometric (alpha from 0.2 to 6, more fine‑grained)
    for alpha in np.linspace(0.2, 6.0, 30):
        families.append((_geom_seq, (n, alpha)))

    # power‑law
    for p in np.linspace(0.1, 3.0, 25):
        families.append((_powerlaw_seq, (n, p)))

    # beta‑like (both parameters up to 5)
    for a in np.linspace(0.5, 5.0, 6):
        for b in np.linspace(0.5, 5.0, 6):
            families.append((_beta_seq, (n, a, b)))

    # linear decay (exponent from 0.1 to 10)
    for e in np.linspace(0.2, 10.0, 20):
        families.append((_linear_decay_seq, (n, e)))

    # random Dirichlet samples
    for _ in range(30):
        families.append((_random_simplex, (n,)))

    for func, args in families:
        if time.time() > deadline:
            break
        seq = func(*args)
        if seq is None or seq.size != n:
            continue
        sc = _objective(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[Init] {func.__name__:12s}  score={sc:.6f}")

    # --------------------------------------------------------------
    # 4. If nothing generated (should not happen) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 5. Projected sub‑gradient descent (aggressive early phase)
    seq = best_seq.copy()
    cur_score = best_score
    step_base = 5.0
    it = 0
    while time.time() < deadline:
        conv = _self_convolve(seq)
        max_val = conv.max()
        # all shifts attaining (or almost) the maximum
        active = np.where(conv >= max_val - 1e-12)[0]
        # average sub‑gradient over the active set
        grad = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += seq[k - idx]
        grad /= max(1, active.size)                 # average

        # back‑tracking line search
        step = step_base
        improved = False
        while step > 1e-6:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            step *= 0.5
        if improved:
            seq, cur_score = cand, cand_score
            if cand_score < best_score:
                best_seq, best_score = cand.copy(), cand_score
                print(f"[Grad {it:05d}] new best {best_score:.6f}")
            step_base = max(step * 1.2, 1e-3)
        else:
            # tiny random mass‑transfer move (always respects simplex)
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)
            if delta > 0.0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq, cur_score = cand, cand_score
                    if cand_score < best_score:
                        best_seq, best_score = cand.copy(), cand_score
                        print(f"[Rand {it:05d}] improved to {best_score:.6f}")
        it += 1
        # stop early if gradient phase no longer yields improvement for a while
        if it % 2000 == 0 and time.time() > deadline - 30:
            break   # give room for the LP phase

    # --------------------------------------------------------------
    # 6. LP‑based refinement – alternating optimisation
    mix_coef = 0.2          # start mixing weight
    while time.time() < deadline:
        conv = _self_convolve(seq)
        M = conv.max()
        # linear programme to find a direction b
        b = _solve_lp(seq, M)
        if b is None:
            break
        # normalise to sum = 1 (any scaling leaves the objective invariant)
        g = b / b.sum()
        # try a convex combination (back‑tracking on the mixing weight)
        t = mix_coef
        success = False
        while t > 1e-4:
            cand = (1.0 - t) * seq + t * g
            cand = _project_onto_simplex(cand)   # numerical safety
            cand_score = _objective(cand)
            if cand_score < cur_score:
                seq, cur_score = cand, cand_score
                success = True
                if cand_score < best_score:
                    best_seq, best_score = cand.copy(), cand_score
                    print(f"[LP] mix t={t:.4f}  new best {best_score:.6f}")
                break
            t *= 0.5
        if not success:
            # mixing did not help – lower the mixing coefficient for future attempts
            mix_coef = max(mix_coef * 0.7, 1e-3)
            # if we cannot make progress for several rounds, break out
            if mix_coef < 1e-4:
                break
        else:
            # after a successful LP step we reset the mixing weight
            mix_coef = min(mix_coef * 1.2, 0.4)

        # occasional cheap gradient steps to polish the new point
        if np.random.rand() < 0.15:
            # one gradient iteration (re‑using code above, but without heavy loops)
            conv = _self_convolve(seq)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-12)[0]
            grad = np.zeros_like(seq)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(k, n - 1)
                idx = np.arange(lo, hi + 1, dtype=int)
                grad[idx] += seq[k - idx]
            grad /= max(1, active.size)
            step = step_base
            while step > 1e-6:
                cand = _project_onto_simplex(seq - step * grad)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq, cur_score = cand, cand_score
                    if cand_score < best_score:
                        best_seq, best_score = cand.copy(), cand_score
                        print(f"[FineGrad] improved to {best_score:.6f}")
                    break
                step *= 0.5

    # --------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
