"""
search for a low‑value step‑function for evaluate_sequence()

Approach
========
* Work on the *shape* of the step‑function only: the objective
  2·n·max(conv(a,a)) / (∑a)² is invariant under a positive scaling of ``a``.
  We therefore stay on the simplex {a_i ≥ 0 , Σ a_i = 1}.
* Replace the non‑smooth “max’’ in the objective by a smooth approximation
      φ_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
  (α grows from 10 → 500 during an optimisation run).
  The gradient of φ_α can be written in closed form using a single
  convolution, which makes each iteration cheap (O(n log n) / O(n²) with
  NumPy’s ``convolve``).
* Perform a projected Adam optimiser on φ_α, projecting after each step
  onto the simplex.
* Run many short, independent optimisation “trials’’ (different lengths,
  different random starts).  The best sequence found overall is returned.
* Simple mutation + re‑projection is occasionally used to shake the
  state and escape shallow local minima.

The algorithm is deliberately different from the previously supplied
gradient‑sub‑gradient/LP scheme, yet respects the same time budget
(≈1000 s) and returns a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector *v* onto the simplex of non‑negative vectors with sum *s*."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted decreasing
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth approximation of max(conv(a,a)) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, alpha: float):
    """
    Return a smooth approximation of max(conv(a,a)) together with its
    gradient w.r.t. *a*.
    The gradient w.r.t. a_j is
        2 * Σ_i   w_{i+j} * a_i
    where w are the softmax weights of the convolution entries.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    max_val = conv.max()
    # stable soft‑max
    shifted = conv - max_val                     # avoid overflow
    exp_vals = np.exp(alpha * shifted)
    sum_exp = exp_vals.sum()
    # smooth max (not used for optimisation, only for diagnostics)
    smooth_max = max_val + np.log(sum_exp) / alpha

    w = exp_vals / sum_exp                       # soft‑max weights, sum to 1
    w_rev = w[::-1]
    # convolution a * w_rev  -> length 3n‑2
    grad_full = np.convolve(a, w_rev)
    # indices that give Σ_i w_{i+j} a_i
    idx_start = 2 * n - 2
    grad = 2.0 * grad_full[idx_start - np.arange(n)]
    return smooth_max, grad, max_val, conv


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the smooth objective
# ----------------------------------------------------------------------
def _smooth_local_descent(
    init: np.ndarray,
    time_limit: float,
    base_lr: float = 0.12,
    alpha_start: float = 10.0,
    alpha_end: float = 500.0,
    print_tag: str = "",
):
    """
    Run a short Adam optimiser on the smooth surrogate of the max‑convolution.
    The simplex Σ a_i = 1 is enforced after each update.
    Returns (best_sequence, best_score) found during the run.
    """
    a = _project_onto_simplex(init.astype(float), 1.0)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    t0 = time.time()
    it = 1
    while time.time() - t0 < time_limit:
        # linearly increase alpha on a log‑scale (more smoothing early)
        frac = (time.time() - t0) / time_limit
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        _, grad, max_val, _ = _smooth_max_and_grad(a, alpha)

        # Adam update (step size scaled by ``base_lr``)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_onto_simplex(a_new, 1.0)

        # true (non‑smoothed) score
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] {print_tag} new best = {score:.6f} (n={len(a)})"
            )

        a = a_new
        it += 1

        # modest decay of learning rate every few hundred steps
        if it % 300 == 0:
            base_lr *= 0.9

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny mutation helper (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, sigma: float) -> np.ndarray:
    """Return a mutated copy of *a* (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, 1.0)
    return b


# ----------------------------------------------------------------------
# Resize helper – linear interpolation / down‑sampling
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to length ``new_n`` preserving non‑negativity."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:  # up‑sample by linear interpolation
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)

# ----------------------------------------------------------------------

        time_left = deadline - time.time()
        if time_left < 8.0:

        if cand_seq is not None and cand_score < best_score:
            best_seq = cand_seq.copy()
            best_score = cand_score
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            # early exit if we already beat the target
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        best_seq = np.full(n, 1.0 / n, dtype=float)

    # ``evaluate_sequence`` is scale‑invariant, but we keep the simplex
    # normalisation (sum = 1) because it yields numerically stable values.
    return best_seq.tolist()
