def weighted_gradient(a, w):
    # w len = 2n-1
    n = a.shape[0]
    w_rev = w[::-1]
    # compute convolution a * w_rev
    L = 1 << ((3*n - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(A * W, n=L)[:3*n-2]
    # gradient = 2 * conv[(2n-1)-1 - np.arange(n)] = 2 * conv[2n-2 - i]
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad

"""
Projected Adam optimisation of a smoothed max‑convolution objective.

The objective 2·n·max(conv(a,a))/sum(a)^2 is scale‑invariant.
We normalise the vector to Σa = √(2n) so that the objective equals
max(conv(a,a)).  To minimise the maximum autocorrelation we replace the
hard max by a smooth log‑sum‑exp (softmax) and perform projected Adam
steps on the simplex.  The softmax weight parameter α is increased
gradually (continuation) so that the optimisation first smooths the
landscape and then focuses on the true maximum.  Multiple independent
restarts with varying lengths are performed until the time budget is
exhausted.  The best sequence found is returned as a plain list.
"""

import time
import numpy as np
from math import sqrt

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}.
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full linear convolution using a real FFT (the result length is 2*n‑1).
# --------------------------------------------------------------------------- #
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # length required for a*a is 2*n-1; we pad to the next power of two.
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the smoothed loss  L = Σ_k w_k (a*a)_k,
# where w_k = exp(α*(conv_k - max(conv))) .
# The gradient is   ∇_i L = 2· Σ_k w_k a_{k-i}
#                = 2· (a ⋆ w)[i]  (cross‑correlation).
# This is obtained via a single FFT‑based convolution of ``a`` with ``w`` reversed.
# --------------------------------------------------------------------------- #
def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ w_k (a*a)_k with respect to a.
    a : (n,)
    w : (2*n-1,)
    Returns an array of shape (n,).
    """
    n = a.shape[0]
    # reverse w for the correlation → convolution
    w_rev = w[::-1]

    # length needed for linear convolution a (n) * w_rev (2n-1) → 3n-2
    L = 1 << ((3 * n - 2).bit_length())

    # FFT of a (shared with the convolution of a with itself)
    A = np.fft.rfft(a, n=L)

    # convolution a * w_rev
    W = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(A * W, n=L)[: 3 * n - 2]

    # indices that give Σ_j a_j w_{i+j}
    idx = (2 * n - 2) - np.arange(n)                     # length n
    grad = 2.0 * conv[idx]
    return grad


# --------------------------------------------------------------------------- #
# Run a single optimisation instance for a given length ``n``.
# Returns the best local vector found (numpy array) and its max‑conv value.
# --------------------------------------------------------------------------- #
def run_instance(n: int,
                 init_vec: np.ndarray,
                 run_deadline: float,
                 keep_best: bool = True) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length ``n`` until ``run_deadline``.
    Returns (best_vector, best_max_conv).
    """
    target_sum = sqrt(2.0 * n)                     # Σ a_i = √(2n)
    a = project_onto_simplex(init_vec.astype(float), target_sum)

    # Adam parameters
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # continuation (soft‑max) parameter
    alpha = 0.5

    best_local = a.copy()
    best_local_val = float('inf')
    it = 0

    while time.time() < run_deadline:
        # ------- convolution of a with itself (for current max) -------
        conv = conv_full_fft(a)
        max_val = conv.max()
        if max_val < best_local_val:
            best_local_val = max_val
            best_local = a.copy()

        # ------- soft‑max weights for the active entries ---------------
        # w_k = exp(alpha * (conv_k - max_val))    (max weight = 1)
        w = np.exp(alpha * (conv - max_val))

        # ------- gradient of the smoothed loss -------------------------
        grad = weighted_gradient(a, w)

        # ------- Adam update -------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ------- feasibility (non‑negativity + simplex) -----------------
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)

        # ------- occasional random shake (helps escape local minima) ----
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.02 * target_sum / n)
            a_new = project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            # slightly reduce learning rate after a shake
            lr *= 0.95

        a = a_new

        # ------- continuation schedule (increase α) --------------------
        if it % 150 == 0 and alpha < 30.0:
            alpha *= 1.25

        # ------- simple learning‑rate decay -----------------------------
        lr *= 0.9999

        # ------- early stop if max value is already very low ------------
        if max_val < 1.4:          # far below target, stop early
            break

    return best_local, best_local_val


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function with minimal evaluation value.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------- #
    # Main outer loop: many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a length.  We allow a broader range (up to 8000) to give
        # the optimiser more freedom when time permits.
        if np.random.rand() < 0.6:
            n = np.random.randint(1200, 3001)      # the range that performed well
        else:
            n = np.random.randint(3000, 8001)      # occasional larger instance

        # Initialise from the global best construction if available,
        # otherwise from a fresh random vector.
        if 'height_sequence_1' in globals() and np.random.rand() < 0.7:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size)
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        # Allocate a slice of the remaining budget for this run (≈8 s)
        run_deadline = min(deadline, time.time() + 8.0)
        local_vec, local_val = run_instance(n, init, run_deadline)

        if local_val < best_score:
            best_score = local_val
            best_seq = local_vec.copy()
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")

        # Early exit if we have already beaten the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if nothing was discovered (highly unlikely).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1000
        target_sum = sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return as a plain Python list as required by the grader.
    return best_seq.tolist()
