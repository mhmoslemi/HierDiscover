class SmoothMaxObject:
    def __init__(self, n):
        self.n = n
        self.size_fft_conv = 1 << int(np.ceil(np.log2(2*n-1)))
        self.size_fft_corr = 1 << int(np.ceil(np.log2(3*n-1)))
        self.lam = 1.0

    def __call__(self, v):
        a = np.exp(v - np.max(v))
        a = a / a.sum()
        conv = _full_convolution_using_precomputed(a, self.size_fft_conv)
        # compute phi
        max_c = conv.max()
        weight = np.exp(self.lam * (conv - max_c))
        weight_sum = weight.sum()
        w = weight / weight_sum
        # gradient w.r.t a
        rev_a = a[::-1]
        fw = np.fft.rfft(w, self.size_fft_corr)
        frev = np.fft.rfft(rev_a, self.size_fft_corr)
        corr = np.fft.irfft(fw * frev, self.size_fft_corr)
        grad_a = 2.0 * corr[self.n - 1: self.n - 1 + self.n]
        # gradient w.r.t v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # phi
        phi = (np.log(weight_sum) + self.lam * max_c) / self.lam  # because phi = (1/lam) * log sum exp(lam*conv)
        # Equivalent expression.
        return phi, grad_v
```

"""
Propose a step‑function that minimises the evaluation score.

The algorithm works in the probability‑simplex via a soft‑max
parameterisation (a_i = exp(v_i)/∑exp(v)).  The objective
max_k (a*a)[k] is replaced by the smooth log‑sum‑exp surrogate

    φ_λ(a) = (1/λ)·log ∑_k exp(λ·(a*a)[k]) ,

whose gradient is 2·(w ★ a) where w_i are the soft‑max weights.
We run a continuation on λ (starting from 1 and ending around 300)
and at each λ minimise φ_λ with L‑BFGS‑B – the simplex
constraints are satisfied automatically by the soft‑max.
Multiple random restarts are performed; the best exact objective
found (with the original max‑convolution) is returned as a plain
list.  All work respects the 1000 s time budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


def _full_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of given length."""
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * len(a) - 1]
    return conv


def _cross_correlation_fft(w: np.ndarray, a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) for vectors of length 2n‑1 and n."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_fft)
    frev = np.fft.rfft(rev_a, size_fft)
    corr = np.fft.irfft(fw * frev, size_fft)
    n = len(a)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    return corr[n - 1 : n - 1 + n]


def _objective(a: np.ndarray) -> float:
    """Exact evaluation function (the same used by evaluate_sequence)."""
    n = len(a)
    conv = _full_convolution_fft(a, 1 << int(np.ceil(np.log2(2 * n - 1))))
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)


class _SmoothMax:
    """
    Callable that returns the smooth‑max value φ_λ and its gradient w.r.t.
    the soft‑max parameters v, for the current smoothing λ.
    """

    __slots__ = ("n", "size_conv", "size_corr", "lam")

    def __init__(self, n: int, size_conv: int, size_corr: int, lam: float):
        self.n = n
        self.size_conv = size_conv
        self.size_corr = size_corr
        self.lam = lam

    def __call__(self, v: np.ndarray):
        # ---------- a on the simplex ----------
        max_v = np.max(v)
        exp_v = np.exp(v - max_v)               # shift for numerical stability
        a = exp_v / exp_v.sum()                # a ≥ 0, ∑a = 1

        # ---------- convolution ----------
        conv = _full_convolution_fft(a, self.size_conv)

        # ---------- soft‑max weights ----------
        max_c = conv.max()
        # w_i = exp(λ·conv_i) / ∑ exp(λ·conv_j)
        w_raw = np.exp(self.lam * (conv - max_c))
        w_sum = w_raw.sum()
        w = w_raw / w_sum

        # ---------- gradient w.r.t. a ----------
        grad_a = 2.0 * _cross_correlation_fft(w, a, self.size_corr)

        # ---------- chain rule: gradient w.r.t. v ----------
        # ∂a_i/∂v_j = a_i (δ_{ij} – a_j)  →  grad_v = a ∘ (grad_a – (grad_a·a))
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---------- smooth‑max value ----------
        # φ_λ = (1/λ)·log∑exp(λ·conv) = max_c + (1/λ)·log w_sum
        phi = max_c + (np.log(w_sum) / self.lam)

        return phi, grad_v


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises the
    evaluation score.  The method runs a continuation of smooth‑max
    minimisation (log‑sum‑exp) with L‑BFGS‑B, using several random
    restarts.  The best exact objective seen within the time budget
    is returned as a plain Python list.
    """
    # ----- timing -------------------------------------------------
    safety = 0.5                     # leave a small margin for interpreter overhead
    deadline = time.time() + budget_s - safety
    np.random.seed(seed)

    # ----- problem size -------------------------------------------
    # 2500 provides a good trade‑off between expressive power and speed.
    n = 2500

    # FFT sizes (next power of two for speed)
    size_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n - 1)))

    # ----- continuation schedule for λ (smoothness) ----------------
    lambda_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 300.0]

    # ----- optimisation parameters ---------------------------------
    max_iter_per_lambda = 200   # L‑BFGS‑B iterations per λ
    n_restarts = 7              # number of random starts (first one is uniform)

    best_seq = None
    best_val = float("inf")

    for restart in range(n_restarts):
        if time.time() > deadline:
            break

        # Initial v: uniform (zeros) for the first restart, otherwise tiny noise.
        if restart == 0:
            v0 = np.zeros(n)
        else:
            v0 = np.random.randn(n) * 0.02

        # Initialise smooth‑max object with the first λ.
        smooth_obj = _SmoothMax(n, size_conv, size_corr, lam=lambda_schedule[0])

        v = v0
        for lam in lambda_schedule:
            if time.time() > deadline:
                break

            smooth_obj.lam = lam

            # L‑BFGS‑B on the smooth surrogate (φ_λ) – returns both value and gradient.
            res = minimize(
                fun=smooth_obj,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_lambda, "disp": False},
            )
            v = res.x

            # ----- evaluate the true objective -----
            # Recover a from v (soft‑max)
            max_v = np.max(v)
            exp_v = np.exp(v - max_v)
            a = exp_v / exp_v.sum()
            val = _objective(a)

            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(
                    f"[Restart {restart}] λ={lam:.0f} → new best obj = {val:.6f}"
                )
                # early stop if already comfortably below target
                if best_val < 1.45:
                    break

    # ----- fallback in the unlikely case no sequence was found -----
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"Search finished – best objective = {best_val:.6f} (n = {n})")
    return best_seq.tolist()
