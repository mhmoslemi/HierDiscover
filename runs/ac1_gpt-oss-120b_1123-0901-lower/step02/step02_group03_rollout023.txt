"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using many short Adam‑style runs on the simplex
{ a_i ≥ 0 , Σ a_i = √(2n) }.  Each run works on a randomly chosen
length n (≈ 1200‑3000) and performs:

* exact convolution via an FFT (O(n log n));
* a sub‑gradient of the active (max‑tying) convolution entries;
* Adam updates with projection onto the simplex;
* occasional random “mass‑swap’’ perturbations (simulated‑annealing).

All runs share the best vector seen so far; the algorithm stops
after the allotted budget (default 1000 s).  The returned list can be
directly fed to the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – radius s = √(2 n)
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("radius must be non‑negative")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full linear convolution via real FFT (fast, accurate)
# --------------------------------------------------------------------------- #
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real) using FFT – length 2·len(a)‑1."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=n_fft)
    c = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return c


# --------------------------------------------------------------------------- #
# Sub‑gradient of the maximum convolution entry (or all ties)
# --------------------------------------------------------------------------- #
def active_subgradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Gradient of Σ_{k∈active} (a∗a)[k] w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad

    The result (if better) is stored into best_sofar dict.
    """
    target_sum = math.sqrt(2.0 * n)          # Σ a_i = √(2n)

    # initialise on the simplex (project in case init is out of range)
    a = project_onto_simplex(init.astype(float), target_sum)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    # schedule for occasional random “shuffles”
    shake_prob = 0.015
    # tolerance for “active” indices – we treat values within 1e‑12 of the max
    active_eps = 1e-12

    while time.time() < deadline:
        # ----- convolution & current maximal entry -----
        conv = conv_full_fft(a)
        cur_max = conv.max()
        # store globally if we improve
        if cur_max < best_sofar["score"]:
            best_sofar["score"] = cur_max
            best_sofar["seq"] = a.copy()
            print(f"[{time.time():.2f}] New global best = {cur_max:.6f} (n={n})")

        # ----- active set (ties) -----
        act = np.where(conv >= cur_max - active_eps)[0]

        # ----- sub‑gradient of the active max -----
        g = active_subgradient(a, act)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – we scale with cur_max to keep steps moderate
        lr = 0.1 * cur_max / (np.sqrt(v_hat).mean() + eps)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection back onto the feasible simplex -----
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)

        # ----- random shake (simulated‑annealing style) -----
        if rng.random() < shake_prob:
            # move a tiny amount of mass from a random index to another
            i, j = rng.integers(0, n, size=2)
            delta = rng.uniform(0.0, a_new[i] * 0.02)   # ≤ 2 % of source
            a_new[i] -= delta
            a_new[j] += delta
            a_new = project_onto_simplex(a_new, target_sum)

        a = a_new

# Main entry point expected by the grader

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    end = start + budget_s - 0.5            # tiny safety margin

    # best solution seen so far (dictionary because we need mutable capture)
    best = {"score": float("inf"), "seq": None}

    # ------------------------------------------------------------------- #
    # Helper: run a single optimisation instance on a random length n
    # ------------------------------------------------------------------- #
    def run_one_instance():
        # length in a sensible interval (the original code used 1200‑3000)
        n = rng.integers(1200, 3001)

        # initialise: either from the global good construction or random
        if "height_sequence_1" in globals() and rng.random() < 0.4:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = rng.random(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = rng.random(n)

        # allocate a slice of the remaining time for this run
        run_deadline = min(end, time.time() + 18.0)   # ~18 s per run
        optimise_fixed_n(n, init, run_deadline, best, rng)

    # ------------------------------------------------------------------- #
    # Main outer loop – many independent restarts
    # ------------------------------------------------------------------- #
    while time.time() < end:
        run_one_instance()
        # early exit – if we ever beat the “target” we stop
        if best["score"] < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if nothing was improved we return a uniform vector
    # ------------------------------------------------------------------- #
    if best["seq"] is None:
        n = 1500
        s = math.sqrt(2.0 * n)
        best["seq"] = np.full(n, s / n, dtype=float)

    # convert to plain Python list as required by the grader
    return best["seq"].tolist()
