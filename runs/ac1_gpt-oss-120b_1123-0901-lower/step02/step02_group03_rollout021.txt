"""
Hybrid optimiser for the step‑function conjecture.

The objective 2 * n * max(conv(a,a)) / (sum(a)**2) reduces to the
maximum entry of the self‑convolution when the L1‑norm is fixed to
√(2n).  The algorithm iteratively improves a candidate sequence by:

* a short projected sub‑gradient phase on the exact max entry,
* a longer phase that minimises a smooth log‑sum‑exp approximation of
  the max using Adam (the gradient is obtained by a single FFT‑based
  convolution),
* occasional random perturbations (simulated‑annealing style) to
  escape shallow local minima,
* many independent restarts with varying lengths (1200–3000) and
  optional seeding from the best known construction,
* a final cheap hill‑climbing refinement on the current global best.

All updates respect the simplex {a_i ≥ 0, Σ a_i = √(2n)} by Euclidean
projection.  The routine respects a global time budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# FFT based full convolution (real input) – returns length (len(a)+len(b)-1)
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Convolution a*a (full, no padding) using a real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _conv_full_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors x and y via FFT."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the (exact) max‑convolution entry using a convolution trick.
# --------------------------------------------------------------------------- #
def _gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """Gradient of the maximal convolution entry (sum over all active k)."""
    n = a.shape[0]
    mask = np.zeros(2 * n - 1, dtype=float)
    mask[active_idxs] = 1.0
    a_rev = a[::-1]
    conv_res = _conv_full_fft_two(mask, a_rev)          # length 3n‑2
    grad = 2.0 * conv_res[n - 1 : n - 1 + n]            # slice of length n
    return grad


# --------------------------------------------------------------------------- #
# Gradient of the smooth (log‑sum‑exp) approximation of the max.
# --------------------------------------------------------------------------- #
def _gradient_smoothed(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Return ∇  (1/β)·log Σ exp(β·conv) ."""
    max_val = conv.max()
    # stability: shift by the maximum
    shift = conv - max_val
    exp_shift = np.exp(beta * shift)
    w = exp_shift / exp_shift.sum()        # probabilities, sum = 1
    a_rev = a[::-1]
    conv_w_a = _conv_full_fft_two(w, a_rev)          # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_w_a[n - 1 : n - 1 + n]
    return grad


# --------------------------------------------------------------------------- #
# Single optimisation run on a given length.
# --------------------------------------------------------------------------- #
def _run_one_instance(n: int,
                      init_vec: np.ndarray,
                      runtime: float,
                      global_best_score: float):
    """
    Optimise a sequence of length n starting from init_vec.
    Returns (best_seq, best_score) found within the allocated runtime.
    """
    target_sum = np.sqrt(2.0 * n)                # Σ a_i = √(2n)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    best_local_seq = a.copy()
    best_local_score = np.inf

    start = time.time()
    deadline = start + runtime

    # Adam state (used after the first few sub‑gradient steps)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.02                               # base learning rate for Adam

    beta = 1.0                               # smoothing parameter
    beta_max = 30.0

    iteration = 0
    # main optimisation loop
    while time.time() < deadline:
        conv = _conv_full_fft(a)
        max_val = conv.max()

        # update best of this run
        if max_val < best_local_score:
            best_local_score = float(max_val)
            best_local_seq = a.copy()
            # early stop if we beat the global bound dramatically
            if best_local_score < 1.45:
                break

        # --- Phase selection -------------------------------------------------
        if iteration < 150:                     # short sub‑gradient warm‑up
            # treat a tiny absolute tolerance to collect all “active” ties
            eps_active = max(1e-12, max_val * 1e-9)
            active = np.where(conv >= max_val - eps_active)[0]
            if active.size == 0:
                # fallback random nudge
                a = _project_onto_simplex(a + np.random.randn(n) * 1e-4, target_sum)
                iteration += 1
                continue

            grad = _gradient_of_active(a, active)
            max_grad = grad.max()
            if max_grad <= 0:
                a = _project_onto_simplex(a + np.random.randn(n) * 1e-4, target_sum)
                iteration += 1
                continue

            # step size proportional to the current value / gradient
            step = 0.5 * max_val / (max_grad + 1e-12)
            a_new = a - step * grad
        else:
            # ----- smooth log‑sum‑exp phase (Adam) -----
            # increase smoothing slowly
            beta = min(beta * 1.018, beta_max)

            grad = _gradient_smoothed(a, conv, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (iteration + 1))
            v_hat = v / (1.0 - beta2 ** (iteration + 1))
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection back onto the feasible simplex -----------------------
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake (simulated‑annealing style)
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.004 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)

        a = a_new
        iteration += 1

    return best_local_seq, best_local_score


# --------------------------------------------------------------------------- #
# Final cheap hill‑climbing refinement (pairwise mass exchange).
# --------------------------------------------------------------------------- #
def _refine_hill_climb(a: np.ndarray, runtime: float):
    """
    Simple stochastic local search: move a tiny amount of mass from one index
    to another if it improves the maximal convolution value.
    """
    n = a.shape[0]
    target_sum = a.sum()
    best_seq = a.copy()
    best_score = np.max(_conv_full_fft(best_seq))

    start = time.time()
    while time.time() - start < runtime:
        i, j = np.random.randint(0, n, size=2)
        if i == j or best_seq[i] <= 1e-12:
            continue
        # move a random fraction of the mass at i to j
        eps = np.random.rand() * best_seq[i] * 0.3
        cand = best_seq.copy()
        cand[i] -= eps
        cand[j] += eps
        # projection is not needed – the L1 sum stays unchanged and positivity holds
        cand_score = np.max(_conv_full_fft(cand))
        if cand_score < best_score:
            best_seq, best_score = cand, cand_score
    return best_seq, best_score

    Returns the best sequence as a plain Python list.

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # If a previous construction is available, keep it for warm‑starting.
    if "height_sequence_1" in globals():
        prev_best = np.asarray(height_sequence_1, dtype=float)
    else:
        prev_best = None

    # ------------------------------------------------------------------- #
    # Outer loop: many independent restarts with different lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size in the range that worked well before.
        n = np.random.randint(1200, 3001)

        # Initialise either from the global best (if long enough) or randomly.
        if prev_best is not None and np.random.rand() < 0.35 and prev_best.size >= n:
            init = prev_best[:n].copy()
        else:
            init = np.random.rand(n)

        # Allocate a slice of the remaining budget for this run.
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        run_time = min(12.0, remaining * 0.9)   # keep a little slack for the final phase

        # Optimise one instance.
        seq_local, score_local = _run_one_instance(n, init, run_time, best_score)

        # Update the global best.
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Final refinement on the current global best (if time permits).
    # ------------------------------------------------------------------- #
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 2.0:
            refined_seq, refined_score = _refine_hill_climb(best_seq, time_left)
            if refined_score < best_score:
                best_seq = refined_seq
                best_score = refined_score
                print(f"[{time.time():.2f}] Refined best = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Fallback – return a uniform distribution if nothing was found.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to a plain Python list as required by the grader.
    return best_seq.tolist()
