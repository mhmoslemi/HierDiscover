def _mirrored_powerlaw_seq(n, p):
    idx = np.arange(n, dtype=np.float64)
    left = (idx + 1) ** (-p)
    right = (n - idx) ** (-p)
    seq = left + right
    return seq / seq.sum()

"""
Propose a step‐function (non‑negative sequence) by:
  * initializing from a diverse set of parametric families and random Dirichlet points,
  * optimising the smooth approximation of  max_k (a*a)[k] using Adam,
    where the smooth max is log‑sum‑exp with a temperature that is annealed,
  * projecting after each update onto the probability simplex (sum = 1, a_i ≥ 0),
  * keeping the best sequence seen under the true objective
    `evaluate_sequence` (i.e. 2·n·max(conv)/sum²).

The optimiser works with FFT based convolutions, so each iteration costs
O(n log n).  With n≈2500 many thousands of iterations fit easily into the
1000 s budget, giving a better bound than the previous simple sub‑gradient
method.
"""

# Simple utilities
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:          # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself (length 2·n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (sum is 1 after normalisation)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1 - α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a‑1} (n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _mirrored_powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    left = (i + 1) ** (-p)
    right = (n - i) ** (-p)
    seq = left + right
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Smooth max + gradient (log‑sum‑exp) ------------------------------------
def _smooth_loss_and_grad(seq: np.ndarray, temperature: float):
    """
    Returns
      loss   – smooth approximation of max_k (a*a)[k],
      grad   – gradient of that loss w.r.t. the sequence,
      cur_max – the actual maximum convolution entry (for the true objective).
    """
    n = seq.shape[0]
    conv = _self_convolve(seq)                 # length 2n‑1
    cur_max = conv.max()

    # stable log‑sum‑exp
    max_val = cur_max
    exp_vals = np.exp(temperature * (conv - max_val))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp                     # soft‑max weights, length 2n‑1

    # Gradient: g_i = 2 * Σ_k w_k * a_{k‑i}  (cross‑correlation of w and a)
    w_rev = w[::-1]                            # reversed weight vector
    m_len = len(w_rev) + n - 1                 # = 3n‑2
    nfft = _next_pow2(m_len)
    A = np.fft.rfft(seq, nfft)
    B = np.fft.rfft(w_rev, nfft)
    conv_rev = np.fft.irfft(A * B, n=nfft)[:m_len]

    idx = (2 * n - 2) - np.arange(n)           # mapping described in analysis
    grad = 2.0 * conv_rev[idx]

    # smooth loss = max + log‑sum‑exp / temperature
    loss = max_val + np.log(sum_exp) / temperature
    return loss, grad, cur_max


# ----------------------------------------------------------------------
# Adam optimiser on the simplex -----------------------------------------
def _adam_optimize(init_seq: np.ndarray, deadline: float,
                   lr: float = 0.2,
                   t0: float = 0.5,
                   t_max: float = 30.0,
                   t_inc: float = 0.02,
                   max_iters: int = 200_000):
    """
    Adam with a smoothed max objective, projection onto the simplex after
    each update.  Returns the best sequence and its exact objective value.
    """
    n = init_seq.shape[0]
    seq = init_seq.copy()
    # Adam state
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # current exact score (sum(seq)==1 ⇒ factor 2·n)
    cur_score = 2.0 * n * _self_convolve(seq).max()
    best_seq, best_score = seq.copy(), cur_score

    temperature = t0
    it = 1
    while time.time() < deadline and it <= max_iters:
        loss, grad, cur_max = _smooth_loss_and_grad(seq, temperature)

        # Adam update (learning‑rate multiplied after bias‑correction)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_onto_simplex(seq - step)

        cand_score = _objective(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[Adam {it:05d}] new global best = {best_score:.6f}")

        # accept if it really improves the true objective
        if cand_score < cur_score:
            seq = cand
            cur_score = cand_score
        else:
            # occasional small random mass transfer to escape stagnation
            if np.random.rand() < 0.02:
                i, j = np.random.choice(n, size=2, replace=False)
                delta = 0.01 * min(seq[i], 0.1)
                tmp = seq.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_onto_simplex(tmp)
                tmp_score = _objective(tmp)
                if tmp_score < cur_score:
                    seq = tmp
                    cur_score = tmp_score

        # anneal temperature to focus on the true max
        temperature = min(temperature + t_inc, t_max)
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Build a collection of initial candidates (parametric families, random)
    and run a fast Adam optimiser (smooth max) from each of them.
    The best sequence found before the budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length that balances expressiveness and speed.
    n = 2500

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to normalise, evaluate and optionally launch optimisation.
    def _handle_candidate(seq):
        nonlocal best_seq, best_score
        if seq is None:
            return
        seq = np.clip(seq, 0.0, 1000.0)
        if seq.sum() == 0.0:
            return
        seq = seq / seq.sum()
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Init] seed score = {score:.6f}")
        # Run the Adam optimiser from this seed if time permits.
        if time.time() < deadline:
            opt_seq, opt_score = _adam_optimize(seq, deadline)
            if opt_score < best_score:
                best_score = opt_score
                best_seq = opt_seq.copy()
                print(f"[Optim] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Seed from the global variable (if any)
    if "height_sequence_1" in globals():
        try:
            _handle_candidate(np.asarray(globals()["height_sequence_1"], dtype=np.float64))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Systematic sweep over parametric families
    #    (break early if the budget runs out).
    for alpha in np.linspace(0.5, 4.0, 12):
        if time.time() > deadline:
            break
        _handle_candidate(_geom_seq(n, alpha))

    for p in np.linspace(0.2, 2.5, 12):
        if time.time() > deadline:
            break
        _handle_candidate(_powerlaw_seq(n, p))

    for a in np.linspace(0.2, 3.0, 5):
        for b in np.linspace(0.2, 3.0, 5):
            if time.time() > deadline:
                break
            _handle_candidate(_beta_seq(n, a, b))

    for p in np.linspace(0.2, 2.5, 6):
        if time.time() > deadline:
            break
        _handle_candidate(_mirrored_powerlaw_seq(n, p))

    # 3. Random Dirichlet points (pure stochastic exploration)
    for _ in range(30):
        if time.time() > deadline:
            break
        _handle_candidate(_random_simplex(n))

    # ------------------------------------------------------------------
    # 4. Fallback – uniform distribution (always feasible)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
