def conv_full(a):

def smooth_max_and_grad(a, tau):
    conv = conv_full(a)
    # For numeric stability: shift by max.
    max_conv = conv.max()
    shifted = conv - max_conv
    exp_shifted = np.exp(tau * shifted)
    sum_exp = exp_shifted.sum()
    # weighting w without shift factor: w_i = exp_tau*(conv_i) / sum(exp)
    # Since we subtracted max, w_i = exp(tau*(conv_i-max)) / sum_exp.
    # Actually we need original w = exp(tau * conv) / sum(exp(tau*conv)).
    # So using shifted yields same after scaling.
    w = exp_shifted / sum_exp
    # compute gradient = 2 * conv(w, a) slice
    conv_wa = np.convolve(w, a)  # length (2n-1 + n -1) = 3n-2
    # grad indices n-1 to n-1 + n - 1 inclusive
    grad = 2.0 * conv_wa[n-1 : n-1 + len(a)]
    smooth = max_conv + (1.0 / tau) * np.log(sum_exp) # Actually smooth = (1/tau)*log( sum(exp(tau*conv)) )
    # Since we used shifted values: sum(exp(tau*(conv-max))) = exp(-tau*max)*sum(exp(tau*conv))
    # So smooth = (1/tau) * (log(exp(-tau*max)*sum(exp(tau*conv)))) = -max + (1/tau)*log(sum_exp)
    # Actually we need smooth = (1/tau)*log(sum(exp(tau*conv))) = max + (1/tau)*log(sum_exp) because sum(exp(tau*conv)) = exp(tau*max)*sum_exp => log = tau*max + log(sum_exp) => divide by tau => max + (1/tau)*log(sum_exp).
    # So we can compute smooth = max_conv + (1.0 / tau) * np.log(sum_exp).
    smooth = max_conv + (1.0 / tau) * np.log(sum_exp)
    return smooth, grad, max_conv

"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The code runs many independent restarts with different lengths.
Each restart performs a constrained Adam optimisation on a *smooth*
approximation of the max‑convolution (log‑sum‑exp).  After each gradient
step the iterate is projected onto the simplex
   {x_i ≥ 0 , Σ x_i = √(2 n)}.
The smoothness parameter τ is increased gradually so that the algorithm
first explores a large region (τ small) and then focuses on the true
max‑convolution (τ large).  The best sequence found before the time
budget expires is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper 1 – Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Helper 2 – smooth max (log‑sum‑exp) of the full convolution and its gradient
# --------------------------------------------------------------------------- #
def _smooth_max_and_grad(a: np.ndarray, tau: float):
    """
    Returns three objects for a non‑negative vector ``a`` and a smoothing
    parameter ``tau`` (> 0):

        smooth   –  (1/tau) * log Σ exp(tau * (a*a)_k)
        grad     –  ∂smooth/∂a  (same shape as ``a``)
        max_conv –  max_k (a*a)_k   (the exact objective value)
    """
    n = a.shape[0]

    # full convolution a * a  (length 2*n-1)
    conv = np.convolve(a, a)
    max_conv = conv.max()

    #--- weights for the log‑sum‑exp (numerically stable) --------------------#
    # w_i ∝ exp(tau * conv_i)
    shifted = conv - max_conv               # max_conv is subtracted to avoid overflow
    exp_shifted = np.exp(tau * shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp               # shape (2*n-1,)

    #--- gradient of smooth max ------------------------------------------------#
    # grad = 2 * (w * a)  where * denotes cross‑correlation.
    # Using the identity   grad_k = 2 * Σ_i w_{i+k} a_i,
    # which is obtained by a convolution of w with a.
    conv_wa = np.convolve(w, a)             # length 3*n-2
    grad = 2.0 * conv_wa[n - 1 : n - 1 + n]  # slice to obtain n components

    #--- smooth value ----------------------------------------------------------#
    # smooth = max_conv + (1/tau) * log Σ exp(tau * (conv_i - max_conv))
    smooth = max_conv + (1.0 / tau) * np.log(sum_exp)

    return smooth, grad, max_conv


# --------------------------------------------------------------------------- #
# Helper 3 – optimise a single instance (fixed length) using Adam
# --------------------------------------------------------------------------- #
def _optimise_instance(
    n: int,
    init_vec: np.ndarray,
    deadline: float,
):
    """
    Runs a constrained Adam optimisation for a given length ``n``.
    Returns the best sequence discovered (numpy array) together with its
    exact max‑convolution value.
    """
    target_sum = np.sqrt(2.0 * n)          # simplex radius
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.25                              # initial learning rate
    tau = 0.5                               # initial smoothing

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iteration = 0

    best_seq = a.copy()
    best_score = np.inf

    # run until the per‑instance deadline is hit
    while time.time() < deadline:
        smooth, grad, max_conv = _smooth_max_and_grad(a, tau)

        # keep the best exact objective seen so far
        if max_conv < best_score:
            best_score = float(max_conv)
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}]  Local best = {best_score:.6f} (n={n}, tau={tau:.2f})"
            )

        # ----- Adam update (minimise smooth) ---------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step                     # gradient descent

        # projection onto the simplex (ensures non‑negativity and Σa = √(2n))
        a = _project_onto_simplex(a, target_sum)

        # ----- schedules -------------------------------------------------------
        if iteration % 40 == 0:
            # increase smoothing (focus more on the true max)
            tau = min(tau * 1.25, 20.0)
            # slowly decay the learning rate
            lr *= 0.97

        # occasional random shake to escape shallow local minima
        if np.random.rand() < 0.005:
            noise = np.random.randn(n) * (target_sum * 0.001 / np.sqrt(n))
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # reset Adam moments after a shake
            m.fill(0.0)
            v.fill(0.0)

        # early exit if we already beat a very good bound
        if best_score < 1.504:
            break

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Searches for a step‑function with the smallest possible objective
    value inside the time budget ``budget_s`` (seconds).  The function
    returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid being cut off while finalising
    overall_deadline = start_time + budget_s - 0.5

    best_global_seq = None
    best_global_score = float("inf")

    # ------------------------------------------------------------------- #
    # Choose a set of problem sizes to explore.
    # Empirically sizes between 1500 and 3000 give a good trade‑off.
    # ------------------------------------------------------------------- #
    possible_ns = list(range(1500, 3001, 100))

    while time.time() < overall_deadline:
        # pick a length at random
        n = int(np.random.choice(possible_ns))

        # initialise: either perturb the previous best construction or start fresh
        if (
            np.random.rand() < 0.35
            and "height_sequence_1" in globals()
            and isinstance(height_sequence_1, (list, tuple, np.ndarray))
        ):
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                # pad with tiny random values
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # Dirichlet‑like random start
            init = np.random.rand(n)

        # give each optimisation run a modest time slice (≈ 8‑10 s)

        # stop early if we have already crossed the target bound
        if best_global_score < 1.503:

    # ------------------------------------------------------------------- #
    # Fallback – in the unlikely event that nothing was found.
    # ------------------------------------------------------------------- #
    if best_global_seq is None:
        n_fallback = 2000
        target_sum = np.sqrt(2.0 * n_fallback)
        best_global_seq = np.full(n_fallback, target_sum / n_fallback, dtype=float)

    # Ensure the return type is a plain list (as required by the grader)
    return best_global_seq.tolist()
