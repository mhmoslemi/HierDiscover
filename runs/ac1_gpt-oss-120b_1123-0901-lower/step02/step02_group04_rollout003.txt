def _lp_refinement(best_seq: np.ndarray, deadline: float) -> tuple:
    try:
        from scipy.optimize import linprog
    except Exception:
        return best_seq, _evaluate(best_seq.tolist())
    a = best_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    while time.time() < deadline - 0.5:

        M = conv.max()
        rows = 2 * n - 1
        # Build constraint matrix A (rows x n)
        # For each k, (a * g)_k = sum_{i} a_i * g_{k - i}
        # This is convolution of a and g. We can build rows efficiently.
        # Use Toeplitz: each row has a_i where col = k - i.
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            cols = np.arange(lo, hi + 1, dtype=int)
            A[k, cols] = a[k - cols]

        g = res.x
        s = g.sum()
        if s == 0:
            break
        g = g * (target / s)
        t = 0.2
        a = (1 - t) * a + t * g
        best_seq, best_score = _maybe_update_candidate(a, best_seq, _evaluate(a.tolist()))
        if best_score < 1.503:

"""
propose_candidate – Adam‑based smoothed max‑convolution minimiser

The evaluation metric for a non‑negative step‑function `a`
```
score = 2·n·max(conv(a,a)) / (sum(a)**2)
```
is equal to `max(conv(a,a))` when the sequence is normalised to
`sum(a)=√(2n)`.  Hence we minimise the infinity‑norm of the
autocorrelation.

The optimiser works in three stages:

1. **Cheap parametric sweep** – simple families (exponential,
   linear, quadratic, triangular) on several lengths.  Each candidate
   is normalised onto the simplex `Σa = √(2n)` and evaluated.

2. **Smooth Adam descent** – we minimise a smoothed version of the
   max‑convolution:
   ```
   M_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
   ```
   whose gradient is a weighted sum of all convolution entries.
   Adam (adaptive‑momentum) updates are applied to the variables,
   followed by projection onto the simplex.  The smoothing parameter
   `α` is annealed from 1 up to 60, allowing large steps early and
   fine tuning later.  The best sequence seen is kept.

3. **Optional LP refinement** – a single linear‑programming step
   (as in the reference SOTA) is performed if SciPy is available.

Random restarts, occasional sub‑gradient polishing and a final
fallback to a uniform sequence guarantee a valid return before the
time budget expires.  All helper functions are defined at module
level (no closures) and the routine respects the supplied `budget_s`
seconds.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto the simplex of non‑negative vectors with sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full convolution a * a (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# Helper: safe call to the external evaluator
# ----------------------------------------------------------------------
def _evaluate(seq: List[float]) -> float:
    """Return the score from `evaluate_sequence` or +inf on error."""
    try:
        return evaluate_sequence(seq)   # provided by the harness
    except Exception:
        return float("inf")

def _maybe_update_candidate(

    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


def _random_dirichlet_start(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with all α=1)."""
    x = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return x


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (softmax with parameter α)
# ----------------------------------------------------------------------
def _grad_softmax_max(a: np.ndarray, alpha: float) -> Tuple[np.ndarray, float]:
    """
    Compute gradient of  M_α(a) = (1/α)·log Σ exp(α·conv_k(a) )
    and also return the (unsmoothed) maximum convolution entry.
    """
    conv = _conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # stable softmax
    shift = conv.max()                           # subtract max to avoid overflow
    exp_term = np.exp(alpha * (conv - shift))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # weights that sum to 1

    # gradient = 2 * (w convolve a_rev)   – slice the centre n entries
    a_rev = a[::-1]
    grad_full = np.convolve(w, a_rev)            # full length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # extract the part that matches a
    return grad, max_conv


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective (no closures)
# ----------------------------------------------------------------------
def _adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> Tuple[np.ndarray, float]:
    """
    Adam descent on the soft‑max‑approximated max‑convolution.
    Returns the possibly improved (best_seq, best_score).
    """
    n = start_a.shape[0]
    target = math.sqrt(2.0 * n)

    # initialise on the simplex
    a = start_a.astype(float)
    a = a / a.sum() * target

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5

    # annealing schedule for α (smoothness)
    schedule = []
    for a_val, steps in [(1.0, 200), (3.0, 200), (10.0, 200), (30.0, 200), (60.0, 200)]:
        schedule.extend([a_val] * steps)

    it = 0
    while time.time() < deadline and it < len(schedule):
        alpha = schedule[it]

        grad, _ = _grad_softmax_max(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = base_lr / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_onto_simplex(a_new, target)

        if it % 5 == 0:
            best_seq, best_score = _maybe_update_candidate(a_new, best_seq, best_score)
            if best_score < 1.503:
                break

        a = a_new
        it += 1

    # one short sub‑gradient polishing phase (helps escape tiny plateaus)
    best_seq, best_score = _subgradient_refine(a, deadline, best_seq, best_score)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple sub‑gradient descent on the true max‑convolution (as fallback)
# ----------------------------------------------------------------------
def _subgradient_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv) – used as a final polish."""
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # momentum variables
    m = np.zeros_like(a)
    beta = 0.9
    base_step = 0.2
    it = 0

    while time.time() < deadline:
        conv = _conv_self_fft(a)
        kmax = int(np.argmax(conv))

        low = max(0, kmax - (n - 1))
        high = min(n - 1, kmax)
        idx = np.arange(low, high + 1, dtype=int)
        partner = kmax - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]          # gradient of the active max entry

        direction = -grad
        m = beta * m + (1.0 - beta) * direction

        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target)

        if it % 5 == 0:
            best_seq, best_score = _maybe_update_candidate(a_new, best_seq, best_score)
            if best_score < 1.503:
                break

        a = a_new
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional linear‑programming refinement (same idea as the SOTA description)
# ----------------------------------------------------------------------
def _lp_refinement(best_seq: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """Perform a single LP‑based refinement if SciPy is present."""
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy not available – nothing to do
        return best_seq, _evaluate(best_seq.tolist())

    a = best_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    while time.time() < deadline - 0.5:
        conv = _conv_self_fft(a)
        M = conv.max()

        rows = 2 * n - 1
        # Build A matrix for (a * g)_k ≤ M
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            cols = np.arange(lo, hi + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(
            c=-np.ones(n),               # maximise Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        g = g * (target / s)               # re‑scale to the simplex

        # convex combination (t modestly chosen)
        t = 0.2
        a = (1.0 - t) * a + t * g

        best_seq, best_score = _maybe_update_candidate(a, best_seq, _evaluate(a.tolist()))
        if best_score < 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main routine required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initial best (use the known construction if available)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    known = globals().get("height_sequence_1")
    if isinstance(known, (list, tuple)):
        known_arr = np.array(known, dtype=float)
        known_arr = np.maximum(0.0, known_arr)  # clip negatives just in case
        n_known = known_arr.shape[0]
        target_known = math.sqrt(2.0 * n_known)
        if known_arr.sum() > 0:
            known_arr = known_arr / known_arr.sum() * target_known
        else:
            known_arr = np.full(n_known, target_known / n_known)
        best_seq, best_score = _maybe_update_candidate(known_arr, None, float("inf"))

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric families
    # ------------------------------------------------------------------
    lengths = list(range(200, 1601, 200)) + [1326, 1500]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-4, -0.3, num=12):
            a = _shape_exp(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_candidate(a, best_seq, best_score)

        # reversed exponential
        for alpha in np.logspace(-4, -0.3, num=12):
            a = _shape_exp_rev(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_candidate(a, best_seq, best_score)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=10):
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_candidate(a, best_seq, best_score)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=10):
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_candidate(a, best_seq, best_score)

        # triangular (symmetric) shape
        a = _shape_triangular(n)
        a = a / a.sum() * target
        best_seq, best_score = _maybe_update_candidate(a, best_seq, best_score)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑based smooth optimisation with random restarts
    # ------------------------------------------------------------------
    if best_seq is not None:
        # polish the current best a bit
        best_seq, best_score = _adam_optimize(best_seq, deadline, best_seq, best_score)

    # random restarts while time permits
    while time.time() < deadline:
        # pick a random length (moderate to keep each run fast)
        n = np.random.randint(300, 2001)
        target = math.sqrt(2.0 * n)

        # initialise from a random Dirichlet point (dense) or a sparse exponential
        if np.random.rand() < 0.5:
            a0 = _random_dirichlet_start(n)
        else:
            alpha = np.random.rand() * 0.02 + 1e-5
            a0 = _shape_exp(n, alpha)

        a0 = a0 / a0.sum() * target

        # run Adam optimiser from this start
        best_seq, best_score = _adam_optimize(a0, deadline, best_seq, best_score)

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – optional LP refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _lp_refinement(best_seq, deadline)

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
