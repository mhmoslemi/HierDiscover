"""
Multi‑length stochastic Adam optimiser with random restarts and a cheap
local‑search phase.  The objective is the maximal value of the
autoconvolution (the evaluate_sequence score when the L1‑norm is fixed).
We explore several lengths, initialise from the supplied “height_sequence_1”
and from random Dirichlet draws, use a sub‑gradient of the max‑convolution
(and occasionally a random active shift) and project onto the feasible set
(non‑negative, fixed sum).  Occasionally small Gaussian jitter is added to
escape plateaus.  After the Adam loop a simple coordinate‑swap hill‑climb
tries to improve the solution further.  The best vector found before the
time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation a*a via FFT and returns the maximal value,
    its index and the whole convolution array.
    """
    n = a.shape[0]
    # n_fft is the next power of two >= 2*n-1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, returns the vector rev where rev[i] = a[k-i]
    if the index is inside the support, otherwise 0.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _subgradient_average(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Average reversed vector over all active shift indices.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= float(active.size)
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 5_000_000,
    jitter_interval: int = 4000,
    random_active_prob: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the max‑convolution objective for a fixed length.
    Returns possibly updated (best_seq, best_score).
    """
    # Normalise the start vector
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # Base learning rate – a modest value that works reasonably for many n
    lr0 = 0.05
    iteration = 0

    # Pre‑compute constant factor of the gradient (coeff = 2 after normalisation)
    coeff = 2.0
    sqrt2 = math.sqrt(2.0)

    while time.time() < deadline and iteration < max_iter:
        # ------------------------------------------------------------
        # 1) compute full auto‑convolution and active set
        M, _, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        # ------------------------------------------------------------
        # 2) sub‑gradient (random active shift with probability `random_active_prob`)
        if active.size and np.random.rand() < random_active_prob:
            k = np.random.choice(active)
            rev = _rev_for_shift(a, k)
        else:
            rev = _subgradient_average(a, active)

        # ------------------------------------------------------------
        # 3) gradient of the objective (M = max convolution, sum fixed)
        const = sqrt2 * M / math.sqrt(n)          # = sqrt(2) * M / sqrt(n)
        grad = coeff * rev - const                # shape (n,)

        # ------------------------------------------------------------
        # 4) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------
        # 5) projection onto feasible set (non‑negative + normalisation)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:  # totally unlikely, but keep a feasible vector
            a = np.full_like(a, target_sum / n)

        # ------------------------------------------------------------
        # 6) occasional jitter to break plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ------------------------------------------------------------
        # 7) current score (equal to M because of normalisation)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {iteration:7d}] new best score = {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that repeatedly moves a small amount of mass
    from one index to another (keeping the sum constant).  Accepts a move
    only if the maximal convolution value decreases.
    """
    a = a_start.copy()
    n_range = np.arange(n)
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        # pick a donor with positive mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # pick a receiver different from donor
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to transfer – up to 5 % of the donor's current mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0:
            continue

        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps

        # keep non‑negativity (donor already ensured)
        M_new, _, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (prop {proposal})")
    return a, best_score
