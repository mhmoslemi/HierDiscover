"""
Propose a non‑negative step‑function (a list of heights) whose
`evaluate_sequence` value is as small as possible.

Algorithm
---------
1. **Parameter sweep** – generate a set of simple analytic families
   (uniform, geometric, arcsine‑law and a one‑parameter power law
   a_i ∝ ((i+½)(n‑i‑½))^{-p}).  All candidates are normalised to sum = 1
   (the evaluation is scale‑invariant).  The best candidate of this
   cheap sweep is used as a starting point.

2. **Exact max‑convolution gradient descent** – the objective for a
   normalised sequence is `score = 2·n·max(a*a)`.  The sub‑gradient of
   `max(a*a)` at the current maximal shift `k*` is known analytically:
   `∂b_{k*}/∂a_i = 2·a_{k*‑i}` for all valid indices.
   We perform Adam‑style updates in the probability simplex
   (projection onto the simplex after each step).

3. **Exploration** – if no improvement is observed for many iterations
   a tiny Gaussian jitter is added and the momentum is reset.

4. **Termination** – the loop runs until the allotted time budget
   (default 1000 s) is exhausted or the target value `1.503` is reached.

The method is fast (a single convolution of length 2000 costs only a few
milliseconds) and focuses optimisation directly on the maximal
convolution entry, which yields a tighter descent direction than the
previous “top‑K” sub‑gradient approach.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        r = 1.0 - factor / n
        if r <= 0:
            continue
        seq = r ** np.arange(n, dtype=np.float64)
        cand.append(seq / seq.sum())

    i = np.arange(n, dtype=np.float64)

    cand.append(arcs / arcs.sum())

    for p in np.arange(0.1, 2.01, 0.1):
        seq = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** p
        cand.append(seq / seq.sum())

    # 5. a few random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # 6. possibly reuse a previously computed solution if it exists
    try:
        global height_sequence_1
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.shape[0] == n:
            cand.append(old / old.sum())
        else:
            # simple truncation / zero‑padding to the required length
            tmp = np.zeros(n, dtype=np.float64)
            m = min(old.shape[0], n)
            tmp[:m] = old[:m]
            if tmp.sum() == 0:
                tmp = np.full(n, 1.0 / n)
            cand.append(tmp / tmp.sum())
    except Exception:
        pass

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. problem size
    n = 2000                                 # keeps each convolution cheap

    # ------------------------------------------------------------------
    # 2. cheap sweep over analytic families
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        # protect against possible NaNs / infinities (should not happen)
        seq = np.clip(seq, 0.0, 1000.0)
        seq = seq / seq.sum()
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[Init] best score after seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. fine‑grained optimisation using the exact max‑convolution gradient
    seq = best_seq.copy()
    # Adam hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    # bookkeeping
    it = 0
    no_improve = 0
    cur_conv = np.convolve(seq, seq)
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max   # because sum(seq)=1

    while time.time() < deadline:
        # --------------------------------------------------------------
        # a) locate the maximal convolution entry and its exact gradient
        kmax = int(np.argmax(cur_conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(seq)
        grad[idx] = 2.0 * seq[kmax - idx]   # ∂max/∂a_i

        # b) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # c) take a step and project back onto the simplex
        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # d) evaluate new objective
        new_conv = np.convolve(new_seq, new_seq)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        # e) keep the globally best sequence
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"[Iter {it:06d}] new best = {best_score:.6f} (max={new_max:.6e})")
            no_improve = 0
        else:
            no_improve += 1

        # f) accept the new point if it does not increase the current score
        #    otherwise keep the old point and decay the learning rate a bit.
        if new_score <= cur_score + 1e-14:
            seq = new_seq
            cur_conv = new_conv
            cur_max = new_max
            cur_score = new_score
        else:
            lr *= 0.99   # modest decay when we step uphill

        # g) occasional jitter to escape plateaus
        if no_improve > 20000:
            jitter = rng.normal(scale=0.001, size=n)
            seq = seq + jitter
            seq = np.clip(seq, 0.0, None)
            if seq.sum() == 0:
                seq = np.full_like(seq, 1.0 / n)
            else:
                seq = seq / seq.sum()
            cur_conv = np.convolve(seq, seq)
            cur_max = cur_conv.max()
            cur_score = 2.0 * n * cur_max
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Iter {it:06d}] stagnation – random jitter applied")

        # h) tiny learning‑rate schedule (slow decay)
        if (it + 1) % 1000 == 0:
            lr *= 0.9995

        # i) early exit if we already beat the target
        if best_score < 1.503:
            print(f"Target reached at iteration {it}, best_score={best_score:.6f}")
            break

        it += 1

    print(f"Finished after {it} iterations. Best objective = {best_score:.6f}")
    # ensure the returned list obeys the required format
    return best_seq.tolist()
