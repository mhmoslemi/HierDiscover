"""
Propose a step‑function (list of non‑negative heights) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation function is scale‑invariant, therefore we keep the
ℓ₁‑norm fixed to √(2 n) (simplex projection).  The optimiser consists of
three stages:

1. **Fast parameter sweep** – generate a diverse set of parametric shapes
   (uniform, triangular, Gaussian, power‑law, random) for several lengths,
   evaluate them and keep the best few.

2. **Smooth‑max descent** – minimise a differentiable approximation of the
   ``max`` using a soft‑max (log‑sum‑exp) with an annealing schedule for the
   smoothing parameter λ.  The gradient of the smooth objective is obtained
   analytically via a convolution, then projected onto the simplex.

3. **Polish** – short projected‑gradient descent on the true max (as in the
   reference code) and an optional LP‑refinement step (if `cvxpy` is present).

The whole routine runs for at most the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helpers for mutation / resizing (reuse from the previous solution)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Projected gradient descent on the true max (same as the reference code)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.05,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """Short projected‑gradient descent (sub‑gradient of the maximal entry)."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()
    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)
        mean_g = np.mean(grad)
        d = -grad + mean_g                 # keeps the sum constant
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs
        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})")

        # occasional shake‑up
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
            step *= 0.9
        else:
            a = a_new

        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score

def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j subject to g ≥ 0 and f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """Few LP‑updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        M = float(np.convolve(a, a).max())
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break
        g = g0 * (target_sum / np.sum(g0))
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})")
    return best_seq, best_score

# Smooth‑max descent (core new idea)
# ----------------------------------------------------------------------
def _smooth_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    lambda_start: float = 0.1,
    lambda_end: float = 20.0,
    base_step: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of max(conv(a,a)):
    M_λ(a) = (1/λ) * log( Σ_i exp( λ * conv_i ) ).
    The gradient is computed via a convolution, then projected onto the simplex.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    n = a.size
    t0 = time.time()
    it = 0
    # linear schedule for λ (slowly increase)
    while time.time() < t0 + time_limit:
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # smoothly increase λ
        frac = min(1.0, it / 2000.0)           # ramp over ~2000 iterations
        lam = lambda_start + (lambda_end - lambda_start) * frac

        # soft‑max weights (numerically stable)
        shifted = conv - max_conv
        exp_vals = np.exp(lam * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth objective:
        #   ∇ = 2 * (a  *  w_rev)[n-1:2n-1][::-1]
        w_rev = w[::-1]
        grad_full = np.convolve(a, w_rev)          # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n][::-1]

        # project direction onto the tangent space (sum stays constant)
        mean_g = np.mean(grad)
        d = -grad + mean_g

        max_abs = np.max(np.abs(d))
        if max_abs == 0:
            break
        eta = base_step * max_conv / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] smooth descent new best = {score:.6f} (n={n})")

        a = a_new
        it += 1
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found (as a Python list of floats).  It combines a cheap
    parametric sweep, a smooth‑max descent, a short projected‑gradient
    refinement and, when possible, an LP‑based update.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # Global best holder (may be initialised from a historic solution)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Started from historic seq, score={sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # Fallback: uniform vector of length 1000
    if best_seq is None:
        n0 = 1000
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic seq – using uniform, score={best_score:.6f}")

    # helper to update the global best safely
    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score
        try:
            sc = float(evaluate_sequence(seq.tolist()))
        except Exception:
            return
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST = {sc:.6f} (n={len(seq)})")

    # ------------------------------------------------------------------
    # Phase 1 – quick parametric sweep (≈ 10 % of the budget)
    # ------------------------------------------------------------------
    param_sweep_budget = min(120.0, budget_s * 0.1)
    sweep_deadline = time.time() + param_sweep_budget
    lengths = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 1800]

    for n in lengths:
        if time.time() > sweep_deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        seeds = []

        # uniform
        seeds.append(np.full(n, target_sum / n, dtype=float))

        # triangular (mountain)
        tri = np.minimum(np.arange(n) + 1, np.arange(n)[::-1] + 1).astype(float)
        tri = tri / tri.sum() * target_sum
        seeds.append(tri)

        # Gaussian families
        x = np.arange(n)
        mu = (n - 1) / 2.0
        for sigma_factor in (0.12, 0.18, 0.25, 0.35):
            sigma = sigma_factor * n
            gauss = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            if gauss.sum() == 0:
                continue
            gauss = gauss / gauss.sum() * target_sum
            seeds.append(gauss)

        # Power‑law / symmetric shapes
        for p in (-1.8, -1.2, -0.6, 0.0, 0.6, 1.2, 1.8):
            shape = (np.arange(n) + 1) ** p * (n - np.arange(n)) ** p
            shape = shape / shape.sum() * target_sum
            seeds.append(shape)

        # a few random points
        for _ in range(3):
            rnd = np.random.rand(n)
            rnd = rnd / rnd.sum() * target_sum
            seeds.append(rnd)

        for seq in seeds:
            if time.time() > sweep_deadline:
                break
            _maybe_update(seq)

    # ------------------------------------------------------------------
    # Phase 2 – iterative refinement (the bulk of the budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # Choose a length (bias towards the current best length)
        if best_seq is not None and np.random.rand() < 0.6:
            n = best_seq.size
            # occasional jitter in length
            if np.random.rand() < 0.3:
                n = int(n * np.random.uniform(0.9, 1.1))
                n = max(200, min(2000, n))
        else:
            n = np.random.randint(200, 2001)

        target_sum = math.sqrt(2.0 * n)

        # Initialise: resize the current global best or fall back to a random shape
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)
            init = init / init.sum() * target_sum

        # Allocate a slice of the remaining time for this iteration
        time_left = deadline - time.time()
        if time_left < 8.0:
            break
        # 1) Smooth‑max descent (most of the allocated time)
        smooth_time = min(0.6 * time_left, 30.0)
        seq_smooth, _ = _smooth_descent(init, target_sum, smooth_time)

        _maybe_update(seq_smooth)

        # 2) Short projected‑gradient descent on the true max (polish)
        remain = deadline - time.time()
        if remain < 5.0:
            break
        seq_local, _ = _local_descent(
            seq_smooth,
            target_sum,
            min(8.0, remain),
            base_step=0.07,
            print_tag="polish",
        )
        _maybe_update(seq_local)

        # 3) Optional LP refinement (cheap for modest n)
        if _CVX_AVAILABLE and n <= 600 and np.random.rand() < 0.25:
            remain = deadline - time.time()
            if remain > 4.0:
                seq_lp, _ = _lp_refine(seq_local, target_sum, min(6.0, remain))
                _maybe_update(seq_lp)

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
