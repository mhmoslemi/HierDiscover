def _full_conv_fft(x, y):
    m = len(x) + len(y) - 1
    size = 1 << ((m - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)[:m]
    return conv

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm uses a **smooth‑max** (soft‑max) surrogate for the
`max(conv)` term.  At each iteration we

  1. compute the full self‑convolution a*a (via an FFT),
  2. form soft‑max weights w_k ∝ exp(λ·conv_k)   (λ grows during the run),
  3. compute the gradient of the smooth‑max,
     ∇ ≈ 2·corr(a, w)  (cross‑correlation of a with the weights),
  4. take a momentum‑accelerated step opposite to the gradient,
  5. project back onto the simplex {x≥0, Σx = √(2n)}.

The smooth‑max starts with a small λ (almost uniform weights) and is
slowly annealed to a large λ, thereby turning the surrogate into the
true max while keeping the optimisation landscape smooth.
Several independent restarts (different lengths, exponents, random
jitter and occasional warm‑starts from the current best) are performed
within the allotted 1000 s budget.  The best sequence found is returned
as a plain Python list.

A cheap local “mass‑transfer” hill‑climber is applied at the end to
polish the final candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of L1‑radius s (v ≥ 0, sum(v) = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – full (linear) convolution via FFT
# ----------------------------------------------------------------------
def _full_conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT (full length)."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << ((m - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)          # numerical safety
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length) using an FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max over convolution entries)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute ∇_a  (log‑sum‑exp(λ·conv))/λ  ≈ 2·corr(a, softmax(λ·conv)).
    """
    # soft‑max weights; subtract max for numerical stability
    max_c = conv.max()
    exps = np.exp(lam * (conv - max_c))
    w = exps / exps.sum()                     # shape (2n‑1,)

    # cross‑correlation a ⋆ w = conv(reversed a, w)
    a_rev = a[::-1]
    corr = _full_conv_fft(a_rev, w)          # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * corr[n - 1:n - 1 + n]       # slice of length n
    return grad


# ----------------------------------------------------------------------
# Core optimisation for a single length using smooth‑max descent
# ----------------------------------------------------------------------
def _optimise_one_length_smooth(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    lam_start: float = 0.01,
    lam_end: float = 15.0,
    base_step: float = 0.25,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max (soft‑max) gradient descent on the max‑convolution objective.
    Returns the best vector found and its (approximate) score (= max(conv)).
    """
    target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2n) → objective = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best observed in this run
    conv = _conv_self_fft(a)
    best_local = a.copy()
    best_score = conv.max()

    # momentum buffer
    velocity = np.zeros_like(a)

    step_factor = base_step
    start_time = time.time()
    iteration = 0

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) current convolution and annealed λ
        # --------------------------------------------------------------
        conv = _conv_self_fft(a)
        max_val = conv.max()
        # linear schedule for λ based on elapsed fraction of the run
        elapsed = time.time() - start_time
        total = run_deadline - start_time
        lam = lam_start + (lam_end - lam_start) * min(elapsed / total, 1.0)

        # --------------------------------------------------------------
        # 2) smooth‑max gradient
        # --------------------------------------------------------------
        grad = _smooth_max_gradient(a, conv, lam)

        # --------------------------------------------------------------
        # 3) momentum + adaptive step size (scale by max gradient)
        # --------------------------------------------------------------
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity)) + 1e-12
        eta = step_factor * max_val / max_grad           # scale step to current max
        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 4) occasional random shake‑up (helps escaping plateaus)
        # --------------------------------------------------------------
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.94

        # --------------------------------------------------------------
        # 5) evaluate new candidate
        # --------------------------------------------------------------
        conv_new = _conv_self_fft(a_new)
        score_new = conv_new.max()

        # --------------------------------------------------------------
        # 6) keep the best of this run
        # --------------------------------------------------------------
        if score_new < best_score:
            best_score = score_new
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  smooth‑best = {best_score:.6f}")

        a = a_new
        step_factor *= 0.9996
        iteration += 1

    return best_local, best_score


# ----------------------------------------------------------------------
# Very cheap local refinement (mass‑transfer hill climbing)
# ----------------------------------------------------------------------
def _local_refine(seq: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between random coordinates to improve max(conv)."""
    start = time.time()
    a = seq.copy()
    n = a.shape[0]
    target_sum = a.sum()          # should already be √(2n)
    best_score = _conv_self_fft(a).max()

    while time.time() - start < time_budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of mass from i to j
        delta = np.random.rand() * a[i] * 0.02
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue                     # keep non‑negativity
        # No need to renormalise – transfer preserves total sum
        score = _conv_self_fft(a_new).max()
        if score < best_score:
            best_score = score
            a = a_new
            print(f"[{time.time():.2f}] local‑refine best = {best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine runs for at most `budget_s` seconds and returns
    the best sequence discovered.

    Strategy
    ----------
    * Randomly pick a length n (800 … 6000).  Larger n gives a richer
      search space but each iteration costs O(n log n); the schedule
      adapts to the remaining time.
    * Initialise with a power‑law a_i ∝ (i+1)^‑p (p≈0.5) plus a tiny jitter.
    * Run the smooth‑max gradient descent described above.  The smoothing
      parameter λ is annealed from a small value to a large one, turning
      the surrogate into the true max while keeping gradients useful.
    * Keep the globally best vector across many independent restarts.
    * Finally, spend a few seconds polishing the best candidate with a
      simple mass‑transfer hill‑climber.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Global restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # keep runs cheap when we are close to the end
        if remaining < 30.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(800, 6000)

        # ----- initialise -------------------------------------------------
        p = np.random.uniform(0.44, 0.57)               # a bit broader than before
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p)

        # tiny random jitter to break symmetry
        init += np.random.rand(n) * 1e-4

        # warm‑start from current best with some probability
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        # ----- allocate a time slice for this run -----------------------
        run_deadline = min(deadline, time.time() + np.random.uniform(8.0, 14.0))

        # ----- optimise --------------------------------------------------
        seq_local, score_local = _optimise_one_length_smooth(
            n,
            init,
            run_deadline,
            lam_start=0.01,
            lam_end=12.0,
            base_step=0.22,
            momentum_beta=0.9,
        )

        # ----- update global best ----------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # early exit if we have already beaten the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final polishing with a cheap local hill‑climber (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None:
        refine_time = min(30.0, deadline - time.time())
        if refine_time > 0.5:
            best_seq, best_score = _local_refine(best_seq, refine_time)
            print(f"[{time.time():.2f}] AFTER refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # Return a plain Python list (the evaluator expects this).
    return best_seq.tolist()
