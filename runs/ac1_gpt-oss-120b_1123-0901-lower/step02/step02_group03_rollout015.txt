    S = np.sqrt(2.0 * n)
    a = project_onto_simplex(init_vec.astype(float), S)
    best_local_seq = a.copy()
    best_local_score = float('inf')
    # hyperparameters
    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    start = time.time()
    while time.time() - start < time_limit:
        conv = conv_fft(a)   # length 2n-1
        max_val = conv.max()
        # compute objective (same as max_val, due to normalization)
        if max_val < best_local_score:
            best_local_score = max_val
            best_local_seq = a.copy()
        # compute active indices (ties)
        eps_active = 1e-12
        active = np.where(conv >= max_val - eps_active)[0]
        # gradient of active entries
        grad = gradient_active(a, active)  # shape (n,)
        # Adam update
        t_iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step   # move opposite direction
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, S)
        # random perturb occasionally
        if np.random.rand() < 0.05:
            noise = np.random.randn(n) * (0.02 * S / n)
            a_new = project_onto_simplex(np.maximum(a_new + noise, 0.0), S)
            lr *= 0.95
        a = a_new
        # decay lr optionally
        lr *= 0.9999
    return best_local_seq, best_local_score

"""
Propose a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation improves the previous method by:
* Using the Adam optimiser (adaptive learning rates) on the sub‑gradient
  of the *active* (largest) convolution entries.
* Updating all of the top‑K convolution values (not only the exact max)
  to flatten the whole convolution profile.
* Adding occasional Gaussian “shakes” and a simple random hill‑climbing
  refinement in the last seconds of the budget.
* Keeping every iterate on the simplex  Σa_i = √(2 n)  by Euclidean projection.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – Σx_i = s, x_i ≥ 0
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast (real) FFT‑based full convolution a * a  → length 2·n‑1
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the (sub‑)gradient of a set of convolution entries.
# For a single index k:  ∂conv[k]/∂a_i = 2 * a_{k-i} (if the partner is inside)
# --------------------------------------------------------------------------- #
def _gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# One optimisation run on a fixed length.
# Returns the best sequence found during the run and its associated score.
# --------------------------------------------------------------------------- #
def _run_one_instance(
    n: int,
    init_vec: np.ndarray,
    end_time: float,
) -> tuple[np.ndarray, float]:
    # target sum = √(2 n)  → the objective equals the maximal convolution entry
    target_sum = np.sqrt(2.0 * n)

    # start from a feasible point on the simplex
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    best_local = a.copy()
    best_score = float("inf")

    # Adam hyper‑parameters
    lr = 0.35
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    # we keep the *top‑K* convolution values (K ≈ 0.2 % of all entries)
    top_k = max(1, int(0.002 * (2 * n - 1)))

    while time.time() < end_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_local = a.copy()

        # indices of the K largest convolution entries
        if top_k == 1:
            active = np.array([np.argmax(conv)], dtype=int)
        else:
            # argpartition is O(N) and gives the K biggest values (unsorted)
            active = np.argpartition(-conv, top_k - 1)[:top_k]

        # sub‑gradient for the selected entries
        grad = _gradient_of_active(a, active)

        # Adam update (move opposite to the sub‑gradient)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # descend
        a = np.maximum(a, 0.0)           # feasibility
        a = _project_onto_simplex(a, target_sum)

        # occasional random perturbation (simulated‑annealing style)
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, target_sum)
            lr *= 0.92                    # slowly shrink the step size after a shake

        # gentle decay of the base learning rate
        lr *= 0.99995

    return best_local, best_score


# --------------------------------------------------------------------------- #
# Small random hill‑climbing refinement that stays on the simplex.
# --------------------------------------------------------------------------- #
def _refine_by_random_swaps(a: np.ndarray, end_time: float) -> np.ndarray:
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    cur = a.copy()
    cur_conv = _conv_full_fft(cur)
    cur_score = cur_conv.max()

    while time.time() < end_time:
        # pick two distinct indices, move a small amount from i → j
        i, j = np.random.choice(n, 2, replace=False)
        if cur[i] <= 1e-12:
            continue
        delta = cur[i] * (np.random.rand() * 0.03 + 0.01)  # 1‑3 % of the mass at i
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # feasibility (no need to re‑project – sum is preserved)
        cand = np.maximum(cand, 0.0)
        # re‑project only if the sum deviated due to numerical noise
        cand_sum = cand.sum()
        if abs(cand_sum - target_sum) > 1e-9:
            cand = _project_onto_simplex(cand, target_sum)

        cand_conv = _conv_full_fft(cand)
        cand_score = cand_conv.max()
        if cand_score + 1e-12 < cur_score:          # accept only improvements
            cur = cand
            cur_score = cand_score
            # keep the best sequence seen so far (for logging)
    return cur


# --------------------------------------------------------------------------- #
# Main entry point – called by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5                     # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------- #
    # Outer loop: many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 6.0:                # not enough time for another full run
            break

        # Length choice – we focus on the interval that gave good results earlier.
        n = np.random.randint(1200, 3001)

        # Initial vector – mixture of random, Dirichlet and a warm‑start.
        if best_seq is not None and np.random.rand() < 0.3:
            src = np.asarray(best_seq, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # pick a random distribution for the start
            mode = np.random.choice(["uniform", "dirichlet", "exp"])
            if mode == "uniform":
                init = np.random.rand(n)
            elif mode == "dirichlet":
                alpha = np.random.rand() * 4 + 0.1
                init = np.random.dirichlet(np.ones(n) * alpha)
            else:  # exponential
                scale = np.random.rand() * 2 + 0.2
                init = np.random.exponential(scale, size=n)

        # Allocate a slice of the remaining budget to this optimisation run.
        run_time = min(12.0, remaining * 0.7)
        seq, score = _run_one_instance(n, init, time.time() + run_time)

        # Quick check with the official evaluator (guards against any hidden changes)
        eval_score = evaluate_sequence(seq.tolist())
        if eval_score < best_score:
            best_score = eval_score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")

        # If we are already under the target we may spend the rest of the budget on refinement.
        if best_score < 1.48:
            # Use the remaining time for a fine‑grained random‑swap refinement.
            refine_end = deadline
            best_seq = _refine_by_random_swaps(best_seq, refine_end)
            # final evaluation
            best_score = evaluate_sequence(best_seq.tolist())
            print(f"[{time.time():.2f}] Refined best = {best_score:.6f} (n={len(best_seq)})")
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (never should be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # expose the best sequence as a global variable for possible reuse
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    return best_seq.tolist()
