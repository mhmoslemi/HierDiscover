"""
Propose a non‑negative step‑function (list of heights) that minimizes the
objective 2·n·max(conv)/sum(a)^2.

Strategy
--------
1. **Explore many lengths** – try n ∈ {1200,1500,2000,2500,3000,4000,5000}.
   Larger n gives more degrees of freedom but also scales the objective,
   therefore we keep the best across all lengths.

2. **Wide parametric seeding** – for each length generate candidates from
   several families:
   * uniform,
   * beta‑like shapes a_i ∝ (i+1)^{a‑1}(n‑i)^{b‑1} with a,b in a broad range
     (including very small exponents → U‑shaped mass),
   * geometric progressions (decreasing and mirrored increasing),
   * piece‑wise constant blocks,
   * alternating even/odd pattern,
   * Dirichlet samples with various concentration parameters.

3. **Fast local optimisation** – starting from the best seed we run a
   projected sub‑gradient descent on the simplex:
   * compute the active shift k* where the self‑convolution attains its maximum,
   * form the sub‑gradient g_i = a_{k*‑i},
   * backtracking line‑search and projection onto the probability simplex,
   * when the line‑search fails, execute a small random mass‑transfer move.
   The step size is adapted dynamically; occasional random moves help escape
   shallow local minima.

4. **Time‑budget aware** – the whole routine stops before the supplied budget
   (default 1000 s) and returns the best sequence found.

All helper functions are top‑level, use only numpy & time, and avoid
closures or lambdas as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself (length 2n‑1) using a real FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Local objective (identical to evaluate_sequence up to clipping)
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv)/sum(seq)^2 for a non‑negative sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1-α/n (α>0)."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    if np.all(powers == 0):
        return None
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a‑1}(n‑i)^{b‑1} (a,b>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # Using np.power with explicit dtype to avoid overflow/underflow warnings.
    left = np.power(i, a - 1.0, where=True)
    right = np.power(n + 1 - i, b - 1.0, where=True)
    seq = left * right
    # Remove non‑finite values that may arise for extreme exponents.
    seq[~np.isfinite(seq)] = 0.0
    total = seq.sum()
    if total <= 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    return seq / total


def _random_simplex(n: int, concentration: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) point on the simplex (α = concentration)."""
    if concentration <= 0.0:
        concentration = 1.0
    x = np.random.gamma(concentration, 1.0, size=n)
    return x / x.sum()
