"""
Search for a non‑negative step‑function that minimizes the provided
`evaluate_sequence` objective.

The algorithm works in two phases:

1. **Fast parametric sweep** – generate many candidate sequences from
   simple families (uniform, symmetric Beta, power‑law, mixtures and
   Dirichlet draws).  Convolution maxima are obtained with an FFT
   (O(n log n)) and the objective is evaluated with the scale‑invariant
   formula `2·n·max(b)`.  The best candidate from this sweep seeds the
   second phase.

2. **Projected gradient descent** – improve the seed by following the
   sub‑gradient of the currently maximal convolution entry
   (the true gradient of the max‑norm).  A heavy‑ball momentum term is
   used, the step size decays slowly, and occasional small Gaussian
   perturbations are added when progress stalls.  After each update the
   iterate is projected onto the probability simplex (non‑negative,
   sum = 1) to respect the problem constraints.

The whole search respects the 1000 s time budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution_with_index(seq):
    """
    Compute the full self‑convolution of `seq` using FFT and return
    (max_value, max_index).  Length of `seq` must be a 1‑D ndarray.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of 2
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    conv = conv[: 2 * n - 1]                 # keep the true length
    idx = int(np.argmax(conv))
    return conv[idx], idx

def _evaluate_fast(seq):
    """Scale‑invariant evaluation: 2·n·max(b) (sum(seq) is assumed =1)."""
    n = seq.shape[0]
    max_b, _ = _max_convolution_with_index(seq)
    s = seq.sum()
    if s < 0.01:                 # guard against numerical loss of mass
        return np.inf
    return 2.0 * n * max_b / (s * s)

        seq = new_seq
        it += 1

    # ------------------------------------------------------------------
    # 5. Final reporting
    final_score = _evaluate_fast(best_seq)
    print(f"Finished. Best objective = {final_score:.6f} (length {n})")
    return best_seq.tolist()
