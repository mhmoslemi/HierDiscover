"""
Propose a step‑function (non‑negative heights) that minimises

      2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective reduces to  max(conv(a,a)) when the vector is projected onto the
simplex Σa_i = √(2n).  The optimiser is a *soft‑max* projected‑gradient descent:

*  At each iteration the convolution is computed.
*  A smooth approximation of the maximum – a soft‑max with a temperature α –
   yields a dense sub‑gradient (weighted sum of the gradients of *all*
   convolution entries).  The gradient is obtained efficiently via a single
   convolution with the weight vector.
*  The step direction is projected back onto the simplex to keep the sum
   constraint and non‑negativity.
*  α is increased slowly so the method evolves from a global “average‑conv”
   descent to a pure max‑conv descent.  The step size decays, occasional
   random mutations and short restarts help escape local minima.
*  Several short runs with different lengths are performed until the time
   budget expires.  The best sequence seen is returned.

The code contains only top‑level functions (no closures) and uses only
numpy, math and time – no external optimisation packages are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return max(a * a) – the true objective when Σa = √(2n)."""
    conv = np.convolve(a, a)
    return float(conv.max())


def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient for the active entry (sub‑gradient of the max)
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Small mutation helper (adds Gaussian noise and re‑projects)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Resize helper – linear interpolation / down‑sampling
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Soft‑max projected gradient descent
# ----------------------------------------------------------------------
def _local_descent_softmax(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.2,
    base_alpha: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Perform a soft‑max based projected gradient descent.
    Returns the best sequence seen and its score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    cur_score = _objective(a)
    best_seq = a.copy()
    best_score = cur_score

    step = base_step
    alpha = base_alpha
    it = 0
    start = time.time()
    n = a.shape[0]

    while time.time() < start + time_limit:
        # ------------------------------------------------------------------
        # compute convolution and soft‑max weights
        # ------------------------------------------------------------------
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # soft‑max weights (max entry gets weight 1, others ≤1)
        w = np.exp(alpha * (conv - max_conv))
        sum_w = w.sum()

        # gradient of the smooth approximation:
        #   grad = 2 * (w * a_rev) convolution / sum_w
        grad_num = np.convolve(w, a[::-1])
        # slice the middle part containing the n useful entries
        mid = n - 1                 # = (len(w) // 2)
        grad_soft = 2.0 * grad_num[mid:mid + n] / sum_w

        # also compute the exact max‑entry sub‑gradient
        _, grad_max, _, _ = _max_conv_and_grad(a)

        # linearly blend from max‑only to soft‑max as alpha grows
        blend = min(1.0, max(0.0, (alpha - base_alpha) / 5.0))
        grad = (1.0 - blend) * grad_max + blend * grad_soft

        # ------------------------------------------------------------------
        # step & projection
        # ------------------------------------------------------------------
        direction = -grad
        norm = np.linalg.norm(direction)
        if norm < 1e-12:          # nothing to do
            break

        eta = step * max_conv / (norm + 1e-12)
        a_new = a + eta * direction
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _objective(a_new)

        # keep the running best
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] soft‑desc new local best = {new_score:.6f} (n={n})"
            )

        # accept if improvement; otherwise very small uphill probability
        if new_score <= cur_score or np.random.rand() < 0.01:
            a = a_new
            cur_score = new_score

        # occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / n
            a = _mutate(a, target_sum, sigma)

        # schedule adaptations
        if it % 200 == 0 and it > 0:
            step *= 0.9
            alpha = min(alpha * 1.1, 10.0)   # cap to avoid extreme exponentials
        it += 1

    return best_seq, best_score

        # pick a length – usually stay close to the current best,
        # but allow moderate exploration
        if np.random.rand() < 0.7 and best_seq is not None:
            n = best_seq.size
            if np.random.rand() < 0.3:
                factor = np.random.uniform(0.85, 1.15)
                n = int(max(200, min(3000, n * factor)))
        else:
            n = np.random.randint(200, 2500)

        target_sum = math.sqrt(2.0 * n)

        # initialise candidate (resize + optional small perturbation)
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        if np.random.rand() < 0.4:
            init = _mutate(init, target_sum, sigma=0.04 * target_sum / n)

        # allocate a slice of the remaining time for this descent run
        time_left = deadline - time.time()
        if time_left < 5.0:
            break
        run_time = min(time_left, 9.0)       # keep runs short for many restarts

        cand_seq, cand_score = _local_descent_softmax(init, target_sum, run_time)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            # early exit if we beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
