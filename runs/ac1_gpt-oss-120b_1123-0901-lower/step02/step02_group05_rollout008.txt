def _softmax_gradient_descent(init, target_sum, time_limit, lam_start=0.5, lam_mul=1.02, lam_max=20.0, base_step=0.5):
    a = _project_onto_simplex(init.astype(float), target_sum)
    best = a.copy()
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf

    t0 = time.time()
    lam = lam_start
    step_fac = base_step
    it = 0
    while time.time() < t0 + time_limit:
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # Compute softmax weights
        # Use stable computation:
        shifted = conv - conv.max()
        exp_vals = np.exp(lam * shifted)
        w = exp_vals / exp_vals.sum()
        # gradient via convolution
        a_rev = a[::-1]
        grad_full = np.convolve(a_rev, w)  # length 3n-2
        grad = 2.0 * grad_full[a.size-1 : a.size-1 + a.size]  # slice length n

        # Determine step size
        max_abs = np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else 1.0
        eta = step_fac * max_conv / max_abs

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate only occasionally to reduce overhead? We'll evaluate each iteration maybe.
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best = a_new.copy()
            print(f"[{time.time():.2f}] Softmax n={len(a)} λ={lam:.2f} new best={score:.6f}")

        # Update
        a = a_new

        lam = min(lam_max, lam * lam_mul)

        it += 1
        if it % 200 == 0:
            step_fac *= 0.9

    return best, best_score

"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluator is invariant under scaling, so we enforce the
normalisation  Σ a_i = √(2 n).  The search uses two complementary
optimisers:

1. **Soft‑max gradient descent** – a smooth approximation of the
   “max‑convolution’’ objective.
   We minimise   f_λ(a) = (1/λ)·log(∑_k exp(λ·(a∗a)_k))
   and slowly increase λ (continuation) so that the landscape
   becomes increasingly like the true max.  The gradient of f_λ
   is a weighted sum of the gradients of every convolution entry,
   which can be obtained cheaply via a single convolution.

2. **Projected sub‑gradient descent** – after the soft‑max phase a
   few iterations of the classic sub‑gradient method (using the
   gradient of the currently maximal entry) finish the optimisation.

Both phases keep the vector in the simplex {x≥0, Σx = √(2n)}.
The outer loop repeatedly creates candidates of varying length,
initialises them either from the current best or from a random
Dirichlet draw, runs the two phases within a time slice and keeps
the globally best sequence found.

The routine respects the 1000 s budget, prints progress and finally
returns the best list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry (used for final refinement)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the sub‑gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # Gradient of the k‑th convolution entry w.r.t. a:
    #   ∂(a∗a)_k / ∂a_i = 2 * a_{k-i}   (if the partner index is inside the array)
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Soft‑max (smooth‑max) gradient descent
# ----------------------------------------------------------------------
def _softmax_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    lam_start: float = 0.5,
    lam_mul: float = 1.02,
    lam_max: float = 30.0,
    base_step: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using a
    soft‑max (log‑sum‑exp) surrogate.  λ is increased gradually.
    Returns the best sequence seen and its true evaluation score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial evaluation
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t0 = time.time()
    lam = lam_start
    step = base_step
    it = 0

    while time.time() < t0 + time_limit:
        # full convolution
        conv = np.convolve(a, a)                     # length 2n‑1
        max_val = conv.max()

        # ----- soft‑max weights -----
        # stable computation: shift by the maximum
        shifted = conv - max_val
        exp_vals = np.exp(lam * shifted)
        w = exp_vals / exp_vals.sum()                # weights sum to 1

        # ----- gradient of the smooth objective -----
        # grad_i = 2 * Σ_j a_j * w_{i+j}
        a_rev = a[::-1]
        grad_full = np.convolve(a_rev, w)            # length 3n‑2
        grad = 2.0 * grad_full[len(a) - 1 : len(a) - 1 + len(a)]

        # ----- step size (scale‑invariant) -----
        max_abs = np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate the new candidate
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] Soft‑max n={len(a)} λ={lam:.2f} "
                f"new best = {score:.6f}"
            )

        # prepare next iteration
        a = a_new
        lam = min(lam_max, lam * lam_mul)

        # simple decay of the step size
        it += 1
        if it % 250 == 0:
            step *= 0.85

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local projected sub‑gradient descent (refinement on the true max)
# ----------------------------------------------------------------------
def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.07,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence seen inside the allotted time.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    t0 = time.time()
    it = 0

    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # ensure zero‑mean direction (keeps the sum constant)
        mean_g = np.mean(grad)
        d = -grad + mean_g

        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] {print_tag} local best = {score:.6f} (n={len(a)})"
            )

        # occasional random shake‑up
        if np.random.rand() < 0.015:
            sigma = 0.012 * target_sum / len(a)
            a = _project_onto_simplex(a + np.random.randn(len(a)) * sigma, target_sum)
            step *= 0.9
        else:
            a = a_new

        # decay step size
        if it % 300 == 0 and it > 0:
            step *= 0.88
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper to resize a sequence while roughly preserving shape
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n by linear interpolation / down‑sampling."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Runs for at most ``budget_s`` seconds and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # slight safety margin

    # ------------------------------------------------------------------
    # Global best holder – initialise from historic solution if present
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Start from historic seq, score={sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # fallback – uniform of moderate size
    if best_seq is None:
        n0 = 1500
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic, uniform start (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper that performs a full optimisation run for a given length
    # ------------------------------------------------------------------
    def _run_one(n: int, init_vec: np.ndarray, remaining: float):
        """
        Optimise a candidate of length n, starting from ``init_vec``.
        ``remaining`` seconds are available for this run.
        Returns (candidate_seq, score) or (None, inf) on failure.
        """
        target = math.sqrt(2.0 * n)

        # allocate time between the two phases
        # 70 % soft‑max, 30 % local refinement (if enough time)
        soft_time = min(remaining * 0.65, 20.0)
        local_time = max(0.0, remaining - soft_time - 0.5)

        # ---- soft‑max phase ----
        cand1, score1 = _softmax_descent(init_vec, target, soft_time)

        # ---- local sub‑gradient refinement ----
        if local_time > 1.0:
            cand2, score2 = _local_descent(cand1, target, local_time,
                                          base_step=0.07,
                                          print_tag=f"refine n={n}")
        else:
            cand2, score2 = cand1, score1

        return cand2, score2

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # pick a length – stay near current best but allow jumps
        if best_seq is not None and np.random.rand() < 0.65:
            n = best_seq.size
            if np.random.rand() < 0.25:
                n = int(n * np.random.uniform(0.85, 1.15))
                n = max(300, min(3000, n))
        else:
            n = np.random.randint(300, 2500)

        target_sum = math.sqrt(2.0 * n)

        # initialise candidate
        if best_seq is not None and np.random.rand() < 0.7:
            init = _resize_sequence(best_seq, n)
        else:
            # random Dirichlet‑like start
            init_raw = np.random.rand(n)
            init = _project_onto_simplex(init_raw, target_sum)

        time_left = deadline - time.time()
        if time_left < 6.0:
            break

        cand_seq, cand_score = _run_one(n, init, time_left)

        if cand_seq is not None and cand_score < best_score - 1e-9:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** NEW GLOBAL BEST = {best_score:.6f} (n={n})"
            )
            # early stop if we have already crossed the target threshold
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Safety net – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
