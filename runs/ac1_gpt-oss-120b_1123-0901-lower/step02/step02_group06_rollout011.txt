def max_convolution(seq):
    n = len(seq)
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    conv = conv[:2*n-1]
    return np.max(conv)

The evaluation function  

\[
E(a)=\frac{2\,n\;\max_k (a*a)[k]}{\bigl(\sum_i a_i\bigr)^2}
\]

is invariant under scaling, therefore we may (and do) keep the
sequence on the probability simplex \(\sum a_i = 1,\;a_i\ge 0\).
With this normalisation  

\[
E(a)=2\,n\;\max_k (a*a)[k] .
\]

Thus the problem is *“make the largest entry of the self‑convolution
as small as possible”*.

The algorithm below is a hybrid of two ideas that were not combined in
the previous code:

1. **Linear‑programming expansion** –  
   Given a current candidate `f`, we maximise the total mass of a new
   non‑negative sequence `g` under the *hard* convolution constraint  

   \[
   (f*g)_k \le M\qquad\forall k,
   \]

   where \(M=\max (f*f)\).  
   The constraints are linear in `g`, therefore a classic LP (SciPy’s
   HiGHS) gives the *most massive* sequence that still respects the
   current bound.  After rescaling `g` to have unit sum we mix it with
   `f`.  Even a tiny mixing step strictly reduces the peak of the
   convolution.

2. **Soft sub‑gradient polishing** –  
   After a handful of LP steps we resort to the gradient‑descent
   routine that was already used successfully.  It quickly sweeps away
   remaining flat plateaus and refines the solution.

Both phases use fast FFT‑based convolution for the objective, but the
LP phase builds the exact Toeplitz matrix (size \((2n-1)\times n\)) only
a few times, keeping the total runtime well below the 1000 s budget.

The whole procedure runs with a fixed length \(n=2000\) (fast enough
for many LP solves, yet large enough to give a good bound).  The
best sequence found is returned as a plain list of Python floats.



```python
"""
Hybrid optimizer for the step‑function evaluation problem.

* Phase 1 – LP expansion:
    For the current sequence f, compute M = max(f*f).
    Solve   max 1ᵀg   subject to   A·g ≤ M,   g ≥ 0,
    where A is the Toeplitz matrix that maps g → (f*g).
    Scale g to have unit sum and mix it with f (mixing factor t).

* Phase 2 – soft sub‑gradient descent (same idea as the reference
  implementation) to polish the result.

The algorithm respects the 1000 s time budget, prints progress
and finally returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:          # all non‑positive → uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a few diverse start‑up sequences (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # 3. arcsine‑shaped law a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. a few random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using FFT (fast, accurate enough)."""
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Compute a non‑negative step‑function with a low evaluation score.
    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1. Problem size (kept moderate for fast LP solves)
    n = 2000

    # ------------------------------------------------------------------
    # 2. Create a pool of promising seeds and pick the best one
    best_seq = None
    best_score = float("inf")
    for cand in _initial_candidates(n, rng):
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f"[Init] best score after seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Phase 1 – Linear‑programming expansion (a few aggressive steps)
    lp_iter = 0
    mix_factor = 0.30                # initial mixing weight (t in the write‑up)
    max_lp_iters = 20                # we never need many LP solves
    while time.time() < deadline - 30 and lp_iter < max_lp_iters:
        # current bound M
        conv_ff = np.convolve(best_seq, best_seq)
        M = float(conv_ff.max())

        # build Toeplitz matrix A such that A @ g = f * g
        col = np.concatenate([best_seq, np.zeros(n - 1, dtype=best_seq.dtype)])
        row = np.concatenate([best_seq[:1], np.zeros(n - 1, dtype=best_seq.dtype)])
        A = toeplitz(col, row)                     # shape (2n‑1, n)

        # LP: maximise 1ᵀg  ↔ minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        # solve – HiGHS is fast and reliable
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP {lp_iter}] linprog failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"[LP {lp_iter}] degenerate solution, abort LP phase.")
            break

        # Rescale to unit sum (the same normalisation as best_seq)
        g = g0 / sum_g0

        # Mix a little bit of g into the current best
        new_seq = (1.0 - mix_factor) * best_seq + mix_factor * g
        # Numerical safety – clip & re‑project onto simplex
        new_seq = np.clip(new_seq, 0.0, None)
        new_seq = _project_onto_simplex(new_seq)

        # Evaluate (exactly, via the provided function)
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[LP {lp_iter}] new best = {best_score:.6f}")
            # after a successful step we can be a bit bolder
            mix_factor = min(0.5, mix_factor * 1.05)
        else:
            # no improvement → become more conservative
            mix_factor = max(0.01, mix_factor * 0.90)

        lp_iter += 1

    # ------------------------------------------------------------------
    # 4. Phase 2 – soft sub‑gradient descent (fine‑tuning)
    seq = best_seq.copy()
    step = 0.20                # initial step size (absolute scale)
    momentum = np.zeros_like(seq)
    beta = 0.9                 # heavy‑ball momentum
    K = 5                      # number of top convolution entries to follow
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # full self‑convolution (exact – needed for correct max)
        conv = np.convolve(seq, seq)

        # indices of K largest entries
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -K)[-K:]

        # accumulate sub‑gradient for the selected shifts
        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += seq[k - idx]

        grad = grad / K

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # gradient step (we minimise → go opposite)
        new_seq = seq - step * momentum
        new_seq = _project_onto_simplex(new_seq)

        # evaluate the candidate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Iter {it:05d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # slowly decay the step size
        step *= 0.9999

        # occasional random jitter if we are stuck
        if no_improve > 30000:
            jitter = rng.normal(scale=0.01, size=n)
            new_seq = new_seq + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq = new_seq / new_seq.sum()
            momentum[:] = 0.0
            no_improve = 0
            print(f"[Iter {it:05d}] stagnation – random perturbation applied")

        seq = new_seq
        it += 1

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
```
