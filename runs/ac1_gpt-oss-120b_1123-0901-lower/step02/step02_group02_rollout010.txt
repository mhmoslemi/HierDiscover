def _optimise_one_length_custom(n, init, run_deadline, rel_tol=5e-4, base_step=0.12, momentum_beta=0.9, extra_local_search=False):

    target_sum = np.sqrt(2.0 * n)

    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    conv = _conv_fft(a)
    max_val = conv.max()
    best_seq = a.copy()
    best_score = 2.0 * n * max_val / (a.sum()**2)

    velocity = np.zeros_like(a)
    step_factor = base_step
    iteration = 0
    rel_tol_curr = rel_tol

    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol_curr)
        grad = _gradient(a, active)

        # momentum
        velocity = momentum_beta * velocity + (1 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity)) if np.max(np.abs(velocity)) > 0 else 1.0
        eta = step_factor * max_val / max_grad
        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92

        # evaluate
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score_new = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            # print progress
            print(f"[{time.time():.2f}] n={n}  local best={score_new:.6f}")

        a = a_new

        # adapt rel_tol: tighten gradually
        if iteration % 200 == 0 and iteration > 0:
            rel_tol_curr = max(rel_tol_curr * 0.7, 1e-5)
            # optional shrink step factor
            step_factor *= 0.95

        # optional extra local search: random pair transfer
        if extra_local_search and (iteration % 50 == 0):
            # propose a random transfer and accept if improves
            i, j = np.random.choice(n, size=2, replace=False)
            eps = a[i] * 0.02  # small fraction
            if eps > 0:
                a_candidate = a.copy()
                a_candidate[i] -= eps
                a_candidate[j] += eps
                # ensure nonnegative
                if a_candidate[i] < 0:
                    a_candidate[i] = 0.0  # this should not happen
                a_candidate = _project_onto_simplex(a_candidate, target_sum)  # ensure sum
                conv_cand = _conv_fft(a_candidate)
                max_val_cand = conv_cand.max()
                score_cand = 2.0 * n * max_val_cand / (a_candidate.sum() ** 2)
                if score_cand < best_score:
                    best_score = score_cand
                    best_seq = a_candidate.copy()
                    a = a_candidate.copy()
                    # optional print
                    print(f"[{time.time():.2f}] n={n} local search improved to {score_cand:.6f}")

        # decay step_factor
        step_factor *= 0.9995
        iteration += 1

    # after loop maybe perform final projection to ensure sum constraint (already done)
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------
1.  Randomly choose a length `n` (typically 800‑5000).
2.  Initialise the vector with a *shape* that is known to give a flat
    autocorrelation:
        • power‑law a_i ∝ (i+1)^‑p   (p≈0.5)  – monotone variant
        • symmetric double‑tail a_i ∝ (i+1)^‑p + (n‑i)^‑p
    Occasionally warm‑start from the best sequence seen so far.
3.  Project the initial vector onto the simplex
        {x_i ≥ 0,  Σ x_i = √(2 n)}.
    The scaling is irrelevant for the objective, but this normalisation
    lets us compare scores directly as ``max(conv)``.
4.  Run a projected momentum gradient descent that minimises the maximum
    convolution entry.
        • The *active* set consists of indices whose convolution value is
          within a (shrinking) relative tolerance of the current maximum.
        • The gradient of the sum of the active entries is computed analytically
          (see `_gradient`).
        • A momentum term (`velocity`) is kept and an adaptive step size
          proportional to ``max(conv) / max(|velocity|)`` is used.
        • Every few iterations a tiny random perturbation is added to help
          escape flat plateaux.
5.  After each descent step a cheap *local transfer* move is attempted:
   a small amount of mass is moved from one coordinate to another;
   the move is kept only if it lowers the objective.
6.  The best vector found for this length is kept; the outer loop runs many
   independent restarts (different lengths, shapes and seeds) until the
   time budget expires.
7.  When the global best is found, a final refinement phase with a tighter
   active‑set tolerance and a smaller step size is performed on that vector.

All heavy work (FFT‑based convolution, simplex projection) is vectorised
and runs in O(n log n).  The routine respects the 1000 s budget and returns
the best sequence discovered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)             # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper 3 – indices whose convolution value is close to the current maximum
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    """Return (max_value, indices) where conv ≥ max – rel_tol·max."""
    max_val = conv.max()
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# Helper 4 – gradient of Σ_{k∈active} conv_k  with respect to a
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of the sum of the active convolution entries."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]          # ∂/∂a_i Σ a_i a_{k‑i}
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length `n`
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    rel_tol_start: float = 1e-2,
    rel_tol_end: float = 5e-5,
    base_step: float = 0.15,
    momentum_beta: float = 0.9,
    extra_local_search: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Projected momentum gradient descent on the max‑convolution objective.
    Returns the best vector found and its score.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)  → objective = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best vector seen in this run
    conv = _conv_fft(a)
    max_val, _ = _active_indices(conv, rel_tol_start)
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    velocity = np.zeros_like(a)

    step_factor = base_step
    iteration = 0
    # linear schedule for the tolerance (coarse → fine)
    rel_tol = rel_tol_start
    tol_decay_iter = max(1, int((run_deadline - time.time()) / 0.2))  # estimate

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) convolution and active set
        # --------------------------------------------------------------
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        # --------------------------------------------------------------
        # 2) gradient of the sum of the active entries
        # --------------------------------------------------------------
        grad = _gradient(a, active)

        # --------------------------------------------------------------
        # 3) momentum + adaptive step size
        # --------------------------------------------------------------
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity))
        if max_grad == 0:
            max_grad = 1.0
        eta = step_factor * max_val / max_grad

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 4) occasional random shake‑up (helps escaping plateaus)
        # --------------------------------------------------------------
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92

        # --------------------------------------------------------------
        # 5) evaluate
        # --------------------------------------------------------------
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        # --------------------------------------------------------------
        # 6) keep global best of this run
        # --------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")

        a = a_new

        # --------------------------------------------------------------
        # 7) optional cheap local transfer move
        # --------------------------------------------------------------
        if extra_local_search and (iteration % 50 == 0):
            i, j = np.random.choice(n, size=2, replace=False)
            eps = a[i] * 0.02                      # move 2 % of mass from i → j
            if eps > 0:
                cand = a.copy()
                cand[i] -= eps
                cand[j] += eps
                cand = _project_onto_simplex(cand, target_sum)
                conv_cand = _conv_fft(cand)
                max_val_cand = conv_cand.max()
                score_cand = 2.0 * n * max_val_cand / (cand.sum() ** 2)
                if score_cand < best_score:
                    best_score = score_cand
                    best_local = cand.copy()
                    a = cand
                    print(f"[{time.time():.2f}] n={n}  local transfer improved to {score_cand:.6f}")

        # --------------------------------------------------------------
        # 8) schedule updates
        # --------------------------------------------------------------
        iteration += 1
        step_factor *= 0.9996        # simulated‑annealing style decay

        # tighten the active‑set tolerance gradually
        if iteration % max(1, int(tol_decay_iter / 10)) == 0:
            rel_tol = max(rel_tol * 0.7, rel_tol_end)

    return best_local, best_score

        if shape_type == 'power':
            init = idx ** (-p)
        else:   # symmetric double‑tail
            init = (idx) ** (-p) + (n - idx + 1) ** (-p)

        # Small jitter to break exact symmetries
        init += np.random.rand(n) * 1e-4

        # Occasionally warm‑start from the best vector seen so far

            if best_seq.size >= n:

    return best_seq.tolist()
