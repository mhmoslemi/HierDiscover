def propose_candidate(...):
    start_time= time.time()
    deadline= start_time+budget_s-0.5

    n = 2000
    rng= np.random.default_rng(seed)

    # generate candidates
    candidates = _initial_candidates(n, rng)
    best_seq,best_score initial as before.

    # optimizer parameters
    seq = best_seq.copy()
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1=0.9
    beta2=0.999
    epsilon = 1e-8
    lr=0.02
    t=0
    no_improve=0

    while time.time() < deadline:
        t+=1
        conv = _conv_self(seq)  # length 2n-1
        # compute smoothed max weights
        # choose alpha schedule
        alpha = 3000.0   # maybe constant
        # shift conv to avoid large exponent
        max_conv = conv.max()
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        rev_seq = seq[::-1]
        conv_rev = _conv_fft(rev_seq, w)  # length 3n-2
        grad_f = 2.0 * conv_rev[n-1:n-1+n]  # gradient of smoothed max f
        grad = 2.0 * n * grad_f             # gradient of objective

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat/(np.sqrt(v_hat) + epsilon)
        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # evaluate objective (the real objective)
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Iter {t}] new best {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedule: maybe decrease lr slowly
        if t % 5000 == 0:
            lr *= 0.95

        # random jitter if stagnation
        if no_improve > 5000:
            jitter = rng.normal(scale=0.005, size=n)
            new_seq = np.clip(new_seq + jitter, 0, None)
            new_seq = new_seq / new_seq.sum()
            seq = new_seq
            m[:] = 0.0
            v[:] = 0.0
            no_improve = 0
            print(f"[Iter {t}] stagnation jitter")
            continue

        seq = new_seq

    print(f"Finished. best {best_score}")
    return best_seq.tolist()

assistantfinal**Solution Overview**

The code below implements a fast stochastic optimisation that works directly on the
shape of the step‑function.  
The objective (after normalising the sum to 1) is  

\[
\Phi(a)=2\,n\;\max_k (a*a)_k,
\]

so we need to make the self‑convolution as flat as possible.
The optimisation uses:

* a **smooth approximation** of the max – the log‑sum‑exp (temperature α) –
  which gives a differentiable objective.
* **Adam** (adaptive‑momentum) updates on the gradient of the smoothed objective.
* **Projection onto the probability simplex** after each update, keeping the
  sequence non‑negative and with sum = 1 (the problem is scale‑invariant).
* **FFT‑based convolutions** – both the self‑convolution and the gradient
  convolution are performed with a single FFT, making a 2000‑dimensional
  iteration cheap (< 2 ms on a modern CPU).
* **Rich initial pool** (uniform, geometric, arcsine, power‑law, random Dirichlet)
  to avoid the uniform stationary point.
* **Stagnation handling** – occasional Gaussian jitter and a learning‑rate decay.

The whole search runs inside the allotted time budget (default 1000 s) and
returns the best sequence discovered.

```python
"""
Propose a non‑negative step‑function minimising
    evaluate_sequence(seq)

The algorithm:
1. Build a diverse set of initial candidates (uniform, geom., arcsine,
   power‑law, Dirichlet draws, optionally a previously found global).
   Keep the best after a cheap evaluation.
2. Optimise with Adam on a smoothed “max” (log‑sum‑exp) of the self‑convolution.
   Convolutions are computed by FFT for speed.
3. After each step project onto the probability simplex (sum=1, a_i≥0).
4. If no improvement for a while, add a small Gaussian perturbation.
5. Continue until the time budget expires and output the best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using FFT (fast for n≈2000)."""
    n = len(a) + len(b) - 1
    L = 1 << ((n - 1).bit_length())   # next power of two
    FA = np.fft.rfft(a, n=L)
    FB = np.fft.rfft(b, n=L)
    c = np.fft.irfft(FA * FB, n=L)
    return c[:n]


def _self_conv(seq: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2n‑1) via FFT."""
    return _conv_fft(seq, seq)


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a list of promising normalised start points."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        r = 1.0 - factor / n
        if r <= 0:
            continue
        geo = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(geo / geo.sum())

    # 3. arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. power‑law families a_i ∝ (i+1)^(-p)
    for p in [0.45, 0.5, 0.55, 0.6, 0.7, 0.8, 0.9, 1.0]:
        powlaw = 1.0 / np.power(i + 1.0, p)
        cand.append(powlaw / powlaw.sum())

    # 5. a few random Dirichlet draws
    for _ in range(6):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # 6. optionally reuse a previously found global solution
    if 'height_sequence_1' in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.size >= n:
            cand.append(old[:n] / old[:n].sum())
        else:
            # linearly interpolate to required length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, old)
            cand.append(interp / interp.sum())

    return cand


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray, conv: np.ndarray = None) -> float:
    """Exact objective (2 n · max(conv)) / (sum(seq)^2)."""
    if conv is None:
        conv = _self_conv(seq)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    max_b = conv.max()
    return 2.0 * len(seq) * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights that (approximately) minimise
    the evaluation function within the allotted time."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – 2000 gives a good trade‑off between quality and speed
    n = 2000

    # ------------------------------------------------------------------
    # 1. initialise pool and pick the best start
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        conv = _self_conv(seq)
        score = _objective(seq, conv)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimiser on the smoothed max (log‑sum‑exp)
    seq = best_seq.copy()
    m = np.zeros_like(seq)          # first moment
    v = np.zeros_like(seq)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.03                       # base learning rate
    t = 0
    no_improve = 0

    # temperature schedule for the smooth max
    alpha_start = 200.0
    alpha_end = 5000.0
    alpha_ramp_iters = 25000        # linear increase

    while time.time() < deadline:
        t += 1

        # self‑convolution (full length 2n‑1)
        conv = _self_conv(seq)

        # ------- smooth max (log‑sum‑exp) -------
        # temperature α grows linearly from α_start to α_end
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / alpha_ramp_iters)
        max_c = conv.max()
        exp_vals = np.exp(alpha * (conv - max_c))      # shift for numerical stability
        w = exp_vals / exp_vals.sum()                 # softmax weights, sum(w)=1

        # ------- gradient of the smooth max -------
        # g = 2 * sum_i w_i * shift_i(a)  (vector of length n)
        # Compute via convolution: rev_seq * w, then take the central block.
        rev = seq[::-1]
        conv_rev = _conv_fft(rev, w)                  # length 3n‑2
        grad_f = 2.0 * conv_rev[n - 1:n - 1 + n]       # gradient of f = smooth max
        grad = 2.0 * n * grad_f                        # gradient of full objective

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # evaluate (cheap exact evaluation)
        conv_new = _self_conv(new_seq)
        score = _objective(new_seq, conv_new)

        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Iter {t:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # learning‑rate decay (very gentle)
        if t % 4000 == 0:
            lr *= 0.97

        # occasional jitter to escape plateaus
        if no_improve > 6000:
            jitter = rng.normal(scale=0.008, size=n)
            new_seq = np.clip(new_seq + jitter, 0.0, None)
            new_seq = new_seq / new_seq.sum()
            seq = new_seq
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Iter {t}] stagnation – jitter applied")
            continue

        seq = new_seq

    print(f"Finished. best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
```
