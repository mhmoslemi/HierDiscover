def _optimize_softmax_len(n, init_a, best_seq, best_score, deadline):
    target_sum = sqrt(2*n)
    a = init_a.copy()
    a = a * (target_sum / a.sum())
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    beta = 0.5  # temperature param
    beta_inc = 1.001
    max_beta = 200.0
    # loop
    while time.time() < deadline:
        # conv self
        M, k_max, conv_full = _conv_max_fft(a)   # we have conv array
        # compute softmax weights
        # shift conv to avoid overflow: subtract max
        c_shift = conv_full - conv_full.max()
        w = np.exp(beta * c_shift)
        w_sum = w.sum()
        w = w / w_sum
        # compute conv of a and w via FFT
        N = 1 << ((n + (2*n-1) - 1).bit_length())  # maybe focus on 3n
        A = np.fft.rfft(a, n=N)
        # w_pad and w_rev pad to size N
        W = np.fft.rfft(w, n=N)
        conv_aw = np.fft.irfft(A * W, n=N)
        # w_rev
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, n=N)
        conv_aw_rev = np.fft.irfft(A * W_rev, n=N)
        # compute gradient components for indices 0..n-1
        L = 2*n - 1
        grad = np.zeros_like(a)
        # conv_aw_slice: conv_aw[0:n] (since conv_aw[j] corresponds to sum_i a_i w_{j-i} ; j can be negative? but we only have conv from 0..)
        conv_aw_slice = conv_aw[:n]
        # conv_aw_rev_slice: conv_aw_rev[L-1 - np.arange(n)]
        idx_rev = L - 1 - np.arange(n)
        conv_aw_rev_slice = conv_aw_rev[idx_rev]
        grad = 2.0 * (conv_aw_slice + conv_aw_rev_slice)
        # embed denominator factor: objective f = 2n * M_smooth / S^2; S fixed,
        # gradient of f w.r.t a = (2n/S^2) * grad
        coeff = 2.0 * n / (target_sum * target_sum)
        grad = coeff * grad
        # also, gradient from denominator? Since sum constraint is enforced via projection after update,
        # we ignore denominator because projection will keep sum constant.
        # Adam update
        iteration += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0 / (1 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # enforce symmetry
        a = (a + a[::-1]) / 2.0
        # project onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # update beta
        beta = min(max_beta, beta * beta_inc)
        # evaluate objective exactly using evaluate_sequence for best check
        cur_obj = evaluate_sequence(a.tolist())
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[len {n} iter {iteration}] new best {best_score:.6f}")
        # occasional jitter
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
    return best_seq, best_score

"""
A multi‑restart soft‑max gradient optimizer for the step‑function problem.

The objective is 2·n·max(a∗a)/(sum a)².  For any non‑negative sequence we can
scale it to sum = √(2n) without changing the value, so after every update we
project onto the simplex { a_i ≥ 0, Σ a_i = √(2n) }.

Instead of using the sub‑gradient of the *max* we smooth the max with a
soft‑max (temperature β).  The gradient of the smoothed objective is cheap to
compute via FFTs:
    w_i ∝ exp(β·( (a∗a)_i – max(a∗a) ))
    grad_j = 2·β_factor·[ (a∗w)_j + (a∗w_rev)_{2n‑2‑j} ] ,
where β_factor = 2n/(√(2n))² = 1.  The“soft‑max’’ gives a dense gradient early
on (β small) and recovers the original max‑gradient later (β → ∞).

The optimizer runs Adam on this gradient, periodically symmetrises (a←(a+aᵀ)/2)
and adds tiny random jitter to escape plateaus.  Several initial shapes
(uniform, exponential, triangular, cosine, the supplied `height_sequence_1`,
or random Dirichlet draws) and several lengths (original length and a few
scaled versions) are tried; the best sequence found before the 1000 s wall‑clock
deadline is returned.

All heavy work is done with FFTs (O(n log n) per iteration), making it fast
enough to perform many thousands of iterations on vectors of a few thousand
elements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation of a (a∗a) using FFT.
    Returns (max_value, index_of_max, full_convolution_array).
    """
    n = a.shape[0]
    # size for convolution a with a (length 2n-1)
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv_full = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    k_max = int(conv_full.argmax())
    return float(conv_full.max()), k_max, conv_full


def _full_conv_fft(x: np.ndarray, y: np.ndarray):
    """Full convolution of real vectors x and y via FFT, returns length len(x)+len(y)-1."""
    L = len(x) + len(y) - 1
    N = 1 << (L - 1).bit_length()
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:L]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float):
    """
    Compute soft‑max weights w_i = exp(beta·(conv_i - max(conv))) / Σ.
    The subtraction of the maximum improves numerical stability.
    """
    shift = conv - conv.max()
    # avoid overflow when beta is large
    exp_vals = np.exp(beta * shift)
    w = exp_vals / exp_vals.sum()
    return w


def _optimize_softmax_len(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
):
    """
    Adam optimizer for a given length `n` starting from `init_a`.
    Updates `best_seq`/`best_score` whenever a better vector is found.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise and normalise
    a = init_a.copy()
    if a.sum() == 0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12          # a bit larger than the previous code
    iteration = 0

    # soft‑max temperature schedule
    beta_temp = 0.4
    beta_inc = 1.0015
    beta_max = 200.0

    # pre‑compute length related constants
    L = 2 * n - 1                # length of autocorrelation
    N_fft = 1 << ((3 * n - 2).bit_length())   # enough for a∗a and a∗w convolutions

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current autocorrelation and max (used as objective value)
        M, _, conv_self = _conv_max_fft(a)        # conv_self length 2n‑1
        # update global best (objective = M because Σa = √(2n) ⇒ denominator = 2n)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:6d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) build soft‑max weights from current autocorrelation
        w = _softmax_weights(conv_self, beta_temp)
        w_rev = w[::-1]

        # --------------------------------------------------------------
        # 3) gradient of the smoothed objective
        #   grad_j = 2 * ( (a∗w)_j + (a∗w_rev)_{L‑1‑j} )
        #   with coefficient 2n/(Σa)² = 1 (because Σa = √(2n))
        A = np.fft.rfft(a, n=N_fft)
        W = np.fft.rfft(w, n=N_fft)
        W_rev = np.fft.rfft(w_rev, n=N_fft)

        conv_aw = np.fft.irfft(A * W, n=N_fft)            # length N_fft
        conv_aw_rev = np.fft.irfft(A * W_rev, n=N_fft)

        # relevant slices for indices 0 … n‑1
        conv_aw_slice = conv_aw[:n]                       # (a∗w)_j
        idx_rev = L - 1 - np.arange(n)                    # index for (a∗w_rev)_{L‑1‑j}
        conv_aw_rev_slice = conv_aw_rev[idx_rev]

        grad_raw = 2.0 * (conv_aw_slice + conv_aw_rev_slice)
        grad = grad_raw                                     # coefficient = 1

        # --------------------------------------------------------------
        # 4) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)                # gentle decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) enforce symmetry, non‑negativity and the sum constraint
        a = (a + a[::-1]) / 2.0
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 6) temperature schedule & occasional jitter
        beta_temp = min(beta_max, beta_temp * beta_inc)

        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_seq, best_score

def _initial_vector(n: int, base: np.ndarray, kind: str, rng: np.random.Generator):
    """
    Produce an initial vector of length n.
    `kind` can be 'base', 'uniform', 'exp', 'tri', 'cos', 'rand'.
    """
    if kind == "base":
        # linear interpolation of the supplied base onto the new length
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        vec = np.interp(new_x, old_x, base)
        vec = np.maximum(vec, 0.0)
    elif kind == "uniform":
        vec = np.ones(n, dtype=float)
    elif kind == "exp":
        # exponential decay, parameter chosen so that the tail is not too short
        decay = math.log(2) / (n / 8.0)          # half‑life ≈ n/8
        vec = np.exp(-decay * np.arange(n))
    elif kind == "tri":
        # symmetric triangle: increase then decrease
        half = (n + 1) // 2
        inc = np.linspace(0.0, 1.0, half, endpoint=False)
        dec = np.linspace(1.0, 0.0, n - half)
        vec = np.concatenate([inc, dec])
    elif kind == "cos":
        vec = 0.5 * (1 + np.cos(np.pi * (np.arange(n) - (n - 1) / 2) / (n - 1)))
    elif kind == "rand":
        vec = rng.random(n) + 0.1                 # avoid exact zeros
    else:
        raise ValueError(f"unknown kind {kind}")

    return vec.astype(float)

    # ------------------------------------------------------------------
    # Initialisation kinds we will try for each length
    init_kinds = ["base", "uniform", "exp", "tri", "cos", "rand"]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[run] length = {n}")

        for kind in init_kinds:
            if time.time() >= deadline:
                break

            # construct the initial vector for this run
            if kind == "base" and n == n0:
                init_a = base.copy()
            else:
                init_a = _initial_vector(n, base, kind, rng)

            # normalise to the correct total sum for this length
            target_sum = math.sqrt(2.0 * n)
            if init_a.sum() == 0:
                init_a = np.full_like(init_a, target_sum / n)
            else:
                init_a = init_a * (target_sum / init_a.sum())

            # run the soft‑max Adam optimiser
            best_seq, best_score = _optimize_softmax_len(
                n=n,
                init_a=init_a,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
            )

    # ------------------------------------------------------------------
    # If nothing was found (should never happen) fall back to uniform.
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
