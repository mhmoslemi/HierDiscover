def _conv_full_fft(a):

def _active_set_mask(conv, rel_tol):
    max_val = conv.max()
    tol = max(rel_tol * max_val, 1e-12)
    mask = np.zeros_like(conv, dtype=np.float64)
    mask[conv >= max_val - tol] = 1.0
    return max_val, mask

def _gradient_from_weights(a, weights):
    # weights length conv_len = 2*n-1
    n = a.shape[0]
    a_rev = a[::-1]
    # Determine FFT size
    conv_len = weights.shape[0]  # should be 2*n-1
    size = 1 << ((n + conv_len - 2).bit_length())  # for length n + conv_len - 1 = 3n-2, but we need power of two >= n + conv_len - 1
    # compute FFTs
    fw = np.fft.rfft(weights, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    corr_full = np.fft.irfft(fw * fa_rev, n=size)[:n + conv_len - 1]
    grad = 2.0 * corr_full[n-1:n-1+n]
    return grad

def _optimise_one_length(n, init, run_deadline, *, rel_tol=5e-4, base_step=0.12, momentum_beta=0.9):
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)
    conv = _conv_full_fft(a)
    max_val, _ = _active_set_mask(conv, rel_tol)
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum()**2)  # equals max_val normalized but compute anyway
    velocity = np.zeros_like(a)
    step_factor = base_step
    iteration = 0
    while time.time() < run_deadline:
        conv = _conv_full_fft(a)
        max_val, mask = _active_set_mask(conv, rel_tol)
        grad = _gradient_from_weights(a, mask)
        # momentum
        velocity = momentum_beta * velocity + (1 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity)) + 1e-12
        eta = step_factor * max_val / max_grad
        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)
        # jitter
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92
        # evaluate
        conv_new = _conv_full_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum()**2)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")
        a = a_new
        step_factor *= 0.9997
        iteration += 1
    return best_local, best_score

"""
Optimise a non‑negative step function to minimise
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines three ideas:

1. **Parameter sweep** – For a given length `n` we try a family
   a_i ∝ (i + 1 + shift)^{-p} with `p` in [0.40, 0.60] and
   `shift` in [0, 8].  The best candidate of the sweep is used as a warm‑start.

2. **Projected (accelerated) gradient descent** – The objective is the
   L∞‑norm of the auto‑convolution.  At each iteration we form the active
   set of indices whose convolution value is within a relative tolerance of
   the current maximum and compute the gradient of the sum of those entries
   using a *single* FFT‑based correlation (O(n log n)).  Momentum and a
   decaying step size are applied, and the iterate is projected back onto the
   simplex {x ≥ 0, Σx = √(2n)}.

3. **Exploration & refinement** – Many independent restarts with random
   lengths, exponents and shifts are performed within the 1000 s budget.
   The globally best vector is finally refined for the remaining time.

The code contains only top‑level functions, respects the time budget and
returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – full linear convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length) using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)   # numerical noise clipping
    return conv


# ----------------------------------------------------------------------
# Helper 3 – active‑set mask (binary weights) for the current max convolution
# ----------------------------------------------------------------------
def _active_set_mask(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    """
    Return (max_value, mask) where mask[k] = 1 if conv[k] is within
    rel_tol·max_val of the maximum, otherwise 0.
    """
    max_val = conv.max()
    tol = max(rel_tol * max_val, 1e-12)
    mask = np.zeros_like(conv, dtype=np.float64)
    mask[conv >= max_val - tol] = 1.0
    return max_val, mask


# ----------------------------------------------------------------------
# Helper 4 – gradient of Σ_{k∈active} conv_k w.r.t. a using FFT correlation
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k weights[k] * conv_k where conv = a * a.
    For binary weights this is exactly the gradient of the sum over the
    active set.  The formula reduces to a correlation between the reversed
    vector a_rev and the weight vector.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # Length needed for linear correlation: n + len(weights) - 1 = 3n-2
    size = 1 << ((n + weights.shape[0] - 2).bit_length())
    fw = np.fft.rfft(weights, n=size)
    fa = np.fft.rfft(a_rev, n=size)
    corr_full = np.fft.irfft(fw * fa, n=size)[: n + weights.shape[0] - 1]
    # Extract the slice that corresponds to gradient components
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length `n`
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    rel_tol: float = 5e-4,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected accelerated gradient descent on the max‑convolution objective.
    Returns the best vector found (as a NumPy array) and the corresponding
    evaluation score.
    """
    target_sum = np.sqrt(2.0 * n)                 # normalisation → objective = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep track of the best vector seen in this run
    conv = _conv_full_fft(a)
    max_val, _ = _active_set_mask(conv, rel_tol)
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    velocity = np.zeros_like(a)
    step_factor = base_step
    iteration = 0

    while time.time() < run_deadline:
        # 1) convolution and active set
        conv = _conv_full_fft(a)
        max_val, mask = _active_set_mask(conv, rel_tol)

        # 2) gradient (sum of active entries)
        grad = _gradient_from_weights(a, mask)

        # 3) momentum + adaptive step size
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # 4) occasional random perturbation
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92

        # 5) evaluate
        conv_new = _conv_full_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        # 6) keep the best local solution
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")

        a = a_new
        step_factor *= 0.9997          # slow annealing of the step size
        iteration += 1

    return best_local, best_score


# ----------------------------------------------------------------------
# Helper – cheap grid search over power‑law parameters
# ----------------------------------------------------------------------
def _grid_search(
    n: int,
    target_sum: float,
    p_vals: np.ndarray,
    shift_vals: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Scan a family a_i ∝ (i+1+shift)^{-p} and return the best vector
    (projected onto the simplex) together with its score.
    """
    best_seq = None
    best_score = np.inf
    idx = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for shift in shift_vals:
            a = (idx + shift) ** (-p)
            a = _project_onto_simplex(a, target_sum)
            conv = _conv_full_fft(a)
            max_val = conv.max()
            score = max_val                     # because target_sum = √(2n) → objective = max_val
            if score < best_score:
                best_score = score
                best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function within the given time budget.  The routine mixes a
    fast parameter sweep, many stochastic restarts and a projected
    accelerated gradient descent with FFT‑based gradients.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # Reasonable range of lengths – large enough to capture the asymptotics,
    # small enough to keep each optimisation cheap.
    min_n, max_n = 800, 3500

    # Pre‑computed grids for the cheap sweep
    p_grid = np.linspace(0.40, 0.60, 11)          # 11 values
    shift_grid = np.linspace(0.0, 8.0, 11)       # 11 values

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a few seconds of work.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # Towards the end we keep the problem size modest to finish cleanly.
        if remaining < 30.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(min_n, max_n + 1)

        target_sum = np.sqrt(2.0 * n)

        # ----- cheap grid search for a good starting shape -------------
        seq0, score0 = _grid_search(n, target_sum, p_grid, shift_grid)

        # ----- occasional warm‑start from the best solution so far -------
        if best_seq is not None and np.random.rand() < 0.3:
            if best_seq.size >= n:
                seq0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                seq0 = np.concatenate([best_seq, pad])

        # ----- allocate a time slice for this run -----------------------
        # 7 s per run works well: many restarts, each still gets enough
        # iterations to improve the candidate.
        run_deadline = min(deadline, time.time() + 7.0)

        # ----- run the optimisation --------------------------------------
        seq_local, score_local = _optimise_one_length(
            n,
            seq0,
            run_deadline,
            rel_tol=np.random.choice([5e-5, 2e-4, 5e-4, 1e-3]),
            base_step=np.random.uniform(0.07, 0.15),
            momentum_beta=0.9,
        )

        # ----- update global best ----------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # Early exit if we have already beaten the target.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final refinement with whatever time is left (if any)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 4.0:
        n_best = best_seq.size
        # run a longer optimisation from the current best
        best_seq, best_score = _optimise_one_length(
            n_best,
            best_seq,
            deadline,
            rel_tol=5e-5,
            base_step=0.08,
            momentum_beta=0.9,
        )
        print(f"[{time.time():.2f}] FINAL refinement score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure we always return *something*.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
