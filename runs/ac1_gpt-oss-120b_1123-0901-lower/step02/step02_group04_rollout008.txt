    # compute stable softmax weights.
    maxc = conv.max()
    expvals = np.exp(beta * (conv - maxc))
    w = expvals / expvals.sum()  # shape 2n -1

    n = a.shape[0]
    grad = np.zeros(n, dtype=float)
    # Loop k.
    for k in range(2*n - 1):
        wk = w[k]
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        # indices i from low to high inclusive.
        # partner = k - i

def compute_softmax_gradient(a, beta):
    conv = _conv_self(a)
    # compute softmax weights
    maxc = conv.max()
    # to avoid overflow when beta large, subtract maxc
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # compute correlation
    corr = _correlation(a, w)  # custom
    # diag correction
    idx = np.arange(a.shape[0], dtype=int)
    diag = w[2*idx] * a
    grad = 2.0 * corr - diag

"""
Propose a step‑function height sequence with a low evaluation score.

The search consists of three stages:

1.  A quick sweep over several low‑dimensional parametric families
    (exponential, linear, quadratic, triangular).  Every candidate is
    normalised onto the simplex Σ a_i = √(2 n) and evaluated with the
    provided ``evaluate_sequence``.  The best candidate so far is kept.

2.  A smooth‑max gradient descent.  Instead of the non‑smooth
    max                                                                                                                                                                                                                     *a* is optimised by minimising a smooth approximation of
    max (a * a).  The smooth approximation is a log‑sum‑exp (soft‑max);
    its gradient can be computed efficiently via an FFT‑based correlation.
    After each gradient step the iterate is projected back onto the simplex.

3.  Optional LP refinement (the same idea as the SOTA description) if
    SciPy is present.

All helper functions are defined at module level; there are no closures
or lambda expressions.  The routine respects the time budget and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex with sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full autocorrelation a*a (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full convolution (autocorrelation) a * a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: correlation a ⋆ w  (full, via FFT)
# ----------------------------------------------------------------------
def _correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear correlation of a (length n) with w (length 2n‑1). Returns length n."""
    n = a.shape[0]
    L = w.shape[0]
    size = 1 << ((n + L - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fwr = np.fft.rfft(w[::-1], n=size)          # w reversed
    prod = Fa * Fwr
    conv_full = np.fft.irfft(prod, n=size)
    conv_full = conv_full[: n + L - 1]
    start = L - 1
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
# Helper: safe wrapper around the external evaluator
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Return the score from evaluate_sequence or +inf on error."""
    try:
        return evaluate_sequence(seq)        # the harness provides this
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Shape factories (non‑negative, not normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(a*a)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  ( 1/β * log Σ_k exp(β * (a*a)_k) ).
    This is the smooth surrogate of the infinity‑norm of the autocorrelation.
    """
    conv = _conv_self(a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    exp_vals = np.exp(beta * (conv - maxc))
    w_sum = exp_vals.sum()
    if w_sum == 0:
        w = np.full_like(exp_vals, 1.0 / exp_vals.size)
    else:
        w = exp_vals / w_sum                  # size 2n‑1, Σ w_k = 1

    # correlation a ⋆ w  gives Σ_k w_k * a_{k-i}
    corr = _correlation_fft(a, w)           # length n

    # diagonal correction: for k = 2i the factor should be 1 rather than 2
    idx = np.arange(a.shape[0], dtype=int)
    diag = w[2 * idx] * a

    grad = 2.0 * corr - diag                # full gradient
    return grad


# ----------------------------------------------------------------------
# Update best sequence if a candidate improves the score
# ----------------------------------------------------------------------
def _update_best(candidate: np.ndarray,
                 best_seq: np.ndarray,
                 best_score: float) -> (np.ndarray, float):
    """Evaluate `candidate`; if it beats `best_score` return the new best."""
    score = _evaluate(candidate.tolist())
    if score < best_score:
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(candidate)})")
        return candidate.copy(), score
    return best_seq, best_score


# ----------------------------------------------------------------------
# Gradient‑descent driver (smooth‑max)
# ----------------------------------------------------------------------
def _run_gradient_descent(init_vec: np.ndarray,
                         deadline: float,
                         global_best_seq: np.ndarray,
                         global_best_score: float) -> (np.ndarray, float):
    """
    Perform projected gradient descent on the smooth‑max objective,
    starting from `init_vec`.  The routine updates the global best
    whenever a better candidate is seen.
    """
    a = init_vec.astype(float)
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    it = 0
    beta = 5.0          # initial smoothness
    best_seq = global_best_seq
    best_score = global_best_score

    while time.time() < deadline:
        grad = _softmax_gradient(a, beta)

        max_abs = np.max(np.abs(grad))
        if max_abs == 0.0:
            break

        # diminishing step size, normalised by the gradient magnitude
        step = 0.4 / math.sqrt(it + 1) / max_abs
        a = a - step * grad
        a = _project_onto_simplex(a, target)

        # periodic evaluation
        if it % 7 == 0:
            best_seq, best_score = _update_best(a, best_seq, best_score)
            if best_score < 1.503:          # early exit threshold
                break

        it += 1
        # increase smoothing gradually (annealing)
        if it % 250 == 0:
            beta = min(beta * 1.6, 300.0)

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list:
    """

    The routine respects the supplied time budget and returns the best
    sequence found as a plain Python list.
    """

    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the previously discovered sequence if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")

        n0 = arr.shape[0]
        target0 = math.sqrt(2.0 * n0)
        if arr.sum() > 0.0:
            arr = arr / arr.sum() * target0
        else:
            arr = np.full(n0, target0 / n0, dtype=float)
        best_seq, best_score = _update_best(arr, best_seq, best_score)

    # ------------------------------------------------------------------
    # 1.  Fast sweep over low‑dimensional parametric families
    # ------------------------------------------------------------------
    lengths = list(range(200, 2001, 200)) + [1326, 1500, 1800]
    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-5, -0.2, num=15):
            a = _shape_exp(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _update_best(a, best_seq, best_score)

        # reversed exponential
        for alpha in np.logspace(-5, -0.2, num=15):
            a = _shape_exp_rev(n, alpha)
            a = a / a.sum() * target
            best_seq, best_score = _update_best(a, best_seq, best_score)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=12):
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _update_best(a, best_seq, best_score)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n * n), num=12):
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _update_best(a, best_seq, best_score)

        # triangular (symmetric) shape
        a = _shape_triangular(n)
        a = a / a.sum() * target
        best_seq, best_score = _update_best(a, best_seq, best_score)

    # ------------------------------------------------------------------
    # 2.  Gradient‑descent refinement (smooth max)
    # ------------------------------------------------------------------
    start_vectors = []

    # a)  The current best (if any)
    if best_seq is not None:
        start_vectors.append(np.array(best_seq, copy=True))

    # b)  Slightly perturbed copies of the best (to escape shallow minima)
    for _ in range(5):
        if time.time() >= deadline or best_seq is None:
            break
        a = np.array(best_seq, copy=True)
        noise = np.random.randn(a.shape[0]) * a.mean() * 0.06
        a = np.maximum(a + noise, 0.0)
        target = math.sqrt(2.0 * a.shape[0])
        a = _project_onto_simplex(a, target)
        start_vectors.append(a)

    # c)  Random restarts with varying lengths (moderate size for speed)
    while time.time() < deadline and len(start_vectors) < 30:
        n = np.random.randint(300, 4001)
        target = math.sqrt(2.0 * n)
        # Dirichlet‑like random point on the simplex (exponential draws)
        a = np.random.exponential(scale=1.0, size=n)
        a = a / a.sum() * target
        start_vectors.append(a)

    # run the descent for every start vector
    for vec in start_vectors:
        if time.time() >= deadline:
            break
        best_seq, best_score = _run_gradient_descent(
            vec, deadline, best_seq, best_score

    # ------------------------------------------------------------------

    except Exception:               # SciPy not installed
        linprog = None

            A = np.zeros((rows, n), dtype=float)

                method="highs",

    # Fallback – uniform sequence if everything else failed
    # ------------------------------------------------------------------

        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _evaluate(best_seq.tolist())

    return best_seq.tolist()
