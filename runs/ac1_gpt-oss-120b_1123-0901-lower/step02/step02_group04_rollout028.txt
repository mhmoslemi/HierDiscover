def _subgrad_descent(state, init_vec, target, deadline, max_iter=10000, K=1):
    a = init_vec.copy()
    n = len(a)
    m = np.zeros_like(a)  # momentum
    beta = 0.9
    base_step = 0.5
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_self(a)
        # get top K indices
        if K==1:
            kmax = int(np.argmax(conv))
            idxs = [kmax]
        else:
            idxs = np.argpartition(conv, -K)[-K:]
        # compute gradient
        grad = np.zeros_like(a)
        for k in idxs:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            # i from low..high
            i_range = np.arange(low, high+1, dtype=int)
            partner = k - i_range
            grad[i_range] += 2.0 * a[partner] / len(idxs)  # average if multiple K
        direction = -grad
        m = beta*m + (1-beta)*direction
        step = base_step / math.sqrt(it+1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0:
            max_abs = 1.0
        eta = step / max_abs
        a_new = a + eta * m
        # Projection onto simplex sum target and nonnegativity
        a_new = _project_onto_simplex(a_new, target)
        # Evaluate periodically
        if it % 5 == 0:
            _maybe_update(state, a_new.tolist())
        a = a_new
        it += 1
        # optional: early break if state['score'] <= 1.4
        if state['score'] < 1.40:
            break

def _random_shape(n):
    choice = np.random.choice(['exp', 'linear', 'quad', 'uniform'])
    if choice == 'exp':
        alpha = np.random.rand() * 0.02 + 1e-5
        a = np.exp(-alpha * np.arange(n))
    elif choice == 'linear':
        beta = np.random.rand() * 2.0 / max(1, n - 1)

def _lp_refine(state, a, target, deadline, max_iters=5):
    n = len(a)
    while time.time() < deadline and max_iters > 0:
        conv = _conv_self(a)
        M = conv.max()
        rows = 2*n - 1
        # Build constraint matrix A: each row k has entries a[k - j] for j in valid range.
        # Create zero matrix rows x n
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high+1, dtype=int)
            A[k, cols] = a[k - cols]
        # Solve LP
        try:
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(rows, M), bounds=(0, None),
                          method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success:
            break
        g = res.x
        # Rescale g to target sum
        s = g.sum()
        if s <= 0:
            break
        g = g * (target / s)
        # Blend
        t = 0.3
        a = (1 - t) * a + t * g
        # Update best sequence
        _maybe_update(state, a.tolist())
        max_iters -= 1
    # return final a
    return a

       a ← Π_{simplex}(a – η · m),

   where `m` is a momentum vector and the gradient is the sum of the
   (sub‑)gradients of the *K* largest convolution entries (K=3 by default).
   The step size decays like 1/√t and the iterate is re‑projected onto the
   simplex after each update.

3. **Iterated LP refinement** – if SciPy is available the LP described in the
   state‑of‑the‑art paper is applied several times.  The programme solves

       maximise Σ gⱼ   s.t.   a * g ≤ M   and   g ≥ 0,

   rescales the optimiser `g` to the same simplex and forms a convex
   combination with the current `a`.  Repeating the step pushes the bound a
   little further.

4. **Fallback** – if nothing else succeeded a uniform sequence of length
   1000 is returned (its bound is exactly 2).

All heavy work is performed with real FFTs (`np.fft.rfft`) for O(n log n)
convolution, and the whole search respects the supplied time budget
(`budget_s` seconds).  Progress is printed to stdout.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex of radius `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full convolution via real FFT (fast for n up to a few thousand)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the (full) linear convolution a * a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: safe call of the external evaluation routine
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Return the score from evaluate_sequence or +inf on error."""
    try:
        return evaluate_sequence(seq)      # provided by the harness
    except Exception:                     # pragma: no cover
        return float("inf")


# ----------------------------------------------------------------------
# Helper: update the global best record (state is a dict)
# ----------------------------------------------------------------------
def _maybe_update(state: dict, cand: list) -> float:
    """Evaluate `cand`; replace state if the score improves."""
    score = _evaluate(cand)
    if score < state["score"]:
        state["score"] = score
        state["seq"] = np.array(cand, copy=True)
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(cand)})")
    return score


# ----------------------------------------------------------------------
# Shape factories (non‑normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=float)


# ----------------------------------------------------------------------
# Sub‑gradient descent with momentum (top‑K max‑lag gradients)
# ----------------------------------------------------------------------
def _run_subgrad(state: dict,
                 init_vec: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 max_iter: int = 20000,
                 K: int = 3) -> None:
    """
    Perform many sub‑gradient steps starting from `init_vec` (already on the simplex).
    The gradient is built from the `K` largest convolution entries.
    """
    a = init_vec.astype(float)
    n = a.shape[0]
    m = np.zeros_like(a)          # momentum
    beta = 0.9                    # momentum decay
    base_step = 0.5               # initial step magnitude

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_self(a)

        # indices of the K largest entries
        if K == 1:
            idxs = [int(np.argmax(conv))]
        else:
            idxs = np.argpartition(conv, -K)[-K:]

        # build sub‑gradient: for each selected lag `k` add 2*a_{k‑i}
        grad = np.zeros_like(a)
        for k in idxs:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            i = np.arange(low, high + 1, dtype=int)
            partner = k - i
            grad[i] += 2.0 * a[partner] / len(idxs)   # average when K>1

        direction = -grad
        m = beta * m + (1.0 - beta) * direction

        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target_sum)

        if it % 5 == 0:                     # cheap periodic evaluation
            _maybe_update(state, a_new.tolist())

        a = a_new
        it += 1

        # early stop if we have already crossed a generous target
        if state["score"] < 1.45:
            break


# ----------------------------------------------------------------------
# Linear‑programming refinement (iterated)
# ----------------------------------------------------------------------
def _lp_refine(state: dict,
               a: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iters: int = 5) -> np.ndarray:
    """
    Perform up to `max_iters` alternating LP refinements as described in the
    state‑of‑the‑art paper.  The procedure stops early if the time budget expires
    or if the LP becomes infeasible.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                     # pragma: no cover
        return a

    n = a.shape[0]
    rows = 2 * n - 1

    # Pre‑allocate the constraint matrix (will be rebuilt each iteration)
    A = np.zeros((rows, n), dtype=float)

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iters:
        conv = _conv_self(a)
        M = conv.max()

        # Build A for the inequalities (a * g)_k ≤ M
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(c=-np.ones(n),                # maximise Σ gⱼ
                      A_ub=A,
                      b_ub=np.full(rows, M),
                      bounds=(0, None),
                      method="highs",
                      options={"presolve": True})
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break

        # Rescale to the same simplex used throughout the search
        g = g * (target_sum / s)

        # Convex combination – blending factor can be moderate
        t = 0.25
        a = (1.0 - t) * a + t * g

        _maybe_update(state, a.tolist())

        iter_cnt += 1
        if state["score"] < 1.40:
            break

    return a


# ----------------------------------------------------------------------
# Main routine required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # tiny safety margin

    # ----- global state -------------------------------------------------
    state = {"seq": None, "score": float("inf")}

    # ----- Phase 0 – broad parametric sweep ----------------------------
    # Lengths from a few hundred up to about 2500 (still cheap for FFT)
    lengths = list(range(400, 2501, 300)) + [1200, 1500, 2000]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # ---- exponential families (α = c / n) -----------------------
        for c in np.linspace(0.5, 5.0, num=15):
            alpha = c / n
            a = _shape_exp(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(state, a.tolist())

        # ---- linear descending (β = c / n) -------------------------
        for c in np.linspace(0.5, 5.0, num=12):
            beta = c / n
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(state, a.tolist())

        # ---- quadratic descending (β = c / n²) --------------------
        for c in np.linspace(0.5, 5.0, num=10):
            beta = c / (n * n)
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(state, a.tolist())

        # ---- uniform (just for completeness) -----------------------
        a = _shape_uniform(n)
        a = a / a.sum() * target
        _maybe_update(state, a.tolist())

        if time.time() >= deadline:
            break

    # ----- Phase 1 – deep sub‑gradient refinement from the current best
    if state["seq"] is not None:
        _run_subgrad(state,
                     init_vec=state["seq"].copy(),
                     target_sum=math.sqrt(2.0 * state["seq"].size),
                     deadline=deadline,
                     max_iter=30000,
                     K=3)

    # ----- Phase 2 – many random restarts with sub‑gradient descent ----
    while time.time() < deadline:
        n = np.random.randint(600, 2501)           # moderate length keeps FFT cheap
        target = math.sqrt(2.0 * n)

        # Randomly pick a base shape
        shape_type = np.random.choice(["exp", "linear", "quad", "uniform"])
        if shape_type == "exp":
            alpha = np.random.rand() * 0.02 + 1e-5
            a0 = _shape_exp(n, alpha)
        elif shape_type == "linear":
            beta = np.random.rand() * 2.0 / max(1, n - 1)
            a0 = _shape_linear_desc(n, beta)
        elif shape_type == "quad":
            beta = np.random.rand() * 1.0 / max(1, n * n)
            a0 = _shape_quad_desc(n, beta)
        else:  # uniform
            a0 = _shape_uniform(n)

        if a0.sum() == 0:
            continue
        a0 = a0 / a0.sum() * target

        _run_subgrad(state,
                     init_vec=a0,
                     target_sum=target,
                     deadline=deadline,
                     max_iter=20000,
                     K=3)

        if state["score"] < 1.45:                 # early exit if very good already
            break

    # ----- Phase 3 – iterated LP refinement (if SciPy is present) -----
    if "seq" in state and state["seq"] is not None and time.time() < deadline:
        a = state["seq"].copy()
        a = _lp_refine(state,
                       a=a,
                       target_sum=math.sqrt(2.0 * a.size),
                       deadline=deadline,
                       max_iters=5)

    # ----- Fallback – uniform sequence ---------------------------------
    if state["seq"] is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        state["seq"] = np.full(n, target / n, dtype=float)

    return state["seq"].tolist()
