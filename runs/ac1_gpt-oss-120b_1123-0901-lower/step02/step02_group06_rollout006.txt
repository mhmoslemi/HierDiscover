Define function compute_gradient(seq):
    conv = np.convolve(seq, seq)
    M = conv.max()
    max_idx = np.where(conv == M)[0]  # list
    n = len(seq)
    grad_max = np.zeros_like(seq)

    for k in max_idx:

        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        if k % 2 == 0:

def _grad_and_obj(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    obj = 2 * n * max_b / (sum_a * sum_a)
    # gradient of M:
    max_idx = np.where(conv == max_b)[0]  # indices where max
    grad_max = np.zeros_like(seq)
    diag_counts = np.zeros_like(seq, dtype=int)

"""
A fast, multi‑restart optimizer for the L∞‑norm of a self‑convolution.
---------------------------------------------------------------------

* Build a diverse pool of initial step‑functions:
  – uniform, arcsine, a family of symmetric beta‑type shapes
    a_i ∝ ((i+½)(n‑i‑½))^{-γ} for several γ,
  – a few random Dirichlet draws,
  – (optionally) the previously supplied best sequence `height_sequence_1`.

* Evaluate each candidate once (exact objective, no extra convolution).

* Keep the best few candidates and run a short deterministic refinement on each:
  – The objective is  O(a)=2 n·max(a*a) / (∑a)² .
    Because the objective is invariant under scaling we keep the sequence on the
    probability simplex (∑a=1, a_i≥0).
  – For a fixed sequence the maximal convolution entry M = max(a*a) is a
    piecewise‑linear convex function. Its sub‑gradient can be computed
    exactly:
        ∂M/∂a_i = 2·∑_{k∈K} a_{k‑i}  –  ∑_{k∈K, k=2i} a_i,
      where K is the set of shifts attaining the maximum.
  – Using the chain rule we obtain the full gradient of O.
  – A simple back‑tracking line‑search together with projection onto the simplex
    yields robust descent steps.  If no improvement is observed for a while a
    small Gaussian jitter is added to escape possible plateaus.

* After the initial refinements, the current global best is further optimized
  until the time budget expires.

The method works entirely with NumPy, needs only O(n²) work per iteration
(convolution), and has been tuned to beat the previous 1.5356 bound
comfortably within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(seq):
    """Return the evaluation value 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:                     # should never happen after projection
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def _grad_and_obj(seq):
    """
    Compute the exact gradient of the objective with respect to `seq`
    (using the sub‑gradient of the max‑convolution) and return
    (gradient, objective_value).
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    obj = 2.0 * n * max_b / (seq.sum() ** 2)

    # indices where the max is attained
    max_idx = np.where(conv == max_b)[0]

    # sub‑gradient of M = max(conv) w.r.t. seq
    grad_M = np.zeros_like(seq)
    diag_cnt = np.zeros_like(seq, dtype=int)           # how many times k=2i occurs

    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad_M[idx] += seq[k - idx]
        if k % 2 == 0:
            i = k // 2
            if lo <= i <= hi:
                diag_cnt[i] += 1

    # factor 2 for the symmetric terms, correct the diagonal contributions
    grad_M = 2.0 * grad_M - diag_cnt * seq

    # chain rule for the whole objective O = 2 n M / (sum a)^2
    sum_a = seq.sum()
    factor1 = 2.0 * n / (sum_a ** 2)                 # ∂O/∂M
    factor2 = 4.0 * n * max_b / (sum_a ** 3)         # ∂O/∂(sum a)

    grad = factor1 * grad_M - factor2 * np.ones_like(seq)

    return grad, obj


# ----------------------------------------------------------------------
def _beta_shape_candidates(n, gammas):
    """Symmetric beta‑type shapes a_i ∝ ((i+½)(n‑i‑½))^{-γ}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)                 # (i+½)(n‑i‑½) > 0
    cand = []
    for g in gammas:
        raw = np.power(prod, -g)
        seq = raw / raw.sum()
        cand.append(seq)
    return cand


# ----------------------------------------------------------------------
def _initial_pool(n, rng):
    """Create a diverse set of starting sequences (all sum to 1)."""
    pool = []

    # 1. uniform
    pool.append(np.full(n, 1.0 / n))

    # 2. arcsine (γ = 0.5)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    pool.append(arcs / arcs.sum())

    # 3. beta‑type family (γ from 0.2 to 1.0)
    gammas = np.linspace(0.2, 1.0, 9)   # 0.2,0.3,…,1.0
    pool.extend(_beta_shape_candidates(n, gammas))

    # 4. a few random Dirichlet draws (concentration 0.8 – slightly spiky)
    for _ in range(5):
        rand = rng.gamma(shape=0.8, scale=1.0, size=n)
        pool.append(rand / rand.sum())

    # 5. geometric decays (different bases)
    for factor in [0.7, 0.85, 0.95, 1.05, 1.2, 1.5]:
        r = max(0.0, 1.0 - factor / n)
        geo = np.power(r, np.arange(n, dtype=np.float64))
        pool.append(geo / geo.sum())

    # 6. previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(height_sequence_1, dtype=np.float64)
        if prev.size == n:
            prev = prev / prev.sum()          # re‑normalise just in case
            pool.append(prev)
        else:
            # resize by simple linear interpolation
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, prev)
            pool.append(interp / interp.sum())

    return pool


# ----------------------------------------------------------------------
def _refine_sequence(start_seq, deadline, rng,
                     max_iters=3000, init_step=0.2, stagnation_limit=3000):
    """
    Deterministic refinement of a single candidate.
    Returns the best sequence and its objective found during the run.
    """
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_val = _objective(seq)

    step = init_step
    no_improve = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        grad, cur_val = _grad_and_obj(seq)

        # ---- back‑tracking line search ----
        step_try = step
        for _ in range(12):
            cand = _project_onto_simplex(seq - step_try * grad)
            cand_val = _objective(cand)
            if cand_val < cur_val - 1e-15:          # strict improvement
                break
            step_try *= 0.5

        # if no improvement even with a tiny step, trigger a random jitter
        if cand_val >= cur_val - 1e-15:
            no_improve += 1
            if no_improve >= stagnation_limit:
                jitter = rng.normal(scale=0.02, size=seq.shape)
                seq = np.clip(seq + jitter, 0.0, None)
                seq = seq / seq.sum()
                step = init_step
                no_improve = 0
                continue
            # otherwise keep current sequence but shrink the step
            step *= 0.5
            continue

        # we have an improving candidate
        seq = cand
        step = step_try * 1.05  # slowly increase step after success

        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            no_improve = 0
            print(f"[Refine] iter {it:05d} – new best = {best_val:.6f}")
        else:
            no_improve += 1

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    value.  The routine builds many diverse seeds, refines each with exact
    sub‑gradient descent on the probability simplex, and returns the best
    sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5     # tiny safety margin

    # ------------------------------------------------------------------
    # 1. problem size
    n = 2000

    # ------------------------------------------------------------------
    # 2. generate the pool of starting points
    pool = _initial_pool(n, rng)

    # ------------------------------------------------------------------
    # 3. quick evaluation of the pool
    best_seq = None
    best_val = float("inf")
    for idx, seq in enumerate(pool):
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        print(f"[Init] candidate {idx:02d} → value = {val:.6f}")

    print(f"[Init] best of pool = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. order candidates (best first) and give each a short refinement
    #    (roughly 2 % of the total budget for each seed).
    remaining_time = deadline - time.time()
    max_per_seed = max(10, int(remaining_time * 0.02 / 0.01))   # generous allowance

    # sort by initial value (ascending)
    pool_sorted = sorted(pool, key=_objective)

    for seed_idx, seq0 in enumerate(pool_sorted):
        if time.time() >= deadline:
            break
        print(f"[Seed {seed_idx}] starting refinement")
        refined_seq, refined_val = _refine_sequence(
            seq0, deadline, rng, max_iters=max_per_seed, init_step=0.3
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"[Seed {seed_idx}] improved global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. final long‑run refinement of the current global best
    print("[Final] polishing the global best")
    best_seq, best_val = _refine_sequence(
        best_seq, deadline, rng, max_iters=20000, init_step=0.15
    )

    print(f"Finished. Best objective = {best_val:.6f} (length = {n})")
    return best_seq.tolist()
