def run_optimization(n, init, time_limit):
    target = np.sqrt(2*n)
    a = project_onto_simplex(init.astype(float), target)
    # algorithm parameters
    # Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5  # initial learning
    # smoothing schedule
    alpha = 0.5
    target_alpha = 10.0
    # step counters
    t = 0
    best_local_seq = a.copy()
    best_local_score = np.inf

    while time.time() < time_limit:
        # evaluate
        conv = conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)  # same as evaluate
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
        # compute gradient with smoothing
        # exponent
        # subtract max to avoid overflow
        diff = conv - max_val
        # compute w
        w = np.exp(alpha * diff)
        Z = w.sum()
        # correlation w with a
        # use numpy correlate valid
        g = np.correlate(w, a, mode='valid')
        grad = 2.0 * g / Z
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # enforce nonnegativity, project to simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target)
        a = a_new

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a smooth‑max surrogate and Adam optimisation on the simplex.
The algorithm:
- Project the current vector onto the simplex {x_i≥0 , Σx_i = √(2n)}.
- Approximate max(conv) by the smooth function
        smax_α(a) = (1/α)·log Σ_i exp(α·(a*a)[i])
  (α is slowly increased during the run).
- The gradient of smax_α can be written in closed form using a single
  correlation, which we evaluate with an FFT‑based convolution.
- Adam with a decaying learning‑rate performs the updates; every
  iteration the vector is re‑projected onto the simplex.
- Many independent restarts with different lengths `n` (≈1500‑3000) are
  executed until the time budget expires. The best sequence found
  (according to the original `evaluate_sequence`) is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – Σ x_i = s , x_i ≥ 0
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full convolution (real FFT) – a (len n) with itself → length 2n‑1
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Full convolution of two (possibly) different length arrays.
# --------------------------------------------------------------------------- #
def _conv_full_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of x and y using FFT (full mode)."""
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << (n.bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the smooth‑max of the self‑convolution.
# --------------------------------------------------------------------------- #
def _smooth_max_grad(a: np.ndarray, alpha: float):
    """
    Returns (grad, smooth_max) where
        smooth_max ≈ max(a*a)  (controlled by alpha)
    and grad = ∇_a smooth_max.
    """
    n = a.shape[0]
    conv = _conv_full_fft(a)
    max_val = conv.max()
    # stabilized exponentials (subtract the max to avoid overflow)
    diff = conv - max_val
    w = np.exp(alpha * diff)               # shape 2n‑1, all >=0
    Z = w.sum() + 1e-30                     # avoid division by zero

    # correlation w with a (valid mode) → length n
    # correlation can be obtained from convolution of w and reversed a
    corr = _conv_full_fft_two(w, a[::-1])
    # valid slice: positions n‑1 … 2n‑2 (inclusive)
    g = corr[n - 1 : 2 * n - 1]

    grad = 2.0 * g / Z
    # smooth approximation of max(conv)
    smooth_max = max_val + (1.0 / alpha) * np.log(Z)
    return grad, smooth_max


# --------------------------------------------------------------------------- #
# Adam optimiser on the simplex for a fixed length n.
# --------------------------------------------------------------------------- #
def _optimize_for_n(
    n: int,
    init_vec: np.ndarray,
    time_deadline: float,
    global_best_score: float,
):
    """
    Perform a single optimisation run for a given length ``n``.
    Returns (best_seq, best_score) discovered in this run.
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6                       # initial learning rate
    alpha = 0.5                    # start smoothing parameter
    alpha_max = 12.0
    t = 0

    best_local_seq = a.copy()
    best_local_score = float("inf")

    while time.time() < time_deadline:
        # ----- evaluate current objective (true max) -----
        conv = _conv_full_fft(a)
        cur_max = conv.max()
        if cur_max < best_local_score:
            best_local_score = float(cur_max)
            best_local_seq = a.copy()
            # also report improvement wrt the *global* best
            if cur_max < global_best_score:
                print(
                    f"[{time.time():.2f}]  NEW GLOBAL BEST = {cur_max:.6f} (n={n})"
                )
        # ----- smoothed gradient step -----
        grad, smooth_max = _smooth_max_grad(a, alpha)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # enforce simplex feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        a = a_new

        # ----- schedule updates -----
        # increase smoothing (make the max sharper)
        if alpha < alpha_max:
            alpha = min(alpha * 1.015, alpha_max)

        # slowly decay learning‑rate
        if t % 200 == 0:
            lr *= 0.97

        # occasional random perturbation to escape plateaux
        if np.random.rand() < 0.008:
            noise = np.random.randn(n) * (0.02 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    return best_local_seq, best_local_score


# --------------------------------------------------------------------------- #
# Main entry point required by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    The best sequence discovered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # keep the overall best sequence / score
    best_seq = None
    best_score = float("inf")

    # possible problem sizes (empirically good range)
    n_choices = [1400, 1600, 1800, 2000, 2200, 2500, 2800, 3000]

    # ------------------------------------------------------------------- #
    # Main outer loop – independent restarts.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a modest slice (≈20–30 s) for one optimisation run
        run_time = min(28.0, remaining * 0.9)
        if run_time < 5.0:
            break

        n = int(np.random.choice(n_choices))
        # initialise: use the global best (if any) or a fresh random vector
        if best_seq is not None and np.random.rand() < 0.33:
            src = np.asarray(best_seq, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        # run a single optimisation instance
        seq, score = _optimize_for_n(
            n=n, init_vec=init, time_deadline=time.time() + run_time, global_best_score=best_score
        )
        if score < best_score:
            best_score = float(score)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}]  *** UPDATED GLOBAL BEST = {best_score:.6f} (n={n}) ***")

        # early exit if we have already broken the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – in the unlikely event that nothing was found.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
