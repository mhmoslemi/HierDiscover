"""
Propose a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation uses three ideas that differ from the previous
baseline:

1. **Smoothed objective** – the non‑smooth max‑norm of the autocorrelation
   is approximated by a log‑sum‑exp with a temperature parameter `t`.  For
   small `t` the function is almost convex, for large `t` it approaches the
   true max.  A continuation schedule (t = 0.5 → 20) is used.

2. **Projected Adam optimisation** – an Adam‑style adaptive step is taken
   on the smoothed objective and the iterates are projected onto the
   simplex `{x ≥ 0, Σx = √(2 n)}` after each step.  This respects the
   homogeneity of the original problem while keeping all heights non‑negative.

3. **Multi‑restart + cheap parametric sweep** – a handful of low‑dimensional
   families (exponential, linear, quadratic, triangular) are tried first,
   then many random restarts (different lengths and random initialisations)
   are performed until the time budget expires.  If SciPy is available a
   final LP refinement (the “SOTA step”) is attempted.

The routine is completely self‑contained, uses only NumPy (and optionally
SciPy) and respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries summing to s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: Full (linear) convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return full convolution of 1‑D real arrays a and b using FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n + m - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Alias for the autocorrelation a*a (full convolution)."""
    return _full_convolution(a, a)


# ----------------------------------------------------------------------
# Helper: safe call to the external evaluation routine
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Return the score from evaluate_sequence or +inf on error."""
    try:
        return evaluate_sequence(seq)   # provided by the harness
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Shape factories (non‑normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _obj_and_grad(a: np.ndarray, t: float):
    """
    Compute the smoothed objective f_t(a) = log_sum_exp(t*conv)/t
    and its gradient w.r.t. a.
    Returns (f, grad) where `grad` has the same shape as `a`.
    """
    # full autocorrelation
    conv = _conv_self(a)                     # length 2n‑1
    max_c = conv.max()

    # stable log‑sum‑exp
    exp_terms = np.exp(t * (conv - max_c))
    Z = exp_terms.sum()
    f = max_c + np.log(Z) / t

    # softmax weights (normalized)
    w = exp_terms / Z    # shape (2n‑1,)

    # gradient = 2 * (w * a_rev) convolution, slice the central window
    a_rev = a[::-1]
    conv_ws_rev = _full_convolution(w, a_rev)   # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_ws_rev[n - 1 : n - 1 + n]  # central n entries
    return f, grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine respects a time budget (default 1000 s) and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the global best
    # ------------------------------------------------------------------
    def _maybe_update(cand: np.ndarray):
        nonlocal best_seq, best_score
        score = _evaluate(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={len(cand)})")

    # ------------------------------------------------------------------
    # 1. Fast sweep over simple parametric families
    # ------------------------------------------------------------------
    lengths = list(range(300, 1601, 200)) + [1200, 1400, 1600]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-4, -0.3, num=10):
            a = _shape_exp(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # reversed exponential
        for alpha in np.logspace(-4, -0.3, num=8):
            a = _shape_exp_rev(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=8):
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=8):
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # triangular (symmetric) shape
        a = _shape_triangular(n)
        if a.sum() != 0:
            a = a / a.sum() * target
            _maybe_update(a)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2. If a global best is present, run a refined Adam optimisation on it
    # ------------------------------------------------------------------
    def _run_adam(start_vec: np.ndarray):
        """
        Perform Adam optimisation of the smoothed objective on `start_vec`.
        The vector is assumed to already satisfy the simplex constraint.
        """
        nonlocal best_seq, best_score
        a = start_vec.astype(float)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        # Adam hyper‑parameters
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.08                      # base learning rate
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # Continuation schedule for temperature t
        t_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 15.0, 20.0]
        it_global = 0

        for t in t_schedule:
            # a few hundred iterations per temperature
            for _ in range(250):
                if time.time() >= deadline:
                    return

                f_val, grad = _obj_and_grad(a, t)

                it_global += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)

                m_hat = m / (1.0 - beta1 ** it_global)
                v_hat = v / (1.0 - beta2 ** it_global)

                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a -= step
                a = _project_onto_simplex(a, target)

                # Periodic evaluation of the true (non‑smoothed) score
                if it_global % 10 == 0:
                    _maybe_update(a)

                # Stop early if we have already reached the target bound
                if best_score < 1.503:
                    return

    # ------------------------------------------------------------------
    # Refine the current best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        _run_adam(np.array(best_seq, copy=True))

    # ------------------------------------------------------------------
    # 3. Random restarts with the Adam optimiser
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length – moderate to keep each iteration cheap
        n = np.random.randint(800, 2501)

        target = math.sqrt(2.0 * n)

        # random exponential‑type seed (often a good baseline)
        alpha = np.random.rand() * 0.02 + 1e-5
        a0 = _shape_exp(n, alpha)
        if a0.sum() == 0:
            continue
        a0 = a0 / a0.sum() * target

        _run_adam(a0)

        if best_score < 1.503:
            break

        # occasionally try a completely random start (uniform)
        if np.random.rand() < 0.15:
            a_rand = np.random.rand(n)
            a_rand = a_rand / a_rand.sum() * target
            _run_adam(a_rand)

    # ------------------------------------------------------------------
    # 4. Optional LP refinement (the classic “LP‑step”) if SciPy exists
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = np.array(best_seq, copy=True)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            rows = 2 * n - 1

            # Build matrix for constraints (a * g)_k ≤ M
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                cols = np.arange(low, high + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),                # maximise Σ g_j
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                break

            g = res.x
            s = g.sum()
            if s <= 0:
                break
            g = g * (target / s)

            # convex combination step (small mixing)
            t_mix = 0.18
            a = (1.0 - t_mix) * a + t_mix * g

            _maybe_update(a)

            if best_score < 1.503 or time.time() + 0.5 > deadline:
                break

    # ------------------------------------------------------------------
    # 5. Fallback – uniform sequence if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
