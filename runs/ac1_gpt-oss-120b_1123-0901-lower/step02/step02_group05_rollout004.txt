def _smooth_gradient(a, tau):
    conv = np.convolve(a, a)  # length 2n-1
    max_val = conv.max()
    # compute weights
    # stable exponent: subtract max * maybe large?
    # w = np.exp(tau * (conv - max_val)) maybe stable (max->0)
    w = np.exp(tau * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # gradient = 2 * convolution of reversed a and w
    a_rev = a[::-1]
    conv_rev = np.convolve(a_rev, w)
    grad = 2 * conv_rev[len(a)-1:len(a)-1+len(a)]  # slice n entries
    return grad

"""
search for a non‑negative step function that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The score equals ``max(conv(a, a))`` when the sequence is normalised to
sum(a)=sqrt(2n).  We therefore minimise the infinity‑norm of the
auto‑convolution under a simplex constraint.

Algorithm
---------

* Multiple random restarts with different lengths (200–2000).
* For each restart we run a **smoothed‑gradient descent**.
  The non‑smooth objective ``max(conv)`` is approximated by the
  log‑sum‑exp of the convolution entries; its gradient is a weighted
  sum of the individual convolution gradients.
  The smoothing parameter τ is increased linearly from a small value
  (≈0.5) to a large value (≈20) so that the optimisation starts
  aggressive and finishes with a near‑exact sub‑gradient step.
* After each gradient step the iterate is projected onto the simplex
  {x ≥ 0, Σx = √(2n)}.
* Occasionally a tiny Gaussian mutation is applied to escape shallow
  local minima.
* The best sequence found so far is kept globally; the optimisation
  loop stops early if the target score 1.503 is beaten.
* A final short refinement is performed on the global best with a
  longer time budget.

The implementation avoids calling ``evaluate_sequence`` inside the
inner loops – the current value of the objective is simply the maximum
convolution entry (the normalisation guarantees the two are identical).
Only the final candidate is evaluated with ``evaluate_sequence`` to
obtain the official score.

The function ``propose_candidate`` respects the 1000 s time‑budget and
returns a list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective: soft‑max of the convolution
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, tau: float):
    """
    Compute a smooth approximation of max(conv(a,a)) and its gradient.

    Returns
    -------
    max_val : float
        The true maximum convolution entry (used for step‑size scaling).
    grad : np.ndarray
        Gradient of the smoothed objective w.r.t. a (shape = a.shape).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    max_val = conv.max()

    # soft‑max weights (stable because we subtract the max)
    w = np.exp(tau * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum

    # gradient_i = 2 * Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    conv_rev = np.convolve(a_rev, w)             # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return max_val, grad


# ----------------------------------------------------------------------
# Very small mutation helper (adds Gaussian noise and re‑projects)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Smooth gradient descent with a schedule for τ
# ----------------------------------------------------------------------
def _local_descent_smooth(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.08,
    tau_start: float = 0.5,
    tau_end: float = 20.0,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Smoothed projected gradient descent.

    Parameters
    ----------
    init : np.ndarray
        Starting point (will be projected onto the simplex).
    target_sum : float
        Desired ℓ₁‑norm (√(2n)).
    time_limit : float
        Seconds allocated to this run.
    base_step : float
        Base step size before scaling.
    tau_start / tau_end : float
        Smoothing parameter schedule.
    print_tag : str
        Prefix used for progress prints.

    Returns
    -------
    best_seq : np.ndarray
        Best sequence seen during the run.
    best_score : float
        Corresponding value of max(conv) (i.e. the evaluation score).
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial score (exact max of the convolution)
    conv = np.convolve(a, a)
    best_score = float(conv.max())
    best_seq = a.copy()

    t0 = time.time()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < t0 + time_limit:
        elapsed = time.time() - t0
        # linear schedule for τ
        tau = tau_start + (tau_end - tau_start) * min(1.0, elapsed / time_limit)

        max_val, grad = _smooth_max_and_grad(a, tau)

        # make the direction sum‑zero (project out the all‑ones component)
        mean_g = np.mean(grad)
        d = -grad + mean_g

        # optional simple momentum
        momentum = 0.9 * momentum + 0.1 * d
        d = momentum

        # adaptive step size (scale by the current max entry)
        max_abs = np.max(np.abs(d))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = base_step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate new point (exact max is cheap)
        conv_new = np.convolve(a_new, a_new)
        score = float(conv_new.max())

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})"
            )

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
        else:
            a = a_new

        # simple decay of the base step every few hundred iterations
        if it % 300 == 0 and it > 0:
            base_step *= 0.9

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple resize (interpolation / down‑sampling) to reuse a known good sequence
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        # pick equally spaced indices
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        # linear interpolation
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Helpers for a few deterministic initial shapes
# ----------------------------------------------------------------------
def _init_sinusoidal(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine shape, all entries positive, normalised to target_sum."""
    x = np.arange(n) + 0.5
    raw = np.sin(np.pi * x / n)          # positive on (0,π)
    raw = raw / raw.sum() * target_sum
    return raw.astype(float)


def _init_exponential(n: int, target_sum: float) -> np.ndarray:
    """Exponential decay shape, normalised to target_sum."""
    raw = np.exp(-np.arange(n) / (n / 5.0))
    raw = raw / raw.sum() * target_sum
    return raw.astype(float)


def _init_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """Random Dirichlet sample (uniform over the simplex)."""
    raw = np.random.dirichlet(np.ones(n))
    return (raw * target_sum).astype(float)

            # occasionally perturb the length a bit
            if np.random.rand() < 0.3:
                n = int(n * np.random.uniform(0.85, 1.15))
                n = max(200, min(2500, n))
        else:
            n = np.random.randint(200, 2100)

        target_sum = math.sqrt(2.0 * n)

        # Prepare an initial vector (one of several plausible shapes)
        init_choice = np.random.rand()
        if best_seq is not None and init_choice < 0.5:
            init = _resize_sequence(best_seq, n)
        elif init_choice < 0.75:
            init = _init_sinusoidal(n, target_sum)
        elif init_choice < 0.9:
            init = _init_exponential(n, target_sum)
        else:
            init = _init_dirichlet(n, target_sum)

        # Run the optimiser for the allotted slice of time
        time_left = deadline - time.time()
        if time_left < 8.0:
            break
        cand_seq, cand_score = _run_one(n, init, time_left)

        # Update the global best if we made progress
        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})")
            if best_score < 1.503:          # target achieved – stop early
                break

    # ------------------------------------------------------------------
    # Final refinement on the best sequence (use remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        n_fin = best_seq.size
        target_fin = math.sqrt(2.0 * n_fin)
        print(f"[{time.time():.2f}] Final refinement (remaining {remaining:.1f}s)")
        best_seq, best_score = _local_descent_smooth(
            best_seq,
            target_fin,
            remaining,
            base_step=0.06,
            tau_start=0.5,
            tau_end=30.0,
            print_tag="final"
        )
        print(f"[{time.time():.2f}] Final refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
