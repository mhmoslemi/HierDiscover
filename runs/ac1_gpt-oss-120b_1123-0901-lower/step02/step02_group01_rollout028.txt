The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `*` is the linear convolution used by NumPy.  
The objective is scale‑invariant, therefore we can (and do) enforce the
simplex constraint \(\sum_i a_i = 1\) and only have to keep the non‑negativity.

The algorithm below is a **projected Adam optimiser** that works on a
smooth approximation of the `max` (log‑sum‑exp) and automatically switches
to the exact sub‑gradient once the smoothing parameter is large enough.
A few different initialisations (geometric, Dirichlet and the previously
found best sequence) are tried; the best sequence over the whole time
budget is returned.

Key tricks that improve over the reference implementation  

* **Adam** instead of plain gradient descent – the adaptive learning rate
  removes the need for a manually decaying step size.  
* **Continuation on the smooth‑max parameter** – λ is increased
  geometrically; when λ becomes large the gradient collapses to the
  true sub‑gradient of the max‑convolution, giving a very precise final
  refinement.  
* **Stagnation handling** – if the best objective has not improved for a
  few seconds we inject a tiny random perturbation and reset the Adam
  moments, helping the search escape shallow minima.  
* **Multiple restarts** – a small pool of diverse starting points
  increases the chance of finding a lower local minimum.  

The whole routine respects the 1000 s time limit, prints progress
information and finally returns a plain Python list ready for
`evaluate_sequence`.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The algorithm is a projected Adam optimiser on a smooth‐max surrogate of
the max‑convolution.  It starts from several diverse initial points
(geometric series, Dirichlet draws, and the previously discovered best
sequence if available), runs a continuation on the smoothing parameter
λ and automatically switches to the exact sub‑gradient once λ is large.
Stagnation is broken by a tiny random shake and a reset of the Adam
moments.  The best sequence seen before the time budget expires is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                       # all numbers non‑positive
        return np.full_like(v, 1.0 / v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using FFT (zero‑padded, length 2·n–1)
def _full_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length, returned as length 2·n‑1)."""
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by evaluate_sequence (scale‑invariant, sum=1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv) for a simplex‑normalised vector a."""
    n = a.shape[0]
    conv = _full_convolution(a, 1 << int(np.ceil(np.log2(2 * n - 1))))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate φλ(a) = (1/λ)·log∑exp(λ·conv)
def _softmax_gradient(
    a: np.ndarray,
    conv: np.ndarray,
    lam: float,
    size_corr: int,
) -> np.ndarray:
    """
    Compute ∇φλ(a) = 2·(w ★ a) where w_k ∝ exp(λ·conv_k).
    """
    # numerically stable soft‑max weights
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:                       # should not happen, but guard
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # cross‑correlation w ★ a (linear, not circular)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    n = a.shape[0]
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used when λ is large)
def _max_convolution_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max_k (a*a)[k]; i.e. 2·shifted a at the
    index where the convolution attains its maximum.
    """
    n = a.shape[0]
    k_max = int(np.argmax(conv))                 # location of the peak
    shift = k_max - (n - 1)                      # true lag (may be negative)

    if shift >= 0:
        # grad[i] = 2·a[i‑shift] for i ≥ shift, else 0
        grad = np.concatenate((np.zeros(shift, dtype=a.dtype), a[: n - shift]))
    else:
        # shift < 0  →  grad[i] = 2·a[i‑shift] = 2·a[i+|shift|]
        s = -shift
        grad = np.concatenate((a[s:], np.zeros(s, dtype=a.dtype)))

    return 2.0 * grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The function respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # Safety margin for the interpreter / print statements
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 2000 gives a good trade‑off between expressivity
    # and speed (FFT size 4096 → very fast).
    n = 2000

    # Pre‑computed FFT sizes
    size_fft = 1 << int(np.ceil(np.log2(2 * n - 1)))        # ≥ 2n−1
    size_corr = 1 << int(np.ceil(np.log2(3 * n - 2)))    # ≥ (2n−1)+n−1

    # --------------------------------------------------------------
    # Storage for the best solution found overall
    best_seq = None
    best_val = float("inf")
    last_improve = time.time()

    # --------------------------------------------------------------
    # Helper: evaluate a candidate and update the global best if needed
    def _maybe_update(candidate):
        nonlocal best_seq, best_val, last_improve
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            last_improve = time.time()
            print(f"[New best]  obj={val:.6f}  time={time.time():.2f}")

    # --------------------------------------------------------------
    # 1. Build a modest pool of diverse initial points
    init_pool = []

    #   a) geometric progressions (a_i ∝ r^i) with many decay rates
    for r in np.linspace(0.90, 0.9999, 20):
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        init_pool.append(seq)

    #   b) random Dirichlet draws
    for _ in range(15):
        seq = np.random.dirichlet(np.ones(n))
        init_pool.append(seq)

    #   c) the previous best sequence if it exists (global variable)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            if prev.size != n:
                # interpolate / down‑sample to length n
                x_old = np.linspace(0, 1, prev.size)
                x_new = np.linspace(0, 1, n)
                prev = np.interp(x_new, x_old, prev)
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            init_pool.append(prev)
    except NameError:
        pass  # height_sequence_1 not defined – ignore

    # --------------------------------------------------------------
    # Evaluate initial pool and keep the best one as a start point
    for seq in init_pool:
        if time.time() > deadline:
            break
        _maybe_update(seq)

    # If nothing survived (highly unlikely) fall back to a uniform start
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2. Optimisation driver – we perform a few independent runs
    #    (different random restarts) and keep the overall best.
    run_id = 0
    while time.time() < deadline:
        run_id += 1
        # ---------- initialise this run ----------
        # Occasionally re‑use the global best as a warm‑start
        if run_id % 3 == 0 and best_seq is not None:
            a = best_seq.copy()
        else:
            a = np.random.dirichlet(np.ones(n))

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.05                    # base learning rate for Adam
        lam = 0.5                    # start of the smooth‑max continuation
        lam_factor = 1.04            # increase λ moderately each iteration
        lam_subgrad_start = 150.0    # switch to exact sub‑gradient beyond this
        iteration = 0

        # ---------- inner optimisation loop ----------
        while time.time() < deadline:
            iteration += 1

            # --- convolution ---
            conv = _full_convolution(a, size_fft)

            # --- decide which gradient to use ---
            if lam < lam_subgrad_start:
                # soft‑max surrogate gradient
                grad = _softmax_gradient(a, conv, lam, size_corr)
            else:
                # exact sub‑gradient of the max‑convolution
                grad = _max_convolution_subgradient(a, conv)

            # --- Adam update (projected) ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            # bias‑correction
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_next = a - step
            a_next = _project_onto_simplex(a_next)

            # --- evaluate exact objective ----------
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                last_improve = time.time()
                print(f"[Iter {iteration:06d} | Run {run_id}] new best obj={val:.6f}")

            # --- accept the new point (always) ---
            a = a_next

            # --- continuation on λ ---
            lam *= lam_factor

            # --- stagnation handling: after ~2 s without improvement,
            #     add a tiny random shake and reset Adam moments.
            if time.time() - last_improve > 2.0:
                noise = (np.random.rand(n) - 0.5) * 1e-4
                a = _project_onto_simplex(a + noise)
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()
                print(f"[Run {run_id}] shake injected (no improvement for 2 s)")

            # Stop the inner loop if we have spent a reasonable chunk of time
            # on this run (≈ 8 s) – leave room for another restart.
            if iteration % 2000 == 0 and time.time() - deadline > -8.0:
                break

        # end of one run
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 3. Return the best sequence found (as a plain list)
    print(f"Finished. Best objective = {best_val:.6f} (length {n})")
    # `evaluate_sequence` will clip negatives and very large numbers,
    # but the projection guarantees non‑negativity already.
    return best_seq.tolist()
```
