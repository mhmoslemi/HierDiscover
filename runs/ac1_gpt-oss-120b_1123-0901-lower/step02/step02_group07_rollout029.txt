def local_optimize(seq, max_iters, deadline):

    cur_score = _objective(cur_seq)

"""
Propose a step‑function (non‑negative sequence) that improves the
evaluation objective 2·n·max(a*a)/(sum a)².

The algorithm follows a multi‑start projected (soft‑max) gradient descent:

1.  Several seeds are built (global seed, uniform, Dirichlet draws,
    a few parametric families, a triangular shape …) and projected onto
    the probability simplex (sum = 1, a_i ≥ 0).

2.  For each seed a local optimisation is performed.
    * The exact self‑convolution `c = a*a` is computed with FFT.
    * A smoothed version of the max‑operator is used:
          F_β(a) = log‑sum‑exp(β·c)/β   (β → ∞ → max(c)).
      Its gradient w.r.t. `a` is
          ∇F_β = Σ_k p_k·a_{k−i}   with   p_k = exp(β·c_k)/Σ exp(β·c).
      The gradient of the whole objective is obtained by the chain rule
      and projected back onto the simplex.
    * A back‑tracking line‑search accepts the first step that reduces the
      *true* objective (the hard max).  If no step succeeds a tiny random
      mass‑transfer move is tried.
    * The smoothing parameter β is slowly increased (from 0.5 up to ~30)
      so that the optimisation first sees a smooth landscape and later
      focuses on the true max‑norm.

3.  The best sequence seen over all seeds and all iterations is returned.
    The routine respects the 1000 s time budget (a safety margin of 0.5 s
    is kept) and prints occasional progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v):
    """Return the Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself (length 2n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
def _convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b."""
    m = a.shape[0] + b.shape[0] - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact evaluation used by the competition (scale‑invariant)."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    return float(2.0 * len(seq) * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Simple parametric families (used as seeds)
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1−α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{‑p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a‑1}(n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _triangular_seq(n: int) -> np.ndarray:
    """Tent shape – rises to the centre then falls symmetrically."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    seq = centre - np.abs(i - centre) + 1.0
    return seq / seq.sum()


def _inverse_triangular_seq(n: int) -> np.ndarray:
    """Inverted tent – smallest in the middle, larger towards the edges."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    seq = np.abs(i - centre) + 1.0
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the given
    evaluation function.  The function is allowed to run for up to
    ``budget_s`` seconds and must return a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Fixed problem size – a larger n gives more freedom but also more
    # work per iteration.  2000 is a good compromise (used by the runner).
    n = 2000

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Collect a pool of starting points.
    seeds = []

    # 1.  Global seed, if it exists.
    if "height_sequence_1" in globals():
        try:
            glob = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            glob = np.clip(glob, 0.0, 1000.0)
            if glob.sum() > 0.0:
                glob = glob / glob.sum()
                seeds.append(glob)
                print("[Init] loaded global seed")
        except Exception:
            pass

    # 2.  Simple deterministic seeds.
    seeds.append(np.full(n, 1.0 / n))          # uniform
    seeds.append(_triangular_seq(n))           # tent
    seeds.append(_inverse_triangular_seq(n))   # inverted tent

    # 3.  Tiny parametric sweep.
    for alpha in np.linspace(0.5, 2.0, 5):
        seq = _geom_seq(n, alpha)
        if seq is not None:
            seeds.append(seq)
    for p in np.linspace(0.2, 2.5, 5):
        seeds.append(_powerlaw_seq(n, p))
    for a in np.linspace(0.5, 3.0, 3):
        for b in np.linspace(0.5, 3.0, 3):
            seeds.append(_beta_seq(n, a, b))

    # 4.  Random Dirichlet draws.
    for _ in range(30):
        seeds.append(_random_simplex(n))

    # ------------------------------------------------------------------
    # Main optimisation loop – run a short local optimiser from each seed.
    for seed_idx, init_seq in enumerate(seeds, start=1):
        if time.time() > deadline:
            break
        seq = _project_onto_simplex(init_seq.copy())
        cur_score = _objective(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[Seed {seed_idx:02d}] initial score = {cur_score:.6f}")

        # Local optimiser parameters
        base_step = 1.0          # initial line‑search step size
        beta = 0.5               # start with a very smooth approximation
        max_iters = 4000

        for it in range(max_iters):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Compute exact self‑convolution (required for the true objective)
            conv = _self_convolve(seq)

            # Smoothed max via log‑sum‑exp
            max_c = conv.max()
            exp_shifted = np.exp(beta * (conv - max_c))
            sum_exp = exp_shifted.sum()
            p_weights = exp_shifted / sum_exp     # softmax probabilities

            # Gradient of the smoothed max (weighted convolution)
            p_rev = p_weights[::-1]
            weighted = _convolve(seq, p_rev)      # length = 3n‑2
            grad_f = weighted[2 * n - 2 - np.arange(n)]

            s = seq.sum()
            # Smooth surrogate of the objective: F = log‑sum‑exp / β
            F = (max_c + np.log(sum_exp) / beta)

            # Full gradient of the original objective (chain rule)
            grad_obj = (2.0 * n / (s * s)) * grad_f - (4.0 * n * F / (s * s * s)) * seq

            # --------------------------------------------------------------
            # Back‑tracking line‑search on the *true* (hard‑max) objective.
            step = base_step
            improved = False
            while step > 1e-8:
                cand = _project_onto_simplex(seq - step * grad_obj)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    improved = True
                    break
                step *= 0.5

            if improved:
                seq = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = seq.copy()
                    print(f"[Seed {seed_idx:02d}][Iter {it:05d}] "
                          f"new best = {best_score:.6f}")
                # successful step → allow a slightly larger step next time
                base_step = max(step * 1.2, 1e-6)
                # increase smoothing (focus more on the true max)
                beta = min(beta * 1.05, 30.0)
                continue

            # --------------------------------------------------------------
            # No improvement via the gradient – random local perturbation.
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)    # move ≤1 % of total mass
            if delta > 0.0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Seed {seed_idx:02d}][Iter {it:05d}] "
                              f"random move improved to {best_score:.6f}")
                # keep the step size moderate after a random move
                base_step = max(base_step * 0.7, 1e-6)

        # end of local optimisation for this seed
        # (continue with the next seed)

    # ------------------------------------------------------------------
    # Fallback – should never happen
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    print(f"Final best objective = {best_score:.6f} (length {n})")
    # convert to plain Python list as required by the harness
    return best_seq.tolist()
