def _max_conv_and_grad_multi(a, epsilon_frac=0.01):
    conv = np.convolve(a, a)
    max_val = conv.max()
    # Indices within epsilon of max
    idxs = np.where(conv >= max_val * (1 - epsilon_frac))[0]
    grad = np.zeros_like(a)
    n = len(a)
    for k in idxs:
        # overlapped region
        low = max(0, k - (n-1))
        high = min(n-1, k)
        i = np.arange(low, high+1)
        j = k - i
        grad[i] += 2.0 * a[j]
    # average
    grad = grad / len(idxs) if len(idxs) > 0 else np.zeros_like(a)
    return max_val, grad, idxs

def _local_descent_multi(init, target_sum, time_limit):
    a = _project_onto_simplex(init.astype(float), target_sum)
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    t0 = time.time()
    step = 0.1
    it = 0
    while time.time() < t0 + time_limit:
        conv = np.convolve(a, a)
        max_val = conv.max()
        # find all near max indices
        eps = 0.001
        near_idx = np.where(conv >= max_val * (1 - eps))[0]
        grad = np.zeros_like(a)
        n = len(a)
        for k in near_idx:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        grad = grad / max(1, near_idx.size)
        # maintain sum
        d = -grad + np.mean(grad)
        # step: adapt
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs
        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate
        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] new best {sc:.6f}")
        # acceptance rule: if sc better else maybe accept with small probability (simulated annealing)
        if sc < best_score or np.random.rand() < 0.001:
            a = a_new
        else:
            a = a_new  # we can still move
        # step decay
        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1
    return best_seq, best_score
```

def _max_conv_and_grad_multi(a: np.ndarray, eps_frac: float = 0.01) -> tuple[float, np.ndarray]:
    conv = np.convolve(a, a)
    max_val = conv.max()
    near_idxs = np.where(conv >= max_val * (1 - eps_frac))[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in near_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if near_idxs.size > 0:
        grad /= near_idxs.size
    return max_val, grad
```

def _local_descent_multi(init, target_sum, time_limit,
                         base_step=0.1, eps_start=0.02, eps_end=0.001,
                         print_tag=""):
    a = _project_onto_simplex(init.astype(float), target_sum)
    # Symmetrize
    a = (a + a[::-1]) / 2.0
    a = a * (target_sum / np.sum(a))
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    start = time.time()
    it = 0
    step = base_step
    m = np.zeros_like(a)  # momentum
    while time.time() < start + time_limit:
        elapsed = time.time() - start
        # linear decay of epsilon
        eps = eps_start + (eps_end - eps_start) * (elapsed / time_limit)
        max_val, grad = _max_conv_and_grad_multi(a, eps_frac=eps)
        # zero-mean direction
        d = -grad + np.mean(grad)
        # momentum update
        beta = 0.9
        m = beta * m + (1 - beta) * d
        # step size scaling
        max_abs = np.max(np.abs(m)) if np.max(np.abs(m)) > 0 else 1.0
        eta = step * max_val / max_abs
        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target_sum)
        # enforce symmetry and re-normalize
        a_new = (a_new + a_new[::-1]) / 2.0
        s = np.sum(a_new)
        if s > 0:
            a_new = a_new * (target_sum / s)
        # evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new best {score:.6f} (n={len(a)})")
        # acceptance: always move
        a = a_new
        # occasional random shake
        if np.random.rand() < 0.01:
            a = _mutate(a, target_sum, sigma=0.01 * target_sum / len(a))
        # decay step
        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1
    return best_seq, best_score

  which are close to the current maximum (ε‑neighbourhood).  The
  direction is mean‑zero, a momentum term is added and the step size
  is adapted to the magnitude of the maximal convolution value.
* After each update the sequence is forced to be symmetric
  (a[i] = a[n‑1‑i]); symmetry is known to help lowering the
  autocorrelation peak.
* An optional LP‑refinement (via cvxpy) implements the “LP‑update’’ from
  the paper sketch.
* A global search loop repeatedly tries different lengths, uses a
  resize‑operator to reuse the current best, and performs occasional
  up‑scaling to longer sequences.

The routine runs for at most ``budget_s`` seconds and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project a vector onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the (non‑negative) simplex of radius *s*."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small mutation (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of *a* (add Gaussian noise then project)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Compute max(conv) and a gradient that aggregates all entries within
# a relative ε‑neighbourhood of that maximum.
# ----------------------------------------------------------------------
def _max_conv_and_grad_multi(a: np.ndarray, eps_frac: float = 0.01):
    """Return (max_val, grad) where grad is the averaged sub‑gradient."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    # indices whose value is within eps_frac of the maximum
    near_idxs = np.where(conv >= max_val * (1 - eps_frac))[0]
    grad = np.zeros_like(a)
    for k in near_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if near_idxs.size > 0:
        grad /= near_idxs.size
    return max_val, grad


# ----------------------------------------------------------------------
# Internal (fast) evaluation – identical to evaluate_sequence for admissible inputs
# ----------------------------------------------------------------------
def _eval(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a ** 2))


# ----------------------------------------------------------------------
# Local descent (multi‑index gradient, momentum, symmetry)
# ----------------------------------------------------------------------
def _local_descent_multi(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.1,
    eps_start: float = 0.02,
    eps_end: float = 0.001,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that reduces *all* near‑maximal convolution entries.
    Returns the best sequence seen and its (internal) evaluation value.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    # enforce symmetry from the outset
    a = (a + a[::-1]) / 2.0
    a *= target_sum / a.sum()
    best_seq = a.copy()
    best_score = _eval(a)

    start = time.time()
    it = 0
    step = base_step
    momentum = np.zeros_like(a)

    while time.time() < start + time_limit:
        elapsed = time.time() - start
        eps = eps_start + (eps_end - eps_start) * (elapsed / time_limit)

        max_val, grad = _max_conv_and_grad_multi(a, eps_frac=eps)

        # zero‑mean direction (keeps the sum constant)
        d = -grad + np.mean(grad)

        # momentum update
        beta = 0.9
        momentum = beta * momentum + (1.0 - beta) * d

        # adapt step size to the magnitude of the direction
        max_abs = np.max(np.abs(momentum))
        if max_abs == 0:
            max_abs = 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * momentum
        a_new = _project_onto_simplex(a_new, target_sum)

        # enforce symmetry and renormalise (symmetrisation may disturb the sum)
        a_new = (a_new + a_new[::-1]) / 2.0
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)

        score = _eval(a_new)

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})")

        # always move forward – occasional random shake‑up
        if np.random.rand() < 0.01:
            sigma = 0.02 * target_sum / len(a)
            a_new = _mutate(a_new, target_sum, sigma)

        a = a_new

        # simple step decay
        if it % 300 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (if cvxpy is installed)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j  s.t. g≥0 and f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """Perform a few LP‑updates starting from *init*."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    best_seq = a.copy()
    best_score = _eval(a)

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        M = np.convolve(a, a).max()
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or g0.sum() == 0:
            break
        g = g0 * (target_sum / g0.sum())
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)
        score = _eval(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Resize a known good sequence to a new length (linear interpolation)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize *seq* to length *new_n* preserving shape."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initialise global best (try historic variable, otherwise uniform)
    # ------------------------------------------------------------------
    global height_sequence_1
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = _eval(cand)
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Start from historic seq, score={sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    if best_seq is None:
        n0 = 1000
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = _eval(best_seq)
        print(f"[{time.time():.2f}] No historic – using uniform, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to run one optimisation round
    # ------------------------------------------------------------------
    def _run_one(n: int, init_vec: np.ndarray, remaining: float):
        target = math.sqrt(2.0 * n)

        # allocate a slice of the remaining time
        run_time = min(remaining, 12.0)

        # first a multi‑index gradient descent
        cand, sc = _local_descent_multi(
            init_vec,
            target,
            run_time,
            base_step=0.07,
            eps_start=0.02,
            eps_end=0.001,
            print_tag=f"n={n}",
        )

        # optional LP refinement (cheap for moderate n)
        if _CVX_AVAILABLE and n <= 600:
            cand2, sc2 = _lp_refine(cand, target, time_limit=5.0, mix=0.15, max_iter=4)
            if sc2 < sc:
                cand, sc = cand2, sc2

        return cand, sc

    # ------------------------------------------------------------------
    # Main global search loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        time_left = deadline - time.time()
        if time_left < 5.0:
            break

        # Choose a length – usually stay near the current best length
        if best_seq is not None and np.random.rand() < 0.6:
            n = best_seq.size
            if np.random.rand() < 0.3:
                n = int(n * np.random.uniform(0.8, 1.2))
                n = max(200, min(4000, n))
        else:
            n = np.random.randint(200, 2500)

        target_sum = math.sqrt(2.0 * n)

        # Initialise candidate – resize the current best or start random
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        # occasional jitter to increase diversity
        if np.random.rand() < 0.15:
            sigma = 0.03 * target_sum / n
            init = _mutate(init, target_sum, sigma)

        cand_seq, cand_score = _run_one(n, init, time_left)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:
                break

        # Occasionally try a larger up‑scaled version of the current best
        if best_seq is not None and np.random.rand() < 0.05:
            n_up = int(best_seq.size * 1.5)
            n_up = max(200, min(4000, n_up))
            target_up = math.sqrt(2.0 * n_up)
            init_up = _resize_sequence(best_seq, n_up)
            # short dedicated run for the larger problem
            up_time = min(15.0, deadline - time.time())
            cand_up, up_score = _local_descent_multi(
                init_up,
                target_up,
                up_time,
                base_step=0.05,
                eps_start=0.02,
                eps_end=0.001,
                print_tag=f"up{n_up}",
            )
            if up_score < best_score:
                best_score = up_score
                best_seq = cand_up.copy()
                print(
                    f"[{time.time():.2f}] *** UP‑SCALE NEW BEST = {best_score:.6f} (n={n_up})"
                )
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
