import time
import numpy as np

def _full_convolution(a, size=None):
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    return conv

def _softmax_weights(conv, lam):
    m = conv.max()
    # scaled for numeric stability
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w = w / w_sum
    return w

"""
Search for a non‑negative step function minimizing

    F(a) = 2·n·max_k (a * a)[k] / (∑ a_i)²

using an Adam optimiser on a smooth‑max surrogate.
The variables are kept in soft‑max (log‑space) so
the probability simplex is satisfied automatically.
Multiple random restarts, a λ‑continuation scheme and occasional
“shakes’’ help to escape shallow local minima.  The routine
honours a 1000 s time budget and finally returns the best sequence

import time

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights  w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    # exponent is non‑positive → no overflow, only possible underflow
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety, should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the original
    probabilities ``a``.  The formula is  ∇φ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

    # sum_a is guaranteed > 0 by construction
