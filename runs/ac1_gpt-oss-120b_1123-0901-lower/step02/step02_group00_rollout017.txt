"""
Optimise non‑negative step‑function heights to minimise `evaluate_sequence`.

Algorithm
---------
1. Load the existing construction (`height_sequence_1`) if present,
   otherwise start from a uniform vector.
2. Run a short‑run Adam optimiser (sub‑gradient of the max‑convolution)
   on a few lengths (original length and a couple of interpolations).
   The best sequence found so far is kept globally.
3. Starting from the best sequence, perform a Frank–Wolfe style refinement:
   * For the current sequence `a` compute M = max(a·a).
   * Build the sparse convolution matrix `C(a)` such that (C(a)·b)_k = (a*b)_k.
   * Solve the linear programme
         maximise   Σ b_i
         subject to  C(a)·b ≤ M·1 ,  b ≥ 0 .
   * If the LP yields a larger total mass than the target ‑√(2n),
     rescale the solution to the target sum and move a fraction `t`
     towards it (convex combination).  Reduce `t` gradually.
   * Repeat while time permits (≈ 10 – 30 LP iterations).
4. Return the best vector discovered (as a plain Python list).

Both stages respect the global 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (M, index_of_max, full_convolution) using FFT."""
    n = a.shape[0]
    # next power of two for efficient FFT, at least 2*n-1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """Average reversed vector a_{k-p} over all active shift indices."""
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active_idxs.size)
    return rev


# ----------------------------------------------------------------------
def _optimize_on_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
):
    """
    Adam optimiser on length `n` starting from `init_a`.
    Updates best_seq / best_score when a better M is found.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    # enforce exact normalisation
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    while time.time() < deadline:
        # 1) convolution, max, active set
        M, k_max, conv_full = _conv_max_fft(a)

        # active set (indices within tolerance of max)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        # 2) sub‑gradient of the objective (which equals M under our normalisation)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # 3) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4) projection onto feasible set (non‑negative + normalisation)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # 5) occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # 6) exact objective value (equals M when normalised)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[ Adam len {n:5d} iter {iteration:7d}] new best M = {best_score:.6f}"
            )

        # optional early stop if the improvement per iteration is tiny
        if iteration % 100000 == 0 and time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray, tol: float = 1e-14):
    """
    Build sparse matrix C where (C @ b) == conv(f, b) (full convolution).
    Rows correspond to k = 0 .. 2n-2, columns to b_j.
    Zero entries (where f_i ≈ 0) are omitted to keep the matrix sparse.
    """
    n = f.shape[0]
    m = 2 * n - 1
    offsets = []
    data = []
    for i, val in enumerate(f):
        if val > tol:
            offsets.append(-i)  # column shift = -i
            data.append(np.full(m - i, val, dtype=np.float64))
    if not offsets:
        # all entries essentially zero – return an explicit zero matrix
        return sp.csr_matrix((m, n), dtype=np.float64)
    # construct in CSC (good for column slicing inside linprog)
    return sp.diags(data, offsets, shape=(m, n), format="csc")

def _lp_refine(a: np.ndarray, deadline: float, max_iter: int = 25):
    """
    Frank–Wolfe style refinement using the linear programme
        maximise Σ b_i   s.t.   (a * b)_k ≤ M   ∀k,   b ≥ 0 .
    The solution b is rescaled to the target sum and blended with the
    current vector `a`.  The procedure is repeated while time permits.
    Returns the best vector found and its max‑convolution value.
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best_vec = a.copy()
    best_M = _conv_max_fft(a)[0]
    step = 0.5  # initial convex‑combination weight

    while time.time() < deadline and step > 1e-5:
        # current bound
        M, _, _ = _conv_max_fft(a)

        # Build convolution matrix C(a)
        C = _build_conv_matrix(a)

        # Linear programme: maximise Σ b_i  ⇔ minimise -Σ b_i
        c = -np.ones(n, dtype=np.float64)  # objective
        ub = np.full(2 * n - 1, M, dtype=np.float64)  # RHS
        bounds = [(0.0, None)] * n

        # Solve with HiGHS – it accepts a sparse A_ub
        res = opt.linprog(
            c,
            A_ub=C,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0, deadline - time.time())},
        )

        if not res.success:
            print(f"[ LP ] iteration {max_iter}: LP failed ({res.message}) – aborting")
            break

        b = res.x
        sum_b = b.sum()

        # If we cannot increase the mass, the bound cannot be improved.
        if sum_b <= target_sum * (1.0 + 1e-12):
            # no slack left – stop
            break

        # Rescale the LP solution to the required total mass
        g = b * (target_sum / sum_b)

        # Candidate new vector (convex combination)
        cand = (1.0 - step) * a + step * g

        # Re‑normalise numerically (tiny drift may appear)
        cand = cand * (target_sum / cand.sum())

        cand_M = _conv_max_fft(cand)[0]

        if cand_M < M:
            # Accept the move
            a = cand
            if cand_M < best_M:
                best_M = cand_M
                best_vec = a.copy()
                print(f"[ LP ] new best M = {best_M:.6f} (step {step:.4f})")
        else:
            # No improvement – shrink step size
            step *= 0.7
            continue

        # Decay step size slowly to fine‑tune near the optimum
        step *= 0.95
        max_iter -= 1

    return best_vec, best_M

"""
Optimise a non‑negative step‑function to minimise `evaluate_sequence`.

Algorithm
---------
1. Load a known construction (`height_sequence_1`) if it exists,
   otherwise start from a uniform vector.
2. Run a short Adam optimiser (sub‑gradient of the max‑convolution)
   on several lengths (original length and a few dilated versions).
   Keep the best sequence discovered.
3. Refine the best sequence with a Frank–Wolfe style loop:
   * For the current vector `a` compute M = max(a * a).
   * Build the sparse convolution matrix `C(a)` so that (C·b)_k = (a * b)_k.
   * Solve the linear programme
         maximise Σ b_i   s.t.   C(a)·b ≤ M·1 ,  b ≥ 0 .
   * If the LP yields a larger total mass than the required sum
     √(2 n), rescale the solution to the target sum and move a fraction
     `t` towards it (convex combination).  Reduce `t` gradually.
   * Repeat while time permits (≈ 10–30 LP steps).  A few random
     restarts are tried if time remains.
4. Optionally finish with a brief Adam polishing phase.
5. Return the best vector found (as a plain Python list).

Both phases respect the global 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (max‑convolution value, index of the max, full convolution)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """Average of the reversed vector a_{k‑p} over all active shift indices."""
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active_idxs.size)
    return rev


# ----------------------------------------------------------------------
def _optimize_on_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
):
    """
    Adam optimiser for a vector of length ``n`` starting from ``init_a``.
    Updates ``best_seq`` and ``best_score`` when a better max‑convolution
    value is found.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    while time.time() < deadline:
        # ---- convolution & active set ---------------------------------
        M, _, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        # ---- sub‑gradient of the objective (equals M after normalisation) --
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # ---- Adam update ----------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection (non‑negative + normalisation) -----------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---- occasional jitter -----------------------------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---- exact objective (M) ---------------------------------------
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[ Adam len {n:5d} iter {iteration:7d}] new best M = {best_score:.6f}"
            )

        # small early‑stop guard (optional)
        if iteration % 200000 == 0 and time.time() >= deadline:
            break

    return best_seq, best_score
