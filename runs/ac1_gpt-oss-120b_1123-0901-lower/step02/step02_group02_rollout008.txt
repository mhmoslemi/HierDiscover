"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The value is invariant under scaling, so we fix the ℓ₁‑norm to
√(2 n).  With that normalisation the objective reduces to `max(conv)`.
The algorithm works as follows:

1. **Search space** – many independent restarts with different lengths
   (800 … 5000) and different initial shapes (power‑law, random Dirichlet,
   warm‑starts from the current global best).

2. **Flatten‑convolution gradient descent** – in each iteration we
   compute the full convolution `c = a * a` and the target average
   `μ = (∑a)² / (2 n - 1)` (≈ 1 with our normalisation).  The loss
   `L = Σ (c_k - μ)²` is minimised.  Its gradient is

        ∇L_i = 4 Σ_k (c_k - μ) a_{k-i}

   which is a cross‑correlation and can be evaluated with two FFTs.
   After a gradient step we project back onto the simplex
   `{x_i ≥ 0, Σ x_i = √(2 n)}`.

3. **Exploration** – occasional random jitter and a decaying step size
   (simulated‑annealing style) help escape plateaus.

4. **Fine‑tuning** – a short phase of the original “active‑set” gradient
   descent (optimising the current maximal convolution entries) is run
   after the flattening stage to shave off the final bits.

The whole search runs for at most `budget_s` seconds (default 1000 s) and
returns the best sequence discovered.  All heavy work uses FFTs, so even
vectors with a few thousand entries are processed quickly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – Linear convolution via FFT (full convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a power‑of‑2 FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper 3 – Cross‑correlation via FFT (full mode)
# ----------------------------------------------------------------------
def _corr_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full cross‑correlation (x ⋆ y)[lag] for lag = -(len(y)-1)…len(x)-1.
    Returned array has length len(x)+len(y)-1.
    """
    lx, ly = x.shape[0], y.shape[0]
    size = 1 << ((lx + ly - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    corr = np.fft.irfft(np.conj(fx) * fy, n=size)
    # slice the valid part (might be larger than needed because of zero‑padding)
    needed = lx + ly - 1
    return corr[:needed]


# ----------------------------------------------------------------------
# Helper 4 – Gradient of the flattening loss L = Σ (c - μ)²
# ----------------------------------------------------------------------
def _flatten_gradient(a: np.ndarray, conv: np.ndarray, mu: float) -> np.ndarray:
    """
    Compute ∇L where L = Σ_k (conv_k - μ)².
    Returns a vector of length len(a).
    """
    diff = conv - mu                     # shape (2n-1,)
    # correlation of diff with a (as described in the analysis)
    corr = _corr_fft(diff, a)            # length 3n-2
    offset = len(diff) - 1               # corresponds to lag = 0
    grad = 4.0 * corr[offset: offset + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Helper 5 – Active‑set gradient (max‑convolution) from the reference code
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4):
    """Return (max_value, indices) where conv ≥ max – rel_tol·max."""
    max_val = conv.max()
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


def _active_gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of Σ_{k∈active} conv_k w.r.t. a (same as the reference)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine for a given length n
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    flatten_step: float = 0.25,
    flatten_decay: float = 0.9995,
    active_step: float = 0.10,
    active_decay: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Finds a good non‑negative vector of length `n`.
    Returns the best vector discovered and its max‑convolution value
    (which equals the evaluation score because Σ a_i = √(2 n)).
    """
    target_sum = np.sqrt(2.0 * n)               # normalisation
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best vector seen in this run
    conv = _conv_fft(a)
    best_local = a.copy()
    best_score = conv.max()                     # with our normalisation

    # Phase‑1: flatten the convolution (L₂ loss)
    step = flatten_step
    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) convolution and flattening gradient
        # --------------------------------------------------------------
        conv = _conv_fft(a)
        mu = (a.sum() ** 2) / (2 * n - 1)       # target average (≈1)
        grad = _flatten_gradient(a, conv, mu)

        # --------------------------------------------------------------
        # 2) gradient step + projection
        # --------------------------------------------------------------
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:                   # stagnation → jitter
            a += np.random.randn(n) * 1e-6
        else:
            eta = step * target_sum / (grad_norm + 1e-12)
            a = a - eta * grad
        a = _project_onto_simplex(a, target_sum)

        # --------------------------------------------------------------
        # 3) occasional random shake‑up (helps escape flat regions)
        # --------------------------------------------------------------
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.0008 / n)
            a = _project_onto_simplex(a + noise, target_sum)
            step *= 0.92

        # --------------------------------------------------------------
        # 4) record improvement
        # --------------------------------------------------------------
        conv = _conv_fft(a)
        cur = conv.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            print(f"[{time.time():.2f}] n={n}  flatten‑best max={cur:.6f}")

        step *= flatten_decay

        # --------------------------------------------------------------
        # 5) early switch to active‑set phase once the max is close to μ
        # --------------------------------------------------------------
        if cur <= mu * 1.05:                    # near‑uniform, switch
            break

    # --------------------------------------------------------------
    # Phase‑2: active‑set (max‑focused) gradient descent (few seconds)
    # --------------------------------------------------------------
    step_active = active_step
    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol=5e-4)
        grad = _active_gradient(a, active)

        grad_max = grad.max() if grad.max() > 0 else 1.0
        eta = step_active * max_val / (grad_max + 1e-12)

        a = a - eta * grad
        a = _project_onto_simplex(a, target_sum)

        if np.random.rand() < 0.02:               # tiny jitter
            a = _project_onto_simplex(a + np.random.randn(n) * (target_sum * 0.0005 / n),
                                      target_sum)
            step_active *= 0.93

        conv = _conv_fft(a)
        cur = conv.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            print(f"[{time.time():.2f}] n={n}  active‑best max={cur:.6f}")

        step_active *= active_decay

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for the lowest possible value of the evaluation function.
    The routine respects the time budget and returns the best sequence
    discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2        # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # length range – larger n gives a lower theoretical floor (≈1)
    MIN_N, MAX_N = 800, 5000

    # optional warm‑start from a previously saved construction
    global height_sequence_1
    try:
        warm_start = np.array(height_sequence_1, dtype=float)
    except Exception:
        warm_start = None

    while time.time() < deadline:
        remaining = deadline - time.time()
        # adapt the chosen length towards the end of the budget
        if remaining < 30.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(MIN_N, MAX_N + 1)

        # --------------------------------------------------------------
        # initialise candidate vector
        # --------------------------------------------------------------
        if np.random.rand() < 0.55:
            # power‑law start a_i ∝ (i+1)^{-p} with p≈0.5
            p = np.random.uniform(0.42, 0.60)
            idx = np.arange(1, n + 1, dtype=np.float64)
            init = idx ** (-p)
        else:
            # random Dirichlet‑like start
            init = np.random.rand(n)

        # small deterministic jitter to break symmetry
        init += np.random.rand(n) * 1e-4

        # occasional warm‑start from the best sequence seen so far
        if best_seq is not None and np.random.rand() < 0.4:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        # also allow the external `height_sequence_1` as a seed
        if warm_start is not None and np.random.rand() < 0.2:
            if warm_start.size >= n:
                init = warm_start[:n].copy()
            else:
                pad = np.random.rand(n - warm_start.size) * 1e-4
                init = np.concatenate([warm_start, pad])

        # --------------------------------------------------------------
        # run a time‑slice optimisation for this length
        # --------------------------------------------------------------
        slice_time = 8.0
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_one_length(
            n,
            init,
            run_deadline,
            flatten_step=0.25,
            flatten_decay=0.9994,
            active_step=0.12,
            active_decay=0.9997,
        )

        # --------------------------------------------------------------
        # update global best
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best max_conv={best_score:.6f} (n={n})")

        # early exit if we already beat the target 1.503 (objective == max_conv)
        if best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # fallback – extremely unlikely to be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # final projection to guarantee the ℓ₁‑norm = √(2 n)
    n = best_seq.shape[0]
    best_seq = _project_onto_simplex(best_seq, np.sqrt(2.0 * n))

    # return a plain Python list as required by the evaluator
    return best_seq.tolist()
