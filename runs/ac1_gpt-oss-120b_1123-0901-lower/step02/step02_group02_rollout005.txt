def _conv_fft_general(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1 -1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)[:n+m-1]
    np.maximum(conv, 0.0, out=conv) # maybe keep sign? Since w positive. a_rev non-negative too.
    return conv

"""
Smooth‑max gradient descent with Adam + final pairwise hill‑climbing.

We minimise   2·n·max(conv(a,a)) / (sum(a)**2)
under the constraints a_i ≥ 0 and Σ a_i = √(2n).
Because the normalisation makes the objective equal to max(conv(a,a)),
the problem is to make the full autocorrelation of a as flat as possible.

The algorithm works as follows:

* Initialise a with a power‑law shape a_i ∝ (i+1)^‑p (p≈0.5).
* Optimise a with a smooth approximation of the max operator:
  f_β(a)= (1/β)·log Σ_k exp(β·conv_k).  Its gradient can be written
  as a single convolution, which we evaluate with an FFT (O(n log n)).
* Adam (first‑moment) updates are used, and the step size is decayed.
  The smoothing parameter β is increased from a small value to a large
  one, so the optimisation first moves in a smooth landscape and later
  focuses on the true maximum.
* After the main phase a cheap random‑pair hill‑climbing (move a small
  amount of mass from one coordinate to another) is performed for a
  short time to escape possible plateaus.
* Many independent restarts with different lengths and exponents are
  performed within the time budget; the best vector encountered is
  returned.

All heavy computations use FFT‑based convolutions, so even thousands of
iterations on vectors of a few thousand entries comfortably fit into the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – full convolution using FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a power‑of‑2 FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    # the vectors are non‑negative, but keep the result as is.
    return conv


# ----------------------------------------------------------------------
# Helper 3 – gradient of the smooth‑max approximation
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a ( (1/β)·log Σ_k exp(β·conv_k) ) efficiently.
    The gradient can be written as
        2 * Σ_k w_k * a_{k‑i}
    where w is the soft‑max over conv.
    """
    n = a.shape[0]

    # soft‑max weights (stable)
    max_c = conv.max()
    shifted = beta * (conv - max_c)           # ≤ 0, no overflow
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation w ⋆ a (equivalently convolution w * a_rev)
    a_rev = a[::-1]
    full = _conv_fft_general(w, a_rev)       # length = (2n‑1)+n‑1 = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Helper 4 – local hill‑climbing (pairwise mass transfer)
# ----------------------------------------------------------------------
def _local_refinement(a: np.ndarray, run_deadline: float) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass between two coordinates if it lowers max(conv)."""
    n = a.shape[0]
    best = a.copy()
    best_max = _conv_fft(best).max()
    while time.time() < run_deadline:
        i = np.random.randint(0, n)
        if best[i] <= 1e-12:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move up to 10 % of a[i] from i to j
        delta = best[i] * np.random.rand() * 0.1
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        conv = _conv_fft(cand)
        m = conv.max()
        if m < best_max:
            best = cand
            best_max = m
    return best, best_max


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length `n`
# ----------------------------------------------------------------------
def _optimise_one_length_smooth(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    beta_start: float = 0.5,
    beta_end: float = 200.0,
    base_lr: float = 0.4,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max Adam optimisation for a given length.
    Returns the best vector and its max‑convolution value.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2n) ⇒ objective = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep best seen in this run
    conv = _conv_fft(a)
    best_max = conv.max()
    best_vec = a.copy()

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-12
    it = 0
    max_iters_est = 2000          # used for beta scheduling, not a hard limit

    while time.time() < run_deadline:
        # ------------------------------------------------------------------
        # 1) current convolution and smooth‑max gradient
        # ------------------------------------------------------------------
        conv = _conv_fft(a)
        max_c = conv.max()

        # schedule β linearly (could be replaced by any schedule)
        progress = min(it / max_iters_est, 1.0)
        beta = beta_start + progress * (beta_end - beta_start)

        grad = _gradient_from_conv(a, conv, beta)

        # ------------------------------------------------------------------
        # 2) Adam update
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_onto_simplex(a_new, target_sum)

        # ------------------------------------------------------------------
        # 3) occasional jitter to escape flat plateaus
        # ------------------------------------------------------------------
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 1e-3 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.92

        # ------------------------------------------------------------------
        # 4) evaluate
        # ------------------------------------------------------------------
        conv_new = _conv_fft(a_new)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max = max_new
            best_vec = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new best max={best_max:.6f}")

        a = a_new
        # decay learning rate slowly
        if it % 500 == 0 and it > 0:
            base_lr *= 0.95
        it += 1

    return best_vec, best_max

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function within the given time budget.

    Strategy:
      • Randomly pick a length n (≈ 800 … 4000) and initialise with a
        power‑law shape a_i ∝ (i+1)^‑p (p≈0.5).
      • Optimise the vector with smooth‑max Adam descent, slowly increasing
        the smoothing parameter β.
      • Apply a cheap pairwise‑mass‑transfer hill‑climbing to polish the
        result.
      • Keep the best vector across many independent restarts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = float("inf")  # this will hold max(conv), i.e. the objective

    # admissible range for the length of the step function
    min_n, max_n = 800, 4000

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a handful of seconds.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a short slice for the main optimiser; keep a tiny
        # reserve for the final refinement.
        if remaining < 20.0:
            n = np.random.randint(800, 1500)
            run_time = min(remaining * 0.6, 5.0)
        else:
            n = np.random.randint(min_n, max_n + 1)
            run_time = min(6.0, remaining * 0.4)

        # ----- initialise -------------------------------------------------
        # a_i = (i+1)^‑p  with a small random jitter
        p = np.random.uniform(0.45, 0.55)
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p)
        init += np.random.rand(n) * 1e-4

        # Occasionally warm‑start from the current global best.
        if best_seq is not None and np.random.rand() < 0.3:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        run_deadline = min(deadline, time.time() + run_time)

        # ----- optimisation ------------------------------------------------
        seq_local, max_local = _optimise_one_length_smooth(
            n,
            init,
            run_deadline,
            beta_start=0.5,
            beta_end=200.0,
            base_lr=0.4,
            beta1=0.9,
            beta2=0.999,
        )

        # ----- local refinement -------------------------------------------
        refine_deadline = min(deadline, time.time() + min(1.0, remaining * 0.1))
        seq_refined, max_refined = _local_refinement(seq_local, refine_deadline)

        # Choose the better of the two local outcomes.
        if max_refined < max_local:
            seq_local, max_local = seq_refined, max_refined

        # ----- update global best -----------------------------------------
        if max_local < best_score:
            best_score = max_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # Early exit if we already beat a comfortable target.
        if best_score < 1.45:   # well below the 1.503 benchmark
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe default.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
