def _fft_convolve(x, y):
    m = len(x) + len(y) - 1
    nfft = _next_pow2(m)
    X = np.fft.rfft(x, nfft)
    Y = np.fft.rfft(y, nfft)
    conv = np.fft.irfft(X * Y, nfft)[:m]
    return conv

"""
Propose a non‑negative step function (list of heights) that minimizes the
objective defined by ``evaluate_sequence``.

Algorithm
---------

1. **Initialization**
   * Choose a moderate length (default 2000, can be increased later).
   * Draw a random point on the probability simplex (Dirichlet(1,…,1)).
   * If a previous best sequence is available in the global variable
     ``height_sequence_1``, use it as an additional starting point.

2. **Smooth max‑gradient descent**
   * The objective for a normalized sequence (sum = 1) is
     ``obj(a) = 2·n·max(a*a)`` where ``a*a`` is the full self‑convolution.
   * Instead of the non‑smooth max we use a temperature‑controlled soft‑max
     over the convolution entries:
         ``w_k ∝ exp((conv_k – max(conv))/τ)``.
     A small uniform component (γ) is mixed into the weights to avoid
     stagnation when the distribution becomes almost one‑hot.
   * The gradient of the weighted sum ``∑_k w_k (a*a)_k`` can be written
     as two ordinary convolutions:
         ``∇a = (a * w)[:n] + rev((a * rev(w)))[(len(w)‑1)‑np.arange(n)]``.
   * An Adam optimiser (with learning‑rate adaptation) updates the vector.
   * After each step the vector is projected back onto the probability
     simplex (non‑negative, sum = 1).

3. **Exploration**
   * When a gradient step fails to improve the objective we decay the
     learning‑rate and occasionally perform a tiny random mass‑transfer
     move (still projected onto the simplex).  This helps escape
     shallow local minima.

4. **Multiple restarts**
   * The whole optimisation is executed repeatedly with fresh random
     starts (or the global seed) until the time budget (≈ 1000 s) is
     exhausted.  The best sequence ever seen is kept.

The code uses FFT‑based convolutions (`rfft`/`irfft`) for O(n log n) speed
and is deliberately written without closures or lambdas as required.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = _next_pow2(size)
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(X * Y, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution (length 2·len(seq)‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Local fast replica of ``evaluate_sequence`` (sum of seq is assumed ≠ 0).
    Returns 2·n·max(conv)/sum(seq)².
    """
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best list found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try to seed from a previously known construction
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()
                s = _objective(init)
                if s < best_score:
                    best_score = s
                    best_seq = init.copy()
                    print(f"[Seed] global prev‑best score={s:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter schedule (length of the sequence)
    # We start with n=2000; after a few restarts we may try n=4000 if time permits.
    lengths = [2000, 4000]

    # ------------------------------------------------------------------
    # Main optimisation loop (multiple restarts)
    for n in lengths:
        if time.time() > deadline:
            break

        # number of independent restarts for this length
        restart_idx = 0
        while time.time() < deadline:
            restart_idx += 1
            # -------------------- initialise a random point on simplex --------------------
            a = np.random.exponential(1.0, size=n)
            a /= a.sum()

            # Occasionally start from the global best (if available)
            if best_seq is not None and restart_idx == 1:
                # resize the global best to current n (pad with zeros if needed)
                if len(best_seq) == n:
                    a = best_seq.copy()
                else:
                    # simple interpolation: keep existing values, add zeros at the end
                    a = np.concatenate([best_seq, np.zeros(n - len(best_seq))])
                    a /= a.sum()

            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()

            # -------------------- Adam optimiser state --------------------
            m = np.zeros_like(a)          # first moment
            v = np.zeros_like(a)          # second moment
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8

            tau = 0.12                   # temperature for the soft‑max
            gamma = 0.015                # uniform mixing weight for w

            it = 0
            last_improve_it = 0
            # --------------------------------------------------------------------
            # Iterative optimisation
            while time.time() < deadline:
                it += 1

                # ----- self‑convolution and soft‑max weights -----
                conv = _self_convolve(a)                         # length 2n‑1
                max_c = conv.max()
                # exponential weighting (stable)

                soft = exp_vec / exp_vec.sum()
                # mix a tiny uniform component to keep the gradient non‑degenerate
                w = (1.0 - gamma) * soft + gamma / len(conv) * np.ones_like(soft)

                conv1 = conv_rev_full[idx]

                grad = conv1 + conv2                               # shape (n,)

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** it)

                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_onto_simplex(a_candidate)

                cand_score = _objective(a_candidate)

                improved = False
                if cand_score < cur_score:
                    improved = True
                else:

                    if np.random.rand() < 0.07:

                        delta = 0.01 * min(a[i], 0.1)     # move at most 1 % of total mass
                        if delta > 0.0:

                            a_tmp[i] -= delta
                            a_tmp[j] += delta
                            a_tmp = _project_onto_simplex(a_tmp)
                            tmp_score = _objective(a_tmp)
                            if tmp_score < cur_score:
                                a_candidate = a_tmp
                                cand_score = tmp_score
                                improved = True

                if improved:

                    cur_score = cand_score
                    last_improve_it = it
                    # slowly increase learning rate after success
                    lr = min(lr * 1.03, 1.2)

                        best_seq = a.copy()

                tau = max(tau * 0.9999, 1e-5)

        n = 2000
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
