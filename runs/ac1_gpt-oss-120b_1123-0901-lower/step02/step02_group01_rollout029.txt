"""
Propose a step‑function sequence with a low evaluation score.

The search consists of two phases.

1.  Continuous optimisation.
    We minimise a smooth approximation of the original objective
    using a projected gradient descent on the log‑sum‑exp surrogate
    of the self‑convolution.  The iterates stay on the probability
    simplex (non‑negative, sum = 1).  This phase quickly finds a good
    local minimum (the previous best 1.519… is reproduced).

2.  Linear‑programming refinement (the “LP‑step’’ from the original
    paper).  For the current sequence *f* we compute
        M = max( f * f ).
    Then we solve the LP
        maximise   Σ b_j
        subject to b_j ≥ 0
                   (f * b)_k ≤ M   for all k,
    which finds a direction *b* that can carry more mass without
    breaking the current convolution bound.  After scaling
        g = b / Σ b    (so Σ g = 1)
    we mix the two sequences
        f←(1‑t)·f + t·g
    with a small step‑size *t*.  Because the bound M does not increase,
    the exact objective usually drops.  The step is repeated until no
    further improvement is observed or the time budget expires.

Both phases respect the 1000 s limit, print progress and finally
return the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all numbers non‑positive → uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a, size=None):
    """Linear self‑convolution a*a (length 2·len(a)-1) using FFT."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _objective(a):
    """Exact objective used by evaluate_sequence (scale‑invariant)."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is forced to 1 by our projection, but keep the formula general
    return float(2.0 * n * max_b / (sum_a * sum_a))

        conv = _full_convolution(seq)
        M = conv.max()

        # build the (2n‑1) × n linear convolution matrix:
        # row k, column i contains f_{k‑i} when the index is in range,
        # otherwise 0.  This can be assembled very compactly:
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i : i + n, i] = seq

        # LP: maximise Σ b_j   subject to A·b ≤ M  and  b ≥ 0
        c = -np.ones(n)                               # linprog minimises, so negate
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[LP {lp_iter}] linprog failed: {res.message}")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        # No improvement possible if the LP cannot increase the total mass
        if sum_b <= 1.0001:
            print(f"[LP {lp_iter}] sum(b)≈{sum_b:.6f} ≤ 1 – stop")
            break

        # scale to the simplex (sum = 1)
        g = b_opt / sum_b

        # try a mixing step; start with a moderate weight
        t = 0.20
        improved = False
        while t > 1e-4:
            new_seq = (1.0 - t) * seq + t * g
            new_seq = _project_onto_simplex(new_seq)
            new_val = _objective(new_seq)
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                seq = new_seq
                improved = True
                print(f"[LP {lp_iter}] mixed t={t:.4f} → obj={new_val:.6f}")
                break
            t *= 0.5                     # halve the step if no improvement

        if not improved:
            print(f"[LP {lp_iter}] no improving mix – stop")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
```
