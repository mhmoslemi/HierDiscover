"""
Improved search for a step‑function minimizing
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines several ideas that differ from the previous
implementation:

* **Soft‑max smoothing of the max‑convolution** – we minimise a
  differentiable surrogate
      f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
  whose gradient can be written as a simple
  correlation of the current sequence with the soft‑max weights.
  By gradually increasing α we move from a smooth average to the true
  L∞ objective, helping to escape shallow local minima.

* **Projected gradient descent** on this smooth surrogate with a
  simplex projection ( Σa = √(2·n) ) after each step.

* **Multi‑scale exploration** – the current best shape is resized to
  different lengths (linear interpolation) and refined again.  This
  re‑uses useful structure while allowing higher resolution.

* **Optional LP‑refinement** (as in the paper) when the problem size is
  modest and cvxpy is available.

* **Light‑weight evolutionary mutations** for occasional global jumps.

All helpers are defined at the top level; the entry point is
`propose_candidate(seed=…, budget_s=1000.0)`.  The routine respects the
time budget and prints progress; the final returned list is guaranteed
to consist of non‑negative floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Very small mutation helper (adds Gaussian noise and re‑projects)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Local projected‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """Return (max_val, grad, max_idx, full_conv) for the current max entry."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient of conv[max_idx] w.r.t. a (sum of overlapping terms)
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.05,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """Fast projected sub‑gradient descent on the true max‑convolution."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()
    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # make the direction sum‑zero before projection
        mean_g = np.mean(grad)
        d = -grad + mean_g
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})")

        # occasional random shake‑up
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
            step *= 0.9
        else:
            a = a_new

        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (uses cvxpy if it can be imported)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve max Σ g_j s.t. g≥0 and f*g ≤ M (linear in g)."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """Perform a few LP‑updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break
        g = g0 * (target_sum / np.sum(g0))
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple resize (interpolation / down‑sampling) to reuse a known good sequence
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Smoothed gradient descent using a soft‑max surrogate for max(conv)
# ----------------------------------------------------------------------
def _smooth_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.2,
    alpha_start: float = 0.2,
    alpha_end: float = 20.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    with a geometric schedule α ∈ [alpha_start, alpha_end].
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t0 = time.time()
    it = 0
    while time.time() < t0 + time_limit:
        elapsed = time.time() - t0
        progress = elapsed / time_limit
        # geometric progression of alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        conv = np.convolve(a, a)
        max_c = conv.max()  # for numerical stability in the softmax
        # soft‑max weights (sum to 1)
        exp_term = np.exp(alpha * (conv - max_c))
        w = exp_term / exp_term.sum()

        # gradient = 2 * correlation(a, w)
        corr = np.correlate(a, w, mode="full")
        offset = len(w) - 1
        grad = 2.0 * corr[offset : offset + a.size]

        # simple normalisation of step length
        max_abs = np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else 1.0
        eta = base_step / max_abs

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] smooth descent α={alpha:.2f} new best = {score:.6f} (n={len(a)})"
            )

        a = a_new
        # modest step decay
        if it % 300 == 0 and it > 0:
            base_step *= 0.9
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function with minimal evaluation score.
    The routine respects the given time budget (seconds) and returns a list
    of floats.  It combines smooth gradient descent, hard max‑gradient
    descent, optional LP refinement and occasional evolutionary mutations.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Global best holder (seed from historic solution if present)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Started from historic seq, score={sc:.6f}")
        except Exception:
            pass

    if best_seq is None:
        # fallback – a uniform sequence of moderate length
        n0 = 800
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic seq – using uniform, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to run a single optimisation attempt for a given length
    # ------------------------------------------------------------------
    def _run_one(n: int, init_vec: np.ndarray, remaining: float):
        target = math.sqrt(2.0 * n)

        # select method probabilistically
        choice = np.random.rand()
        # cheap LP updates only for modest n
        if _CVX_AVAILABLE and n <= 600 and choice < 0.12:
            run_time = min(remaining, 12.0)
            cand, sc = _lp_refine(init_vec, target, run_time)
            return cand, sc

        if choice < 0.55:
            # smooth descent – our main new ingredient
            run_time = min(remaining, 22.0)
            cand, sc = _smooth_descent(
                init_vec, target, run_time, base_step=0.25, alpha_start=0.2, alpha_end=15.0
            )
            return cand, sc
        elif choice < 0.85:
            # hard max‑gradient descent (fast local refinement)
            run_time = min(remaining, 14.0)
            cand, sc = _local_descent(
                init_vec, target, run_time, base_step=0.07, print_tag=f"n={n}"
            )
            return cand, sc
        else:
            # tiny evolutionary step: mutate then short smooth descent
            sigma = 0.03 * target / n
            mutated = _mutate(init_vec, target, sigma)
            run_time = min(remaining, 10.0)
            cand, sc = _smooth_descent(
                mutated,
                target,
                run_time,
                base_step=0.2,
                alpha_start=0.1,
                alpha_end=8.0,
            )
            return cand, sc

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # Choose a length – favour current best but allow jumps
        if best_seq is not None and np.random.rand() < 0.65:
            n = best_seq.size
            if np.random.rand() < 0.35:
                n = int(n * np.random.uniform(0.85, 1.15))
                n = max(200, min(3000, n))
        else:
            n = np.random.randint(200, 2800)

        target_sum = math.sqrt(2.0 * n)

        # Initial candidate: resized best or random
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        time_left = deadline - time.time()
        if time_left < 8.0:
            break
        # Give each run a slice of the remaining budget (up to ~30 s)
        slice_time = min(time_left * 0.2, 30.0)

        cand_seq, cand_score = _run_one(n, init, slice_time)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final polish: a short hard max‑gradient descent on the best shape
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            target = math.sqrt(2.0 * best_seq.size)
            cand_seq, cand_score = _local_descent(
                best_seq,
                target,
                remaining,
                base_step=0.05,
                print_tag="final polish",
            )
            if cand_seq is not None and cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(f"[{time.time():.2f}] Final polish improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
