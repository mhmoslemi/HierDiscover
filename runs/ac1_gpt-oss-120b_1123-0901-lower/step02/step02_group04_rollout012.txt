assistantfinal**Solution Overview**

The evaluation function is  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

with a non‑negative step‑function `a`.  
The score is *scale invariant* – multiplying `a` by any constant does not change the value – therefore we can always project a candidate onto the simplex  

\[
\{a\;|\;a_i\ge0,\;\sum_i a_i = 1\}
\]

without affecting the objective.

The optimisation problem is convex (maximum of a family of convex quadratics).  
A fast sub‑gradient method works well, but the naïve version that uses only the single maximal index can stall.  
We improve it by **smoothing the max** with a soft‑max (log‑sum‑exp) and taking the corresponding gradient, which is a weighted sum of all convolution entries.  
The gradient can be obtained with a single convolution, so each iteration costs only `O(n log n)`.

The complete search consists of three stages:

1. **Parametric families** – cheap sweeps over exponentials, linear/quadratic decays, triangles, sines, …  
   every candidate is projected onto the simplex and evaluated.

2. **Smoothed sub‑gradient descent** – many random restarts (different lengths, Dirichlet‑drawn starts, exponential starts).  
   The descent uses momentum, a diminishing step size and an increasing smoothing parameter `τ` which gradually turns the soft‑max into the true max.

3. **Linear‑programming refinement** (if SciPy is available) – the classic SOTA LP step that maximises the sum of a new vector `g` under the convolution bound.  
   The result is rescaled to the simplex and blended with the current best vector.

All helper functions are defined at module level and avoid closures, as required.

```python
"""
Propose a step‑function (non‑negative heights) minimising

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

1.  Scan a few parametric families (exponential, linear, quadratic,
    triangular, sinusoid) for several lengths.  Each candidate is
    projected onto the simplex Σa_i = 1 and evaluated; the best so far
    is kept.

2.  Perform many smoothed sub‑gradient descents with momentum.
    The descent uses a soft‑max weight vector `w = softmax(τ·conv)`,
    the gradient of `∑ w_k (a*a)_k` being `2 * (a ⋆ w)` (cross‑correlation).
    `τ` starts small (smooth) and is increased each iteration, turning
    the direction into the true sub‑gradient of the max.
    Random restarts with different lengths and random initial vectors
    (Dirichlet, exponential, piecewise‑linear) explore the space.

3.  Optional LP refinement (SciPy highs).  The LP maximises Σg_j
    under the convolution constraint `(a * g)_k ≤ M` where
    `M = max(a*a)`.  The optimum `g` is rescaled to Σg=1 and blended
    with `a` (mixing factor 0.2).  This frequently yields a tiny
    improvement.

The routine respects the given time budget (default 1000 s) and
returns the best list of floats discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of vector `v` onto the simplex of sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT for speed)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b (length len(a)+len(b)-1) via FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n.bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
# Smoothed gradient of the soft‑max of the convolution
# ----------------------------------------------------------------------
def _smoothed_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k w_k (a * a)[k]   (Σ w_k = 1, w_k ≥ 0).

    The gradient is   2 * (a ⋆ w)   where “⋆” denotes cross‑correlation.
    It can be computed by a single convolution with the reversed weight
    vector.
    """
    # reverse the weight vector
    w_rev = w[::-1]
    # full convolution a * w_rev (length = 3·n‑2)
    conv_full = np.convolve(a, w_rev)
    # offset = len(w_rev)‑1 = 2·n‑2
    offset = len(w_rev) - 1
    # take the slice that corresponds to correlation indices 0…n‑1
    g = conv_full[offset - np.arange(a.shape[0])]
    return 2.0 * g


# ----------------------------------------------------------------------
# Wrapper for the external evaluation routine
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Call the provided evaluate_sequence, return +inf on any error."""
    try:
        return evaluate_sequence(seq)
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Update the global best (no closures – just return new values)
# ----------------------------------------------------------------------
def _maybe_update(best_seq, best_score, candidate):
    """Return possibly updated (best_seq, best_score) after evaluating."""
    score = _evaluate(candidate)
    if score < best_score:
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(candidate)})")
        return candidate.copy(), score
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed sub‑gradient descent with momentum
# ----------------------------------------------------------------------
def _run_subgrad(init_vec: np.ndarray,
                 target_sum: float,
                 best_seq,
                 best_score,
                 deadline: float):
    """
    Perform many iterations of smoothed sub‑gradient descent starting
    from `init_vec` (already on the simplex Σ=target_sum).  Returns the
    possibly improved (best_seq, best_score).
    """
    a = init_vec.astype(float)
    n = a.shape[0]

    # momentum tracker
    m = np.zeros_like(a)
    beta = 0.9          # momentum decay
    base_step = 0.5

    # smoothing parameter τ – starts gentle, grows slowly
    tau = 0.3
    tau_inc = 0.0004

    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)                 # full autocorrelation
        max_val = conv.max()

        # ------------------------------------------------------------------
        # Update best from time to time (every few iterations)
        # ------------------------------------------------------------------
        if it % 5 == 0:
            cand = a.tolist()
            best_seq, best_score = _maybe_update(best_seq, best_score, cand)
            if best_score < 1.503:           # target already reached
                return best_seq, best_score

        # ------------------------------------------------------------------
        # Soft‑max weights over the convolution entries
        # ------------------------------------------------------------------
        # subtract max_val for numerical stability
        w = np.exp(tau * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ------------------------------------------------------------------
        # Gradient of the smooth surrogate (2·a⋆w)
        # ------------------------------------------------------------------
        grad = _smoothed_gradient(a, w)

        # sub‑gradient direction (we minimise the max)
        direction = -grad
        m = beta * m + (1.0 - beta) * direction

        # diminishing step size
        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        # projected update
        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new
        it += 1
        tau += tau_inc

    # final best update before leaving the loop
    best_seq, best_score = _maybe_update(best_seq, best_score, a.tolist())
    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               best_seq,
               best_score,
               deadline: float):
    """Perform a few LP refinement steps; return possibly improved (best_seq, best_score)."""
    try:
        from scipy.optimize import linprog
    except Exception:
        return best_seq, best_score

    n = a.shape[0]
    target_sum = a.sum()                # invariant -> should be 1.0 but keep general
    M = _conv_self(a).max()

    # Build convolution matrix A where (a * g)_k = Σ_i A[k,i]·g_i
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        cols = np.arange(lo, hi + 1, dtype=int)
        A[k, cols] = a[k - cols]

    # LP loop – we stop after a few successful refinements or when time runs out
    while time.time() < deadline - 0.5:
        res = linprog(
            c=-np.ones(n),               # maximise Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        # Rescale to the simplex Σ = target_sum
        g = g * (target_sum / s)

        # Convex combination – moderate mixing
        t = 0.2
        a = (1.0 - t) * a + t * g

        # Update best
        best_seq, best_score = _maybe_update(best_seq, best_score, a.tolist())
        if best_score < 1.503:
            break

        # recompute M for the next LP round
        M = _conv_self(a).max()

    return best_seq, best_score
```
