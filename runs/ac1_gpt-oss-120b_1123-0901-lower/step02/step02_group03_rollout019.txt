def smooth_max_and_gradient(a, alpha):
    conv = _conv_full_fft(a)
    max_val = conv.max()
    # compute softmax weights for convex combination
    # shift conv by subtract max to avoid overflow
    exp_vals = np.exp(alpha * (conv - max_val))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    # compute weighted gradient
    n = a.shape[0]
    # compute correlation using FFT
    # using real FFT for speed
    # Compute rfft of a and w (zero-padded to size n_fft)
    n_fft = 1 << ((3 * n - 2).bit_length())  # length enough for correlation length 3n-2
    A = np.fft.rfft(a, n=n_fft)
    W = np.fft.rfft(w, n=n_fft)
    # correlation a * w: irfft(A * conj(W))
    corr = np.fft.irfft(A * np.conj(W), n=n_fft)
    # gradient: 2 * corr[:n] (first n elements correspond to lag j>=0)
    grad = 2.0 * corr[:n]
    return conv, max_val, grad

def run_optimization(n, init_a, budget_run_seconds):
    target_sum = sqrt(2*n)
    a = project_onto_simplex(init_a, target_sum)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # initial learning rate
    alpha = 5.0  # smoothing parameter
    iteration = 0
    while time.time() < run_deadline:
        # compute conv
        n_fft = 1 << ((3 * n - 2).bit_length())
        A = np.fft.rfft(a, n=n_fft)
        conv = np.fft.irfft(A * A, n=n_fft)[:2*n-1]
        max_val = conv.max()
        # update best if cur_score < best
        # compute softmax gradient
        exp_vals = np.exp(alpha * (conv - max_val))
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp
        # Compute w FFT
        W = np.fft.rfft(w, n=n_fft)
        # correlation gradient
        corr = np.fft.irfft(A * np.conj(W), n=n_fft)[:n]
        grad = 2.0 * corr
        # adam update (descend)
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # enforce nonnegativity and simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)
        # occasional random perturb

            a_new = project_onto_simplex(np.maximum(a_new + eps_noise, 0.0), target_sum)

def _smooth_max_and_gradient(a, alpha):
    n = a.shape[0]
    n_fft = 1 << ((3 * n - 2).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    max_val = conv.max()
    # compute softmax weights
    # shift for stability
    exp_vals = np.exp(alpha * (conv - max_val))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    # compute gradient via correlation
    W = np.fft.rfft(w, n=n_fft)
    corr = np.fft.irfft(A * np.conj(W), n=n_fft)[: n]
    grad = 2.0 * corr
    return conv, max_val, grad

"""
Propose a step‑function (non‑negative heights) that minimizes the maximal
self‑convolution value

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm uses many independent restarts, each run applying Adam on a
smooth approximation of the “max” (soft‑max) of the full convolution.
The smoothness parameter α is increased gradually, which lets the optimiser
explore the landscape early on and later focus on the true maximal entry.
Gradients are obtained analytically via FFT‑based correlation
(gradient_j = 2 * Σ_k w_k a_{k‑j}), so each iteration costs only a few FFTs.
Occasional random perturbations and a final refinement stage on the best
candidate improve the chance of beating the target bound 1.503 within the
1000‑second budget.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Global storage for the best solution found so far.
# --------------------------------------------------------------------------- #
best_seq_global = None          # numpy array, the best sequence
best_score_global = float("inf")  # objective value (max convolution entry)

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------------------- #
# Fast full linear convolution via FFT (real‑valued).
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using a power‑of‑two real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv

# --------------------------------------------------------------------------- #
# Smooth‑max of a convolution together with its gradient (Adam‑compatible).
# --------------------------------------------------------------------------- #
def _smooth_max_and_gradient(a: np.ndarray, alpha: float) -> tuple[np.ndarray,
                                                                  float,
                                                                  np.ndarray]:
    """
    Returns (conv, max_val, grad) where
        conv    – full convolution a*a,
        max_val – max(conv),
        grad    – ∇_a  (soft‑max_α(conv))  =  2 * corr(a, w)
    with w_i = exp(α (conv_i - max_val)) / Σ_j exp(α (conv_j - max_val)).
    """
    n = a.shape[0]
    # Size large enough for both convolution (2n‑1) and correlation (3n‑2)
    n_fft = 1 << ((3 * n - 2).bit_length())

    # FFT of a
    A = np.fft.rfft(a, n=n_fft)

    # ----- convolution a*a -------------------------------------------------
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    max_val = conv.max()

    # ----- soft‑max weights -------------------------------------------------
    # Stabilise the exponentials by subtracting max_val.
    exp_vals = np.exp(alpha * (conv - max_val))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp                     # shape (2n‑1,)

    # ----- gradient = 2 * correlation(a, w) -------------------------------
    W = np.fft.rfft(w, n=n_fft)
    corr = np.fft.irfft(A * np.conj(W), n=n_fft)[: n]   # first n lags
    grad = 2.0 * corr

    return conv, max_val, grad

# --------------------------------------------------------------------------- #
def _run_one_instance(n: int, init_vec: np.ndarray, run_deadline: float) -> None:
    """
    Optimises a sequence of length ``n`` starting from ``init_vec``.
    Updates the global best_seq_global / best_score_global when a better
    candidate is discovered.
    """
    global best_seq_global, best_score_global

    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2n)  (scale‑invariant)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    # ---- Adam hyper‑parameters --------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr = 0.05                # base learning rate
    alpha = 5.0              # start with a smooth max
    it = 0

    while time.time() < run_deadline:
        # ----- evaluate smooth max and its gradient -------------------------
        conv, max_val, grad = _smooth_max_and_gradient(a, alpha)

        # ----- keep global best --------------------------------------------
        if max_val < best_score_global:

            best_seq_global = a.copy()
            print(f"[{time.time():.2f}] New best (n={n}) = {best_score_global:.6f}")
