"""
Propose a non‑negative step‑function (a probability vector) that tries to
beat the previous best bound ≈ 1.5449.  The search consists of three stages:

1. **Broad parametric sweep** – a few hundred candidates from simple
   families (geometric, power‑law, Beta‑like, random Dirichlet with different
   concentration parameters and a few custom “mirror” shapes) are generated
   and the best one (lowest objective) is kept.

2. **Projected Adam‑style optimisation** – starting from the best seed,
   we minimise  2·n·max(conv(seq,seq))/sum(seq)²  by gradient descent on the
   simplex.  At each iteration the active shifts (those attaining the
   maximal convolution value, up to a small tolerance) are collected and a
   sub‑gradient is built from them.  An Adam‑like momentum update is taken,
   then the vector is projected back onto the probability simplex
   (non‑negative, unit sum).  If no improvement is obtained after a few
   back‑tracking reductions of the step size, a tiny random mass‑transfer
   move is tried to escape flat regions.

3. **Return the best sequence found before the time budget expires**.
   All heavy work is done with an FFT based convolution (O(n log n)) and
   strict time‑checks, so the routine comfortably fits into the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (see https://arxiv.org/abs/1101.6081)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself (length 2·len(seq)‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (fast local version)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# ------------- parametric families ------------------------------------
def _geom_seq(n: int, r: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r**i (r in (0,1])."""
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p} (p may be negative)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.maximum(seq, 0.0)
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like a_i ∝ (i+1)^{a-1} (n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq = np.maximum(seq, 0.0)
    return seq / seq.sum()


def _random_simplex(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet(α,…,α) draw (α controls sparsity)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _mirror_seq(base: np.ndarray) -> np.ndarray:
    """Create a symmetric (palindromic) sequence of even length."""
    half = base
    full = np.concatenate([half, half[::-1]])
    return full / full.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function that minimises evaluate_sequence().
    Returns the best list found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a moderate length – 2000 gives a good trade‑off.
    n = 2000

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Global seed (if present)
    if "height_sequence_1" in globals():
        try:
            g = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            g = np.clip(g, 0.0, 1000.0)
            if g.sum() > 0.0:
                g = g / g.sum()
                sc = _objective(g)
                if sc < best_score:
                    best_score = sc
                    best_seq = g.copy()
                    print(f"[Init] global seed score={sc:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Broad parametric sweep (early exit when time runs out)
    # --------------------------------------------------------------
    # geometric (r in (0,1])
    for r in np.logspace(np.log10(0.02), np.log10(0.999), 30):
        if time.time() > deadline:
            break
        seq = _geom_seq(n, r)
        if seq is None:
            continue
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[Init] geom r={r:.4f}  score={sc:.6f}")

    # power‑law (p from -1.5 to 5)
    for p in np.linspace(-1.5, 5.0, 30):
        if time.time() > deadline:
            break
        seq = _powerlaw_seq(n, p)
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[Init] power p={p:.3f}  score={sc:.6f}")

    # beta‑like (a,b in [0.05,5] – log‑spaced)
    a_vals = np.logspace(np.log10(0.05), np.log10(5.0), 15)
    b_vals = np.logspace(np.log10(0.05), np.log10(5.0), 15)
    for a in a_vals:
        for b in b_vals:
            if time.time() > deadline:
                break
            seq = _beta_seq(n, a, b)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[Init] beta a={a:.3f} b={b:.3f}  score={sc:.6f}")

    # random Dirichlet draws with various concentrations
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for alpha in alphas:
        for _ in range(30):
            if time.time() > deadline:
                break
            seq = _random_simplex(n, alpha)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[Init] Dirichlet α={alpha:.2f}  score={sc:.6f}")

    # simple mirror construction (take a short random vector and mirror it)
    for _ in range(20):
        if time.time() > deadline:
            break
        half = _random_simplex(n // 2, 0.5)
        seq = _mirror_seq(half)
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[Init] mirror  score={sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2. Local optimisation – projected Adam‑style gradient descent
    seq = best_seq.copy()
    cur_score = best_score

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    iter_cnt = 0
    step_base = 0.8      # initial step magnitude
    tolerance_active = 1e-9   # how close a shift must be to the max to be active

    while time.time() < deadline:
        # --------------------------------------------------------------
        # a) convolution and active shifts
        conv = _self_convolve(seq)
        max_val = conv.max()
        # active shifts: those whose value is within a tiny additive tolerance
        # (or relative tolerance to avoid floating‑point noise)
        act_mask = conv >= (max_val - tolerance_active)
        active_shifts = np.nonzero(act_mask)[0]

        # b) sub‑gradient (sum of contributions of all active shifts)
        n_seq = seq.shape[0]
        grad = np.zeros_like(seq)
        for k in active_shifts:
            lo = max(0, k - (n_seq - 1))
            hi = min(k, n_seq - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += seq[k - idx]   # g_i = a_{k-i} for each active k

        # c) Adam update (direction = -grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter_cnt + 1))
        v_hat = v / (1.0 - beta2 ** (iter_cnt + 1))
        direction = -m_hat / (np.sqrt(v_hat) + eps)

        # d) back‑tracking line‑search on the step magnitude
        step = step_base
        improved = False
        while step > 5e-7:
            cand = _project_onto_simplex(seq + step * direction)
            cand_score = _objective(cand)
            if cand_score < cur_score - 1e-12:     # strict improvement
                improved = True
                break
            step *= 0.5

        if improved:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, seq.copy()
                print(f"[Iter {iter_cnt:06d}] new best = {best_score:.6f}")
            # allow a slightly larger base step after a successful move
            step_base = max(step * 1.2, 5e-7)
        else:
            # ----------------------------------------------------------
            # No descent – perform a tiny random mass‑transfer move.
            i, j = np.random.choice(n_seq, 2, replace=False)
            delta = 0.02 * min(seq[i], 0.1)   # up to 2 % of the total mass
            if delta > 0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score - 1e-12:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score, best_seq = cand_score, seq.copy()
                        print(f"[Iter {iter_cnt:06d}] random move -> {best_score:.6f}")

        iter_cnt += 1

        # Occasionally (every ~500 iterations) inject a very small isotropic jitter
        # to shake the optimisation out of possible plateaux.
        if iter_cnt % 500 == 0 and time.time() < deadline:
            noise = np.random.normal(scale=1e-4, size=n_seq)
            cand = seq + noise
            cand = _project_onto_simplex(cand)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                seq = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score, best_seq = cand_score, seq.copy()
                    print(f"[Iter {iter_cnt:06d}] jitter improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
