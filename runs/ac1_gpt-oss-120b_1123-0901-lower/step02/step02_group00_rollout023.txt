def optimize_vector(a, target_sum, start_iteration, max_iters, deadline):
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum/n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = start_iteration
    best_local_score = float('inf')
    best_local_a = None

    while time.time() < deadline and iteration < max_iters:
        # compute conv via FFT
        N = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(A*A, n=N)[:2*n-1]
        M = conv.max()
        # active indices
        tol = M * 1e-12 + 1e-12
        active = np.where(conv >= M - tol)[0]
        # subgradient
        rev = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]

"""
Search for a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Try several lengths (the original length and a few scaled versions).
* For each length start several independent runs from different random
  initial vectors (uniform, exponential or an interpolated copy of the
  known best sequence).  The sum of the vector is fixed to
  sqrt(2·n) – the normalisation used by ``evaluate_sequence``.
* Inside a run a projected Adam optimiser is used on the exact
  sub‑gradient of the objective (the max‑convolution).  The optimisation
  proceeds in three phases:
    1.  Coarse phase – relatively large learning‑rate and occasional
        Gaussian jitter to escape plateaus.
    2.  Fine phase – learning‑rate decays, jitter magnitude shrinks.
    3.  Local random‑perturbation phase – small Gaussian moves are
        accepted only if they improve the objective (simple hill‑climb).
* The best vector seen over all runs (including the supplied global
  ``height_sequence_1`` if it exists) is returned.  The whole search is
  bound by the supplied time budget (default 1000 s).
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: full convolution and its maximum (via FFT)
def _full_conv_max(a: np.ndarray):
    """Return (max value, full convolution) of a with itself."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# -------------------------------------------------------------
# Helper: sub‑gradient of the max‑convolution
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Compute the average reversed vector that corresponds to the
    sub‑gradient of max_k (a*a)[k] for the active indices.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active_idxs.size)
    return rev

# Adam optimiser on a fixed length
def _optimize_vector(
    a: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
):
    """
    Run a projected Adam optimiser on ``a`` (which is assumed to be
    non‑negative).  The L1‑norm is kept equal to ``target_sum`` after each
    update.  Returns the best vector seen, its score (max‑convolution), and
    the number of performed iterations.
    """
    n = a.shape[0]

    # initialise: non‑negative & normalised
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.03                     # base learning rate (tuned empirically)

    # bookkeeping
    best_a = a.copy()
    # initial score
    cur_max, _ = _full_conv_max(a)
    best_score = cur_max

    # pre‑compute FFT size (constant for the whole run)
    N = 1 << ((2 * n - 1).bit_length())

    it = 0
    while it < max_iters and time.time() < deadline:
        # ---------- convolution ----------
        A = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
        M = float(conv.max())

        # ---------- active set ----------
        # numerical tolerance – relative to the magnitude of M
        tol = max(1e-14, M * 1e-12)
        active = np.where(conv >= M - tol)[0]

        # ---------- sub‑gradient ----------
        rev = _subgradient(a, active)          # shape (n,)

        # ---------- Adam update ----------
        it += 1
        m = beta1 * m + (1.0 - beta1) * rev
        v = beta2 * v + (1.0 - beta2) * (rev * rev)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)            # mild decay

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection (non‑negative + renorm) ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:                                   # should never happen
            a = np.full_like(a, target_sum / n)

        # ---------- occasional jitter ----------
        if it % 2500 == 0:
            # jitter magnitude shrinks with time
            jitter_scale = 0.015 * target_sum / n * (0.85 ** (it // 2500))
            jitter = np.random.randn(n) * jitter_scale
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---------- keep best ----------
        if M < best_score:
            best_score = M
            best_a = a.copy()
            # optional: print progress (kept concise)
            print(f"[len {n:5d} iter {it:6d}] new local best = {best_score:.6f}")

        # ---------- early exit if time is almost exhausted ----------
        if time.time() > deadline - 0.5:
            break

    # ---------- local random perturbation phase ----------
    # simple hill‑climb with tiny moves; cheap because we already have NFFT ready
    for _ in range(120):
        # generate a tiny Gaussian perturbation
        sigma = 0.004 * target_sum / n
        a_try = np.maximum(best_a + np.random.randn(n) * sigma, 0.0)
        a_try = a_try * (target_sum / a_try.sum())
        M_try, _ = _full_conv_max(a_try)
        if M_try < best_score:
            best_score = M_try
            best_a = a_try.copy()

    return best_a, best_score, it

    base = base * (target_sum0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # Choose a few lengths to explore.
    # The original length plus scaled versions give extra degrees of freedom.
    length_factors = [1.0, 1.2, 1.5, 2.0, 3.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    # keep them in a reasonable range
    lengths = [l for l in lengths if 300 <= l <= 8000]

    # -----------------------------------------------------------------
    # Main stochastic multi‑restart loop
    # -----------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1

        # pick a length (randomly to diversify the search)
        n = int(np.random.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        # -------------------------------------------------------------
        # initialise a candidate vector for this run
        if n == n0 and restart % 5 == 0:
            # from the current global best, with a little noise
            a0 = best_seq.copy()
            jitter = np.random.randn(n) * 0.02 * target_sum / n
            a0 = np.maximum(a0 + jitter, 0.0)
        elif np.random.rand() < 0.45:
            # linear interpolation of the base shape onto the new grid
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, base)
            a0 = np.maximum(a0, 0.0)
        else:
            # random Dirichlet / exponential start – encourages sparsity
            if np.random.rand() < 0.5:
                a0 = np.random.exponential(scale=1.0, size=n)
            else:
                a0 = np.random.rand(n)
        a0 = a0 * (target_sum / a0.sum())

        # -------------------------------------------------------------
        # decide how many iterations we can afford for this run
        # (the longer the better, but we also want many restarts)
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        # allocate a modest slice of the remaining budget
        max_iters = int(min(25000, remaining * 30))   # rough 30 iters / second

        # -------------------------------------------------------------
        # run the optimiser
        a_opt, score_opt, performed = _optimize_vector(
            a0, target_sum, max_iters, deadline
        )

        # update global best if needed
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[global] restart {restart:3d}, len={n:5d}, "
                f"local_iters={performed}, new best={best_score:.6f}"
            )

    # -----------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    # -----------------------------------------------------------------
    # Final report
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
