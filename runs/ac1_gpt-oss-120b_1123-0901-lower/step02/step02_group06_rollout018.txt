"""
Propose a non‑negative step‑function (a list of heights) that minimizes
the provided ``evaluate_sequence`` function.

Algorithm
---------
1. **Problem size** – we fix a length `n` (default 2000).  The sequence is
   normalised so that its ℓ₁–norm equals `sqrt(2·n)`.  With this scaling
   the evaluation function returns exactly the peak of the self‑convolution,
   i.e. it equals `max (f∗f)`.  Minimising that peak is the core task.

2. **Initialisation** – if a global variable ``height_sequence_1`` is
   available we start from it (renormalised).  Otherwise we generate a
   small pool of candidates (uniform, geometric, arcsine, few random
   Dirichlet draws) and keep the best one.

3. **Linear‑programming step** – for the current sequence `f` we compute
   `M = max(f∗f)`.  We then solve the LP

        maximise   1ᵀ g
        subject to  f∗g ≤ M   (element‑wise)
                    g ≥ 0

   The convolution `f∗g` is linear in `g`, therefore the constraints are
   ordinary linear inequalities.  The solution `g₀` is scaled to the same
   ℓ₁‑norm as `f` (i.e. `sqrt(2·n)`) giving `g`.  A convex combination
   `f←(1‑t)·f + t·g` is accepted if the new peak is strictly smaller than
   `M`.  The step size `t` is adapted by a simple back‑tracking line search.

   This LP‑based move pushes mass into positions where `f` is small while
   keeping the convolution bound, which provably cannot increase the peak
   and normally yields a strict decrease.

4. **Sub‑gradient refinement** – once the LP stops improving (or the time
   budget is nearly exhausted) we switch to a “soft” sub‑gradient descent
   that directly minimises the peak of `f∗f`.  The gradient of the maximum
   convolution entry is accumulated over the `K` largest shifts, momentum
   is applied, and the iterate is projected back onto the simplex
   `{x≥0, Σx = sqrt(2·n)}`.  The step size decays slowly.

5. **Time‑bounded execution** – the whole routine runs until the allotted
   time (`budget_s`) is almost exhausted.  The best sequence found
   (lowest objective value) is returned as a plain Python list.

The returned sequence has sum `sqrt(2·n)`.  Because of the scaling
property of the evaluation function, ``evaluate_sequence`` will output the
peak of its self‑convolution, which is the quantity we minimise.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Returns a new vector with the same shape as `v`.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive – return the centre of the simplex
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_onto_simplex_target(v, target_sum):
    """Project `v` onto {x ≥ 0, Σx = target_sum}."""
    w = _project_onto_simplex(v)          # sum = 1
    return w * target_sum


# ----------------------------------------------------------------------
def _initial_candidates(n, rng):
    """Create a small set of diverse start‑up sequences (ℓ₁‑norm = 1)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        r = 1.0 - factor / n
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # 3. arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. a few random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def _build_conv_matrix(f):
    """
    Build the matrix `A` such that (A @ g) == f∗g  (full linear convolution).
    Shape of `A` is (2·n‑1, n) where `n = len(f)`.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def _solve_lp_for_g(f, M, target_sum):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to A @ g ≤ M·1    (where A encodes f∗g)
                    g_i ≥ 0

    Returns the scaled vector `g` with ℓ₁‑norm = target_sum,
    or ``None`` if the solver fails.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:
        # SciPy not available – signal failure to caller
        return None

    n = len(f)
    A = _build_conv_matrix(f)                 # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximize Σ g_i  →  minimize -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    g_raw = np.asarray(res.x, dtype=np.float64)
    sum_raw = g_raw.sum()
    if sum_raw <= 0.0:
        return None

    scale = target_sum / sum_raw
    g = g_raw * scale
    # Clip tiny negatives caused by numerical noise
    g = np.clip(g, 0.0, None)
    return g


# ----------------------------------------------------------------------
def _subgradient_descent(seq, target_sum, deadline,
                         step0=0.1, beta=0.9, K=5, max_iter=10_000_000):
    """
    Soft sub‑gradient descent on the peak of the self‑convolution.
    Works directly on the simplex with ℓ₁‑norm = target_sum.
    """
    n = len(seq)
    step = step0 * target_sum
    momentum = np.zeros_like(seq)
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        conv = np.convolve(seq, seq)

        # indices of the K largest convolution entries
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -K)[-K:]

        # sub‑gradient for the selected shifts
        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += seq[k - idx]        # contribution a_{k-i}
        grad = grad / K

        # heavy‑ball momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # gradient step and projection back onto the simplex
        new_seq = seq - step * momentum
        new_seq = _project_onto_simplex_target(new_seq, target_sum)

        sc = evaluate_sequence(new_seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = new_seq.copy()
            # print improvement occasionally
            if it % 5000 == 0:
                print(f"[SGD] iter {it:06d} new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # slowly decay the step size
        step *= 0.9999

        # occasional jitter if we have been stuck for a long time
        if no_improve > 30_000:
            jitter = np.random.default_rng().normal(scale=0.01 * target_sum, size=n)
            new_seq = new_seq + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq = _project_onto_simplex_target(new_seq, target_sum)
            momentum[:] = 0.0
            no_improve = 0
            print(f"[SGD] iter {it:06d} stagnation – random perturbation applied")

        seq = new_seq
        it += 1

    return best_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, budget_s - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # problem size – try to keep the same length as the best known solution
    try:
        init_from_global = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_from_global.size
        print("[Info] Using length from `height_sequence_1` :", n)
    except Exception:
        n = 2000
        print("[Info] Falling back to default length:", n)

    target_sum = math.sqrt(2.0 * n)          # scaling that makes eval == max conv

    # ------------------------------------------------------------------
    # build an initial feasible point (ℓ₁‑norm = target_sum)
    if 'init_from_global' in locals():
        f = init_from_global.copy()
        s = f.sum()
        if s <= 0.0:
            f = np.full(n, target_sum / n)
        else:
            f = f * (target_sum / s)
        init_score = evaluate_sequence(f.tolist())
        print(f"[Init] Using `height_sequence_1`, score = {init_score:.6f}")
    else:
        best_seq = None
        best_score = float("inf")
        for cand in _initial_candidates(n, rng):
            cand = cand * target_sum                     # enforce ℓ₁‑norm
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
        f = best_seq
        init_score = best_score
        print(f"[Init] best seed score = {init_score:.6f}")

    # ------------------------------------------------------------------
    # keep track of the overall best sequence
    best_f = f.copy()
    best_obj = evaluate_sequence(best_f.tolist())
    print(f"[Start] objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming driven improvement
    t = 0.2                     # initial mixing factor
    max_lp_iters = 30
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        conv = np.convolve(f, f)
        M = conv.max()
        # solve LP to obtain a new direction g
        g = _solve_lp_for_g(f, M, target_sum)
        if g is None:
            print("[LP] solver failed – exiting LP phase")
            break

        # try to mix with the current f; back‑track on t if needed
        t_cur = t
        improved = False
        while t_cur > 1e-6:
            new_f = (1.0 - t_cur) * f + t_cur * g
            new_score = evaluate_sequence(new_f.tolist())
            if new_score < best_obj - 1e-12:
                # accept the step
                f = new_f
                best_f = new_f.copy()
                best_obj = new_score
                print(f"[LP {lp_iter:02d}] t={t_cur:.4f} → new best = {best_obj:.6f}")
                improved = True
                # increase step size a little for the next LP iteration
                t = min(t_cur * 1.2, 0.9)
                break
            else:
                t_cur *= 0.5

        if not improved:
            # could not improve – shrink global step size
            t *= 0.5
            if t < 1e-5:
                # inject a small random jitter to escape a flat region
                jitter = rng.normal(scale=0.01 * target_sum, size=n)
                f = f + jitter
                f = np.clip(f, 0.0, None)
                f = _project_onto_simplex_target(f, target_sum)
                t = 0.2
                print(f"[LP {lp_iter:02d}] jitter injected")
        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent for fine tuning (if time permits)
    if time.time() < deadline:
        print("[Phase 2] starting sub‑gradient descent")
        best_f = _subgradient_descent(best_f, target_sum, deadline,
                                      step0=0.2, beta=0.9, K=7)

        best_obj = evaluate_sequence(best_f.tolist())
        print(f"[Finish] final objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    return best_f.tolist()
