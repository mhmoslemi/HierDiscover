"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Several lengths are tried (the original length and a few larger ones).
* For every length several initial vectors are generated:
      – the previous best construction (if any);
      – uniform, linear, quadratic and sinusoidal “window’’ shapes;
      – a random Dirichlet draw.
* Each start is refined by Adam on the exact sub‑gradient of the
  ``max`` of the self‑convolution (the gradient uses *all* indices
  whose value is within a small relative tolerance of the maximum,
  not only the single maximiser).  After each step the vector is
  projected onto the feasible set (non‑negative, fixed ℓ₁‑norm).
* Jitter is added regularly to avoid plateaus and a simple
  “no‑improvement‑for‑a‑while’’ restart is performed.
* The best sequence (lowest exact score) found before the wall‑clock
  deadline (default 1000 s) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray):
    """
    Full convolution a * a using an FFT of length a power of two.
    Returns (conv_array, max_value, index_of_max).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k_max = int(conv.argmax())
    return conv, float(conv.max()), k_max


def _active_set(conv: np.ndarray, M: float, rel_tol: float = 0.999):
    """
    Return indices of ``conv`` that are within ``rel_tol`` of the maximum.
    ``rel_tol`` close to 1 keeps only the top few values.
    """
    # ``rel_tol`` is a relative tolerance (e.g. 0.999 means within 0.1% of max)
    thr = max(1e-12, M * rel_tol)
    return np.where(conv >= thr)[0]


def _subgradient(a: np.ndarray, active: np.ndarray):
    """
    Sub‑gradient of the (smoothed) max‑convolution.
    For each active shift k we add the reversed slice a_{k‑i} to the
    gradient, then average over the active set.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size:
        rev /= float(active.size)
    return rev


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 10_000_000,
    rel_tol: float = 0.999,
    jitter_interval: int = 2000,
):
    """
    Adam optimiser on the exact max‑convolution objective.
    Returns the best vector encountered and its associated score.
    """
    n = a0.shape[0]
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * (target_sum / a.sum())          # enforce normalisation

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.09           # a bit larger than the reference code
    decay = 1e-4

    best_a = a.copy()
    # because we normalise Σa = sqrt(2n) the objective equals the maximal
    # convolution value.
    conv, M, _ = _conv_full_fft(a)
    best_score = float(M)

    iteration = 0
    last_improve = 0

    while time.time() < deadline and iteration < max_iters:
        # ------------------------------------------------------------------
        # 1) convolution and active set
        conv, M, _ = _conv_full_fft(a)
        active = _active_set(conv, M, rel_tol=rel_tol)

        # ------------------------------------------------------------------
        # 2) sub‑gradient of the (smoothed) max
        rev = _subgradient(a, active)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # ------------------------------------------------------------------
        # 3) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + decay * iteration)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # 4) Projection (non‑negative + ℓ₁ normalisation)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:                               # extremely unlikely
            a = np.full_like(a, target_sum / n)

        # ------------------------------------------------------------------
        # 5) keep the best
        if M < best_score:
            best_score = M
            best_a = a.copy()
            last_improve = iteration
            print(
                f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}"
            )

        # ------------------------------------------------------------------
        # 6) occasional jitter to shake the iterates
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # ------------------------------------------------------------------
        # 7) restart if stuck for a long while
        if iteration - last_improve > 1_000_000:
            # re‑initialise from the current best with a fresh Adam state
            a = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration

    return best_a, best_score

def _initial_vectors(n: int, base: np.ndarray | None):
    """
    Produce a list of diverse initial vectors of length ``n``.
    ``base`` may be ``None`` – in that case it is ignored.
    All vectors are normalised to Σ = sqrt(2 n).
    """
    target_sum = math.sqrt(2.0 * n)
    init_list = []

    # 1) the previous best if it matches the length
    if base is not None and base.shape[0] == n:
        init_list.append(base.copy())

    # 2) uniform
    uni = np.full(n, target_sum / n)
    init_list.append(uni)

    # 3) linear decreasing (triangular) – useful baseline
    lin = np.linspace(1.0, 0.0, n)
    lin = np.maximum(lin, 0.0)
    lin *= (target_sum / lin.sum())
    init_list.append(lin)

    # 4) quadratic decreasing
    quad = (np.linspace(1.0, 0.0, n) ** 2)
    quad = np.maximum(quad, 0.0)
    quad *= (target_sum / quad.sum())
    init_list.append(quad)

    # 5) Hann window (sinusoidal, zero at ends)
    idx = np.arange(n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
    hann = np.maximum(hann, 0.0)
    hann *= (target_sum / hann.sum())
    init_list.append(hann)

    # 6) Random Dirichlet (exponential draws)
    rng = np.random.default_rng()
    for _ in range(3):
        rand = rng.exponential(scale=1.0, size=n)
        rand = np.maximum(rand, 0.0)
        rand *= (target_sum / rand.sum())
        init_list.append(rand)

    return init_list
