"""
Step‑function optimizer using a smooth‑max surrogate and Adam.

The algorithm works with a probability vector (sum = 1, non‑negative) on a
grid of size *n*.  At each iteration a fast FFT based convolution is used
to obtain the self‑convolution `c = a*a`.  Instead of the nonsmooth
`max(c)` we minimise a smooth approximation

    S_α(c) = (1/α)·log ∑_k exp(α·c_k) ,

whose gradient is a weighted convolution of the current sequence with the
soft‑max weights of the convolution entries.  The weights concentrate on
the largest entries as `α` grows, so the method first smooths the
objective and later focuses on the true peak.

An Adam optimiser provides adaptive step sizes; after each update the
vector is projected onto the probability simplex by simple clipping and
renormalisation (which is fast and keeps the feasible set).  The learning
rate and the smoothness parameter `α` are slowly decayed / increased,
respectively.  A small jitter is added after long stagnation to escape
local minima.

The whole loop runs until the given time budget (≈ 1000 s) expires and the
best sequence found is returned.  The objective is evaluated with the
provided ``evaluate_sequence`` only at the end; a fast FFT based estimator
is used during optimisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.  Used only as a safety net; the main loop uses
    simple clipping+renorm for speed.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:              # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator):
    """
    Generate a small collection of diverse, already normalised candidates.
    """
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # 3. arcsine law (heavy at the boundaries)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. a few random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # 5. optionally reuse the previous best if it exists

        if prev.ndim == 1 and len(prev) > 1:
            prev = np.clip(prev, 0, None)
            s = prev.sum()
            if s > 0:
                cand.append(prev / s)
    except NameError:
        pass

    return cand


# ----------------------------------------------------------------------
def _fast_objective(seq: np.ndarray, n: int, conv_len: int, L: int):
    """
    Fast evaluation of 2·n·max(conv) under the assumption sum(seq)=1.
    """
    A = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(A * A, n=L)[:conv_len]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list of heights (floats) that aims to minimise
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. problem size (a good compromise between expressiveness and speed)
    n = 4000                        # feel free to tweak; 4 k works fast with FFT
    conv_len = 2 * n - 1
    L = 1 << ((3 * n - 2).bit_length())   # power‑of‑two >= 3n‑2 for linear conv.

    # ------------------------------------------------------------------
    # 2. initialise a pool of candidates and pick the best one
    best_seq = None
    best_score = np.inf
    for cand in _initial_candidates(n, rng):
        seq = np.clip(cand, 0, None)
        if seq.sum() == 0:
            continue
        seq = seq / seq.sum()                    # enforce simplex
        score = _fast_objective(seq, n, conv_len, L)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best fast‑objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation with a smooth‑max surrogate
    cur_seq = best_seq.copy()
    m = np.zeros_like(cur_seq)          # first moment
    v = np.zeros_like(cur_seq)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                           # base learning rate
    alpha_smooth = 200.0                # start with a very soft max
    alpha_target = 20000.0
    last_improve = 0
    iteration = 0
    max_iters_without_improve = 30000   # after this we add jitter

    while time.time() < deadline:
        # ----- FFT of the current sequence ---------------------------------
        A = np.fft.rfft(cur_seq, n=L)

        # ----- self‑convolution (needed for the smooth‑max) ----------------
        conv = np.fft.irfft(A * A, n=L)[:conv_len]
        max_conv = conv.max()

        # ----- smooth‑max weights -----------------------------------------
        # w_k ∝ exp(α·(conv_k – max_conv)); the max entry receives weight 1.
        diff = conv - max_conv
        # avoid overflow – shift by the maximum (which is 0) is already done
        w = np.exp(alpha_smooth * diff)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / conv_len)
        else:
            w /= w_sum

        # ----- gradient of the smooth‑max ---------------------------------
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A * W, n=L)[:n]          # (a * w)[0:n]
        grad = 2.0 * grad_full                           # sub‑gradient of max

        # ----- Adam update -------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projected gradient step (clip + renorm) --------------------
        new_seq = cur_seq - step
        new_seq = np.clip(new_seq, 0.0, None)
        s = new_seq.sum()
        if s < 1e-12:
            new_seq = np.full_like(new_seq, 1.0 / n)
        else:
            new_seq = new_seq / s

        # ----- evaluate the fast objective for the new point ----------------
        cur_obj = _fast_objective(new_seq, n, conv_len, L)

        # ----- keep the global best ----------------------------------------
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = new_seq.copy()
            last_improve = iteration
            print(f"[Iter {iteration:06d}] new best fast‑obj = {best_score:.6f}")

        # ----- stagnation handling -----------------------------------------
        if iteration - last_improve > max_iters_without_improve:
            # small random jitter + reset Adam moments
            jitter = rng.normal(scale=0.001, size=n)
            new_seq = best_seq + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq /= new_seq.sum()
            cur_seq = new_seq
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[Iter {iteration:06d}] stagnation – jitter applied")
            continue

        # ----- increase the smoothness (focus more on the true max) -------
        if alpha_smooth < alpha_target:
            alpha_smooth = min(alpha_target, alpha_smooth * 1.001)

        # ----- move to the next iterate ------------------------------------
        cur_seq = new_seq

    # ------------------------------------------------------------------
    # 4. final evaluation with the official function (the judge will do this
    #    automatically, but we print it for our own information).
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished. Best official objective = {final_score:.6f} (length {n})")
    return best_seq.tolist()
