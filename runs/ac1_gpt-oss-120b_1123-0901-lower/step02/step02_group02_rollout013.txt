def _cross_corr(w, a):
    # w length = 2n-1, a length n
    n = a.shape[0]
    m = len(w)
    size = 1 << ((m + n - 1 - 1).bit_length())  # next power of two

    return grad

"""
Propose a non‑negative step‑function (list of heights) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm blends two complementary optimisation strategies:

1. **Soft‑max – gradient descent**  
   For a given length `n` we minimise the smooth surrogate
   `L_β(a)=∑_k exp(β·(conv_k−max(conv)))·conv_k`.  Its gradient can be
   written as a single cross‑correlation, which we evaluate with an FFT.
   A decreasing learning‑rate together with Euclidean projection onto the
   simplex `{x ≥ 0, Σx = √(2n)}` keeps the iterates feasible.
   The temperature `β` is slowly increased, turning the surrogate into the
   true `max` as optimisation proceeds.

2. **Active‑set refinement (fallback)**  
   After the main search we optionally run a few iterations of the classic
   active‑set projected gradient descent (the method that produced the
   previous 1.520 bound).  This often squeezes a little extra improvement.

Many independent restarts with random lengths (800 … 8000) and random
initial power‑law shapes are performed until the global time budget (≈1000 s)
is exhausted.  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – full linear convolution a*a using a power‑of‑2 FFT
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length) via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper – indices whose convolution value is close to the current maximum
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    max_val = conv.max()
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# Helper – gradient of Σ_{k∈active} conv_k w.r.t. a (used by fallback method)
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# Helper – cross‑correlation needed for the soft‑max gradient
# ----------------------------------------------------------------------
def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the vector `g` of length n with
        g[k] = Σ_i w[i] * a[i - k]   (out‑of‑range indices give 0)
    using FFTs.  This is exactly the gradient (up to the factor 2) of
    Σ_i w[i]·conv_i with respect to a.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    # Full linear convolution of reversed w with a.
    w_rev = w[::-1]
    size = 1 << ((m + n - 1 - 1).bit_length())   # next power of two ≥ m+n‑1
    fw = np.fft.rfft(w_rev, n=size)
    fa = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fw * fa, n=size)[:m + n - 1]
    # Extract the needed n entries:
    # grad[k] = conv_full[m‑1‑k]   for k = 0 … n‑1
    idx = m - 1 - np.arange(n, dtype=np.int64)
    grad = conv_full[idx]
    return grad


# ----------------------------------------------------------------------
# Core optimiser – soft‑max based projected gradient descent for a fixed n
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    base_lr: float = 0.4,
    momentum: float = 0.9,
    beta_start: float = 0.8,
    beta_inc: float = 0.02,
    beta_max: float = 20.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max surrogate of max(conv) for a vector of length n.
    Returns the best vector found in this run and its (true) objective value.
    """
    target_sum = np.sqrt(2.0 * n)           # Σ a_i = √(2n) ⇒ objective = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best seen locally
    conv = _conv_fft(a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_local = a.copy()

    velocity = np.zeros_like(a)
    beta = beta_start
    lr = base_lr

    while time.time() < run_deadline:
        # ----- convolution & soft‑max weights -----
        conv = _conv_fft(a)
        max_val = conv.max()
        diff = conv - max_val                # ≤ 0
        w = np.exp(beta * diff)              # w[max] = 1, others ∈ (0,1]

        # ----- gradient of the weighted sum -----
        grad = 2.0 * _cross_corr(w, a)        # see derivation above

        # ----- momentum update -----
        velocity = momentum * velocity + (1.0 - momentum) * grad

        # ----- adaptive step size (scale‑invariant) -----
        max_abs = np.max(np.abs(velocity))
        if max_abs == 0:
            max_abs = 1.0
        eta = lr * max_val / max_abs

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # ----- occasional jitter to escape shallow plateaus -----
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a_new = _project_onto_simplex(a_new + noise, target_sum)
            lr *= 0.94

        # ----- evaluate true objective -----
        conv_new = _conv_fft(a_new)
        score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score:.6f}")

        a = a_new
        # anneal temperature and learning rate
        beta = min(beta + beta_inc, beta_max)
        lr *= 0.9995

    return best_local, best_score


# ----------------------------------------------------------------------
# Core optimiser – classic active‑set projected gradient descent (fallback)
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    rel_tol: float = 5e-4,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the true max‑convolution objective."""
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    conv = _conv_fft(a)
    max_val, _ = _active_indices(conv, rel_tol)
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    velocity = np.zeros_like(a)
    step_factor = base_step

    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)
        grad = _gradient(a, active)

        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity))
        if max_grad == 0:
            max_grad = 1.0
        eta = step_factor * max_val / max_grad

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92

        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  active‑set local best = {score:.6f}")

        a = a_new
        step_factor *= 0.9997

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine runs for at most `budget_s` seconds and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # plausible range of lengths – larger n gives a finer “flat” autocorrelation
    min_n, max_n = 800, 8000

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a few seconds of work.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # shrink problem size a little when we are close to the end
        if remaining < 20.0:
            n = np.random.randint(800, 2000)
        else:
            n = np.random.randint(min_n, max_n + 1)

        # ---- initial shape ------------------------------------------------
        # a_i ∝ (i+1)^{-p} with a random exponent around 0.5
        p = np.random.uniform(0.45, 0.55)
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p)
        init += np.random.rand(n) * 1e-4   # tiny jitter

        # ---- occasional warm‑start from the current global best ----------
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        # ---- allocate a time slice for this run -------------------------
        run_deadline = min(deadline, time.time() + 6.0)

        # ---- run the soft‑max optimiser ---------------------------------
        seq_local, score_local = _optimise_one_length_softmax(
            n,
            init,
            run_deadline,
            base_lr=0.4,
            momentum=0.9,
            beta_start=0.8,
            beta_inc=0.025,
        )

        # ---- keep the overall best ---------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})")

        # optional early exit if we beat the target comfortably
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final polishing with the classic active‑set method (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n_best = best_seq.size
        extra_deadline = min(deadline, time.time() + 5.0)
        seq_refined, refined_score = _optimise_one_length(
            n_best,
            best_seq,
            extra_deadline,
            rel_tol=5e-4,
            base_step=0.08,
            momentum_beta=0.92,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] POLISH global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – in the extremely unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
