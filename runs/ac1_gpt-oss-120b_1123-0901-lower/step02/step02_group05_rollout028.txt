def _max_conv_and_grad_fft(a):
    n = a.shape[0]
    # compute conv via FFT
    size = 1
    while size < n + n - 1:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]
    # gradient: derivative of a_i * a_{k-i}: 2*a_{k-i}
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high+1)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv
```

    beta1 = 0.9
    beta2 = 0.999

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # temperature schedule: start at tau0 = max(initial max conv, 1.0)
    tau = None

    start = time.time()
    while time.time() - start < time_budget:
        # compute convolution and softmax weights
        b = _conv_full(a)  # efficient conv using fft if needed
        max_b = float(b.max())
        if tau is None:
            tau = max_b  # initialize temperature
        # compute softmax weights
        # shift by max for numerical stability
        shifted = b - max_b
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # compute gradient = sum_k w_k * grad_{k}
        # grad_k = 2 * a shifted
        # Efficiently via convolution a * w_rev
        g = 2 * _conv_weighted_gradient(a, w)
        # Enforce zero-mean
        g = g - np.mean(g)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project onto simplex
        a = _project_onto_simplex(a, target_sum)

        # evaluate current
        try:
            score = float(evaluate_sequence(a.tolist()))
        except:
            score = math.inf

            best_seq = a.copy()

"""
Annealed projected‑gradient descent for the step‑function problem.

The objective is
    2 * n * max(conv(a,a)) / (sum(a)**2)

Because the evaluation is scale‑invariant we keep the ℓ₁‑norm fixed to
√(2n) (the same normalisation used by the reference code).  The algorithm
projects all iterates onto the simplex {a ≥ 0, Σa = √(2n)}.

Instead of optimising the (non‑smooth) max we use a smooth “soft‑max’’:
    Sτ(b) = τ·log∑ₖexp(bₖ/τ)

Its gradient is a weighted sum of the gradients of every convolution entry,
with weights proportional to exp(bₖ/τ).  As τ → 0 the soft‑max converges to
the true max, so we start with a relatively large temperature and decay it.
The gradient of a single entry bₖ is 2·a shifted, therefore the full soft‑max
gradient can be computed by a single convolution
    g = 2·conv(a, w_rev)
where w are the soft‑max weights and w_rev = w[::-1].

An Adam‑style update (with momentum and adaptive step size) is applied,
followed by Euclidean projection onto the simplex.  The routine runs many
independent restarts with different sequence lengths (up to a few thousand)
and keeps the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT for speed on larger vectors)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (length 2·len(a)-1) using FFT when beneficial."""
    n = a.shape[0]
    if n <= 1500:
        return np.convolve(a, a)
    # next power of two for zero‑padding
    m = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_k w_k ∇_a (a∗a)_k efficiently.
    Returns a vector of the same length as a.
    """
    n = a.shape[0]
    # w_rev has length 2n‑1
    w_rev = w[::-1]
    if n <= 1500:
        conv = np.convolve(a, w_rev)
    else:
        m = 1 << ((n + w_rev.size - 1).bit_length())
        fa = np.fft.rfft(a, n=m)
        fw = np.fft.rfft(w_rev, n=m)
        conv = np.fft.irfft(fa * fw, n=m)
    # slice conv[n‑1 : n‑1+n] and reverse to obtain the gradient
    g = conv[n - 1 : n - 1 + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Small mutation helper (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Annealed Adam descent on the soft‑max of the convolution
# ----------------------------------------------------------------------
def _annealed_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Perform a projected Adam‑descent using a soft‑max approximation of max(conv).
    Returns the best sequence seen and its evaluation score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial evaluation
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    n = a.size
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # base learning rate – we scale it with the typical size of a entry
    base_lr = 0.25 * target_sum / n

    # start temperature with the current maximal convolution entry
    b = _conv_full(a)
    tau = float(b.max()) if b.size else 1.0

    start = time.time()
    while time.time() - start < time_limit:
        # ------------------------------------------------------------------
        # Compute convolution and soft‑max weights
        # ------------------------------------------------------------------
        b = _conv_full(a)            # length 2n‑1
        max_b = float(b.max())
        # temperature decay – slowly shrink towards the true max
        tau = max(tau * 0.998, 0.02 * max_b)

        # stable soft‑max (subtract max before exp)
        shifted = b - max_b
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # Gradient of soft‑max (weighted sum of per‑shift gradients)
        # ------------------------------------------------------------------
        g = _weighted_gradient(a, w)

        # keep the update in the tangent space of the simplex (zero‑sum)
        g = g - np.mean(g)

        # ------------------------------------------------------------------
        # Adam update + projection
        # ------------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_onto_simplex(a, target_sum)

        # ------------------------------------------------------------------
        # Evaluate
        # ------------------------------------------------------------------
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] {print_tag} new best = {score:.6f} (n={n})"
            )

        # occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.01:
            sigma = 0.01 * target_sum / n
            a = _mutate(a, target_sum, sigma)

        # stop early if we already beat the target bound
        if best_score < 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Resize helper (linear interpolation / down‑sampling)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)
