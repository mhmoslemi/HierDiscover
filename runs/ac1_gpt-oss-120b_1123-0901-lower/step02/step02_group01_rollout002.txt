"""
Propose a step‑function sequence with a low evaluation score.

The algorithm combines three ideas:

1. **Large support** – we use n≈4096.  With more points the theoretical
   lower bound of the objective (≈1.0) becomes reachable.

2. **Smooth‑max surrogate + Adam** – we minimise a differentiable
   approximation of the max‑convolution using the soft‑max
   (log‑sum‑exp).  Adam with simplex projection gives fast, stable
   convergence.

3. **Continuation & random shakes** – we start with a smooth
   surrogate (small λ) and increase λ slowly, while injecting tiny
   random perturbations when progress stalls.

The code respects the 1000 s time budget, prints progress and finally
returns the best sequence found (as a plain Python list).  If a global
variable ``height_sequence_1`` exists it is resampled to the chosen
length and used as an additional starting point.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Full linear self‑convolution via FFT (real‑to‑real)
def _full_convolution(a, size=None):
    """Return a*a (linear convolution, length 2·len(a)-1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 3.  Exact objective used by evaluate_sequence (sum(a) is forced =1)
def _objective(a):
    """2·n·max(conv) when sum(a)=1 (a is non‑negative)."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# 4.  Gradient of the smooth‑max (log‑sum‑exp) surrogate
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Gradient of  (1/lam)·log∑_k exp(lam·conv_k)
    with respect to a (linear cross‑correlation w ★ a, w is soft‑max).
    """
    n = len(a)
    # stable soft‑max
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# 5.  Resample an existing sequence to a new length (linear interpolation)
def _resample_sequence(old_seq, new_n):
    old_seq = np.asarray(old_seq, dtype=np.float64)
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, old_seq)
    # enforce simplex
    new_seq = _project_onto_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    evaluate_sequence().  Returns the best list of floats found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length – a power of two gives cheap FFTs.
    n = 4096                     # ≈4 k points (good trade‑off)

    # Pre‑compute FFT sizes (powers of two)
    size_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))  # enough for correlation

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 2. Gather a few cheap starting points.
    #    a) resample the historic construction if available
    old = globals().get("height_sequence_1")
    if old is not None:
        cand = _resample_sequence(old, n)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Init from height_sequence_1] obj={val:.6f}")

    # b) geometric sequences a_i ∝ r^i   (r close to 1)
    for r in np.linspace(0.990, 0.9999, 12):
        if time.time() > deadline:
            break
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] r={r:.5f}  obj={val:.6f}")

    # c) a few random Dirichlet draws
    for _ in range(10):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init rand] obj={val:.6f}")

    # If for some reason we still have no candidate, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3. Optimisation loop – Adam + continuation on λ.
    seq = best_seq.copy()
    # Adam state
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.15               # base learning rate
    lr_decay = 0.9999       # gentle decay

    lam = 1.0               # start with a smooth surrogate
    lam_factor = 1.005      # increase λ slowly

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ---- 3a) self‑convolution
        conv = _full_convolution(seq, size=size_conv)

        # ---- 3b) soft‑max gradient
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_corr)

        # ---- 3c) Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- 3d) projection onto simplex
        new_seq = _project_onto_simplex(seq - step)

        # ---- 3e) exact objective (fast, no evaluate_sequence call)
        val = _objective(new_seq)

        # keep best ever
        if val < best_val:
            best_val, best_seq = val, new_seq.copy()
            last_improve = time.time()
            print(f"[Iter {it:06d}] new best obj={val:.6f}")

        # ---- 3f) occasional shake when stuck >2 s
        if time.time() - last_improve > 2.0:
            noise = (np.random.rand(n) - 0.5) * 1e-4
            new_seq = _project_onto_simplex(new_seq + noise)
            last_improve = time.time()  # reset timer after shake

        # ---- 3g) update parameters
        seq = new_seq
        lr *= lr_decay
        lam *= lam_factor

        # optional early exit when we already beat the target comfortably
        if best_val < 1.30 and time.time() - deadline > -30:  # 30 s left
            # give a short tail for fine‑tuning
            pass

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # Return as plain Python list – the evaluator will handle the final clipping.
    return best_seq.tolist()
