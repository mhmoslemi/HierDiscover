"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

with a fixed time budget (default 1000 s).

Algorithm
---------

* Initialise from the historic best sequence if present, otherwise a uniform
  distribution (sum = √(2 n) so that the evaluation reduces to max(conv)).
* Repeatedly pick a length n (mostly around the current best length) and
  generate a starting vector (resize the current best or a fresh uniform
  vector).  A small Gaussian mutation is applied to give diversity.
* On each candidate we run a **hill‑climbing** optimiser that preserves the
  L1‑sum exactly.  At each step a random source index i (with a[i] > 0) moves
  a random fraction of its mass to a random destination j.  The convolution
  array is updated *incrementally* using the exact formula

        conv ← conv − 2·Δ·shift(i) + 2·Δ·shift(j)
                + Δ²·e_{2i} + Δ²·e_{2j} − 2·Δ²·e_{i+j}

  where ``shift(k)`` is the vector a shifted by k positions.
  The new maximum of the convolution is accepted if it improves the score,
  otherwise it is accepted with a temperature‑controlled probability
  (simulated‑annealing style) to escape local minima.
* The best sequence seen during the hill‑climb is kept.  When the time budget
  is nearly exhausted a final short projected‑gradient descent (the helper
  from the reference implementation) refines the result.
* The globally best sequence across all runs is returned.

The incremental update makes each hill‑climbing move O(n) (≈ 3000 operations
for n≈1500) and allows millions of trial moves within the allotted time,
far more than a naïve recomputation of the full convolution.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Resize a sequence by linear interpolation / subsampling
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Very small mutation helper (adds Gaussian noise and re‑projects)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Incremental hill‑climbing optimiser
# ----------------------------------------------------------------------
def _hill_climb(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    max_alpha: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Perform many cheap moves that keep the L1‑sum exactly target_sum.
    Returns the best sequence seen and its evaluation score.
    """
    a = init.astype(float).copy()
    n = a.size
    # ensure exact sum
    diff = target_sum - a.sum()
    if abs(diff) > 1e-12:
        a = _project_onto_simplex(a, target_sum)

    # full convolution and current maximum
    conv = np.convolve(a, a)
    cur_max = float(conv.max())
    best_seq = a.copy()
    best_val = cur_max

    start = time.time()
    it = 0

    # simulated‑annealing schedule (exponential cooling)
    temp_start = cur_max * 0.5 + 1e-6
    temp_end = cur_max * 0.001 + 1e-9

    while time.time() - start < time_limit:
        # elapsed fraction for temperature decay
        frac = (time.time() - start) / time_limit
        temp = temp_start * ((temp_end / temp_start) ** frac)

        # pick a source index with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # pick a random fraction of the source mass to move
        alpha = np.random.rand() * max_alpha
        delta = alpha * a[i]          # 0 < delta ≤ a[i]

        # store a copy of the convolution for a possible revert
        conv_old = conv.copy()

        # linear part of the update (uses the *old* a)
        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a

        # quadratic part (indices are always inside the array)
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        # apply the move to a
        a[i] -= delta
        a[j] += delta

        new_max = float(conv.max())

        # accept if improvement, otherwise accept with Metropolis probability
        accept = False
        if new_max < cur_max:
            accept = True
        elif temp > 0 and np.random.rand() < math.exp(-(new_max - cur_max) / temp):
            accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
                print(
                    f"[{time.time():.2f}] hillclimb new best max={best_val:.6f} (n={n})"
                )
        else:
            # revert all changes
            conv = conv_old
            a[i] += delta
            a[j] -= delta

        # occasional full recomputation to keep numerical drift in check
        if it % 2000 == 0:
            conv = np.convolve(a, a)
            cur_max = float(conv.max())
        it += 1

    # final evaluation via the provided routine (adds the 2·n / sum² factor)
    try:
        score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        score = math.inf
    return best_seq, score
