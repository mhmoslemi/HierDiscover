def _optimize_on_length(n, init_a, best_seq, best_score, deadline):
    target_sum = math.sqrt(2.0 * n)
    a = init_a.astype(float)
    # ensure nonnegative
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a * (target_sum / a.sum())
    else: a = np.full_like(a, target_sum / n)
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    # schedule for temperature
    beta_temp = 0.1  # initial
    beta_max = 30.0  # final
    beta_inc = 1e-4  # per iteration increment

    while time.time() < deadline:
        # compute convolution via FFT
        conv = conv_fft(a)
        M = conv.max()
        # softmax weights for smoothing
        beta_val = min(beta_max, beta_temp + beta_inc * iteration)
        # shift for stability
        conv_shift = conv - conv.max()
        exp_vals = np.exp(beta_val * conv_shift)
        w = exp_vals / exp_vals.sum()
        # gradient of M_beta:
        grad_M = grad_M_fft(a, w)  # shape n
        # const factor:
        const = 2.0 * n / (target_sum ** 2)
        grad = const * grad_M

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        lr = lr0 / (1 + 1e-4 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # jitter occasional
        if iteration % 4000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0)

            cur_score = const * M  # M is from conv

                best_score = cur_score
                best_seq = a.copy()
                print(f"[len {n:5d} iter {iteration:7d}] new best score = {best_score:.6f}")

    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
* Use several candidate lengths (the supplied length and a few scaled
  versions).  For each length run a few independent Adam optimisations.
* The optimisation minimises a *smooth* approximation of the maximum
  convolution value:
        M_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
  (soft‑max).  As β grows the smooth function approaches the true
  maximum, while for small β the landscape is much smoother and easier
  to explore.
* Exact gradient of M_β can be written in closed form:
        ∂M_β/∂a = 2·(a_rev ★ w)   with   w_k = exp(β·c_k)/Σ exp(β·c)
  where c = a*a (full convolution) and ★ denotes convolution.
  Both convolutions are evaluated with FFT (O(N log N)).
* Adam updates the parameters, then we project onto the feasible set:
  clip to ≥0 and renormalise so that Σ a_i = √(2·N) (the standard
  normalisation used by the scoring function).
* The temperature β is increased slowly (annealing) so that the optimiser
  first moves in a smooth landscape and later focuses on the true max.
* Several restarts with random initialisations and with the provided
  ``height_sequence_1`` are performed; the best sequence found over all
  runs is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution using FFT (real input)
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a (length 2*len(a)-1)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth maximum M_β with respect to a.
# Uses FFT to compute 2·(a_rev ★ w) where w are the soft‑max weights.
def _grad_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of M_β for a given weight vector w (length 2n‑1)."""
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # convolution length = n + (2n-1) - 1 = 3n-2
    conv_len = 3 * n - 2
    N = 1 << ((n + (2 * n - 1) - 1).bit_length())
    A = np.fft.rfft(a_rev, n=N)
    B = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(A * B, n=N)
    conv = conv[:conv_len]              # keep only needed part
    # slice the middle segment (indices n-1 … n-1+n-1)
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for a *single* length.
def _adam_optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on a vector of length ``n`` starting from ``init_a``.
    Updates and returns the global best sequence/score.
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the starting point
    a = np.maximum(init_a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.04                # base learning rate
    iteration = 0

    # Temperature schedule for the soft‑max
    beta_start = 0.2
    beta_end = 25.0
    beta_inc_per_it = (beta_end - beta_start) / 300_000.0  # reach β≈β_end after ~3e5 iters

    # constant factor in the objective (see evaluate_sequence)
    const = 2.0 * n / (target_sum ** 2)

    while time.time() < deadline:
        # -------- convolution and smooth‑max weights ----------
        conv = _conv_fft(a)                # shape 2n-1
        M_exact = conv.max()               # exact max (for reporting)

        # temperature for this iteration
        beta = beta_start + beta_inc_per_it * iteration
        if beta > beta_end:
            beta = beta_end

        # soft‑max weights (stable version)
        shift = conv - conv.max()
        exp_vals = np.exp(beta * shift)
        w = exp_vals / exp_vals.sum()

        # -------- gradient of the smooth objective ----------
        grad_M = _grad_smooth_max(a, w)     # ∂M_β/∂a
        grad = const * grad_M               # full gradient of the score

        # -------- Adam update ----------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)     # very slow decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # -------- projection onto feasible set ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional tiny jitter to escape plateaus
        if iteration % 4000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # -------- periodic exact evaluation ----------
        if iteration % 1500 == 0:
            cur_score = const * M_exact          # identical to evaluate_sequence when normalised
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[len {n:5d} iter {iteration:7d}] "
                    f"new best score = {best_score:.6f}"
                )

    # final check with the last a
    conv = _conv_fft(a)
    final_score = const * conv.max()
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()
        print(f"[len {n:5d}] final improvement to {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin

    # --------------------------------------------------------------
    # Load a previously found construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)                # simple fallback

    n0 = base.shape[0]
    target0 = math.sqrt(2.0 * n0)
    base = np.maximum(base, 0.0)
    base = base * (target0 / base.sum())

    # evaluate the initial candidate
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Choose a range of lengths (including the original)
    length_factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]
    lengths = sorted({max(300, int(n0 * f)) for f in length_factors})
    lengths = [min(l, 15000) for l in lengths]   # cap to keep FFT reasonable

    # --------------------------------------------------------------
    # Main optimisation loop: for each length we perform a few restarts
    restarts_per_len = 3

    for idx, n in enumerate(lengths):
        # allocate a fair share of the remaining time to this length
        now = time.time()
        remaining_lengths = len(lengths) - idx
        if now >= deadline:
            break
        length_deadline = now + (deadline - now) / max(1, remaining_lengths)

        for restart in range(restarts_per_len):
            if time.time() >= deadline:
                break

            # ---- initialise candidate vector for this run ----
            if restart == 0 and n == n0:
                init_a = base.copy()
            elif restart == 0:
                # linear interpolation of the base shape onto the new grid
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, base)
                init_a = np.maximum(init_a, 0.0)
            else:
                # random positive start (log‑normal for better spread)
                init_a = np.random.lognormal(mean=0.0, sigma=1.0, size=n)
                init_a = np.maximum(init_a, 0.0)

            # normalise to the required sum √(2n)
            target_sum = math.sqrt(2.0 * n)
            init_a = init_a * (target_sum / init_a.sum())

            print(f"[run] length={n}, restart={restart+1}")

            # ---- run Adam optimisation for this length ----
            best_seq, best_score = _adam_optimize_length(
                n=n,
                init_a=init_a,
                best_seq=best_seq,
                best_score=best_score,
                deadline=length_deadline,
            )

    # --------------------------------------------------------------
    # In case nothing was found (should never happen), fall back to uniform.
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return plain python list as required by the harness.
    return best_seq.tolist()
