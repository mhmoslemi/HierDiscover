    # function to compute objective (max conv) and score
    def objective(a):
        n = a.shape[0]
        # conv via FFT
        N = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(A*A, n=N)[:2*n-1]
        return conv.max()

    # evaluate candidate (full evaluate_sequence)
    def evaluate_full(a):
        return evaluate_sequence(a.tolist())

    # record initial
    n0 = base_seq.shape[0]
    base_seq = base_seq * (math.sqrt(2*n0) / base_seq.sum())
    init_score = evaluate_full(base_seq)
    best_seq = base_seq.copy()
    best_score = init_score
    print(f"[init] length {n0} score {best_score}")

    # define lengths (including base) up to max_len
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0]
    max_len = 8000
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [min(max_len, max(300, L)) for L in lengths]
    # ensure unique
    lengths = list(dict.fromkeys(lengths))

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # generate a few starting points (including interpolation of best_seq)
        starts = []
        # 1. interpolation of global best to new length
        old_x = np.linspace(0, 1, best_seq.shape[0])
        new_x = np.linspace(0, 1, n)
        inter = np.interp(new_x, old_x, best_seq)
        inter = np.maximum(inter, 0.0)
        inter = inter * (target_sum / inter.sum())
        starts.append(inter)

        # 2. uniform
        uniform = np.full(n, target_sum / n)
        starts.append(uniform)

        # 3. few random Dirichlet draws
        for _ in range(3):
            alpha = np.random.uniform(0.1, 5.0)  # random concentration
            rand = np.random.dirichlet(np.full(n, alpha))
            rand = rand * target_sum
            starts.append(rand)

        # process each start
        for idx_start, init_a in enumerate(starts):
            if time.time() >= deadline:
                break
            print(f"[run] length {n} start {idx_start+1}/{len(starts)}")

                best_score=best_score,

            )

    final_score = evaluate_full(best_seq)

def _adam_optimize(a0, n, target_sum, best_seq, best_score, time_deadline):
    a = a0.copy()
    # Normalise
    a = a * (target_sum / a.sum())
    # Adam state
    m, v = np.zeros_like(a), np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.2   # try larger
    iteration = 0
    last_improve_iter = 0
    jitter_interval = 2000
    while time.time() < time_deadline:
        iteration += 1
        # FFT convolution
        N = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(A*A, n=N)[:2*n-1]
        M = conv.max()
        # active set
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        # compute rev subgradient (average over active)
        idx = np.arange(n)
        rev = np.zeros_like(a)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        rev /= max(1, active.size)

        coeff = 4.0 * n / (target_sum ** 2)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # gradient clipping
        max_norm = 0.1 * target_sum  # adjustable
        gnorm = np.linalg.norm(grad)
        if gnorm > max_norm:
            grad = grad * (max_norm / gnorm)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        lr = lr0 / (1 + 1e-5 * iteration)  # slower decay

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a = np.full_like(a, target_sum / n)

        # jitter
        if iteration % jitter_interval == 0:
            sigma = 0.01 * target_sum / np.sqrt(n)  # scale
            jitter = np.random.randn(n) * sigma
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # compute current score which equals M (since normalized)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best {best_score:.6f}")

        # optional early stop if stagnant
        if iteration - last_improve_iter > 200000:
            # restart with small perturbation to avoid waste
            # apply random mask to mix
            a = best_seq.copy()
            # add small noise
            a = np.maximum(a + np.random.randn(n) * 0.001 * target_sum, 0.0)
            a *= target_sum / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            iteration = 0
            last_improve_iter = 0

    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Several candidate lengths are explored (original length and a few larger
  interpolated versions, up to 8000 entries).
* For *each* length we perform a number of *independent restarts*.
  The initial vectors are generated from:
    – interpolation of the current global best,
    – a uniform vector,
    – a few Dirichlet‑distributed random vectors (different concentration
      parameters).
* Every restart runs an Adam optimiser on the exact sub‑gradient of the
  max‑convolution (the active shifts where the convolution attains its
  maximum).  The update is followed by projection onto the feasible set:
  non‑negative entries and a fixed ℓ₁‑norm  √(2·n).  The projection keeps the
  objective equal to the maximal convolution entry, i.e.  ``score = M``.
* Several stabilisation tricks are added:
    – gradient clipping,
    – a decaying learning‑rate,
    – occasional Gaussian jitter,
    – a “stagnation reset’’ that injects a small random perturbation if
      no improvement has been seen for many iterations.
* The best sequence seen over **all** runs (any length, any restart) is
  returned.  The whole search respects the 1000 s wall‑clock budget.

The implementation uses only NumPy and the supplied ``evaluate_sequence``
function; it runs completely in memory and works on a single thread
(although the harness may schedule it on up to two CPUs).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray):
    """Return (full convolution array, maximal value, index of a max)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    k_max = int(conv.argmax())
    return conv, float(conv.max()), k_max


def _subgradient(a: np.ndarray, active: np.ndarray):
    """
    Sub‑gradient of the max‑convolution.
    Returns a vector of length ``len(a)`` equal to the average of the
    reversed shift vectors for all active convolution indices.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= float(active.size)
    return rev


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    rng: np.random.Generator,
    lr0: float = 0.15,
    jitter_interval: int = 3000,
):
    """
    Run Adam on a single start vector.
    Returns the (possibly updated) global best sequence and its score.
    """
    a = a0 * (target_sum / a0.sum())          # enforce the ℓ₁‑norm

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # stagnation detection
    last_improve = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ---- convolution -------------------------------------------------
        conv, M, _ = _conv_full_fft(a)

        # ---- active set --------------------------------------------------
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        rev = _subgradient(a, active)

        # ---- sub‑gradient ------------------------------------------------
        coeff = 4.0 * n / (target_sum ** 2)          # 4n / S²
        const = 4.0 * n * M / (target_sum ** 3)      # 4nM / S³
        grad = coeff * rev - const

        # ---- gradient clipping -------------------------------------------
        max_norm = 0.12 * target_sum          # empirical; keeps steps sane
        gnorm = np.linalg.norm(grad)
        if gnorm > max_norm:
            grad = grad * (max_norm / gnorm)

        # ---- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-5 * iteration)   # gentle decay

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto feasible set --------------------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s
        else:                                   # practically never happens
            a.fill(target_sum / n)

        # ---- occasional jitter -------------------------------------------
        if iteration % jitter_interval == 0:
            sigma = 0.015 * target_sum / math.sqrt(n)
            jitter = rng.normal(0.0, sigma, size=n)
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # ---- record improvement -------------------------------------------
        cur_score = M                               # objective == M (normalised)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = iteration
            print(f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}")

        # ---- stagnation handling -----------------------------------------
        if iteration - last_improve > 300_000:
            # inject a modest random perturbation and reset moments
            a = best_seq.copy()
            sigma = 0.02 * target_sum / math.sqrt(n)
            a = np.maximum(a + rng.normal(0.0, sigma, size=n), 0.0)
            a *= target_sum / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration

    return best_seq, best_score

    """

    start_time = time.time()
    # leave a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------

    base = base * (target0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length = {n0}, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) choose a handful of lengths (≤ 8000) to explore
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0]   # enlarge the search space
    max_len = 8000
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [min(max_len, max(300, L)) for L in lengths]
    # ensure uniqueness while preserving order
    lengths = list(dict.fromkeys(lengths))

    # ------------------------------------------------------------------
    # 3) optimisation over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- generate a pool of starting points for this length ----------
        starts = []

        # a) interpolate the current global best
        old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        interp = np.interp(new_x, old_x, best_seq)
        interp = np.maximum(interp, 0.0)
        interp *= target_sum / interp.sum()
        starts.append(interp)

        # b) uniform vector (baseline)
        starts.append(np.full(n, target_sum / n))

        # c) a few random Dirichlet draws with varying concentration
        for _ in range(4):
            alpha = rng.uniform(0.1, 5.0)               # random concentration
            rand = rng.dirichlet(np.full(n, alpha))
            rand *= target_sum
            starts.append(rand)

        # ------------------------------------------------------------------
        # run an Adam optimiser for each start vector
        for idx, init_a in enumerate(starts):
            if time.time() >= deadline:
                break
            print(f"[run] length={n:5d} start {idx + 1}/{len(starts)}")

            # randomise learning‑rate a little for diversity
            lr0 = rng.uniform(0.08, 0.22)
            jitter_int = rng.integers(2000, 5000)

            best_seq, best_score = _adam_optimize(
                a0=init_a,
                n=n,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                rng=rng,
                lr0=lr0,
                jitter_interval=jitter_int,
            )

    # ------------------------------------------------------------------
    # 4) final evaluation with the official function (exact, includes
    #    clipping and the 2·n factor – now we know the objective equals
    #    the maximum convolution because of the normalisation).
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] final score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
