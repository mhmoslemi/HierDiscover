def adam_optimize(n, a0, best_seq, best_score, deadline):
    a = a0.copy()
    target_sum = math.sqrt(2.0 * n)
    # Ensure normalization (should be already)
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0=0.2
    iteration=0
    jitter_interval=2000
    while time.time() < deadline:
        M, _, conv = _conv_max_fft(a)
        # Determine active set (max - tolerance) to include near ties
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        rev = _subgradient(a, active)
        coeff = 4.0 * n / (target_sum**2)
        const = 4.0 * n * M / (target_sum**3)
        grad = coeff * rev - const
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0 / (1.0 + 5e-5 * iteration)  # slower decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.005 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        # compute score (equal to M)
        cur_score = M
        if cur_score < best_score:

"""
Propose a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
* Load the previously discovered construction (``height_sequence_1``) if it
  exists, otherwise start from a uniform vector.
* Consider several lengths (the original length and several scaled versions
  up to a configurable maximum, e.g. 8‑12 k entries).  More degrees of
  freedom give a lower theoretical bound.
* For each length perform a few *restarts*:
    - initialise the vector by interpolating the base shape and adding a
      modest random jitter;
    - run a projected Adam optimiser that minimises the maximal
      convolution value M = max (a*a).  The sub‑gradient for the max‑
      convolution is computed exactly (average of contributions from all
      active shifts).
    - after each update the vector is projected onto the feasible set
      (non‑negative entries, fixed ℓ₁‑norm √(2n)).
    - occasional Gaussian jitter helps escape plateaus.
* Keep the best sequence (smallest M) seen so far.
* After the length‑loops perform a cheap stochastic hill‑climbing
  refinement: repeatedly move a tiny amount of mass from a random index
  to another and keep the change if the maximal convolution decreases.
* The whole procedure respects the 1000 s wall‑clock budget; any
  remaining time is used for the final local search.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full convolution a * a via FFT and return (max value, full array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A                                     # element‑wise product
    conv = np.fft.irfft(B, n=N)
    conv = conv[: 2 * n - 1]                     # keep the valid part
    M = float(conv.max())
    return M, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of the max‑convolution for a set of active shift indices.
    Returns a vector of length len(a) equal to the average reversed slice.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active_idxs.size)   # average when several shifts tie
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    a0: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
):
    """
    Run Adam optimisation on a vector of length ``n`` starting from ``a0``.
    Updates and returns the global best sequence/score.
    """
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * (target_sum / a.sum())          # enforce the normalisation

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2                               # a slightly larger base LR
    iteration = 0
    jitter_interval = 2000                  # more frequent jitter

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution, maximal value and active set
        M, conv_full = _conv_max_fft(a)
        # treat all indices numerically equal to the max as active
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        # --------------------------------------------------------------
        # 2) sub‑gradient of the objective (objective == M after normalisation)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # --------------------------------------------------------------
        # 3) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of the learning rate
        lr = lr0 / (1.0 + 5e-5 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 4) projection onto feasible set (non‑negative + normalisation)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            # Extremely unlikely – fallback to uniform vector
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 5) occasional jitter to break plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.005 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 6) evaluate (objective == M after the normalisation)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, best_score: float, deadline: float,
                max_trials: int = 50000):
    """
    Simple stochastic hill‑climbing: repeatedly move a tiny amount of mass
    from one index to another and keep the change if the maximal
    convolution decreases.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = a.sum()

    for trial in range(max_trials):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 1e-12:
            continue

        # move a random fraction (up to 30 %) of the mass from i to j
        frac = np.random.rand() * 0.3
        delta = frac * a[i]

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # compute new maximal convolution
        M_new, _ = _conv_max_fft(a_new)

        if M_new < best_score - 1e-9:
            a = a_new
            best_score = M_new
            best_seq = a.copy()
            print(f"[hill] trial {trial:5d} new best = {best_score:.6f}")

    return a, best_score
