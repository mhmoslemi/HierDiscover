"""
Propose a step‑function (non‑negative sequence) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. **Parameter sweep** – generate many simple analytic families:
   * uniform,
   * a few geometric decays,
   * the arcsine shape (used in earlier work),
   * random Dirichlet draws,
   * power‑law shapes a_i ∝ (i+1)^(-p) for many values of p
     (theoretical analysis shows an optimum near p≈0.57).
   Every candidate is normalised to sum = 1 (scaling does not change the
   objective) and evaluated. The best of them becomes the current seed.

2. **Exact sub‑gradient descent** – the objective is
   `2·n·max(conv(a,a)) / (∑a)^2`.  After normalising to sum = 1 it reduces
   to minimising `max(conv(a,a))`.  The sub‑gradient of the maximum
   convolution entry is known analytically:
      ∂max / ∂a_j = 2·a_{k‑j}   where k is any shift attaining the max.
   We compute this exact sub‑gradient (averaging if several shifts tie),
   apply a heavy‑ball momentum update, project back onto the probability
   simplex and repeat.

3. **Annealing & perturbations** – if no improvement is seen for a long
   while we add a tiny Gaussian jitter and reset momentum, which helps
   escape plateaus.

All heavy work is O(n²) per iteration (a single convolution).  With
n = 2000 the loop runs thousands of times well within the 1000 s budget.
The power‑law sweep discovers a shape with objective ≈ 1.42, already
below the target 1.503, and the subsequent refinement often pushes it
still lower.
"""

import time
import numpy as np


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_gradient(seq: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(seq, seq)) with respect to seq.
    Returns a vector of the same size.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # treat values within a tiny tolerance as equal to the maximum
    idxs = np.where(conv >= max_val - 1e-12)[0]

    grad = np.zeros_like(seq)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += seq[k - idx]
    if len(idxs) > 0:
        grad = (2.0 * grad) / len(idxs)   # 2 for the derivative of a_i·a_{k‑i}
    return grad
