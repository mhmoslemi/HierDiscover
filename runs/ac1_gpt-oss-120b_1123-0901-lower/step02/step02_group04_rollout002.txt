Propose a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The strategy is a multi‑phase black‑box optimiser:

1. **Broad parametric sweep** – exponential, linear, quadratic, triangular,
   sinusoidal and Hann windows for several lengths.  Every candidate is
   projected onto the simplex Σa_i = √(2 n) and evaluated.

2. **Smooth‑max Adam optimisation** – the non‑smooth objective
   `max(conv)` is approximated by a soft‑max
   `Lβ(a)= (1/β)·log Σ exp(β·conv_k)`.  Its gradient can be written as a
   single convolution, so each Adam step costs only two FFTs.
   The temperature β is annealed from a small value (≈ 1) to a large value
   (≈ 30) so that the optimisation first moves in a smooth landscape and
   finally refines the true max‑objective.

# ----------------------------------------------------------------------
# Simplex projection – Σx = s,  x ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b by FFT (real input)."""
    n = a.shape[0]
    m = b.shape[0]
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n + m - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute normalized soft‑max weights w_k = exp(β·(conv_k‑max))/Σexp.
    The subtraction of max prevents overflow.
    """
    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Lβ(a) = (1/β)·log Σ exp(β·conv_k)  w.r.t a.
    It is the cross‑correlation of a with the weight vector.
    Implements g_i = Σ_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    # reverse the weight vector for correlation → convolution
    w_rev = weights[::-1]
    # full convolution w_rev * a, then take the central slice
    conv = _conv_full(w_rev, a)                     # length = 3·n‑2
    offset = 2 * n - 2
    grad = conv[offset: offset + n]
    return grad

def _adam_optimize(init_vec: np.ndarray,
                   best_holder: dict,
                   deadline: float,
                   start_beta: float = 1.0,
                   end_beta: float = 30.0,
                   max_iters: int = 200_000,
                   eval_every: int = 10,
                   lr: float = 0.5) -> None:
    """
    Run Adam optimisation from ``init_vec`` (already on the simplex).
    The function updates ``best_holder`` whenever a better sequence is found.
    """
    a = init_vec.astype(float).copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta_ratio = end_beta / start_beta if start_beta > 0 else 1.0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and smooth‑max weights
        conv = _conv_self(a)
        # anneal β exponentially (log‑linear schedule)
        beta = start_beta * (beta_ratio ** (it / max_iters))
        w = _smooth_weights(conv, beta)

        # gradient of the smooth‑max loss
        grad = _gradient_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # project back onto the simplex (keeps sum = target and non‑negativity)
        a = _project_onto_simplex(a, target)

        if it % eval_every == 0:
            score = _evaluate(a.tolist())
            if score < best_holder["score"]:
                best_holder["score"] = score
                best_holder["seq"] = a.copy()
                print(f"[{time.time():.2f}] Adam iter {it} – new best = {score:.6f} (n={n})")
                if score < 1.503:
                    return  # target reached – stop early

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a small safety margin

    # holder for the global best
    best = {"seq": None, "score": float("inf")}

    # ------------------------------------------------------------------
    # 0.  Take the previous SOTA construction if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        a0 = np.asarray(prev, dtype=float)
        if a0.ndim == 1 and a0.size > 0:
            n0 = a0.shape[0]
            target0 = math.sqrt(2.0 * n0)
            a0 = _project_onto_simplex(a0, target0)
            _maybe_update(a0, best)

    # ------------------------------------------------------------------
    # 1.  Parametric sweep over several families (cheap)
    # ------------------------------------------------------------------
    lengths = list(range(200, 1801, 200)) + [1326, 1500, 1700]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-4, -0.2, num=14):
            a = _shape_exp(n, alpha)
            a = a / a.sum() * target
            _maybe_update(a, best)

        # reversed exponential
        for alpha in np.logspace(-4, -0.2, num=14):
            a = _shape_exp_rev(n, alpha)
            a = a / a.sum() * target
            _maybe_update(a, best)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=12):
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a, best)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=12):
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a, best)

        # triangular
        a = _shape_triangular(n)
        a = a / a.sum() * target
        _maybe_update(a, best)

        # sine‑squared
        a = _shape_sine_sq(n)
        a = a / a.sum() * target
        _maybe_update(a, best)

        # Hann window
        a = _shape_hann(n)
        a = a / a.sum() * target
        _maybe_update(a, best)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2.  Adam optimisation from the current best (if any)
    # ------------------------------------------------------------------
    if best["seq"] is not None:
        _adam_optimize(best["seq"], best, deadline,
                       start_beta=1.0, end_beta=30.0,
                       max_iters=120_000, eval_every=8, lr=0.4)

    # ------------------------------------------------------------------
    # 3.  Random restarts with Adam optimisation
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length that keeps FFT cheap (≤ 3000)
        n = np.random.randint(300, 2501)
        target = math.sqrt(2.0 * n)

        # 3a – dense Dirichlet start
        a_dense = _rand_dirichlet_start(n, target)
        _adam_optimize(a_dense, best, deadline,
                       start_beta=1.0, end_beta=30.0,
                       max_iters=80_000, eval_every=10, lr=0.5)

        if time.time() >= deadline:
            break

        # 3b – sparse spikes start (helps explore extreme shapes)
        a_sparse = _rand_sparse_start(n, target, k=np.random.randint(3, 10))
        _adam_optimize(a_sparse, best, deadline,
                       start_beta=0.5, end_beta=20.0,
                       max_iters=60_000, eval_every=12, lr=0.6)

        if best["score"] < 1.503:
            break

    # ------------------------------------------------------------------
    # 4.  Optional linear‑programming refinement (SciPy, if available)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    if linprog is not None and best["seq"] is not None and time.time() < deadline:
        a = best["seq"].copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        # Pre‑build the convolution‑matrix structure only once (size 2n‑1 × n)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]

        # LP loop – a few updates
        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()

            res = linprog(c=-np.ones(n),
                          A_ub=A,
                          b_ub=np.full(rows, M),
                          bounds=(0, None),
                          method="highs",
                          options={"presolve": True})
            if not res.success:
                break

            g = res.x
            s = g.sum()
            if s <= 0:
                break
            g = g * (target / s)   # rescale to the simplex

            # convex blend – keep steps modest to stay stable
            t = 0.2
            a = (1 - t) * a + t * g
            # Update the matrix A for the new a (very cheap)
            for k in range(rows):
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                cols = np.arange(low, high + 1, dtype=int)
                A[k, cols] = a[k - cols]

            _maybe_update(a, best)

            if best["score"] < 1.503:
                break
            if time.time() + 0.5 >= deadline:
                break

    # ------------------------------------------------------------------
    # 5.  Fallback – uniform distribution if nothing succeeded
    # ------------------------------------------------------------------
    if best["seq"] is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best["seq"] = np.full(n, target / n, dtype=float)

    return best["seq"].tolist()
