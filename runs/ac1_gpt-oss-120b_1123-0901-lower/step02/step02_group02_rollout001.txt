"""
Propose a low‑value step‑function.

Algorithm
---------
1. **Simplex projection** – we always keep the vector on the
   simplex  Σ a_i = √(2·n) , a_i ≥ 0.  This makes the objective equal
   to max(conv(a,a)), which is easier to monitor.

2. **FFT based convolution** – full convolution a*a and the
   gradient w.r.t. the “active” entries are computed with a single
   real FFT (O(n log n)).

3. **Projected (Nesterov‑)gradient descent** – each iteration
   * finds the set of indices whose convolution value is within a
     relative tolerance of the current maximum,
   * builds a (sub‑)gradient of the weighted sum of those entries,
   * updates with momentum,
   * projects back onto the simplex,
   * occasionally adds a tiny random shake‑up to escape plateaus.

   The tolerance starts a little larger and is multiplied by 0.999
   each iteration so that the optimisation first moves quickly and
   later refines the solution.

4. **Coarse‑to‑fine schedule** – we start with a moderate length,
   optimise it, then repeatedly **upsample** the current best vector
   (linear interpolation) to a larger length and continue optimisation.
   Random restarts with different power‑law exponents are also performed
   while time permits.

All heavy work is vectorised; the total running time is limited by
the supplied `budget_s` (≈ 1000 s).  The routine returns the best sequence
found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex of radius `s` (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – full convolution via real FFT (a * a)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of `a` with itself using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # eliminate tiny numerical negatives
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper 3 – active set (indices whose conv value is near the max)
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float) -> tuple[float, np.ndarray]:
    """Return (max_value, indices) where conv >= max - rel_tol * max."""
    max_val = conv.max()
    # avoid empty set for extremely tiny maxima
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# Helper 4 – gradient of weighted sum of active convolution entries
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient constructed from the active convolution entries."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        # indices i that contribute to conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        # ∂/∂a_i Σ a_i a_{k-i} = a_{k-i}
        grad[idx] += 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# Core optimisation for a fixed length n
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    rel_tol_start: float = 5e-3,
    rel_tol_end: float = 5e-4,
    base_step: float = 0.14,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov) gradient descent for a given length.
    Returns the best vector found and the associated score.
    """
    target_sum = math.sqrt(2.0 * n)               # Σ a_i = √(2n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best local solution
    conv = _conv_fft(a)
    max_val, _ = _active_indices(conv, rel_tol_start)
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    # momentum buffer
    velocity = np.zeros_like(a)

    # tolerance schedule (exponential decay)
    rel_tol = rel_tol_start
    decay_factor = 0.9995

    step_factor = base_step
    iteration = 0

    while time.time() < run_deadline:
        # ------------------------------------------------------------------
        # 1. convolution and active set
        # ------------------------------------------------------------------
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        # ------------------------------------------------------------------
        # 2. gradient (sub‑gradient of weighted active sum)
        # ------------------------------------------------------------------
        grad = _gradient(a, active)

        # ------------------------------------------------------------------
        # 3. Nesterov‑like momentum update
        # ------------------------------------------------------------------
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        # simple step‑size normalisation
        max_grad = np.max(velocity) if np.max(velocity) > 0 else 1.0
        eta = step_factor * max_val / max_grad

        # Nesterov look‑ahead point
        a_look = a - eta * velocity
        a_new = _project_onto_simplex(a_look, target_sum)

        # ------------------------------------------------------------------
        # 4. occasional random shake‑up (helps escaping flat regions)
        # ------------------------------------------------------------------
        if np.random.rand() < 0.025:
            noise = np.random.randn(n) * (target_sum * 1e-3 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.93   # become a little more cautious after a shake

        # ------------------------------------------------------------------
        # 5. evaluate new candidate
        # ------------------------------------------------------------------
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        # ------------------------------------------------------------------
        # 6. keep best local solution
        # ------------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score:.6f}")

        a = a_new
        # --------------------------------------------------------------
        # 7. schedule updates
        # --------------------------------------------------------------
        rel_tol = max(rel_tol_end, rel_tol * decay_factor)
        step_factor *= 0.9998
        iteration += 1

    return best_local, best_score


# ----------------------------------------------------------------------
# Upsample (or truncate) a sequence to a new length using linear interpolation
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `seq` to length `new_n` by linear interpolation (or truncation)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        return seq[:new_n].copy()
    # interpolation
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """

    The routine respects the given time budget (seconds) and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Coarse‑to‑fine schedule: progressively larger lengths
    # ------------------------------------------------------------------
    length_schedule = [1024, 2048, 4096, 8192, 16384]   # will be clipped by time
    for n in length_schedule:

        remaining = deadline - time.time()

        if best_seq is not None:

        else:

            p = np.random.uniform(0.45, 0.55)
            idx = np.arange(1, n + 1, dtype=np.float64)

        seq_local, score_local = _optimise_one_length(
