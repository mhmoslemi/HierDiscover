def _project_onto_simplex(v):
    """Projection onto probability simplex."""
    # copy
    v = np.asarray(v, dtype=np.float64)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        # all entries are <= 0
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that minimizes

    F(a) = 2·n·max_k (a * a)[k] / (∑a_i)²

subject to a_i ≥ 0 and ∑a_i = 1 (scale‑invariant).

Algorithm
---------

* **Initial pool** – a few geometric decays and a handful of
  Dirichlet samples give diverse starting points.
* **Projected Adam** – we minimise a smooth surrogate of the max,
  namely the log‑sum‑exp of the self‑convolution.  The gradient of the
  surrogate is obtained by a single FFT‑based cross‑correlation.
  Adam’s adaptive steps are applied, and after each step the vector
  is projected back onto the probability simplex.
* **Continuation on λ** – the soft‑max temperature λ starts small
  (smooth) and is increased multiplicatively each iteration up to a
  cap (λ≈50).  This gradually turns the surrogate into the true max.
* **Stagnation handling** – if no improvement is seen for a few
  seconds we add a tiny random perturbation and reset Adam’s moment
  estimates, helping the search escape shallow plateaus.
* **Time‑budget aware** – the routine runs until the supplied time
  budget (≈1000 s) expires, keeping the best sequence ever seen.

The whole optimisation runs in O(n log n) per iteration (FFT) and
fits comfortably within the time limit for n≈3000‑4000.  The returned
list satisfies the strict checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (scale‑invariant)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 in our workflow
    return float(2.0 * n * max_b / (sum_a * sum_a))

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length that gives a good trade‑off between expressivity
    #    and speed.  3000 works well on a typical 2‑core machine.
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # 2. Build a modest pool of diverse starting points and keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # geometric decays
    for r in np.linspace(0.92, 0.998, 12):
        seq = np.power(r, np.arange(n, dtype=np.float64))
        seq /= seq.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom r={r:.3f}] obj={val:.6f}")

    # few random Dirichlet draws
    for _ in range(10):
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init dir] obj={val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation core – projected Adam on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    size_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))  # for w ★ a

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12          # base learning rate
    beta1_pow = beta1
    beta2_pow = beta2

    # λ (soft‑max temperature) schedule
    lam = 5.0
    lam_inc = 1.015
    lam_max = 45.0

    # state
    seq = best_seq.copy()
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    it = 0
    last_improve = time.time()

    # small function to recompute objective only when needed
    def _eval(s):
        return _objective(s)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # a) self‑convolution
        conv = _full_convolution(seq, size=size_conv)

        # b) soft‑max weights (smooth max)
        cmax = conv.max()
        # exp(lam*(conv-cmax)) is numerically safe (≤1)
        w = np.exp(lam * (conv - cmax))
        w_sum = w.sum()
        if w_sum == 0:   # numerical safety (should not happen)
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # c) gradient = 2·(w ★ a)
        rev_seq = seq[::-1]
        fw = np.fft.rfft(w, size_corr)
        frev = np.fft.rfft(rev_seq, size_corr)
        corr = np.fft.irfft(fw * frev, size_corr)
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # d) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        seq_new = seq - step
        seq_new = _project_onto_simplex(seq_new)

        # e) evaluate exact objective
        val = _eval(seq_new)

        # f) keep the best ever seen
        if val < best_val:
            best_val = val
            best_seq = seq_new.copy()
            last_improve = time.time()
            print(f"[Iter {it:06d}] new best obj={val:.6f}")

            # early exit if target is already beaten
            if best_val < 1.503:
                print("Target reached – exiting early.")
                break

        # g) stagnation handling – random shake and Adam reset
        if time.time() - last_improve > 6.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            seq_new = _project_onto_simplex(seq_new + noise)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta1
            beta2_pow = beta2
            last_improve = time.time()
            print(f"[Iter {it:06d}] shake triggered")

        # h) prepare for next iteration
        seq = seq_new
        lam = min(lam_max, lam * lam_inc)
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
