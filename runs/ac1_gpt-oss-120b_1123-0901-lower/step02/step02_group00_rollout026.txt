            p_new = np.maximum(p_new,0)
            # renormalize to sum 1
            p_new = p_new / p_new.sum()
        else:
            S = math.sqrt(2.0 * n)
            p_new = best_seq / S

        # ensure p_new sum=1

        # run EG for this length
        best_p, best_score_n = _optimize_eg_length(n, p_new, best_score, deadline)
        # convert p to a
        S_new = math.sqrt(2.0 * n)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[len {n}] improved global best score={best_score:.6f}")

"""
Improved search for a low‑peak self‑convolution step function.

Algorithm
---------
* Represent the vector a as a probability vector p with Σp = 1 and
  a = S·p where S = √(2·n).  The objective becomes  2·n·max_k (p∗p)_k .
* Optimise p by Exponentiated‑Gradient (mirror descent on the simplex).
  For a given p we compute the full auto‑convolution conv = p∗p
  (FFT, O(n log n)) and the maximal value M.
* Instead of a hard max we use a soft‑max weight
      w_k = exp(β·(conv_k−M)) , β growing from a small to a moderate value.
  The sub‑gradient of the smoothed maximum is
      rev_i = Σ_k w_k·p_{k−i}   (a correlation, also obtained by FFT).
* The EG update is performed on log‑probabilities:
      logp ← logp − η·(4 n·rev)       (η is a decreasing step‑size).
  Afterwards logp is normalised back to a probability vector.
* A multiscale schedule is employed: start with a modest length,
  optimise, then up‑sample the current best solution to a larger length
  and continue optimising.  Lengths up to 8000 are explored.
* Jitter (small Gaussian noise on logp) is added periodically to
  escape plateaus.  The best sequence found over all lengths and
  iterations is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Compute the full self‑convolution (p∗p) using a single FFT of size N.
    Returns the real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i} for i=0..n−1.
    This is a correlation of p with w and is obtained via FFT.
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    # FFT of p (size N already chosen ≥ n+L−1)
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weights
    W_rev = np.fft.rfft(w[::-1], n=N)
    # convolution of p with reversed w
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep the linear part
    # Extract rev = conv_rev[L‑n : L] and reverse
    rev = conv_rev[L - n : L][::-1]
    return rev


def _optimize_eg_length(
    n: int,
    p_start: np.ndarray,
    best_score_global: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector ``p`` (sum = 1) and its score
    (2·n·max(p∗p)).
    """
    # parameters
    max_iters = 200_000                     # safety cap
    jitter_interval = 3000
    jitter_scale = 0.001
    beta_start = 0.1
    beta_end = 10.0
    # linear schedule for beta across the first 80 k iterations
    beta_len = 80_000

    p = p_start.copy()
    logp = np.log(p + 1e-30)                # keep in log‑space

    # FFT size that works for both the self‑convolution (2n‑1) and the
    # correlation with the weight vector (n+2n‑2 = 3n‑2)
    N = _next_pow2(3 * n - 2)

    best_local_score = float("inf")
    best_local_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution (p∗p) -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()

        # current objective (no need to call evaluate_sequence each time)
        cur_score = 2.0 * n * M

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- soft‑max weights for a smoother sub‑gradient -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))          # w_k ≤ 1, w_max = 1

        # ----- weighted rev (correlation) -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smoothed max -----
        grad = 4.0 * n * rev                    # ∇_p (2·n·M_β)

        # ----- step size (η decays) -----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # ----- EG update in log‑space -----
        logp -= eta * grad

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)      # avoid overflow
        p = exp_shift / exp_shift.sum()

    return best_local_p, best_local_score
