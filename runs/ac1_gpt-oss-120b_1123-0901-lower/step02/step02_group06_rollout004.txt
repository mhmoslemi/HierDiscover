"""
Search for a non‑negative step‑function (a probability vector) that
minimises ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Broad initialisation** – a pool of candidate distributions is
   generated (uniform, geometric, arcsine, a dense grid of Beta‑shaped
   densities and a few random Dirichlet draws).  The best of these
   becomes the starting point.

2. **Exact sub‑gradient descent on the maximal autocorrelation entry**.
   For a sequence `a` the objective (when ∑a = 1) is
   `obj = 2·n·max(a*a)`.  The sub‑gradient of `max(a*a)` is obtained
   by summing the contributions of the convolution indices that attain
   the maximum.  A heavy‑ball momentum term is used, an adaptive step‑
   size is increased after a successful step and halved after a failure.
   The iterate is projected back onto the probability simplex after each
   update.

3. **Late‑stage random perturbations** – if progress stalls for many
   iterations a small Gaussian jitter is added to escape the basin,
   after which the descent continues.

All heavy work (convolution, gradient, projection) is performed with
NumPy; no external optimisation packages are required.  The routine
stops when the allocated wall‑time (`budget_s`) is exhausted and returns
the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector ``v`` onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_candidate(n: int, a: float, b: float) -> np.ndarray:
    """
    Return a discretised Beta(a,b) density on ``n`` points, normalised
    to sum to one.  The points are taken at the centres of the bins.
    """
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    # avoid exact zeros/ones – the shift by 0.5 guarantees 0 < x < 1
    seq = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    # numerical problems when a or b are very small can yield infinities;
    # replace them by a large finite number before normalising.
    seq[np.isinf(seq)] = np.max(seq[np.isfinite(seq)]) * 1e6
    seq = seq / seq.sum()
    return seq

    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # 3. arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. a grid of Beta‑shaped densities
    alphas = [0.2, 0.3, 0.4, 0.5, 0.8, 1.0, 1.5]
    betas  = [0.2, 0.3, 0.4, 0.5, 0.8, 1.0, 1.5]
    for a in alphas:
        for b in betas:
            # Skip the case (0.5,0.5) which is already the arcsine shape.
            if a == 0.5 and b == 0.5:
                continue
            seq = _beta_candidate(n, a, b)
            cand.append(seq)

    # 5. a few random Dirichlet draws (parameter 1 → uniform Dirichlet)
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Internal objective implemented directly: for a normalised
    sequence (Σa = 1) it equals ``2·n·max(conv(a,a))``.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _gradient_max(seq: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max(conv(seq,seq))`` w.r.t. ``seq``.
    The gradient of the convolution entry `k` is 2·seq[k‑i] for each
    overlapping index `i`.  Only the indices attaining the maximal
    value contribute.
    """
    n = len(seq)
    max_val = conv.max()
    # use a tight tolerance to collect all (numerically) maximal entries
    mask = np.isclose(conv, max_val, atol=1e-12, rtol=0.0)
    max_idx = np.flatnonzero(mask)

    grad = np.zeros_like(seq)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * seq[k - idx]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑valued step‑function within ``budget_s`` seconds.
    Returns a list of non‑negative heights (they sum to 1 by construction).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. problem size
    n = 2000

    # ------------------------------------------------------------------
    # 2. generate initial pool (including a possible external solution)
    candidates = _initial_candidates(n, rng)

    # If the notebook already supplied a previous construction, use it too.
    # ``height_sequence_1`` – if present – is assumed to be a Python list.
    if 'height_sequence_1' in globals():
        external = np.asarray(height_sequence_1, dtype=np.float64)
        external = np.clip(external, 0.0, None)
        if external.sum() < 0.01:               # guard against an empty seed
            external = np.full_like(external, 1.0 / external.size)
        else:
            external = external / external.sum()
        # Adjust length to ``n`` (truncate or pad with zeros)
        if external.size > n:
            external = external[:n]
        elif external.size < n:
            external = np.concatenate([external,
                                       np.zeros(n - external.size,
                                                dtype=np.float64)])
        candidates.append(external)

    # ------------------------------------------------------------------
    # 3. pick the best start from the pool
    best_seq = None
    best_score = float('inf')
    for seq in candidates:
        seq = np.clip(seq, 0.0, None)
        if seq.sum() == 0.0:
            continue
        seq = seq / seq.sum()
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best objective from seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. exact sub‑gradient descent (heavy‑ball momentum)
    seq = best_seq.copy()
    momentum = np.zeros_like(seq)
    beta_m = 0.9               # momentum factor
    step = 0.5                 # initial step size
    it = 0
    no_improve = 0

    while time.time() < deadline:
        conv = np.convolve(seq, seq)          # full self‑convolution
        grad = _gradient_max(seq, conv)       # exact sub‑gradient of max

        # heavy‑ball update
        momentum = beta_m * momentum + (1.0 - beta_m) * grad

        # propose a new point, then project onto the simplex
        cand = seq - step * momentum
        cand = _project_onto_simplex(cand)

        cand_score = _objective(cand)

        if cand_score < best_score - 1e-12:
            # improvement – accept, enlarge step a little
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand
            step *= 1.02
            no_improve = 0
            print(f"[Iter {it:07d}] new best = {best_score:.6f}")
        else:
            # no improvement – shrink step and stay at previous point
            step *= 0.5
            no_improve += 1

        # ------------------------------------------------------------------
        # 5. occasional random jitter to escape flat plateaus
        if no_improve > 30000:
            jitter = rng.normal(scale=0.01, size=n)
            seq = seq + jitter
            seq = np.clip(seq, 0.0, None)
            seq = seq / seq.sum()
            momentum.fill(0.0)
            step = 0.5
            no_improve = 0
            print(f"[Iter {it:07d}] stagnation – random jitter applied")

        it += 1

    # ------------------------------------------------------------------
    # 6. a quick random‑local‑search phase with any remaining time
    while time.time() < deadline:
        # move a tiny random amount between two random coordinates
        cand = best_seq.copy()
        i, j = rng.choice(n, 2, replace=False)
        eps = min(cand[i], rng.random() * 0.02)   # at most 2 % of i’s mass
        if eps > 0:
            cand[i] -= eps
            cand[j] += eps
        cand = _project_onto_simplex(cand)
        cand_score = _objective(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[Random] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"Finished. Best objective = {final_obj:.6f} (length {n})")
    # final sanity‑check with the official evaluator (optional)
    # eval_check = evaluate_sequence(best_seq.tolist())
    # print(f"Evaluator check = {eval_check}")

    return best_seq.tolist()
