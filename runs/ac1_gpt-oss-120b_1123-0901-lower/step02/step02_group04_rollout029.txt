def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    def _update_best(a):
        nonlocal best_seq, best_score
        score = _score(a)  # quick internal
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # use external evaluate for safety (like rounding)
            external = _evaluate_score(a.tolist())
            # if external better (maybe due to clipping) use that:
            if external < best_score:
                best_score = external
                best_seq = a.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={len(a)})")
        return

    # Phase 1: parametric families
    lengths = list(range(200, 2401, 200))
    # add random lengths around 1200
    lengths += [1326, 1500, 1800]
    for n in lengths:
        target_sum = 1.0
        # Exponential
        for alpha in np.logspace(-4, -0.5, num=12):
            a = np.exp(-alpha * np.arange(n, dtype=float))
            a = a / a.sum() * target_sum
            _update_best(a)
        # Inverse power
        for p in np.linspace(0.5, 3.0, num=10):
            i = np.arange(1, n+1, dtype=float)
            a = 1.0 / (i**p)
            a = a / a.sum() * target_sum
            _update_best(a)
        # Linear decreasing
        for beta in np.linspace(0.0, 2.0/(n-1) if n>1 else 0.0, num=8):
            i = np.arange(n, dtype=float)
            a = np.maximum(0., 1. - beta * i)
            if a.sum() == 0: continue
            a = a / a.sum() * target_sum
            _update_best(a)
        # Quadratic
        for beta in np.linspace(0., 1.0/(n**2), num=8):
            i = np.arange(n, dtype=float)
            a = np.maximum(0., 1. - beta * i**2)
            if a.sum() == 0: continue
            a = a / a.sum() * target_sum
            _update_best(a)
        # Triangular
        a = np.minimum(np.arange(n, dtype=float)+1., n - np.arange(n, dtype=float))
        a = a / a.sum() * target_sum
        _update_best(a)

        if time.time() >= deadline:
            break

    # Phase 2: smoothed gradient descent
    # Start from best or from some param families if best_seq None
    if best_seq is None:
        # fallback initialization random
        n = np.random.randint(500, 1500)
        a = np.random.rand(n)
        a = a / a.sum()
        best_seq = a
        best_score = _score(a)
    else:
        a = best_seq.copy()
        n = a.shape[0]

    # schedule of beta values
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    for beta in beta_schedule:
        # inner loops
        while time.time() < deadline:
            conv, grad = _smooth_grad(a, beta)
            t_iter += 1
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t_iter)
            v_hat = v / (1 - beta2**t_iter)
            step = 0.1 / (np.sqrt(v_hat) + eps)  # base step 0.1
            a = a - step * m_hat
            # projection onto simplex sum=1
            a = _project_onto_simplex(a, 1.0)
            # evaluate occasionally
            if t_iter % 5 == 0:
                _update_best(a)
            # termination condition within inner loop
            if best_score < 1.45:  # early stop
                break
        if best_score < 1.45:
            break

    # Phase 3: local random perturbations (simulated annealing)
    # Use current best as starting point
    a = best_seq.copy()
    n = a.shape[0]
    temp = 1.0
    while time.time() < deadline:
        # propose perturb
        i,j = np.random.choice(n, size=2, replace=False)
        eps_move = np.random.rand() * 0.01  # small fraction of total sum (since sum=1)
        if a[i] < eps_move:
            continue
        a_new = a.copy()
        a_new[i] -= eps_move
        a_new[j] += eps_move
        # no need to project, sum unchanged, nonnegativity ensured
        new_score = _score(a_new)
        cur_score = _score(a)
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / temp):
            a = a_new
            _update_best(a)
        # cool temperature
        temp *= 0.9999
        # stop if temperature too low
        if temp < 1e-6:
            break

"""
propose_candidate
-----------------

A three‑phase optimizer for the “max‑convolution” bound

The routine works completely offline (no I/O) and respects the given
time‑budget.  It consists of

1.  A cheap sweep over several parameterised families (exponential,
    inverse‑power, linear, quadratic, triangular).  Every candidate is
    normalised onto the probability simplex (Σa_i = 1) and the cheap
    analytic score 2·n·max(conv)/Σa² is evaluated.  The best candidate
    so far is kept.

2.  A continuation method on a smooth approximation of the
    ``max``‑operator (log‑sum‑exp).  For a fixed smoothing parameter β we
    compute the gradient of  f_β(a)=log‑sum‑exp(β·(a*a))  by means of
    FFT‑based correlation, then perform a projected Adam step on the
    simplex.  β is increased (0.5 → 20) so that the optimisation first
    follows a smooth landscape and later focuses on the true maximum.
    The current best ever sequence is updated whenever an improvement
    is observed.

3.  A few rounds of random “mass‑transfer’’ moves (simulated‑annealing
    style) that can escape shallow local minima.  Each move preserves
    the ℓ₁‑norm, keeps the vector non‑negative and is accepted if it
    improves the cheap score or with a temperature‑dependent probability.

Optionally a linear‑programming refinement (the SOTA “LP‑step”) is
performed if SciPy is available.  The whole search stops as soon as the
deadline is reached and the best sequence discovered is returned as a
plain list of floats ready for the evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto the simplex of sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full convolution a * a using real FFT (fast for n up to a few 10⁴)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the (full) convolution a * a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: cheap evaluation of the objective (scale‑invariant)
#          score = 2 * n * max(conv) / (sum(a) ** 2)
# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Fast score based on the current vector (no external checks)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Helper: smooth gradient (log‑sum‑exp) of the max‑convolution
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, beta: float):
    """
    Return (conv, grad) where ``conv`` is a*a (full) and ``grad`` is the
    gradient of the smoothed max function
        f_β(a)= (1/β)·log Σ_k exp(β·conv_k)
    """
    n = a.shape[0]
    # 1) full convolution
    conv = _conv_self(a)

    # 2) softmax weights (stable version)
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()                       # shape (2n‑1,)

    # 3) grad = 2·correlation(a, w)  (correlation = convolution with reversed w)
    w_rev = w[::-1]
    size = 1 << ((n + w_rev.shape[0] - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w_rev, n=size)
    corr = np.fft.irfft(Fa * Fw, n=size)
    corr = corr[: n + w_rev.shape[0] - 1]       # length 3n‑2
    offset = w_rev.shape[0] - 1                # = 2n‑2
    grad = 2.0 * corr[offset : offset + n]      # length n

    return conv, grad


# ----------------------------------------------------------------------
# Helper: a single LP refinement step (if SciPy is present)
# ----------------------------------------------------------------------
def _lp_refinement(a: np.ndarray, max_iter: int, deadline: float):
    """Run up to ``max_iter`` LP refinements using the current best ``a``."""
    try:
        from scipy.optimize import linprog
    except Exception:     # SciPy unavailable – nothing to do
        return a

    n = a.shape[0]
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()
        rows = 2 * n - 1

        # Build the constraint matrix A such that (a * g)_k ≤ M
        # (a * g)_k = Σ_i a_i g_{k-i}
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            cols = np.arange(lo, hi + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(
            c=-np.ones(n),                # maximise Σ g_j  ↔ minimise -Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
        if not res.success:
            break
        g = res.x
        if g.sum() <= 0:
            break
        # normalise g to the probability simplex (sum = 1)
        g = g / g.sum()
        # convex combination with a modest mixing factor
        t = 0.2
        a = (1.0 - t) * a + t * g
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5       # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric families
    # ------------------------------------------------------------------
    target_sum = 1.0                                 # work on the simplex Σa_i = 1
    lengths = list(range(200, 2600, 200)) + [1326, 1500, 1800]
    for n in lengths:
        # exponential decay
        for alpha in np.logspace(-4, -0.5, num=12):
            a = np.exp(-alpha * np.arange(n, dtype=float))
            a = a / a.sum() * target_sum
            score = _cheap_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (exp) = {best_score:.6f} (len={n})")

        # inverse‑power law
        for p in np.linspace(0.5, 3.0, num=10):
            i = np.arange(1, n + 1, dtype=float)
            a = 1.0 / (i ** p)
            a = a / a.sum() * target_sum
            score = _cheap_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (pow) = {best_score:.6f} (len={n})")

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=8):
            i = np.arange(n, dtype=float)
            a = np.maximum(0.0, 1.0 - beta * i)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target_sum
            score = _cheap_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (lin) = {best_score:.6f} (len={n})")

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=8):
            i = np.arange(n, dtype=float)
            a = np.maximum(0.0, 1.0 - beta * i * i)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target_sum
            score = _cheap_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (quad) = {best_score:.6f} (len={n})")

        # triangular shape
        a = np.minimum(np.arange(n, dtype=float) + 1.0, n - np.arange(n, dtype=float))
        a = a / a.sum() * target_sum
        score = _cheap_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New best (tri) = {best_score:.6f} (len={n})")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – smooth continuation (Adam on log‑sum‑exp)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but fall back to a random probability vector
        n = np.random.randint(500, 1500)
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()

    a = best_seq.copy()
    n = a.shape[0]

    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    for beta in beta_schedule:
        while time.time() < deadline:
            conv, grad = _smooth_grad(a, beta)
            t_iter += 1
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            step = 0.15 / (np.sqrt(v_hat) + eps)          # base step size
            a = a - step * m_hat
            # keep a on the probability simplex
            a = _project_onto_simplex(a, target_sum)

            # occasional evaluation
            if t_iter % 5 == 0:
                sc = _cheap_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] New best (smooth) = {best_score:.6f} (len={n})")

            # very early exit if we already beat the target
            if best_score < 1.45:
                break
        if best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – random local moves (simulated‑annealing style)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    temperature = 1.0
    while time.time() < deadline:
        i, j = np.random.choice(n, size=2, replace=False)
        eps_move = np.random.rand() * 0.02       # at most 2 % of the total mass
        if a[i] < eps_move:
            continue
        a_new = a.copy()
        a_new[i] -= eps_move
        a_new[j] += eps_move
        cur_sc = _cheap_score(a)
        new_sc = _cheap_score(a_new)
        accept = False
        if new_sc < cur_sc:
            accept = True
        else:
            prob = math.exp((cur_sc - new_sc) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            a = a_new
            if new_sc < best_score:
                best_score = new_sc
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (anneal) = {best_score:.6f} (len={n})")
        temperature *= 0.9999
        if temperature < 1e-6:
            break

    # ------------------------------------------------------------------
    # Phase 4 – optional LP refinement (mirrors the SOTA “LP‑step”)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    a = _lp_refinement(a, max_iter=4, deadline=deadline)
    sc = _cheap_score(a)
    if sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"[{time.time():.2f}] New best (LP) = {best_score:.6f} (len={a.shape[0]})")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # final fallback – uniform distribution of length 1000
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=float)

    return best_seq.tolist()
