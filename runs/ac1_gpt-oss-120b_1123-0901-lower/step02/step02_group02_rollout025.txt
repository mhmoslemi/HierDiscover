def _sa_optimize(a, target_sum, deadline, init_temp=0.1, decay=0.9995, delta_scale=0.3, min_temp=1e-8):
    n = len(a)
    # compute conv via _conv_fft
    conv = _conv_fft(a)
    max_val = conv.max()
    best_a = a.copy()
    best_score = max_val
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_max = max_val
    T = init_temp
    iteration = 0
    while time.time() < deadline:
        i = np.random.randint(0, n)
        if cur_a[i] == 0:
            continue  # cannot transfer
        # choose destination j != i
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # delta fraction of a_i
        delta = cur_a[i] * delta_scale * np.random.rand()
        # optionally also random sign? We'll always move from i to j (decrease i)
        # Ensure delta <= cur_a[i]
        # compute updates
        a_i_old = cur_a[i]
        a_j_old = cur_a[j]
        # update a
        cur_a[i] = a_i_old - delta
        cur_a[j] = a_j_old + delta
        # update conv array using formulas
        # indices:
        idx2i = 2*i
        idxij = i + j
        idx2j = 2*j
        # update conv at these indices
        # we need old conv to compute new via delta.
        # We'll compute new directly, but easier: compute conv change using formulas.
        # However we need to access cur_conv (old) values.
        # Compute change:
        # delta_conv[2i] = -2*delta*a_i_old + delta**2
        # delta_conv[2j] = 2*delta*a_j_old + delta**2
        # delta_conv[i+j] = 2*delta*(a_i_old - a_j_old) - delta**2
        # However note a_i_old and a_j_old are the values before move.
        # apply:
        conv2i_new = cur_conv[idx2i] + (-2*delta*a_i_old + delta*delta)
        conv2j_new = cur_conv[idx2j] + (2*delta*a_j_old + delta*delta)
        convij_new = cur_conv[idxij] + (2*delta*(a_i_old - a_j_old) - delta*delta)
        # assign
        cur_conv[idx2i] = conv2i_new
        cur_conv[idx2j] = conv2j_new
        cur_conv[idxij] = convij_new
        # other indices unchanged.
        # compute new max
        new_max = cur_conv.max()
        # acceptance
        if new_max < cur_max or np.random.rand() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
            # accept
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = cur_a.copy()
            # maybe increase temperature? Not needed
        else:
            # reject: revert a and conv
            cur_a[i] = a_i_old
            cur_a[j] = a_j_old
            # revert conv update
            cur_conv[idx2i] = cur_conv[idx2i] - (-2*delta*a_i_old + delta*delta)
            cur_conv[idx2j] = cur_conv[idx2j] - (2*delta*a_j_old + delta*delta)
            cur_conv[idxij] = cur_conv[idxij] - (2*delta*(a_i_old - a_j_old) - delta*delta)
        # temperature decay
        T = max(T * decay, min_temp)
        iteration += 1
        # Optional occasional re-randomization of delta_scale
        if iteration % 1000 == 0:
            # perhaps adjust delta_scale revolve
            delta_scale *= 0.99  # slowly reduce step size
    return best_a, best_score

"""
Propose a non‑negative step‑function (vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a two‑phase optimiser:

1. **Initial shape** – a power‑law a_i ∝ (i+1)^{-p} (p≈0.5) projected onto
   the simplex {x_i ≥ 0, Σ x_i = √(2n)}.  Optionally the current global best
   can be truncated / padded and used as a warm‑start.

2. **Simulated‑annealing fine‑tuning** – the vector is improved by
   repeatedly transferring a random amount δ from one index to another
   (keeping the total sum exactly constant).  The convolution
   a∗a is updated *in‑place* in O(1) using the exact algebraic change
   for the three affected coefficients (2i, i+j, 2j).  A temperature
   schedule coupled with a slowly shrinking step‑size allows the
   optimiser to escape plateaus and drive the maximum autocorrelation
   towards its theoretical lower bound (≈ 1 for large n).

Multiple independent restarts with different lengths (800 … 6000) are
performed until the time budget expires.  The best sequence discovered
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – full linear convolution via FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)             # remove tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper 3 – the simulated‑annealing optimiser (constant‑sum moves)
# ----------------------------------------------------------------------
def _sa_optimize(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine ``a`` (already projected onto the simplex) by a fast
    simulated‑annealing scheme that preserves the sum exactly.
    Returns the best vector found and its max‑convolution value.
    """
    n = a.shape[0]
    target_sum = a.sum()                # should be √(2 n)
    conv = _conv_fft(a)
    cur_max = conv.max()

    best_a = a.copy()
    best_score = cur_max

    cur_a = a.copy()
    cur_conv = conv.copy()

    # temperature and step‑size schedules
    temp = max(cur_max * 0.05, 1e-4)     # start a few percent of the objective
    min_temp = 1e-8
    step_factor = 0.5                   # fraction of a_i used for a move
    iteration = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) pick a source index i (must have some mass) and a destination j
        # ------------------------------------------------------------------
        i = np.random.randint(n)
        if cur_a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ------------------------------------------------------------------
        # 2) propose a transfer δ from i to j (δ ≤ a_i)
        # ------------------------------------------------------------------
        delta = cur_a[i] * step_factor * np.random.rand()
        if delta <= 0.0:
            continue

        a_i_old = cur_a[i]
        a_j_old = cur_a[j]

        # ------------------------------------------------------------------
        # 3) apply the move (sum stays constant)
        # ------------------------------------------------------------------
        cur_a[i] = a_i_old - delta
        cur_a[j] = a_j_old + delta

        # ------------------------------------------------------------------
        # 4) update the convolution in O(1) using the exact algebraic change
        # ------------------------------------------------------------------
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        d2i = -2.0 * delta * a_i_old + delta * delta          # change at 2i
        d2j =  2.0 * delta * a_j_old + delta * delta          # change at 2j
        dij =  2.0 * delta * (a_i_old - a_j_old) - 2.0 * delta * delta  # at i+j

        cur_conv[idx2i] += d2i
        cur_conv[idx2j] += d2j
        cur_conv[idxij] += dij

        new_max = cur_conv.max()

        # ------------------------------------------------------------------
        # 5) Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = cur_a.copy()
        else:
            # revert the move
            cur_a[i] = a_i_old
            cur_a[j] = a_j_old
            cur_conv[idx2i] -= d2i
            cur_conv[idx2j] -= d2j
            cur_conv[idxij] -= dij

        # ------------------------------------------------------------------
        # 6) cool down and shrink the step size
        # ------------------------------------------------------------------
        temp = max(temp * 0.9995, min_temp)
        step_factor = max(step_factor * 0.9995, 0.001)

        iteration += 1

        # ------------------------------------------------------------------
        # 7) occasional jitter to kick the sampler out of a deep plateau
        # ------------------------------------------------------------------
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (target_sum * 5e-4 / n)
            cur_a = best_a + jitter
            cur_a = np.maximum(cur_a, 0.0)
            cur_a = _project_onto_simplex(cur_a, target_sum)
            cur_conv = _conv_fft(cur_a)
            cur_max = cur_conv.max()

    # final projection to respect the exact simplex constraint
    best_a = _project_onto_simplex(best_a, target_sum)
    return best_a, best_score

    best_seq = None
    best_score = float("inf")

    # length range – larger n gives a lower theoretical average
    min_n, max_n = 800, 6000

    while time.time() < deadline:
        remaining = deadline - time.time()

        # choose a problem size; shrink a little when time is short
        if remaining < 20.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(min_n, max_n + 1)

        # ------------------------------------------------------------------
        # 1) initial vector
        # ------------------------------------------------------------------
        # start from the global best with some probability (warm‑start)
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])
        else:
            # power‑law with a random exponent around 0.5
            p = np.random.uniform(0.45, 0.55)
            idx = np.arange(1, n + 1, dtype=np.float64)
            init = idx ** (-p)
            # tiny random jitter to break perfect symmetry
            init += np.random.rand(n) * 1e-4

        # project onto the simplex Σ a_i = √(2 n)
        target_sum = np.sqrt(2.0 * n)
        init = _project_onto_simplex(init, target_sum)

        # ------------------------------------------------------------------
        # 2) allocate a slice of the budget for this run
        # ------------------------------------------------------------------
        run_deadline = min(deadline, time.time() + 12.0)   # ≤12 s per restart

        # ------------------------------------------------------------------
        # 3) optimise the candidate with simulated annealing
        # ------------------------------------------------------------------
        seq_local, max_conv = _sa_optimize(init, run_deadline)

        # the objective that evaluate_sequence will compute is
        #   2 * n * max_conv / (sum(seq_local) ** 2)
        # because we keep the sum equal to √(2 n) this boils down to max_conv.
        score_local = max_conv

        # ------------------------------------------------------------------
        # 4) keep the global best
        # ------------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})")

        # early stop if we have already crossed a comfortable margin
        if best_score < 1.45:
            break

    # ----------------------------------------------------------------------
    # 5) fallback – should never be needed, but guarantees a valid return
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # convert to plain Python list as required by the evaluator
    return best_seq.tolist()
