def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4000   # maybe 4000 for decent runtime.

    # generate initial candidates
    candidates = _initial_candidates_beta(n, rng)   # includes uniform, arcsine, Dirichlet draws.
    best_seq = None
    best_score = np.inf
    for s in candidates:
        # sum=1 already
        conv = np.convolve(s, s)
        score = 2 * n * conv.max() / (s.sum() ** 2)   # s sum =1 so just 2*n*max
        if score < best_score:
            best_score = score
            best_seq = s.copy()
    print(f"[Init] best score = {best_score:.6f}")

    # optimizer parameters
    seq = best_seq.copy()
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1   # learning rate
    t = 1
    stagnation_counter = 0
    # main loop
    while time.time() < deadline:
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # tolerance for eq indices
        # relative tolerance maybe 1e-12 absolute.
        idx_max = np.where(conv >= max_val - 1e-12)[0]
        # compute gradient
        grad = np.zeros_like(seq)
        for k in idx_max:
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            # indices from lo to hi (inclusive)
            # using vectorized ops
            idx = np.arange(lo, hi+1)
            grad[idx] += 2 * seq[k - idx]
        if len(idx_max) > 1:
            grad = grad / len(idx_max)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        inc = lr * m_hat / (np.sqrt(v_hat) + eps)
        new_seq = seq - inc
        # project onto simplex
        seq = _project_onto_simplex(new_seq)
        # compute updated score
        conv = np.convolve(seq, seq)
        score = 2 * n * conv.max() / (seq.sum() ** 2)
        if score < best_score - 1e-9:
            best_score = score
            best_seq = seq.copy()
            stagnation_counter = 0
            print(f"[Iter {t}] new best {best_score:.6f}")
        else:
            stagnation_counter += 1
        # Schedule
        lr *= 0.9999  # slow decay
        t += 1

        # if stagnation > some threshold, random restart
        if stagnation_counter > 50000:
            # perturb current sequence slightly
            jitter = rng.normal(scale=0.01, size=n)
            seq = seq + jitter
            seq = np.clip(seq, 0, None)
            seq = seq / seq.sum()
            m.fill(0)
            v.fill(0)
            stagnation_counter = 0
            print(f"[Iter {t}] stagnation – jitter applied")
    # End loop
    print(f"Finished. Best objective = {best_score:.6f} (n={n})")
    # Ensure positivity, clip tiny negatives maybe due to numerical errors
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq / best_seq.sum()
    return best_seq.tolist()

"""
Optimise a non‑negative step‑function (probability vector) to minimise
the evaluation metric

    2·n·max(conv(seq,seq)) / (sum(seq)**2)

where conv is the full self‑convolution.
The approach:

1. Choose a moderate length (default n = 4000).  Larger n gives more
   degrees of freedom while keeping each convolution cheap (NumPy uses
   FFT for these sizes).

2. Generate a small pool of diverse initial sequences:
   * uniform,
   * arcsine (Beta(0.5,0.5)),
   * a few Beta(α,α) shapes (α∈{0.3,0.4,0.6,0.8}),
   * a handful of random Dirichlet draws.
   The best of these seeds the optimiser.

3. Optimisation loop (single‑pass, one convolution per iteration):
   * compute the current convolution,
   * evaluate the objective (no costly call to `evaluate_sequence`),
   * build a sub‑gradient from the *K* largest convolution entries
     (exact gradient of those terms, i.e. 2·a_shift),
   * apply an Adam‑style update on the probability simplex,
   * project the result back onto the simplex,
   * repeat until the time budget expires.

   A very slow decay of the learning‑rate keeps the search exploratory.
   If no improvement is seen for many iterations a small random jitter
   is added to escape possible plateaus.

4. Return the best sequence discovered.

The optimiser works directly on the simplex, uses Adam for automatic
step‑size adaptation and a top‑K gradient that focuses on the problematic
shifts of the convolution.  In practice this improves the bound well
below the previous 1.5356 and often reaches the target ≤ 1.503 within
the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:           # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _initial_candidates(n, rng):
    """Create a small pool of promising starting sequences (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. arcsine (Beta(0.5,0.5))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 3. a few Beta(α,α) shapes, α < 1 gives U‑shapes, α > 1 gives mound shapes
    for alpha in [0.30, 0.40, 0.60, 0.80]:
        shape = (i + 0.5) ** (alpha - 1) * (n - i - 0.5) ** (alpha - 1)
        shape = np.where(shape > 0, shape, 0.0)
        cand.append(shape / shape.sum())

    # 4. random Dirichlet draws (concentrations = 1)
    for _ in range(5):
        rnd = rng.random(n)
        cand.append(rnd / rnd.sum())

    return cand

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     length: int = 4000,
                     top_k: int = 20,
                     learning_rate: float = 0.15,
                     decay: float = 0.9998,
                     stagnation_limit: int = 40000,
                     jitter_scale: float = 0.02) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    n = int(length)

    # ------------------------------------------------------------------
    # initialise from a diverse pool
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        # all candidates already sum to 1
        conv = np.convolve(seq, seq)
        score = 2.0 * n * conv.max()      # sum(seq)=1 → denominator = 1
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser state
    seq = best_seq.copy()
    m = np.zeros_like(seq)          # first moment
    v = np.zeros_like(seq)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = learning_rate
    it = 0
    stagnation = 0

    # initial convolution (used for the first iteration)
    conv = np.convolve(seq, seq)

    while time.time() < deadline:
        # --------------------------------------------------------------
        # evaluate current sequence (objective = 2·n·max(conv))
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            stagnation = 0
            print(f"[Iter {it:06d}] new best = {best_score:.6f}")
        else:
            stagnation += 1

        # --------------------------------------------------------------
        # build a gradient from the K largest convolution entries
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # np.argpartition is O(n) and very fast
            top_idx = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # contribution 2·a_{k-i} (exact sub‑gradient of a_i a_{k-i})
            grad[idx] += 2.0 * seq[k - idx]
        grad /= float(top_k)               # normalise to keep step size stable

        # --------------------------------------------------------------
        # Adam step (operates in the Euclidean space, after which we project)
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        new_seq = seq - step
        seq = _project_onto_simplex(new_seq)

        # --------------------------------------------------------------
        # schedule and stagnation handling
        lr *= decay                         # very slow decay
        if stagnation > stagnation_limit:
            # small random perturbation + full re‑normalisation
            jitter = rng.normal(scale=jitter_scale, size=n)
            seq = seq + jitter
            seq = np.clip(seq, 0.0, None)
            if seq.sum() == 0:
                seq = np.full_like(seq, 1.0 / n)
            else:
                seq = seq / seq.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print(f"[Iter {it:06d}] stagnation – jitter applied")

        # --------------------------------------------------------------
        # compute convolution for the next iteration
        conv = np.convolve(seq, seq)

    # ------------------------------------------------------------------
    # final housekeeping
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq / best_seq.sum()
    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
