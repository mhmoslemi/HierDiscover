"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1. Load the previously discovered construction (``height_sequence_1``) or fall
   back to a uniform vector.
2. For several candidate lengths (original and a few multiples) do:
   a. Interpolate the base shape onto the new length and normalise the sum to
      ``sqrt(2·n)`` (the optimum of the objective is simply the maximum of
      the autocorrelation after this normalisation).
   b. **Alternating linear‑programming phase** – repeat until time runs out
      or no improvement:
        i.   Compute the current maximal self‑convolution ``M``.
        ii.  Determine the *active* indices – those where the convolution is
             within a tiny relative tolerance of ``M`` (typically only a few
             points).
        iii. Solve a linear program that **maximises** ``∑ g_i`` under the
             constraints ``(f * g)_k ≤ M`` for all active indices and
             ``g_i ≥ 0`` (upper bound 1000 is imposed to keep the LP bounded).
        iv.  Rescale the LP solution to the required total sum ``S = sqrt(2n)``.
        v.   Perform a line‑search on the mixing parameter ``α`` and replace

                 f ← (1−α)·f + α·g_rescaled

             (both vectors have sum ``S``).  The search accepts the first
             ``α`` (starting at 0.5 and halving repeatedly) that does **not**
             increase the maximal autocorrelation.  If none works the LP
             phase stops.
   c. After the LP phase a **short Adam fine‑tuning** (few × 10⁴ iterations)
      is applied to polish the result.
3. Keep the globally best sequence (lowest ``evaluate_sequence`` value) over
   all lengths and return it.

The LP uses only the currently active constraints, so each linear program
contains at most a few dozen rows – it is extremely fast even for n≈3000.
The outer loop is limited by a 1000 s wall‑clock budget, with a small safety
margin.

The implementation relies only on ``numpy`` and ``scipy.optimize.linprog``;
no filesystem or network access is performed.
"""

import time
import math
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (max convolution value, index of the maximum, full array)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]  # full convolution
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 1e-6, max_active: int = 30):
    """Indices where ``conv`` is within ``rel_tol`` of its maximum."""
    M = conv.max()
    idx = np.where(conv >= M * (1.0 - rel_tol))[0]
    if idx.size > max_active:
        idx = idx[:max_active]
    return idx


# ----------------------------------------------------------------------
def _build_active_matrix(f: np.ndarray, active_idx: np.ndarray):
    """
    Build the matrix A where each row corresponds to a convolution
    constraint (f * g)_k ≤ M for a given active index k.
    """
    n = f.shape[0]
    m = active_idx.size
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        A[row, j] = f[i]
    return A


# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float, active_idx: np.ndarray, bound: float = 1000.0):
    """
    Solve
        maximise    Σ g_i
        subject to  (f * g)_k ≤ M   for k in active_idx
                    0 ≤ g_i ≤ bound
    Returns the optimal g and a success flag.
    """
    n = f.shape[0]
    A = _build_active_matrix(f, active_idx)
    b = np.full(A.shape[0], M, dtype=np.float64)

    # maximise Σ g_i  → minimise - Σ g_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, bound) for _ in range(n)]

    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x, True
    else:
        return None, False


# ----------------------------------------------------------------------
def _mix_step(f: np.ndarray, g_scaled: np.ndarray, M_old: float):
    """
    Try mixing coefficients α = 0.5, 0.25, 0.125, … until the maximal
    autocorrelation does not increase.
    Returns (new_f, new_M, α) or (None, None, None) if no α works.
    """
    alphas = [0.5]
    while alphas[-1] > 1e-4:
        alphas.append(alphas[-1] / 2.0)

    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * g_scaled
        M_new, _, _ = _conv_max_fft(f_new)
        # accept a non‑increase (tiny tolerance to avoid numerical noise)
        if M_new <= M_old * (1.0 + 1e-12):
            return f_new, M_new, alpha
    return None, None, None


# ----------------------------------------------------------------------
def _adam_refine(a: np.ndarray, deadline: float, init_score: float):
    """
    A tiny Adam optimiser – only a few hundred thousand iterations –
    to polish a given vector while keeping the sum fixed.
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # make sure we start from a correctly normalised vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.03
    it = 0
    best_score = init_score
    best_vec = a.copy()

    while time.time() < deadline:
        it += 1
        # ----- convolution and sub‑gradient -------------------------------------------------
        M, _, conv_full = _conv_max_fft(a)
        # active set (strict – only the exact maximum)
        active = np.where(conv_full >= M - 1e-12)[0]
        rev = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        rev /= max(1, active.size)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # ----- Adam update -----------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection -------------------------------------------------------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:  # extremely unlikely
            a = np.full_like(a, target_sum / n)

        # occasional jitter to avoid plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- bookkeeping -----------------------------------------------------------------
        cur_score = float(2.0 * n * M / (target_sum ** 2))   # equals M after normalisation
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(
                f"[adam] n={n:5d} iter={it:7d}  best={best_score:.6f}"
            )
        # stop condition is enforced by the outer deadline
    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the wall‑clock deadline.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small margin for the harness
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the previous construction (if any); otherwise start with a modest uniform vector.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")

    # evaluate the initial candidate
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    init_score = float(evaluate_sequence(base.tolist()))
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a few multiples of the original size, capped at 8000.
    length_factors = [1.0, 1.25, 1.5, 2.0]
    lengths = sorted({max(300, int(n0 * f)) for f in length_factors})
    lengths = [min(l, 8000) for l in lengths]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise the vector for this length -----------------------------------------
        if n == n0:
            f = base.copy()
        else:
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            f = np.interp(new_x, old_x, base)
            f = np.maximum(f, 0.0)

        target_sum = math.sqrt(2.0 * n)
        f = f * (target_sum / f.sum())

        # initial score for this length
        M_cur, _, _ = _conv_max_fft(f)
        cur_score = float(2.0 * n * M_cur / (target_sum ** 2))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[len {n:5d}] new global best (init) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Alternating LP + mixing phase ---------------------------------------------------------
        outer_iter = 0
        while time.time() < deadline:
            outer_iter += 1
            # compute current max‑convolution and active set
            M_cur, _, conv_full = _conv_max_fft(f)
            active_idx = _active_indices(conv_full, rel_tol=1e-6, max_active=30)

            # solve LP
            g_opt, ok = _lp_max_sum(f, M_cur, active_idx, bound=1000.0)
            if not ok:
                print("[lp] infeasible – terminating LP phase")
                break

            sum_g = g_opt.sum()
            if sum_g <= 0.0:
                break

            # rescale to the required total sum
            g_scaled = g_opt * (target_sum / sum_g)

            # try to mix – ensure the maximal autocorrelation does not increase
            f_new, M_new, alpha = _mix_step(f, g_scaled, M_cur)
            if f_new is None:
                # no admissible mixing factor → stop the LP loop
                break

            f = f_new
            cur_score = float(2.0 * n * M_new / (target_sum ** 2))

            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(
                    f"[len {n:5d} lp iter {outer_iter:3d}] α={alpha:.5f}  score={best_score:.6f}"
                )

            # stop early if the improvement becomes negligible
            if abs(M_cur - M_new) < 1e-12:
                break

        # ------------------------------------------------------------------
        # Short Adam fine‑tuning (optional – only if time permits)
        remaining = deadline - time.time()
        if remaining > 30.0:   # need at least a few seconds for Adam
            print(f"[finetune] length={n}  remaining={remaining:.1f}s")
            f_refined, refined_score = _adam_refine(f, deadline, best_score)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = f_refined.copy()
                print(f"[finetune] new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed.
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform sequence")

    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
