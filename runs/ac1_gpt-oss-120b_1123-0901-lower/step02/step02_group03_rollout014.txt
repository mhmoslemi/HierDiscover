    a = simplex_projection(a0, target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    for stage_alpha in schedule_alpha:
        for step_i in range(max_iter_per_stage):
            t+=1
            # compute conv
            conv = _conv_full_fft(a)
            max_conv = conv.max()
            # evaluate and update best
            if max_conv < best_val: update best
            # compute softmax weights
            shift = max_conv
            w = np.exp(stage_alpha * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0:
                # fallback to active max
                active = np.where(conv >= max_conv - 1e-12)[0]
                grad = _gradient_of_active(a, active)
            else:

                # compute gradient using convolution with reversed a
                a_rev = a[::-1]
                # full conv of w_norm and a_rev
                grad_full = _full_conv_fft(w_norm, a_rev)  # length 3n-2
                # slice central n entries

            a = np.maximum(a, 0.0)

            if time.time() > time_limit: break

def _full_conv_fft(x, y):
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X*Y, n=n_fft)[:n]
    return conv

def conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]; n_fft = 1 << ((2*n-1).bit_length()); A = np.fft.rfft(a, n=n_fft); conv = np.fft.irfft(A*A, n=n_fft)[:2*n-1]; return conv

def full_conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X*Y, n=n_fft)[:n]
    return conv

def compute_softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    # conv length = 2*n-1
    # gradient = 2 * conv_full_fft(w_norm, a_rev)[(n-1):(2*n-1)]
    n = a.shape[0]
    shift = conv.max()
    # compute weights
    # scale term might be huge if alpha large; use shift for stability
    exp_terms = np.exp(alpha * (conv - shift))
    w_sum = exp_terms.sum()
    if w_sum == 0.0:
        # fallback: active max gradient
        # use zero tolerance
        active = np.where(conv >= shift - 1e-12)[0]
        # compute gradient via active set as used earlier
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        return grad
    w_norm = exp_terms / w_sum
    a_rev = a[::-1]
    conv_full = full_conv_fft(w_norm, a_rev)
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad

def adam_step(a: np.ndarray, grad: np.ndarray, m: np.ndarray, v: np.ndarray,
              t: int, lr: float = 0.05,
              beta1: float = 0.9, beta2: float = 0.999, eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    m_new = beta1 * m + (1 - beta1) * grad
    v_new = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m_new / (1 - beta1 ** t)
    v_hat = v_new / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    return a_new, m_new, v_new

"""
Proposes a non‑negative step function that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works inside the given time budget (default 1000 s).

*   The vector is kept on the simplex Σa_i = √(2 n) (the objective is then
    exactly the maximal autocorrelation value).
*   A smooth surrogate of the “max” – the log‑sum‑exp – is used.  For a
    temperature α the surrogate is
        φ_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Its gradient can be expressed as a convolution, which we evaluate with
    FFTs.
*   An Adam optimiser (with projection onto the simplex) descends the
    surrogate.  The temperature is increased in stages (continuation
    method) so the algorithm first explores a smoothed landscape and then
    focuses on the true max.
*   Many independent restarts with different lengths (≈1500–6000) are run
    until the time limit is hit.  The best sequence found is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# 1.  Projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of vector `v` onto the simplex of radius `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# 2.  FFT based full convolutions
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution of `a` with itself (real FFT)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _full_conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two arbitrary real vectors `x` and `y`."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


# --------------------------------------------------------------------------- #
# 3.  Gradient of the smoothed max (log‑sum‑exp) using FFTs
# --------------------------------------------------------------------------- #
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of φ_α(a) = (1/α)·log Σ exp(α·(a*a)_k) w.r.t. `a`.
    Returns a vector of length n.
    """
    n = a.shape[0]
    max_val = conv.max()

    # soft‑max weights – stable with a shift by the maximum
    exp_terms = np.exp(alpha * (conv - max_val))
    w_sum = exp_terms.sum()
    if w_sum == 0.0:                     # fallback to ordinary sub‑gradient
        active = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        return grad

    w = exp_terms / w_sum                 # normalized weights, length 2n‑1
    a_rev = a[::-1]                       # reversed a
    conv_full = _full_conv_fft(w, a_rev)  # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


# --------------------------------------------------------------------------- #
# 4.  One Adam update (including the projection back onto the simplex)
# --------------------------------------------------------------------------- #
def _adam_update(a, grad, m, v, t, lr=0.05,
                 beta1=0.9, beta2=0.999, eps=1e-8,
                 target_sum=None):
    """Perform a single Adam step and re‑project onto the simplex."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    if target_sum is not None:
        a_new = project_onto_simplex(a_new, target_sum)
    return a_new, m, v


# --------------------------------------------------------------------------- #
# 5.  Optimisation of a single instance (fixed length `n`)
# --------------------------------------------------------------------------- #
def _optimise_instance(n: int, init_vec: np.ndarray,
                       deadline: float,
                       global_best_seq, global_best_val):
    """
    Run the Adam‑based optimisation for a fixed problem size `n`.
    Returns possibly updated (best_seq, best_val).
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2n)
    a = project_onto_simplex(init_vec.astype(float), target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # temperature schedule (α) – early smooth, later sharp
    schedule = [(0.5, 400), (1.0, 400), (2.0, 400),
                (5.0, 400), (10.0, 400), (20.0, 400), (40.0, 400)]

    base_lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    last_improve_iter = 0
    iter_counter = 0
    max_no_improve = 1200                # stop when nothing improves for a while

    for alpha, max_iter in schedule:
        for _ in range(max_iter):
            if time.time() >= deadline:
                return global_best_seq, global_best_val

            conv = _conv_full_fft(a)
            cur_max = conv.max()

            # update global best
            if cur_max < global_best_val:
                global_best_val = float(cur_max)
                global_best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (n={n}) = {global_best_val:.6f}")
                last_improve_iter = iter_counter
                if global_best_val < 1.503:
                    return global_best_seq, global_best_val

            # gradient of the smoothed objective
            grad = _softmax_gradient(a, conv, alpha)

            # Adam step
            t += 1
            a, m, v = _adam_update(a, grad, m, v, t,
                                   lr=base_lr,
                                   beta1=beta1, beta2=beta2,
                                   eps=eps,
                                   target_sum=target_sum)

            # occasional random shake (simulated‑annealing style)
            if np.random.rand() < 0.01:
                noise = np.random.randn(n) * (0.002 * target_sum / n)
                a = project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
                base_lr *= 0.9                 # slowly cool after a shake

            # learning‑rate decay
            base_lr *= 0.9999

            iter_counter += 1
            if iter_counter - last_improve_iter > max_no_improve:
                # stagnated – abort the inner loop early
                break
        if iter_counter - last_improve_iter > max_no_improve:
            break

    return global_best_seq, global_best_val
