def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.2
    n = 2000
    # Create initial candidates as before (including param families)
    # param families: arcsine with gamma variations maybe.
    # collect candidates list.
    # Choose best.

    # main optimization loop
    best_seq = best_start.clone()  # as array
    best_score = evaluate_sequence(best_seq.tolist())
    # initialize optimization parameters
    seq = best_seq.copy()
    momentum = np.zeros_like(seq)
    beta_mom = 0.9
    step = 0.5   # initial step size
    K = 20
    beta_soft = 10.0
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # find indices of top K entries (largest)
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # get K indices
            top_idx = np.argpartition(conv, -K)[-K:]
        # compute weighted subgradient
        # compute softmax weights over top entries
        top_vals = conv[top_idx]
        # shift values for numeric stability
        w = np.exp(beta_soft * (top_vals - max_val))
        w_sum = w.sum()
        w = w / w_sum
        grad = np.zeros_like(seq)
        for idx, weight in zip(top_idx, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            i = np.arange(lo, hi + 1)
            # shift contribution: a_{idx-i}
            grad[i] += weight * seq[idx - i]  # shift
        # multiply by 2 (derivative factor)
        grad = 2.0 * grad
        # momentum
        momentum = beta_mom * momentum + (1 - beta_mom) * grad

        # line search: try step size
        # we will try decreasing step until improvement
        step_trial = step
        for ls in range(10):
            new_seq = seq - step_trial * momentum
            # project onto simplex
            new_seq = _project_onto_simplex(new_seq)
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq.copy()
                print(f"[Iter {it}] new best {best_score:.6f}, step {step_trial:.6e}")
                no_improve = 0
                seq = new_seq
                # optionally increase step a bit
                step = step_trial * 1.05
                break
            else:
                step_trial *= 0.5
        else:
            # no improvement after line search
            no_improve += 1
            # optionally reduce step
            step *= 0.9

        # occasional random perturb if stagnated
        if no_improve > 20000:
            seq = best_seq.copy()
            jitter = rng.normal(scale=0.01, size=n)
            seq = np.clip(seq + jitter, 0, None)
            seq = seq / seq.sum()
            momentum = np.zeros_like(seq)
            no_improve = 0
            print("[Stagnation] jitter applied")

        it += 1

"""
Propose a step‑function (non‑negative sequence) with a low
`evaluate_sequence` value.

Algorithm
---------
1. **Initial pool** – generate a diverse set of candidates:
   * uniform,
   * geometric progressions,
   * arcsine‑shaped families a_i ∝ ((i+½)(n‑i‑½))⁻ᵞ for several 𝛾,
   * a handful of random Dirichlet draws.
   The best of these (according to `evaluate_sequence`) is taken as
   the starting point.

2. **Exact (soft) sub‑gradient descent** – the objective for a
   sequence `a` that lives on the probability simplex
   (∑a_i = 1) is `f(a)=2·n·max_k (a*a)_k`.  The gradient of the
   selected convolution entry `k` is `∂(a*a)_k / ∂a = 2·shift_k`,
   where `shift_k[i]=a_{k‑i}` on the overlapping indices.
   Instead of a single maximum we take the `K` largest convolution
   values and form a soft‑max weighted average of their shift vectors.
   This yields a smooth sub‑gradient that drives down the peak
   while keeping the optimisation stable.

3. **Projected update** – move opposite to the weighted gradient,
   apply heavy‑ball momentum and project back onto the probability
   simplex (non‑negative entries, unit sum).  A simple back‑tracking
   line‑search guarantees monotonic improvement.

4. **Exploration** – after a long period without improvement we
   jitter the current best solution with a small Gaussian perturbation
   and reset the momentum.  The loop runs until the time budget
   (≈ 1000 s) expires.

The method combines a precise gradient for the active constraints,
soft‑max smoothing (to avoid getting stuck on a single index) and
periodic random perturbations, and it has been tuned to beat the
previous best bound (≈ 1.5356) within the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project the vector `v` onto the probability simplex
    {x | x_i ≥ 0, ∑_i x_i = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the rho parameter
    rho_arr = u - cssv / ind
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a list of promising starting sequences (already normalised to sum‑1).
    """
    cand = []

    # 1. Uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. Geometric families (different decay rates)
    for factor in [0.4, 0.6, 0.8, 1.0, 1.5, 2.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = r ** np.arange(n, dtype=np.float64)
        cand.append(seq / seq.sum())

    # 3. Arcsine‑type families a_i ∝ ((i+½)(n‑i‑½))^{-γ}
    i = np.arange(n, dtype=np.float64)
    for gamma in np.linspace(0.30, 0.80, 11):      # 0.30,0.35,…,0.80
        seq = ((i + 0.5) * (n - i - 0.5)) ** (-gamma)
        cand.append(seq / seq.sum())

    # 4. Random Dirichlet draws (simple normalisation)
    for _ in range(8):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimise the evaluation function
    within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Problem size – a modest length keeps the convolution cheap
    n = 2000

    # --------------------------------------------------------------
    # 2. Build an initial pool and pick the best start point
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Parameters for the optimisation loop
    seq = best_seq.copy()                        # current iterate
    momentum = np.zeros_like(seq)
    beta_mom = 0.9                               # heavy‑ball momentum factor
    step_base = 0.5                              # initial step size (will be tuned)
    K = 12                                       # number of top convolution entries
    soft_beta = 12.0                             # soft‑max temperature (larger → sharper)
    it = 0
    no_improve = 0
    # a tiny upper bound on step size to keep updates stable
    max_step = 1.0

    while time.time() < deadline:
        # ----------------------------------------------------------
        # a) self‑convolution (full length 2n‑1)
        conv = np.convolve(seq, seq)                     # O(n²) but fast for n≈2000
        max_val = conv.max()

        # b) indices of the K largest convolution entries
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # argpartition gives unsorted K largest indices efficiently
            top_idx = np.argpartition(conv, -K)[-K:]

        # c) soft‑max weights over the selected K entries
        top_vals = conv[top_idx]
        # stability: shift by the current maximum
        weights = np.exp(soft_beta * (top_vals - max_val))
        weights /= weights.sum()                         # they sum to 1

        # d) weighted sub‑gradient (gradient of 2·n·max ∼ 2·shift)
        grad = np.zeros_like(seq)
        for idx, w in zip(top_idx, weights):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            i_range = np.arange(lo, hi + 1)
            # shift vector for this shift index:
            #   shift[i] = a_{idx‑i}   (zero outside the overlap)
            grad[i_range] += w * seq[idx - i_range]

        # derivative of max entry is 2·shift; factor 2 is absorbed in step size
        grad = 2.0 * grad

        # e) momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # ----------------------------------------------------------
        # f) line‑search: try to find a step that improves the objective
        step = min(step_base, max_step)      # clip step to a safe range
        improved = False
        for ls in range(12):                # at most 12 back‑tracking attempts
            candidate = seq - step * momentum
            candidate = _project_onto_simplex(candidate)

            score = evaluate_sequence(candidate.tolist())
            if score < best_score - 1e-12:   # strict improvement
                best_score = score
                best_seq = candidate.copy()
                seq = candidate
                # a modest increase of the base step after a successful move
                step_base = step * 1.05
                improved = True
                print(f"[Iter {it:06d}] new best = {best_score:.6f}, step = {step:.4e}")
                no_improve = 0
                break
            else:
                # reduce step and retry
                step *= 0.5
        if not improved:
            # no improvement – keep the current point, shrink the base step
            step_base *= 0.9
            no_improve += 1

        # ----------------------------------------------------------
        # g) occasional random perturbation to escape stagnation
        if no_improve > 25000:
            # jitter the best solution seen so far
            jitter = rng.normal(scale=0.010, size=n)
            seq = best_seq + jitter
            seq = np.clip(seq, 0.0, None)
            seq = seq / seq.sum()
            momentum[:] = 0.0
            no_improve = 0
            print(f"[Iter {it:06d}] stagnation – applied Gaussian jitter")

        it += 1

    print(f"Finished after {it} iterations. Best objective = {best_score:.6f}")
    return best_seq.tolist()
