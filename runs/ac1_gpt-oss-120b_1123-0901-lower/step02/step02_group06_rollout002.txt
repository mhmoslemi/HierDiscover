"""
Propose a non‑negative step‑function (probability vector) that minimises
the objective returned by ``evaluate_sequence``.

Algorithm
---------
1. **Initial pool** – generate a few analytically‑motivated candidates:
   * uniform,
   * arcsine law,
   * symmetric Beta‑shaped families  a_i ∝ (i+½)^{α‑1}(n‑i‑½)^{β‑1}
     for several α,β ∈ (0,1],
   * a handful of random Dirichlet draws,
   * (if available) the previously found best sequence ``height_sequence_1``.
   The best of these seeds the optimisation.

2. **Soft‑max gradient descent** – we minimise a smooth approximation of
   ``max_k (a*a)[k]``:
   ```
       L(a) = (1/γ)·log Σ_k exp(γ·(a*a)[k])      (γ>0)
   ```
   The gradient of L is a weighted sum of the gradients of the individual
   convolution entries, where the weights are the soft‑max probabilities.
   Using the identity
       ∂(a*a)[k]/∂a_j = 2·a_{k‑j}  (zero when the index is out of range),
   the gradient can be obtained by a single correlation, i.e. a convolution
   of the reversed sequence with the soft‑max weight vector.

   An Adam‑style adaptive step is employed, followed by a projection onto
   the probability simplex (non‑negative entries that sum to one).

3. **Exploration** – the smoothness parameter γ is increased slowly so that
   the optimisation starts with a broad view of the landscape and finishes
   focusing on the true maximum.  If no improvement is observed for many
   iterations a small Gaussian jitter is added to escape possible plateaus.

The whole routine respects the 1000 s wall‑time budget, prints progress
and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta‑shaped distribution on {0,…,n‑1}:
        a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}
    Returned vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = i + 0.5                # distance from left boundary
    y = n - i - 0.5            # distance from right boundary
    seq = np.power(x, alpha - 1.0) * np.power(y, beta - 1.0)
    # Guard against NaNs when alpha or beta = 1 (exponent 0)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse set of seed vectors (already normalised)."""
    cand: list[np.ndarray] = []

    # 1. Uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. Arcsine law (α = β = 0.5)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 3. Symmetric Beta families (α = β) with several exponents
    for a in [0.2, 0.3, 0.35, 0.4, 0.45, 0.5, 0.6, 0.7, 0.8, 0.9]:
        cand.append(_beta_shape(n, a, a))

    # 4. A few asymmetric Beta shapes
    for a, b in [(0.3, 0.7), (0.7, 0.3), (0.25, 0.75), (0.75, 0.25)]:
        cand.append(_beta_shape(n, a, b))

    # 5. Random Dirichlet draws
    for _ in range(6):
        r = rng.random(n)
        cand.append(r / r.sum())

    # 6. Possibly reuse the previously discovered best sequence
    try:
        # ``height_sequence_1`` is defined outside this file in the evaluation
        # environment.  If it exists we simply normalise it and add it.
        global height_sequence_1
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size == n:
            cand.append(prev / prev.sum())
    except Exception:
        pass

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low ``evaluate_sequence`` value.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for printing / finalisation
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1. Problem size – a modest length that still allows fine shapes
    n = 2000

    # ------------------------------------------------------------------
    # 2. Seed the optimisation with a small pool of candidates
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        # direct evaluation (the helper does all safety checks)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best score after seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Soft‑max gradient descent (Adam variant) on the smooth max
    seq = best_seq.copy()                     # current iterate
    m = np.zeros_like(seq)                    # first‑moment accumulator
    v = np.zeros_like(seq)                    # second‑moment accumulator
    beta1, beta2 = 0.9, 0.999                 # Adam parameters
    eps = 1e-8
    step0 = 0.25                               # base learning rate
    gamma_start = 5.0                         # initial soft‑max temperature
    gamma_inc = 1e-5                          # how fast we sharpen the max
    it = 0
    no_improve = 0

    # Pre‑compute the convolution for the starting point (used in the loop)
    conv = np.convolve(seq, seq)

    while time.time() < deadline:
        # ---------- soft‑max weights for the current convolution ----------
        max_conv = conv.max()
        gamma = gamma_start + gamma_inc * it * (it + 1) * 0.5   # grows ~ O(it²)
        # stabilise exponentials by subtracting the max before exp
        exp_arg = gamma * (conv - max_conv)
        w = np.exp(exp_arg, dtype=np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:                     # degenerate – fall back to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                         # now a probability vector

        # ---------- gradient of the smooth max ----------
        # g = 2 * correlation(seq, w)   →   convolution of reversed seq with w
        grad_rev = np.convolve(seq[::-1], w)
        grad = 2.0 * grad_rev[n - 1 : n - 1 + n]   # slice length n

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = step0 / (np.sqrt(v_hat) + eps)

        new_seq = seq - step * m_hat
        new_seq = _project_onto_simplex(new_seq)   # enforce feasibility

        # ---------- evaluate new candidate ----------
        conv_new = np.convolve(new_seq, new_seq)
        score = 2.0 * n * conv_new.max() / (new_seq.sum() ** 2)

        if score < best_score - 1e-9:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Iter {it:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # ---------- occasional random perturbation ----------
        if no_improve > 25000:
            jitter = rng.normal(scale=0.01, size=n)
            new_seq = new_seq + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = np.full(n, 1.0 / n)
            else:
                new_seq /= new_seq.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Iter {it:06d}] jitter applied (stagnation)")

        # Prepare for next iteration
        seq = new_seq
        conv = conv_new
        it += 1

    print(f"Finished after {it} iterations. Best objective = {best_score:.6f}")
    # Return the best sequence found (as plain python list)
    return best_seq.tolist()
