```
def simplex_projection(v):
    # v: numpy array
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # Compute rho
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    for i in range(iterations):
        conv = np.convolve(a, a)
        max_conv = conv.max()

"""
propose_candidate
=================

Search for a discrete step‑function (non‑negative sequence) that minimises
the supplied ``evaluate_sequence``.  The method is a smooth‑gradient descent
on the *log‑sum‑exp* approximation of the maximum of the auto‑convolution.
The optimisation is performed on the probability simplex (the entries sum to
one) using multiplicative mirror‑descent updates, which keep the iterates
strictly non‑negative.  Fast FFT based convolutions make each iteration
~O(n log n) even for n≈6000, allowing thousands of iterations within the
budget.  The algorithm starts from the classic arcsine construction (the
known near‑optimal analytical solution) and then tries to improve it by
continuous optimisation; occasional random re‑initialisations help to escape
possible plateaus.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: FFT based convolution (real input → real output)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Helper: generate the (unnormalised) arcsine distribution of length n
# ----------------------------------------------------------------------
def _arcsine_raw(n: int) -> np.ndarray:
    """Return the raw (unnormalised) arcsine coefficients a_k = C·(2k choose k)/4^k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return a


# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (sum==1, a>=0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # Find the largest rho where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` value than the baseline arcsine construction.

    The routine respects a global ``height_sequence_1`` if present, and falls
    back to a fresh arcsine start otherwise.  It performs a smooth‑gradient
    descent on the log‑sum‑exp surrogate of the max convolution, using FFT
    convolutions for speed, and mirrors‑descent updates that stay on the
    probability simplex.
    """
    np.random.seed(seed)

    # -------------------------------------------------------------
    # 1.  Choose the length of the sequence.
    # -------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        # try to reuse the length of the previously found best sequence
        try:
            n = len(height_sequence_1)  # type: ignore[name-defined]
        except Exception:
            n = 6027                     # default used in the last run

    # -------------------------------------------------------------
    # 2.  Initialise the distribution.
    #    Prefer the stored sequence if it exists, otherwise use the
    #    analytical arcsine recipe (normalised to sum==1).
    # -------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.shape[0] != n:
            raise ValueError
        # normalise to a probability vector (sum==1)
        a = init.copy()
        a = a / np.maximum(a.sum(), 1e-12)
        print("[init] using global height_sequence_1")
    except Exception:
        raw = _arcsine_raw(n)
        a = raw / raw.sum()          # sum == 1
        print("[init] arcsine start, length =", n)

    # Ensure simplex feasibility
    a = _project_simplex(a)

    # -------------------------------------------------------------
    # 3.  Evaluate initial score (for logging).
    # -------------------------------------------------------------
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[start] score={best_score:.6f}")

    # -------------------------------------------------------------
    # 4.  Parameters for the optimisation loop.
    # -------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    iteration = 0
    # smoothing parameter for the log‑sum‑exp surrogate
    beta = 5.0
    beta_inc = 1.02          # slowly make the surrogate sharper
    # mirror‑descent step size (multiplicative factor)
    eta = 2000.0
    eta_decay = 0.9995
    # controls how often we randomise the iterate if we stall
    stall_counter = 0
    stall_limit = 500

    # -------------------------------------------------------------
    # 5.  Main optimisation loop.
    # -------------------------------------------------------------
    while time.time() < deadline:
        # ----- 5.1  compute auto‑convolution via FFT -----------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_conv = conv.max()

        # ----- 5.2  smooth approximation of the max (log‑sum‑exp) ---
        # w_k = exp(beta * (conv_k - max_conv)),   sum(w)=Z
        w = np.exp(beta * (conv - max_conv))
        Z = w.sum()
        if Z == 0.0:                     # numerical safeguard

            w[:] = 1.0 / w.size

        # ----- 5.3  gradient of the smooth surrogate ----------------
        # grad = 2 * (rev(a) * (w / Z))  (convolution, slice the centre)

        # a_new proportional to a * exp(-eta * grad), then renormalise

        # numerical safety – avoid all‑zero vectors
        if a_new.sum() == 0.0:

        a_new = a_new / a_new.sum()               # back to simplex

        cur_score = evaluate_sequence(a_new.tolist())

        # ----- 5.6  accept improvement --------------------------------
        if cur_score < best_score:

            best_seq = a_new.copy()
            stall_counter = 0
            print(f"[improve] it={iteration:05d}, score={best_score:.6f}")

        # ----- 5.7  update the iterate (always move) -----------------
        a = a_new

        # ----- 5.8  schedule parameters --------------------------------
        beta = min(beta * beta_inc, 1000.0)   # keep beta bounded
        eta *= eta_decay

        # ----- 5.9  check for stagnation -------------------------------

            a = a + perturb

            a = a / a.sum()
            stall_counter = 0
            print(f"[re‑seed] it={iteration:05d}")

    # -------------------------------------------------------------
    # 6.  Final log and return.
    # -------------------------------------------------------------
    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
