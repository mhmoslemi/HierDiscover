def project_simplex(v):
    # v is vector
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a new candidate sequence for the self‑convolution bound.

The algorithm combines several ideas that are complementary to the
projected sub‑gradient method used previously.

1.  The coefficients are kept on the probability simplex
    (non‑negative, sum = 1) by an exact Euclidean projection.
2.  The objective  max(conv(a,a))  is approximated by the smooth
    log‑sum‑exp (softmax) of the convolution values.  For a large
    smoothing parameter β this approximation becomes the true max,
    while for smaller β the gradient is well‑behaved.
3.  Adam (adaptive moments) is used to obtain a robust step‑size
    schedule.  After each Adam update the vector is projected back onto
    the simplex.
4.  Multiple random restarts are performed until the overall time
    budget (≈ 1000 s) is exhausted.  Each restart starts from a
    different initial distribution (central‑binomial, Dirichlet
    samples, or a random perturbation of the previous best).  The best
    sequence seen across all restarts is returned.

The method works purely with NumPy’s FFT (O(n log n) per iteration) and
fits easily within the allowed resource limits.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level – required by the platform)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Exact Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Weights w_k = exp(β·conv_k) / Σ exp(β·conv)  (log‑sum‑exp smoothing)."""
    if beta <= 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    cmax = conv.max()
    # subtract the max to prevent overflow
    exp_term = np.exp(beta * (conv - cmax))
    sum_exp = exp_term.sum()
    return exp_term / sum_exp


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  (1/β)·log Σ exp(β·conv)  w.r.t a.
    Implements   g_i = 2· Σ_k w_k·a_{k-i} .
    """
    n = len(a)
    m = len(w)                     # = 2·n‑1
    # convolution of a with reversed w (w_rev = w[::-1])
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fw_rev = np.fft.rfft(w[::-1], size)
    conv_aw_rev = np.fft.irfft(fa * fw_rev, size)[: n + m - 1]
    # g_i = 2·conv_aw_rev[m‑1‑i]   (see derivation in the analysis)
    idx = m - 1 - np.arange(n)
    grad = 2.0 * conv_aw_rev[idx]
    return grad


def _central_binomial_initial(n: int) -> np.ndarray:
    """Return the truncated central‑binomial (1/4)^i·C(2i,i) sequence, projected to Σ=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return _project_onto_simplex(a)


def _dirichlet_initial(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) sample and project onto the simplex (already on it)."""
    sample = rng.gamma(alpha, 1.0, size=n)
    return sample / sample.sum()


# ----------------------------------------------------------------------
# Core optimisation routine (inner loop)
# ----------------------------------------------------------------------


def _optimize_one_restart(
    n: int,
    max_iters: int,
    beta_start: float,
    beta_end: float,
    lr: float,
    rng: np.random.Generator,
    init_mode: str = "central",
    warm_start: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Run Adam + soft‑max smoothing for a single random restart.
    Returns the best vector found (as a NumPy array) and its score.
    """
    # --------------------------------------------------------------
    # Initialise a
    # --------------------------------------------------------------
    if warm_start is not None:
        a = warm_start.copy()
    elif init_mode == "central":
        a = _central_binomial_initial(n)
    elif init_mode == "dirichlet_small":
        a = _dirichlet_initial(n, alpha=0.2, rng=rng)
    elif init_mode == "dirichlet_medium":
        a = _dirichlet_initial(n, alpha=2.0, rng=rng)
    else:  # pure uniform + tiny noise
        a = _project_onto_simplex(rng.random(n))

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # keep the best
    conv = _full_convolution(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # schedule for smoothing parameter β (linear growth)
    if max_iters > 1:
        beta_inc = (beta_end - beta_start) / (max_iters - 1)
    else:
        beta_inc = 0.0

    for t in range(1, max_iters + 1):
        # current smoothing value
        beta = beta_start + (t - 1) * beta_inc

        # convolution and soft‑max weights
        conv = _full_convolution(a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # compute gradient of the smoothed objective
        w = _softmax_weights(conv, beta)
        grad = _gradient_softmax(a, w)

        # Adam update (note the descent direction is -grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_onto_simplex(a)

    # final evaluation
    conv = _full_convolution(best_a)
    best_score = 2.0 * n * conv.max()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence().  The routine performs many random restarts of a
    smooth Adam optimisation (log‑sum‑exp smoothing of the self‑convolution
    peak).  The best sequence seen before the time budget expires is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction is available we reuse its
    # length (it is usually well‑chosen).  Otherwise we fall back to a
    # moderately large value.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        n = 6000

    # keep the overall best vector
    overall_best_a = None
    overall_best_score = float("inf")

    # parameters that work well in practice
    max_iters_per_restart = 2000          # a few thousand Adam steps per run
    beta_start = 0.5                      # start with a very smooth approximation
    beta_end = 30.0                       # end with a fairly sharp max
    lr = 0.02                             # base learning rate for Adam

    # ------------------------------------------------------------------
    # Main outer loop – keep launching restarts until time runs out.
    # ------------------------------------------------------------------
    restart_id = 0
    while time.time() < deadline:
        # pick an initialisation scheme (randomised)
        mode = rng.choice(
            ["central", "dirichlet_small", "dirichlet_medium", "noise"]
        )
        warm = None
        # occasionally use the current best as a warm start (with tiny jitter)
        if overall_best_a is not None and rng.random() < 0.15:
            warm = overall_best_a + rng.normal(scale=1e-4, size=n)
            warm = _project_onto_simplex(warm)

        # run the optimiser
        a_opt, score_opt = _optimize_one_restart(
            n=n,
            max_iters=max_iters_per_restart,
            beta_start=beta_start,
            beta_end=beta_end,
            lr=lr,
            rng=rng,
            init_mode=mode,
            warm_start=warm,
        )
        restart_id += 1

        # keep the best over all restarts
        if score_opt < overall_best_score:
            overall_best_score = score_opt
            overall_best_a = a_opt.copy()
            print(
                f"[restart {restart_id:3d}] new global best score = {overall_best_score:.6f} (mode={mode})"
            )

        # safeguard against running too many restarts with a tiny budget left
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as a Python list of floats)
    # ------------------------------------------------------------------
    if overall_best_a is None:
        # this should never happen, but fallback to a trivial uniform vector
        overall_best_a = np.full(n, 1.0 / n)

    print(f"Finished search. Global best score = {overall_best_score:.6f}")
    return overall_best_a.tolist()
