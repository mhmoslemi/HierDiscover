def propose_candidate(seed=42, budget_s=1000, length=None, **kwargs):
    import time, numpy as np, scipy.optimize, scipy.sparse

    np.random.seed(seed)
    # Determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = length if length is not None else 2000

    # initialize as arcsine
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * ((2*k-1)*(2*k)) / (4.0*k*k)
    a /= a.sum()   # normalize sum=1

    best_seq = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(best_seq, best_seq))
    print("[init] n=", n, "score=", best_score)

    start_time = time.time()
    #Parameters
    max_outer = 10
    t = 0.5  # mixing factor
    while time.time() - start_time < budget_s - 5 and max_outer>0:
        # compute current M
        conv = np.convolve(a, a)
        M = conv.max()
        # Solve LP to maximize sum(g) s.t. conv_f_g <= M
        # Build constraint matrix A (2n-1 x n)
        rows = []
        cols = []
        data = []
        for i in range(n):
            coeff = a[i]
            # positions k where this coeff appears: k = i + j for j in [0,n-1]
            # rows range from i to i + n - 1 inclusive (size n)
            rows.extend(list(range(i, i + n)))
            cols.extend(list(range(n)))
            data.extend([coeff]*n)
        # convert to sparse
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # upper bounds
        b = np.full(2*n-1, M, dtype=np.float64)
        # LP.
        res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                                    bounds=[(0, None)]*n, method='highs',
                                    options={"presolve": True, "time_limit": budget_s - (time.time()-start_time) - 1})
        if not res.success:
            print("[lp] failed:", res.message)
            break
        g0 = res.x
        sum_g0 = g0.sum()
        print(f"[lp] sum_g={sum_g0:.6f}")
        if sum_g0 <= 1.000001:
            print("[lp] no improvement possible")
            break
        # scale to sum 1
        g = g0 / sum_g0
        # mix
        a_new = (1 - t) * a + t * g
        a_new /= a_new.sum()   # re-normalize to sum 1 (should be 1 anyway)
        # compute new score
        conv_new = np.convolve(a_new, a_new)
        new_score = 2.0 * n * conv_new.max()
        print(f"[iter] new_score={new_score:.6f}, best={best_score:.6f}")

        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            # optional reduce mixing factor to accelerate
        else:
            # no improvement, break
            print("[iter] no improvement, stop")
            break
        max_outer -= 1

    # optional hill climbing final
    # Further simple hill climbing of small transfers
    deadline = time.time() + budget_s - 5
    it=0
    while time.time() < deadline and it<2000:
        i,j = np.random.choice(n,2,replace=False)
        if a[i] > a[j]:
            src,dst=i,j
        else:
            src,dst=j,i
        delta = 0.01 * min(a[src], a[dst])
        if delta==0: it+=1; continue
        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta
        a_candidate /= a_candidate.sum()
        new_score = 2.0 * n * np.max(np.convolve(a_candidate, a_candidate))
        if new_score < best_score:
            a = a_candidate
            best_score = new_score
            best_seq = a_candidate.copy()
            print(f"[hill] it={it}, score={best_score:.6f}")
        it+=1

    print("[final] best_score=", best_score)
    return best_seq.tolist()

"""
propose_candidate
=================

Search for a step‑function (non‑negative sequence) with a low value of

    2 * n * max(conv(a, a)) / (sum a)^2

where `conv` is the full discrete convolution.  The algorithm consists of
three stages:

1. **Start from the analytic arcsine construction** (the known asymptotic
   optimum).  The vector is normalised so that its sum equals 1.

2. **Conditional‑gradient (Frank–Wolfe) improvement** – a linear programming
   sub‑problem is solved:
   maximise ∑ gᵢ  subject to  gⱼ≥0  and  (f∗g)ₖ ≤ M   (M = max(f∗f)).
   The solution can carry more total mass than the current f while respecting
   the same bound M.  After normalising the solution to unit sum we mix it
   with the current iterate (mixing coefficient t≈0.5).  Each successful mix
   strictly reduces the objective.

3. **Simulated‑annealing hill‑climbing** – random transfers of mass between
   two entries are performed; moves that improve the score are accepted,
   otherwise they are accepted with a temperature‑dependent probability.
   This phase extracts any remaining improvement that the linear‑program
   step may miss.

The whole routine respects the 1000 s time budget, reports intermediate
scores and finally returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper: evaluation of the objective (the same formula used in
# `evaluate_sequence` when the input already satisfies the security checks)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))  (the denominator is 1 because `seq` is
    normalised to sum==1)."""
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming sub‑problem (step 2)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   1·gᵀ·1
        subject to  A·g ≤ M·1   (A encodes the convolution f∗g)
                    g ≥ 0
    where ``M = max(f∗f)``.
    Returns the optimal g (not normalised) or ``None`` if the solver fails.
    """
    n = f.size
    rows = []          # row indices of non‑zero entries in A
    cols = []          # column indices
    data = []          # values (always a coefficient from f)

    # Build the Toeplitz‑like matrix A where A[k, j] = f[k‑j] for 0≤k‑j<n
    # For a fixed column j, the non‑zero rows are j … j+n‑1 (inclusive)
    # with coefficient f[0], f[1], … , f[n‑1] respectively.
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(f.tolist())

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  ⇔ minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Highs is fast and respects a time‑limit (seconds)
    remaining = max(0.0, budget - (time.time() - start_time) - 0.5)
    lp_res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": remaining},
    )
    if not lp_res.success:
        return None
    return lp_res.x


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing (step 3)
# ----------------------------------------------------------------------
def _anneal(
    seq: np.ndarray,
    best_score: float,
    time_limit: float,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass while using a simple annealing
    schedule.  Returns the best sequence encountered."""
    n = seq.size
    a = seq.copy()
    cur_score = best_score
    best_seq = a.copy()
    best_val = cur_score

    # temperature schedule – starts fairly warm and cools slowly
    T0 = 1e-2
    Tmin = 1e-6
    decay = (Tmin / T0) ** (1.0 / max_iters)

    T = T0
    it = 0
    while time.time() < time_limit and it < max_iters:
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # choose a transfer amount – a random fraction of the *source*
        # (bounded to keep the destination non‑negative)
        src, dst = (i, j) if a[i] > a[j] else (j, i)
        if a[src] == 0.0:
            it += 1
            continue
        # transfer up to 30 % of the source mass (large enough to explore)
        delta = np.random.rand() * 0.30 * a[src]

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # renormalise – numerical drift may appear after many moves
        cand /= cand.sum()

        cand_score = _score(cand)

        improve = cand_score < cur_score
        if improve or np.random.rand() < np.exp(-(cand_score - cur_score) / T):
            a = cand
            cur_score = cand_score
            if cand_score < best_val:
                best_val = cand_score
                best_seq = cand.copy()
                print(f"[anneal] it={it:05d}, new best={best_val:.6f}")

        # temperature decay
        T = max(T * decay, Tmin)
        it += 1

    return best_seq, best_val

    """
