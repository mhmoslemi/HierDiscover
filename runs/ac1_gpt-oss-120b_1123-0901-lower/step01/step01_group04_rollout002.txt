"""
propose_candidate
=================

A deterministic algorithm that improves the classical arcsine construction.
It alternates between a linear‑programming step (maximising the total mass
that can be added without raising any convolution entry above the current
maximum) and a convex combination with the current step‑function.
The LP enforces the convolution constraints
    (f * b)_k ≤ M   (k = 0,…,2n−2)
and the variables b_k are non‑negative.
Mixing the new distribution back into f reduces the quantity

    2·n·max_k (f*f)_k

which is exactly the value returned by ``evaluate_sequence`` (the sum is kept
equal to 1, therefore the denominator is unchanged).

After a few LP iterations a lightweight random hill‑climbing phase tweaks the
result until the time budget expires.  The whole routine runs under the
provided ``budget_s`` (default 1000 s) and returns a plain list of non‑negative
coefficients.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    # -------------------------------------------------------------
    # 1.  Setup – length of the sequence and random seed
    # -------------------------------------------------------------
    np.random.seed(seed)
    # Prefer the length of a previously found sequence if it exists.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000                             # modest size – faster LPs
    # a manual override
    if "length" in kwargs:
        n = int(kwargs["length"])
    # keep the problem tractable (memory ≈ n²)
    n = max(500, min(n, 3000))

    # -------------------------------------------------------------
    # 2.  Initialise with the discrete arcsine distribution
    # -------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()                     # normalise to sum = 1

    # -------------------------------------------------------------
    # 3.  Helper to obtain the maximal value of the self‑convolution
    # -------------------------------------------------------------
    def max_conv(x: np.ndarray) -> float:
        """
        Compute max_k (x*x)_k using an FFT based convolution.
        The returned value is the raw maximum (the denominator is
        assumed to be 1, i.e. sum(x) == 1).
        """
        m = x.size
        N = 1 << ((2 * m - 1).bit_length())   # next power of two
        fx = np.fft.rfft(x, N)
        conv = np.fft.irfft(fx * fx, N)[: 2 * m - 1]
        return float(conv.max())

    # -------------------------------------------------------------
    # 4.  Record the best sequence seen so far
    # -------------------------------------------------------------
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # -------------------------------------------------------------
    # 5.  Pre‑compute the index pattern for the Toeplitz convolution matrix
    # -------------------------------------------------------------
    # row indices  i + j   (i = 0..n-1, j = 0..n-1)
    row_idx = np.add.outer(np.arange(n), np.arange(n)).ravel()
    # column indices are simply the second index j repeated n times
    col_idx = np.repeat(np.arange(n), n)

    # -------------------------------------------------------------
    # 6.  Main optimisation loop (LP + mixing)
    # -------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # small safety margin
    iteration = 0
    # step size for convex combination (will be decreased)
    t = 0.5
    # store bounds for linprog once
    bounds = [(0.0, None)] * n

    while time.time() < deadline and iteration < 30:
        M = max_conv(a)               # current worst convolution value
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter {iteration:02d}] improved score = {best_score:.6f}")

        # ---- Linear programme: maximise sum(b)  subject to  (a * b) ≤ M  ----
        # Build the sparse Toeplitz matrix of the convolution operator.
        data = np.tile(a, n)                  # each column receives a copy of a
        A_ub = sp.csr_matrix((data, (row_idx, col_idx)),
                             shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Objective: minimise -∑ b_i   (i.e. maximise the sum)
        c = -np.ones(n, dtype=np.float64)

        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={"presolve": True})
        if not res.success:
            print(f"[iter {iteration:02d}] LP failed – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            print(f"[iter {iteration:02d}] trivial LP solution, stop.")
            break

        # Normalise the new vector to keep the overall sum equal to 1
        g = b / sum_b

        # Convex combination – gradually shrink the step size
        a = (1.0 - t) * a + t * g
        a /= a.sum()                     # guard against numerical drift

        # Reduce the mixing parameter a little each round
        t = max(0.05, t * 0.85)

        iteration += 1

    # -------------------------------------------------------------
    # 7.  A light random hill‑climbing phase (exploits any remaining slack)
    # -------------------------------------------------------------
    it = 0
    while time.time() < deadline and it < 3000:
        i, j = np.random.choice(n, size=2, replace=False)
        # move a small amount from the larger entry to the smaller one
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        delta = 0.005 * min(a[src], a[dst])
        if delta == 0.0:
            it += 1
            continue
        a2 = a.copy()
        a2[src] -= delta
        a2[dst] += delta
        # quick evaluation – sum stays ≈1, so score = 2·n·max(conv)
        M2 = max_conv(a2)
        score2 = 2.0 * n * M2
        if score2 < best_score:
            best_score = score2
            best_seq = a2.copy()
            a = a2
            print(f"[hill {it:04d}] improved score = {best_score:.6f}")
        it += 1

    # -------------------------------------------------------------
    # 8.  Final sanity check and return
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score (reported) = {best_score:.6f}, "
          f"evaluate_sequence() = {final_score:.6f}")
    return best_seq.tolist()
