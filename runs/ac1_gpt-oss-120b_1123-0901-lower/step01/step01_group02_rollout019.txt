def compute_conv(a):
    return np.convolve(a, a)

def max_gradient_step(a, conv, S, lr):
    n = a.shape[0]
    M = conv.max()
    k_max = int(conv.argmax())
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]
    c = 2.0 * M / S  # constant term for gradient derived earlier
    grad = 2.0 * rev - c
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new = a_new * (S / sum_new)
    else:
        a_new = a.copy()
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new

def softmax_gradient_step(a, conv, S, lr, beta):
    n = a.shape[0]
    # compute weights
    scaled = beta * conv
    max_scaled = scaled.max()
    expw = np.exp(scaled - max_scaled)  # shape 2n-1
    # compute cross-correlation
    corr = np.correlate(expw, a, mode='full')
    # gradient for each i:
    grad = 2.0 * corr[(n-1) - np.arange(n)]
    # apply step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new = a_new * (S / sum_new)
    else:
        a_new = a.copy()
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new

def random_move_step(a, S, move_factor):
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = a[i] * move_factor * np.random.rand()
    if delta <= 0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.maximum(a_new, 0.0)
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new = a_new * (S / sum_new)
    else:
        a_new = a.copy()
    np.clip(a_new, 0.0, 1000.0, out=a_new)

def optimizer(a_initial, deadline):
    a = a_initial.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    conv = np.convolve(a, a)
    M = conv.max()
    best_a = a.copy()
    best_M = M

    lr0 = 0.6
    decay = 5e-5
    beta = 0.2
    beta_inc = 0.001
    move_factor = 0.2

        # adapt learning rate
        lr = lr0 / (1.0 + decay * iter)

        # choose step
        r = np.random.rand()
        if r < 0.7:
            # softmax gradient
            a_new = softmax_gradient_step(a, conv, S, lr, beta)
        elif r < 0.9:
            # max gradient
            a_new = max_gradient_step(a, conv, S, lr)
        else:
            # random move
            a_new = random_move_step(a, S, move_factor)

        # compute new conv and max
        conv_new = np.convolve(a_new, a_new)
        M_new = conv_new.max()

        # accept if improved or small chance...
        if M_new <= M:
            accept = True
        else:
            # probability to accept worsening moves (annealing)
            # use temperature based on beta maybe
            T = max(0.001, (beta + 0.1) * 0.01)  # heuristic
            accept = np.random.rand() < np.exp(-(M_new - M) / (T * S))
        if accept:
            a = a_new
            conv = conv_new
            M = M_new
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
                print(f"[{time.time() - start:.1f}s] iter {iter} new best M={best_M:.6f}")
        else:
            # occasionally accept a worse move to escape plateau
            if np.random.rand() < 0.001:
                a = a_new
                conv = conv_new
                M = M_new

        # schedule adjustments
        iter += 1
        # increase beta (focus on max)
        beta = min(beta + beta_inc, 5.0)
        # periodically reduce move_factor
        if iter % 5000 == 0:
            move_factor = max(move_factor * 0.9, 0.001)
        # jitter occasionally
        if iter % 2000 == 0:
            jitter = np.random.randn(n) * 0.001 * S / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())
            conv = np.convolve(a, a)
            M = conv.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
    return best_a, best_M

"""
Hybrid optimizer for the step‑function “max‑convolution” problem.

The objective after fixing the sum Σa_i = √(2 n) is simply
    E(a) = max_k (a * a)[k]            (a*a is the discrete convolution).

The algorithm mixes three complementary moves:

1. **Soft‑max weighted gradient step** – uses a smooth approximation of the
   max (soft‑max) to push down all large convolution entries at once.
2. **Exact‑max gradient step** – the classic gradient derived from the index
   of the current maximum (as in the reference implementation).
3. **Random mass‑transfer moves** – pick two indices i≠j and move a small
   amount of mass from i to j, allowing the search to escape local minima.

The optimizer keeps the non‑negative orthant, re‑normalises the vector after
each move (so the sum stays √(2 n)), and accepts every improving move.
Worsening moves are occasionally accepted with a tiny probability (simulated
annealing) to encourage exploration.  Periodic “jitter” and optional
symmetrisation further help to avoid stagnation.

Multiple restarts (uniform, triangular and the supplied `height_sequence_1`)
are performed until the wall‑clock budget (≈1000 s) is exhausted.
The best sequence seen is returned.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the self‑convolution a*a (full length)."""
    return np.convolve(a, a)


def _max_gradient_step(a: np.ndarray, conv: np.ndarray, S: float, lr: float) -> np.ndarray:
    """
    Gradient step derived from the index of the current maximum.
    grad = 2 * rev - (2 * M) / S   where rev[p] = a_{k_max-p} (zero outside).
    """
    n = a.shape[0]
    M = conv.max()
    k_max = int(conv.argmax())

    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]

    const = (2.0 * M) / S
    grad = 2.0 * rev - const

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)

    s = a_new.sum()
    if s > 0.0:
        a_new = a_new * (S / s)
    else:
        a_new = a.copy()

    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


def _softmax_gradient_step(a: np.ndarray, conv: np.ndarray, S: float,
                           lr: float, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate  L = log Σ exp(beta * conv)
    which is  grad_i = 2 * Σ_k w_k * a_{k-i}
    with w_k = exp(beta * conv_k) / Σ_j exp(beta * conv_j).
    The convolution with the reversed vector is used for efficiency.
    """
    n = a.shape[0]

    # soft‑max weights (stable)
    shifted = beta * (conv - conv.max())
    w = np.exp(shifted)
    w /= w.sum()

    # reversed a
    a_rev = a[::-1]

    # w * a_rev  (full convolution) – length = (2n‑1) + n‑1 = 3n‑2
    conv_wa = np.convolve(w, a_rev)

    # gradient entries are taken from indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_wa[(n - 1):(n - 1 + n)]

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)

    s = a_new.sum()
    if s > 0.0:
        a_new = a_new * (S / s)
    else:
        a_new = a.copy()

    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


def _random_move_step(a: np.ndarray, S: float, move_factor: float) -> np.ndarray:
    """
    Transfer a random fraction of mass from index i to a different index j.
    move_factor controls the maximal fraction moved.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a  # nothing to do

    delta = a[i] * move_factor * np.random.rand()
    if delta <= 0.0:
        return a

    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.maximum(a_new, 0.0)

    s = a_new.sum()
    if s > 0.0:
        a_new = a_new * (S / s)
    else:
        a_new = a.copy()

    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


# ------------------------------------------------------------
# Core optimisation loop (single restart)
# ------------------------------------------------------------
def _run_optimisation(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a hybrid optimisation starting from `a_start`.
    Returns the best (lowest max‑convolution) sequence seen and its max value.
    """
    a = a_start.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    conv = _conv_self(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M

    it = 0
    lr0 = 0.8          # initial learning‑rate
    decay = 5e-5       # decay of the learning‑rate
    beta = 0.2         # initial soft‑max temperature
    beta_inc = 0.001   # increase per iteration (focuses on true max)
    move_factor = 0.25 # start with relatively large random moves

    start = time.time()
    while time.time() < deadline:
        # learning‑rate schedule
        lr = lr0 / (1.0 + decay * it)

        # pick a move type
        r = np.random.rand()
        if r < 0.65:
            a_new = _softmax_gradient_step(a, conv, S, lr, beta)
        elif r < 0.90:
            a_new = _max_gradient_step(a, conv, S, lr)
        else:
            a_new = _random_move_step(a, S, move_factor)

        # evaluate the candidate
        conv_new = _conv_self(a_new)
        M_new = conv_new.max()

        # accept if it improves, otherwise accept very rarely
        if M_new <= M:
            accept = True
        else:
            # tiny simulated‑annealing probability
            T = max(0.001, (beta + 0.1) * 0.02)
            accept = np.random.rand() < math.exp(-(M_new - M) / T)

        if accept:
            a, conv, M = a_new, conv_new, M_new
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
                print(f"[{time.time() - start:.1f}s] it {it:06d} new best M = {best_M:.6f}")

        # occasional aggressive jitter to leave shallow plateaus
        if it % 2000 == 0 and it > 0:
            jitter = np.random.randn(n) * (0.001 * S / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())
            conv = _conv_self(a)
            M = conv.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
                print(f"[{time.time() - start:.1f}s] jitter improved M = {best_M:.6f}")

        # optional symmetrisation (helps in many runs)
        if it % 5000 == 0 and it > 0:
            a = 0.5 * (a + a[::-1])
            a = a * (S / a.sum())
            conv = _conv_self(a)
            M = conv.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
                print(f"[{time.time() - start:.1f}s] symmetrisation improved M = {best_M:.6f}")

        # parameter schedules
        beta = min(beta + beta_inc, 5.0)          # make the soft‑max sharper over time
        if it % 4000 == 0 and it > 0:
            move_factor = max(move_factor * 0.85, 0.001)

        it += 1

    return best_a, best_M


# ------------------------------------------------------------
# Public entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s`
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # -----------------------------------------------------------------
    # Determine a default length (use the known construction if present)
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore
        n = base_seq.shape[0]
    except Exception:
        n = 2100
        base_seq = np.ones(n, dtype=float)

    # Normalise to the required sum √(2 n)
    target_sum = math.sqrt(2.0 * n)
    base_seq = base_seq * (target_sum / base_seq.sum())
    base_seq = np.clip(base_seq, 0.0, 1000.0)

    # -----------------------------------------------------------------
    # Prepare a handful of diverse starting points
    starters = []

    # 1) the known SOTA construction
    starters.append(base_seq.copy())

    # 2) uniform distribution
    uniform = np.full(n, 1.0, dtype=float)
    uniform = uniform * (target_sum / uniform.sum())
    starters.append(uniform)

    # 3) triangular shape (high in the centre, zero on the edges)
    x = np.arange(n, dtype=float)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) * 2.0 / (n - 1)
    tri = np.maximum(tri, 0.0)
    tri = tri * (target_sum / tri.sum())
    starters.append(tri)

    # 4) a random start (drawn from an exponential to create a few large values)
    rand = np.random.exponential(scale=1.0, size=n)
    rand = rand * (target_sum / rand.sum())
    starters.append(rand)

    best_overall_seq = None
    best_overall_M = float("inf")

    # -----------------------------------------------------------------
    # Run independent optimisation runs (restarts) until time runs out
    for start_idx, a0 in enumerate(starters):
        if time.time() >= deadline:
            break
        print(f"[init] start {start_idx} (length={n})")
        # Give each run roughly an equal share of the remaining budget,
        # but also allow early termination if a very good solution is found.
        run_deadline = min(deadline, time.time() + (budget_s / 4.0))

        a_best, M_best = _run_optimisation(a0, run_deadline)

        print(f"[run {start_idx}] best M = {M_best:.6f}")

        if M_best < best_overall_M:
            best_overall_M = M_best
            best_overall_seq = a_best.copy()
            print(f"[global] new overall best M = {best_overall_M:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence (optional but handy)
    if best_overall_seq is None:
        # Fallback – should never happen
        best_overall_seq = uniform.copy()
        best_overall_M = _conv_self(best_overall_seq).max()

    try:
        final_score = evaluate_sequence(best_overall_seq.tolist())
        print(f"[final] evaluation score = {final_score:.6f}")
    except Exception:
        print("[final] evaluation failed – returning raw sequence")

    # Return as plain Python list
    return best_overall_seq.tolist()
