def propose_candidate(seed=42, budget_s=1000):

    deadline = time.time() + budget_s - 0.5

    # Start with length n: choose 2000 for high potential
    n = 2000
    # initial sequences (list of candidates)
    factors = [0.5, 0.8, 1.0, 1.3, 1.5, 1.8, 2.0, 2.5, 3.0]
    best_seq = None
    best_score = float('inf')
    for factor in factors:
        if time.time() > deadline:
            break
        eps = factor / n
        r = 1 - eps
        if r <= 0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # Also uniform
    seq = np.full(n, 1.0 / n)
    score = evaluate_sequence(seq.tolist())
    if score < best_score:
        best_score = score
        best_seq = seq.copy()

    # main descent
    seq = best_seq.copy()
    step = 0.5
    it = 0
    no_improve = 0
    while time.time() < deadline:
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # determine active indices
        rel_tol = 1e-4  # tolerance on relative max
        tol = max(max_val * rel_tol, 1e-12)
        active = np.where(conv >= max_val - tol)[0]
        act = np.zeros_like(conv)
        act[active] = 1.0
        # gradient via convolution
        grad_full = np.convolve(seq[::-1], act)
        grad = grad_full[n-1:n-1+n]
        # Normalise gradient magnitude to avoid too small updates
        grad_norm = np.linalg.norm(grad, 1)  # L1 norm
        if grad_norm == 0:
            break
        # compute step size adaptively
        step_size = step / (grad_norm)  # ensures update magnitude roughly step
        # try a few step scales
        improved = False
        for factor_scale in [1.0, 0.5, 0.25]:

            score = evaluate_sequence(new_seq.tolist())

"""
Propose a step‑function (non‑negative sequence) that beats the uniform baseline.
The algorithm combines three ideas:

1.   A geometric seed (a_i ∝ r^i) with several decay factors – the best
     seed is kept as a starting point.
2.   Projected (sub‑)gradient descent on the *soft‑max* of the self‑convolution.
     The gradient of a smooth approximation to max_k (a*a)[k] is
         g = 2 * (a_rev ★ w)   where w is the soft‑max weight vector.
     After each step the vector is projected back onto the probability simplex.
3.   An occasional linear‑programming “refinement” step (the LP described in the
     paper).  With the current a we maximise Σ b_j under the constraints
     (a*b)[k] ≤ M (M = max convolution of a).  The optimal b is mixed with a
     and projected onto the simplex.

The whole search runs for at most 1000 s and keeps the best sequence seen.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project v (any shape) onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_constraint_matrix(a):
    """Build the (2n‑1) × n matrix for the constraints (a * b)[k] ≤ M."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1, dtype=np.int32)
        vals = a[k - idx]          # a_{k‑i}
        rows.extend([k] * len(idx))
        cols.extend(idx.tolist())
        data.extend(vals.tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Main entry point for the competition.  Returns a list of non‑negative
    heights (the step‑function) that yields the smallest objective found
    within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1.  Choose a working length.
    n = 2000                     # large enough to get near the theoretical limit

    # --------------------------------------------------------------
    # 2.  Initialise from several geometric families (plus the uniform one)
    best_seq = None
    best_score = float("inf")
    factors = [0.5, 0.8, 1.0, 1.3, 1.5, 1.8, 2.0, 2.5, 3.0]

    for factor in factors:
        if time.time() > deadline:
            break
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()               # already on the simplex
        score = evaluate_sequence(seq.tolist())
        print(f"[Init] factor={factor:4.2f}  eps={eps:.6e}  score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # uniform baseline
    uni = np.full(n, 1.0 / n)
    score = evaluate_sequence(uni.tolist())
    if score < best_score:
        best_score = score
        best_seq = uni.copy()
    print(f"[Init] uniform  score={score:.6f}")

    # --------------------------------------------------------------
    # 3.  Projected gradient descent on a smooth max (soft‑max)
    seq = best_seq.copy()
    step = 0.02                     # initial L2 step size
    alpha = 400.0                   # smoothing parameter for soft‑max
    it = 0
    no_improve = 0
    max_no_improve = 20000

    while time.time() < deadline:
        # self‑convolution
        conv = np.convolve(seq, seq)          # length 2n‑1
        max_val = conv.max()

        # soft‑max weights (stable version)
        shifted = conv - max_val               # max becomes 0
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of the smooth‑max objective:  g = 2 * (rev(seq) ★ w)
        # rev(seq) is seq[::-1]; the convolution length is 3n‑2
        grad_full = np.convolve(seq[::-1], w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice length n

        # normalise the direction (L2)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        direction = grad / norm

        # try a few step sizes (line‑search like)
        improved = False
        for factor in (1.0, 0.5, 0.25):
            new_seq = seq - step * factor * direction
            new_seq = _project_onto_simplex(new_seq)
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score - 1e-8:
                best_score = score
                best_seq = new_seq.copy()
                seq = new_seq
                improved = True
                step *= 1.02                     # slightly increase step after success
                no_improve = 0
                print(f"[Iter {it:06d}] new best = {best_score:.6f}")
                break
        if not improved:
            step *= 0.99                         # shrink step if no progress
            no_improve += 1

        # occasional random jitter to escape flat regions
        if it % 5000 == 0 and it > 0:
            jitter = np.random.normal(scale=0.005, size=n)
            seq = _project_onto_simplex(seq + jitter)

        # stochastic restart if we have stagnated for a while
        if no_improve >= max_no_improve:
            # restart from a fresh Dirichlet draw
            rnd = np.random.rand(n)
            seq = _project_onto_simplex(rnd)
            step = 0.02
            no_improve = 0
            print(f"[Iter {it:06d}] restart (stagnation)")

        it += 1

    # --------------------------------------------------------------
    # 4.  One LP refinement (if time permits)
    if time.time() < deadline - 5.0:
        try:
            A = _build_constraint_matrix(seq)
            M = np.convolve(seq, seq).max()
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            # maximise Σ b_j  → minimise -Σ b_j
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub,
                          bounds=(0, None), method='highs')
            if res.success and res.x.sum() > 0:
                b = res.x
                b = b / b.sum()                 # bring to the simplex
                t = 0.15                         # mixing parameter
                mixed = (1 - t) * seq + t * b
                mixed = _project_onto_simplex(mixed)
                score = evaluate_sequence(mixed.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = mixed.copy()
                    print(f"[LP] improved score = {best_score:.6f}")
        except Exception as exc:
            # LP failed – we simply ignore it
            print(f"[LP] exception: {exc}")

    # --------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    # evaluate expects a list
    return best_seq.tolist()
