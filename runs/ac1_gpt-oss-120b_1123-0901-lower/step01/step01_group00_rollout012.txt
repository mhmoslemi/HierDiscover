"""
Propose a non‑negative step‑function (vector) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a **soft‑max gradient descent** with a projection onto the
simplex {x_i ≥ 0, ∑x_i = √(2 n)}.

* The objective is scale‑invariant, therefore we keep the ℓ₁‑norm of the
  vector fixed to S = √(2 n).  With this normalisation the evaluation function
  equals the maximal convolution entry, i.e. score = max(conv(a,a)).
* At each iteration we compute the full convolution `c = a * a`.  
  Instead of moving only in the direction of the worst entry we use a
  smooth approximation of the max:
        f_β(c) = (1/β)·log Σ exp(β·c_k) .
  Its gradient is a weighted sum of the gradients of every convolution entry,
  where the weight of entry *k* is proportional to `exp(β·c_k)`.
* The gradient of `c_k = Σ_i a_i a_{k−i}` w.r.t. `a` is `2·a_{k−i}` on the
  overlap region.  Using the weights we obtain the descent direction efficiently
  via a second convolution.
* After a step we project back onto the simplex (Euclidean projection).  A
  momentum term (Nesterov‑style) and occasional random perturbations help to
  escape shallow local minima.
* The temperature `β` is increased slowly (continuation method) so that the
  optimisation first flattens the whole autocorrelation and later concentrates
  on the highest peak.
* Several independent restarts with different lengths (≈ 1 000–2 200) are
  performed within the allotted time.  The best sequence seen so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full convolution using NumPy (fast C implementation)
# ----------------------------------------------------------------------
def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution of two 1‑D arrays."""
    return np.convolve(x, y)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds and
    keeps the best sequence discovered.

    The optimisation uses a soft‑max based gradient descent on the maximum
    convolution entry, with periodic restarts and a continuation schedule for the
    temperature parameter β.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Parameters (can be tweaked on the fly)
    # ------------------------------------------------------------------
    # Lengths we will try – they are large enough to give a good lower bound
    # but still cheap enough for many iterations.
    candidate_lengths = [900, 1200, 1500, 1800, 2100]

    # Global hyper‑parameters for the inner loop
    base_step = 0.3               # base step size (later normalised by ||grad||)
    beta_start = 0.5
    beta_end = 12.0
    beta_ramp = 2500              # iterations over which β grows to beta_end
    max_iters_per_run = 4000
    momentum_coeff = 0.9
    noise_prob = 0.012            # probability of injecting a jitter
    noise_scale = 2e-3            # relative magnitude of the jitter

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # Target ℓ₁‑norm (the normalisation used by the evaluation)
        S = np.sqrt(2.0 * n)

        # ----- initialise the vector -----------------------------------
        if 'height_sequence_1' in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                a0 = src[:n].copy()
            else:
                # pad with tiny random values to reach the required length
                pad = np.random.rand(n - src.size) * 1e-4
                a0 = np.concatenate([src, pad])
        else:
            a0 = np.random.rand(n)

        # project onto the simplex to enforce feasibility
        a = _project_onto_simplex(a0, S)

        # velocity for momentum (initially zero)
        v = np.zeros_like(a)

        # ----------------------------------------------------------------
        # inner optimisation loop
        # ----------------------------------------------------------------
        it = 0
        while time.time() < deadline and it < max_iters_per_run:
            # 1) current convolution and current score (max entry)
            conv = _conv_full(a, a)               # length 2n‑1
            max_conv = conv.max()

            # keep the globally best solution seen so far
            if max_conv < best_score:
                best_score = max_conv
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best → {best_score:.6f} (n={n})")

            # early exit if we already beat the target
            if best_score < 1.503:
                break

            # 2) continuation schedule for β (soft‑max temperature)
            if it < beta_ramp:
                beta = beta_start + (beta_end - beta_start) * (it / beta_ramp)
            else:
                beta = beta_end

            # 3) soft‑max weights (stable by subtracting max)
            #    w_k ∝ exp(β·(conv_k – max_conv))
            w = np.exp(beta * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:                     # numerically unlikely
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # 4) gradient of the smoothed objective.
            #    grad_i = 2 * Σ_k w_k·a_{k‑i}  (indices outside [0,n‑1] are ignored)
            #    This is exactly a convolution of the reversed weight vector with a.
            w_rev = w[::-1]
            grad_full = _conv_full(w_rev, a)      # length 3n‑2
            idx = 2 * n - 2 - np.arange(n)        # slice that yields the needed entries
            grad = 2.0 * grad_full[idx]           # shape (n,)

            # 5) occasional random jitter to escape flat plateaus
            if np.random.rand() < noise_prob:
                jitter = noise_scale * S / n * np.random.randn(n)
                a_jitter = a + jitter
                a = _project_onto_simplex(a_jitter, S)

            # 6) momentum update (Nesterov‑style)
            #    Normalise the step by the gradient magnitude so that the update
            #    size is roughly comparable across iterations.
            grad_norm = np.linalg.norm(grad) + 1e-12
            step = base_step / grad_norm
            v = momentum_coeff * v - step * grad
            a_new = a + v

            # 7) re‑project onto the simplex to maintain feasibility
            a = _project_onto_simplex(a_new, S)

            it += 1

        # stop early if the target has been reached
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: uniform vector if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n, dtype=float)

    # The evaluation routine clips negatives and caps values at 1000, but
    # our construction already respects those bounds.
    return best_seq.tolist()
