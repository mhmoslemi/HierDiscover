"""
Simulated‑annealing optimizer for the step‑function evaluation.

The objective `evaluate_sequence` is scale‑invariant, therefore we first normalize
the candidate to have unit ℓ¹‑norm (∑ a_i = 1).  With this normalization the
score reduces to 2·n·max(a*a) and the problem becomes “minimise the maximum
entry of the self‑convolution subject to a_i ≥ 0 and ∑ a_i = 1”.

We start from the analytically good “inverse‑square‑root’’ (central‑binomial)
shape, normalise it, and then perform a long simulated‑annealing run.  Each
move transfers a small amount of mass from one index to another, keeping the
total sum exactly 1.  An uphill move is accepted with probability exp(−Δ/T);
the temperature T decays geometrically.  The best sequence seen during the run
is returned.

The search is completely self‑contained, uses only NumPy and the provided
`evaluate_sequence` function, and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """Return a NumPy array a_0,…,a_{n‑1} with
    a_i = binom(2*i,i) / 4**i = (2i)!/(i!²·4**i)."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence a_{i+1}=a_i * (2*i+1)/(2*i+2)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def max_convolution(seq: np.ndarray) -> float:
    """Fast helper: maximum entry of the full convolution seq*seq."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function coefficients.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best coefficient list found within the time limit.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Length of the problem.  Use the length of the previously found
    #    sequence if it exists; otherwise fall back to a moderate size.
    # ----------------------------------------------------------------------
    try:

        start_seq = np.array(height_sequence_1, dtype=float)
        # ensure non‑negativity and normalise – the judge will do the same.
        start_seq = np.maximum(start_seq, 0.0)
        start_seq /= max(start_seq.sum(), 1e-12)
    except Exception:
        n = 3400
        start_seq = generate_inverse_sqrt_sequence(n)

    # ----------------------------------------------------------------------
    # 2. Normalise to unit ℓ¹‑norm (makes the objective a pure max‑convolution).
    # ----------------------------------------------------------------------
    total = start_seq.sum()
    if total < 1e-12:
        start_seq = np.full(n, 1.0 / n)
    else:
        start_seq = start_seq / total

    # ----------------------------------------------------------------------
    # 3. Initial evaluation.
    # ----------------------------------------------------------------------
    cur_seq = start_seq.copy()
    try:
        cur_score = evaluate_sequence(cur_seq.tolist())
    except Exception:
        # fallback – should never happen.
        cur_score = float('inf')
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ----------------------------------------------------------------------
    # 4. Simulated annealing parameters.
    # ----------------------------------------------------------------------
    temperature = 0.05                # initial temperature
    decay = 0.9995                    # geometric decay per iteration
    min_temp = 1e-6
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    it = 0
    while time.time() < deadline:
        it += 1

        # pick two distinct indices – i (donor) preferably large, j (receiver)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # if donor is (almost) empty, skip this move
        if cur_seq[i] <= 1e-14:
            continue

        # delta is a random fraction (0–20 %) of the donor amount
        frac = rng.random() * 0.2
        delta = cur_seq[i] * frac

        # propose new sequence (mass transfer i → j)
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate candidate
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # acceptance test (Metropolis)
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temperature, min_temp)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{it}] new best score={best_score:.6f}")

        # cool down
        temperature = max(temperature * decay, min_temp)

        # occasional temperature reset to keep some exploration
        if it % 20000 == 0:
            temperature = max(temperature, 0.01)

    print(f"[final] iterations={it}, best score={best_score:.6f}")

    # Return a plain Python list (any scaling is fine – the evaluator will
    # re‑normalise internally).  We keep the sequence as‑is.
    return best_seq.tolist()
