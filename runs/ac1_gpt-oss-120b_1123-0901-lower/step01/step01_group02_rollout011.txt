def _gradient_step_active(a, target_sum, lr):
    n = a.shape[0]
    b = np.convolve(a, a)
    M = b.max()
    # tolerance: consider values within 1e-9 relative
    eps = 1e-12

    # compute rev_sum
    rev_sum = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx

def project_simplex(v, s):
    # ensure v is np.ndarray
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

"""Propose a step‑function sequence using active‑set projected gradient
descent with random restarts.

The objective after normalising the sum to √(2n) is simply the
maximum value of the discrete convolution a*a.  The algorithm
maintains a feasible non‑negative vector with the required sum,
repeatedly takes a projected gradient step based on the
sub‑gradient of the maximum convolution entry (averaged over all
active indices), and periodically restarts from a new random sparse
support.  Random jitter and occasional larger perturbations are used
to escape plateaus.  The best sequence found before the time budget
expires is returned.
"""

import time
import math
import numpy as np

def _convolution_max(a: np.ndarray):
    """Return (max_value, indices_where_max, full_convolution)."""
    b = np.convolve(a, a)
    M = b.max()
    # include all indices whose value is within a tiny tolerance of the max
    eps = max(1e-12, M * 1e-12)
    k_max = np.where(b >= M - eps)[0]
    return M, k_max, b

def _project_to_target_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip negative entries to zero and rescale so that the sum equals target_sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        # Should not happen, fall back to uniform distribution
        v = np.full_like(v, target_sum / v.size)
    return v

def _gradient_step_active(a: np.ndarray, target_sum: float, lr: float) -> np.ndarray:
    """Projected gradient step using the sub‑gradient of the max‑convolution."""
    n = a.shape[0]
    M, active, _ = _convolution_max(a)

    # accumulate the reversed contributions for all active indices
    rev_sum = np.zeros_like(a)
    idx = np.arange(n)

    for k in active:
        j = k - idx                     # positions that contribute for this k
        mask = (j >= 0) & (j < n)
        rev_sum[mask] += a[j[mask]]

    if active.size > 0:
        grad = (2.0 * rev_sum) / active.size   # average sub‑gradient
    else:
        grad = np.zeros_like(a)

    a_new = a - lr * grad
    a_new = _project_to_target_sum(a_new, target_sum)

    # Enforce the hard limits required by evaluate_sequence
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new

def _random_sparse_vector(n: int, target_sum: float,
                         min_spikes: int = 80, max_spikes: int = 250) -> np.ndarray:
    """Create a random non‑negative vector with a sparse support."""
    m = np.random.randint(min_spikes, max_spikes + 1)

    a = np.zeros(n, dtype=float)
    a[pos] = 1.0
    a = _project_to_target_sum(a, target_sum)
    return a
