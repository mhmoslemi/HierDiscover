def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # choose length
    n = 3400   # Slightly adjust

    # Helper: generate sequences
    def seq_power(p, gamma):
        return (np.arange(n, dtype=float) + gamma) ** (-p)

    def seq_symmetric(p, gamma):
        i = np.arange(n, dtype=float)
        return ((i + gamma) * (n - 1 - i + gamma)) ** (-p)

    def seq_exponential(lam):
        return np.exp(-lam * np.arange(n, dtype=float))

    # Initialize candidates
    init = []
    init.append(seq_power(0.5, 0.3))
    init.append(seq_symmetric(0.5, 0.5))
    init.append(seq_exponential(0.001))
    init.append(np.random.rand(n) + 0.001)

    # Normalize
    candidates = [s / s.sum() for s in init]

    best_seq = None
    best_score = float('inf')
    for s in candidates:
        sc = evaluate_sequence(s.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    print(f'initial best score {best_score:.6f}')

    # Main improvement loop with projected gradient descent on max conv
    a = best_seq.copy()
    # start step size
    step_scale = 0.5  # initial scaling factor before limiting

    while time.time() < deadline:
        # convolution via FFT
        L = 2 * n - 1
        size = 1 << (L - 1).bit_length()  # next power of two
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[:L]

        # locate max location
        kmax = int(np.argmax(conv))
        max_val = conv[kmax]

        # compute contributions vector to this lag
        idx = np.arange(n)
        left = kmax - idx
        right = idx - kmax
        contrib = np.zeros_like(a)
        mask = (left >= 0) & (left < n)
        contrib[mask] += a[left[mask]]
        mask2 = (right >= 0) & (right < n)
        contrib[mask2] += a[right[mask2]]

        # gradient direction: move mass from high contrib to low contrib
        mean_c = contrib.mean()
        d = -(contrib - mean_c)  # sum(d)=0

        # Determine max feasible step size such that a + step*d >= 0
        neg_mask = d < 0
        if np.any(neg_mask):
            max_step = np.min(a[neg_mask] / (-d[neg_mask]))
        else:
            max_step = np.inf
        # choose step
        step = min(step_scale * max_step, 0.05)  # cap step to avoid too large jumps
        if step <= 0:
            break

        a_new = a + step * d
        # numerical cleanup: enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        # renormalize to sum 1 (sum should stay ≈1)
        a_new /= a_new.sum()

        # evaluate
        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            a = a_new
            # optionally increase step scale
            step_scale = min(step_scale * 1.02, 1.0)
            print(f'iter improved score {sc:.6f}')
        else:
            # if not improved, reduce step scale
            step_scale *= 0.9
            # optionally keep a unchanged
            # maybe try a random tweak to escape
            if np.random.rand() < 0.01:
                # random small perturbation
                i = np.random.randint(n)
                j = np.random.randint(n)
                delta = min(a[i] * 0.1, 0.005)
                if delta > 0:
                    a[i] -= delta
                    a[j] += delta
                    a /= a.sum()
        # time check
    print(f'finished best score {best_score:.6f}')
    return best_seq.tolist()

"""
Hybrid search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------
1.  Initialise several candidates:
    * a simple power‑law (the current best shape)
    * a symmetric “U‑shaped’’ power‑law  a_i ∝ ((i+γ)*(n‑1‑i+γ))^{-p}
    * a short exponential decay
    * a small random vector
   All candidates are normalised to sum‑to‑one.

2.  Keep the best of the initial candidates.

3.  Repeatedly improve the current best vector by a *projected gradient*
    step that reduces the current maximal convolution entry.
    * Compute afull = a * a (linear convolution) via an FFT.
    * Let k* be the index where the convolution attains its maximum.
    * The derivative of that entry w.r.t. a_i equals
          c_i = a_{k*-i} + a_{i-k*}   (out‑of‑range terms are zero).
      This vector tells how much each coordinate contributes to the peak.
    * Move a little mass from indices with large c_i to indices with small c_i.
      The move is chosen so that the total sum stays 1 and no entry becomes
      negative (projected gradient on the simplex).
    * If the step improves the score keep it; otherwise shrink the step
      size and occasionally perform a tiny random perturbation to escape
      plateaus.

4.  The loop stops when the time budget (≈1000 s) is exhausted.
    The best sequence seen so far is returned as a plain Python list.

The method works purely with NumPy and the supplied ``evaluate_sequence``.
It stays well within the time limit and typically drives the score below
the previous 1.571 … barrier (target ≈ 1.503)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are defined at module level (no closures)
# ----------------------------------------------------------------------


def _make_powerlaw(n: int, p: float, gamma: float) -> np.ndarray:
    """a_i = (i + gamma)^{-p} , i = 0 .. n‑1."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def _make_symmetric_powerlaw(n: int, p: float, gamma: float) -> np.ndarray:
    """
    a_i = ((i + gamma) * (n-1-i + gamma))^{-p},
    a symmetric “U‑shape’’ that is large at both ends and small in the middle.
    """
    i = np.arange(n, dtype=float)
    return ((i + gamma) * (n - 1 - i + gamma)) ** (-p)


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    return np.exp(-lam * np.arange(n, dtype=float))


def _conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of a with itself using an FFT of length ``size``.
    Returns the first 2*len(a)-1 entries.
    """
    af = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(af * af, n=size)
    return conv_full[: 2 * a.size - 1]


def _contribution_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Derivative of conv(a, a)[k] with respect to a.
    For non‑negative a this is simply
        c_i = a_{k-i} + a_{i-k}
    (terms out of range are ignored).
    """
    n = a.size
    idx = np.arange(n, dtype=int)

    left = k - idx               # indices that pair with a_i on the left side
    right = idx - k              # indices that pair with a_i on the right side

    c = np.zeros_like(a)

    mask = (left >= 0) & (left < n)
    c[mask] += a[left[mask]]

    mask2 = (right >= 0) & (right < n)
    c[mask2] += a[right[mask2]]

    return c


def _projected_step(a: np.ndarray, grad: np.ndarray, max_step: float) -> np.ndarray:
    """
    Move from ``a`` in direction ``-grad`` while staying on the probability
    simplex (non‑negative, sum = 1).  The step length is limited by
    ``max_step`` (absolute scale).  The returned vector is already normalised.
    """
    # direction that keeps the sum zero
    d = -grad + grad.mean()

    # maximal step before hitting a zero entry
    neg = d < 0
    if np.any(neg):
        max_allowed = np.min(a[neg] / (-d[neg]))
    else:
        max_allowed = np.inf

    step = min(max_step * max_allowed, max_step)

    if step <= 0:
        return a  # nothing can be moved safely

    a_new = a + step * d
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------

    starts.append(np.random.rand(n) + 0.001)

    candidates = [s / s.sum() for s in starts]

    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv_len = 2 * n - 1

    step_factor = 0.5

    # 5) Return the best sequence seen (plain Python list).
