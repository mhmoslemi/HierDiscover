"""
Propose a candidate step‑function using a fast incremental convolution update
and simulated‑annealing hill‑climbing.

The algorithm
---------------
* Initialise a non‑negative sequence.  By default we use the discrete arcsine
  distribution (known to be close to optimal) or a previously found global
  variable ``height_sequence_1`` if it exists.
* Keep the sequence normalised to sum ≈ 1 (scale‑invariant for the objective).
* Repeatedly propose a small transfer of mass from a “large” index to a
  “small” index.  The source index is chosen among the entries that contribute
  most to the currently maximal convolution entry; the destination is chosen
  uniformly at random.
* The convolution of the current sequence with itself is stored and updated
  incrementally in O(n) time using the fact that moving a mass δ from `src`
  to `dst` changes the convolution by a known linear term plus a tiny quadratic
  correction.
* Accept a move if it improves the objective, otherwise accept with a
  temperature‑dependent probability (simulated annealing).  The temperature
  decays linearly over the allotted time.
* Periodically recompute the full convolution to avoid drift from floating‑
  point error and keep track of the best sequence seen.
* Return the best sequence as a plain ``list[float]``.

This method performs many cheap iterations (hundreds of thousands in the
1000‑second budget) and typically pushes the evaluation below the arcsine
baseline (≈ 1.571).  The code respects all constraints of ``evaluate_sequence``.
"""

import time
import math
import numpy as np


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Return the evaluation score given a convolution array."""
    if total < 0.01:
        return float("inf")
    return float(2.0 * n * conv.max() / (total * total))


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) that tries to minimise
    ``evaluate_sequence``.  The function is deterministic for a given ``seed``.
    """
    np.random.seed(seed)

    # -------------------------------------------------
    # 1.  Determine the length of the sequence.
    # -------------------------------------------------
    length = kwargs.get("length")
    if length is None:
        try:
            length = len(height_sequence_1)  # type: ignore[name-defined]
        except Exception:
            length = 6027
    n = int(length)

    # -------------------------------------------------
    # 2.  Initialise the sequence.
    # -------------------------------------------------
    try:
        a = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if a.shape[0] != n:
            raise ValueError
    except Exception:
        # discrete arcsine distribution: a_k = C·(2k choose k)/4**k
        k = np.arange(n, dtype=np.float64)
        # use recurrence to avoid huge binomials
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
        a /= a.sum()                     # normalise to sum = 1

    # ensure non‑negative and normalised (tiny numerical adjustments)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a[:] = 1.0 / n
    else:
        a /= a_sum

    # -------------------------------------------------
    # 3.  Pre‑compute the full convolution and initial score.
    # -------------------------------------------------
    conv = np.convolve(a, a)
    cur_score = _score_from_conv(conv, a.sum(), n)
    best_seq = a.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # -------------------------------------------------
    # 4.  Simulated‑annealing hill climbing.
    # -------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin
    pad = n - 1                               # amount of zero padding on each side
    iteration = 0
    # temperature schedule (linear cooling)
    T_start = 0.02 * best_score   # a few percent of the score
    T_end = 1e-5 * best_score

    while time.time() < deadline:
        # recompute full convolution occasionally to keep numerical error low
        if iteration % 5000 == 0 and iteration > 0:
            conv = np.convolve(a, a)

        # -------------------------------------------------
        # 4a.  Identify the shift with maximal convolution value.
        # -------------------------------------------------
        k_max = int(conv.argmax())                       # shift where max occurs
        # range of indices that actually contribute to this shift
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)

        # pick a source index among those that contribute the most
        if i_range.size > 0:
            products = a[i_range] * a[k_max - i_range]
            src = int(i_range[products.argmax()])
        else:
            src = np.random.randint(n)

        # pick a destination distinct from src (preferably outside the overlapping region)
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        # -------------------------------------------------
        # 4b.  Propose moving a fraction of mass from src to dst.
        # -------------------------------------------------
        # choose a random fraction; allow relatively large moves to explore the space
        frac = np.random.rand() * 0.5 + 0.05          # 5 % … 55 % of src mass
        delta = frac * a[src]
        if delta <= 0.0:
            iteration += 1
            continue

        # incremental update of the convolution
        # a shifted view: a_shift(idx)[k] = a[k-idx] (zero outside)
        a_padded = np.pad(a, (pad, pad), mode="constant")
        # slices giving the shifted versions
        s_src = a_padded[pad - src : pad - src + 2 * n - 1]
        s_dst = a_padded[pad - dst : pad - dst + 2 * n - 1]
        diff = s_dst - s_src                       # a_shift(dst) - a_shift(src)

        # new convolution (linear part)
        new_conv = conv + 2.0 * delta * diff

        # quadratic corrections from Δ*Δ
        # positions: 2*src, 2*dst, src+dst
        pos_src = 2 * src
        pos_dst = 2 * dst
        pos_sum = src + dst
        if pos_src < new_conv.size:
            new_conv[pos_src] += delta * delta
        if pos_dst < new_conv.size:
            new_conv[pos_dst] += delta * delta
        if pos_sum < new_conv.size:
            new_conv[pos_sum] -= 2.0 * delta * delta

        # -------------------------------------------------
        # 4c.  Compute the new objective value.
        # -------------------------------------------------
        new_score = _score_from_conv(new_conv, a.sum(), n)   # sum unchanged by the transfer

        # -------------------------------------------------
        # 4d.  Acceptance test (simulated annealing).
        # -------------------------------------------------
        # temperature decreasing linearly with elapsed time
        elapsed = time.time() - start_time
        t_frac = min(1.0, elapsed / budget_s)
        T = (1.0 - t_frac) * T_start + t_frac * T_end

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # apply the move
            a[src] -= delta
            a[dst] += delta
            # enforce non‑negativity (numerical safety)
            a[src] = max(a[src], 0.0)
            a[dst] = max(a[dst], 0.0)

            conv = new_conv
            cur_score = new_score

            # keep track of the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[improve] it={iteration:06d}, score={best_score:.6f}")

        # occasionally renormalise to combat drift
        if iteration % 10000 == 0 and iteration > 0:
            total = a.sum()
            if total != 0.0:
                a /= total

        iteration += 1

    # -------------------------------------------------
    # 5.  Final evaluation (using the provided function for safety).
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] iterations={iteration}, best score={final_score:.6f}")

    return best_seq.tolist()
