def project_onto_simplex(v):
    """Projects onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_onto_simplex(v):
    """Euclidean projection of vector onto simplex {x >=0, sum(x)=1}."""
    # Implements algorithm from Duchi et al., 2008.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all nonpositive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function using a smoothed‐max gradient descent.

The evaluation metric 2·n·max(conv(a,a)) / (sum(a)^2) is homogeneous,
so we keep Σa_i = 1.  The algorithm works with the following ideas:

*   Use the soft‑max (log‑sum‑exp) surrogate of the peak of the
    self‑convolution:   F_p(a)= (1/p)·log Σ_k exp(p·conv_k).
    Its gradient can be written in closed form as a correlation,
    which we compute efficiently with FFTs.

*   Start from the known good construction (the global variable
    ``height_sequence_1``) or, if unavailable, from the central‑binomial
    coefficients a_i = C(2i,i)/4^i.

*   Iterate:
        – compute conv = a*a (linear convolution) and the current score;
        – form a soft‑max weight w_k ∝ exp(p·(conv_k−max_conv))   (p>0);
        – the gradient of F_p is 2·corr(w,a) and is obtained by a single
          FFT–based correlation;
        – take a projected (onto the probability simplex) gradient step
          with a diminishing step‑size; a simple back‑tracking line‑search
          is used to guarantee non‑increase of the true objective;
        – slowly increase the smoothing parameter p, turning the surrogate
          into the true max‑operator.

*   Occasionally inject a tiny random perturbation to escape plateaus
  and keep the best sequence seen so far.

The whole routine runs until the allotted CPU time (~1000 s) is exhausted
and returns the best coefficient list found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())
