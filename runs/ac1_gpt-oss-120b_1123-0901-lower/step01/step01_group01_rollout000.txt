def objective(seq):
    seq = np.asarray(seq, dtype=np.float64)

    if sum_a < 0.01:

"""
Propose a step‑function (non‑negative sequence) with a low value of the
evaluation function.

Approach
--------
1. **Geometric sweep** – generate a family of decreasing geometric
   sequences a_i ∝ r^i with r = 1 - factor / n.  The factor is swept in a
   fairly dense range and the best of them is kept as a seed.

2. **Exponentiated (mirror) gradient descent** – the objective is
   f(a)=2·n·max_k (a∗a)[k] (the denominator ∑a_i² cancels if we keep
   ∑a_i=1).  The sub‑gradient of the max is the average of the gradients
   of all *active* shifts (those attaining the maximal convolution value).
   We update multiplicatively:
        a ← a ⊙ exp( -η·g )
   and renormalise to the probability simplex (∑a_i=1).  This guarantees
   non‑negativity and avoids a separate projection routine.

   The step size η decays slowly; the update is cheap (the active set is
   typically of size 1).

3. **Greedy random local search** – a simple hill‑climbing phase that
   repeatedly transfers a small amount of mass from one coordinate to
   another.  Only improving moves are kept (no temperature schedule is
   required for the modest budget).

The whole routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence found (as a plain Python
list).  No external files or networks are accessed.
"""

import numpy as np

# Helper: evaluate the objective directly (same as evaluate_sequence)
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²)  (assumes seq ≥ 0)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return 2.0 * len(seq) * max_b / (sum_a * sum_a)

def _active_gradient(seq: np.ndarray, conv: np.ndarray) -> np.ndarray:

    Compute the sub‑gradient of  max_k (seq*seq)[k]  at the current point.
    The gradient is the average of the individual gradients for all shifts
    k that attain the maximum value.

        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # grad_i = a_{k-i}  (the same as in the reference implementation)
        grad[idx] += seq[k - idx]

    if len(active) > 0:
        grad /= len(active)          # average over the active set
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The function respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a moderate length – large enough for a good bound,
    #    small enough for many fast convolutions.
    n = 2000

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 2. Sweep geometric families.
    #    Theoretical optimum is factor≈2, but we explore a wider range.
    factors = np.linspace(0.8, 3.2, 24)   # 24 candidates → cheap
    for factor in factors:
        if time.time() > deadline:
            break
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()          # normalise to sum = 1
        score = _objective(seq)              # cheap local evaluation
        # keep the best geometric candidate
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        print(f"[Init] factor={factor:5.2f}  r={r:.6f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Optionally try a few random Dirichlet seeds (they can help escape
    #    pathological local minima).  The number is tiny so we stay within
    #    the budget.
    for _ in range(6):
        if time.time() > deadline:
            break
        seq = rng.dirichlet(alpha=np.full(n, 0.5))
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        print(f"[Random] score={score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated gradient descent on the active‑set sub‑gradient.
    #    This phase keeps the sum fixed (simple normalisation) and works
    #    directly on the simplex.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    seq = best_seq.copy()
    # step‑size schedule – start moderate, decay slowly
    eta0 = 0.12
    it = 0
    while time.time() < deadline:
        conv = np.convolve(seq, seq)
        # sub‑gradient of the maximum convolution entry
        grad = _active_gradient(seq, conv)

        # Polyak‑like step size would be huge for flat regions,
        # therefore we cap it to a safe maximum.
        eta = eta0 / (1.0 + it * 1e-4)   # very slow decay
        eta = min(eta, 0.5)

        # multiplicative (mirror) update – guarantees non‑negativity
        seq = seq * np.exp(-eta * grad)
        seq /= seq.sum()                 # projection onto the simplex

        # occasional evaluation (every 20 iterations)
        if it % 20 == 0:
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[EGD {it:05d}] new best = {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 5. Greedy random local search (hill climbing).
    #    Transfer a small fraction of mass between two random coordinates.
    seq = best_seq.copy()
    move_frac = 0.025                     # proportion of the source mass to move
    hill_it = 0
    while time.time() < deadline:
        i = rng.integers(n)
        if seq[i] < 1e-12:
            continue
        # pick a distinct target index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = min(seq[i], move_frac * rng.random())
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # quick objective (no need to call evaluate_sequence repeatedly)
        score = _objective(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"[Hill {hill_it:06d}] new best = {best_score:.6f}")
        else:
            # keep the current sequence unchanged (greedy acceptance)
            pass

        hill_it += 1

    # ------------------------------------------------------------------
    # 6. Final report.
    print(f"Final best objective = {best_score:.6f} (length {n})")
    # Convert to plain Python list for the external evaluator.
    return best_seq.tolist()
