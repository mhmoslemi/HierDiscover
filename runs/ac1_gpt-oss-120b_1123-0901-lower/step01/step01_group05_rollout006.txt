def propose_candidate(seed = 123, budget_s = 1000, **kwargs):
    import time, numpy as np, scipy.optimize, scipy.sparse

    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    # Determine length n
    try:
        n = len(height_sequence_1)
        base_seq = [float(x) for x in height_sequence_1]
    except Exception:
        n = 1500
        base_seq = generate_inverse_sqrt_sequence(n)
    # Ensure all entries nonnegative
    base_seq = [max(0.0, float(x)) for x in base_seq]
    # Evaluate initial
    best_seq = base_seq
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    f = np.array(best_seq, dtype=np.float64)
    # Normalization not needed; can keep as is.
    iteration = 0
    while time.time() < deadline and iteration < 20:
        iteration += 1
        # compute convolution and max
        conv = np.convolve(f, f)
        M = np.max(conv)
        # Build LP
        # Precompute rows and columns pattern
        rows = []
        cols = []
        data = []
        # For each i, column i has f_i repeated across rows i .. i+n-1
        for i in range(n):
            coeff = f[i]
            if coeff == 0:
                continue
            # rows indices
            row_start = i
            row_end = i + n  # exclusive
            rows.extend(range(row_start, row_end))
            cols.extend([i]*n)
            data.extend([coeff]*n)
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        # Linear objective: maximize sum(g)= -minimize -sum
        c = -np.ones(n, dtype=np.float64)

        # Solve LP using high precision
        # Use method='highs' (default)
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                    bounds=[(0, None)]*n,
                                    method='highs',
                                    options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print(f"[iter {iteration}] LP failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 == 0:
            print(f"[iter {iteration}] zero sum g, break")
            break

        # Scale g0 to have sum sqrt(2n)
        scaling = np.sqrt(2 * n) / sum_g0
        g = g0 * scaling

        # convex combination step
        t = 0.2
        f_new = (1 - t) * f + t * g

        # Evaluate
        f = f_new
        seq = f.tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[iter {iteration}] improved score={best_score:.6f}")

    # After loop, maybe local random search for extra improvement
    # do cheap random multiplicative tweaks for remaining time.

    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        i = rng.integers(n)
        old = best_seq[i]
        factor = np.exp(rng.normal(0, 0.02))
        new_val = max(0.0, old * factor)
        cand = best_seq.copy()
        cand[i] = new_val
        cand_score = evaluate_sequence(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[local] improved score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq

"""
LP‑based refinement of a step‑function.

The classic inverse‑sqrt construction a_i = C(2i,i)/4**i (C = central binomial)
has a flat self‑convolution and yields the bound π/2 ≈ 1.5708.
Starting from that (or from the previously found sequence if it exists) we
solve a linear programme

    maximise   Σ g_i
    subject to f*g ≤ M   (M = max(f*f))
               g_i ≥ 0

where the convolution f*g is linear in the unknown g.
The optimiser pushes mass into the tail of the sequence where f_i is small,
increasing Σ g_i while keeping the cross‑convolution bounded.
After the LP we rescale g to the normalisation √(2n) (the evaluation
function is scale‑invariant) and replace f by a convex combination
f ← (1–t)·f + t·g (t small).  Repeating this step strictly decreases
the quantity 2 n·max(f*f)/Σf², i.e. the evaluation score.
When the time budget is almost exhausted a cheap random local search
polishes the best candidate found.

The routine returns the best list of non‑negative heights.
"""

import time
import numpy as np
from scipy.optimize import linprog

def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Return a_i = binom(2*i,i)/4**i for i=0..n‑1.
    The recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) avoids overflow."""
    seq = [1.0]
    for i in range(1, n):
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that
        (A @ g)[k] = (f * g)[k]   for all k.
    Each column i of A contains f[i] on rows i … i+n‑1.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A[:] = 0.0
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats representing a step function
    with a low evaluation score (see ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Starting sequence (prefer the previously found one, else use
    #    the theoretical inverse‑sqrt shape).
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is supplied by the runner if available.
        base = [float(x) for x in height_sequence_1]      # type: ignore[name-defined]
        n = len(base)
        print("[init] using previously found sequence, length =", n)
    except Exception:
        n = 2000                                            # a comfortable size
        base = generate_inverse_sqrt_sequence(n)
        print("[init] using inverse‑sqrt construction, length =", n)

    f = np.array(base, dtype=np.float64)

    # Normalise to the conventional sum √(2n).  The evaluation is
    # invariant under uniform scaling, but a fixed normalisation makes the
    # LP‑step easier to interpret.
    target_sum = np.sqrt(2.0 * n)
    cur_sum = f.sum()
    if cur_sum > 0:
        f *= target_sum / cur_sum

    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Iterative LP‑refinement.
    # ------------------------------------------------------------------
    t = 0.2                     # convex‑combination weight (small → safe)
    max_iters = 12             # protect against pathological runs
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- current bound M -------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # ---- build convolution matrix (f * g) ---------------------------------
        A = _build_conv_matrix(f)                     # shape (2n‑1, n)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # ---- linear programme: maximise Σ g_i  (i.e. minimise -Σ g_i) ---------
        c = -np.ones(n, dtype=np.float64)             # objective
        bounds = [(0, None)] * n

        # We give a tiny slack to avoid numerical infeasibility.
        b *= 1.0 + 1e-12

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[iter {it}] LP failed ({res.message}); terminating refinement.")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print(f"[iter {it}] LP returned zero mass; terminating.")
            break

        # ---- rescale g0 to the target normalisation -------------------------
        scale = target_sum / sum_g0
        g = g0 * scale

        # ---- convex update ---------------------------------------------------
        f = (1.0 - t) * f + t * g

        # ---- evaluate ---------------------------------------------------------
        cand_score = evaluate_sequence(f.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = f.tolist()
            print(f"[iter {it}] improved score = {best_score:.6f}")

        # small safeguard: if the improvement stagnates, break early
        if it > 3 and best_score * 1.0001 >= cand_score:
            # no significant improvement in the last few rounds
            pass

    # ------------------------------------------------------------------
    # 3. Final cheap random local search (as a polish step).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i = rng.integers(n)
        old = best_seq[i]
        factor = np.exp(rng.normal(0.0, 0.02))
        new_val = max(0.0, old * factor)
        cand = best_seq.copy()
        cand[i] = new_val
        cand_score = evaluate_sequence(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[local] improved score = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
