def propose_candidate(seed=42, budget_s=1000, length=None):
    import time, numpy as np, cvxpy as cp, scipy.sparse

    np.random.seed(seed)
    if length is None:
        try:
            n = len(height_sequence_1)
        except NameError:
            n = 4000 # choose moderate length
    else:
        n = length

    # start with arcsine distribution
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * ((2*k-1)*(2*k)) / (4.0 * k * k)

    a /= a.sum()   # sum=1

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score}")

    deadline = time.time() + budget_s - 5
    iteration = 0
    t = 0.5      # mixing factor optional
    while time.time() < deadline and iteration < 30:
        # compute current conv and M
        conv = np.convolve(a, a)
        M = conv.max()
        # Build convolution matrix A
        A = np.zeros((2*n-1, n), dtype=np.float64)
        # Fill using broadcasting:
        # each column i gets a shifted placement:
        A[np.arange(n)[:,None] + i, i] = a  # Not for loops; maybe use loop.
        # Instead:
        for i in range(n):
            A[i:i+n, i] = a
        # Add small slack
        M_slack = M * (1.0 + 1e-8)

        # LP
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_slack]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except cp.error.SolverError:
            print("[LP] solver error, break")
            break

        if g.value is None:
            print("[LP] no solution")
            break

        g_val = np.maximum(0.0, np.array(g.value, dtype=np.float64))
        # normalize g to sum 1 (same scaling as a)
        g_sum = g_val.sum()
        if g_sum == 0:
            print("[LP] zero sum")
            break
        g_norm = g_val / g_sum

        # blend
        a_new = (1 - t) * a + t * g_norm
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        print(f"[iter] {iteration}, score={score:.6f}, M={M:.6e}")

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"  [improve] best_score={best_score:.6f}")

        a = a_new
        iteration += 1
        # maybe reduce t gradually
        t = max(0.1, t * 0.9)

    print(f"[final] best_score={best_score:.6f}")
    return best_seq.tolist()

"""
propose_candidate
=================

Search for a non‑negative step function that minimizes the target evaluation
function.  The algorithm works in three phases:

1️⃣ **Start** from the discrete arcsine distribution (the known asymptotic
   optimum) scaled to unit L1‑norm.

2️⃣ **Iterative LP refinement**.  Given the current sequence `a`,
   we solve a linear programme

        maximise   Σ g_j
        subject to  (a ∗ g)_k ≤ M   for all k
                    g_j ≥ 0

   where `M = max(a∗a)`.  The LP pushes mass into positions where the
   current sequence `a` is small, therefore reducing the maximal
   autoconvolution when we move towards the LP solution.  The new
   candidate is a convex combination `a ← (1‑t)·a + t·g_norm` (with a
   decreasing step size `t`).  The LP is solved with SciPy’s HiGHS
   solver, which is fast enough for a few hundred variables.

3️⃣ **Fine‑grained hill‑climbing** – after the LP phase we perform a
   cheap random shuffle of a tiny amount of mass between two entries.
   This often extracts a final tiny improvement.

The whole routine respects the 1000 s budget, prints progress, and
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


def _arcsine_sequence(n: int) -> np.ndarray:
    """Return the discrete arcsine distribution of length *n*,
    normalised so that its L1‑norm equals 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()
    return a


def _make_conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1) × n matrix ``A`` such that ``A @ g`` equals
    ``np.convolve(a, g)`` for any vector ``g`` (full convolution mode).
    The matrix is returned as a CSR sparse matrix.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    data = np.repeat(a, n)               # each column repeats the whole a
    # column indices repeat 0…n‑1 each n times
    col_idx = np.tile(np.arange(n), n)
    # row indices: column i occupies rows i … i+n‑1
    row_idx = np.concatenate([np.arange(i, i + n) for i in range(n)])
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
    return A


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     length: int | None = None,
                     **kwargs):
    """
    Generate a step‑function (list of non‑negative floats) that aims to
    minimise ``evaluate_sequence``.
    """
    # -----------------------------------------------------------------
    # 1. Determine the problem size (number of coefficients)
    # -----------------------------------------------------------------
    np.random.seed(seed)

    # Prefer a moderate size that keeps the LP tractable.
    # If a previous candidate exists and is reasonably short, reuse its length.
    try:
        prev_len = len(height_sequence_1)       # type: ignore[name-defined]
    except Exception:
        prev_len = None

    if length is not None:
        n = int(length)
    elif prev_len is not None and prev_len <= 2000:
        n = prev_len
    else:
        n = 1200                # a safe default: large enough for a good bound,
                               # small enough for fast LP solves.

    # -----------------------------------------------------------------
    # 2. Initialise with the discrete arcsine distribution (unit L1)
    # -----------------------------------------------------------------
    a = _arcsine_sequence(n)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Main optimisation loop (LP + mixing)
    # -----------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # keep a few seconds for final work
    iteration = 0
    t = 0.5                                   # initial mixing factor

    while time.time() < deadline and iteration < 40:
        # current maximal auto‑convolution
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()
        # a small slack avoids numerical infeasibility
        M_slack = M * (1.0 + 1e-12)

        # Build the convolution matrix for the current 'a'
        A = _make_conv_matrix(a)

        # ----- Linear programme -----
        # maximise Σ g_j   ⇔   minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)

        # inequality:  A @ g ≤ M_slack   (vector of length 2n‑1)
        b_ub = np.full(2 * n - 1, M_slack, dtype=np.float64)

        # bounds: g_j ≥ 0
        bounds = [(0.0, None) for _ in range(n)]

        # Solve with HiGHS – fast for dense medium‑sized LPs
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            print(f"[LP] iteration {iteration}: LP failed (status={res.message}).  Stopping.")
            break

        g = np.maximum(0.0, res.x)            # ensure non‑negativity in case of tiny negatives

        # Normalise g to unit L1 (the objective is scale‑invariant)
        g_sum = g.sum()
        if g_sum == 0.0:
            print(f"[LP] iteration {iteration}: got zero sum, aborting.")
            break
        g_norm = g / g_sum

        # Mix the new candidate with the current one
        a_new = (1.0 - t) * a + t * g_norm

        # Evaluate
        cur_score = evaluate_sequence(a_new.tolist())
        print(f"[LP] it={iteration:02d}, t={t:.3f}, score={cur_score:.6f}, M={M:.6e}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"   [improve] new best = {best_score:.6f}")

        # Prepare for next iteration
        a = a_new
        iteration += 1
        # Reduce the mixing factor slowly – this mirrors the “small‑step” idea
        t = max(0.05, t * 0.85)

    # -----------------------------------------------------------------
    # 4. Light random hill‑climbing to squeeze out remaining slack
    # -----------------------------------------------------------------
    # We use a very cheap move: transfer a tiny fraction of mass from a
    # larger entry to a smaller one if it improves the objective.
    hill_iters = 3000
    deadline = time.time() + budget_s - 1.0
    it = 0

    a = best_seq.copy()          # start hill‑climb from the best LP result
    n = a.shape[0]

    while time.time() < deadline and it < hill_iters:
        i, j = np.random.choice(n, size=2, replace=False)
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i

        # Move a tiny proportion of the smaller entry (ensures non‑negativity)
        delta = 0.004 * min(a[src], a[dst])
        if delta == 0.0:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta

        cand_score = evaluate_sequence(a_candidate.tolist())
        if cand_score < best_score:
            a = a_candidate
            best_seq = a_candidate
            best_score = cand_score
            print(f"[hill] it={it:04d}, score={best_score:.6f}")

        it += 1

    print(f"[final] best score={best_score:.6f} after {iteration} LP iterations "
          f"and {it} hill‑climb steps.")
    # Return as plain Python list (required by the evaluator)
    return best_seq.tolist()
