def full_convolution(p):
    n = len(p)
    m = 2*n - 1
    n_fft = 1 << ((m - 1).bit_length())
    # use np.fft.rfft for real
    P = np.fft.rfft(p, n=n_fft)
    conv = np.fft.irfft(P * P, n=n_fft)[:m]
    return conv

          p ← p * exp(-η * g)     (η decreases like η₀/√t),
      then project back onto the probability simplex (renormalise).
    * Keep the best sequence seen (lowest true score).

4.  **Fine local search** – after the gradient phase, spend the remaining time
    on cheap random perturbations of single entries (log‑normal factors) and
    accept only improving moves.

The evaluation function is scale‑invariant, therefore we keep the sum of the
coefficients equal to one throughout the optimisation.  All operations stay in
memory, use only NumPy (no external file or network access) and respect the
1000 s budget.

The returned list can be any non‑negative scaling of the optimiser’s `p`
because `evaluate_sequence` normalises the scale internally.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classic inverse‑square‑root sequence.
# ----------------------------------------------------------------------
def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """Return a NumPy array a_0,…,a_{n‑1} with a_i = binom(2*i,i)/4**i."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        # recurrence a_{i} = a_{i-1} * (2*i-1)/(2*i)
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


# ----------------------------------------------------------------------
# Convolution helpers that use FFT (O(n log n) and vectorised).
# ----------------------------------------------------------------------
def full_convolution(x: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of a real vector with itself via FFT."""
    n = x.size
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())       # next power of two
    X = np.fft.rfft(x, n=n_fft)
    conv = np.fft.irfft(X * X, n=n_fft)[:m]
    return conv


def weighted_gradient(conv: np.ndarray, p: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max objective.
    `conv` must be the full convolution p ⋆ p.
    Returns a vector of length len(p) containing 2·Σ_k w_k·p_{k-i}.
    """
    n = p.size
    # Soft‑max weights, normalised to avoid overflow.
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))          # w_max == 1
    # Convolution w ⋆ rev(p)  (only the middle n entries are needed)
    rev_p = p[::-1]
    m = w.size + rev_p.size - 1
    n_fft = 1 << ((m - 1).bit_length())
    W = np.fft.rfft(w, n=n_fft)
    Rev = np.fft.rfft(rev_p, n=n_fft)
    tmp = np.fft.irfft(W * Rev, n=n_fft)[:m]
    grad = 2.0 * tmp[n - 1 : n - 1 + n]        # extract central window
    return grad


# ----------------------------------------------------------------------
# Scoring (exact, no call to evaluate_sequence – cheap because sum(p)=1)
# ----------------------------------------------------------------------
def score_from_p(p: np.ndarray) -> float:
    """Return the evaluation score for a probability vector p (sum(p)=1)."""
    n = p.size
    conv = full_convolution(p)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best bound (~1.5709).  The algorithm combines a smooth‑max
    gradient descent with a final stochastic local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 3400                     # sensible default if not provided

    # ------------------------------------------------------------------
    # 2. Base distribution: inverse‑square‑root shape, normalised to sum=1.
    # ------------------------------------------------------------------
    base = generate_inverse_sqrt_sequence(n)
    p = base.astype(np.float64)
    p /= p.sum()                    # now a probability vector

    best_p = p.copy()
    best_score = score_from_p(p)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 3. Phase‑1: smooth‑max gradient descent.
    # ------------------------------------------------------------------
    beta = 6.0                     # controls how sharp the soft‑max is
    eta0 = 1.0                     # initial step‑size for the mirror step
    it = 0
    phase1_deadline = start_time + 0.6 * budget_s   # spend ~60 % of time

    while time.time() < phase1_deadline:
        it += 1
        conv = full_convolution(p)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[improve:grad] iter {it}, score {best_score:.6f}")

        # Gradient of the smooth max
        g = weighted_gradient(conv, p, beta)

        # Decreasing step‑size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(it)
        factor = np.exp(-eta * g)
        p = p * factor

        # Projection onto the simplex (simple normalisation)
        s = p.sum()
        if s <= 0.0 or not np.isfinite(s):
            # numerical trouble – re‑initialise from the best known point
            p = best_p.copy()
            continue
        p /= s

        # Occasionally inject a tiny random jitter to avoid stagnation
        if it % 250 == 0:
            indices = rng.choice(n, size=min(20, n), replace=False)
            jitter = np.exp(rng.normal(0.0, 0.04, size=indices.shape))
            p[indices] *= jitter
            p /= p.sum()

        # Safety break if time is exhausted
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Phase‑2: cheap random local search (single‑entry perturbations).
    # ------------------------------------------------------------------
    it2 = 0
    while time.time() < deadline:
        it2 += 1
        i = rng.integers(n)
        old = best_p[i]

        # Propose a log‑normal multiplicative perturbation
        factor = np.exp(rng.normal(0.0, 0.025))
        new_val = max(0.0, old * factor)

        cand = best_p.copy()
        cand[i] = new_val
        cand /= cand.sum()                     # keep it a probability vector
        cand_score = score_from_p(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_p = cand
            print(f"[improve:local] iter {it2}, idx {i}, score {best_score:.6f}")

        # Optional small random shake every 500 trials
        if it2 % 500 == 0:
            idxs = rng.choice(n, size=min(10, n), replace=False)
            shake = np.exp(rng.normal(0.0, 0.03, size=idxs.shape))
            best_p[idxs] *= shake
            best_p /= best_p.sum()

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (any positive scaling is fine).
    # ------------------------------------------------------------------
    print(f"[final] best score ≈ {best_score:.6f}")
    # Scale up a bit (optional) – not required, but keeps numbers comfortably
    # away from 0 without affecting the evaluation.
    final_seq = (best_p * 100.0).tolist()
    return final_seq
