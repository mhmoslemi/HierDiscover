"""
Propose a candidate step‑function (non‑negative sequence) that improves on the
inverse‑square‑root construction.  The algorithm consists of three stages:

1.  Choose a length (by default the length of the previously found best
    sequence) and build the analytic “inverse‑sqrt’’ sequence
    a_i = binom(2*i,i)/4**i, which yields a score ≈ π/2 ≈ 1.5708.

2.  Run a cheap gradient‑based descent that tries to flatten the
    discrete convolution.  At each iteration the index *k* where the
    convolution attains its maximum is identified.  The gradient of the
    evaluation function (with the active index *k* fixed) can be written
    analytically and points from entries that heavily contribute to the
    maximal convolution towards those that contribute little.  A small
    step is taken, the sequence is projected back onto the non‑negative
    orthant and re‑scaled so that its ℓ₁‑norm stays constant (the objective
    is scale‑invariant).  The step size is adapted: it is increased after a
    successful move and halved after a failure.

3.  Interleave the gradient updates with very cheap random multiplicative
    perturbations of single entries (the “log‑normal’’ move from the
    reference implementation).  Only moves that strictly improve the
    evaluation score are kept.

The loop runs until the allocated time budget (default 1000 s) is exhausted.
The best sequence found so far is returned.
"""

import time
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Create a_i = binom(2*i,i) / 4**i  (≈ 1/√i).  Uses the recurrence
    a_{i+1}=a_i*(2*i+1)/(2*i+2) to avoid large intermediate numbers."""
    seq = [1.0]                     # a_0 = 1
    for i in range(1, n):
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq

def _gradient_step(seq: np.ndarray, lr: float) -> np.ndarray:
    """One gradient step that (approximately) follows the derivative of the
    evaluation function assuming the current maximal convolution index stays
    fixed.  The step is projected onto the non‑negative orthant and the
    ℓ₁‑norm is restored (scale‑invariant objective)."""
    n = seq.size
    S = seq.sum()
    # convolution and its maximal index
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))               # index where max occurs
    max_b = conv[k]

    # partner[i] = a_{k-i} if the index is inside the array, else 0
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    partner = np.zeros_like(seq)
    partner[mask] = seq[j[mask]]

    # gradient of 2*n*max_b / S^2  (with k fixed)
    grad = 4.0 * n * partner / (S * S) - 4.0 * n * max_b / (S ** 3)

    # take a step, keep non‑negative entries
    new_seq = seq - lr * grad
    np.maximum(new_seq, 0.0, out=new_seq)

    # restore original ℓ₁‑norm (objective is scale‑invariant)
    new_sum = new_seq.sum()
    if new_sum > 0.0:
        new_seq *= (S / new_sum)
    return new_seq
