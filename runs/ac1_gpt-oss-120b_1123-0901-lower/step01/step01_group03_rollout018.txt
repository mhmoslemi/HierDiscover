        conv = _full_convolution(a)
        max_val = conv.max()

        w = np.exp(beta * (conv_scaled - conv_scaled.max()))

                cur_score = new_score
                if new_score < best_score:

                break

                eta_i *= 0.5

def _simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros? fallback uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a coefficient sequence for the self‑convolution bound.

The algorithm is a smoothed‑max gradient descent on the quantity
    M(a) = 2·n·max(conv(a,a))   (the evaluation metric with Σa=1).

* Initialise a with the central‑binomial coefficients (discretised
  (1‑z)^‑1/2) and project onto the probability simplex.
* At each iteration:
    – Compute the full self‑convolution via a single FFT.
    – Form a soft‑max over the convolution values (β grows slowly) to
      obtain a weight vector w that concentrates on the near‑maximal
      entries.
    – The gradient of the smoothed max is 2·n·corr(a,w) where
      corr is the discrete correlation; this is evaluated efficiently
      with another FFT (using w reversed).
    – An Adam‑style update is taken, then the vector is projected back
      onto the simplex.
    – If the new vector does not improve the objective, the step is
      rejected and a tiny random jitter is added to escape plateaus.
    – The best vector seen so far is kept.
* The smoothing parameter β is increased from a small value up to a
  moderate maximum, turning the soft‑max into a true max as the run
  proceeds.
* The loop runs until the time budget (≈1000 s) is exhausted.

The method uses only O(n log n) work per iteration and works well for
lengths of several thousand entries.  It typically improves on the
π/2 bound (≈1.57086) and often reaches the target ≈1.503.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (simple scaling, fast)
# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Scale a non‑negative vector to have sum 1 (leaving zeros unchanged)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Helper: full linear convolution a*a using a single FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(a, size):
    """Return a*a (length 2·len(a)-1) using FFT of given size."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * len(a) - 1]
    return conv, fa

        conv_scaled = conv * n                     # scale ≈ O(1)

        # safe soft‑max (subtract the max to avoid overflow)
        w = np.exp(beta * (conv_scaled - max_scaled))
        w_sum = w.sum()

            w = np.full_like(w, 1.0 / len(w))
