```
def project_simplex(v):
    # v: array shape (n,)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np
import time

def project_onto_simplex(v):
    # Duchi et al. Euclidean projection onto simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def full_convolution_self(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def correlation_valid(w, a):
    # Return linear correlation (valid) of w with a, length len(a)
    n = a.shape[0]
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, size)
    A_rev = np.fft.rfft(a[::-1], size)
    conv = np.fft.irfft(W * A_rev, size)[: len(w) + n - 1]
    # extract slice
    start = n - 1
    return conv[start:start + n]

def evaluate_metric(a):
    # assume a is already normalized to sum 1
    conv = full_convolution_self(a)
    max_val = conv.max()
    return 2.0 * a.shape[0] * max_val
```

    The algorithm combines a multi‑active sub‑gradient descent on the peak
    of the self‑convolution with a smooth log‑sum‑exp gradient descent.
    """
    np.random.seed(seed)
    end_time = time.time() + budget_s - 5.0

    # Choose a working length.
    if 'height_sequence_1' in globals():
        base_seq = np.array(height_sequence_1, dtype=np.float64)
        n = min(len(base_seq), 2000)   # cap for speed
        a = base_seq[:n].copy()
    else:
        n = 2048
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    # project onto simplex (sum = 1)
    a = project_onto_simplex(a)

    best_a = a.copy()
    best_score = evaluate_metric(best_a)
    print(f"Initial length {n}, score {best_score:.6f}")

    # ------------------------------------------------------------------ #
    # Phase 1 – multi‑active sub‑gradient descent on the raw maximum.
    # ------------------------------------------------------------------ #
    eta0 = 0.8
    iteration = 0
    active_eps = 1e-12

    while time.time() < end_time * 0.45:
        conv = full_convolution_self(a)
        max_val = conv.max()
        # active mask (indices where conv is within eps of the max)
        mask = (conv >= max_val - active_eps * max_val).astype(np.float64)
        grad = 2.0 * correlation_valid(mask, a)

        # simple back‑tracking line search
        eta = eta0 / np.sqrt(iteration + 1.0)
        improved = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_max = full_convolution_self(a_new).max()
            if new_max <= max_val * (1.0 + 1e-14):
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            # random jitter to escape stagnation
            a = project_onto_simplex(a + np.random.randn(n) * 1e-4)

        iteration += 1
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Phase1 it {iteration:6d}: better score {best_score:.6f}")

        if iteration % 2000 == 0:

"""
Propose a candidate step‑function sequence that improves the known bound.

The algorithm works in three stages:

1. Initialise a feasible probability vector (non‑negative, sum = 1).  If a
   previously found sequence ``height_sequence_1`` exists we reuse its first
   few thousand entries, otherwise we start from the central‑binomial
   coefficients (the discrete arcsine distribution).

2. Phase 1 – multi‑active sub‑gradient descent on the *peak* of the
   self‑convolution.  Instead of using a single maximising index we take an
   active set of all indices whose convolution value is within a tiny
   tolerance of the maximum and move in the summed sub‑gradient direction.
   After each step the vector is projected onto the probability simplex.

3. Phase 2 – smooth log‑sum‑exp gradient descent.  The objective
   ``max(conv)`` is approximated by a softmax with a temperature
   parameter ``s``; the gradient of the smoothed objective is the
   correlation of the softmax weights with the current vector.  ``s`` is
   increased slowly, turning the smooth problem into the original
   nonsmooth one.  Again we project onto the simplex after each update.

4. (Optional) a tiny greedy hill‑climbing loop that swaps a small amount
   of mass between two random positions if it improves the metric.

All heavy operations (convolution, correlation) are performed with FFTs,
so each iteration runs in O(n log n) time.  The whole search runs for up to
the allotted 1000 s and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level as required
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = 1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _valid_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation (valid) of w with a.
    Returns an array of length len(a) where
        result[i] = Σ_j w[i + j] * a[j] .
    """
    n = a.shape[0]
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, size)
    A_rev = np.fft.rfft(a[::-1], size)   # a reversed
    full = np.fft.irfft(W * A_rev, size)[: len(w) + n - 1]
    start = n - 1
    return full[start:start + n]        # length n


def _metric(a: np.ndarray) -> float:
    """Evaluation metric for a (assumed already normalised to sum = 1)."""
    conv = _full_self_convolution(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the given metric.
    The function may use the global ``height_sequence_1`` if it exists.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # -------------------------
    # 1. Choose a working length
    # -------------------------
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
        n = min(len(base), 2000)          # limit size for speed
        a = base[:n].copy()
    else:
        n = 2048
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    # Normalise to the probability simplex Σa = 1
    a = _project_onto_simplex(a)

    best_a = a.copy()
    best_score = _metric(best_a)
    print(f"Initial length {n}, score {best_score:.6f}")

    # -------------------------------------------------
    # 2. Phase 1 – multi‑active sub‑gradient descent
    # -------------------------------------------------
    eta0 = 0.8
    it = 0
    active_rel_eps = 1e-6

    while time.time() < deadline * 0.45:
        conv = _full_self_convolution(a)
        max_val = conv.max()
        # active set: indices whose value is within a relative tolerance
        mask = (conv >= max_val * (1.0 - active_rel_eps)).astype(np.float64)
        # sub‑gradient = 2·correlation(mask, a)
        grad = 2.0 * _valid_correlation(mask, a)

        # simple back‑tracking line search
        eta = eta0 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = _project_onto_simplex(a_new)
            new_max = _full_self_convolution(a_new).max()
            if new_max <= max_val * (1.0 + 1e-14):
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            # tiny random kick to escape a possible stall
            a = _project_onto_simplex(a + np.random.randn(n) * 1e-4)

        it += 1
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Phase 1  it {it:6d}: improved to {best_score:.6f}")

        # occasional larger perturbation
        if it % 2000 == 0:
            a = _project_onto_simplex(a + np.random.randn(n) * 0.003)

    # -------------------------------------------------
    # 3. Phase 2 – smoothed log‑sum‑exp gradient descent
    # -------------------------------------------------
    eta0 = 0.4
    it2 = 0
    s = 1.0                     # smoothing strength, grows slowly

    while time.time() < deadline:
        conv = _full_self_convolution(a)
        max_val = conv.max()

        # softmax weights (numerically stable)
        shifted = conv - max_val
        w = np.exp(s * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        grad = 2.0 * _valid_correlation(w, a)

        eta = eta0 / np.sqrt(it2 + 1.0)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new)
        new_max = _full_self_convolution(a_new).max()

        if new_max <= max_val:
            a = a_new
        else:
            # back‑track a little
            eta *= 0.5
            a_new = a - eta * grad
            a = _project_onto_simplex(a_new)

        # increase smoothing (approach the true max)
        s *= 1.001

        it2 += 1
        score = 2.0 * n * new_max
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Phase 2  it {it2:6d}: new best {best_score:.6f}")

        # occasional random jitter
        if it2 % 1500 == 0:
            a = _project_onto_simplex(a + np.random.randn(n) * 0.004)

    # -------------------------------------------------
    # 4. Optional tiny hill‑climbing refinement
    # -------------------------------------------------
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a tiny amount of mass from i to j
        delta = min(best_a[i], 0.001)
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand)
        cand_score = _metric(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            print(f"Hill‑climb improved to {best_score:.6f}")

    print(f"Finished. Best score {best_score:.6f} (n = {n})")
    # Return as a plain Python list of floats
    return best_a.tolist()
