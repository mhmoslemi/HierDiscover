"""
Propose a step‑function (non‑negative sequence) with a low
`evaluate_sequence` value.

Algorithm
---------
1. Choose a moderate length n (here 2000 – cheap enough for many
   convolution evaluations).
2. Initialise a pool of diverse candidates:
   * uniform,
   * several geometric progressions (different decay rates),
   * an arcsine‑shaped distribution (the continuous arcsine law on [0,1]),
   * a few random Dirichlet draws.
   The best of these is kept as the starting point.
3. Refine the current best with a *soft* sub‑gradient descent.
   At each iteration we
   * compute the full self‑convolution,
   * select the K (default 5) shifts with the largest values,
   * accumulate the exact sub‑gradient of those K terms,
   * apply a momentum update,
   * step opposite to the gradient,
   * project back onto the probability simplex (sum = 1, all entries ≥0).
   The step size decays slowly; momentum helps to slide past flat
   plateaus.  Occasionally (after a long period without improvement) a
   small random perturbation is added to escape local minima.
4. Continue until the time budget (≈ 1000 s) expires, then return the
   best sequence found.

The method keeps the evaluation cost low (one O(n²) convolution per
iteration) while exploring many directions in the feasible space, and
empirically reaches an objective well below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:           # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n, rng):
    """Generate a list of promising starting sequences (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # 3. arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. a few random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimizes the evaluation
    function within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. problem size
    n = 2000                                 # modest size, fast conv.

    # ------------------------------------------------------------------
    # 2. generate initial pool and pick the best start
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best score after seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. refinement – soft sub‑gradient descent
    seq = best_seq.copy()
    step = 0.20                # initial step size (absolute scale)
    momentum = np.zeros_like(seq)
    beta = 0.9                 # momentum factor
    K = 5                      # number of top convolution entries to follow
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # full convolution (length 2n‑1)
        conv = np.convolve(seq, seq)

        # indices of K largest entries
        if K >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -K)[-K:]

        # accumulate sub‑gradient for the selected shifts
        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # contribution a_{k-i} for each i in the overlap
            grad[idx] += seq[k - idx]

        # average over the K chosen shifts (keeps scale stable)
        grad = grad / K

        # momentum update (classical heavy‑ball)
        momentum = beta * momentum + (1.0 - beta) * grad

        # gradient step (we minimise, so go opposite to gradient)
        new_seq = seq - step * momentum
        new_seq = _project_onto_simplex(new_seq)

        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Iter {it:05d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # step size schedule – very slow decay (so we keep exploring)
        step *= 0.9999

        # occasional random perturbation if we have stagnated too long
        if no_improve > 30000:
            # add a small Gaussian jitter and re‑normalise
            jitter = rng.normal(scale=0.01, size=n)
            new_seq = new_seq + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq = new_seq / new_seq.sum()
            momentum[:] = 0.0
            no_improve = 0
            print(f"[Iter {it:05d}] stagnation – random perturbation applied")

        seq = new_seq
        it += 1

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
