def objective_u(u, lam):
    # compute a = softmax of u
    u_max = np.max(u)
    e = np.exp(u - u_max)   # subtract max for stability
    a = e / e.sum()
    # compute conv
    size = 1 << ((2*n-1).bit_length())
    # Actually conv compute length 2*n-1 and we need max.
    # Use FFT for conv.
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_val = conv.max()
    # compute p weights
    shifted = conv - max_val   # non-positive
    exp_term = np.exp(lam * shifted)   # <=1
    Z = exp_term.sum()
    smoothed_max = max_val + np.log(Z) / lam
    # compute p = exp_term / Z   (size 2n-1)
    p = exp_term / Z

    # compute term1, term2 via correlation
    # compute correlation using FFT as described
    size2 = 1 << ((len(a) + len(p) - 1).bit_length())
    fa2 = np.fft.rfft(a, size2)
    fp2 = np.fft.rfft(p, size2)
    corr_full = np.fft.irfft(fa2 * np.conj(fp2), size2)[:len(a) + len(p) - 1]
    center = len(p) - 1
    term1 = corr_full[center:center+n]  # i from 0..n-1
    term2 = corr_full[center-n+1:center+1][::-1]  # reversed
    grad_a = term1 + term2

    # compute gradient w.r.t u
    scalar = grad_a.dot(a)
    grad_u = a * (grad_a - scalar)

    # objective
    obj = 2.0 * n * smoothed_max
    return obj, grad_u

def softmax(u):
    u_max = np.max(u)
    e = np.exp(u - u_max)
    return e / e.sum()

def objective_and_grad(u, n, lam):
    a = softmax(u)
    # convolution a*a
    size1 = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fa * fa, size1)[:2*n-1]
    max_val = conv.max()
    # smoothed max via log-sum-exp
    shifted = conv - max_val
    exp_term = np.exp(lam * shifted)   # stable
    Z = exp_term.sum()
    smoothed_max = max_val + np.log(Z) / lam
    p = exp_term / Z
    # correlation via FFT
    size2 = 1 << ((n + (2*n-1) -1).bit_length())  # n + 2n -2 = 3n-2

    fp2 = np.fft.rfft(p, size2)

    dot = np.dot(grad_a, a)
    grad_u = a * (grad_a - dot)
    obj = 2.0 * n * smoothed_max
    return obj, grad_u

"""
Propose a candidate step‑function using a smoothed ∞‑norm objective and
projected L‑BFGS on the soft‑max parametrisation (u → a = softmax(u)).
The objective is 2·n·max(conv(a,a)) with a_i ≥ 0, Σa_i = 1.
A log‑sum‑exp surrogate (parameter λ) makes the peak differentiable.
We start from the classic central‑binomial (arcsine) distribution,
run a few L‑BFGS stages while increasing λ, and finish with a short
random‑swap hill‑climb.  The whole search stops before the allotted
budget (≈1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(u: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector that sums to 1."""
    m = np.max(u)
    e = np.exp(u - m)
    return e / e.sum()


def _conv_self(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a*a using FFT, length = 2·len(a)‑1."""
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)
    return conv[: 2 * a.size - 1]


def _correlation(a: np.ndarray, p: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ p) where
    (a ⋆ p)[ℓ] = Σ_j a_j · p_{j+ℓ}.
    Returns the full length = len(a)+len(p)‑1 array.
    """
    fa = np.fft.rfft(a, size_fft)
    fp = np.fft.rfft(p, size_fft)
    corr = np.fft.irfft(fa * np.conj(fp), size_fft)
    return corr[: a.size + p.size - 1]


def _objective_and_grad(u: np.ndarray, n: int, lam: float):
    """
    Returns (objective, gradient) for the smoothed peak objective.
    Parameters
    ----------
    u   : real vector (log‑probabilities)
    n   : length of the step function
    lam : smoothing parameter (λ > 0)
    """
    # probabilities a_i (sum = 1, a_i > 0)
    a = _softmax(u)

    # ---- 1. self‑convolution and smoothed max -----------------------
    size_c = 1 << ((2 * n - 1).bit_length())            # next power of 2
    conv = _conv_self(a, size_c)                        # length 2n‑1
    max_val = conv.max()

    # log‑sum‑exp (max is shifted to zero for numerical stability)
    shifted = conv - max_val                 # ≤ 0
    exp_term = np.exp(lam * shifted)        # in [0,1]
    Z = exp_term.sum()
    smoothed_max = max_val + np.log(Z) / lam

    # ---- 2. gradient w.r.t. a (soft‑max variables) -----------------
    p = exp_term / Z                         # weights, same length as conv

    size_corr = 1 << ((n + (2 * n - 1) - 1).bit_length())   # >= 3n‑2
    corr_full = _correlation(a, p, size_corr)                # length 3n‑2
    centre = 2 * n - 2                                        # index of lag 0

    # term1_i = Σ_j a_j·p_{i+j}   (i = 0…n‑1)
    term1 = corr_full[centre : centre + n]
    # term2_i = Σ_j a_j·p_{i‑j}
    term2 = corr_full[centre - n + 1 : centre + 1][::-1]

    grad_a = term1 + term2               # ∂/∂a of the smoothed max

    # ---- 3. chain rule for the soft‑max parametrisation -------------
    dot = np.dot(grad_a, a)
    grad_u = a * (grad_a - dot)          # ∂/∂u

    # ---- 4. full objective (the quantity that evaluate_sequence returns) --
    obj = 2.0 * n * smoothed_max
    return obj, grad_u

    """Scalar objective for scipy.optimize (no gradient)."""
    obj, _ = _objective_and_grad(u, n, lam)
    return obj


def _fun_grad(u, n, lam):
    """Gradient of the objective for scipy.optimize."""
    _, grad = _objective_and_grad(u, n, lam)
    return grad
