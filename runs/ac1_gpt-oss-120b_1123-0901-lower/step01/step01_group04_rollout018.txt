"""
propose_candidate
=================

Generate a non‑negative step function (a list of floats) that minimises the
provided ``evaluate_sequence`` objective.

The method has three stages:

1. **Initial guess** – an accurate discrete version of the arcsine
   distribution (the known asymptotically optimal shape).
2. **Linear‑programming spreading** – repeatedly solve a small LP
   to push mass into positions where the current function is small while
   keeping the convolution bound unchanged.  This is the exact discrete analogue
   of the iterative scheme described in the statement.
3. **Local simulated‑annealing refinement** – cheap random moves that
   transfer a fraction of mass from one index to another; moves that
   improve the score are always kept, worse moves are occasionally accepted
   with a temperature that decays over time.

The whole search is bounded by the supplied *budget_s* (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# optional heavy libraries – they are available in the execution environment
import cvxpy as cp
try:
    from scipy.linalg import toeplitz
except Exception:                     # pragma: no cover
    toeplitz = None


def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Return a probability vector of length *n* that approximates the continuous
    arcsine density on [0,1] by integrating the pdf over each sub‑interval.
    The vector sums to one.
    """
    # CDF of the arcsine law: F(x) = (2/π)·arcsin(√x)
    k = np.arange(1, n + 1, dtype=np.float64) / n
    j = np.arange(0, n, dtype=np.float64) / n
    a = (2.0 / np.pi) * (np.arcsin(np.sqrt(k)) - np.arcsin(np.sqrt(j)))
    a /= a.sum()
    return a


def _conv_max(a: np.ndarray) -> float:
    """Return max(convolve(a, a)).  The sum of ``a`` is assumed to be 1."""
    return np.convolve(a, a).max()


def _lp_spread(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve a linear program that maximises the ℓ₁‑norm of a vector ``b`` while
    keeping the linear convolution ``f * b`` below the current bound ``M``.
    The returned vector is normalised to sum to one.
    """
    n = f.shape[0]

    # Build the (2n‑1) × n Toeplitz matrix that represents the linear convolution.
    if toeplitz is not None:
        # first column = f followed by n‑1 zeros, first row = f[0] then zeros
        col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
        T = toeplitz(col, row)[: 2 * n - 1, :]
    else:   # fallback – still works for the modest n we use
        T = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(2 * n - 1):
            lo = max(0, i - (n - 1))
            hi = min(i, n - 1)
            # indices j = lo … hi, coefficient f[i‑j]
            T[i, lo : hi + 1] = f[i - np.arange(lo, hi + 1)]

    # Linear programme -------------------------------------------------
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]                     # component‑wise ≤ M
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is fast for dense problems of this size
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False,
               max_iters=5000)
    if b.value is None:
        # numerical failure – fall back to the current f
        return f.copy()
    b_vec = np.asarray(b.value).flatten()
    if b_vec.sum() == 0:
        return f.copy()
    # Normalise to unit ℓ₁‑norm so we stay on the feasible manifold
    return b_vec / b_vec.sum()


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Main entry point used by the evaluator.  Returns a list of non‑negative
    heights.  The algorithm respects the time limit indicated by *budget_s*.
    """
    start = time.time()
    np.random.seed(seed)

    # -------------------------------------------------------------
    # 1.  Choose a length.
    # -------------------------------------------------------------
    if "length" in kwargs:
        n = int(kwargs["length"])
    else:
        try:
            n = len(height_sequence_1)          # type: ignore[name-defined]
        except Exception:
            n = 1500                               # moderate size, fast LP

    # -------------------------------------------------------------
    # 2.  Initial guess – discrete arcsine.
    # -------------------------------------------------------------
    a = _arcsine_sequence(n)
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_max(a)              # sum(a) == 1
    print(f"[init] n={n}, score={best_score:.6f}")

    # -------------------------------------------------------------
    # 3.  LP‑based spreading iterations.
    #    Each iteration keeps the current convolution bound M
    #    while enlarging the ℓ₁‑norm of a vector b that lives in the
    #    ‘low‑f’ region; a convex combination then reduces the bound.
    # -------------------------------------------------------------
    mix_factor = 0.6                 # start with a strong move
    for it in range(12):            # a dozen iterations are enough for a big gain
        if time.time() - start > budget_s - 5.0:
            break

        M = _conv_max(a)                         # current worst convolution value
        b = _lp_spread(a, M)                     # b is already normalised (sum = 1)

        # Convex combination – keep the ℓ₁‑norm = 1.
        a = (1.0 - mix_factor) * a + mix_factor * b

        cur_score = 2.0 * n * _conv_max(a)
        print(f"[lp] it={it:02d}, mix={mix_factor:.3f}, score={cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # slowly reduce the mixing weight – fine‑tune later stages
        mix_factor = max(0.1, mix_factor * 0.85)

    # -------------------------------------------------------------
    # 4.  Simulated‑annealing hill‑climbing on the best candidate.
    # -------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    temperature = 0.02 * cur_score               # initial temperature
    it = 0
    deadline = start + budget_s - 0.5

    while time.time() < deadline:
        # pick two distinct positions
        src, dst = np.random.choice(n, size=2, replace=False)

        # move a random fraction (up to 20 %) of the source mass
        max_move = min(a[src], 0.20)
        delta = max_move * np.random.rand()
        if delta == 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_score = 2.0 * n * _conv_max(a_new)

        # Metropolis acceptance
        if (new_score < cur_score or
                np.random.rand() < math.exp(-(new_score - cur_score) / temperature)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[SA improve] it={it:05d}, score={best_score:.6f}")

        it += 1
        # temperature schedule
        if it % 1000 == 0:
            temperature *= 0.9
            temperature = max(temperature, 1e-9)

    print(f"[final] total iterations={it}, best score={best_score:.6f}")
    # -------------------------------------------------------------
    # 5.  Return plain Python list.
    # -------------------------------------------------------------
    return best_seq.tolist()
