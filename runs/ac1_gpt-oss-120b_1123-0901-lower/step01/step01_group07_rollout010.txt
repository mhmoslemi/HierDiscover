def gamma_guess(p, n):
    x = 1-p
    beta = math.gamma(x)*math.gamma(x)/math.gamma(2*x)
    term = beta * (n**(1-2*p))
    if term <= 0:

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time()+budget_s-0.5
    n = 5000

    def make_seq(p,g):
        # raw shape
        i = np.arange(n, dtype=float)
        raw = (i + g) ** (-p)
        # scaling factor to avoid clipping
        max_raw = raw.max()
        if max_raw > 1000.0:
            raw = raw * (1000.0 / max_raw)
        return raw

    best_seq = None
    best_score = float('inf')
    best_p = None
    best_g = None

    # generate candidates
    p_vals = np.linspace(0.51,0.89,39)  # include many
    for p in p_vals:
        # compute gamma default guess
        try:
            guess = gamma_guess(p,n)
        except Exception:
            guess = 1.0
        # check also range around guess
        for factor in np.linspace(0.5,2.0,5):
            g = guess * factor
            if g <=0:
                continue
            seq = make_seq(p,g)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score=score; best_seq=seq.copy(); best_p=p; best_g=g
                print(f"[grid] p={p:.4f}, γ={g:.4f}, score={score:.6f}")
        # Also test a few random gamma values
        for _ in range(3):
            g = 10**np.random.uniform(-2,1) # gamma in [0.01,10]
            seq = make_seq(p,g)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score=score; best_seq=seq.copy(); best_p=p; best_g=g
                print(f"[grid random] p={p:.4f}, γ={g:.4f}, score={score:.6f}")

        if time.time()>deadline:
            break

    # local refinement using simulated annealing
    if best_p is None:
        best_p, best_g = 0.5, 1.0
        best_seq = make_seq(best_p,best_g)
        best_score = evaluate_sequence(best_seq.tolist())

    # parameters for annealing
    iteration = 0
    max_iter = 20000
    sigma_p = 0.05
    sigma_g = 0.5
    temperature = 1.0
    while time.time() < deadline and iteration < max_iter:
        iteration+=1
        # propose new candidate
        p_new = best_p + np.random.randn()*sigma_p
        g_new = best_g + np.random.randn()*sigma_g
        # keep in bounds
        if not (0.5 < p_new < 0.95 and 0.01 < g_new < 20):
            continue
        seq = make_seq(p_new,g_new)
        score = evaluate_sequence(seq.tolist())
        # acceptance criteria
        delta = score - best_score
        if delta < 0 or np.exp(-delta/temperature) > np.random.rand():
            # accept as new current (not necessarily best)
            # For simplicity we treat best as current
            # Accept modifications
            if score < best_score:
                best_score=score
                best_seq=seq.copy()
                best_p=p_new
                best_g=g_new
                print(f"[refine] new best p={p_new:.4f}, γ={g_new:.4f}, score={score:.6f}")
        # reduce temperature & step sizes gradually
        temperature *= 0.9999
        sigma_p *= 0.9999
        sigma_g *= 0.9999

    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} with p={best_p:.4f}, γ={best_g:.4f}")
    return result

"""
Propose a candidate step‑function (non‑negative heights) that beats the previous
best score (~1.57) and aims for the target 1.503.

Algorithm
---------
1.  Use the analytically‑motivated family  
        a_i = (i + γ) ** (‑p) ,  i = 0 … n‑1
    together with an *automatic scaling* step so that the built‑in
    clipping to 1000 in `evaluate_sequence` never distorts the shape.
    Scaling does not change the objective because the score is invariant
    under a constant factor.

2.  For a given exponent `p` the optimal shift `γ` (so that the maximal
    convolution entry is the central term rather than the trivial a₀²)
    satisfies roughly  

        γ ≈ (B(1‑p,1‑p)·n^{1‑2p})^{‑1/(2p)} ,

    where `B` is the beta function.  This provides a good starting value
    for each `p`.

3.  Perform a coarse grid search over `p ∈ [0.55,0.88]`.  For every `p`
    evaluate several `γ` values around the analytical guess (factors
    0.5 … 2.0) and a few random perturbations.  Keep the best pair.

4.  Refine the best pair with a simple stochastic hill‑climber /
    simulated‑annealing loop that proposes Gaussian perturbations of
    `p` and `γ`.  The step sizes and temperature are slowly decayed.
    The loop runs until the time budget is exhausted.

5.  Return the best sequence found (as a plain Python list).  All
    numbers are non‑negative, never exceed 1000, and the whole search
    respects the 1000 s limit.

The implementation uses only `numpy`, `math`, `time`, and the supplied
`evaluate_sequence`.  No file I/O or external solvers are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _beta(x: float) -> float:
    """Beta(x,x) = Gamma(x)^2 / Gamma(2*x).  x>0."""
    return math.gamma(x) * math.gamma(x) / math.gamma(2.0 * x)


def _gamma_guess(p: float, n: int) -> float:
    """
    Rough analytic value of γ that makes a₀² comparable to the central
    convolution term for the exponent p.
    """
    x = 1.0 - p                     # 0 < x < 0.5
    if x <= 0.0:
        return 1.0
    beta = _beta(x)
    term = beta * (n ** (1.0 - 2.0 * p))   # ≈ a₀² when γ is the right size
    # Guard against zero / negative (should not happen for the chosen ranges)
    if term <= 0.0:
        return 1.0
    gamma_val = term ** (-1.0 / (2.0 * p))
    # keep γ away from zero to avoid huge intermediate values
    return max(gamma_val, 1e-6)


def _make_sequence(p: float, gamma: float, n: int) -> np.ndarray | None:
    """

    Returns None if the raw vector contains NaN/Inf.
    """

    # Protect against overflow / invalid numbers
    if np.any(np.isnan(raw)) or np.any(np.isinf(raw)):
        return None

    max_raw = raw.max()
    if max_raw > 1000.0:
        raw = raw * (1000.0 / max_raw)   # uniform scaling – invariant for the score
    # Ensure non‑negativity (the formula already yields non‑negative values)
    return raw
