```
    target_sum = np.sqrt(2.0 * n)
    # Make a sum to target
    a = project_simplex(a, target_sum)

    # schedule
    start = time.time()
    while time.time() < time_limit:
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)
        T = max(0.001, (1 - frac) * cur_score)  # temperature bound relative to score
```

```
def lp_refine(a, t=0.5):
    n = a.shape[0]
    target_sum = a.sum()
    M = max_convolution(a)
    # Build convolution matrix A of shape (2*n-1, n)
    rows = 2*n - 1
    A = np.zeros((rows, n))
    for k in range(rows):
        i_low = max(0, k - (n-1))
        i_high = min(n-1, k)
        # for i in [i_low, i_high], partner = k - i
        # A[k,i] = a[partner]
        # This is row k picks a[partner]
        # Could fill via loop.
        A[k, i_low:i_high+1] = a[k - np.arange(i_low, i_high+1)]
    # Solve LP: maximize sum(g) -> minimize -1^T g
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if not res.success:
        return a
    g = res.x
    # Combine
    new_a = (1 - t) * a + t * g
    # Project onto simplex with target sum
    new_a = project_onto_simplex(new_a, target_sum)
    return new_a
```

assistantfinal**Solution Overview**

The objective  

\[
F(a)=\frac{2n\;\max (a*a)}{(\sum a_i)^2}
\]

is *scale‑invariant*: multiplying all entries of `a` by a constant does not change `F`.  
Therefore we can fix the ℓ₁‑norm of the sequence to  

\[
\sum a_i = \sqrt{2n},
\]

in which case `F(a)` simplifies to the maximum convolution value `M = max(a*a)`.  
Our job is thus to find a non‑negative vector on the simplex that makes the
largest entry of its autocorrelation (`np.convolve(a,a)`) as small as possible.

The previous runner used a simple projected gradient descent.  
To push the bound further we combine three ideas:

1. **Projected Simulated Annealing** – a global stochastic search that moves
   mass from one coordinate to another while staying on the simplex.
   The temperature controls the size of the move and the acceptance
   probability; a linear cooling schedule lets us escape shallow minima
   early and then converge.

2. **Occasional Gaussian perturbations** – with a small probability the whole
   vector is perturbed by a tiny zero‑mean Gaussian noise and re‑projected.
   This helps the search jump between distant basins.

3. **Linear‑programming refinement** – after the annealing phase we solve the
   LP described in the statement:

   \[
   \max\; \mathbf 1^\top g \quad\text{s.t.}\quad g\ge 0,\;
   (f*g)_k\le M\ \forall k ,
   \]

   where `f` is the best vector found so far and `M` its current maximal
   convolution value.  The LP pushes as much mass as possible into a new
   direction while respecting the same maximal autocorrelation.  The
   resulting `g` is mixed with `f` (convex combination) and projected back
   onto the simplex.  Repeating this step a few times reduces the bound
   further.

All heavy work (convolution, projection, LP) is performed with NumPy
and SciPy only; the whole routine stops automatically after the given
time budget (default 1000 s) and returns the best sequence discovered.

```python
"""
Propose a non‑negative height sequence that minimises the evaluation metric.

Algorithm
---------
1.   Fix the ℓ₁‑norm to √(2n) → the score equals the maximal entry of
     the convolution a*a.
2.   For many random problem sizes n we start from the known good construction
     (or a random point) and project it onto the simplex.
3.   Simulated annealing moves a small amount of mass from one index to another,
     always staying on the simplex.  Temperature decreases linearly; a
     Metropolis accept/reject step allows uphill moves.
     Occasionally a global Gaussian perturbation is added.
4.   After the annealing phase a short linear‑programming refinement is applied:
     maximise Σ g_i s.t. g≥0 and f*g ≤ M (component‑wise).  The new vector
     f←(1‑t)f+t g (t≈0.5) is projected back onto the simplex.
5.   The best sequence found over the whole time budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray) -> float:
    """
    Returns max_k (a * a)[k] using an FFT based convolution.
    a must be a 1‑D real array.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power‑of‑two for efficient FFT
    N = 1 << ((conv_len - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    # keep only the needed part
    return conv[:conv_len].max()


def _score_normalised(a: np.ndarray) -> float:
    """
    For a vector that already satisfies Σ a = √(2·n) the score equals the
    maximal convolution entry.
    """
    n = a.shape[0]
    S = a.sum()
    if S < 0.01:
        return float("inf")
    M = _max_convolution(a)
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------------------
# Simulated annealing – global stochastic optimisation
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Perform simulated annealing on the simplex.
    Returns the best vector found and its (normalised) score.
    """
    n = a0.shape[0]
    target_sum = math.sqrt(2.0 * n)
    a = _project_onto_simplex(a0, target_sum)

    cur_score = _max_convolution(a)          # because Σa = target_sum
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    # initial temperature proportional to the current score
    T0 = max(cur_score, 0.001)

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        # linear cooling to a tiny temperature
        T = (1.0 - frac) * T0 + 0.001 * frac

        # ----- propose a move that preserves the sum -----
        # pick a source index with positive mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # maximal transferable mass from i depends on temperature
        max_move = a[i]
        step_limit = T * target_sum / n           # O(T) scaling
        max_delta = min(max_move, step_limit)
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta      # amount moved from i to j

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate the proposal
        new_score = _max_convolution(a_new)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.2f}]  SA new best = {best_score:.6f} (n={n})")

        # ----- occasional global perturbation (helps escaping deep minima) -----
        if np.random.rand() < 0.005:
            sigma = T * target_sum / (10.0 * n)
            a = a + sigma * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, target_sum)
            cur_score = _max_convolution(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}]  SA perturbation best = {best_score:.6f} (n={n})")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP step” from the statement)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, mix: float = 0.5) -> np.ndarray:
    """
    Perform one LP refinement step.
    The LP maximises Σ g_i under g≥0 and (a * g)_k ≤ M for all k.
    The result g is mixed with a (convex combination) and projected back.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy not available – skip refinement
        return a

    n = a.shape[0]
    M = _max_convolution(a)

    rows = 2 * n - 1
    # Build the convolution matrix A where (A @ g)_k = (a * g)_k
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        A[k, idx] = a[partner]

    # maximise Σ g_i  <=> minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a

    g = res.x
    # convex combination and re‑projection onto the simplex
    target_sum = a.sum()
    a_new = (1.0 - mix) * a + mix * g
    a_new = _project_onto_simplex(a_new, target_sum)
    return a_new
```
