def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

heights whose sum is 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine (beta(1/2,1/2)) coefficients
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^(-1/2) = Σ C(2i,i)/4^i z^i."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Optional: linear‑programming step (SciPy)
# ----------------------------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float):
    """Solve max Σ b_i  s.t. (a*b)_k ≤ M for all k, b ≥ 0.
    Returns the optimal b as a NumPy array, or None if the LP fails."""
    try:
        from scipy.optimize import linprog
    except Exception:                     # SciPy not installed – give up
        return None

    n = a.shape[0]
    m = 2 * n - 1

    # Build the Toeplitz‐like constraint matrix.
    # Column i is the vector a placed on rows i … i+n‑1.
    A = np.zeros((m, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a

    # Linear program: maximise 1ᵀb  ⇔  minimise –1ᵀb
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n
    b_ub = np.full(m, M, dtype=float)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0, seed: int = 42, **kwargs):
    """
    Produce a candidate step‑function (non‑negative sequence).

    Parameters
    ----------
    budget_s : float, optional
        Overall time budget in seconds.
    seed : int, optional
        Random seed – used for the stochastic refinement.
    **kwargs
        Ignored – kept for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum == 1.0) that minimises the evaluation function.
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction exists we honour its
    # length (up to a sensible ceiling); otherwise we fall back to 2000.
    # ------------------------------------------------------------------
    try:
        n = min(2500, len(height_sequence_1))   # pylint: disable=undefined-variable
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Start from the arcsine coefficients (the analytically known
    # “constant‑convolution’’ shape) and normalise to sum == 1.
    # ------------------------------------------------------------------
    coeffs = _arcsine_coeffs(n)
    a = coeffs / coeffs.sum()          # a now lies on the probability simplex

    # Helper to compute the score used by evaluate_sequence (sum(a) == 1):
    def _score(vec):
        return float(2 * len(vec) * np.convolve(vec, vec).max())

    best_seq = a.copy()
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Main optimisation loop – alternating LP + mixing.
    # The loop stops when time runs out, when no progress is made or
    # after a modest number of LP iterations (the LP is the expensive step).
    # ------------------------------------------------------------------
    max_lp_iters = 30
    t_mix = 0.3                       # initial mixing factor

    for it in range(max_lp_iters):
        # time guard
        if time.time() - start > budget_s - 0.5:
            break

        # current bound
        conv = np.convolve(a, a)
        M = conv.max()

        # solve LP to obtain a vector b that maximises Σb under (a*b)≤M
        b = _lp_max_sum(a, M)
        if b is None:
            # LP not available – fall back to pure stochastic search
            break

        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # normalise b to the simplex (same total mass as a)
        b_norm = b / sum_b

        # Attempt a mixing step.  If the bound does not improve we shrink t.
        improved = False
        for _ in range(8):          # try a few decreasing t’s
            new_a = (1.0 - t_mix) * a + t_mix * b_norm
            new_score = _score(new_a)
            if new_score < best_score - 1e-12:
                a = new_a
                best_seq = a.copy()
                best_score = new_score
                improved = True
                # a slight increase of t for the next iteration can speed up
                t_mix = min(0.5, t_mix * 1.05)
                break
            else:
                t_mix *= 0.5        # back‑off
                if t_mix < 1e-4:
                    break
        if not improved:
            # No improvement with the current LP solution – stop early.
            break

    # ------------------------------------------------------------------
    # Light stochastic polishing: random pairwise mass shifts.
    # This cheap step often squeezes out a few extra digits.
    # ------------------------------------------------------------------
    try:
        max_iters = 5000
        for _ in range(max_iters):
            if time.time() - start > budget_s - 0.5:
                break

            # pick two distinct indices and move a tiny amount epsilon
            i, j = np.random.choice(n, 2, replace=False)
            if best_seq[i] <= 0:
                continue
            eps = best_seq[i] * 0.02 * np.random.rand()   # up to 2 % of the source
            if eps <= 0:
                continue

            cand = best_seq.copy()
            cand[i] -= eps
            cand[j] += eps                     # sum unchanged, non‑negativity stays

            sc = _score(cand)
            if sc < best_score:
                best_seq = cand
                best_score = sc
    except Exception:
        # If anything goes wrong we just keep the current best_seq.
        pass

    # ------------------------------------------------------------------
    # Final logging – the evaluator will also re‑compute the score.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())   # pylint: disable=undefined-variable
        print(f"[propose_candidate] length={n}, final_score={final_score:.6f}")
    except Exception:
        # evaluate_sequence may not be importable in a sandbox; ignore.
        pass

    return best_seq.tolist()
