"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The basic idea is to use a *scaled* power‑law shape

    a_i = (i + α·n) ** (‑p) · exp(‑λ·i)           i = 0 … n‑1

where  

* **p**   – power‑law exponent (≈0.5…0.85)  
* **α**   – shift expressed as a fraction of the length n (α·n is the usual γ)  
* **λ**   – tiny exponential decay that gives us an extra degree of freedom  
* **n**   – length of the sequence (larger n reduces discretisation error).

For each triple (p,α,λ) we build the sequence, call the provided
`evaluate_sequence`, and keep the best result.  The search consists of  

1. A coarse grid over (p,α,λ).  
2. A stochastic refinement that perturbs the best parameters
   (log‑normal perturbation for α, Gaussian for the others).  
3. A very cheap occasional change of the length *n* (few preset values).

All loops respect the global time budget (default 1000 s).  The function
returns the best sequence found as a plain Python list (the scorer expects
that format).  The algorithm stays completely inside the allowed
environment – only `numpy` and the standard library are used.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper – build a candidate sequence
# --------------------------------------------------------------
def _make_seq(n: int, p: float, alpha: float, lam: float = 0.0) -> np.ndarray:
    """
    a_i = (i + alpha*n) ** (-p) * exp(-lam*i)
    """
    gamma = alpha * n
    i = np.arange(n, dtype=np.float64)
    seq = (i + gamma) ** (-p)
    if lam != 0.0:
        seq = seq * np.exp(-lam * i)
    return seq


# --------------------------------------------------------------
# Main entry point for the competition framework
# --------------------------------------------------------------
def propose_candidate(seed: int = 2024,
                     budget_s: float = 1000.0,
                     **kwargs) -> list:
    """
    Search for a low‑score step function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Hard time budget (seconds).  The function stops a little before the
        limit to avoid being killed.
    **kwargs
        Ignored – present only for compatibility with the previous code.

    Returns
    -------
    list
        Heights of the best step function found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.15   # safety margin

    # -----------------------------------------------------------------
    # 1) Choose a reasonable start length.  Larger n reduces discretisation
    #    error but costs a bit more time per evaluation.
    # -----------------------------------------------------------------
    n_candidates = [3000, 4000, 5000]          # we will try these lengths later
    n = n_candidates[1]                       # primary length (4000)

    best_seq = None
    best_score = float('inf')
    best_params = None   # (p, alpha, lam, n)

    # -----------------------------------------------------------------
    # 2) Coarse grid search over (p, alpha, lam)
    # -----------------------------------------------------------------
    p_vals = np.linspace(0.50, 0.85, 18)               # exponent
    alpha_vals = np.geomspace(0.001, 2.0, 30)          # shift as fraction of n
    lam_vals = [0.0, 5e-4, 1e-3, 2e-3, 5e-3]           # tiny exponential decay

    print("[grid] starting coarse search")
    for p in p_vals:
        for alpha in alpha_vals:
            for lam in lam_vals:
                if time.time() > deadline:
                    break
                seq = _make_seq(n, p, alpha, lam)
                # evaluate expects a plain list
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p, alpha, lam, n)
                    print(f"[grid] new best p={p:.4f}, α={alpha:.4f}, λ={lam:.5f}, "
                          f"n={n}, score={score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # -----------------------------------------------------------------
    # 3) Occasionally try a different length (very cheap – only a few runs)
    # -----------------------------------------------------------------
    for n_alt in n_candidates:
        if n_alt == n:
            continue
        # reuse current best parameters, only change n
        p, alpha, lam, _ = best_params
        seq = _make_seq(n_alt, p, alpha, lam)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_params = (p, alpha, lam, n_alt)
            print(f"[len] improved length n={n_alt}, score={score:.6f}")

    # -----------------------------------------------------------------
    # 4) Stochastic refinement around the current best parameters
    # -----------------------------------------------------------------
    # tuning of step sizes – chosen by quick experiments
    refinement_iters = 15000
    print("[refine] starting stochastic refinement")
    for it in range(refinement_iters):
        if time.time() > deadline:
            break

        # unpack best parameters
        p_best, alpha_best, lam_best, n_best = best_params

        # propose new parameters (log‑normal perturbation for alpha)
        p_new = p_best + np.random.randn() * 0.012
        # keep alpha positive and explore multiplicatively
        log_alpha = np.log(alpha_best) + np.random.randn() * 0.08
        alpha_new = np.exp(log_alpha)
        lam_new = lam_best + np.random.randn() * 4e-4
        n_new = n_best

        # occasionally (5 % of iterations) try a different length
        if np.random.rand() < 0.05:
            n_new = int(np.random.choice(n_candidates))

        # enforce simple bounds
        if not (0.48 < p_new < 0.92):
            continue
        if not (5e-4 < alpha_new < 5.0):
            continue
        if lam_new < 0.0:
            lam_new = 0.0
        if lam_new > 0.01:
            lam_new = 0.01

        seq = _make_seq(n_new, p_new, alpha_new, lam_new)
        # Avoid sequences that are essentially zero (evaluate_sequence would
        # reject them because the sum would be <0.01)
        if np.sum(seq) < 0.02:

        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_params = (p_new, alpha_new, lam_new, n_new)

    # 5) Finalisation – make sure we return a plain Python list
    # -----------------------------------------------------------------
    result = best_seq.tolist()

          f"with p={best_params[0]:.4f}, α={best_params[1]:.4f}, "
          f"λ={best_params[2]:.5f}, n={best_params[3]}")
    return result
