    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    try:
        a = np.array(height_sequence_1, dtype=float)

        a = np.ones(2100, dtype=float)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print("[init] n=",n,"score",best_score)

    # Functions
    def conv_max(arr):
        b = np.convolve(arr, arr)
        return b.max(), int(b.argmax()), b

    def build_A(f):

        # each row k = sum_i f[i] * g_{k-i}
        # A[k,i] = f[i] if 0 <= k-i < n else 0
        # use broadcasting
        # we can fill by loops
        A = np.zeros((2*n-1, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = f[i]
        return A

    # Precompute maybe? Not needed each iteration (f changes)
    
    iteration = 0
    while time.time() < deadline:
        # compute M
        M, k_max, conv_f_f = conv_max(a)
        # early exit if already good enough
        if M <= 1.503+1e-6:
            print("[early] target reached at iter",iteration)
            best_seq = a.copy()
            break

        # Build LP matrix A
        A = np.zeros((2*n-1, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = a[i]  # conv(f,g)

        # b vector = M (same for all rows)
        b_vec = np.full(2*n-1, M, dtype=float)

        # Solve LP: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n, dtype=float)
        bounds = [(0, None) for _ in range(n)]
        # Not using huge memory, but okay
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            print("[lp] failed at iter", iteration, "message", res.message)
            # fallback: random perturbation
            jitter = np.random.randn(n) * 0.01 * target_sum / n
            a = np.maximum(a + jitter, 0)
            a = a * (target_sum / a.sum())

        g_opt = res.x
        sum_g = g_opt.sum()
        # scaling now
        if sum_g <= 0:
            print("[lp] zero sum")
            break

        g_norm = g_opt * (target_sum / sum_g)

        # try mixing parameters
        improved = False
        for t in (0.5, 0.3, 0.1, 0.05):
            a_new = (1 - t) * a + t * g_norm
            # ensure non-negative and sum fixed (should be)

            if M_new < M * (1 - 1e-8):

"""
Iterative LP + gradient refinement search for low‑autocorrelation step functions.

The objective of `evaluate_sequence` (under the usual normalisation
∑a_i = √(2n)) equals the maximal autocorrelation value

    M(a) = max_k (a * a)[k] .

We therefore try to minimise M.

Algorithm
---------
1. Start from the best known construction (`height_sequence_1` or a uniform
   vector) and normalise the sum to √(2n).
2. Repeat while time permits
   a) Compute the current peak M = max_k (a*a)[k].
   b) Solve the linear programme

        maximise   ∑ g_i
        subject to  (a * g)[k] ≤ M   for all k = 0 … 2n‑2
                    g_i ≥ 0

      The constraints are linear in the unknown vector g because a is fixed.
      The LP gives a direction g that can increase the total mass without
      raising the current peak.
   c) Rescale g to the required total sum S = √(2n) and try several convex
      combinations a′ = (1‑t)·a + t·g (t ∈ {0.5,0.3,0.1,0.05}).  Keep the one that
      reduces M.
   d) Apply a short projected‑gradient descent (the exact gradient of M is
      known) to polish the new vector.
   e) Enforce symmetry a[i] = a[n‑1‑i] (the optimum is known to be symmetric)
      and clip to the limits required by `evaluate_sequence`.
   f) Keep the best sequence seen so far.
3. Return the best sequence (its maximal convolution equals the reported
   score).

The LP is solved with SciPy’s HiGHS implementation; if SciPy is not
available a pure gradient‑descent fallback is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray):
    """Return (M, k_max, full_conv) where M = max convolution a*a."""
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())
    M = float(conv.max())
    return M, k_max, conv

# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray, target_sum: float, lr: float) -> np.ndarray:
    """
    One projected gradient step for the max‑autocorrelation objective.
    The exact gradient at the peak index k_max is used.
    """
    n = a.shape[0]
    S = a.sum()
    if S == 0.0:
        return a.copy()

    M, k_max, _ = _conv_max(a)

    # reversed part a_{k_max-p}
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]

    # gradient of M (see the derivation in the statement)
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)               # non‑negative orthant
    sum_new = a_new.sum()
    if sum_new > 0.0:
        a_new *= target_sum / sum_new            # renormalise to S
    else:
        a_new = a.copy()

    np.clip(a_new, 0.0, 1000.0, out=a_new)       # safety limits
    return a_new

# ----------------------------------------------------------------------
def _gd_refine(a: np.ndarray, target_sum: float,
               max_iter: int = 500,
               lr0: float = 0.5,
               decay: float = 1e-4,
               jitter_interval: int = 1000) -> np.ndarray:
    """Several gradient steps with a decaying learning rate."""
    a_cur = a.copy()
    for it in range(max_iter):
        lr = lr0 / (1.0 + decay * it)
        a_cur = _gradient_step(a_cur, target_sum, lr)

        # tiny random jitter every few hundred steps
        if jitter_interval and it % jitter_interval == 0 and it > 0:
            jitter = np.random.randn(a_cur.size) * 0.001 * target_sum / a_cur.size
            a_cur = np.maximum(a_cur + jitter, 0.0)
            a_cur *= target_sum / a_cur.sum()
    return a_cur

# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[n‑1‑i] (simple arithmetic mean)."""
    return (a + a[::-1]) * 0.5

# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A_ub for the LP:
        (f * g)[k] = (A_ub @ g)[k] .
    Shape is (2n‑1, n).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]         # contribution of g_i to all admissible shifts
    return A

# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP that maximises ∑g_i subject to (f * g)[k] ≤ M.
    Returns the optimal g (numpy array) or None if the LP fails.
    """
    if _opt is None:       # SciPy unavailable – cannot solve LP
        return None

    n = f.size
    A = _build_conv_matrix(f)                 # (2n‑1) × n
    b = np.full(A.shape[0], M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)         # maximise sum -> minimise -sum
    bounds = [(0.0, None) for _ in range(n)]

    # HiGHS is fast and supports a time limit; we give it a few seconds.
    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                           method='highs',
                           options={'presolve': True, 'time_limit': 5.0})
    except Exception:                         # pragma: no cover
        return None

    if not res.success:
        return None
    return np.array(res.x, dtype=np.float64)

# ----------------------------------------------------------------------
def _random_jitter(a: np.ndarray, target_sum: float,
                   scale: float = 0.001) -> np.ndarray:
    """Add a tiny zero‑mean Gaussian perturbation and renormalise."""
    jitter = np.random.randn(a.size) * scale * target_sum / a.size
    a_new = np.maximum(a + jitter, 0.0)
    a_new *= target_sum / a_new.sum()
    return a_new

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed (for reproducibility).
    budget_s : float
        Wall‑clock time budget in seconds.  The function stops a few seconds
        earlier to guarantee a return value.
    kwargs : dict
        Unused – kept for API compatibility.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Initialise a starting point
    base = globals().get('height_sequence_1')
    if base is None:
        base = np.ones(2100, dtype=np.float64)    # safe fallback
    a = np.array(base, dtype=np.float64)

    n = a.size
    target_sum = math.sqrt(2.0 * n)               # normalisation used by the paper
    a = a * (target_sum / a.sum())                # enforce the required sum
    a = _symmetrize(a)

    best_seq = a.copy()
    best_M, _, _ = _conv_max(a)
    best_score = best_M
    print(f"[init] n={n}, initial M={best_score:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Compute current peak and stop early if we are already below target
        cur_M, _, _ = _conv_max(a)
        if cur_M <= 1.503 + 1e-9:
            print(f"[early] target reached at iter {iteration}, M={cur_M:.6f}")
            best_seq = a.copy()
            best_score = cur_M
            break

        # --------------------------------------------------------------
        # 1) Try to improve using the LP step (if SciPy is available)
        if _opt is not None:
            g_opt = _solve_lp_for_g(a, cur_M)
        else:
            g_opt = None

        improved = False
        if g_opt is not None and g_opt.sum() > 0.0:
            # Scale g to the required total sum
            g_norm = g_opt * (target_sum / g_opt.sum())
            g_norm = _symmetrize(g_norm)

            # Check whether simply adopting g_norm already improves the peak
            g_M, _, _ = _conv_max(g_norm)
            if g_M < cur_M - 1e-8:
                a = g_norm
                cur_M = g_M
                improved = True
                print(f"[iter {iteration:06d}] LP direct improvement M={cur_M:.6f}")

            if not improved:
                # Try convex combinations with the current a
                for t in (0.5, 0.3, 0.1, 0.05):
                    a_mix = (1.0 - t) * a + t * g_norm
                    a_mix = _symmetrize(a_mix)
                    a_mix = np.clip(a_mix, 0.0, 1000.0)
                    a_mix = a_mix * (target_sum / a_mix.sum())
                    mix_M, _, _ = _conv_max(a_mix)
                    if mix_M < cur_M - 1e-8:
                        a = a_mix
                        cur_M = mix_M
                        improved = True
                        print(f"[iter {iteration:06d}] LP mix t={t:.2f} M={cur_M:.6f}")
                        break

        # --------------------------------------------------------------
        # 2) If LP did not yield improvement, perform a few gradient steps
        if not improved:
            # Gradient descent with a modest number of steps
            a = _gd_refine(a, target_sum, max_iter=400,
                           lr0=0.4, decay=5e-5, jitter_interval=0)
            a = _symmetrize(a)
            cur_M, _, _ = _conv_max(a)

        # --------------------------------------------------------------
        # 3) Occasionally add a small random perturbation to escape plateaus
        if iteration % 2000 == 0 and iteration > 0:
            a = _random_jitter(a, target_sum, scale=0.002)
            a = _symmetrize(a)
            cur_M, _, _ = _conv_max(a)

        # --------------------------------------------------------------
        # Keep the best ever seen
        if cur_M < best_score:
            best_score = cur_M
            best_seq = a.copy()
            print(f"[iter {iteration:06d}] new best M={best_score:.6f}")

        iteration += 1

    # Final report
    print(f"[done] iterations={iteration}, best M={best_score:.6f}")
    # Convert to plain Python list for the harness
    return best_seq.tolist()
