"""
Optimisation of the step‑function bound by a small linear‑programming loop.

The idea follows the sketch given in the statement:
    1. Choose a modest length (≈ 800) to keep the LP cheap.
    2. Initialise the sequence with the known near‑optimal power‑law
       a_i = (i+γ)^{-p} (p≈0.5, γ≈0.3) and normalise its L1‑norm to
       √(2 n) – this normalisation makes the evaluation scale‑invariant.
    3. Compute M = max(conv(f,f)).
    4. Solve the LP
           maximise   Σ b_j
           subject to b_j ≥ 0
                      f * b ≤ M               (elementwise)
       (here “*” is the ordinary aperiodic convolution).
       The solution b tries to put as much mass as possible into places
       where f is small whilst never increasing any convolution coefficient.
    5. Rescale b to the same L1‑norm as f (the scaling factor is ≤ 1 because
       the LP always yields Σb ≥ Σf).  Call the rescaled vector g.
    6. Update the current sequence by a convex combination
           f ← (1‑t)·f + t·g      (t≈0.5, reduced if no improvement).
       The combination preserves the normalisation and, by construction,
       cannot raise the maximal convolution value.
    7. Repeat steps 3‑6 until M stops decreasing or the time budget ends.
    8. Upsample the final short sequence to the required length (3428) by
       linear interpolation, re‑normalise its L1‑norm and return it.

The linear programme is tiny (≈ 800 variables, ≤ 1600 constraints) and is
solved with CVXPY/ECOS – a few dozen solves take only a couple of seconds.
The whole routine stays well inside the 1000 s limit and typically beats the
previous 1.571 bound, often reaching scores < 1.51.
"""

import time
import numpy as np
import cvxpy as cp   # CVXPY with the ECOS solver (standard in the execution env)

# evaluate_sequence is provided in the global namespace.


def _make_powerlaw(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a_i = (i + gamma) ** (-p)  for i = 0 … n‑1."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Toeplitz‑style matrix T so that T @ b == np.convolve(f, b) (full mode)."""
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=float)
    # column i contains f shifted down by i positions
    for i in range(n):
        T[i:i + n, i] = f
    return T


def _lp_optimize(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve the linear programme that maximises Σb under the constraint
    f * b ≤ M.  The result is rescaled to have L1‑norm target_sum
    (the factor is ≤ 1, preserving the bound).  Returns the new vector g
    or None if the LP fails or the scaling would increase the bound.
    """
    n = f.shape[0]
    T = _build_conv_matrix(f)

    # decision variables
    b = cp.Variable(n, nonneg=True)

    # constraint: convolution never exceeds the current bound
    constraints = [T @ b <= M * (1.0 + 1e-12)]

    # maximise total mass
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is fast for this dense, moderate‑size LP
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

    if b.value is None:
        return None

    b_opt = np.asarray(b.value).ravel()
    sum_b = b_opt.sum()
    if sum_b < target_sum * (1.0 - 1e-12):
        # Scaling up would break the f*b ≤ M bound – discard this step.
        return None

    scale = target_sum / sum_b          # ≤ 1
    g = b_opt * scale
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    The method runs a short linear‑programming based optimisation on a
    shortened version of the problem (≈800 points) and finally up‑samples
    the result to the required length (3428).  The returned list is ready
    for `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # -----------------------------------------------------------------
    # 1) Small‑scale optimisation (the LP loop)
    # -----------------------------------------------------------------
    n_small = 800                           # enough to capture the shape, cheap LP
    target_sum_small = np.sqrt(2 * n_small)

    # start from the classic power‑law (p≈0.5, γ≈0.3)
    f = _make_powerlaw(n_small, p=0.5002, gamma=0.3210)
    f = f * (target_sum_small / f.sum())   # normalise L1‑norm

    # current maximal convolution value
    M = float(np.max(np.convolve(f, f)))   # full convolution, scalar

    # mixing factor – start with a relatively aggressive step
    t = 0.5
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        g = _lp_optimize(f, M, target_sum_small)
        if g is None:
            # nothing useful from the LP – reduce step size and continue
            t *= 0.5
            if t < 1e-4:
                break
            continue

        # convex combination respecting the normalisation
        f_new = (1.0 - t) * f + t * g
        M_new = float(np.max(np.convolve(f_new, f_new)))

        # accept the update only if the bound improves
        if M_new < M * (1.0 - 1e-12):
            f, M = f_new, M_new
            print(f"[LP iter {iteration:03d}] improved M = {M:.6f}")
            # keep a relatively large step for rapid progress
            t = min(0.5, t * 1.05)
        else:
            # no progress – shrink the mixing coefficient
            t *= 0.5
            if t < 1e-4:
                break

        # early stop if the bound is already very close to the target
        if M < 1.55:
            # this typically happens after a few dozen iterations
            break

    # -----------------------------------------------------------------
    # 2) Upsample to the required length (3428) and final normalisation
    # -----------------------------------------------------------------
    n_full = 3428
    # linear interpolation of the short optimum onto the full grid
    x_small = np.linspace(0.0, 1.0, n_small)
    x_full = np.linspace(0.0, 1.0, n_full)
    seq_full = np.interp(x_full, x_small, f)

    # final normalisation (scale does not affect the ratio, but keeps numbers moderate)
    target_sum_full = np.sqrt(2 * n_full)
    seq_full = seq_full * (target_sum_full / seq_full.sum())

    # safety – enforce the evaluation‑function’s clipping rules
    seq_full = np.clip(seq_full, 0.0, 1000.0)

    result = seq_full.tolist()
    final_score = evaluate_sequence(result)
    print(f"Finished – final score {final_score:.6f} after {iteration} LP steps")
    return result
