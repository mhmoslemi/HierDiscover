def hill_climb(a, conv, n, start_time, total_budget):
    # a: np.ndarray, conv: np.ndarray, both with sum=1.
    # Simulated annealing / greedy.
    best_score = 2 * n * conv.max()
    best_seq = a.copy()
    # parameters
    temperature = 0.01
    min_temp = 1e-8
    cooling_rate = 0.9995
    step_factor = 0.2
    it = 0
    no_improve = 0
    deadline = start_time + total_budget - 0.5
    while time.time() < deadline:
        # recompute max index each iteration
        p = conv.argmax()
        # compute gradient w
        L = max(0, p - (n-1))
        U = min(n-1, p)
        w = np.zeros_like(a)
        idx = np.arange(L, U+1)
        w[L:U+1] = 2.0 * a[p - idx]
        # choose src (largest w among indices with a>eps)
        eps = 1e-12
        candidates = np.where(a > eps)[0]
        if candidates.size == 0:
            break
        src = candidates[np.argmax(w[candidates])]
        # choose dst (smallest w)
        # We allow any index, but prefer where a not near 1 (some upper bound)
        dst_candidates = np.arange(n)
        # ensure dst != src
        if len(dst_candidates) > 1:
            # compute sorted candidates by w
            sorted_idx = np.argsort(w)
            # pick first that is not src
            for cand in sorted_idx:
                if cand != src:
                    dst = cand
                    break
        else:
            dst = src
        if src == dst:
            # fallback random
            src, dst = np.random.choice(n, size=2, replace=False)
        # compute delta
        delta = step_factor * a[src]
        if delta < eps:
            # try smaller step
            delta = step_factor * (a[src] + a[dst]) * 0.5
        if delta <= 0:
            # no move possible
            continue
        # store old a for conv update
        a_old = a.copy()
        # compute new convex
        conv_new = conv.copy()

        conv_new[src:src+n] -= 2.0 * delta * a_old
        conv_new[dst:dst+n] += 2.0 * delta * a_old

        conv_new[2*dst] += delta * delta
        conv_new[src+dst] -= 2.0 * delta * delta

        accept = False
        if new_score < best_score:

                pass

            a = a_old
            # conv remains unchanged
            no_improve += 1
        # temperature cooling
        temperature = max(min_temp, temperature * cooling_rate)
        # gradually decay step_factor

            step_factor = 0.2

def build_arcsine(n):
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * ((2*k-1)*(2*k)) / (4.0*k*k)
    a /= a.sum()
    return a

def compute_conv(a):
    return np.convolve(a, a)

    conv_new[dst:dst+n] += 2.0 * delta * a_old
    conv_new[2*src] += delta * delta
    conv_new[2*dst] += delta * delta
    conv_new[src+dst] -= 2.0 * delta * delta
    return conv_new

"""
Gradient‑guided hill climbing for the “step‑function” evaluation problem.

The search starts from the discrete arcsine distribution –
a classically optimal shape with asymptotic constant π/2.
At each iteration we locate the index `p` where the current
convolution `a*a` attains its maximum.  The contribution of each
coefficient `a[i]` to that peak is `2*a[p-i]` (when the index lies
in the overlap range).  We therefore move a small amount of mass
from the index with the largest contribution to the index with the
smallest contribution.  A tiny simulated‑annealing acceptance rule
allows occasional uphill moves, and a periodic full recomputation
of the convolution avoids drift.  If the search stalls we apply a
random jitter and restart the local search.

The routine respects the overall time budget (default 1000 s), works
with any length (uses the length of `height_sequence_1` if available,
otherwise tries a few candidates), and finally returns a plain
Python list of non‑negative numbers suitable for `evaluate_sequence`.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper 1 – Build the (unnormalised) arcsine coefficients and normalise.
# ----------------------------------------------------------------------
def build_arcsine(n: int) -> np.ndarray:
    """Return a length‑`n` non‑negative array whose entries sum to 1.
    The shape is the discrete arcsine distribution:
        a_k ∝ C(2k,k)/4**k .
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # recurrence a[k] = a[k‑1] * ((2k‑1)(2k)) / (4 k²)
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()                 # normalise so that sum(a) == 1
    return a

# ----------------------------------------------------------------------
# Helper 2 – Full convolution (used only for occasional recomputation).
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution a*a."""
    return np.convolve(a, a)

# ----------------------------------------------------------------------
# Helper 3 – Efficient rank‑2 update of the convolution after moving
#            a small amount δ from src to dst.
# ----------------------------------------------------------------------
def update_convolution(conv: np.ndarray, a_old: np.ndarray,
                       src: int, dst: int, delta: float) -> np.ndarray:
    """
    Return the convolution after the transformation
        a' = a_old  – δ·e_src + δ·e_dst .
    This uses the exact identity  (a+d)*(a+d) = a*a + 2*a*d + d*d .
    """
    n = a_old.shape[0]
    conv_new = conv.copy()
    # contribution of 2*a*d  (pairwise cross terms)
    conv_new[src:src + n] -= 2.0 * delta * a_old
    conv_new[dst:dst + n] += 2.0 * delta * a_old
    # contribution of d*d  (the three point‑mass terms)
    conv_new[2 * src] += delta * delta
    conv_new[2 * dst] += delta * delta
    conv_new[src + dst] -= 2.0 * delta * delta
    return conv_new

# ----------------------------------------------------------------------
# Helper 4 – One hill‑climbing run for a fixed length.
# ----------------------------------------------------------------------
def hill_climb(start_a: np.ndarray, start_conv: np.ndarray,
               deadline: float) -> Tuple[np.ndarray, float]:
    """
    Perform a gradient‑guided hill climbing on `start_a`.
    Returns the best sequence found and its associated score.
    The routine stops when `deadline` (a Unix timestamp) is reached.
    """
    a = start_a.copy()
    conv = start_conv.copy()
    n = a.shape[0]

    # current best
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    # annealing / step‑size schedules
    temperature = 0.02
    step_factor = 0.20
    min_temp = 1e-8
    cooling = 0.99995

    it = 0
    no_improve = 0
    eps = 1e-12

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify the active peak of the convolution.
        # --------------------------------------------------------------
        p = int(conv.argmax())

        # --------------------------------------------------------------
        # Build the subgradient w_i = 2 * a[p-i] (zero outside the overlap).
        # --------------------------------------------------------------
        L = max(0, p - (n - 1))
        U = min(n - 1, p)
        w = np.zeros(n, dtype=np.float64)
        idx = np.arange(L, U + 1, dtype=int)
        w[L:U + 1] = 2.0 * a[p - idx]

        # --------------------------------------------------------------
        # Choose source (largest contribution) and destination (smallest).
        # --------------------------------------------------------------
        positive = np.where(a > eps)[0]
        if positive.size == 0:
            break
        src = positive[int(np.argmax(w[positive]))]
        dst = int(np.argmin(w))
        if dst == src:
            # fall back to a random different index
            dst = np.random.choice(np.delete(np.arange(n), src))

        # --------------------------------------------------------------
        # Propose moving a tiny fraction of mass from src to dst.
        # --------------------------------------------------------------
        delta = step_factor * a[src]
        if delta < eps:
            # if src is already tiny, pick a smaller step based on both ends
            delta = step_factor * (a[src] + a[dst]) * 0.5
            if delta < eps:
                # nothing sensible to move – give up this iteration
                it += 1
                continue

        # --------------------------------------------------------------
        # Perform the update (keeping a copy for possible rejection).
        # --------------------------------------------------------------
        a_old = a.copy()
        conv_new = update_convolution(conv, a_old, src, dst, delta)

        # apply the mass transfer
        a[src] -= delta
        a[dst] += delta

        # --------------------------------------------------------------
        # Evaluate the new objective (exact – sum(a) stays 1).
        # --------------------------------------------------------------
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # Acceptance decision (greedy + occasional uphill moves).
        # --------------------------------------------------------------
        accept = False
        if new_score < best_score - 1e-12:
            accept = True                     # strict improvement
        else:
            # simulated‑annealing style uphill acceptance
            prob = np.exp((best_score - new_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            conv = conv_new
            if new_score < best_score - 1e-12:
                # record a new best
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
                print(f"[improve] it={it:05d}, score={best_score:.6f}")
            else:
                # uphill step – keep the move but do not update “best”
                pass
        else:
            # reject → restore the old state
            a = a_old
            # conv stays unchanged

        # --------------------------------------------------------------
        # Scheduling updates.
        # --------------------------------------------------------------
        it += 1
        temperature = max(min_temp, temperature * cooling)
        step_factor = max(0.01, step_factor * 0.9999)

        # periodic full recomputation to avoid accumulated numeric drift
        if it % 2000 == 0:
            conv = full_convolution(a)

        # restart if we have been idle for too long
        if new_score >= best_score - 1e-12:
            no_improve += 1
        else:
            no_improve = 0
        if no_improve > 5000:
            # small random jitter + full recompute
            jitter = np.random.normal(scale=0.001, size=n)
            jitter = np.where(a + jitter < 0, -a, jitter)   # keep non‑negative
            a = a + jitter
            a = np.maximum(a, 0)
            a /= a.sum()                                   # re‑normalise
            conv = full_convolution(a)
            best_a = a.copy()
            best_score = 2.0 * n * conv.max()
            step_factor = 0.20
            temperature = 0.02
            no_improve = 0
            print(f"[restart] it={it}, new score={best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point required by the competition harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total wall‑clock seconds available for the search.
    **kwargs
        Optional keyword ``length`` to force a particular sequence length.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a set of candidate lengths.
    # ------------------------------------------------------------------
    try:
        # Prefer the length of a previously found sequence if present.
        preferred_len = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        preferred_len = None

    # Base list of lengths (short enough for many iterations, long enough
    # to be close to the asymptotic optimum).  Feel free to extend.
    candidate_lengths = [2000, 4000, 6027]
    if preferred_len is not None:
        candidate_lengths = [preferred_len] + candidate_lengths

    # If the caller forces a length, honour it.
    forced = kwargs.get("length")
    if forced is not None:
        candidate_lengths = [int(forced)]

    # Divide the total budget evenly across the lengths (a bit of slack at the end).
    per_len_budget = max(0.1, budget_s / max(1, len(candidate_lengths)))

    best_overall_seq = None
    best_overall_score = float("inf")
    best_overall_n = None

    for idx, n in enumerate(candidate_lengths):
        # Remaining time check – leave a small safety margin.
        time_left = start_time + budget_s - time.time()
        if time_left < 5.0:
            break

        # ------------------------------------------------------------------
        # Initialise from the arcsine distribution.
        # ------------------------------------------------------------------
        a = build_arcsine(n)
        conv = full_convolution(a)
        init_score = 2.0 * n * conv.max()
        print(f"[init] n={n}, initial score={init_score:.6f}")

        # ------------------------------------------------------------------
        # Run the local search for this length.
        # ------------------------------------------------------------------
        deadline = start_time + (idx + 1) * per_len_budget - 0.5
        if deadline - time.time() < 0.1:
            deadline = time.time() + time_left - 0.2
        best_a, best_score = hill_climb(a, conv, deadline)

        print(f"[finished] n={n}, best score={best_score:.6f}")

        if best_score < best_overall_score:
            best_overall_score = best_score
            best_overall_seq = best_a.copy()
            best_overall_n = n

        # Stop early if we are out of budget.
        if time.time() - start_time > budget_s - 0.3:
            break

    # Fallback – if nothing succeeded (unlikely), return the default construction.
    if best_overall_seq is None:
        best_overall_seq = build_arcsine(6027)

    print(f"[final] best n={best_overall_n}, best score={best_overall_score:.6f}")
    return best_overall_seq.tolist()
