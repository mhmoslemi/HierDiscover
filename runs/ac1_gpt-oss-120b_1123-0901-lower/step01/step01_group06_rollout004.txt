#!/usr/bin/env python
"""
propose_candidate
=================

Search for a non‑negative step function (a probability vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The search is based on a simple two‑parameter family

    a_i ∝ (i + δ)^{-p} ,   i = 0,…,n-1

where p∈[0.5,0.9] controls the decay and δ≥1 moves the mass away from the
first index.  For each (p,δ) the vector is normalised to sum to 1 and its
score is obtained with the provided ``evaluate_sequence`` function.
A coarse grid is scanned, the best candidate is kept, and a quick random
pairwise mass‑redistribution hill‑climber refines it while time permits.

The method is O(N·G) for a grid of size G (≈3 k for the defaults) and
fits easily into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: power‑law coefficients (i+δ)^{-p}
# ----------------------------------------------------------------------
def _powlaw_coeffs(n: int, p: float, delta: float) -> np.ndarray:
    """Return the first n terms of (i+δ)^{-p} as a NumPy array."""
    # Using float64 throughout – values stay safely bounded for the
    # parameter ranges used below.
    idx = np.arange(n, dtype=float)
    return (idx + delta) ** (-p)


# ----------------------------------------------------------------------
# Optional baseline: arcsine coefficients (1‑z)^(-1/2)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^(-1/2).  a_0 = 1, a_{i+1}=a_i·(2i+1)/(2(i+1))."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0, seed: int = 42, **kwargs):
    """
    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Time budget (seconds).  The algorithm stops before the limit
        and returns the best sequence found.
    seed : int, optional
        Random seed for the stochastic refinement stage.
    **kwargs
        Unused, accepted for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum == 1.0) that minimises the evaluation function.
    """

    start = time.time()

    # Choose a length.  Re‑use a previous construction length if
    # available; otherwise fall back to a comfortable default.

    try:
        n = len(height_sequence_1)  # pylint: disable=undefined-variable
    except Exception:
        n = 2000

    best_seq = None
    best_score = float("inf")

    # 1. Baseline: arcsine coefficients (the previous state‑of‑the‑art).
    # --------------------------------------------------------------
    try:
        coeffs = _arcsine_coeffs(n)
        seq = (coeffs / coeffs.sum()).tolist()
        sc = evaluate_sequence(seq)            # pylint: disable=undefined-variable
        best_seq = seq
        best_score = sc
        print(f"[baseline] arcsine score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2. Grid search over the two‑parameter power‑law family.
    #    p in [0.55, 0.90]  – controls decay.
    #    δ in [1.0, 5.0]   – shifts the first index.
    # --------------------------------------------------------------

    # evaluate for n≈2000.

        if s <= 0.0:
            continue
        new_seq = [x / s for x in new_seq]

        sc = evaluate_sequence(new_seq)  # pylint: disable=undefined-variable
        if sc < best_score:
            best_score = sc
            best_seq = new_seq
            if it % 500 == 0:   # occasional progress report
                print(f"[refine] iter {it:5d} → new best = {sc:.6f}")

    # --------------------------------------------------------------
    # Final report
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)  # pylint: disable=undefined-variable
        print(f"[propose_candidate] length={n}, final_score={final_score:.6f}")
    except Exception:
        pass

    return best_seq
