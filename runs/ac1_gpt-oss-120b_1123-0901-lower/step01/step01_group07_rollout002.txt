def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, math, random
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # base length
    n_base = 4000
    best_score = float('inf')
    best_params = None
    best_seq = None

    # Parameter ranges
    p_vals = np.linspace(0.55, 0.85, 16) # step 0.02
    gamma_vals = np.logspace(-1, 2, 30) # 0.1 to 100
    # grid search
    for p in p_vals:
        for gamma in gamma_vals:
            if time.time() > deadline: break
            seq = _make_sequence(n_base, p, gamma)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_params = (p,gamma)
                best_seq = seq.copy()
                print(f"[grid] p={p:.4f}, gamma={gamma:.4f}, score={score:.6f}")

    # refine with Nelder-Mead or random perturb
    if best_params is None:
        best_params = (0.5, 0.3)
        best_seq = _make_sequence(n_base, *best_params)
        best_score = evaluate_sequence(best_seq.tolist())
    # local optimization using simple random walk with annealing
    p_cur, gamma_cur = best_params
    # step sizes
    step_p = 0.02
    step_gamma = 0.5
    iter = 0
    while time.time() < deadline:
        iter+=1
        # generate proposal around current
        p_new = p_cur + np.random.randn()*step_p
        gamma_new = gamma_cur * np.exp(np.random.randn()*0.2) # log perturbation
        # enforce bounds
        if not (0.3 < p_new < 0.95) or not (0.01 < gamma_new < 1000):
            continue
        seq = _make_sequence(n_base, p_new, gamma_new)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            p_cur, gamma_cur = p_new, gamma_new
            print(f"[refine] iter={iter}, p={p_new:.5f}, gamma={gamma_new:.5f}, score={score:.6f}")
            # optionally shrink step sizes after improvement
            step_p = max(step_p*0.9, 0.001)
            step_gamma = max(step_gamma*0.9, 0.01)
        else:
            # occasional acceptance? not needed.
            # gradually shrink step sizes
            if iter % 1000 == 0:
                step_p = max(step_p*0.95, 0.001)
                step_gamma = max(step_gamma*0.95, 0.01)

    # After optimization, test larger n options to see improvement
    for n_candidate in [5000, 8000, 12000]:
        if time.time() > deadline: break
        seq = _make_sequence(n_candidate, p_cur, gamma_cur)
        score = evaluate_sequence(seq.tolist())
        print(f"[n_test] n={n_candidate}, score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[update] new best with n={n_candidate}")

    # final output ensure list of python floats
    result = best_seq.tolist()
    print(f"Finished – best score {best_score:.6f} with p={p_cur:.6f}, gamma={gamma_cur:.6f}, n={len(best_seq)}")
    return result

The target function is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

    a_i = (i + γ) ** (‑p)
the continuous‑limit bound is
    2·(1‑p)²·B(1‑p,1‑p)

3.  **Length tuning** – after a good shape has been found we try a few

def _make_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return an array a_i = (i + gamma) ** (-p), i = 0 … n‑1."""
    # Using float64 throughout; negative or NaN values are filtered later
    return (np.arange(n, dtype=float) + gamma) ** (-p)
