"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm starts from the best known sequence (or from the central‑binomial
coefficients) and performs a projected gradient descent on a smooth
approximation of the max‑convolution.  The smooth objective is the
log‑sum‑exp of the self‑convolution, whose gradient can be written in closed
form and computed efficiently with two FFTs.  At each iteration we

  * evaluate the convolution a*a,
  * form a soft‑max weight vector w (the smoother becomes sharper over time),
  * compute the gradient  ∇ F(a) = 2·(a_rev * w)[n‑1 : 2n‑1]  by FFT,
  * take a step a←a−η ∇ F(a) and project exactly onto the probability simplex,
  * keep the best vector seen.

Learning‑rate η decays as 1/√t, the smoothing parameter α ramps from a small
value to a large one, and occasional tiny Gaussian kicks help escape plateaus.
The whole routine runs until the given time budget (≈ 1000 s) is exhausted
and returns the best sequence found.
"""

import numpy as np
import time
import math
import random

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # Find the rho index (the last positive entry of u - cssv/ind)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is non‑positive; return the uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max‑convolution objective:
        g_j = 2 * Σ_i a_i * w_{i+j}
    Computed by FFT as a convolution of a_rev and w.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((3 * n - 2).bit_length())  # enough for len(a_rev)+len(w)-1
    fa_rev = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv_rev = np.fft.irfft(fa_rev * fw, size)[: 3 * n - 2]
    grad = 2.0 * conv_rev[n - 1 : 2 * n - 1]   # slice of length n
    return grad

# Main optimisation routine
# ----------------------------------------------------------------------
