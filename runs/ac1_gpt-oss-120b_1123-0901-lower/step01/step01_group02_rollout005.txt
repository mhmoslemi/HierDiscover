import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑value step‑function using a smoothed‑max gradient descent
    with momentum and occasional random perturbations.
    The objective is   E(a) = 2 * n * max_k (a * a)[k] / (sum a)^2 .
    By fixing sum(a) = sqrt(2 n) the objective equals max_k (a * a)[k].
    A soft‑max approximation of the maximum provides a differentiable loss,
    whose gradient can be computed by a single convolution.
    The algorithm performs projected gradient descent on this loss,
    rescales to keep the sum fixed and clips to the bounds required by
    ``evaluate_sequence``.  The current best sequence is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # Load a starting sequence if the competition provides one.
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.shape[0]
        print("[info] Using provided height_sequence_1 (n={})".format(n))
    except Exception:
        n = 2100
        base = np.ones(n, dtype=float)
        print("[info] No provided sequence – using uniform vector (n={})".format(n))

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Helper: normalise a vector to the required sum and clip to feasible range.
    def normalise_and_clip(vec):
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s > 0.0:
            vec = vec * (target_sum / s)
        # clip to the limits used by evaluate_sequence
        np.clip(vec, 0.0, 1000.0, out=vec)
        return vec

    # initialise from the best known construction
    a = normalise_and_clip(base.copy())
    # best-so-far
    best_seq = a.copy()
    # because sum is exactly target_sum, the score equals the max convolution entry
    conv = np.convolve(a, a)
    best_score = float(conv.max())
    print("[init] initial score = {:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # Hyper‑parameters
    lr0 = 0.3                # base learning rate
    decay = 1e-5             # learning‑rate decay
    beta = 0.9               # momentum factor
    vel = np.zeros_like(a)   # momentum buffer
    # temperature for the soft‑max; we slowly raise it to focus on the true max
    alpha = 0.5
    alpha_max = 20.0
    alpha_inc = 1e-5        # incremental increase per iteration

    iteration = 0
    last_improve = 0
    # after this many iterations without improvement we do a random restart
    patience = 200_000

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # current convolution and objective
        conv = np.convolve(a, a)
        cur_score = float(conv.max())

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = iteration
            print("[{:07d}] new best = {:.6f}".format(iteration, best_score))

        # ------------------------------------------------------------------
        # compute soft‑max weights on the convolution entries
        conv_max = conv.max()
        # stabilise exponential by subtracting max
        exp_arg = alpha * (conv - conv_max)
        # avoid overflow/underflow – numpy handles it gracefully
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to a hard max weight
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ------------------------------------------------------------------
        # gradient of the smoothed objective: g = 2 * (w * a)_{valid}
        # where * denotes convolution.  The “valid” mode yields exactly n elements.
        grad = 2.0 * np.convolve(w, a, mode='valid')

        # momentum update
        vel = beta * vel + (1.0 - beta) * grad

        # adaptive learning rate
        lr = lr0 / (1.0 + decay * iteration)

        # projected gradient step
        a_new = a - lr * vel
        a_new = normalise_and_clip(a_new)

        # occasionally inject a very small random jitter to shake the system
        if iteration % 5000 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            a_new = normalise_and_clip(a_new + jitter)

        # evaluate the candidate (we can reuse the convolution we already need)
        conv_new = np.convolve(a_new, a_new)
        new_score = float(conv_new.max())

        # accept if the move is not worse, otherwise keep the old vector;
        # a tiny probability of accepting a worse move helps escape shallow plateaus.
        if new_score <= cur_score or np.random.rand() < 0.001:
            a = a_new
        # otherwise stay with the previous a

        # ------------------------------------------------------------------
        # increase the temperature slowly – this sharpens the soft‑max.
        alpha = min(alpha_max, alpha + alpha_inc)

        # check for stagnation → random restart
        if iteration - last_improve > patience:
            print("[{:07d}] stagnation detected – random restart".format(iteration))
            a = normalise_and_clip(np.random.rand(n))
            vel = np.zeros_like(a)
            last_improve = iteration

"""
A smoothed‑max gradient descent optimiser for the step‑function problem.

The evaluation function is
    E(a) = 2 * n * max_k (a * a)[k] / (sum a)^2 .
If we keep the total sum fixed to sqrt(2 n) then E(a) = max_k (a * a)[k].
The algorithm therefore minimises the maximum entry of the full convolution
of the vector with itself.

The non‑smooth “max” is replaced by a differentiable soft‑max:
    L(a) = (1/α) log Σ_k exp(α·(a*a)[k]) .
Its gradient can be written in closed form as
    ∂L/∂a = 2 * ( w  *  a )_valid
where w_k = exp(α·c_k) / Σ_j exp(α·c_j) are soft‑max weights for the
convolution entries c = a*a and *_valid is a valid‑mode convolution.

We apply projected gradient descent with momentum, adapt the learning rate,
gradually increase the temperature α (so the approximation sharpens),
and add occasional random jitter / restarts to escape plateaus.
All vectors are kept non‑negative, clipped to the limits required by
`evaluate_sequence`, and re‑scaled after each step to the fixed sum.

The best sequence seen during the run is returned.
"""

import math
import numpy as np

    target_sum = math.sqrt(2.0 * n)

    def normalise_and_clip(vec: np.ndarray) -> np.ndarray:

        s = vec.sum()

        # final clipping to the bounds required by evaluate_sequence

    alpha_max = 20.0    # final temperature

    while time.time() < deadline:

        conv = np.convolve(a, a)
        cur_score = float(conv.max())

        # update global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = iteration
