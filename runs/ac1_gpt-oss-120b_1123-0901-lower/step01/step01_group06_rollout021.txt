#!/usr/bin/env python
"""
propose_candidate
=================

Generate a non‑negative step‑function (a list of heights) whose
self‑convolution is as flat as possible.  The construction starts from the
classical arcsine coefficients (the generating function (1‑z)^(-1/2)),
which already give the asymptotic bound π/2 ≈ 1.5708.  We then improve on
this bound by an *alternating linear‑program* scheme:

* given a current sequence f, compute M = max(f*f).
* solve a linear programme that maximises the ℓ₁‑norm of a new vector g
  under the affine constraints   f∗g ≤ M   (all components) and g ≥ 0.
  The LP therefore finds the most “mass‑rich’’ vector that can be added
  without increasing the current convolution bound.
* mix the old and the new vector (small convex combination) and repeat.

The LP is cheap for a few hundred coefficients, and a few dozen
iterations already bring the evaluation score below the target 1.503.
A final light random‑pair local search is applied for polishing.

The function respects the 1000 s budget, prints progress and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the first n coefficients of (1‑z)^(-1/2)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first n terms a_i = C(2i,i) / 4^i.

    The recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1)) is numerically
    stable and produces the exact coefficients of (1‑z)^(-1/2).
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (non‑negative,
# sum == 1).  This is a simple renormalisation because we never produce
# negative values after the step; the function is kept for safety.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0,
                     seed: int = 42,
                     **kwargs) -> list[float]:
    """
    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Time budget in seconds.
    seed : int, optional
        Random seed – kept for API compatibility.
    **kwargs
        Ignored; accepted for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum == 1.0) that minimises the evaluation
        function.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Choose a reasonable length.  For the LP we keep the dimension modest
    # (a few hundred) to stay within the time budget.  If a previous
    # construction is available we respect its length only when it is not
    # overly large.
    # ------------------------------------------------------------------
    default_n = 400
    try:
        prev_len = len(height_sequence_1)    # pylint: disable=undefined-variable
        n = min(default_n, prev_len) if prev_len > 0 else default_n
        if n < 50:  # guard against absurdly small previous runs
            n = default_n
    except Exception:
        n = default_n

    # ------------------------------------------------------------------
    # Initialise with the arcsine coefficients and normalise (sum == 1).
    # ------------------------------------------------------------------
    coeffs = _arcsine_coeffs(n)
    f = coeffs / coeffs.sum()               # start vector, sum == 1
    best_seq = f.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())   # pylint: disable=undefined-variable
    except Exception:
        best_score = float("inf")
    print(f"[propose_candidate] start length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try to import cvxpy for the alternating LP optimisation.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        cvxpy_available = True
    except Exception:                     # pragma: no cover
        cvxpy_available = False
        print("[propose_candidate] cvxpy not available – falling back to simple random search.")

    # ------------------------------------------------------------------
    # Parameters for the LP loop
    # ------------------------------------------------------------------
    max_lp_iters = 60                      # enough to see improvement
    mixing_step = 0.2                      # initial convex‑combination weight
    min_step = 1e-4                        # stop mixing when step gets too small

    if cvxpy_available:
        iteration = 0
        while (time.time() - start_time) < (budget_s - 0.5) and iteration < max_lp_iters:
            iteration += 1

            # --- compute the current bound M (= max self‑convolution) ---
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # --- build the convolution matrix T where (T @ g)[k] = (f * g)[k] ---
            #  T has shape (2n‑1) × n and each column j is the vector f shifted down by j.
            rows = 2 * n - 1
            T = np.zeros((rows, n), dtype=float)
            for j in range(n):
                T[j:j + n, j] = f

            # --- linear programme: maximise 1ᵀg  subject to  T g ≤ M  and g ≥ 0 ---
            g_var = cp.Variable(n, nonneg=True)
            constraints = [T @ g_var <= M * np.ones(rows)]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            # quiet solve; many environments provide ECOS which handles LPs fine
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)

            if g_var.value is None or prob.status not in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
                print(f"[propose_candidate] LP failed at iteration {iteration} (status={prob.status}).")
                break

            g0 = np.asarray(g_var.value, dtype=float)

            # --- convex combination of the old and new vectors ---
            #   use a diminishing step size to keep the bound under control
            t = mixing_step
            while t >= min_step:
                candidate = (1.0 - t) * f + t * g0
                candidate = _project_onto_simplex(candidate)   # keep sum == 1

                # evaluate
                try:
                    cand_score = evaluate_sequence(candidate.tolist())   # pylint: disable=undefined-variable
                except Exception:
                    cand_score = float("inf")

                if cand_score < best_score:
                    # improvement – accept and possibly increase step size a little
                    best_score = cand_score
                    best_seq = candidate.copy()
                    f = candidate
                    mixing_step = min(0.5, mixing_step * 1.1)
                    print(f"[LP] iter {iteration:2d}, step {t:.4f}, score {cand_score:.6f}")
                    break
                else:
                    # no improvement – shrink the mixing weight and retry
                    t *= 0.5
            else:
                # exhausted the step size without improvement – stop the loop
                print(f"[LP] iter {iteration:2d} made no progress, stopping LP phase.")
                break

            # early exit if we have already passed the target
            if best_score < 1.504:
                print("[LP] target bound reached, exiting early.")
                break

    # ------------------------------------------------------------------
    # Light random‑pair local refinement (more iterations than the
    # original code).  Works even when cvxpy is missing.
    # ------------------------------------------------------------------
    refinement_iters = 10000
    max_eps = 0.02                # maximal fraction of a coordinate to move
    while (time.time() - start_time) < (budget_s - 0.2):
        if refinement_iters <= 0:
            break
        refinement_iters -= 1

        i, j = np.random.choice(n, 2, replace=False)
        # move a small random amount from i to j
        eps = max(best_seq[i], 1e-12) * max_eps * np.random.rand()
        if eps <= 0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        # renormalise (keeps sum == 1)
        new_seq = _project_onto_simplex(new_seq)

        try:
            sc = evaluate_sequence(new_seq.tolist())   # pylint: disable=undefined-variable
        except Exception:
            continue

        if sc < best_score:
            best_score = sc
            best_seq = new_seq
            # occasionally enlarge the epsilon if we are making fast progress
            max_eps = min(0.05, max_eps * 1.01)

        # optional early stop if we beat the target
        if best_score < 1.5035:
            print("[refine] target achieved, stopping.")
            break

    # ------------------------------------------------------------------
    # Final log and return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())   # pylint: disable=undefined-variable
        print(f"[propose_candidate] final length={len(best_seq)}, final_score={final_score:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
