            sum_b = b.sum()
            if sum_b <= 0:

            if score_b < best_score:
                best_score = score_b
                best_a = b_norm.copy()
                print(f"LP improvement: score {best_score:.6f}")
            # update a to b_norm for next iteration
            a = b_norm
        else:
            print("LP failed:", res.message)
            break

    # Return best sequence as Python list
    print(f"Finished. Best metric = {best_score:.6f}")
    return best_a.tolist()

            s2 = float(2.0 * n * _full_convolution(a2).max())
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
                a = a2
                print(f"Iter {iteration}: exact‑subgrad improvement {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Optional linear‑program refinement (only if SciPy is present).
    # --------------------------------------------------------------
    if _SCIPY_AVAILABLE:
        while time.time() < deadline:
            conv = _full_convolution(a)
            M = conv.max()
            A = _build_convolution_matrix(a)
            K = A.shape[0]

            # maximise Σ b_j  →  minimise -Σ b_j
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(K, M, dtype=np.float64)

            # give the solver a tiny per‑call time budget
            remaining = max(0.0, deadline - time.time())
            opts = {"time_limit": remaining}

            try:
                res = linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=[(0, None)] * n,
                              method="highs",
                              options=opts)
            except Exception as e:  # pragma: no cover
                print("LP solver raised an exception:", e)
                break

            if not res.success:
                print("LP did not succeed:", res.message)
                break

            b = res.x
            s_b = b.sum()
            if s_b <= 0.0:
                break
            b_norm = b / s_b
            lp_score = _metric(b_norm)
            if lp_score < best_score:
                best_score = lp_score
                best_a = b_norm.copy()
                a = b_norm
                print(f"LP refinement improvement: metric {best_score:.6f}")
            else:
                # No further gain – exit the LP phase.
                break

    # --------------------------------------------------------------
    # 5. Return the best sequence found.
    # --------------------------------------------------------------
    print(f"Finished. Best metric = {best_score:.6f}")
    return best_a.tolist()
