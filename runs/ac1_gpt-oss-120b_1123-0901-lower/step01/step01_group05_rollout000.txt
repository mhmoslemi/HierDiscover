def generate_arcsine_sequence(n, p=1.0, eps=1e-6):
    i = np.arange(n, dtype=float)
    denominator = np.sqrt((i + eps) * (n-1 - i + eps))
    seq = denominator ** (-p)
    # Equivalent to denominator^-p = 1/(denominator^p)
    # Actually denominator = sqrt(...), then seq = denominator**(-p) = ((i+eps)*(n-1-i+eps))^{-p/2}
    # That's what we need.

    # Avoid zeros (shouldn't) but to safe.
    seq = np.maximum(seq, 0.0)
    # Scale so that max <= 1000
    max_val = np.max(seq)
    if max_val > 1000:
        seq = seq * (1000.0 / max_val)
    return seq.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)

    try: n = len(height_sequence_1)
    except NameError: n = 3000

    # base ranges
    p_vals = np.linspace(0.6, 1.4, 17) # from 0.6 to 1.4 step 0.05
    eps_vals = [1e-4, 1e-5, 1e-6, 1e-7]

    best_seq = None
    best_score = float('inf')

    deadline = time.time() + budget_s - 0.5

    def eval_seq(seq):
        return evaluate_sequence(seq)

    # Evaluate grid
    for eps in eps_vals:
        for p in p_vals:
            seq = generate_arcsine_sequence(n, p, eps)
            # rescale to avoid >1000
            maxv = max(seq)
            if maxv > 1000:
                factor = 1000.0 / maxv
                seq = [x * factor for x in seq]
            score = eval_seq(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[grid] eps={eps:.1e} p={p:.3f} score={score:.6f}")

            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # Follow-up random local search: mutate few positions multiplicative factor, keep if improvement.
    while time.time() < deadline:
        i = rng.integers(n)
        # jitter factor drawn from lognormal distribution
        factor = np.exp(rng.normal(0, 0.01))
        new_seq = best_seq.copy()
        new_val = max(0.0, new_seq[i] * factor)
        new_seq[i] = new_val
        # optionally adjust scale to maintain max bound
        maxv = max(new_seq)
        if maxv > 1000:
            scale = 1000.0 / maxv
            new_seq = [x * scale for x in new_seq]
        score = eval_seq(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq
            print(f"[local] improve score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
