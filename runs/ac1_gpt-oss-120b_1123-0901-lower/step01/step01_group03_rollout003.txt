def project_simplex(v):
    v = np.asarray(v)
    if v.ndim != 1: raise ...
    # shift to ensure positive
    # use sorting descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected gradient descent on a smoothed version of the peak of the
self‑convolution.

The evaluation metric is 2·n·max(conv(a,a)) / (sum(a)²).  Because the
objective is homogeneous we keep the sequence on the probability simplex
(sum = 1, a_i ≥ 0).  The non‑smooth max is replaced by a log‑sum‑exp
approximation with a temperature ε.  The gradient of the smoothed
objective is

    g_i = 2·∑_k w_k·a_{k‑i} ,   w_k ∝ exp(conv_k/ε) ,  ∑_k w_k = 1 .

Both the convolution a*a and the weighted convolution needed for the
gradient are performed with FFTs (O(N log N)).  A simple back‑tracking line‑
search guarantees monotone improvement of the true (unsmoothed) peak.
Occasional random kicks help to escape shallow plateaus.  The best vector
found before the time budget expires is returned as a plain Python list.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using FFT of given size."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(w: np.ndarray, a_rev: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed max objective:
        g_i = 2 * Σ_k w_k * a_{k-i}
    using FFT.  a_rev is a reversed copy of the current a.
    """
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    full = np.fft.irfft(fw * fa_rev, size)[: w.size + a_rev.size - 1]
    n = a_rev.size
    # slice the part that corresponds to i = 0 … n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _initial_binomial(n: int) -> np.ndarray:
    """Central‑binomial coefficients a_i = C(2i,i)/4^i (unnormalised)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a = _project_onto_simplex(a)          # enforce sum = 1
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative coefficient vector that minimises
    `evaluate_sequence`.  The method uses a smoothed max objective,
    projected gradient descent on the probability simplex and occasional
    random perturbations.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (a few seconds are kept as safety margin).

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length and initialise the sequence.
    # ------------------------------------------------------------------
    try:
        # Use the previously discovered construction if it exists.
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        a = _project_onto_simplex(init_seq)
    except NameError:
        # Fallback – central‑binomial coefficients, length ≈ 6000.
        n = 6000
        a = _initial_binomial(n)

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT size (must accommodate convolutions of length up to
    #    3·n‑2 when forming the gradient).
    # ------------------------------------------------------------------
    fft_size = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3. Optimisation loop.
    # ------------------------------------------------------------------
    best_a = a.copy()
    # initial metric (exact, using the definition)
    conv = _full_convolution_fft(a, fft_size)
    best_score = 2.0 * n * conv.max()
    print(f"Initial score (n={n}): {best_score:.6f}")

    # Adam‑style accumulators (optional, can be switched off)
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1, beta2 = 0.9, 0.999
    epsilon_adam = 1e-8

    eta0 = 0.5                     # base step size
    no_improve_counter = 0

    iteration = 0
    while time.time() < deadline:
        # -------- compute (a*a) and the current peak ------------------
        conv = _full_convolution_fft(a, fft_size)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # store the best solution seen so far
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # -------- temperature for smoothing (10 % of the current peak) --
        eps = max(max_val * 0.10, 1e-12)

        # ---------- compute softmax weights over the convolution ----------
        shift = conv - max_val                 # ≤ 0
        w = np.exp(shift / eps)
        w_sum = w.sum()
        if w_sum == 0.0:                      # should not happen, but guard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the smoothed objective --------------------
        a_rev = a[::-1]
        grad = _gradient_from_weights(w, a_rev, fft_size)

        # ---------- Adam‑style moment updates ----------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # ---------- adaptive step size (back‑tracking) --------------------
        eta = eta0 / math.sqrt(iteration + 1.0)
        improved = False
        for _ in range(5):
            # Adam update direction (scaled like a simple gradient step)
            step = eta * (m_hat / (np.sqrt(v_hat) + epsilon_adam))
            a_candidate = a - step
            a_candidate = _project_onto_simplex(a_candidate)

            # evaluate the *true* peak after the trial step
            conv_cand = _full_convolution_fft(a_candidate, fft_size)
            max_cand = conv_cand.max()
            if max_cand <= max_val - 1e-12:   # genuine improvement
                a = a_candidate
                improved = True
                no_improve_counter = 0
                break
            # if not better, shrink step size and retry
            eta *= 0.5

        if not improved:
            no_improve_counter += 1
            # occasional random kick when we are stuck
            if no_improve_counter % 2000 == 0:
                noise = np.random.randn(n) * 5e-4
                a = _project_onto_simplex(a + noise)

        # occasional broader perturbation to explore new basins
        if iteration % 5000 == 0 and iteration > 0:
            a = _project_onto_simplex(a + np.random.randn(n) * 0.001)

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations. Best score = {best_score:.6f}")
    return best_a.tolist()
