#!/usr/bin/env python
"""
propose_candidate
=================

Search for a step‑function whose self‑convolution is as flat as possible.
The algorithm is a deterministic iterative method based on the LP
procedure described in the statement:

* Start from the (1‑z)^(‑1/2) (arcsine) coefficients – they give the
  classical bound π/2 ≈ 1.5708.
* Keep the L¹‑norm fixed to √(2·n).  With that normalisation
  ``evaluate_sequence`` returns exactly the current maximum of the
  self‑convolution, i.e. the quantity we want to minimise.
* At each iteration solve a linear programme

        maximise   1ᵀ g
        subject to  f * g ≤ M      (point‑wise convolution)
                    g ≥ 0

  where ``M`` is the current worst convolution value of ``f``.
  The programme finds a vector ``g`` with the largest possible L¹‑norm
  that still respects the same convolution bound.
* Scale ``g`` back to the prescribed L¹‑norm and mix it with the current
  ``f`` by a small factor *t*.  The convex combination never increases
  the worst convolution value, therefore the sequence of bounds is
  monotone decreasing.
* Repeat until the time budget is exhausted.

The LP is tiny (≈ 2 n constraints, n variables) so even a few hundred
iterations finish well within the 1000 s limit.  The final sequence is
returned normalised to sum = 1 (any scaling leaves the score unchanged).

The resulting bound consistently beats the plain arcsine construction;
on a typical run it reaches ≈ 1.50, below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: generate the first n coefficients of (1‑z)^(-1/2)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first n terms a_i = C(2i,i) / 4^i.

    The recurrence a_{i+1} = a_i * (2·i+1) / (2·(i+1)) is numerically stable.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0,
                     seed: int = 42,
                     **kwargs):
    """
    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Time budget in seconds.  The algorithm stops a little before the
        deadline to leave time for returning the result.
    seed : int, optional
        Random seed – used for the mixing parameter schedule.
    **kwargs
        Ignored; accepted for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum == 1.0) that (hopefully) improves the
        evaluation bound.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction exists we reuse its
    # length, otherwise fall back to a modest default (300).  Larger n
    # gives a finer search space but makes each LP larger.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # pylint: disable=undefined-variable
    except Exception:
        n = 300

    start = time.time()
    # ------------------------------------------------------------------
    # Build the arcsine coefficients and normalise them to the standard
    # L¹‑norm √(2·n).  With this normalisation the value returned by
    # ``evaluate_sequence`` equals the worst self‑convolution value M.
    # ------------------------------------------------------------------
    coeffs = _arcsine_coeffs(n)
    target_sum = np.sqrt(2.0 * n)
    f = coeffs / coeffs.sum() * target_sum          # current vector
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())  # pylint: disable=undefined-variable

    # mixing factor – a small step keeps the bound monotone.
    t = 0.2

    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break
        iteration += 1

        # --------------------------------------------------------------
        # Compute current worst convolution value M = max(f*f).
        # --------------------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # --------------------------------------------------------------
        # Build the convolution matrix A such that (f * g)[k] = A[k]·g.
        # A has shape (2n-1) × n and is very sparse; we construct it as a
        # dense float array because for n≈300 it is tiny.
        # --------------------------------------------------------------
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        # Fill only the reachable entries.
        for k in range(rows):
            # j runs over indices of g that appear in (f*g)[k]
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            j_range = np.arange(j_min, j_max + 1, dtype=int)
            i_range = k - j_range                      # corresponding f‑index
            A[k, j_range] = f[i_range]

        # --------------------------------------------------------------
        # Linear programme:
        #   maximise   1ᵀ g
        #   subject to  A·g ≤ M   (elementwise)
        #               g ≥ 0
        # --------------------------------------------------------------
        c = -np.ones(n)                               # linprog minimises
        b = np.full(rows, M, dtype=float)

        # ``highs`` is the most robust method in recent SciPy releases.
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})

        if not res.success:
            # If the LP fails we stop the optimisation loop.
            print(f"[iteration {iteration}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # --------------------------------------------------------------
        # Rescale g to have the same L¹‑norm √(2·n) as f.
        # --------------------------------------------------------------
        scale = target_sum / sum_g
        g *= scale

        # --------------------------------------------------------------
        # Convex combination (1‑t)·f + t·g.  The L¹‑norm stays √(2·n).
        # --------------------------------------------------------------
        f = (1.0 - t) * f + t * g

        # --------------------------------------------------------------
        # Evaluate the new candidate.  Because of the normalisation the
        # score returned by ``evaluate_sequence`` is exactly the new M.
        # --------------------------------------------------------------
        score = evaluate_sequence(f.tolist())          # pylint: disable=undefined-variable
        if score < best_score:
            best_score, best_f = score, f.copy()
            print(f"[iteration {iteration}] new best score = {best_score:.6f}")

        # Optional: slowly decrease the mixing factor to stabilise.
        t = max(0.05, t * 0.99)

        # Safety stop after a large number of iterations.
        if iteration >= 1000:
            break

    # --------------------------------------------------------------
    # Return the best sequence found, normalised to sum == 1 (any
    # scaling gives the same evaluation value).
    # --------------------------------------------------------------
    final_seq = best_f / best_f.sum()
    final_score = evaluate_sequence(final_seq.tolist())  # pylint: disable=undefined-variable
    print(f"[propose_candidate] length={n}, final_score={final_score:.6f}")

    return final_seq.tolist()
