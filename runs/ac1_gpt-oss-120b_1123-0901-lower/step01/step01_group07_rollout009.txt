def fast_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    # compute convolution using FFT
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    # conv length size; we only need first (2n-1) values
    conv = conv[:2*n-1]
    max_conv = np.max(conv)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_seq * sum_seq)

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three phases:

1. **Coarse grid search** over several sequence lengths (n) and the
   two‑parameter power‑law family a_i = (i + γ)^‑p.
   A fast FFT‑based convolution (`fast_score`) is used to evaluate millions
   of candidates quickly.

2. **Random refinement** of the best (n, p, γ) triple.  The parameters are
   perturbed with a Gaussian walk while keeping them in sensible ranges.

3. **Hill‑climbing on the actual coefficients**.  Starting from the best
   power‑law shape we repeatedly move a small amount of mass from one index
   to another (preserving the total sum).  After each move the fast score is
   recomputed; improvements are kept.  The step size is gradually reduced.

All operations respect the non‑negativity requirement, use only
numpy/scipy (no I/O) and honour the 1000 s time budget.  The returned list
contains the heights of the best sequence found.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog  # not used directly but kept for possible extensions

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective using FFT (much faster than
# numpy.convolve for repeated calls).
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return the value of 2 * n * max(conv(seq, seq)) / (sum(seq) ** 2).

    The convolution is performed with an FFT of power‑of‑two length to
    obtain the linear (non‑circular) convolution.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    if n == 0:
        return float("inf")
    # linear convolution length = 2*n-1, pad to next power of two
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    conv = conv[: 2 * n - 1]                # keep only the linear part
    max_conv = np.max(conv)                # non‑negative by construction
    sum_seq = np.sum(seq)
    if sum_seq < 0.01:                     # mirrors the safeguard in evaluate_sequence
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq * sum_seq))

# Main entry point – the function that will be called by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑score step function under the given time budget.
    Returns a plain Python list of heights (the length is part of the
    optimisation).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – coarse grid over (n, p, γ)
    # ------------------------------------------------------------------
    # a handful of lengths – short enough for many evaluations, long enough
    # to allow the asymptotic behaviour of the power law.
    n_values = [500, 800, 1200, 1700, 2300, 3000, 3400, 4000]
    best_score = float("inf")
    best_seq = None
    best_params = None  # (n, p, gamma)

    for n in n_values:
        # p around the theoretically promising region, but we also allow a
        # modest overshoot in both directions.
        p_vals = np.linspace(0.45, 0.85, 17)           # step 0.025
        gamma_vals = np.linspace(0.05, 5.0, 19)        # step ≈0.26
        for p in p_vals:
            for gamma in gamma_vals:
                if time.time() > deadline:
                    break
                seq = (np.arange(n, dtype=float) + gamma) ** (-p)
                score = fast_score(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (n, p, gamma)
                    print(f"[grid] n={n:4d}, p={p: .4f}, γ={gamma: .4f} → score={score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – random refinement of the triple (n, p, γ)
    # ------------------------------------------------------------------
    if best_params is None:          # very unlikely – safeguard
        best_params = (3428, 0.5, 0.3)
        best_seq = (np.arange(best_params[0]) + best_params[2]) ** (-best_params[1])
        best_score = fast_score(best_seq)

    # number of refinement attempts – chosen to fit comfortably in the
    # remaining time budget.
    refinement_iters = 20000
    for _ in range(refinement_iters):
        if time.time() > deadline:
            break
        n0, p0, g0 = best_params
        # keep n within a reasonable window (200 – 5000)
        n_new = int(np.clip(n0 + rng.integers(-30, 31), 200, 5000))
        p_new = p0 + rng.normal() * 0.015
        g_new = g0 + rng.normal() * 0.20
        if not (0.3 < p_new < 0.9 and 0.0 < g_new < 10.0):
            continue
        seq = (np.arange(n_new, dtype=float) + g_new) ** (-p_new)
        score = fast_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_params = (n_new, p_new, g_new)
            print(f"[refine] n={n_new:4d}, p={p_new: .4f}, γ={g_new: .4f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – direct hill‑climbing on the coefficients
    # ------------------------------------------------------------------
    # Work with a normalized version (sum == 1) – the objective is scale‑invariant.
    seq = best_seq.copy()
    seq /= np.sum(seq)                     # now sum == 1
    n = seq.shape[0]
    cur_score = fast_score(seq)

    # mass‑exchange step size (fraction of the source entry)
    step_frac = 0.10
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a *fraction* of the mass from j to i
        t = rng.random() * step_frac
        delta = t * seq[j]
        if delta == 0.0:
            continue
        new_seq = seq.copy()
        new_seq[i] += delta
        new_seq[j] -= delta
        if new_seq[j] < 0:
            continue
        new_score = fast_score(new_seq)
        if new_score < cur_score:
            seq = new_seq
            cur_score = new_score
            # keep track of the globally best sequence
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[hill] iter={it:6d} → new best score={new_score:.6f}")
        # slowly anneal the step size
        it += 1
        if it % 1000 == 0:
            step_frac = max(step_frac * 0.9, 1e-4)

    # The final sequence is already non‑negative; we return it as a plain list.
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f}, length {len(result)}")
    return result
