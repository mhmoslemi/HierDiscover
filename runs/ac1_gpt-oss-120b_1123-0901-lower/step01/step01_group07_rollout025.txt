def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # ... security checks ...
    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Propose a candidate sequence that beats the previous 1.571 bound.

Approach
--------
We start from a simple power‑law shape (a_i = (i+γ)^‑½), which already gives a
good flat self‑convolution.  Then we improve it by an iterative linear‑program
procedure described in the state‑of‑the‑art paper:

* Given the current step‑function f, compute M = max(f*f).
* Solve a linear program:
      maximise   Σ g_j
      subject to g_j ≥ 0  and  (f ∗ g)_k ≤ M   for every convolution index k.
  The constraints are linear in g because f is fixed.
* The LP tries to increase the ℓ₁‑mass of the function while keeping the
  infinity‑norm of the convolution unchanged.
* If the LP yields a sequence g with a strictly larger sum, we replace f by g
  (or by a small convex combination of f and g) and repeat.
* The objective 2·n·max(conv)/sum² is invariant under scaling, so we can work
  with the raw vectors.  The loop stops when no improvement is possible or the
  time budget (≈ 1000 s) is exhausted.

The LP is small (n≈300) and each solve takes only a few hundred milliseconds,
so dozens of iterations easily fit into the budget.  The final sequence is
returned as a plain Python list of floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# cvxpy is optional but highly desirable for the LP step.
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _make_powerlaw(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a_i = (i + gamma) ** (-p) for i = 0 .. n-1."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def _max_convolution(seq: np.ndarray) -> float:
    """Return max of the linear convolution seq * seq."""
    return float(np.max(np.convolve(seq, seq)))


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve: maximise Σ g_j  subject to g ≥ 0 and (f ∗ g)_k ≤ M for all k.
    Returns the optimal g as a NumPy array, or None if the problem fails.
    """
    if cp is None:               # cvxpy not available – give up on the LP.
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    # Build the linear constraints (f * g)_k ≤ M.
    # Each row of the convolution matrix corresponds to a shifted copy of f.
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        jdx = k - idx                     # indices of g that pair with f[idx]
        # Linear expression: Σ f[idx] * g[jdx]
        expr = cp.sum(cp.multiply(f[idx], g[jdx]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        return None

    if g.value is None:
        return None
    # Numerical safety – enforce non‑negativity and clip tiny negatives.
    return np.maximum(0.0, np.array(g.value, dtype=float))


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` score.  The method uses a power‑law seed and an
    iterative linear‑program refinement (up to the time budget).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise with a classic power‑law (p≈0.5).  A moderate length
    #    (n≈300) keeps the LP cheap while still giving a good score.
    # ------------------------------------------------------------------
    n = 300
    p = 0.50
    gamma = 0.30
    f = _make_powerlaw(n, p, gamma)

    best_seq = f.copy()
    best_score = float('inf')
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, p={p:.4f}, γ={gamma:.4f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Iterative LP refinement.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Compute current maximal convolution value.
        M = _max_convolution(f)

        # Solve LP to increase ℓ₁‑mass without raising the max convolution.
        g = _solve_lp(f, M)
        if g is None:
            print("[LP] Solver failed or not available – aborting refinement.")
            break

        # If the LP did not increase the sum, we are at a fixed point.
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f * (1.0 + 1e-8):
            print(f"[iter {iteration}] No ℓ₁‑mass increase (sum_f={sum_f:.6f}).")
            break

        # Update rule – a modest convex combination keeps the bound steady.
        t = 0.15                      # step size; small enough to stay safe
        f_new = (1.0 - t) * f + t * g

        # Evaluate the new candidate.
        score = evaluate_sequence(f_new.tolist())
        print(f"[iter {iteration}] sum_f={sum_f:.6f}, sum_g={sum_g:.6f}, "
              f"score={score:.6f}")

        # Accept improvement.
        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            print(f"   ** new best (iter {iteration}) score={best_score:.6f}")

        # Prepare for next iteration.
        f = f_new

        # Stop early if we already beat the target by a comfortable margin.
        if best_score < 1.40:
            print(f"[stop] Target comfortably beaten (score={best_score:.6f}).")
            break

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} with length {len(result)}")
    return result
