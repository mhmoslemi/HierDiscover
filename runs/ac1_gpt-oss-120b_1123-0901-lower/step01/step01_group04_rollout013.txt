Algorithm:

```
def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Determine lengths list
    lengths = []
    try:
        lengths.append(len(height_sequence_1))
    except Exception:
        pass
    # Add a few other lengths
    lengths += [500, 1000, 2000, 3000, 5000, 8000, 10000]
    # Unique & sorted, ensure not too large
    lengths = [l for l in sorted(set(lengths)) if l <= 10000]
    best_score = float('inf')
    best_seq = None
    # function to compute fast score
    def fast_score(a):
        n = a.shape[0]
        size = 2*n - 1
        fa = np.fft.fft(a, size)
        conv = np.fft.ifft(fa * fa).real
        max_conv = conv.max()
        sum_a = a.sum()
        if sum_a < 1e-8:
            return np.inf
        return 2.0 * n * max_conv / (sum_a * sum_a)
    # candidate generation loops
    for n in lengths:
        if time.time() > deadline:
            break
        # arcsine via binomial recurrence
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            a[k] = a[k-1] * ((2.0*k - 1.0) * (2.0*k)) / (4.0 * k * k)
        a /= a.sum()
        score = fast_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[init arcsine] n={n}, score={score:.6f}")
        # try a range of exponents p
        for p in np.linspace(0.35, 0.68, 13):
            if time.time() > deadline:
                break
            i = np.arange(1, n+1, dtype=np.float64)
            a2 = i ** (-p) + (n - i + 1) ** (-p)  # (n-i+1) because i starts at 1
            a2 /= a2.sum()
            score2 = fast_score(a2)
            if score2 < best_score:
                best_score = score2
                best_seq = a2.copy()
                print(f"[exponent] n={n}, p={p:.3f}, score={score2:.6f}")
        # hill climbing on this n using best found for it
        # start from best_seq_local (maybe best_seq global)
        # we will run a simple random swap hill climbing for limited steps.
        a_local = best_seq if best_seq.shape[0] == n else a.copy()
        # number of local iterations based on time left
        it = 0
        while time.time() < deadline and it < 2000:
            # propose move
            i, j = np.random.choice(n, size=2, replace=False)
            # amount proportional to min of the two
            delta = 0.01 * min(a_local[i], a_local[j])
            if delta == 0:
                it += 1
                continue
            # move from larger to smaller
            if a_local[i] > a_local[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            a_new = a_local.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            # project: sum unchanged but we ensure nonnegatives (the move ensures nonnegatives)
            new_score = fast_score(a_new)
            # accept if improvement
            if new_score < best_score - 1e-9:
                a_local = a_new
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[improve] n={n}, it={it}, score={new_score:.6f}")

    final_seq = best_seq.tolist()

"""
propose_candidate
=================

Generate a step‑function (non‑negative sequence) that improves on the
classic arcsine construction.  The algorithm works in three phases:

1. **Exploratory grid** –  several sequence lengths are tried and, for
   each length, a family of symmetric “inverse‑power” shapes
   a_i ∝ (i+1)^‑p + (n‑i)^‑p (p≈0.5 gives the arcsine law).  The best
   candidate found by a cheap FFT‑based score is kept.

2. **Local deterministic refinement** –  starting from the current best
   sequence we repeatedly identify the convolution shift(s) where the
   maximum occurs, compute a sub‑gradient that tells which entries
   contribute most to that maximum, and move a small fraction of mass
   from the worst contributing index to the best contributing one.
   The step size decays slowly; the loop stops when time runs out.

3. **Final sanity check** –  the resulting sequence is handed to the
   provided ``evaluate_sequence`` function and returned as a plain
   Python list.

The whole search respects the supplied ``budget_s`` (default 1000 s) and
returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """
    Return the maximum value of the full convolution a*a using FFT.
    a is assumed to be non‑negative and one‑dimensional.
    """
    n = a.shape[0]
    size = 2 * n - 1
    fa = np.fft.fft(a, size)
    conv = np.fft.ifft(fa * fa).real
    return float(conv.max())


def _fast_score(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` (no safety checks).  The score
    is 2·n·max(conv(a,a))/sum(a)^2.
    """
    n = a.shape[0]
    max_conv = _conv_max(a)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_for_max_shifts(a: np.ndarray, max_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the objective (the current maximal convolution value)
    with respect to a.  For each shift s in ``max_shifts`` the contribution
    to the gradient at index i is a[s‑i] (if the index is inside the
    sequence).  The factor 2 from the derivative of a_i a_j is omitted
    because it is common to all components and does not affect the
    direction of steepest descent.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for s in max_shifts:
        # i runs over all indices where both i and s‑i are valid
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[i_range] += a[s - i_range]
    return grad


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list[float] representing the best step‑function found within the
    time budget.  The function makes no assumptions about the existence of
    a global ``height_sequence_1``; if it exists its length is tried first.
    """
    # ------------------------------------------------------------------
    # 1.  Basic configuration & time management
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # lengths we will explore – always try the length of a possible
    # provided solution, then a handful of other sizes.
    lengths = []
    try:
        lengths.append(len(height_sequence_1))   # type: ignore[name-defined]
    except Exception:
        pass
    # a set of auxiliary lengths (must stay modest to keep LP/FFT cheap)
    lengths += [400, 800, 1200, 2000, 3000, 5000, 8000]
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 2.  Global best bookkeeping
    # ------------------------------------------------------------------
    best_seq = None          # numpy array
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 3.  Exploration over lengths and simple parametric families
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        # ---- (a)  classic arcsine obtained from the binomial recursion ----
        a_arc = np.empty(n, dtype=np.float64)
        a_arc[0] = 1.0
        for k in range(1, n):
            a_arc[k] = a_arc[k - 1] * ((2.0 * k - 1.0) * (2.0 * k)) / (4.0 * k * k)
        a_arc /= a_arc.sum()
        sc = _fast_score(a_arc)
        if sc < best_score:
            best_score, best_seq = sc, a_arc.copy()
            print(f"[init] n={n:5d}  arcsine score={sc:.6f}")

        # ---- (b)  symmetric “inverse‑power’’ families a_i ∝ (i+1)^‑p + (n-i)^‑p ----
        #   try a modest grid around the optimal continuous exponent p=0.5
        for p in np.linspace(0.38, 0.62, 13):
            if time.time() > deadline:
                break
            i = np.arange(1, n + 1, dtype=np.float64)                      # 1…n
            a_pow = (i ** (-p)) + ((n - i + 1) ** (-p))
            a_pow /= a_pow.sum()
            sc = _fast_score(a_pow)
            if sc < best_score:
                best_score, best_seq = sc, a_pow.copy()
                print(f"[grid] n={n:5d}  p={p:.3f}  score={sc:.6f}")

        # ---- (c)  a tiny random perturbation of the current best of this length
        #        (useful to escape mild discretisation artefacts)
        if best_seq is not None and best_seq.shape[0] == n:
            a_local = best_seq.copy()
            it = 0
            while time.time() < deadline and it < 2000:
                i, j = np.random.choice(n, size=2, replace=False)
                # transfer a modest fraction from the larger entry to the smaller one
                if a_local[i] > a_local[j]:
                    src, dst = i, j
                else:
                    src, dst = j, i
                delta = 0.01 * a_local[src]           # 1 % of the source mass
                if delta <= 0:
                    it += 1
                    continue
                a_new = a_local.copy()
                a_new[src] -= delta
                a_new[dst] += delta
                sc_new = _fast_score(a_new)
                if sc_new < best_score - 1e-12:
                    a_local = a_new
                    best_seq = a_new.copy()
                    best_score = sc_new
                    print(f"[local] n={n:5d}  it={it:04d}  score={sc_new:.6f}")
                it += 1

    # ------------------------------------------------------------------
    # 4.  Deterministic sub‑gradient refinement on the global best sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No candidate sequence was generated.")
    a = best_seq.copy()
    n = a.shape[0]

    step_factor = 0.02          # initial fraction of the source mass to move
    it = 0
    while time.time() < deadline and it < 100000:
        # compute convolution max and the shift(s) attaining it
        n = a.shape[0]
        size = 2 * n - 1
        fa = np.fft.fft(a, size)
        conv = np.fft.ifft(fa * fa).real
        max_val = conv.max()
        max_shifts = np.where(conv == max_val)[0]

        # sub‑gradient for the active shifts
        grad = _gradient_for_max_shifts(a, max_shifts)

        # source = index that contributes most to the current maximum,
        # destination = index that contributes least.
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            break

        delta = step_factor * a[src]          # move a small fraction of src
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # compute the new objective (fast version)
        sc_new = _fast_score(a_new)
        if sc_new < best_score - 1e-12:
            a = a_new
            best_seq = a_new.copy()
            best_score = sc_new
            print(f"[refine] it={it:05d}  score={sc_new:.6f}")
        # decay the step size gently
        step_factor *= 0.9995
        it += 1

    # ------------------------------------------------------------------
    # 5.  Final evaluation via the official ``evaluate_sequence`` call
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"[final] fast_score={best_score:.6f}  evaluate_sequence={final_score:.6f}")

    return final_list
