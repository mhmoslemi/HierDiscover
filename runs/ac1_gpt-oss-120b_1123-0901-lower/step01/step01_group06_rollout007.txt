"""
propose_candidate
=================

Generate a step‑function whose self‑convolution is extremely flat by
concatenating several independent copies of the optimal *arcsine* (1‑z)^(-1/2)
coefficients.  Each copy has a self‑convolution bounded by 1, so the global
maximum stays ≈1 while the total ℓ¹‑mass grows linearly with the number of
copies.  Because the evaluation function is

    E(a) = 2·n·max(a*a) / (sum(a)²),

the denominator grows quadratically with the number of copies, whereas the
numerator grows only linearly (the factor n).  The resulting bound for k
copies (each of length L, separated by a gap of L zeros) is

    E_k = π·(2k‑1) / (2k²)      (asymptotically, with max≈1).

Even k = 2 gives E ≈ 1.18, well below the target 1.503; k = 3 brings it
down to ≈0.87.  The implementation tries a few values of k (subject to a
maximum total length) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the first n coefficients of (1‑z)^(-1/2) (the arcsine series)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i) / 4^i.

    Recurrence: a_{i+1} = a_i * (2·i+1) / (2·(i+1)).
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0,
                     seed: int = 42,
                     **kwargs):
    """
    Build a sequence by concatenating several copies of the arcsine coefficients.
    Returns the sequence (list[float]) with the smallest evaluation score found
    within the time budget.
    """
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Base length L: reuse the length of the previous best construction
    # if available, otherwise fall back to a reasonable default.
    # --------------------------------------------------------------
    try:

    # --------------------------------------------------------------
    base_coeffs = _arcsine_coeffs(L).tolist()

    # --------------------------------------------------------------
    # Search over a small number of copies.
    # We keep the total length below a safe bound (≈20000) to stay within
    # the 1000 s time limit for the convolution.
    # --------------------------------------------------------------
    MAX_TOTAL_LEN = 20000
    best_score = float('inf')
    best_seq = None

    # Upper bound on copies we can afford given the length limit.
    max_copies = max(1, (MAX_TOTAL_LEN // L + 1) // 2)   # solve (2k‑1)L ≤ MAX_TOTAL_LEN
    max_copies = min(max_copies, 6)                     # a tiny ceiling for safety

    for k in range(1, max_copies + 1):
        # stop early if we are out of budget
        if time.time() - start > budget_s - 0.5:
            break

        # Gap length = L (zero padding between copies).  This guarantees
        # that the cross‑convolution between two distinct copies reproduces

        for i in range(k):
