Projection onto simplex: There is algorithm to project onto the probability simplex: sort descending, find threshold θ.

Pseudo:

```
def proj_simplex(v):
    # projects v onto {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    np.random.seed(seed)
    n = kwargs.get('length', len(height_sequence_1) if defined else 6027)
    
    # initial shape: arcsine
    a_arcsine = compute_arcsine(n)  # as before 
    best_seq = a_arcsine
    best_score = compute_score(best_seq)
    print(f"[init] length={n}, arcsine score={best_score}")

    # explore mixture with uniform
    uniform = np.full(n, 1.0 / n)
    mix_factors = np.linspace(0, 1, 101)  # 0-1 step 0.01

    for lam in mix_factors:
        cand = (1 - lam) * a_arcsine + lam * uniform
        # normalization not needed because both sum to 1
        score = compute_score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[mix] lambda={lam:.3f}, improved score={best_score:.6f}")

    # explore delta shape
    deltas = np.geomspace(1e-6, 10, num=30)  # log spaced
    for delta in deltas:
        i = np.arange(n, dtype=np.float64)
        denom = np.sqrt((i + delta) * (n - 1 - i + delta))
        cand = 1.0 / denom
        cand_sum = cand.sum()
        cand /= cand_sum
        score = compute_score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[delta] delta={delta:.6g}, improved score={best_score:.6f}")

    # local hill climbing from best_seq
    deadline = time.time() + budget_s - 0.5  # safety margin
    it = 0
    a = best_seq.copy()
    n = len(a)
    while time.time() < deadline and it < 200000:
        # choose random pair indices
        i, j = np.random.choice(n, 2, replace=False)
        # random proportion to move, e.g., up to 10% of source
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        # move fraction
        max_move = 0.1 * a[src]  # up to 10% of source
        # random delta uniform [0, max_move]
        delta = np.random.random() * max_move
        if delta <= 0:
            it += 1
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # compute score quickly via convolution compute_score
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            a = a_new
            best_seq = a_new
            print(f"[hill] it={it}, move src={src} dst={dst} delta={delta:.6g}, score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}")

def max_convolution(seq):
    n = len(seq)
    # use FFT with length m = 2^next_pow2(2*n-1)
    m = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, m)
    conv = np.fft.irfft(f * f, m)
    # conv length m, but only first 2*n-1 are valid
    return conv[:2*n-1].max()

def hill_climb(a, time_deadline, max_iters=100000):
    n = len(a)
    best = a.copy()
    best_score = compute_score(best)
    it = 0
    while time.time() < time_deadline and it < max_iters:
        i, j = np.random.choice(n, size=2, replace=False)
        # decide move direction: from larger to smaller
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        # choose move fraction up to 10% of source
        max_move = 0.1 * a[src]
        if max_move == 0:
            it += 1
            continue
        delta = np.random.random() * max_move
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # project onto simplex may not be needed because sum unchanged, but ensure nonneg
        a_new = proj_simplex(a_new)  # small overhead
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new
            a = a_new  # continue from this improved candidate
            # prints occasional
            if it % 1000 == 0:
                print(f"[hill] it={it}, new best={best_score:.6f}")
        it += 1
    return best, best_score

"""
propose_candidate
=================

Search for a step‑function (non‑negative sequence) that improves on the
standard arcsine construction.  The algorithm works in three phases:

1. **Generation of diversified candidates** –  
   * the exact arcsine sequence,
   * a family  a_i = 1/√((i+δ)(n‑1−i+δ))   (δ controls the edge regularisation),
   * power‑law shapes a_i = (i+1)^(−p)    (p sweeps the decay),
   * a small mixture of the arcsine with the uniform distribution.

   All candidates are normalised to have unit sum, their objective
   ``2·n·max(conv(a,a))`` is evaluated (the denominator is 1) and the best
   one is kept.

2. **Local stochastic hill‑climbing** –  
   Starting from the best candidate, random pairwise mass transfers are
   performed (up to 10 % of the source entry).  A move is kept if the
   evaluation improves.  The search runs until the time budget expires.

3. **Return the best sequence** –  
   The final sequence is returned as a plain Python list; a validation call
   to ``evaluate_sequence`` is printed for sanity.

The whole routine respects the supplied time budget (default 1000 s) and
only uses NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    # Fast projection via sorting
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Return uniform distribution if projection fails
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(v):
    """Return max of the linear convolution v * v (full length)."""
    n = v.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fv = np.fft.rfft(v, m)
    conv = np.fft.irfft(fv * fv, m)[: 2 * n - 1]
    return conv.max()


def compute_score(v):
    """Evaluate the objective 2·n·max(conv(v,v)) / (sum v)^2."""
    total = v.sum()
    if total < 1e-12:
        return float("inf")
    maxc = max_convolution(v)
    return float(2.0 * len(v) * maxc / (total * total))


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def generate_arcsine(n):
    """Exact discrete arcsine coefficients (binomial(2k,k)/4^k)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()
    return a


def generate_delta(n, delta):
    """a_i = 1 / sqrt((i+δ)*(n‑1‑i+δ)), then normalised."""
    i = np.arange(n, dtype=np.float64)
    denom = np.sqrt((i + delta) * (n - 1 - i + delta))
    a = 1.0 / denom
    a /= a.sum()
    return a


def generate_power_law(n, p):
    """a_i ∝ (i+1)^{‑p}, then normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = i ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Local optimisation (simple hill‑climbing)
# ----------------------------------------------------------------------


def hill_climb(start, deadline, max_iters=200000):
    """Stochastic pairwise mass transfer improving the objective."""
    a = start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # move a tiny fraction from the larger entry to the smaller one
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i

        max_move = 0.1 * a[src]                     # up to 10 % of source
        if max_move <= 0.0:
            it += 1
            continue

        delta = np.random.random() * max_move
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # numerical safety – re‑project onto the simplex
        a_new = proj_simplex(a_new)

        s = compute_score(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            a = a_new                                   # continue from improvement
            if it % 1000 == 0:
                print(f"[improve] it={it:05d}, score={best_score:.6f}")
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Produce a candidate sequence of non‑negative heights.
    The routine explores several parametric families, keeps the best
    candidate, and finally refines it with a cheap hill‑climbing loop.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Determine length
    # ------------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)      # use the previously found sequence if present
        except Exception:
            n = 6027                        # fallback length (matches the last run)
    print(f"[setup] sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Build a pool of base candidates
    # ------------------------------------------------------------------
    candidates = []

    # exact arcsine
    arcsine_seq = generate_arcsine(n)
    candidates.append(("arcsine", arcsine_seq))

    # delta‑family (regularising the singularities)
    for delta in np.geomspace(1e-4, 10.0, num=30):
        cand = generate_delta(n, delta)
        candidates.append((f"delta_{delta:.3g}", cand))

    # power‑law family
    for p in np.linspace(0.30, 0.80, num=26):
        cand = generate_power_law(n, p)
        candidates.append((f"pow_{p:.3f}", cand))

    # small mixture of arcsine with uniform (lambda in [0,0.2])
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for lam in np.linspace(0.0, 0.20, num=11):
        cand = (1 - lam) * arcsine_seq + lam * uniform
        candidates.append((f"mix_{lam:.3f}", cand))

    # ------------------------------------------------------------------
    # 3. Evaluate all candidates (fast objective, no scaling needed)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for name, seq in candidates:
        sc = compute_score(seq)
        print(f"[candidate] {name:12s} score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    print(f"[init] best initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local stochastic optimisation (hill climbing)
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 1.0          # keep a safety margin
    refined_seq, refined_score = hill_climb(best_seq, deadline)

    if refined_score < best_score:
        best_seq = refined_seq
        best_score = refined_score

    print(f"[final] best score after hill‑climb = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Verify with the provided evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[verify] evaluate_sequence returns {ev:.6f}")
    except Exception as exc:
        print("[verify] evaluate_sequence failed:", exc)

    # ------------------------------------------------------------------
    # 6. Return plain list as required by the evaluator
    # ------------------------------------------------------------------
    return best_seq.tolist()
