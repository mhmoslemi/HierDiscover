import time
import math
import numpy as np
from scipy.optimize import linprog

def _catalan_sequence(n):
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _conv(a):
    return np.convolve(a, a)

def _top_k_indices(conv, K):
    # returns K indices with largest values
    if K >= len(conv):
        return np.arange(len(conv))
    # argpartition gives unsorted top K
    idx = np.argpartition(-conv, K-1)[:K]
    # sort them descending to maybe more stable
    idx = idx[np.argsort(-conv[idx])]
    return idx

def _rev_average(a, top_k):
    n = a.shape[0]
    rev_sum = np.zeros_like(a)
    idx = np.arange(n)
    for k in top_k:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev_sum[mask] += a[j[mask]]
    rev_avg = rev_sum / len(top_k)
    return rev_avg

def _project_and_normalise(x, target_sum):
    """Project onto non‑negative orthant, clip to bounds, and renormalise."""
    x = np.maximum(x, 0.0)
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s > 0:
        x = x * (target_sum / s)
    return x

def _adam_step(a, target_sum, m, v, it, lr0=0.5, K=5, beta1=0.9, beta2=0.999, eps=1e-8):
    n = a.shape[0]
    S = a.sum()
    conv = _conv(a)
    M = conv.max()
    top = _top_k_indices(conv, K)
    rev_avg = _rev_average(a, top)
    # gradient of the objective (E = 2 n M / S^2)
    grad = (4.0 * n) / (S * S) * rev_avg - (4.0 * n * M) / (S ** 3)
    # Adam update
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)
    lr = lr0 / (1.0 + 1e-4 * it)   # decaying learning rate
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = _project_and_normalise(a_new, target_sum)
    return a_new, m, v, M, conv

def _lp_improve(a, target_sum, M, t=0.1):
    """
    Solve a linear programme that maximises sum(b) under the
    constraint (a * b) <= M element‑wise.
    Return the mixed vector (1‑t)*a + t*g where g is the scaled LP solution.
    """
    n = a.shape[0]
    # Build constraint matrix A (2n‑1 rows, n cols)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # each column j contributes a[j] on rows j .. j+n-1
    for j in range(n):
        A[j:j+n, j] = a
    # Upper bound for each row = M
    ub = np.full(rows, M, dtype=float)
    # Objective: maximise sum(b) -> minimise -sum(b)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None) for _ in range(n)]
    # Solve with HiGHS (fast interior point)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        # fail gracefully
        return None
    b0 = res.x
    sum_b = b0.sum()
    if sum_b <= 0:
        return None
    # Scale to same total sum

    # Convex combination
    a_mix = (1.0 - t) * a + t * g
    a_mix = _project_and_normalise(a_mix, target_sum)
    return a_mix

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first n coefficients of the series
        c_i = binom(2 i, i) / 4**i   (i = 0, 1, …)
    which satisfies (c * c)[k] ≈ 1 for k away from the borders.
    The recurrence  c_{i+1}=c_i * (2 i+1) / (2 i+2)  is used.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a."""
    return np.convolve(a, a)


def _top_k_indices(conv: np.ndarray, K: int) -> np.ndarray:
    """Indices of the K largest entries of `conv`."""
    if K >= conv.size:
        return np.arange(conv.size)
    idx = np.argpartition(-conv, K - 1)[:K]          # unsorted
    # sort them descending – makes debugging easier
    idx = idx[np.argsort(-conv[idx])]
    return idx


def _rev_average(a: np.ndarray, top_k: np.ndarray) -> np.ndarray:
    """Average of the reversed‑shift vectors for the shifts in `top_k`."""
    n = a.shape[0]
    rev_sum = np.zeros_like(a)
    idx = np.arange(n)
    for k in top_k:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev_sum[mask] += a[j[mask]]
    return rev_sum / len(top_k)


def _project_and_normalise(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project onto the feasible box (0 ≤ x ≤ 1000), then renormalise the vector
    so that its L1‑norm equals `target_sum`.
    """
    x = np.maximum(x, 0.0)
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s > 0.0:
        x = x * (target_sum / s)
    return x


# ----------------------------------------------------------------------
# Adam optimiser that focuses on the K largest convolution entries ----
def _adam_step(a: np.ndarray,
               target_sum: float,
               m: np.ndarray,
               v: np.ndarray,
               it: int,
               lr0: float = 0.5,
               K: int = 7,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8):
    """
    Perform one projected Adam step on the evaluation functional.
    The gradient is taken as the average over the K currently largest
    convolution entries.
    Returns the new vector, the updated Adam moments, the current max
    convolution value, and the raw convolution array (needed for diagnostics).
    """
    n = a.shape[0]
    S = a.sum()
    conv = _conv(a)
    M = conv.max()

    # indices of the K largest entries of the convolution
    top = _top_k_indices(conv, K)

    # average of the reversed‑shift vectors for those indices
    rev_avg = _rev_average(a, top)

    # gradient of  E = 2 n M / S**2
    grad = (4.0 * n) / (S * S) * rev_avg - (4.0 * n * M) / (S ** 3)

    # Adam update
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)

    # simple learning‑rate decay (same schedule as the reference code)
    lr = lr0 / (1.0 + 1e-4 * it)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    a_new = a - step
    a_new = _project_and_normalise(a_new, target_sum)

    return a_new, m, v, M, conv


# ----------------------------------------------------------------------
# Linear‑programming refinement -----------------------------------------
def _lp_improve(a: np.ndarray,
                target_sum: float,
                M_bound: float,
                t: float = 0.12):
    """
    Solve the LP
        maximise   1ᵀ b
        subject to (a * b)[k] ≤ M_bound   for all k,
                  b ≥ 0.
    The result `b` is scaled to the required total sum and mixed with `a`
    using a convex combination weight `t`.
    Returns the mixed vector or None if the LP fails.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz‑like matrix A where
    #   (A b)[k] = Σ_i a_i b_{k‑i}
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a                               #   a_i contributes on rows j…j+n‑1

    ub = np.full(rows, M_bound, dtype=float)           # every row ≤ M_bound
    c = -np.ones(n, dtype=float)                       # maximise sum(b)  → minimise –sum(b)
    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:   # safety net – any unexpected LinProg error
        return None

    if not res.success:
        return None

    b0 = res.x
    sum_b = b0.sum()
    if sum_b <= 0.0:
        return None

    # Scale to the required normalisation and mix
    g = b0 * (target_sum / sum_b)
    a_mix = (1.0 - t) * a + t * g
    a_mix = _project_and_normalise(a_mix, target_sum)
    return a_mix
