Standard projection onto simplex algorithm:

def project_simplex(v):
    # ensure sum=1, v>=0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv/np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

#!/usr/bin/env python
"""
propose_candidate
=================

Iterative optimisation of a non‑negative step function.
The construction starts from the analytically optimal “arcsine”
coefficients (a_i = C(2i,i)/4^i) and then repeatedly improves the
self‑convolution peak by solving a small linear programme (LP).  The LP
maximises the ℓ₁‑norm of a new candidate distribution **g**, while keeping
the mixed convolution (f∗g) below the current worst convolution value *M*.
A convex combination ``f ← (1‑t)·f + t·g`` (with t≈0.2) lowers the
maximum of ``f∗f``.  After a handful of LP steps a cheap random‑pair
refinement is performed.  The whole routine runs within the 1000 s
budget and returns the best sequence found.

The algorithm is deliberately written without closures, with all helpers
at module level and only standard scientific Python libraries (`numpy`,
`scipy.sparse` and `cvxpy`).  If `cvxpy` is unavailable the code falls
back to a simple projected‑gradient descent.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine coefficients (C(2i,i)/4^i)
# ----------------------------------------------------------------------
def arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first *n* coefficients of (1‑z)^(-1/2).

    Recurrence:
        a[0] = 1
        a[i] = a[i‑1] * (2*i‑1) / (2*i)   for i ≥ 1
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x)=1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: solve the LP  max sum(g)  s.t.  (f∗g)_k ≤ M , g ≥ 0
# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float):
    """Return a feasible distribution g (sum(g)=1) with maximal ℓ₁‑norm
    under the convolution constraints (f∗g)≤M.
    """
    import cvxpy as cp
    import scipy.sparse as sp

    n = f.shape[0]
    # Build sparse convolution matrix A where (A @ g)_k = (f∗g)_k
    # Row k contains f_i for each i such that 0≤k-i<n.
    A = sp.lil_matrix((2 * n - 1, n), dtype=float)
    for i in range(n):
        A[i : i + n, i] = f[i]
    A = A.tocsr()

    g = cp.Variable(n, nonneg=True)
    # Vector of the same bound for every row
    bounds = M * np.ones(2 * n - 1, dtype=float)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= bounds])
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
               abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)

    if g.value is None:
        raise RuntimeError("LP did not converge")
    g_val = np.asarray(g.value).flatten()
    # Normalise to a probability vector
    g_val = np.maximum(g_val, 0.0)
    total = g_val.sum()
    if total <= 0:
        # fallback to uniform if something went wrong
        g_val = np.full(n, 1.0 / n)
    else:
        g_val = g_val / total
    return g_val


# ----------------------------------------------------------------------
# Helper: simple projected‑gradient descent (fallback)
# ----------------------------------------------------------------------
def gradient_descent_refine(a: np.ndarray, budget: float, start: float):
    """Perform a few thousand sub‑gradient steps to lower max(f∗f)."""
    n = a.shape[0]
    best = a.copy()
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()

    max_iters = 20000
    for it in range(max_iters):
        if time.time() - start > budget - 0.5:
            break

        conv = np.convolve(a, a)
        max_val = conv.max()
        # indices where the max is attained (within tolerance)
        idxs = np.where(conv >= max_val - 1e-12)[0]

        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # gradient entry = 2 * a_{k-i}
            rev = a[k - np.arange(i_start, i_end + 1)]
            grad[i_start : i_end + 1] += 2.0 * rev
        grad /= len(idxs)

        # step size schedule
        eta = 0.5 / np.sqrt(it + 1.0)
        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        # evaluate improvement
        new_score = 2.0 * n * np.convolve(a_new, a_new).max()
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
        a = a_new
    return best
