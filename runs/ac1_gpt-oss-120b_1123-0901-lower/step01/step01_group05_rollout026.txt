    # set region thresholds

    max_frac_multi = 0.02   # for multi-target

    deadline = time.time() + budget_s - 0.2
    it = 0
    while time.time() < deadline:

        remaining = deadline - time.time()

        T = end_temp + (start_temp - end_temp) * frac
        move_type = rng.random()
        cand = best_seq.copy()
        if move_type < 0.4:
            # single transfer early->mid
            i = rng.integers(0, early_end)
            j = rng.integers(mid_start, mid_end)
            delta = cand[i] * rng.uniform(0, max_frac_single)
            cand[i] -= delta
            cand[j] += delta
        elif move_type < 0.7:
            # multi transfer from early to many mids
            i = rng.integers(0, early_end)
            delta_total = cand[i] * rng.uniform(0, max_frac_multi)
            cand[i] -= delta_total
            m = rng.integers(5, 15)  # number of targets
            targets = rng.integers(mid_start, mid_end, size=m)
            add = delta_total / m
            for t in targets:
                cand[t] += add
        else:
            # local multiplicative perturb
            idx = rng.integers(n)
            factor = np.exp(rng.normal(0, max_factor_local))
            cand[idx] = max(0.0, cand[idx] * factor)

        # evaluate
        try:
            score = evaluate_sequence(cand)
        except Exception:
            continue

        # acceptance
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[improve] iter {it} score={best_score:.6f}")

"""
Propose a candidate step‐function sequence.

The baseline is the “inverse‑square‑root’’ (central binomial) sequence,
which gives a convolution that is exactly flat (value 1) for all
indices k ≤ n‑1.  This shape maximises the total sum under the constraint
max(conv) ≤ 1, yielding the classic asymptotic score π/2 ≃ 1.571.
To beat this value we must allow the maximum of the convolution to drop
below 1 while keeping the total sum as large as possible.

The optimizer performs a simulated‑annealing style stochastic search
starting from the baseline.  The main move is a *mass‑transfer*:
a small amount of weight is moved from an early index (where the entry
contributes strongly to many convolution terms) to a middle index
(where its impact on the maximum is milder).  Because the total sum is
conserved, any reduction in the peak of the convolution directly lowers
the evaluation score.

Additional moves (single‑entry multiplicative perturbations and
multi‑target transfers) increase exploration.  The algorithm runs for the
entire time budget (≈ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Produce a_i = binom(2*i,i)/4**i = 1/√(π i) (exact recurrence)."""
    seq = [1.0]                     # a_0 = 1
    for i in range(1, n):
        # recurrence a_{i} = a_{i-1} * (2*i-1)/(2*i)
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq

    convolution while keeping the total sum large.  Returns a list of
    non‑negative floats.

        best_seq = [1.0] * n

        # ------------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(cand)   # type: ignore[name-defined]
        except Exception:
            continue  # illegal candidate, skip

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[improve] iter {it:06d} new best score={best_score:.6f}")
        else:
            # simulated‑annealing acceptance of worse candidates
            if temp > 0.0:
                accept_prob = np.exp((best_score - cand_score) / temp)
                if rng.random() < accept_prob:
                    best_seq = cand

        it += 1

    # ----------------------------------------------------------------------
    # 7. Return the best sequence discovered.
    # ----------------------------------------------------------------------
    print(f"[final] iterations={it}, best score={best_score:.6f}")
    return list(best_seq)
