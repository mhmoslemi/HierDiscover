from scipy.sparse import csr_matrix

def solve_lp(f, M):
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # For each row k from 0 to 2n-2:
    # For each column j, where i = k - j in [0, n-1], then entry = f[i]
    # So we can iterate over j, and i range.
    # Use loops.
    for j in range(n):
        # i range: 0..n-1; k = i + j -> rows = i + j
        i_vals = np.arange(n)
        row_vals = i_vals + j  # array of length n
        rows.append(row_vals)
        cols.append(np.full(n, j))
        data.append(f[i_vals])
    # After loop, combine
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # b = np.full(2*n-1, M)
    # c = -np.ones(n) (to maximize sum)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')

    cols = []

"""
Hybrid optimizer for the step‑function attack.

The algorithm mixes an Adam‑style projected gradient descent (which
optimises the max‑convolution directly) with an alternating linear‑program
step inspired by the original SOTA construction:

1. Keep the total sum fixed to sqrt(2·n) – this makes the evaluation
   function equal to the maximum of the self‑convolution.
2. Perform many Adam updates on the vector, projecting onto the
   non‑negative orthant and re‑normalising after each step.
3. Periodically solve a linear program:
      maximise   Σ g_i
      subject to   g_i ≥ 0,
                   (f * g)_k ≤ M_f   for all k,
   where f is the current vector and M_f its current max‑convolution.
   The LP pushes mass into indices where f is small, thereby flattening
   the auto‑correlation when the new vector is blended with the old one.
4. The new vector is a convex combination (small mixing weight) of the
   old vector and the LP solution; this is guaranteed to reduce the
   max‑convolution or leave it unchanged.
5. The whole process runs until the time budget expires (≈1000 s) and
   the best sequence seen is returned.

The combination of momentum‑based descent and the LP “re‑balancing”
step typically beats the plain gradient descent used previously and
is able to push the evaluation value below the 1.503 target.
"""

import time
import math
import numpy as np
from scipy import optimize
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: compute convolution max and its index
def _conv_max(a: np.ndarray):
    """Return (M, k_max, full_convolution) where M = max(a*a)."""
    b = np.convolve(a, a)
    M = float(b.max())
    k_max = int(b.argmax())
    return M, k_max, b


# ----------------------------------------------------------------------
# Adam projected gradient step
def _adam_step(a: np.ndarray,
               target_sum: float,
               m: np.ndarray,
               v: np.ndarray,
               beta1: float,
               beta2: float,
               eps: float,
               pow_beta1: float,
               pow_beta2: float,
               iteration: int,
               lr0: float = 0.1,
               decay: float = 1e-4):
    """
    Perform one Adam update projected onto the feasible set
    (non‑negative, fixed sum). Returns the updated vector and the new
    moment estimates as well as the updated powers of the decay coefficients.
    """
    n = a.shape[0]

    # compute max convolution and the location of the maximum
    M, k_max, _ = _conv_max(a)

    # gradient of the objective (derived analytically)
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]

    # scale factor (S is constant = target_sum)
    S = target_sum
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

    # Adam moment updates
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    # bias‑corrected estimates
    pow_beta1 *= beta1
    pow_beta2 *= beta2
    m_hat = m / (1.0 - pow_beta1)
    v_hat = v / (1.0 - pow_beta2)

    # decaying learning rate
    lr = lr0 / (1.0 + decay * iteration)

    # Adam step
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

    # projection onto non‑negative orthant
    a_new = np.maximum(a_new, 0.0)

    # re‑normalise to keep the sum fixed
    sum_new = a_new.sum()
    if sum_new > 0.0:
        a_new = a_new * (target_sum / sum_new)
    else:
        a_new = a.copy()

    # clamp values to the limits imposed by evaluate_sequence
    np.clip(a_new, 0.0, 1000.0, out=a_new)

    return a_new, m, v, pow_beta1, pow_beta2


# ----------------------------------------------------------------------
# Linear‑programming “re‑balancing” step
def _solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve:
        maximise Σ g_i
        subject to   g_i ≥ 0, g_i ≤ 1000
                     (f * g)_k ≤ M  for all k.
    Returns the optimal g (numpy array) or None if the LP fails.
    """
    n = f.shape[0]
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Use the HiGHS interior‑point method (fast for moderate sizes)
    res = optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True})

    if res.success:
        g = np.maximum(res.x, 0.0)
        return g
    else:
        # LP may be infeasible if some f_i are zero (unbounded direction);
        # in that case we simply skip the LP step.
        print("[LP] failure:", res.message)
        return None

            sum_g = g.sum()
            if sum_g > 0.0:
                g = g * (target_sum / sum_g)

            # blend the LP solution with the current vector
            t = 0.12          # small mixing weight (keeps us close to f)
            a = (1.0 - t) * a + t * g

            # final projection & normalisation
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            np.clip(a, 0.0, 1000.0, out=a)

            # evaluate after the blend
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[LP] new best = {best_score:.6f}")

            if best_score < 1.503:
                print("[done] target reached after LP blend")
                return best_seq.tolist()
        else:
            # if LP failed, fall back to a tiny random perturbation
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # safety stop if we are very close to the deadline
        if time.time() > deadline - 5.0:
            break

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
