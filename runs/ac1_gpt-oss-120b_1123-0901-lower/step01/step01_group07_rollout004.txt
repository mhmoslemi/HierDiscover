"""
Improved step‑function search.

The algorithm starts from a good power‑law construction (or the
`height_sequence_1` variable if it exists) and then performs a fast
hill‑climbing that repeatedly moves a little amount of mass from the
indices that contribute most to the current maximal aperiodic
autocorrelation towards indices that contribute least.  The score
`2·n·max(conv(a,a))/sum(a)²` is evaluated with a cheap FFT‑based routine,
so thousands of iterations fit comfortably into the 1000 s budget.
Only improvements are kept; occasional random perturbations help to
escape shallow local minima.  The best sequence found is returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence for valid inputs)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the same value as ``evaluate_sequence`` but much faster.

    The function assumes ``seq`` contains only finite non‑negative numbers
    (the search code enforces that).  It uses an FFT based convolution.
    """
    n = seq.shape[0]
    # length of the full (aperiodic) convolution
    conv_len = 2 * n - 1
    # next power of two for an efficient FFT
    m = 1 << (conv_len - 1).bit_length()
    # real‑to‑real FFT
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)[:conv_len]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

# Core optimisation routine
# ----------------------------------------------------------------------

                      budget_s: float = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 1000).
    **kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns

    list[float]
        The best height sequence discovered.
    """

    deadline = time.time() + budget_s - 0.1   # safety margin

    # 1. Initial sequence (use the provided one if available)
    # --------------------------------------------------------------
    try:

        init_seq = np.array(height_sequence_1, dtype=float)
        n = init_seq.shape[0]

    seq = np.clip(init_seq, 0.0, 1000.0)

    best_seq = seq.copy()

    print(f"[init] length={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Main hill‑climbing loop
    # --------------------------------------------------------------
    # control parameters (they adapt during the run)
    step_fraction = 0.25                 # initial fraction of source mass to move
    min_step = 1e-8                      # absolute lower bound on moved mass
    decay_rate = 0.9995                  # gradual reduction of step size
    random_move_prob = 0.02              # occasional random jitter
    iteration = 0
    no_improve = 0

    # pre‑computed FFT size (does not change because we keep n fixed)
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()

    while time.time() < deadline:
        iteration += 1

        # ----- a) compute full convolution via FFT and locate the peak -----
        fft_seq = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(fft_seq * fft_seq, n=fft_len)[:conv_len]
        max_val = float(conv.max())
        k_max = int(conv.argmax())          # index of the current maximum

        # ----- b) partner values for the maximal lag ----------------------
        # partner[j] = a_{k_max - j}   (zero when the index is out of range)
        idx = k_max - np.arange(n)          # vector of partner indices

        final_score = best_score
        print(f"[final] best score (fast) = {final_score:.6f}")

    return best_seq.tolist()
