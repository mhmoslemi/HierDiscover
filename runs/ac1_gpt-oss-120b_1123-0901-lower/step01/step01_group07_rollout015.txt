"""
Propose a candidate sequence of non‑negative heights that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm explores a richer family of shapes than the simple power‑law used
before.  Each candidate has the form

        a_i = w * (i + γ)^(‑p) + (1‑w) * 1                (i = 0 … n‑1)

where
    p     – decay exponent (≈0.5 … 0.9),
    γ     – a small shift (log‑uniformly sampled,
                              0.05 … 50),
    w     – mixing weight between the power‑law and a flat background
            (0 … 1).

The scale of the vector does not influence the objective, therefore no
additional normalisation is required – the raw values can be fed directly to
``evaluate_sequence``.

Search strategy (1000 s budget):
1. A coarse grid over (p,γ) with w=1 to locate a good region.
2. A large random sampling of (p,γ,w) covering the whole domain.
3. A final local refinement using Gaussian perturbations around the best
   parameters found.

All candidates are evaluated with the given ``evaluate_sequence`` function.
The best sequence (as a plain Python list) is returned.
"""

import time
import numpy as np
import random

# The evaluation function is available in the global namespace
# from the surrounding environment:
#     evaluate_sequence(sequence: list[float]) -> float


def _make_sequence(n: int, p: float, gamma: float, w: float = 1.0) -> np.ndarray:
    """
    Construct the candidate vector.

    Parameters
    ----------
    n : int
        Length of the sequence.
    p : float
        Power‑law exponent (p > 0).
    gamma : float
        Shift of the power‑law (γ > 0).
    w : float
        Mixing weight (0 ≤ w ≤ 1).  w = 1 gives the pure power‑law,
        w = 0 yields a constant vector.

    Returns
    -------
    np.ndarray
        The sequence of heights (float64).
    """
    indices = np.arange(n, dtype=np.float64) + gamma
    power = indices ** (-p)
    if w < 1.0:
        # add a flat background of height 1 (any constant works because of
        # scale‑invariance, 1 keeps numbers in a comfortable range)
        seq = w * power + (1.0 - w) * 1.0
    else:
        seq = power
    # use the same clipping policy as evaluate_sequence (clamp to [0,1000])
    # – this is only to avoid extremely large intermediate values that would
    #   be truncated later.
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000,
                     length: int = 3428,
                     **kwargs):
    """
    Search for a low‑score step function.

    The search respects a hard time limit (`budget_s`).  It returns a list of
    heights (default length 3428) that yields a small value of
    ``evaluate_sequence``.
    """
    random.seed(seed)
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    n = length

    # best result bookkeeping
    best_seq = None
    best_score = float('inf')
    best_params = None   # tuple (p, gamma, w)

    # --------------------------------------------------------------
    # 1) Coarse grid (pure power‑law, w = 1)
    # --------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.85, 31)          # 0.55 … 0.85
    gamma_grid = np.logspace(np.log10(0.05), np.log10(30), 31)  # 0.05 … 30

    for p in p_grid:
        for gamma in gamma_grid:
            if time.time() > deadline:
                break
            seq = _make_sequence(n, p, gamma, w=1.0)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_params = (p, gamma, 1.0)
                print(f"[grid] new best p={p:.4f}, γ={gamma:.4f}, w=1.0, score={score:.6f}")
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2) Broad random sampling (p,γ,w)
    # --------------------------------------------------------------
    # we will try many random triples; the loop stops when time runs out.
    # The ranges are deliberately generous.
    while time.time() < deadline:
        # exponents favour the region known to be good but still allow exploration
        p = random.uniform(0.5, 0.9)

        # gamma is sampled log‑uniformly to cover several orders of magnitude
        log_gamma = random.uniform(np.log10(0.01), np.log10(100.0))
        gamma = 10 ** log_gamma

        w = random.uniform(0.0, 1.0)

        # small pragmatic cut‑offs to keep the shape reasonable
        if p <= 0 or gamma <= 0:
            continue

        seq = _make_sequence(n, p, gamma, w)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_params = (p, gamma, w)
            print(f"[rand] improved p={p:.4f}, γ={gamma:.4f}, w={w:.4f}, score={score:.6f}")

    # --------------------------------------------------------------
    # 3) Local refinement around the current best parameters
    # --------------------------------------------------------------
    if best_params is None:
        # fallback to the original construction if everything failed
        best_params = (0.5002, 0.3210, 1.0)
        best_seq = _make_sequence(n, *best_params[:2], w=best_params[2])
        best_score = evaluate_sequence(best_seq.tolist())

    p_best, gamma_best, w_best = best_params
    # Standard deviation for the Gaussian proposals – shrink slowly
    sigma_p = 0.02
    sigma_gamma = 0.3
    sigma_w = 0.1

    refinement_iters = 12000
    for iteration in range(refinement_iters):
        if time.time() > deadline:
            break

        # propose a perturbation
        p_new = p_best + np.random.randn() * sigma_p
        gamma_new = gamma_best * np.exp(np.random.randn() * sigma_gamma)  # log‑normal step
        w_new = w_best + np.random.randn() * sigma_w

        # enforce bounds
        if not (0.45 < p_new < 0.95):
            continue
        if not (0.01 < gamma_new < 200.0):
            continue
        if not (0.0 <= w_new <= 1.0):
            continue

        seq = _make_sequence(n, p_new, gamma_new, w_new)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            p_best, gamma_best, w_best = p_new, gamma_new, w_new
            print(f"[refine] iter {iteration:05d} → p={p_new:.4f}, γ={gamma_new:.4f}, "
                  f"w={w_new:.4f}, score={score:.6f}")

        # slowly reduce the step sizes to focus the search
        if iteration % 2000 == 0 and iteration > 0:
            sigma_p *= 0.9
            sigma_gamma *= 0.9
            sigma_w *= 0.9

    # final safety – convert to plain Python list
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} "
          f"with p={p_best:.4f}, γ={gamma_best:.4f}, w={w_best:.4f}")
    return result
