    M = b.max()

def _gradient_step(a: np.ndarray, target_sum: float, lr: float):
    n = a.shape[0]
    S = a.sum()
    if S == 0.0:
        return a.copy()
    M, k_max, _ = _conv_max(a)
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    sum_new = a_new.sum()
    if sum_new > 0.0:
        a_new = a_new * (target_sum / sum_new)
    else:
        a_new = a.copy()
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new

    deadline = start_time + budget_s - 5.0  # safety margin

    # Initial sequence
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.shape[0]
    except Exception:
        n = 2100
        base = np.random.rand(n)

    base = np.clip(base, 0.0, 1000.0)
    target_sum = math.sqrt(2.0 * n)
    base = base * (target_sum / base.sum())
    f = base.copy()

    # Keep best solution
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Pre‑compute static row/col indices for the convolution matrix.
    # rows = i + j where i is index of f (column), j runs 0..n-1 (row offset)
    rows = (np.arange(n)[:, None] + np.arange(n)).reshape(-1)
    cols = np.repeat(np.arange(n), n)

    # Main loop
    iteration = 0
    grad_lr0 = 0.7
    grad_decay = 2e-4
    t = 0.2                # mixing factor for LP updates
    t_decay = 0.995        # slowly reduce mixing
    grad_steps_between_lp = 400   # number of cheap gradient steps
    last_improve = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) A burst of cheap gradient steps
        for _ in range(grad_steps_between_lp):
            # adapt learning rate
            lr = grad_lr0 / (1.0 + grad_decay * iteration)
            f = _gradient_step(f, target_sum, lr)
            # evaluate
            cur_score = evaluate_sequence(f.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                last_improve = iteration
                print(f"[iter {iteration}] new best after grad = {best_score:.6f}")
            iteration += 1
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 2) Linear‑program update
        M, _, _ = _conv_max(f)
        # Build the constraint matrix A with current f as coefficients.
        # data = f[i] repeated n times for each column i.
        data = np.repeat(f, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

            eps = 1e-9 * M

        if not res.success:
            print(f"[iter {iteration}] LP failed, terminating.")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print(f"[iter {iteration}] LP produced zero mass, terminating.")
            break

        # Scale g to the fixed total mass
        g = g0 * (target_sum / sum_g0)

        # Mix the old and the LP‑generated sequence
        f_new = (1.0 - t) * f + t * g
        # Re‑normalise (numerical drift)
        f_new = f_new * (target_sum / f_new.sum())

        # Evaluate the mixed sequence
        cur_score = evaluate_sequence(f_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f_new.copy()
            last_improve = iteration
            print(f"[iter {iteration}] new best after LP = {best_score:.6f}")

        if cur_score <= best_score:
            f = f_new
        else:
            # occasional acceptance to escape plateaus
            if np.random.rand() < 0.02:
                f = f_new

        # Decay mixing factor slowly
        t = max(0.01, t * t_decay)

        iteration += 1

    print(f"[done] total iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()

"""
Hybrid optimization for the step‑function bound.

The algorithm alternates cheap projected‑gradient steps with an exact
linear‑program update that maximises the total mass of an auxiliary
sequence g while keeping the convolution f * g bounded by the current
maximum M = max(f * f).  This is the block‑coordinate descent described
in the problem statement; the LP step yields the optimal “direction”,
and a convex combination with the current iterate reduces the bound.
Gradient steps provide fast local refinement between LP solves.
The routine runs for at most `budget_s` seconds and returns the best
sequence seen.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper: max of the auto‑convolution and the index where it occurs
def _conv_max(a: np.ndarray):
    """Return (M, k_max, full_conv) where M = max(a*a)."""
    b = np.convolve(a, a)
    M = b.max()
    k_max = int(b.argmax())
    return M, k_max, b


# ----------------------------------------------------------------------
# Simple projected gradient descent step (same as the reference code)
def _gradient_step(a: np.ndarray, target_sum: float, lr: float):
    """
    One projected gradient descent step for the objective
    E(a) = 2·n·max(a*a) / (sum a)^2  with the sum fixed to `target_sum`.
    """
    n = a.shape[0]
    S = a.sum()
    if S == 0.0:
        return a.copy()

    M, k_max, _ = _conv_max(a)

    # gradient of max(a*a)[k_max] w.r.t. a_p is 2·a_{k_max-p}
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]

    # derived analytical gradient of E
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)                     # projection onto ℝ₊
    sum_new = a_new.sum()
    if sum_new > 0.0:
        a_new = a_new * (target_sum / sum_new)        # renormalise
    else:
        a_new = a.copy()

    np.clip(a_new, 0.0, 1000.0, out=a_new)            # respect evaluate_sequence limits
    return a_new


# ----------------------------------------------------------------------

    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    The method combines:
      * many cheap projected‑gradient steps,
      * an exact linear‑program update (maximise total mass of g under
        f*g ≤ M), and
      * occasional random jitter / occasional acceptance of worse moves
        to avoid stagnation.

    The function runs until the time budget expires and returns the best
    sequence encountered.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # Initialise from the best known construction if available.
    try:
        base = np.array(height_sequence_1, dtype=float)   # global from previous runs
        n = base.shape[0]
    except Exception:
        n = 2100                                          # fallback length

    base = np.clip(base, 0.0, 1000.0)
    target_sum = math.sqrt(2.0 * n)                       # normalisation used in the paper
    base = base * (target_sum / base.sum())
    f = base.copy()

    # initial evaluation
    cur_score = evaluate_sequence(f.tolist())
    best_seq = f.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute the static row/column pattern for the convolution matrix.
    # rows[k,i] = i + k where k runs 0..n-1, i runs 0..n-1 → shape (2n‑1, n)

    grad_decay = 2e-4            # decay of the learning rate

    t_decay = 0.997              # slowly shrink the mixing factor

        # --------------------------------------------------------------

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:

            break

        # Scale g to the fixed total mass
        g = g0 * (target_sum / sum_g0)

        # Mix old and new sequence
        f_new = (1.0 - t) * f + t * g
        f_new = f_new * (target_sum / f_new.sum())   # renormalise (tiny drift)

        # Occasionally add a minute jitter to keep the search lively
        if iteration % 50 == 0:
            jitter = np.random.randn(n) * 1e-4 * target_sum / n
            f_new = np.maximum(f_new + jitter, 0.0)
            f_new = f_new * (target_sum / f_new.sum())

        cand_score = evaluate_sequence(f_new.tolist())

        # update global best if improved
        if cand_score < best_score:

            best_seq = f_new.copy()
            last_improve_iter = iteration

        # otherwise accept with a tiny probability
        if cand_score <= cur_score or np.random.rand() < 0.05:
            f = f_new
            cur_score = cand_score

        t = max(min_t, t * t_decay)

        if iteration - last_improve_iter > 5000:

            f = f[perm]
            cur_score = evaluate_sequence(f.tolist())
            last_improve_iter = iteration
            print(f"[iter {iteration}] performed random re‑shuffle")

        iteration += 1

    print(f"[done] total iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
