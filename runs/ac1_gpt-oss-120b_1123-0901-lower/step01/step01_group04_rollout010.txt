"""
propose_candidate
=================

Generate a non‑negative step function that gets as close as possible to the
theoretically optimal constant π/2 for ``evaluate_sequence``.

The construction uses the discrete arcsine distribution

    a_k ∝ (2k choose k) / 4**k ,   k = 0,…,n‑1

which is known to minimise the ℓ∞‑norm of the self‑convolution.
Because the bound is approached only asymptotically, we increase the length
of the sequence while time permits – each increase makes the score
slightly smaller (the error decays roughly like 1/n).

After a short “hill‑climbing” phase (tiny mass transfers between random
positions) we return the best sequence found within the allotted budget.
All numbers are non‑negative, the routine respects any existing global
``height_sequence_1`` and never uses closures or lambda expressions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: discrete arcsine coefficients of length n (sum normalised to 1)
# ----------------------------------------------------------------------
def arcsine_normalised(n: int) -> np.ndarray:
    """Return an n‑vector a with a_k ∝ (2k choose k) / 4**k  and  Σa = 1."""
    if n == 1:
        return np.ones(1, dtype=np.float64)
    ks = np.arange(1, n, dtype=np.float64)               # 1 … n‑1
    # recurrence factor  a_{k} = a_{k‑1} * ((2k‑1)(2k)) / (4 k²)
    factor = ((2.0 * ks - 1.0) * (2.0 * ks)) / (4.0 * ks * ks)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    a[1:] = np.cumprod(factor)                           # a_k = Π_{i=1}^k factor_i
    a /= a.sum()                                         # normalise to sum == 1
    return a


# ----------------------------------------------------------------------
# Helper: evaluate a candidate (list of floats) via the supplied function
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Wrapper around the external ``evaluate_sequence``."""
    # The external function expects a plain Python list.
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.

    The algorithm works in three stages:

    1. **Initial length** – use the length of an existing global
       ``height_sequence_1`` when present; otherwise start with a modest
       default (≈ 3000).

    2. **Length expansion** – keep doubling the length while time permits,
       rebuilding the arcsine distribution each time.  Because the
       asymptotic constant is π/2, each enlargement yields a small improvement.

    3. **Light hill‑climbing** – for the final length, perform a bounded
       number of random mass‑transfer moves (move 1 % of the source entry to a
       random destination) and keep any move that strictly lowers the score.

    The best sequence found before the time budget expires is returned.
    """
    start_time = time.time()
    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1. Determine the starting length
    # --------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)            # type: ignore[name-defined]
        except Exception:
            n = 3000                               # reasonable default

    # --------------------------------------------------------------
    # 2. Build the first arcsine sequence and evaluate it
    # --------------------------------------------------------------
    best_seq = arcsine_normalised(n)
    best_score = _score(best_seq)
    print(f"[init] length={n}, score={best_score:.9f}")

    # --------------------------------------------------------------
    # 3. Expand the length while we still have time.
    #    We double the size each round (capped at 120 000) because the
    #    improvement behaves like O(1/n).
    # --------------------------------------------------------------
    max_len = 120_000
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed - 0.5          # safety margin
        if time_left < 0.4:
            break

        new_n = min(n * 2, max_len)
        if new_n == n:
            break

        cand_seq = arcsine_normalised(new_n)
        cand_score = _score(cand_seq)
        print(f"[grow] length={new_n}, score={cand_score:.9f}")

        # The longer the sequence the better (or at worst equal),
        # so we always accept it as the new baseline.
        best_seq = cand_seq
        best_score = cand_score
        n = new_n

        # If we reached the hard cap we stop expanding.
        if n == max_len:
            break

    # --------------------------------------------------------------
    # 4. Light hill‑climbing on the final sequence.
    #    Move a tiny fraction (1 %) of mass from a larger entry to a
    #    smaller one and keep the change if it reduces the score.
    # --------------------------------------------------------------
    deadline = time.time() + (budget_s - (time.time() - start_time) - 0.5)
    it = 0
    max_iters = 20_000                      # more than enough for short runs
    while time.time() < deadline and it < max_iters:
        # pick two distinct indices uniformly at random
        i, j = np.random.choice(n, size=2, replace=False)

        # always transfer from the larger to the smaller entry
        if best_seq[i] < best_seq[j]:
            i, j = j, i

        # transfer 1 % of the source (guaranteed to stay non‑negative)
        delta = 0.01 * best_seq[i]
        if delta == 0.0:
            it += 1
            continue

        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_score = _score(new_seq)
        if new_score < best_score:
            best_seq = new_seq
            best_score = new_score
            print(f"[improve] it={it:05d}, score={best_score:.9f}")

        it += 1

    print(f"[final] length={len(best_seq)}, best_score={best_score:.9f}, "
          f"iterations={it}")

    # --------------------------------------------------------------
    # 5. Return a plain Python list as required by the scorer.
    # --------------------------------------------------------------
    return best_seq.tolist()
