def compute_max_conv(b):
    n = b.shape[0]
    size = 1 << ((2*n-1).bit_length())
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(B * B, size)[:2*n-1]
    return conv.max()

```
def hill_climb(seq, cur_score, deadline):

    size = 1 << ((2*n-1).bit_length())
    conv = compute_full_conv(seq)

    cur_max = conv.max()
    best_seq = seq.copy()
    best_max = cur_max
    best_score = cur_score

    step = 0.1  # initial proportion of src mass to move.
    it = 0
    while time.time() < deadline and it < max_iter:
        idx = int(np.argmax(conv))

        # valid ks
        low = max(0, idx - (n-1))
        high = min(idx, n-1)
        ks = np.arange(low, high+1)
        # contributions
        contrib = seq[ks] * seq[idx - ks]
        src = ks[int(np.argmax(contrib))]

        # Choose dst: index where partner value small
        # compute partner values:
        partner_vals = np.full(n, np.inf)
        # compute a_{idx - j} for all j
        # Actually we need values a_{idx - j} for each j (if in range else inf)
        # We'll compute for all indices:
        i_minus = idx - np.arange(n)
        mask = (i_minus >=0) & (i_minus < n)
        partner_vals[mask] = seq[i_minus[mask]]
        # Now choose dst with minimal partner but not src.
        # Among all indices (including those outside region where partner undefined -> treat as 0)
        # Set partner for out-of-range indices as 0 (since they don't affect conv[idx]).
        partner_vals[~mask] = 0.0
        # Exclude src
        partner_vals[src] = np.inf
        dst = int(np.argmin(partner_vals))

        # compute delta
        delta = step * seq[src]
        if delta < 1e-12:
            # No mass to move, reduce step or break
            step *= 0.5
            it += 1
            continue

        # maintain non-negative
        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # compute new conv efficiently via FFT
        new_conv = compute_full_conv(new_seq)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score - 1e-9:
            seq = new_seq
            conv = new_conv
            cur_max = new_max
            best_seq = new_seq.copy()
            best_score = new_score
```

assistantfinal**Solution Overview**

The evaluation

```
score(a) = 2·n·max(conv(a,a)) / (sum(a)²)
```

is *scale‑invariant*: multiplying the whole vector by a constant does not change the
score.  Hence we may (and do) keep the vector normalised to `sum(a)=1` and minimise  

```
f(a) = 2·n·max(conv(a,a))        (  a ≥ 0 ,  sum(a)=1 )
```

`conv(a,a)` is the discrete auto‑convolution.  For a smooth, U‑shaped density the
maximum of the convolution is small – the continuous arcsine density
`Beta(½,½)` achieves the classic bound `π/2 ≈ 1.5708`.  For a finite discrete
length the constant can be a little larger, therefore we search for a better
discrete distribution.

The optimiser works in two stages  

1. **global sampling** – a small pool of well‑known families (arcsine,
   uniform, symmetric Beta distributions, random Dirichlet vectors) is
   evaluated fast with an FFT based convolution.
2. **local refinement** – starting from the best vector a deterministic
   hill‑climbing move is performed repeatedly:

   * find the index `i*` where the convolution attains its current maximum,
   * locate the component `src` that contributes most to this maximum
     (`src` maximises `a[src]·a[i*‑src]`),
   * choose a destination `dst` where the partner entry
     `a[i*‑dst]` is *as small as possible* (preferably outside the support of
     `i*` so that the move does **not** increase the current maximum),
   * move a modest fraction `δ = step·a[src]` from `src` to `dst`,
   * accept the move only if the new maximum decreases (otherwise the step
     size is reduced).

   The step size is adapted during the run; the loop stops when the time
   budget expires.

All heavy work is done with NumPy / FFT – a full convolution costs only
`O(n log n)` and is fast enough for thousands of iterations on a vector with
a few thousand entries.

The final vector is normalised again (to avoid tiny rounding errors) and
returned as a plain Python list; the provided `evaluate_sequence` will compute
the exact score.

```python
"""
propose_candidate
=================

Search for a non‑negative step‑function (list of floats) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of

1. a quick global sweep over several families (arcsine, uniform, Beta(α,α)
   for many α, and a few random Dirichlet vectors);
2. a deterministic hill‑climbing phase that repeatedly moves a small amount
   of mass from the entry that contributes most to the current maximal
   convolution value to an entry that contributes least (or not at all) to
   that same convolution entry.

Fast convolution is performed with an FFT (O(n log n)).  The whole search is
time‑limited by the supplied ``budget_s`` argument (default 1000 s).  The
returned list is guaranteed to contain only non‑negative numbers; its
evaluation is logged for diagnostics.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures, no lambdas
# ----------------------------------------------------------------------


def _fft_max_conv(seq: np.ndarray, size_fft: int) -> float:
    """
    Return max(conv(seq,seq)) using an FFT of length ``size_fft``.
    ``seq`` is assumed to be 1‑D, real and already normalised (sum = 1).
    """
    ft = np.fft.rfft(seq, size_fft)
    conv = np.fft.irfft(ft * ft, size_fft)[: 2 * len(seq) - 1]
    # tiny negative values can appear because of rounding – clip them
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def _full_fft_conv(seq: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Compute the full autocorrelation (conv(seq,seq)) via FFT.
    Result length is 2·len(seq)‑1.
    """
    ft = np.fft.rfft(seq, size_fft)
    conv = np.fft.irfft(ft * ft, size_fft)[: 2 * len(seq) - 1]
    return np.maximum(conv, 0.0)


def _beta_sequence(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the discrete grid { (k+0.5)/n }_{k=0}^{n‑1},
    normalised to sum 1.
    """
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    # pdf ∝ t^{α‑1}(1‑t)^{α‑1}
    pdf = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0:               # safety – should not happen
        pdf[:] = 1.0
        pdf_sum = n
    pdf /= pdf_sum
    return pdf


def _initial_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine coefficients a₀=1, a_{k+1}=a_k·((2k+1)(2k+2))/(4(k+1)²)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2.0 * k - 1.0) * (2.0 * k)) / (4.0 * k * k)
    a /= a.sum()
    return a


def _evaluate_fast(seq: np.ndarray, n: int, size_fft: int) -> float:
    """
    Fast version of evaluate_sequence for a normalised sequence (sum = 1):
    returns 2·n·max(conv(seq,seq)).
    """
    return 2.0 * n * _fft_max_conv(seq, size_fft)


def _hill_climb(start_seq: np.ndarray,
                n: int,
                size_fft: int,
                deadline: float) -> np.ndarray:
    """
    Deterministic hill‑climbing starting from ``start_seq``.
    The algorithm repeatedly moves a fraction of mass from the entry that
    contributes most to the current maximal convolution value to an entry
    that contributes least (preferably not at all) to that same convolution
    entry.  The step size is adapted automatically.
    """
    seq = start_seq.copy()
    seq = np.maximum(seq, 0.0)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        seq[:] = 1.0 / n
    else:
        seq /= seq_sum                     # keep sum == 1

    # pre‑compute FFT length (next power of two ≥ 2·n‑1)
    conv_full = _full_fft_conv(seq, size_fft)
    cur_max = float(conv_full.max())
    cur_score = 2.0 * n * cur_max

    best_seq = seq.copy()
    best_score = cur_score

    # initial step – proportion of the source entry that will be moved
    step = 0.08
    it = 0
    max_iter = 200_000                 # hard cap, real stop is time‑limit

    while time.time() < deadline and it < max_iter:
        # --------------------------------------------------------------
        # 1. locate the index where the convolution attains its maximum
        # --------------------------------------------------------------
        idx = int(np.argmax(conv_full))

        # --------------------------------------------------------------
        # 2. identify the source entry: the one that contributes most to
        #    conv[idx] (i.e. maximises a[k]·a[idx‑k])
        # --------------------------------------------------------------
        low = max(0, idx - (n - 1))
        high = min(idx, n - 1)
        ks = np.arange(low, high + 1, dtype=np.int64)
        contrib = seq[ks] * seq[idx - ks]
        src = ks[int(np.argmax(contrib))]

        # --------------------------------------------------------------
        # 3. choose a destination entry that has the smallest partner value
        #    for the same convolution index (ideally zero, i.e. outside the
        #    overlap region of idx).  This guarantees that moving mass from
        #    src to dst does not increase conv[idx].
        # --------------------------------------------------------------
        i_minus = idx - np.arange(n, dtype=np.int64)
        mask = (i_minus >= 0) & (i_minus < n)
        partner = np.empty(n, dtype=np.float64)
        partner.fill(0.0)                     # outside the range → partner = 0
        partner[mask] = seq[i_minus[mask]]
        partner[src] = np.inf                 # src cannot be dst
        dst = int(np.argmin(partner))

        # --------------------------------------------------------------
        # 4. move a small amount of mass
        # --------------------------------------------------------------
        delta = step * seq[src]
        if delta <= 0.0:
            # source exhausted – shrink step and continue
            step *= 0.5
            it += 1
            continue

        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # safeguard against tiny negative values caused by rounding
        if new_seq[src] < 0.0:
            it += 1
            continue

        # --------------------------------------------------------------
        # 5. evaluate the proposal (full convolution via FFT)
        # --------------------------------------------------------------
        new_conv = _full_fft_conv(new_seq, size_fft)
        new_max = float(new_conv.max())
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # 6. accept if the score improves, otherwise shrink the step size
        # --------------------------------------------------------------
        if new_score < best_score - 1e-12:
            seq = new_seq
            conv_full = new_conv
            cur_max = new_max
            best_seq = new_seq.copy()
            best_score = new_score
            # a successful move – gently increase step size again
            step = min(step * 1.1, 0.5)
            print(f"[hill] it={it:05d} score={best_score:.6f}")
        else:
            # no improvement – become more conservative
            step *= 0.7

        it += 1

    # final normalisation (protect against accumulated rounding errors)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / n
    else:
        best_seq /= total
    return best_seq

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) yields a low value
    for `evaluate_sequence`.  The function respects the time budget, prints
    progress information, and uses the global variable ``height_sequence_1``
    (if present) only for the length hint.
    """
    start_time = time.time()
    # leave a small safety margin for the caller
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # 1. decide on the length of the sequence
    # ------------------------------------------------------------------
    np.random.seed(seed)
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)        # type: ignore[name-defined]
        except Exception:
            n = 6000                         # a comfortable default

    # ------------------------------------------------------------------
    # 2. pre‑compute the FFT size (next power of two >= 2·n‑1)
    # ------------------------------------------------------------------
    size_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 3. keep the current best candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def _consider(cand: np.ndarray, label: str):
        nonlocal best_seq, best_score
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            return
        cand = cand / cand.sum()
        sc = _evaluate_fast(cand, n, size_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{label}] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. a handful of well‑known families
    # ------------------------------------------------------------------
    # 4.1  arcsine (the classic construction)
    _consider(_initial_arcsine(n), "arcsine")

    # 4.2  uniform distribution
    _consider(np.full(n, 1.0 / n, dtype=np.float64), "uniform")

    # 4.3  the global sequence if it exists and has the correct length
    try:
        h1 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if h1.shape[0] == n:
            _consider(h1, "global")
    except Exception:
        pass

    # 4.4  symmetric Beta(α,α) for a range of α around the arcsine exponent 0.5
    alphas = np.concatenate((
        np.linspace(0.35, 0.65, 31),                 # fine grid around 0.5
        np.linspace(0.10, 0.30, 11),
        np.linspace(0.70, 1.50, 17)
    ))
    for a in alphas:
        if time.time() >= deadline:
            break
        _consider(_beta_sequence(a, n), f"beta_{a:.3f}")

    # 4.5  a few random Dirichlet draws (purely exploratory)
    rng = np.random.default_rng(seed + 12345)
    for _ in range(30):
        if time.time() >= deadline:
            break
        _consider(rng.dirichlet(np.ones(n)), "random")

    # ------------------------------------------------------------------
    # 5. local refinement – hill climbing from the current best sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        # extremely unlikely, but fall back to uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # hill climb respects the remaining time
    remaining_deadline = deadline
    refined_seq = _hill_climb(best_seq, n, size_fft, remaining_deadline)

    # ------------------------------------------------------------------
    # 6. final evaluation with the official function (only for logging)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(refined_seq.tolist())
    print(f"[final] length={n}  score={final_score:.6f}")

    return refined_seq.tolist()
```
