"""
Propose a candidate step‑function (non‑negative heights) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The following implementation explores the two‑parameter family

    a_i = (i + γ) ** (‑p) , i = 0 … n‑1

where the exponent *p* is allowed to be > 0.5.  For p>0.5 the dominant
term of the convolution is the “central’’ value  
    D(p,γ) = Σ a_i·a_{n‑1‑i}
instead of the edge term a_0².  When γ is large enough a_0² ≤ D and the
score depends only on *p*:

    score(p) = 2·(1‑p)²·B(1‑p,1‑p)          (B = beta‑function)

The minimum of this analytic bound is attained near p≈0.65 giving a
theoretical score ≈1.35, well below the current best ≈1.57.

The search therefore consists of:
  1) a coarse grid over (p,γ) – p in [0.55,0.85] and γ on a
     logarithmic scale up to 30;
  2) a focused random refinement around the best parameters found.
The whole routine respects a 1000 s wall‑time budget and returns the
best sequence discovered.
"""

import time
import numpy as np

# evaluate_sequence is provided in the global namespace.


def _make_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return the power‑law sequence a_i = (i + γ) ** (‑p)."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def propose_candidate(seed: int = 123, budget_s: float = 1000) -> list[float]:
    """Search for a low‑score step function.

    The function respects a hard time limit (`budget_s`).  It returns a list
    of heights (length 3428) that yields a small value of
    ``evaluate_sequence()``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # safety margin

    # Length used in the reference solution – large enough to exhibit
    # the asymptotic behaviour but still cheap to evaluate.
    n = 3428

    best_seq = None
    best_score = float("inf")
    best_p = None
    best_gamma = None

    # ------------------------------------------------------------------
    # 1) Coarse grid search
    # ------------------------------------------------------------------
    # p slightly above 0.5 (the point where the two contributions to the
    # convolution balance) up to 0.85.
    p_vals = np.linspace(0.55, 0.85, 31)          # 31 values
    # γ on a log‑scale – the analytic bound suggests γ≈(n^{(2p‑1)/(2p)}) is
    # sufficient to move the max from a₀² to the central term.
    gamma_vals = np.logspace(np.log10(0.2), np.log10(30.0), 30)

    for p in p_vals:
        for gamma in gamma_vals:
            if time.time() > deadline:
                break
            seq = _make_sequence(n, p, gamma)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
                best_gamma = gamma
                print(f"[grid] new best p={p:.5f}, γ={gamma:.5f}, score={score:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Random local refinement around the best pair found
    # ------------------------------------------------------------------
    if best_seq is None:          # very unlikely, but keep a safe fallback
        best_p, best_gamma = 0.65, 2.0
        best_seq = _make_sequence(n, best_p, best_gamma)
        best_score = evaluate_sequence(best_seq.tolist())

    # refinement parameters

    for _ in range(refinement_iters):
        if time.time() > deadline:

    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} with p={best_p:.5f}, γ={best_gamma:.5f}")
