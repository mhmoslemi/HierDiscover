def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np
    import time
    import math
    from numpy.fft import rfft, irfft
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # Choose length
    try:
        n = len(height_sequence_1)
        print(f"Using length from existing sequence: n={n}")
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        init_seq = init_seq.clip(min=0)   # ensure nonnegative
    except NameError:
        n = 4000
        print(f"No prior sequence found. Using n={n}")
        # central binomial
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (2.0*i-1.0) / (2.0*i)
        init_seq = a

    # Normalise sum to 1
    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0:
        init_seq = np.full_like(init_seq, 1.0/n)
    init_seq = init_seq / init_seq.sum()

    # initialise parameters for Adam
    u = np.log(init_seq + 1e-12)  # log-odds
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # learning rate

    best_a = init_seq.copy()
    best_score = 2.0 * n * np.max(full_convolution_fft(best_a))
    best_iter = 0
    iteration = 0

    # pre compute fft size for convolution of length 2n-1
    size_conv = 1 << ((2*n-1).bit_length())
    size_corr = 1 << ((3*n-2).bit_length())  # for correlation length 3n-2

    while time.time() < deadline:
        # softmax
        u_max = np.max(u)
        exp_u = np.exp(u - u_max)
        a = exp_u / exp_u.sum()

        # compute convolution a*a via FFT
        fa = rfft(a, size_conv)
        conv = irfft(fa * fa, size_conv)[:2*n-1]

        # compute smooth max
        # schedule beta
        beta = 5.0 + min(iteration/500.0, 25.0)  # increases up to 30
        # compute stabilized exponentials
        z = beta * conv
        z_max = np.max(z)
        expz = np.exp(z - z_max)
        sum_expz = expz.sum()
        weights = expz / sum_expz

        # compute gradient wrt a via correlation
        # correlation of weights (len 2n-1) with a_rev
        ar = a[::-1]
        fw = rfft(weights, size_corr)
        far = rfft(ar, size_corr)
        corr = irfft(fw * far, size_corr)[: (2*n-1) + n - 1]  # length = 3n-2
        # slice grad_a
        grad_a = 2.0 * corr[n-1 : n-1 + n]

        # gradient w.r.t u (softmax parameterization)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate real metric every few iterations
        if iteration % 100 == 0 or time.time() > deadline:
            max_conv = conv.max()
            score = 2.0 * n * max_conv
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_iter = iteration
                print(f"Iter {iteration:6d}: new best score {best_score:.6f}")

        # occasional random perturbation to avoid stagnation
        if iteration % 2000 == 0:
            # small jitter in u
            jitter = np.random.randn(n) * 0.01
            u += jitter

    print(f"Finished after {iteration} iterations. Best score {best_score:.6f} (found at iter {best_iter})")
    return best_a.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works directly on the probability simplex by parameterising
the sequence a with a soft‑max of a real vector u (a_i = exp(u_i)/∑exp).
We minimise a smooth approximation of the peak of the self‑convolution:
    φβ(a) = (1/β)·log Σ_k exp(β·(a*a)_k)   (β → ∞ gives the actual max).

Gradient of φβ can be written in closed form using a correlation of the
soft‑max weights with the current a.  An Adam optimiser (adaptive moments)
updates u, while a small learning‑rate schedule and occasional random
perturbations avoid plateaus.  The best sequence (according to the true
metric) encountered during the run is returned.

If a previously discovered sequence `height_sequence_1` is available it is
used as the starting point; otherwise the central‑binomial coefficients
(the Taylor series of (1‑z)⁻¹ᐟ²) are employed.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper functions (top‑level, no closures)
# -------------------------------------------------------------


def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    u_max = np.max(u)
    exp_u = np.exp(u - u_max)
    return exp_u / exp_u.sum()


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    `size` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * len(a) - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray, size: int) -> np.ndarray:
    """
    Correlation of the weight vector w (length 2·n‑1) with a.
    Returns the full discrete correlation of length 3·n‑2.
    Implemented as conv(w, a[::-1]).
    """
    ar = a[::-1]                     # reversed a
    fw = np.fft.rfft(w, size)
    far = np.fft.rfft(ar, size)
    corr = np.fft.irfft(fw * far, size)[: len(w) + len(a) - 1]
    return corr


# -------------------------------------------------------------
# Main optimisation routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a probability distribution)
    that (hopefully) yields a low evaluation score.
    The routine runs up to `budget_s` seconds, using a smoothed‑max
    Adam optimisation on the self‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ---------------------------------------------------------
    # 1. Choose a length and an initial sequence
    # ---------------------------------------------------------
    try:
        # Use the previously found construction if it exists
        n = len(height_sequence_1)               # type: ignore[name-defined]
        print(f"[init] Using existing sequence of length n={n}.")
        init_seq = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
    except Exception:
        # Fallback: central‑binomial coefficients (known good start)
        n = 4000
        print(f"[init] No previous sequence – using n={n} with central binomial start.")
        init_seq = np.empty(n, dtype=np.float64)
        init_seq[0] = 1.0
        for i in range(1, n):
            init_seq[i] = init_seq[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    # -----------------------------------------------------------------
    # 2. Normalise to the probability simplex (sum = 1, a_i ≥ 0)
    # -----------------------------------------------------------------
    init_seq = np.maximum(init_seq, 0.0)
    s = init_seq.sum()
    if s <= 0.0:
        init_seq = np.full_like(init_seq, 1.0 / n)
    else:
        init_seq = init_seq / s

    # -----------------------------------------------------------------
    # 3. Adam optimiser on the soft‑max parametrisation
    # -----------------------------------------------------------------
    # u is the unconstrained log‑parameter; a = softmax(u)
    u = np.log(init_seq + 1e-16)          # avoid log(0)
    m = np.zeros_like(u)                 # first moment
    v = np.zeros_like(u)                 # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                             # base learning rate

    # Pre‑compute FFT sizes (powers of two) for speed
    size_conv = _next_pow2(2 * n - 1)
    size_corr = _next_pow2(3 * n - 2)

    best_a = init_seq.copy()
    conv_best = _full_convolution_fft(best_a, size_conv)
    best_score = 2.0 * n * conv_best.max()
    best_iter = 0

    it = 0
    while time.time() < deadline:
        it += 1

        # -------------------------------------------------
        # a = soft‑max(u)  (feasible point on the simplex)
        # -------------------------------------------------
        a = _softmax(u)

        # -------------------------------------------------
        # Self‑convolution a*a
        # -------------------------------------------------
        conv = _full_convolution_fft(a, size_conv)

        # -------------------------------------------------
        # Smoothed max (log‑sum‑exp) with a gradually increasing β
        # -------------------------------------------------
        beta = 5.0 + min(it / 500.0, 25.0)          # β grows from 5 to ~30
        z = beta * conv
        z_max = np.max(z)
        expz = np.exp(z - z_max)                    # stabilised exponentials
        w = expz / expz.sum()                       # softmax weights, Σw = 1

        # -------------------------------------------------
        # Gradient of the smoothed max w.r.t. a:
        #   grad_a = 2 * (w ⋆ a)   where ⋆ is correlation
        # -------------------------------------------------
        corr = _correlation_fft(w, a, size_corr)    # length 3n‑2
        grad_a = 2.0 * corr[n - 1 : n - 1 + n]       # slice of length n

        # -------------------------------------------------
        # Chain rule through the soft‑max parametrisation
        #   ∂φ/∂u_i = a_i * (grad_a_i - ⟨a,grad_a⟩)
        # -------------------------------------------------
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # -------------------------------------------------
        # Adam update on u
        # -------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # -------------------------------------------------
        # Keep the best true score seen so far (computed via the exact max)
        # -------------------------------------------------
        if it % 100 == 0:
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_iter = it
                print(f"[{it:6d}] new best score {best_score:.6f}")

        # -------------------------------------------------
        # Occasional random kick to avoid stagnation
        # -------------------------------------------------
        if it % 2000 == 0:
            u += np.random.randn(n) * 0.01

    # ---------------------------------------------------------
    # Final tidy‑up: one short projected sub‑gradient refinement
    # (often gives a tiny extra push)
    # ---------------------------------------------------------
    a = best_a.copy()
    eta0 = 0.1
    for sub_it in range(500):
        conv = _full_convolution_fft(a, size_conv)
        max_val = conv.max()
        k_max = int(conv.argmax())
        # sub‑gradient for the true max
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n, k_max + 1)
        j_range = k_max - np.arange(i_start, i_end)
        grad[i_start:i_end] = 2.0 * a[j_range]
        # simple diminishing step
        step = eta0 / math.sqrt(sub_it + 1.0)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        # accept if not worse
        conv_new = _full_convolution_fft(a_new, size_conv)
        if conv_new.max() <= max_val + 1e-12:
            a = a_new
        else:
            # add tiny random jitter instead
            a = np.maximum(a + np.random.randn(n) * 1e-4, 0.0)
            a = a / a.sum()

    final_score = 2.0 * n * _full_convolution_fft(a, size_conv).max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
        print(f"[final] improved by sub‑gradient phase: score {best_score:.6f}")

    print(f"Finished after {it} main iterations (+500 sub‑grad). "
          f"Best score {best_score:.6f} (found at iter {best_iter}).")
    # Return plain Python floats
    return best_a.tolist()
