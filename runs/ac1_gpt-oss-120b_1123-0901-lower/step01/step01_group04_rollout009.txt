propose_candidate

Return a non‑negative step‑function (list of floats) that tries to minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

    # real‑to‑real FFT because a is real‑valued
    A = np.fft.rfft(a, n=fft_len)
    C = A * A
    conv = np.fft.irfft(C, n=fft_len)[: 2 * len(a) - 1]
    # numerical noise may produce tiny negative values; ignore them
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Generate a candidate sequence for the evaluation function.

    Parameters
    ----------
    seed: int
        Random seed for any stochastic component.
    budget_s: float
        Time budget (seconds) for the optimisation loop.
    length: int, optional
        Desired length of the sequence.  If omitted the function tries to
        reuse ``height_sequence_1`` (if it exists) and otherwise defaults
        to 20000 – larger than the previous run to obtain a tighter
        asymptotic bound.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------
    # 1. Determine the length of the sequence.
    # ------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)      # type: ignore[name-defined]
        except Exception:
            n = 20000                       # longer than the previous run

    # ------------------------------------------------------------
    # 2. Build the discrete arcsine distribution (Catalan numbers).
    # ------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence for a_k = C * (2k choose k) / 4**k  (unnormalised)
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)

    # Normalise so that the sum equals 1 (scaling does not affect the score)
    a /= a.sum()

    # ------------------------------------------------------------
    # 3. Helper data for fast convolution (FFT length = next power of two)
    # ------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # smallest power of two ≥ conv_len

    # ------------------------------------------------------------
    # 4. Initial score
    # ------------------------------------------------------------
    max_b = _fft_convolution_max(a, fft_len)
    best_score = 2.0 * n * max_b / (a.sum() ** 2)
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.9f}")

    # ------------------------------------------------------------
    # 5. Projected gradient descent on the active (max‑convolution) index.
    # ------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin
    it = 0
    step_factor = 0.02                       # relative amount moved each step
    no_improve = 0

    while time.time() < deadline:
        # current convolution and its peak
        conv = np.convolve(a, a)               # cheap – np.convolve falls back to FFT for this size
        max_idx = int(conv.argmax())
        max_val = float(conv.max())

        # gradient of max_val w.r.t. a_i  (g_i = 2*a_{max_idx-i} if inside bounds)
        idx = np.arange(n)
        rev = max_idx - idx
        valid = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[rev[valid]]

        # pick source (largest gradient) and destination (smallest gradient)
        src = int(np.argmax(g))
        # avoid selecting src again for destination
        mask = np.arange(n) != src
        if not np.any(mask):
            break
        dst = int(np.argmin(np.where(mask, g, np.inf)))

        # if no beneficial move is possible, try a small random perturbation
        if g[src] <= g[dst] + 1e-15:
            if np.random.rand() < 0.01:
                i, j = np.random.choice(n, 2, replace=False)
                src, dst = (i, j) if a[i] > a[j] else (j, i)
                delta = step_factor * min(a[src], a[dst])
                if delta == 0.0:
                    it += 1
                    continue
            else:
                # no direction can improve the current peak – stop
                break
        else:
            delta = step_factor * a[src]          # move a fraction of the source mass

        # enforce non‑negativity
        if delta > a[src]:
            delta = a[src]

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # evaluate the new candidate
        new_max = _fft_convolution_max(new_a, fft_len)
        new_score = 2.0 * n * new_max / (new_a.sum() ** 2)

        if new_score < best_score:
            a = new_a
            best_seq = new_a.copy()
            best_score = new_score
            no_improve = 0
            # occasional status output
            if it % 500 == 0:
                print(f"[improve] it={it:05d}, src={src}, dst={dst}, "
                      f"delta={delta:.3e}, score={best_score:.9f}")
        else:
            # reject the move; occasionally shrink the step size to escape plateaus
            no_improve += 1
            if no_improve >= 2000:
                step_factor *= 0.5
                no_improve = 0

        it += 1

    print(f"[final] iterations={it}, best score={best_score:.9f}")
    # ------------------------------------------------------------
    # 6. Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------
    return best_seq.tolist()
