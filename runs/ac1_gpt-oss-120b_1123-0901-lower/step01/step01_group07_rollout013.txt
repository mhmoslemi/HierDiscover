"""
Propose a candidate step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Broad random search** – 4 families of simple analytic shapes are
   sampled (pure power‑law, power‑law with exponential taper,
   piece‑wise power‑law, and early‑region scaling).  For each randomly
   chosen length `n` a sequence is built, normalised (sum = 1) and
   evaluated with `evaluate_sequence`.  The best sequence found so far is
   kept.

2. **Gradient‑descent style polishing** – starting from the best
   sequence we repeatedly identify the lag `k` where the autocorrelation
   `conv(a, a)` is maximal.  The contribution of each coefficient to this
   maximum is `a[k‑i]`.  We shrink the coefficients proportionally to
   these contributions (`a ← a·exp(‑α·grad)`) and renormalise.
   This simple descent quickly flattens the autocorrelation peak.

3. **Local parametric refinement** – the parameters of the family that
   produced the best sequence are perturbed with a small Gaussian noise
   (respecting the original bounds).  Whenever an improvement is found
   the parameters (and therefore the sequence) are updated.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence discovered as a plain Python list.
"""

import time

# Parameter specifications for the four families
# ----------------------------------------------------------------------
PARAM_RANGES = {
    # (low, high) for each parameter in order
    "A": [(0.40, 0.70), (0.10, 5.00)],                     # p, gamma
    "B": [(0.40, 0.70), (0.10, 5.00), (0.0, 0.02)],       # p, gamma, lambda
    "C": [(0.50, 1.20), (0.30, 0.60), (0.10, 5.00), (0.10, 0.90)],  # p1, p2, gamma, cut_frac
    "D": [(0.40, 0.70), (0.10, 5.00), (0.01, 0.15), (0.05, 0.90)],  # p, gamma, k_frac, scale
}

def _seq_power_law(params, n):
    p, gamma = params
    i = np.arange(n, dtype=float)
    seq = (i + gamma) ** (-p)
    return seq

def _seq_power_law_exp(params, n):
    p, gamma, lam = params
    i = np.arange(n, dtype=float)
    seq = (i + gamma) ** (-p) * np.exp(-lam * i)
    return seq

def _seq_piecewise(params, n):
    p1, p2, gamma, cut_frac = params
    cut = int(np.clip(round(cut_frac * n), 1, n - 1))
    i = np.arange(n, dtype=float)
    seq = np.empty(n, dtype=float)
    seq[:cut] = (i[:cut] + gamma) ** (-p1)
    seq[cut:] = (i[cut:] + gamma) ** (-p2)
    return seq

    k = int(np.clip(round(k_frac * n), 1, n - 1))
    i = np.arange(n, dtype=float)

    seq[:k] *= scale
    return seq

    "B": _seq_power_law_exp,
    "C": _seq_piecewise,
    "D": _seq_early_scaling,
}

def _normalise(seq: np.ndarray) -> np.ndarray:
    """Scale to L1 = 1 while protecting against a zero sum."""
    total = seq.sum()
    if total <= 0.0:
        return None
    return seq / total


def _random_params(family: str, rng: np.random.Generator):
    """Draw a random parameter vector for the given family."""
    low_high = PARAM_RANGES[family]
    return [rng.uniform(lo, hi) for lo, hi in low_high]


def _perturb_params(params, ranges, rng, scale=0.05):
    """Gaussian perturbation of parameters inside the allowed ranges."""
    new = []
    for val, (lo, hi) in zip(params, ranges):
        sigma = scale * (hi - lo)
        v = val + rng.normal(0.0, sigma)
        v = max(lo, min(hi, v))
        new.append(v)
    return new


def _build_sequence(family: str, params, n: int):
    """Construct and normalise a sequence for the given family."""
    seq = SEQ_BUILDERS[family](params, n)
    seq = np.clip(seq, 0.0, None)          # safety, though generator never goes negative
    seq = _normalise(seq)
    return seq


# ----------------------------------------------------------------------
# Gradient‑descent style polishing of a concrete sequence
# ----------------------------------------------------------------------
def _polish_sequence(seq: np.ndarray, time_deadline: float, start_score: float):
    """
    Perform a simple fixed‑step descent that reduces the peak of the
    aperiodic autocorrelation.  The routine stops when time runs out or
    after a reasonable number of iterations.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = start_score
    alpha = 0.25               # step size for the exponential update

    # Estimate a generous number of iterations but respect the deadline
    max_iter = 3000
    for it in range(max_iter):
        now = time.time()
        if now >= time_deadline:
            break

        conv = np.convolve(seq, seq)            # length 2n‑1
        k_max = int(np.argmax(conv))

        # Gradient of the maximal lag w.r.t. each coefficient:
        #   d/d a_i  sum_j a_j a_{k_max-j} = a_{k_max-i}  (if index valid)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(seq)
        grad[mask] = seq[idx[mask]]

        # Exponential decay step guarantees positivity
        seq = seq * np.exp(-alpha * grad)

        # Re‑normalise to keep the scale fixed (doesn't affect the score)
        total = seq.sum()
        if total > 0.0:
            seq /= total

        # Periodically evaluate – cheap enough to do every few steps
        if it % 20 == 0:
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                # tighten the step if we are already very good
                alpha = max(0.05, alpha * 0.9)

    return best_seq, best_score

    print(f"Finished search – best score {best_score:.6f} (length {len(result)})")
    return result
