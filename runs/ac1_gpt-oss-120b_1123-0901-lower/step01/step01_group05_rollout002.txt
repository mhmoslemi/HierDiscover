"""
Propose a candidate step‐function sequence.

The algorithm starts from a known good shape (inverse‑square‑root) and
iteratively reduces the maximal value of the self‑convolution.
For a fixed ℓ₁‑norm the evaluation score is proportional to that maximal
convolution value, therefore we minimise it.

Each iteration solves a linear programme

    maximise   Σ g_i
    subject to  g_i ≥ 0
                (f * g)_k ≤ M   for all k

where M = max(f * f) of the current sequence f.
The LP yields a vector g that carries as much mass as possible without
exceeding the current bound M.  After scaling g to the same ℓ₁‑norm as f we
mix f ← (1–t)·f + t·g with a small step size t; the new self‑convolution
cannot exceed M and usually becomes smaller.  The process repeats until the
time budget is exhausted.  A cheap random local search is then used for
fine‑tuning.

If cvxpy is unavailable the code falls back to a simple stochastic refinement
similar to the reference implementation.
"""

def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """
    Returns a NumPy array a_0,…,a_{n‑1} with
        a_i = binom(2*i,i) / 4**i .
    The recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) is used to avoid overflow.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence using i (the new index)
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def conv_matrix_sparse(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A such that A @ g == np.convolve(f, g).
    f is a 1‑D array of length n; the resulting matrix has shape
    (2*n-1) x n.  Each column i contains a copy of f shifted down by i.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Number of non‑zero entries = n * n
    data = np.tile(f, n)                     # (n*n,)  – the values
    # Build row and column index arrays

        col_inds[pos:pos + n] = i
        pos += n
    A = coo_matrix((data, (row_inds, col_inds)), shape=(rows, n)).tocsr()
    return A

            A = conv_matrix_sparse(f)

            # Linear programme: maximise Σ g_i  s.t.  g ≥ 0, A·g ≤ M.
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M]
            objective = cp.Maximize(cp.sum(g))
            prob = cp.Problem(objective, constraints)

            try:
                # ECOS handles moderate‑size dense LPs well; we keep a modest
                # iteration limit to stay within the time budget.
                prob.solve(solver=cp.ECOS, max_iters=5000, warm_start=True,
                           verbose=False)
            except Exception as exc:  # pragma: no cover
                print(f"[lp] iteration {iteration} failed ({exc}) – falling back.")
                break

            if prob.status not in ("optimal", "optimal_inaccurate"):
                print(f"[lp] iteration {iteration} status {prob.status} – stop.")
                break

            g_opt = np.asarray(g.value).flatten()
            sum_g = np.sum(g_opt)
            if sum_g <= 0.0:
                print(f"[lp] iteration {iteration} produced zero mass – stop.")
                break

            # Scale g to the same ℓ₁‑norm as f.
            g_scaled = g_opt * (sum_f / sum_g)

            # -----------------------------------------------------------------
            # 3. Mix f with the scaled g.  A simple back‑tracking line search on
            #    the mixing weight t is employed.
            # -----------------------------------------------------------------
            mixed = False
            t = 0.5
            while t > 1e-4:
                f_new = (1.0 - t) * f + t * g_scaled
                M_new = np.max(np.convolve(f_new, f_new))
                if M_new < M - 1e-12:          # strict improvement
                    f = f_new
                    best_seq = f.copy()
                    best_M = M_new
                    best_score = evaluate(best_seq.tolist())
                    print(f"[iter {iteration}] t={t:.4f} → M={M_new:.6f} score={best_score:.6f}")
                    mixed = True
                    break
                t *= 0.5

            if not mixed:
                # Could not improve M with this g – abort the LP loop.
                print(f"[iter {iteration}] no improving step size – terminate LP phase.")
                break
    else:
        print("[info] cvxpy not available – skipping LP phase.")

    # --------------------------------------------------------------
    # 4. Final cheap stochastic refinement (time‑limited).
    # --------------------------------------------------------------
    # Remaining time (leave a small safety margin).
    remaining = start_time + budget_s - time.time() - 0.2
    if remaining > 0:
        deadline = time.time() + remaining
        n = best_seq.shape[0]
        while time.time() < deadline:
            # Random multiplicative perturbation of a single coefficient.
            i = rng.integers(n)
            factor = math.exp(rng.normal(0.0, 0.015))
            cand = best_seq.copy()
            cand[i] = max(0.0, cand[i] * factor)

            # Re‑normalise ℓ₁‑norm (scale invariance makes this optional,
            # but it keeps the numbers in a similar range).
            cand = cand * (sum_f / np.sum(cand))

            cand_score = evaluate(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand
                best_M = np.max(np.convolve(best_seq, best_seq))
                print(f"[local] improved score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found.
    # --------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} with n={len(best_seq)}")
    return best_seq.tolist()
