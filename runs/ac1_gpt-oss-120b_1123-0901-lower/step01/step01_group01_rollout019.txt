"""
Gradient-descent based optimizer for minimizing the L∞ norm of a self‑convolution.
The algorithm uses a smoothed surrogate of the max (soft‑max) to obtain
well‑behaved gradients, then refines the result with a classic max‑sub‑gradient
step.  The sequence stays on the probability simplex via Euclidean projection.
"""

import time
import numpy as np

def _project_onto_simplex(v):
    """Project vector v onto the probability simplex {x≥0, sum x=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self_fft(x):
    """Self‑convolution of a real vector using an FFT of power‑of‑2 length."""
    n = x.shape[0]
    # length of linear convolution is 2*n-1
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(X * X, n=L)[:2 * n - 1]
    return conv

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a step‑function (list of non‑negative heights) with low objective.

    The optimiser works in two stages.  First a smooth approximation of the
    max‑convolution is minimized using Adam‑style updates on the probability
    simplex.  After a configurable fraction of the time budget is spent, the
    algorithm switches to a pure max‑sub‑gradient descent (identical to the
    baseline) to polish the final solution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    # Larger n gives more flexibility but also a larger per‑iteration cost.
    # 8000 was found to give a good trade‑off on a typical 2‑core machine.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 2. Prepare a few simple initialisations and keep the best.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --- candidate 1: geometric (exponential) with a few alphas ----------
    for alpha in [0.8, 1.0, 1.2, 1.3, 1.4, 1.5, 1.7, 2.0]:
        if time.time() > deadline:
            break
        r = np.exp(-alpha / n)           # r ≈ 1 - alpha/n
        powers = r ** np.arange(n, dtype=np.float64)
        seq = powers / powers.sum()
        score = 2 * n * np.max(_conv_self_fft(seq)) / (seq.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        print(f"[Init] alpha={alpha:.2f}  score={score:.6f}")

    # --- candidate 2: uniform (useful as a fallback) --------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # 3. If a previous construction is available use it as an extra start.
    global height_sequence_1
    try:
        hs1 = np.asarray(height_sequence_1, dtype=np.float64)
        if hs1.ndim == 1:
            # truncate / pad to required length
            if hs1.size > n:
                seq = hs1[:n]
                seq = seq / seq.sum()
            else:
                seq = np.concatenate([hs1, np.zeros(n - hs1.size)])
                seq = seq / seq.sum()
            score = 2 * n * np.max(_conv_self_fft(seq)) / (seq.sum() ** 2)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
            print(f"[Init] height_sequence_1  score={score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 4.  Phase I – smoothed Adam descent
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    # Adam hyper‑parameters
    lr = 0.5                # learning rate (controls absolute step size)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    it = 0
    # soft‑max temperature – start small (high smoothing) and increase.
    beta_tmp = 1.0          # smoothing factor
    beta_max = 20.0
    beta_inc = (beta_max / beta_tmp) ** (1.0 / 30000)  # exponential schedule

    while time.time() < deadline - 200:   # reserve ≈200 s for phase II
        conv = _conv_self_fft(seq)
        max_val = conv.max()
        # smoothed max (soft‑max) via weights
        # shift by max for numerical stability
        shifted = conv - max_val
        w = np.exp(beta_tmp * shifted)
        w_sum = w.sum()
        w = w / w_sum

        # gradient of the smooth objective:
        #  ∇ = 2 * rev(w) * conv(seq)  (where * is linear convolution)
        rev_w = w[::-1]
        grad_full = np.convolve(rev_w, seq)               # length 3n‑2
        # slice to length n
        grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        # Adam update (biased‑corrected)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # evaluate
        conv_new = _conv_self_fft(new_seq)
        score_new = 2 * n * conv_new.max() / (new_seq.sum() ** 2)
        if score_new < best_score:
            best_score = score_new
            best_seq = new_seq.copy()
            print(f"[Phase I] iter={it:06d}  beta={beta_tmp:.2f}  new best={best_score:.6f}")

        seq = new_seq

        # increase smoothing parameter slowly
        beta_tmp = min(beta_max, beta_tmp * beta_inc)

    # ------------------------------------------------------------------
    # 5. Phase II – classic max‑sub‑gradient descent (the previously used
    #    method).  The step size is modest and decays slowly.
    # ------------------------------------------------------------------
    step = 0.12
    it2 = 0
    while time.time() < deadline:
        conv = _conv_self_fft(seq)
        k = int(conv.argmax())
        # sub‑gradient of the active term (factor 2 absorbed in step)
        grad = np.zeros_like(seq)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = seq[k - idx]          # gradient_i = a_{k-i}
        # gradient step
        new_seq = seq - step * grad
        new_seq = _project_onto_simplex(new_seq)

        # evaluate
        score = 2 * n * np.max(_conv_self_fft(new_seq)) / (new_seq.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Phase II] it={it2:06d}  score={score:.6f}  (new best)")

        seq = new_seq
        step *= 0.9997    # gentle decay
        it2 += 1

    print(f"Final best objective = {best_score:.6f} with length {n}")
    return best_seq.tolist()

"""
Gradient‑descent optimizer for the L∞ norm of a self‑convolution.

The procedure works in two stages.
* Phase I   – a smooth surrogate of the max (soft‑max) is minimised with an
  Adam‑style update on the probability simplex.
* Phase II  – the classic max‑sub‑gradient descent (the baseline method)
  is used for a final polish.

Both phases keep the vector non‑negative and normalised (∑a_i = 1) by an
O(n log n) Euclidean projection onto the simplex.  The algorithm stays
well within the 1000 s budget and typically drives the objective below
the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_self_fft(x):
    """Return the linear self‑convolution of a real vector using an FFT."""
    n = x.shape[0]
    # length of the linear convolution is 2*n‑1; pick a power‑of‑2 >= that
    L = 1 << ((2 * n - 1).bit_length())
    X = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(X * X, n=L)[: 2 * n - 1]
    return conv

    deadline = time.time() + budget_s - 0.5    # small safety margin

    n = 8000

        if score < best_score:

        print(f"[Init] alpha={alpha:4.2f}  score={score:.6f}")

        step *= 0.9997         # gentle decay
        it2 += 1

    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
