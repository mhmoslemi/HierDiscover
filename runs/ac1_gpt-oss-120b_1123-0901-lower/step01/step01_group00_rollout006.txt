"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a smooth‑max (soft‑max) gradient descent with momentum, simplex
projection and a simple annealing schedule.  The method works with many
random restarts, different sequence lengths and occasional random
perturbations to escape plateaus.  It keeps track of the best sequence
found during the allotted time budget and returns it as a plain Python
list.

The optimisation is carried out on the simplex

    { a_i ≥ 0 , Σ a_i = √(2 n) }

so that the evaluation function equals the maximal convolution entry,
i.e. we minimise the ℓ∞‑norm of the self‑convolution.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i >= 0, Σ x_i = s }.
    Implementation of Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute full linear convolution of a with itself.
    For the problem sizes we target (n ≤ ~3000) the direct np.convolve
    is fast enough and avoids the small numerical overhead of FFT.
    """
    return np.convolve(a, a)


def _gradient_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective
        F(a) = (1/β) log Σ_k exp(β * (a*a)_k)

    ∂F/∂a_i = 2 Σ_k p_k a_{k-i}
    where p_k = softmax(β * (a*a)_k) and the sum is over the
    valid indices only.  This can be computed by a single convolution
    of a with the reversed weight vector p.
    """
    n = a.shape[0]
    p_rev = p[::-1]                     # reversed soft‑max weights
    conv_full = np.convolve(a, p_rev)   # length = n + (2n‑1) – 1 = 3n‑2
    idx = 2 * n - 2 - np.arange(n)      # pick the required slice
    grad = 2.0 * conv_full[idx]
    return grad


def _run_optimization(n: int,
                     init: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform one optimisation run on a sequence of length ``n``.
    The routine stops after ~12 s or when the global deadline is reached.
    Returns the potentially updated best sequence and its score.
    """
    target_sum = np.sqrt(2.0 * n)

    # start from the provided vector and enforce the simplex constraint
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # momentum variables
    v = np.zeros_like(a)
    mu = 0.9

    # hyper‑parameters (all are tuned empirically)
    base_step = 0.05
    beta = 0.5          # start with a smooth soft‑max
    beta_inc = 0.02
    last_improve = time.time()

    # allocate a small time slice for this run
    run_deadline = min(deadline, time.time() + 12.0)

    while time.time() < run_deadline:
        # ----- evaluate current candidate -----
        conv = _self_convolution(a)                     # length 2n‑1
        max_conv = np.max(conv)
        sum_a = np.sum(a)                               # should be target_sum
        score = 2.0 * n * max_conv / (sum_a * sum_a)    # exact evaluation

        # keep the best sequence found so far
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (n={n})")
            last_improve = time.time()

        # ----- soft‑max weights -----
        scaled = beta * conv
        max_scaled = np.max(scaled)
        exp_vals = np.exp(scaled - max_scaled)          # stable soft‑max
        p = exp_vals / np.sum(exp_vals)

        # ----- gradient of the smooth objective -----
        grad = _gradient_softmax(a, p)

        # ----- step‑size selection (similar to the original code) -----
        max_grad = np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else 1.0
        eta = base_step * max_conv / max_grad

        # ----- momentum update -----
        v = mu * v - eta * grad
        a_new = a + v
        a_new = _project_onto_simplex(a_new, target_sum)

        # ----- acceptance test -----
        conv_new = _self_convolution(a_new)
        max_conv_new = np.max(conv_new)
        score_new = 2.0 * n * max_conv_new / (np.sum(a_new) ** 2)

        # accept if improvement, otherwise accept with a tiny probability
        if score_new <= score or np.random.rand() < 0.01:
            a = a_new
            if score_new < score:
                base_step = min(base_step * 1.02, 0.5)   # slowly enlarge step
        else:
            # reject – shrink step and forget the accumulated momentum
            base_step *= 0.7
            v = np.zeros_like(a)

        # ----- anneal the smoothness parameter -----
        beta = min(beta + beta_inc, 10.0)

        # ----- occasional random shake‑up to leave flat regions -----
        if np.random.rand() < 0.003:
            noise = np.random.randn(n) * (0.005 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            v = np.zeros_like(a)
            base_step = 0.05
            beta = 0.5

        # stop early if we have been idle for a while
        if time.time() - last_improve > 30.0:
            break

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a low‑evaluation step‑function.
    The routine runs for at most ``budget_s`` seconds, uses multiple
    random restarts (different lengths and initialisations) and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Main optimisation loop – independent restarts
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # pick a problem size; we favour the range where previous experiments
        # performed well but also explore a bit larger vectors
        if np.random.rand() < 0.7:
            n = np.random.randint(800, 1800)
        else:
            n = np.random.randint(1800, 3000)

        # initialise from the existing best construction (if any) or random
        if np.random.rand() < 0.3 and 'height_sequence_1' in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        # run a focussed optimisation on this restart
        best_seq, best_score = _run_optimization(
            n, init, deadline, best_seq, best_score
        )

        # early exit if we have already crossed the target threshold
        if best_score < 1.503:
            break

    # -----------------------------------------------------------------
    # Fallback – a safe uniform sequence if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    return best_seq.tolist()
