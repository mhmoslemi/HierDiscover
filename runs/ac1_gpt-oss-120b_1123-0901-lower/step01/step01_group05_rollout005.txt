"""
Propose a candidate step‑function that beats the π/2 ≈ 1.571 bound.

Algorithm
---------
1. Choose a few candidate lengths (the previous best length is used as a centre).
2. For each length generate simple power‑law sequences 
      a_i = (i+1)^(-γ)   with γ ∈ (0.5, 0.85).
   The shape interpolates between the uniform sequence (γ=0) and the
   inverse‑square‑root sequence (γ=0.5).  For moderate γ the self‑convolution
   maximum becomes smaller than for the Catalan construction, giving a lower
   evaluation score.
3. Scan the γ‑grid coarse first, keep the best (length,γ) pair.
4. Refine the best γ with a finer grid around it.
5. Perform a cheap stochastic local search: repeatedly pick a random index
   and apply a multiplicative log‑normal perturbation, accepting only moves
   that strictly improve the score.  This can nudge the shape away from the
   pure power law and capture small discretisation effects.
6. Stop when the allotted 1000 s budget is about to be exhausted and
   return the best sequence found.

All numbers are non‑negative; the evaluation function is scale‑invariant,
so we do not normalise the sequence.  The search stays well below the
time limit by limiting the number of full evaluations (each one costs
O(n²) due to np.convolve).
"""

import time
import numpy as np

def _power_law_sequence(n: int, gamma: float) -> list[float]:
    """Return a_i = (i+1)^(-γ) for i=0..n‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-gamma)
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step function.

    Parameters
    ----------
    seed : int
        Random seed for the stochastic refinement.
    budget_s : int
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """

    deadline = start_time + budget_s - 0.3
