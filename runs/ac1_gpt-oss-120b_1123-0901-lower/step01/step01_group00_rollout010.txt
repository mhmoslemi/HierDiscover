def compute_score(a: np.ndarray) -> float:

    n = len(a)
    sum_a = a.sum()
    if sum_a < 0.01: return inf
    # compute conv via FFT
    # zero pad to next power of two maybe.
    size = 2*n -1
    # compute FFT length sufficiently large (next power of two)
    L = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa*fa, L)[:size]
    max_b = conv.max()
    return 2*n*max_b/(sum_a**2)

def run_gradient_descent(a_init, target_sum, deadline):
    a = a_init.copy()
    best_seq_local = a.copy()
    best_score_local = compute_score(a)
    # gradient descent parameters
    step_factor = 0.05
    while time.time() < deadline:
        # compute convolution and max index(s)
        # conv via FFT for speed, but also compute gradient contributions.
        # Let's compute convolution as conv = np.fft.ifft(fft(a)*fft(a)).real using FFT. But need to compute for specific index only.
        # We'll compute conv via np.fft.rfft as earlier.
        n = len(a)
        L = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa*fa, L)[:2*n-1]
        max_val = conv.max()
        # find all indices with value close to max (within tolerance)
        tolerance = 1e-12 # but for floating we can use relative tolerance: e.g., max - conv > 1e-8 * max
        # We'll get indices where conv >= max_val - 1e-8 * max_val maybe.
        mask = conv >= max_val - 1e-8 * max_val
        indices = np.where(mask)[0]
        # compute gradient as sum over each max index:
        grad = np.zeros_like(a)
        for k in indices:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        # step size: adapt based on max gradient magnitude
        max_grad = np.max(np.abs(grad))
        if max_grad == 0:
            break
        eta = step_factor * max_val / max_grad
        a_new = a - eta * grad
        # project onto simplex
        a_new = project_onto_simplex(a_new, target_sum)
        # evaluate
        score_new = compute_score(a_new)
        if score_new < best_score_local:
            best_score_local = score_new
            best_seq_local = a_new.copy()
        # update a:
        a = a_new
        # optionally adapt step size
        # after certain iterations maybe reduce step

        if time.time() > deadline:
            break

    return best_seq_local, best_score_local

"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation works in three stages:

1. **Parametric search** – generate candidates of the form
   a_i ∝ (i+1)^{p1}·(n-i)^{p2} (a beta‑like shape).  The parameters
   (n, p1, p2) are sampled randomly; each candidate is projected onto the
   simplex Σa_i = √(2n).  The best candidate is kept.

2. **Projected gradient descent** – starting from the best shape, perform
   a few iterations of the classic “move opposite to the gradient of the
   maximal convolution entry” step, always projecting back onto the simplex.
   This refines the candidate and removes small local peaks.

3. **Simple stochastic local search** – a (1+1)‑ES style hill‑climber that
   repeatedly moves a random amount of mass from one coordinate to another.
   Improvements are kept and the global best sequence is updated.

All heavy work (convolution, objective) is done with an FFT‑based
implementation (`compute_score`) which is ~10× faster than `np.convolve`
for the sizes we use (n≈200‑1700).  The routine runs for at most the
provided time budget (default 1000 s) and finally returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s }.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (the scaling invariant objective).
    Returns 2·n·max(conv(a,a)) / (sum(a)^2).
    """
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:                     # same safeguard as evaluate_sequence
        return float('inf')
    # length of linear convolution
    conv_len = 2 * n - 1
    # next power‑of‑two for efficient FFT
    L = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[:conv_len]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def beta_shape(n: int, p1: float, p2: float) -> np.ndarray:
    """
    Return a non‑negative vector of length n with entries proportional to
    (i+1)^{p1}·(n-i)^{p2}, i = 0…n‑1.
    """
    # work in log‑space to avoid overflow for large exponents
    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.log(i) * p1
    right = np.log(n - i + 1) * p2
    a = np.exp(left + right)
    return a


# ------------------------------------------------------------
# Optimisation primitives
# ------------------------------------------------------------

def gradient_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the index attaining the maximal convolution.
    Stops when the deadline is reached.  Returns (best_vector, best_score).
    """
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = project_onto_simplex(a, target_sum)

    best_vec = a.copy()
    best_sc = compute_score(a)

    step_factor = 0.07          # base step size, tweaked experimentally

    while time.time() < deadline:
        # ----- convolution & max index(s) -----
        conv_len = 2 * n - 1
        L = 1 << (conv_len - 1).bit_length()
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, L)[:conv_len]
        max_val = conv.max()
        # treat all indices that are virtually maximal to smooth the gradient
        eps = 1e-12 * max_val
        max_idx = np.where(conv >= max_val - eps)[0]

        # ----- gradient of the selected max‑entry(s) -----
        grad = np.zeros_like(a)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = np.max(np.abs(grad))
        if max_grad == 0:
            break

        eta = step_factor * max_val / max_grad
        a_new = a - eta * grad
        a_new = project_onto_simplex(a_new, target_sum)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_vec = a_new.copy()
            # optional verbose line (can be commented out if too chatty)
            print(f"[{time.time():.2f}] GD improvement → {best_sc:.6f} (n={n})")

        # simple acceptance rule – always move (helps escape flat zones)
        a = a_new

    return best_vec, best_sc


def stochastic_local_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very simple (1+1)‑ES style hill‑climber.  Moves a random amount of mass
    from one coordinate to another and keeps improvements.  Uses a linear
    cooling schedule to occasionally accept uphill moves.
    """
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = project_onto_simplex(a, target_sum)

    best_vec = a.copy()
    best_sc = compute_score(a)

    start = time.time()
    while time.time() < deadline:
        # ----- propose a neighbour -----
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if best_vec[i] == 0.0:
            continue
        # move up to 30 % of the source entry
        delta = best_vec[i] * np.random.rand() * 0.3
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta          # total sum stays constant

        sc = compute_score(cand)

        # linear temperature (start ~0.1·best_sc, end →0)
        elapsed = time.time() - start
        T = max(0.0, 0.1 * best_sc * (1 - elapsed / (deadline - start)))
        accept = sc < best_sc or np.random.rand() < np.exp((best_sc - sc) / max(T, 1e-12))

        if accept:
            best_vec = cand
            best_sc = sc
            print(f"[{time.time():.2f}] Local search → {best_sc:.6f} (n={n})")

    return best_vec, best_sc


# ------------------------------------------------------------
# Main entry point used by the outer harness
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a low‑value step sequence.
    Returns the best sequence (as a plain list) found before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # margin for safety

    best_seq = None
    best_score = float('inf')
    target_sum = None

    # --------------------------------------------------------
    # Stage 1 – random beta‑shapes
    # --------------------------------------------------------
    stage1_deadline = deadline - 15.0        # leave some time for refinement
    while time.time() < stage1_deadline:
        n = np.random.randint(200, 1700)    # length of the candidate
        # sample exponents in a fairly wide range
        p1 = np.random.uniform(-4.0, 4.0)
        p2 = np.random.uniform(-4.0, 4.0)

        a = beta_shape(n, p1, p2)
        # avoid a pathological all‑zero vector
        if a.sum() == 0:
            continue
        # project onto the required simplex (Σa = √(2n))
        target = np.sqrt(2.0 * n)
        a = project_onto_simplex(a, target)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            target_sum = target
            print(f"[{time.time():.2f}] New shape best = {best_score:.6f} "
                  f"(n={n}, p1={p1:.2f}, p2={p2:.2f})")

        # occasionally try a random uniform restart (helps diversity)
        if np.random.rand() < 0.02:
            a = np.random.rand(n)
            a = project_onto_simplex(a, target)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                target_sum = target
                print(f"[{time.time():.2f}] Random restart best = {best_score:.6f} (n={n})")

    # --------------------------------------------------------
    # Stage 2 – gradient descent refinement
    # --------------------------------------------------------
    if best_seq is not None:
        for _ in range(3):                     # a few GD passes
            if time.time() >= deadline:
                break
            gd_deadline = min(deadline, time.time() + 8.0)
            refined, sc = gradient_descent(best_seq.copy(), gd_deadline)
            if sc < best_score:
                best_score = sc
                best_seq = refined.copy()
                target_sum = np.sqrt(2.0 * refined.size)
                print(f"[{time.time():.2f}] GD refined best = {best_score:.6f} (n={refined.size})")

    # --------------------------------------------------------
    # Stage 3 – stochastic local search (tiny hill‑climber)
    # --------------------------------------------------------
    if best_seq is not None:
        ls_deadline = deadline
        refined, sc = stochastic_local_search(best_seq.copy(), ls_deadline)
        if sc < best_score:
            best_score = sc
            best_seq = refined.copy()
            print(f"[{time.time():.2f}] Final local‑search best = {best_score:.6f}")

    # --------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # --------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # Clip to the limits enforced by evaluate_sequence (0…1000) and
    # convert to a plain Python list.
    best_seq = np.clip(best_seq, 0.0, 1000.0).tolist()
    return best_seq
