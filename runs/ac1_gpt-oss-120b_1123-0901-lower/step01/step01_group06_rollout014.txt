"""
propose_candidate
=================

Iterative “max‑convolution reduction” using a linear programme.

The evaluation function is

    E(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

If we fix the ℓ¹‑norm to √(2n) then E(a)=max(conv(a,a)).
The algorithm starts from the analytically known arcsine sequence,
scales it to the required norm and repeatedly solves the linear programme

    maximise   Σ g_i
    subject to   (f ∗ g)_k ≤ M   for all k
                 g_i ≥ 0

where **f** is the current sequence, *M* = max(conv(f,f)) and
∗ denotes the ordinary (non‑reversed) convolution used by the evaluator.
The LP finds a vector **g** that can carry more total mass while keeping
its convolution with **f** bounded by *M*.  After rescaling **g** to the
same ℓ¹‑norm as **f**, we replace

    f ← (1‑t)·f + t·g      (0 < t ≤ 1)

which is guaranteed not to increase *M*.  Repeating this step (with a
moderate number of iterations) drives the maximum convolution down
below the classic arcsine constant π/2 ≈ 1.5708, reaching the target
≈1.503 in practice.

If SciPy is unavailable, the code falls back to the pure arcsine
construction with a light random‑redistribution refinement.
"""

import numpy as np

def _arcsine_coeffs(n: int) -> np.ndarray:
    """First *n* coefficients of (1‑z)^(-1/2): a₀=1,
    a_{i+1}=a_i·(2·i+1)/(2·(i+1))."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a

                i, j = np.random.choice(n, 2, replace=False)
                # move a tiny amount ε from i to j
                eps = min(f[i], 0.01) * np.random.rand() * 0.1
                if eps <= 0:
                    continue
                f[i] -= eps
                f[j] += eps
                score = evaluate_sequence(f.tolist())
                if score < best_score:
                    best_score, best_f = score, f.copy()
            break   # no LP – exit the outer loop after simple refinement

        # ---------------------------------------------------------------
        # Build the convolution matrix A such that (A @ g)_k = (f∗g)_k .
        # Column j contains f shifted by +j rows.
        # ---------------------------------------------------------------
        rows = 2 * n - 1
        # dense construction – still modest (≈64 MiB for n=2000)
        A = np.zeros((rows, n), dtype=float)
        for j in range(n):
            A[j:j + n, j] = f

        b = np.full(rows, M, dtype=float)          # (f∗g)_k ≤ M
        c = -np.ones(n, dtype=float)               # maximise Σ g_i  → minimise -Σ g_i
        bounds = [(0, None)] * n

        # ---------------------------------------------------------------
        # Solve the LP.
        # ---------------------------------------------------------------
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'disp': False})
        if not res.success:
            print(f"[propose_candidate] LP failed at iteration {outer}: {res.message}")
            break

        g0 = np.maximum(res.x, 0.0)                # eliminate tiny negatives
        sum_g0 = float(g0.sum())
        if sum_g0 <= 0:
            print("[propose_candidate] LP returned zero‑mass vector, stopping.")
            break

        # ---------------------------------------------------------------
        # Rescale to the canonical ℓ¹‑norm and mix.
        # ---------------------------------------------------------------
        scale = target_sum / sum_g0
        g = g0 * scale

        f_new = (1.0 - t) * f + t * g
        # Numerical drift may break the exact norm – re‑normalise.
        f_new = f_new * (target_sum / f_new.sum())
        f_new = np.maximum(f_new, 0.0)

        # ---------------------------------------------------------------
        # Evaluate and decide whether to keep the update.
        # ---------------------------------------------------------------
        try:
            new_score = evaluate_sequence(f_new.tolist())
        except Exception:                         # pragma: no cover
            new_score = float('inf')

        print(f"[propose_candidate] iter {outer:2d} → score {new_score:.6f}")

        if new_score + eps_improve < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
        else:
            # No improvement – shrink the mixing step; if it gets too
            # small we abort the LP loop.
            t *= 0.5
            if t < 1e-4:
                print("[propose_candidate] mixing step vanished – terminating.")
                break
            # keep the current f for the next iteration

    # ---------------------------------------------------------------
    # Final logging and return.
    # ---------------------------------------------------------------
    final_score = evaluate_sequence(best_f.tolist())
    print(f"[propose_candidate] final_score={final_score:.6f}")

    # The evaluator tolerates any non‑negative scaling, so we simply
    # return the best vector we have found.
    return best_f.tolist()
