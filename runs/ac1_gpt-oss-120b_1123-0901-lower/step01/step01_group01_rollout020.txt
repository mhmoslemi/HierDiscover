    # conv is full conv array len 2n-1
    # find indices of top_k values

    entries (default k = 3) and the update is performed with a
    multiplicative (exponentiated‑gradient) step followed by a projection
    onto the probability simplex (simple normalisation).

3.  **Simulated‑annealing fine‑tuning** – random transfers of a small
    amount of probability between two coordinates are accepted either
    if they improve the score or with a temperature‑dependent
    probability.  This stage explores the neighbourhood of the best
    point found so far.

The whole routine respects the 1000 s time budget, prints occasional
progress messages and finally returns the best sequence found as a
plain Python list (the evaluator expects a list, not a NumPy array).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate candidate sequences from a few parametric families
def _make_candidate(n, kind, **params):
    """Return a normalised non‑negative vector of length n."""
    if kind == "geometric":
        factor = params["factor"]
        r = 1.0 - factor / n
        if r <= 0:
            seq = np.full(n, 1.0 / n)
        else:
            seq = r ** np.arange(n, dtype=np.float64)
    elif kind == "power":
        p = params["p"]
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    elif kind == "shifted_power":
        p = params["p"]
        s = params["s"]
        seq = ((np.arange(1, n + 1, dtype=np.float64) + s) ** (-p))
    elif kind == "sym_power":
        p = params["p"]
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = (i * (n - i + 1)) ** (-p)
    elif kind == "gaussian":
        sigma = params["sigma"]
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        seq = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif kind == "sine":
        seq = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    elif kind == "root_inv":
        # classic a_i ∝ 1/√(i+1)
        seq = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    else:
        raise ValueError(f"unknown kind {kind}")

    # safeguard against negative or all‑zero vectors
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total <= 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    evaluate_sequence().  The method combines a quick parametric sweep,
    an exponentiated‑gradient descent on the worst convolution entries,
    and a final simulated‑annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a modest length – large enough to give a fine
    #     resolution but small enough for fast convolutions.
    n = 800

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2.  Sweep simple families (geometric, power, symmetric power,
    #     Gaussian, sine, 1/√i).  Keep the best one.
    families = []

    # geometric – factor roughly 2/n is known to be good
    for factor in np.linspace(0.5, 5.0, 20):
        families.append(("geometric", {"factor": factor}))

    # pure power law a_i ∝ (i+1)^(-p)
    for p in np.linspace(0.2, 1.0, 20):
        families.append(("power", {"p": p}))

    # symmetric power law a_i ∝ ((i+1)*(n-i))^(-p)
    for p in np.linspace(0.2, 1.0, 20):
        families.append(("sym_power", {"p": p}))

    # shifted power law (to avoid the singularity at i=0)
    for p in [0.5, 0.6]:
        for s in [0.0, 1.0, 5.0]:
            families.append(("shifted_power", {"p": p, "s": s}))

    # Gaussian‑like bells, sigma expressed as a fraction of n
    for sigma_factor in np.linspace(0.1, 0.5, 10):
        families.append(("gaussian", {"sigma": sigma_factor * n}))

    # sine shape (positive half‑wave)
    families.append(("sine", {}))

    # classic 1/√i shape
    families.append(("root_inv", {}))

    # ------------------------------------------------------------------
    # Evaluate every candidate – stop early if time runs out.
    for kind, pars in families:
        if time.time() > deadline:
            break
        seq = _make_candidate(n, kind, **pars)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Init] kind={kind:12s} params={pars}  score={score:.6f}")

    # Fallback to uniform if nothing survived (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[Init] fallback uniform  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Exponential‑gradient descent on the max‑convolution.
    #     We use the top‑k largest convolution entries to obtain a
    #     richer sub‑gradient.
    seq = best_seq.copy()
    cur_score = best_score
    eta = 0.5               # initial (multiplicative) step size
    top_k = 3               # how many peak shifts participate in the gradient
    iter_eg = 0

    # reserve roughly the last 30 s for the annealing phase
    while time.time() < deadline - 30.0:
        conv = np.convolve(seq, seq)                     # length 2n‑1
        # indices of the k largest convolution values
        if top_k >= len(conv):
            peak_idxs = np.arange(len(conv))
        else:
            peak_idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        # build the aggregated gradient
        grad = np.zeros_like(seq)
        for k in peak_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # gradient of (a*a)[k] w.r.t a_i is 2*a_{k-i}; factor 2 absorbed in η
            grad[idx] += seq[k - idx]

        # multiplicative (exponentiated) update, then renormalise
        mult = np.exp(-eta * grad)
        seq_new = seq * mult
        seq_new = seq_new / seq_new.sum()

        # evaluate the new candidate
        new_score = evaluate_sequence(seq_new.tolist())

        if new_score < best_score:
            best_score = new_score
            best_seq = seq_new.copy()
            print(f"[EGD {iter_eg:05d}] new best = {best_score:.6f}")

        # accept the move if it improves the current point;
        # otherwise keep the old point (simple hill‑climb)
        if new_score < cur_score:
            seq = seq_new
            cur_score = new_score

        # decay the step size very slowly
        eta *= 0.9995
        iter_eg += 1

    # ------------------------------------------------------------------
    # 4.  Simulated‑annealing fine‑tuning.
    #     Randomly relocate a tiny amount of probability mass.
    cur_seq = seq.copy()
    cur_score = evaluate_sequence(cur_seq.tolist())
    it_sa = 0
    T0 = 0.02                     # initial temperature
    while time.time() < deadline:
        # pick two distinct positions and move a small chunk from i → j
        i, j = np.random.choice(n, 2, replace=False)
        max_delta = min(cur_seq[i], 0.02)          # at most 2 % of the total mass
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑normalise (tiny numerical drift)
        cand = cand / cand.sum()

        cand_score = evaluate_sequence(cand.tolist())

        # linear cooling schedule
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 0.5)
        T = max(1e-8, T0 * (1.0 - progress))

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / T):
            cur_seq = cand
            cur_score = cand_score

        # Global best tracking
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[SA {it_sa:06d}] new best = {best_score:.6f}")

        it_sa += 1

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    # evaluator expects a plain Python list
    return best_seq.tolist()
