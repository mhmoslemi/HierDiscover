"""
Propose a step‑function (non‑negative sequence) that improves the
objective 2 n·max(conv(a,a)) / (sum(a)^2).  The algorithm works in three
phases:

1. **Initial guess** – a short exponential tail (geometric progression)
   with a decay chosen from a small sweep.  The vector is projected onto
   the probability simplex so the sum equals 1.

2. **Smoothed max‑convolution minimisation** – we treat the objective as
   `max(conv)` and replace the nondifferentiable max by a soft‑max
   (log‑sum‑exp) with temperature α.  The gradient of the smooth
   approximation is a convolution of the current sequence with the
   soft‑max weights and is computed efficiently with a single
   NumPy convolution.  An Adam optimiser (with momentum and adaptive
   step‑sizes) updates the vector, after each step projecting back onto
   the simplex.

3. **Fine‑grained sub‑gradient polishing** – when the time budget is
   almost exhausted we switch to a classic projected sub‑gradient step
   using only the active shift (the index where the convolution attains
   its maximum).  This removes the remaining smoothing bias and often
   yields a lower final objective.

The whole routine respects the 1000 s budget, prints periodic progress
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x) = 1}
def _project_onto_simplex(v):
    """Project a (possibly negative) vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # find the last positive rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a coefficient sequence that minimises the evaluation function.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length – a few thousand gives a good trade‑off.
    n = 4000

    # ------------------------------------------------------------------
    # 2. Build a geometric start (a_i ∝ r^i) and keep the best of a few
    #    decay factors.  The factor c = ε·n controls the decay r = 1‑c/n.
    best_seq = None
    best_score = float("inf")
    for c in [0.8, 1.0, 1.2, 1.4, 1.6, 1.8, 2.0]:
        if time.time() > deadline:
            break
        eps = c / n
        r = max(0.0, 1.0 - eps)
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()               # sum = 1
        # a tiny jitter helps escaping degenerate plateaus
        seq += 1e-7 * np.random.rand(n)
        seq = _project_onto_simplex(seq)
        score = evaluate_sequence(seq.tolist())
        print(f"[Init] c={c:4.2f}  r={r:.6f}  score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # If for any reason we have no start, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    seq = best_seq.copy()

    # ------------------------------------------------------------------
    # 3. Adam optimisation on a smoothed max‑convolution.
    #    Parameters are chosen conservatively to stay within the time budget.
    lr = 0.2                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0
    alpha = 80.0                 # temperature for the soft‑max (higher → sharper)

    while time.time() < deadline:
        # convolution a*a
        conv = np.convolve(seq, seq)                 # length = 2n‑1
        max_conv = conv.max()

        # soft‑max weights approximating the arg‑max
        w_raw = np.exp(alpha * (conv - max_conv))
        w = w_raw / w_raw.sum()                      # sum_{k} w_k = 1

        # gradient = 2 * Σ_k w_k * ∂conv_k/∂a   (see analysis)
        # Implemented via a single convolution with the reversed weight vector.
        w_rev = w[::-1]
        corr = np.convolve(seq, w_rev)               # length = 3n‑2
        # slice the part that corresponds to the required indices
        grad = 2.0 * corr[(2 * n - 2) - np.arange(n)]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        seq_new = seq - step
        seq_new = _project_onto_simplex(seq_new)

        # evaluate and keep the best
        score = evaluate_sequence(seq_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq_new.copy()
            print(f"[Adam {t:06d}] new best score = {best_score:.6f}")

        seq = seq_new

        # slowly reduce the smoothing temperature – helps final convergence
        if t % 5000 == 0:
            alpha = max(1.0, alpha * 0.85)

    # ------------------------------------------------------------------
    # 4. Final polishing with a classic projected sub‑gradient method.
    #    Uses the active shift (the one attaining the maximum) as gradient.
    step = 0.12
    it = 0
    while time.time() < deadline:
        conv = np.convolve(seq, seq)
        k = int(conv.argmax())                     # active shift
        # sub‑gradient of (a*a)[k]  (exactly a_{k‑i} for valid i)
        grad = np.zeros_like(seq)

        hi = min(n - 1, k)

        grad[idx] = seq[k - idx] * 2.0            # factor 2 for symmetry

        seq_new = seq - step * grad
        seq_new = _project_onto_simplex(seq_new)

        score = evaluate_sequence(seq_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq_new.copy()
            print(f"[SubGrad {it:06d}] new best score = {best_score:.6f}")

        seq = seq_new
        step *= 0.9995          # gentle decay
        it += 1

    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
