"""
Propose a step‑function (non‑negative sequence) that beats the uniform
construction.  The search works in three stages:

1.  Scan a few geometric progressions a_i ∝ r^i (r = 1‑c/n) and keep the
    best one – this reproduces the “geometric seed’’ used previously.
2.  Run a projected Adam–style optimisation that minimizes a smooth
    approximation of max_k (a*a)[k].  The smooth max is realised by a
    soft‑max (log‑sum‑exp) with a temperature τ that is slowly reduced.
    After each Adam update the vector is projected onto the probability
    simplex (non‑negative, sum = 1) using the O(n log n) algorithm.
3.  Occasionally inject a tiny random perturbation (simulated‑annealing
    style) to help escape shallow plateaux.

All operations are O(n) or O(n log n); with n≈2000 the inner loop runs
many tens of thousands of iterations well within the 1000 s budget.
The routine returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project a (possibly negative) vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:           # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

                v.fill(0.0)
                print(f"[Iter {it:06d}] random perturbation injected")

    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
