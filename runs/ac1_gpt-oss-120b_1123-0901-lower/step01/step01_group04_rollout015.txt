import time
import numpy as np

def fft_convolve(a, b, L=None):
    """Linear convolution of real 1‑D arrays a and b via FFT.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    if L is None:
        L = 1 << ((len(a) + len(b) - 1).bit_length())
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    c = np.fft.irfft(A * B, L)
    return c[:len(a) + len(b) - 1]

def softmax_weights(conv, beta):
    """Return the soft‑max of beta*conv as a probability vector."""
    # shift for numerical stability
    z = beta * conv
    z_max = z.max()
    e = np.exp(z - z_max)
    return e / e.sum()

def project_onto_simplex(v):
    """Simple projection onto the probability simplex {x≥0, sum x = 1}.
    Here we just zero‑out negatives and renormalise – sufficient for our
    heavily regularised updates.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fall back to uniform distribution to keep the iterate feasible
        n = v.shape[0]
        return np.full(n, 1.0 / n)
    return v / s

def evaluate_obj(a, n):
    """Fast evaluation of the true objective using FFT convolution.
    Returns 2·n·max(conv(a,a)).
    """
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[:2 * n - 1]
    conv = np.clip(conv, 0.0, None)
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``. The algorithm consists of three stages:

    1.  Initialise with the discrete arcsine distribution (the known asymptotic
        optimum) and optionally improve it by a cheap one‑dimensional
        mixture/parameter sweep.
    2.  Refine the best candidate with a smooth‑max gradient descent
        (Adam optimiser).  The smooth‑max turns the non‑differentiable
        ``max(conv)`` into a weighted sum, allowing us to take a gradient
        that pushes mass away from the currently dominant convolution index.
        The temperature (soft‑max β) is increased slowly so that the method
        first smooths the landscape and later focuses on the true maximum.
    3.  The best sequence seen throughout the run is returned.
    """

    # 1) Determine sequence length and initialise a starting point.
    # -----------------------------------------------------------------

    n = kwargs.get("length")

        except Exception:
            n = 6027  # fallback used in earlier runs

    a_arc[0] = 1.0

    a_arc /= a_arc.sum()                     # normalised sum = 1

    a_uni = np.full(n, 1.0 / n, dtype=np.float64)

    # ---- 1c) pick a feasible starting sequence --------------------
    best_seq = a_arc.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, base score={best_score:.6f}")

    # 2) Cheap parametric sweeps (mixtures and Beta‑like shapes)
    # -----------------------------------------------------------------
    # Mixture of arcsine and uniform

        cand = lam * a_arc + (1.0 - lam) * a_uni
        cand = project_onto_simplex(cand)      # safe‑guard against rounding
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[mix] λ={lam:.2f}, score={sc:.6f}")

    # Simple symmetric power‑law family a_k ∝ (x(1‑x))^{α}
    # x = (k+0.5)/n lies in (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # exponents to try (negative values produce U‑shaped densities)
    exponents = np.array([-0.9, -0.8, -0.7, -0.6, -0.55, -0.5,
                          -0.45, -0.4, -0.35, -0.3, -0.25, -0.2,
                          -0.15, -0.1, -0.05, 0.0])
    for a_exp in exponents:
        # avoid exact zeros at the ends by adding a tiny epsilon
        eps = 1e-12
        vals = (np.maximum(x, eps) ** a_exp) * (np.maximum(1.0 - x, eps) ** a_exp)
        cand = vals / vals.sum()
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[beta] α={a_exp:.2f}, score={sc:.6f}")

    # 3) Smooth‑max gradient descent with Adam (soft‑max temperature β)
    # -----------------------------------------------------------------
    a = best_seq.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # FFT lengths (different for the two convolutions we need)
    L_conv = 1 << ((2 * n - 1).bit_length())       # for a*a
    L_corr = 1 << ((3 * n - 2).bit_length())       # for a * w_rev

    start = time.time()
    deadline = start + budget_s - 0.5
    it = 0
    beta = 2.0                # start temperature
    beta_max = 200.0

    while time.time() < deadline:

        conv = np.fft.irfft(A_conv * A_conv, L_conv)[:2 * n - 1]
        conv = np.clip(conv, 0.0, None)   # numerical safety

        # ---- 3b) smooth‑max weighting -----------------------------------
        w = softmax_weights(conv, beta)

        # ---- 3c) gradient of the smooth‑max objective -------------------
        # w_rev = w reversed
        w_rev = w[::-1]

        W_corr = np.fft.rfft(w_rev, L_corr)

        g = full[n - 1:2 * n - 1][::-1]   # size n
        grad = 4.0 * n * g                # ∇ O_β

        # ---- 3d) Adam update -------------------------------------------
        it += 1
        lr = 0.02 / (beta + 1.0)          # modest decay as β grows
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = project_onto_simplex(a_new)

        if it % 30 == 0:
            cur_score = evaluate_sequence(a_new.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_new.copy()
                print(f"[improve] it={it:05d}, β={beta:.1f}, score={cur_score:.6f}")

        # update temperature (slow geometric growth)
        if it % 150 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.07)

        a = a_new

    # -----------------------------------------------------------------
    # 4) Final reporting
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] it={it}, best_score={final_score:.6f}")
    return best_seq.tolist()
