```
def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # choose length
    n = len(height_sequence_1) if defined else 5000

    # init a
    if defined: a = np.array(height_sequence_1, dtype=float)
    else: init central binomial for n.

    # normalize sum to 1
    a = a / a.sum()
    # rename: shape length n.

    # Adam parameters
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for alpha and learning rate
    alpha = 10.0
    lr = 0.05  # base step size

    best_a = a.copy()
    best_score = float('inf')
    iteration = 0

    while time.time() < deadline:
        # compute conv
        conv = full_convolution_fft(a, a)
        max_val = conv.max()
        # compute metric
        score = 2*n*max_val/(a.sum()**2)  # same as 2*n*max_val since sum=1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iteration} new best {best_score}")

        # compute softmax weights
        # scaling: using exponent but safe
        tmp = conv - max_val
        w = np.exp(alpha * tmp)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback uniform
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum

        # gradient via correlation
        a_rev = a[::-1]
        # conv_w_ar = full_convolution_fft(w, a_rev)
        conv_w_ar = full_convolution_fft(w, a_rev)
        # get segment n-1:n-1+n
        g = 2.0 * conv_w_ar[n-1: n-1+n]

        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # clip negatives and renormalize sum to 1
        a = np.maximum(a, 0)
        s = a.sum()
        if s <= 0:
            # random restart
            a = np.random.rand(n)
            s = a.sum()
        a = a / s

        # schedule updates
        # increase alpha slowly
        alpha *= 1.001  # mild increase

        # optional decreasing lr
        if iteration % 10000 == 0 and iteration > 0:
            lr *= 0.9

        # random perturbation sometimes
        if iteration % 5000 == 0 and iteration > 0:
            a += np.random.randn(n) * 1e-3
            a = np.maximum(a, 0)
            a = a / a.sum()

        iteration += 1
    # Return best_a.tolist()
    return best_a.tolist()
```

def _full_convolution_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

def _project_onto_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        # uniform distribution fallback
        return np.full_like(v, 1.0 / len(v))
    return v / s

"""
Optimise a non‑negative step‑function (a list of heights) to minimise

    2·n·max(conv(a, a)) / (sum(a)**2)

The metric is homogeneous of degree 0, therefore we keep the vector on the
probability simplex (sum = 1).  The objective max(conv) is non‑smooth; we
approximate it by the smooth log‑sum‑exp (soft‑max) with an inverse temperature
α that is slowly increased.  The gradient of the smooth surrogate is a
weighted sum of the gradients of each convolution entry and can be computed
by a single FFT‑based correlation.  Adam (adaptive moment) updates keep the
step size well‑scaled, while a simple projection onto the simplex enforces
non‑negativity and normalisation.

The optimiser runs until the time budget expires and returns the best vector
seen.  It can start from the previously discovered `height_sequence_1` if
available; otherwise it initialises with the central‑binomial coefficients
(the discrete analogue of (1‑z)^‑½), which are known to give a very flat
self‑convolution.
"""

import time
import math
import numpy as np


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (real‑to‑real)."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, sum(x) = 1} by clipping and renormalising."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / s


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a coefficient sequence that minimises the evaluation metric.
    Uses a smooth‑max objective with Adam updates and a slowly increasing
    temperature parameter.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length and an initial vector.
    # ------------------------------------------------------------------
    try:
        # if the previous run left a variable, reuse its length and values
        n = len(height_sequence_1)          # type: ignore[name-defined]
        a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        # default: a few thousand entries – enough degrees of freedom, cheap FFT
        n = 5000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            # a_i = a_{i‑1}·(2i‑1)/(2i)  – central‑binomial coefficients
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    # Normalise to the simplex (sum = 1); the metric is homogeneous.
    a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # 2. Adam state and hyper‑parameters.
    # ------------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr = 0.05                     # base learning rate
    alpha = 10.0                  # inverse temperature for the soft‑max
    alpha_factor = 1.001          # geometric increase each iteration

    best_a = a.copy()
    best_score = float("inf")
    iteration = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- self‑convolution and metric ----
        conv = _full_convolution_fft(a, a)          # length 2·n‑1
        max_val = conv.max()
        score = 2.0 * n * max_val                    # sum(a)=1 ⇒ metric = 2·n·max

            best_a = a.copy()

        w_sum = w.sum()

            w[np.argmax(conv)] = 1.0
            w_sum = 1.0

        a_rev = a[::-1]                              # reversed a
        grad_conv = _full_convolution_fft(w, a_rev)  # length 3·n‑2

    # 4. Return the best coefficient list (plain Python floats).
    # ------------------------------------------------------------------
    return best_a.tolist()
