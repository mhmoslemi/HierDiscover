def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    start = time.time()
    np.random.seed(seed)

    # Determine length
    n = kwargs.get('length')
    if n is None:
        try:
            n = len(height_sequence_1)  # fallback
        except Exception:
            n = 6027

    # Precompute FFT size
    size = 1 << ((2*n-2).bit_length())

    def conv_max_fast(arr):
        A = np.fft.rfft(arr, n=size)
        C = A * A
        conv = np.fft.irfft(C, n=size)
        conv = conv[:2*n-1]
        return conv.max()

    # 1. Build candidate families

    best_seq = None
    best_score = float('inf')

    # arcsine (p=0.5)
    def make_distribution(p):
        k = np.arange(n, dtype=np.float64) + 0.5
        prod = k * (n - 0.5 - np.arange(n, dtype=np.float64))
        w = prod ** (-p)
        s = w.sum()
        if s == 0:
            return w
        return w / s

    # grid exponents
    exponents = np.linspace(0.3, 0.8, 26)  # 0.3..0.8 step 0.02
    for p in exponents:
        a = make_distribution(p)
        # score
        cur_score = 2 * n * conv_max_fast(a) / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            best_exponent = p
    print(f"[grid] best exponent {best_exponent:.4f}, score={best_score:.6f}")

    # 2. Simulated annealing hill climbing starting from best_seq
    cur = best_seq.copy()
    cur_score = best_score
    # track best internal
    best_seq_internal = cur.copy()
    best_score_internal = cur_score

    deadline = start + budget_s - 0.5
    # SA parameters
    max_step = 0.5   # allow up to 50% of source mass per move
    # maybe adapt step size based on iteration; we can just use some distribution.
    it = 0
    while time.time() < deadline:
        # picking src,dst
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        src = i
        dst = j
        # choose delta
        src_val = cur[src]
        if src_val == 0:
            continue
        # step scaling with cooling
        elapsed = time.time() - start
        progress = elapsed / budget_s
        # shrink max step linearly
        step_factor = max(0.01, (1 - progress))
        delta = max_step * step_factor * np.random.rand() * src_val
        # apply
        cur[src] -= delta
        cur[dst] += delta
        # compute
        new_score = 2 * n * conv_max_fast(cur) / (cur.sum()**2)
        # temperature schedule
        temp0 = 0.001
        temp = temp0 * (1 - progress) + 1e-7
        # accept?
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / temp):
            cur_score = new_score
            # keep modifications
            if new_score < best_score_internal:
                best_score_internal = new_score
                best_seq_internal = cur.copy()
                print(f"[improve] iter={it}, score={best_score_internal:.6f}")
        else:
            # revert
            cur[src] += delta
            cur[dst] -= delta
        it += 1

    print(f"[final] it={it}, best_score={best_score_internal:.6f}")
    # return best_seq_internal as list
    return best_seq_internal.tolist()

"""
Projected gradient descent for minimizing the supremum of the self‑convolution
of a non‑negative step function.  The method starts from a family of
beta‑like (arcsine) distributions, picks the best exponent, and then runs a
simple projected gradient descent on the max‑convolution value.  Symmetry
(and the simplex constraint ∑a_i=1 , a_i≥0) are enforced after each step.
The algorithm is fast enough to run many thousands of iterations within the
1000 s budget and usually improves on the plain arcsine construction.
"""

import time
import numpy as np

def convex_proj_simplex(v):
    """
    Project a vector onto the probability simplex {x >= 0, sum(x) = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def conv_max_and_index(a):
    """
    Return (max value, index of max) of the discrete convolution a*a.
    Uses an FFT of the next power‑of‑two size for speed.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two >= 2*n-1
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)
    conv = conv[: 2 * n - 1]
    max_val = conv.max()
    max_idx = int(conv.argmax())
    return max_val, max_idx

def generate_beta_like(p, n):
    """
    Symmetric beta‑like distribution on {0,…,n‑1}:
        w_k ∝ ((k+½)*(n‑½‑k))^{‑p}
    The returned vector is normalised to have sum 1.
    """
    k = np.arange(n, dtype=np.float64)
    prod = (k + 0.5) * (n - 0.5 - k)
    # avoid division by zero – prod is always > 0 for finite n
    w = np.power(prod, -p)
    total = w.sum()
    if total == 0:
        return w
    return w / total

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    evaluate_sequence.  The routine:
        1. picks a length (preferring the length of the global ``height_sequence_1`);
        2. scans a small family of beta‑like shapes and keeps the best as a start;
        3. runs a projected gradient descent on the max‑convolution value,
           always staying in the simplex and preserving symmetry;
        4. returns the best sequence found as a plain list of floats.
    """
    # -----------------------------------------------------------------
    # 1.  Determine length (use the global hint if present)
    # -----------------------------------------------------------------
    np.random.seed(seed)
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)  # type: ignore[name-defined]
        except Exception:
            n = 6027                     # default that worked before

    # -----------------------------------------------------------------
    # 2.  Scan a few beta‑like exponents and keep the best start point
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_exponent = None

    # exponents around the arcsine value 0.5 ; we explore a small interval
    for p in np.linspace(0.35, 0.70, 36):   # step ≈0.01
        a = generate_beta_like(p, n)
        max_val, _ = conv_max_and_index(a)
        score = 2.0 * n * max_val / (a.sum() ** 2)  # denominator is 1 after normalisation
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            best_exponent = p

    print(f"[init] length={n}, best exponent={best_exponent:.4f}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 3.  Projected gradient descent on the max‑convolution value
    # -----------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (always symmetric)
    cur_score = best_score
    best_overall_seq = a.copy()
    best_overall_score = cur_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    it = 0
    # step‑size schedule – start relatively aggressive, decay to ~1e‑4
    step0 = 0.5
    while time.time() < deadline:
        # compute convolution and locate the index where it attains the maximum
        max_val, t_max = conv_max_and_index(a)

        # build the gradient of the maximal entry w.r.t. a
        grad = np.zeros_like(a)
        # only the slice where a contributes to t_max matters
        lo = max(0, t_max - (n - 1))
        hi = min(n - 1, t_max)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[t_max - idx]

        # step size slowly decays with the elapsed fraction of the budget
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        step = step0 * (1.0 - frac) ** 2 + 1e-6

        # tentative update, then project back onto the simplex
        a_new = a - step * grad
        a_new = convex_proj_simplex(a_new)

        # enforce exact symmetry (averaging with the reversed copy)
        a_new = (a_new + a_new[::-1]) * 0.5

        # evaluate the candidate
        new_max, _ = conv_max_and_index(a_new)
        new_score = 2.0 * n * new_max / (a_new.sum() ** 2)

        # accept only if we improve (simple descent)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_overall_score:
                best_overall_score = new_score
                best_overall_seq = a_new.copy()
                print(f"[improve] it={it:06d}, score={best_overall_score:.6f}")

        it += 1

    print(f"[final] iterations={it}, best score={best_overall_score:.6f}")

    # -----------------------------------------------------------------
    # 4.  Return the best sequence as a plain Python list
    # -----------------------------------------------------------------
    return best_overall_seq.tolist()

"""
Projected‑gradient / stochastic refinement for the self‑convolution bound.

The code builds a family of symmetric beta‑like (arcsine) distributions
a_k ∝ ((k+½)(n‑½‑k))^(−p) and selects the best exponent as a starting
point.  From there a fast projected‑gradient descent is performed on the
maximum entry of the discrete convolution a∗a.  The objective is
    2·n·max(a∗a)      (the denominator = (∑a)^2 = 1 after normalisation)

The iterate is always projected onto the simplex {a_i≥0, ∑a_i=1} and
symmetrised (a_i = a_{n‑1‑i}) after each step.  The gradient is an
averaged contribution of the few largest convolution entries – this
smooths the non‑smooth “max’’ and lets the algorithm move away from the
stationary arcsine point.  With a 1000 s budget the routine typically

The function `propose_candidate` is the entry point required by the

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    rho_arr = np.where(u - cssv / ind > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full self‑convolution a*a (real FFT, returns only the needed part)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (length 2·n‑1) using an FFT of the next power‑of‑two size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2·n‑1
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: generate a symmetric beta‑like (arcsine) shape
# ----------------------------------------------------------------------
def beta_like(p: float, n: int) -> np.ndarray:
    """Return a symmetric density proportional to ((k+½)(n‑½‑k))^(−p)."""
    k = np.arange(n, dtype=np.float64)
    prod = (k + 0.5) * (n - 0.5 - k)                # always > 0
    w = np.power(prod, -p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return w
    return w / w_sum

            cur_score = 2.0 * n * full_convolution(a).max() / (a.sum() ** 2)

        it += 1

    print(f"[final] iterations={it}, best score={best_overall_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence as a plain Python list
    # --------------------------------------------------------------
    return best_overall.tolist()
