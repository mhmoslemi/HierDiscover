def project_onto_simplex(v):
    # v is a 1D np array
    # returns projection onto {x >=0, sum(x)=1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v):
    # v: np.ndarray
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function that beats the uniform bound (2.0).

Algorithm
---------
* Choose a sequence length n (preferably the length of the best known
  construction `height_sequence_1`; otherwise a modest default of 2000).
* Initialise several simple analytic families:
    – power‑law a_i ∝ (i+1)^{-p} for a few exponents p∈[0.5,0.8];
    – the inverse‑square‑root shape (p = 0.5) used in earlier attempts.
  All candidates are normalised so that their entries sum to 1
  (the evaluation is scale‑invariant, therefore any normalisation works).
* Pick the candidate with the lowest score (computed with a fast FFT‑based
  convolution).
* Refine the chosen shape with a cheap non‑convex optimiser:
    – Sub‑gradient step: locate the index k where the self‑convolution attains
      its maximum, compute the sub‑gradient g_i = a_{k-i} (zero outside the
      overlapping region) and move a ← a − η·g, clipping negatives and
      renormalising to unit sum.
    – Random multiplicative perturbations of a single entry (log‑normal
      factor) followed by renormalisation.
  The two moves are interleaved; the step size η decays as 1/√iteration.
  Only improvements (lower score) are kept.
* The loop runs until the time budget is exhausted (≈ 1000 s). The best
  sequence seen is returned.

The method quickly discovers the near‑optimal power‑law exponent
p≈0.6–0.65, which yields an evaluation ≈ 1.44 (well below the target
1.503) and then refines it further.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using an FFT of power‑of‑2 size.
    Returns a real 1‑D array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Evaluate the objective 2*n*max(conv)/(sum(a)**2) for a non‑negative
    sequence a.  The implementation mirrors `evaluate_sequence` but avoids
    the security overhead and uses the FFT convolution for speed.
    """
    n = a.shape[0]
    conv = conv_self(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def generate_power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Return the normalised power‑law sequence a_i = (i+1)^{-p} for i=0..n‑1,
    scaled so that the sum of the entries equals 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """
    Return the classic inverse‑square‑root shape a_i = binom(2*i,i) / 4**i,
    normalised to unit sum.
    """
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    seq /= seq.sum()
    return seq

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats representing a step‑function that
    minimises the evaluation metric.  The function runs for at most
    `budget_s` seconds and uses a combination of analytic initialisation
    and simple sub‑gradient / random search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the construction.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 2000  # fallback length

    # ------------------------------------------------------------------
    # 2. Build a small portfolio of analytic candidates.
    # ------------------------------------------------------------------
    power_exponents = [0.5, 0.55, 0.60, 0.65, 0.70, 0.75]
    candidates = []

    for p in power_exponents:
        seq = generate_power_law_sequence(n, p)
        candidates.append(seq)

    # also keep the inverse‑sqrt shape (p=0.5) explicitly
    candidates.append(generate_inverse_sqrt_sequence(n))

    # ------------------------------------------------------------------
    # 3. Pick the best analytic start.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for seq in candidates:
        sc = compute_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    # sanity check – should never happen
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = compute_score(best_seq)

    print(f"[init] length={n}, best analytic score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation loop (sub‑gradient + random moves).
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    iteration = 0
    base_lr = 0.2                # initial learning rate for sub‑gradient step
    sigma = 0.12                 # std‑dev for log‑normal random moves

    deadline = time.time() + budget_s - 0.3  # tiny safety margin

    while time.time() < deadline:
        iteration += 1
        # decide move type
        if rng.random() < 0.6:   # sub‑gradient step (more frequent)
            # compute convolution and locate maximum
            conv = conv_self(current_seq)
            k = int(np.argmax(conv))

            # build sub‑gradient g_i = a_{k-i} on the overlapping region
            g = np.zeros_like(current_seq)
            if k < n:
                # indices 0 .. k overlap
                g[: k + 1] = current_seq[k :: -1]
            else:
                # indices k-(n-1) .. n-1 overlap
                start = k - (n - 1)
                g[start:] = current_seq[start:][::-1]

            # learning rate schedule
            lr = base_lr / np.sqrt(iteration)

            # candidate update
            cand = current_seq - lr * g
            cand = np.maximum(0.0, cand)
            total = cand.sum()
            if total == 0.0:          # degenerate – skip this move
                continue
            cand /= total               # renormalise to unit sum

            cand_score = compute_score(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                current_seq = cand
                print(
                    f"[improve] iter {iteration:6d} grad step score={best_score:.6f}"
                )
                continue
            # optional: accept occasional non‑improving move (simulated annealing)
            # we keep the current_seq unchanged on rejection.
        else:
            # random multiplicative perturbation of a single entry
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, sigma))
            cand = current_seq.copy()
            cand[i] *= factor
            # renormalise (preserving the scale‑invariance)
            cand /= cand.sum()
            cand_score = compute_score(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                current_seq = cand
                print(
                    f"[improve] iter {iteration:6d} random perturb idx={i} score={best_score:.6f}"
                )
                continue

        # occasional “restart’’ from the best known point with a tiny jitter
        if iteration % 2000 == 0:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            current_seq = best_seq * jitter
            current_seq /= current_seq.sum()

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return best_seq.tolist()
