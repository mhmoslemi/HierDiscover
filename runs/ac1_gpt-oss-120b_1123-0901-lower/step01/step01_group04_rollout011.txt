def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Gradient–exponential descent algorithm for minimizing the evaluation function.
Starts from the arcsine (optimal‑asymptotic) construction, then repeatedly:
  * computes the full auto‑convolution,
  * builds a soft‑max distribution over the convolution indices,
  * updates the coefficients by a multiplicative (exponentiated‑gradient) step,
  * renormalises onto the probability simplex,
  * keeps the best sequence seen.
A cheap directed hill‑climbing phase finishes the search.
"""

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    # v is a 1‑d numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the largest rho such that u_rho > cssv_rho / rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a, nfft):
    """Fast full convolution a*a using a pre‑computed FFT length."""
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:2 * a.size - 1]
    # numerical noise may produce tiny negative values
    return np.maximum(conv, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Gradient‑exponential descent + directed hill‑climbing.
    Returns a list of non‑negative floats (the step‑function coefficients)
    that minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)

    # -------------------------------------------------------------
    # 1. Determine sequence length
    # -------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)  # type: ignore[name-defined]
        except Exception:
            n = 6000                     # a comfortable default
    # -------------------------------------------------------------
    # 2. Initialise with the arcsine construction
    # -------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()                     # now sum(a) == 1

    # -------------------------------------------------------------
    # 3. Basic bookkeeping
    # -------------------------------------------------------------
    best_seq = a.copy()
    conv = _full_convolution(a, 1 << ((2 * n - 1).bit_length()))
    best_score = float(2 * n * conv.max())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -------------------------------------------------------------
    # 4. Parameters for the exponential‑gradient phase
    # -------------------------------------------------------------
    nfft = 1 << ((2 * n - 1).bit_length())
    beta = 5.0            # smoothness of the soft‑max (will grow)
    eta = 0.2             # step size for the multiplicative update
    beta_inc = 1.001      # slowly increase beta so the algorithm
                          # focuses more on the true maximum
    eta_decay = 0.9999    # gently reduce the step size

    start = time.time()
    deadline = start + budget_s - 0.5   # leave a safety margin
    it = 0

    # -------------------------------------------------------------
    # 5. Exponential‑gradient optimisation loop
    # -------------------------------------------------------------
    while time.time() < deadline and it < 30000:
        # full auto‑convolution
        conv = _full_convolution(a, nfft)
        max_val = conv.max()
        # soft‑max probabilities (shifted by max to avoid overflow)
        probs = np.exp(beta * (conv - max_val))
        probs_sum = probs.sum()
        if probs_sum == 0:
            # numeric underflow – fall back to uniform weights
            probs = np.full_like(probs, 1.0 / probs.size)
        else:
            probs /= probs_sum

        # gradient of the smooth max (2·a * probs convolution)
        g = 2.0 * np.correlate(probs, a, mode='valid')  # length n

        # multiplicative (exponentiated‑gradient) update
        a_new = a * np.exp(-eta * g)

        # renormalise onto the simplex (keeps sum = 1 and non‑negativity)
        a_new = _project_simplex(a_new)

        # evaluate the true objective (no approximation)
        conv_new = _full_convolution(a_new, nfft)
        score_new = float(2 * n * conv_new.max())

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            a = a_new
            print(f"[exp‑grad] it={it:05d}, score={best_score:.6f}")
        else:
            # keep the old a, but still move a a little to avoid stagnation
            a = a_new

        # update parameters
        beta = min(beta * beta_inc, 80.0)
        eta *= eta_decay

        it += 1

    # -------------------------------------------------------------
    # 6. A final directed hill‑climbing phase (swap mass)
    # -------------------------------------------------------------
    it_hc = 0
    while time.time() < deadline and it_hc < 5000:
        # compute convolution and the index where it is maximal
        conv = _full_convolution(a, nfft)
        idx = conv.argmax()
        # partner array for the current maximal index
        partner = idx - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        w = np.zeros_like(a)
        w[mask] = a[partner[mask]]

        # contributions to the maximal entry
        contrib = a * w
        src = np.argmax(contrib)               # source of a large contribution
        # candidates for the destination – we prefer indices that hardly
        # participate in the maximal entry (small w)
        min_w = w.min()
        dst_candidates = np.where(w == min_w)[0]
        dst = np.random.choice(dst_candidates)

        # move a tiny fraction of mass
        eps = 0.01 * a[src]                     # 1 % of the source coefficient
        if eps <= 0.0:
            it_hc += 1
            continue
        a_try = a.copy()
        a_try[src] -= eps
        a_try[dst] += eps

        # keep feasibility (non‑negativity) – the source reduction is safe
        conv_try = _full_convolution(a_try, nfft)
        score_try = float(2 * n * conv_try.max())

        if score_try < best_score:
            a = a_try
            best_seq = a_try.copy()
            best_score = score_try
            print(f"[hill‑climb] it={it_hc:04d}, score={best_score:.6f}")

        it_hc += 1

    # -------------------------------------------------------------
    # 7. Return result
    # -------------------------------------------------------------
    print(f"[final] total iterations={it+it_hc}, best score={best_score:.6f}")
    return best_seq.tolist()

"""
Gradient‑exponential descent + a short directed hill‑climb.

The routine starts from the classic arcsine construction (asymptotically
optimal for the bound π/2) and then repeatedly:
  1. computes the full auto‑convolution a∗a,
  2. builds a soft‑max distribution over the convolution indices,
  3. takes a multiplicative (exponentiated‑gradient) step that moves mass
     away from the indices that dominate the current maximum,
  4. renormalises the coefficients onto the probability simplex.

When the time budget is almost exhausted a tiny deterministic hill‑climbing
phase swaps a small amount of mass from the most harmful coefficient to a
coefficient that hardly participates in the maximal convolution entry.
The best sequence seen during the whole run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]                       # last index satisfying the condition
    theta = cssv[rho] / (rho + 1)       # optimal threshold
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: fast full convolution a∗a using a pre‑computed FFT length
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, nfft: int) -> np.ndarray:
    """Compute the full linear convolution a*a via FFT (real‑valued)."""
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * a.size - 1]
    # eliminate tiny negative artefacts caused by rounding
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Gradient‑exponential descent followed by a tiny directed hill‑climb.
    Returns a list of non‑negative floats that (approximately) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1. Length of the sequence
    # --------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)      # type: ignore[name-defined]
        except Exception:
            n = 6000                        # safe default

    # --------------------------------------------------------------
    # 2. Initialise with the arcsine construction (the known asymptotic
    #    optimum).  The coefficients are scaled to sum to 1.
    # --------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()                           # now Σa_i = 1

    # --------------------------------------------------------------
    # 3. Book‑keeping
    # --------------------------------------------------------------
    nfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2n‑1
    conv = _full_convolution(a, nfft)
    best_score = float(2 * n * conv.max())
    best_seq = a.copy()
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Parameters for the exponential‑gradient phase
    # --------------------------------------------------------------
    beta = 5.0          # smoothness of the soft‑max (will slowly rise)
    eta = 0.20          # step size for the multiplicative update
    beta_inc = 1.001    # geometric increase of beta
    eta_decay = 0.9999  # gentle decay of the step size

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a short safety margin
    it = 0

    # --------------------------------------------------------------
    # 5. Exponential‑gradient optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline and it < 30000:
        # (a) full auto‑convolution
        conv = _full_convolution(a, nfft)
        max_val = conv.max()

        # (b) soft‑max probabilities (shifted by max to avoid overflow)
        probs = np.exp(beta * (conv - max_val))
        probs_sum = probs.sum()
        if probs_sum == 0.0:                # extreme under‑flow → fallback
            probs = np.full_like(probs, 1.0 / probs.size)
        else:
            probs /= probs_sum

        # (c) gradient of the smoothed max:  g = 2·a ⊛ probs
        #    (np.correlate with mode='valid' implements the needed
        #     cross‑correlation a ↔ probs)
        g = 2.0 * np.correlate(probs, a, mode='valid')   # length n

        # (d) multiplicative (exponentiated‑gradient) update
        a_new = a * np.exp(-eta * g)

        # (e) renormalise onto the simplex (keeps Σa = 1, a ≥ 0)
        a_new = a_new / a_new.sum()          # all entries stay positive

        # (f) evaluate the true (non‑approximated) objective
        conv_new = _full_convolution(a_new, nfft)
        score_new = float(2 * n * conv_new.max())

        # (g) accept any improvement; otherwise keep the previous a
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            a = a_new
            print(f"[exp‑grad] it={it:05d}, score={best_score:.6f}")
        else:
            # still move forward – the exponential step is cheap and
            # provides a smooth exploration of the landscape
            a = a_new

        # (h) schedule updates
        beta = min(beta * beta_inc, 80.0)
        eta *= eta_decay
        it += 1

    # --------------------------------------------------------------
    # 6. Final directed hill‑climbing phase (swap a tiny amount of mass)
    # --------------------------------------------------------------
    it_hc = 0
    while time.time() < deadline and it_hc < 5000:
        conv = _full_convolution(a, nfft)
        idx = int(conv.argmax())                     # index of the current maximum

        # partner index for the maximal entry:  partner = idx – i
        partner = idx - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        w = np.zeros_like(a)
        w[mask] = a[partner[mask]]                   # weight of each coefficient in the max entry

        # contribution of each coefficient to the maximal entry
        contrib = a * w
        src = int(np.argmax(contrib))                # source of the largest contribution

        # destination – choose a coefficient that hardly participates in the max entry
        min_w = float(w.min())
        dst_candidates = np.where(w == min_w)[0]
        dst = int(np.random.choice(dst_candidates))

        # move a small fraction (1 %) of the source mass to the destination
        eps = 0.01 * a[src]
        if eps == 0.0:
            it_hc += 1
            continue
        a_try = a.copy()
        a_try[src] -= eps
        a_try[dst] += eps

        # evaluate the move
        conv_try = _full_convolution(a_try, nfft)
        score_try = float(2 * n * conv_try.max())

        if score_try < best_score:
            a = a_try
            best_seq = a_try.copy()
            best_score = score_try
            print(f"[hill‑climb] it={it_hc:04d}, score={best_score:.6f}")

        it_hc += 1

    # --------------------------------------------------------------
    # 7. Return the best sequence found
    # --------------------------------------------------------------
    print(f"[final] total iterations={it + it_hc}, best score={best_score:.6f}")
    return best_seq.tolist()
