            if score < best_score:

            # find active indices
            # tolerance based on relative diff
            tol = 1e-12
            active = np.where(conv >= max_b - tol)[0]
            # compute gradient
            grad = np.zeros_like(a)
            for k in active:
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                idx = np.arange(low, high + 1, dtype=int)
                partner = k - idx
                grad[idx] += 2.0 * a[partner]
            # adapt step size

            if max_grad == 0:

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a fast FFT based convolution, a projected sub‑gradient descent on
the active (maximal) convolution entries and a small amount of simulated
annealing.  The algorithm works with many independent restarts of
different lengths (≈1200‑3000).  Each inner optimisation run keeps the
vector on the simplex {x_i≥0, Σx_i = √(2n)} – this makes the objective
scale‑invariant, so the objective equals the maximal convolution value.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Compute gradient of the *maximal* convolution entry (or a set of ties).
# grad_i = Σ_{k∈active} 2 * a_{k-i}   (only when the index is inside the overlap)
# --------------------------------------------------------------------------- #
def _gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    The best sequence discovered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    best_seq = None            # numpy array of the best candidate
    best_score = float("inf")  # objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Helper: run a single optimisation instance on a given length ``n``.
    # ------------------------------------------------------------------- #
    def _run_one_instance(n: int, init_vec: np.ndarray):
        nonlocal best_seq, best_score

        target_sum = np.sqrt(2.0 * n)           # Σ a_i = √(2n)  (scale‑invariant)
        a = _project_onto_simplex(init_vec.astype(float), target_sum)

        # hyper‑parameters for this run
        base_lr = 0.25                          # base learning rate
        momentum = np.zeros_like(a)            # heavy‑ball momentum
        beta_mom = 0.9
        iter_cnt = 0

        # allocate a slice of the remaining time for this run
        run_deadline = min(deadline, time.time() + 14.0)   # ~14 s per run

        while time.time() < run_deadline:
            # ----- convolution & current score ---------------------------------
            conv = _conv_full_fft(a)
            max_val = conv.max()
            cur_score = float(max_val)          # equals evaluate_sequence() value
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (n={n})")

            # ----- identify all active (max‑tying) indices --------------------
            # tolerance relative to the max (tiny, only true ties usually)
            eps = 1e-12
            active = np.where(conv >= max_val - eps)[0]

            # ----- gradient of the active max entry (sub‑gradient) ------------
            grad = _gradient_of_active(a, active)

            max_grad = grad.max()
            if max_grad <= 0.0:
                # nothing to descend – break to avoid division by zero
                break

            # step size – similar spirit to the original code
            eta = base_lr * max_val / max_grad

            # heavy‑ball momentum update
            momentum = beta_mom * momentum - eta * grad
            a_new = a + momentum

            # enforce feasibility
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_onto_simplex(a_new, target_sum)

            # ----- occasional random shake (simulated‑annealing style) --------
            if np.random.rand() < 0.02:
                noise = np.random.randn(n) * (0.01 * target_sum / n)
                a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
                base_lr *= 0.9            # slowly shrink step after a shake

            a = a_new
            iter_cnt += 1
            # a tiny decay of the base learning rate helps final fine‑tuning
            base_lr *= 0.99995

    # ------------------------------------------------------------------- #
    # Main outer loop – many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.  Larger n gives more freedom but each
        # iteration costs a bit more; we stay between 1200 and 3000.
        n = np.random.randint(1200, 3001)

        # Initialise either from the existing good construction (if any)
        # or from a fresh random point.
        if np.random.rand() < 0.35 and "height_sequence_1" in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        _run_one_instance(n, init)

        # Early exit if we already beat the target bound (1.503)
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if for some reason nothing was found.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to plain Python list as required by the grader
    return best_seq.tolist()
