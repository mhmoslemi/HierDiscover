The linear program: maximize sum_j b_j subject to b_j >= 0, and (f * b)_k <= M for all k, where M = max convolution of f with itself. def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _convolve_self(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv

    sum_a = a.sum()
    if sum_a < 0.01:

    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a**2))

    eps = alpha / n

"""
Propose a step‑function (non‑negative sequence) that beats the uniform
construction (objective ≈ 2) and improves on the previous best
≈ 1.5449.  The algorithm consists of three stages:

1. **Broad parametric sweep** – generate a few hundred candidates from
   simple parametric families (geometric, power‑law, beta‑like and random
   Dirichlet).  The best of these seeds the next stage.

2. **Aggressive projected gradient descent** – repeatedly locate the
   active shift `k* = argmax (a*a)[k]`, take the sub‑gradient
   `g_i = a_{k*-i}` (the reversed fragment that builds the maximal
   convolution entry) and try a back‑tracking line‑search on the step
   size.  After each trial the vector is projected back onto the
   probability simplex (non‑negative, sum = 1).  The line‑search accepts
   the first step that strictly improves the objective, otherwise a
   small random mass‑transfer move is attempted.

3. **Final output** – the best sequence found before the time budget
   expires is returned as a plain Python list.

The implementation uses a fast FFT based convolution (O(n log n)) and
tight time‑checks, so it comfortably fits into the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast self‑convolution using FFT (real inputs)
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Return the full convolution of seq with itself (length 2n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    # real FFT is a little faster
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective used by evaluate_sequence (computed locally for speed)
def _objective(seq: np.ndarray) -> float:
    """Compute 2·n·max(conv)/sum(seq)^2 for a non‑negative sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1 - alpha/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a-1}(n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    # guard against under‑flow when a or b are < 1
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()
