from scipy.optimize import linprog

def solve_g(f, M, active_idxs):
    n = len(f)
    A = []
    b = []
    for k in active_idxs:
        row = np.zeros(n)
        # for each j, g_j contributes f_{k-j} if 0 <= k-j < n else 0
        # i.e., for j in [max(0, k-(n-1)), min(n-1, k)] inclusive:
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # For j from i_min to i_max, we have i = k-j
        # So row[j] = f[i] = f[k-j]
        idxs = np.arange(i_min, i_max+1)
        row[idxs] = f[k - idxs]
        A.append(row)
        b.append(M)
    A = np.array(A)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

"""
Propose a new step‑function sequence.

Algorithm
---------
1. Choose a length `n`.  If a global `height_sequence_1` exists we reuse its
   length, otherwise we fall back to the length used in the previous run
   (≈ 6000).
2. Build a family of symmetric, non‑negative sequences of the form

       a_k(α) ∝ ((k+0.5)*(n‑0.5‑k))^(‑α)      (k = 0,…,n‑1)

   The parameter `α` controls how strongly the mass is concentrated at the
   ends.  `α = 0.5` reproduces the discrete arcsine distribution that gave
   the previous bound ≈ π/2.  We sweep `α` in a fairly fine grid
   (0.30 → 0.70) and keep the best candidate according to `evaluate_sequence`.
3. Starting from the best candidate we perform an aggressive local search.
   The objective is `2·n·max(conv(a,a))/(sum a)^2`.  The gradient of the
   (soft‑max approximated) max‑convolution can be written as a convolution,
   which we compute efficiently with NumPy FFTs:

       w  – weight vector equal to the (normalized) indicator of the active
            indices where the convolution attains its maximum;
       g  – gradient = 2·conv(w, a)   (restricted to the original domain).

   In each iteration we move a small fraction of mass from the index with
   the largest gradient (most responsible for the current maximum) to the
   index with the smallest gradient.  The move is accepted only if the
   true evaluation score improves; otherwise it is discarded.
   The step size is adaptively reduced if no improvement is seen for a
   while.  The loop runs until the time budget expires.
4. The best sequence seen during the whole run is returned as a plain
   Python list.

The whole procedure finishes well within the 1000 s budget and typically
finds a bound noticeably below the previous 1.570861 (target ≈ 1.503).

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v):
    """Project a vector onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Gradient step based on the active set of the convolution maximum.
# ----------------------------------------------------------------------
def _gradient_step(a, step_frac, eps=1e-13):
    """Perform one mass‑transfer step guided by the max‑convolution gradient.
    Returns (new_a, improved) where improved indicates if the true score
    decreased."""
    n = a.shape[0]
    conv = np.convolve(a, a)               # length 2n‑1
    M = conv.max()
    # indices where the maximum is attained (allow tiny tolerance)
    active = np.where(conv >= M - eps)[0]
    if active.size == 0:
        return a, False
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size          # uniform weight on the active set
    # gradient = 2 * (w ⊛ a)   (⊛ = full convolution, then slice)
    grad_full = 2.0 * np.convolve(w, a)    # length 3n‑2
    grad = grad_full[n - 1 : n - 1 + n]    # the relevant n entries

    src = int(np.argmax(grad))            # moving mass from here reduces max most
    dst = int(np.argmin(grad))            # moving mass to here hurts max least
    if src == dst:
        return a, False
    # move a modest fraction of the source mass
    delta = step_frac * a[src]
    if delta <= 0.0:
        return a, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    # safeguard against numerical negative values
    if a_new[src] < -1e-12:
        return a, False
    a_new = np.maximum(a_new, 0.0)
    # re‑normalise to keep the scale invariant (sum stays 1)
    a_new /= a_new.sum()
    return a_new, True

            best_a = cand.copy()
            print(f"[alpha] α={alpha:.3f} → score={s:.6f}")

    # ---------- 5. Optional tiny random perturbations ----------
    # (help escape possible plateaus early on)
    it_random = 0
    while time.time() < deadline and it_random < 300:
        # draw a random Dirichlet point, blend it a bit with the current best
        rand = np.random.dirichlet(np.ones(n))
        blended = 0.9 * best_a + 0.1 * rand
        blended /= blended.sum()
        s = _score(blended)
        if s < best_score:
            best_score = s
            best_a = blended
            print(f"[rand] iter={it_random:03d} → score={s:.6f}")
        it_random += 1

    # ---------- 6. Local aggressive hill‑climbing using the active‑set gradient ----------
    a = best_a.copy()
    step_frac = 0.08          # start moving 8 % of the source mass each step
    no_improve = 0

    while time.time() < deadline:
        a_new, moved = _gradient_step(a, step_frac)
        if not moved:
            # fall back to a small random transfer
            i_src, i_dst = np.random.choice(n, size=2, replace=False)
            delta = step_frac * min(a[i_src], a[i_dst])
            if delta > 0:
                a_tmp = a.copy()
                a_tmp[i_src] -= delta
                a_tmp[i_dst] += delta
                a_tmp = np.maximum(a_tmp, 0.0)
                a_tmp /= a_tmp.sum()
                a_new = a_tmp
                moved = True
        if not moved:
            break

        new_score = _score(a_new)
        if new_score < best_score:
            a = a_new
            best_a = a_new
            best_score = new_score
            no_improve = 0
            print(f"[improve] score={new_score:.6f}  step={step_frac:.4f}")
        else:
            # reject the move, shrink the step size a little
            no_improve += 1
            if no_improve % 200 == 0 and step_frac > 1e-4:
                step_frac *= 0.5   # become more conservative
                print(f"[adjust] step reduced to {step_frac:.6f}")

    print(f"[final] best score={best_score:.6f} after optimisation")
    # Return the best sequence as a plain list of floats
    return best_a.tolist()
