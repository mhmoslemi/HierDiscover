"""
Propose a candidate step‑function sequence.

The core idea is to start from the “inverse‑square‑root’’ (central‑binomial)
sequence, which already gives the classic bound π/2≈1.571.
We then improve it by a few iterations of a linear‑programming refinement
described in the problem statement:

*   With the current sequence `f` (normalised so that Σf = √(2·n)),
    compute M = max(f*f).

*   Solve the linear program  
        maximise Σ g_j
        s.t.   g_j ≥ 0
               (f * g)_k ≤ M   for every convolution index k

    This is linear because `f` is fixed; the convolution constraints are
    written as a dense matrix `C` with rows C[k]·g = (f*g)[k].

*   Rescale the optimal `g` back to the same ℓ¹‑norm as `f`
    (let S = √(2·n)) and mix the two sequences:
        f ← (1‑t)·f + t·g_scaled      (t≈0.3‑0.5)

    The mixing step strictly reduces the maximal convolution value, hence
    improves the evaluation score.

We repeat the LP‑mixing loop a few times (as long as time permits) and
finish with a cheap random local search (multiplicative perturbations of
single entries) to squeeze a little more improvement.

All numbers are kept non‑negative; the evaluation function is scale‑invariant
so we keep the ℓ¹‑norm fixed to √(2·n) throughout the optimisation.
"""

import time
import numpy as np

# cvxpy is optional – if unavailable we fall back to the simple local search.
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:   # pragma: no cover
    _CVXPY_AVAILABLE = False


def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Return a list a_0,…,a_{n‑1} with a_i = C(2i,i)/4**i.
    The recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) avoids large intermediate numbers."""
    seq = [1.0]       # a_0 = 1
    for i in range(1, n):
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix C such that (C @ g)[k] = (f * g)[k] for all k.
    Row k uses the coefficients f_{k‑j} where they are defined.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=float)
    for j in range(n):
        # affect rows k = j, j+1, …, j+n‑1  (but not beyond rows‑1)
        length = min(n, rows - j)
        C[j:j + length, j] = f[:length]
    return C


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Choose a length.  Use the previously found sequence if it exists,
    #    but cap the size to keep the LP tractable (≈ 2 000 – 3 000).
    # ----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        n = min(n, 2500)                    # safety cap for the LP
        if n < 500:
            n = 500
    except Exception:
        n = 2000

    # ----------------------------------------------------------------------
    # 2. Base sequence – the “optimal’’ inverse‑square‑root shape.
    #    Normalise it to the convention Σa = √(2·n) which makes the score
    #    equal to the maximal convolution value.
    # ----------------------------------------------------------------------
    base = np.array(generate_inverse_sqrt_sequence(n), dtype=float)
    target_sum = np.sqrt(2.0 * n)
    base = base * (target_sum / base.sum())
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    deadline = time.time() + budget_s - 0.5   # tiny safety margin
    max_lp_iters = 12                         # a handful of LP refinements
    t_mix = 0.4                               # mixing weight for each LP step

    # ----------------------------------------------------------------------
    # 3. LP‑based refinement loop (if cvxpy is available).
    # ----------------------------------------------------------------------
    if _CVXPY_AVAILABLE:
        while time.time() < deadline and max_lp_iters > 0:
            # current maximal convolution M (also the current score)
            conv_f = np.convolve(best_seq, best_seq)
            M = float(conv_f.max())

            # Build convolution matrix for the constraints f * g ≤ M.
            C = _build_conv_matrix(best_seq)

            # Linear program: maximise sum(g) subject to C @ g ≤ M, g ≥ 0.
            g = cp.Variable(n, nonneg=True)
            constraints = [C @ g <= M * np.ones(2 * n - 1)]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                           verbose=False)
            except Exception as e:          # pragma: no cover
                print("[LP] solver exception:", e)
                break

            if g.value is None:            # pragma: no cover
                print("[LP] no solution returned")
                break

            g_val = np.maximum(np.array(g.value).ravel(), 0.0)
            sum_g = g_val.sum()
            if sum_g == 0.0:                # pragma: no cover
                print("[LP] degenerate zero‑sum solution")
                break

            # Rescale g to the target ℓ¹‑norm.
            g_scaled = g_val * (target_sum / sum_g)

            # Mix the old and the new sequence.
            new_seq = (1.0 - t_mix) * best_seq + t_mix * g_scaled
            new_score = evaluate_sequence(new_seq.tolist())

            print(f"[LP] iter left={max_lp_iters-1}  M={M:.6f}"
                  f"  sum_g={sum_g:.3f}  new_score={new_score:.6f}")

            if new_score < best_score - 1e-7:
                best_seq = new_seq
                best_score = new_score
                print(f"[improve] score={best_score:.6f}")
                # keep the current mixing weight
            else:
                # no progress – reduce the mixing weight to be more conservative
                t_mix *= 0.5
                print(f"[LP] no improvement, reducing mix weight to {t_mix:.3f}")
                if t_mix < 0.01:
                    break

            max_lp_iters -= 1
    else:
        print("[info] cvxpy not available – skipping LP refinement")

    # ----------------------------------------------------------------------
    # 4. Light‑weight random local search for the remaining time.
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        i = rng.integers(n)
        old = best_seq[i]
        # small log‑normal multiplicative perturbation
        factor = np.exp(rng.normal(0.0, 0.02))
        new_val = max(0.0, old * factor)
        cand = best_seq.copy()
        cand[i] = new_val
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[local] improve idx={i} score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
