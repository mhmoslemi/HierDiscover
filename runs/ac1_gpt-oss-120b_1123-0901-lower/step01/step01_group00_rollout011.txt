def propose_candidate(seed=0, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None; best_score = inf

    while time.time() < deadline:
        n = rng.integers(300, 3000)   # random length
        S = np.sqrt(2.0 * n)

        # initial a
        if best_seq is not None and n == len(best_seq):
            a = np.array(best_seq, copy=True)
        else:
            # start from random Dirichlet like distribution
            a = rng.random(n)
        a = _project_onto_simplex(a, S)

        # optimization loop
        beta = 0.5
        step = 0.1  # initial step size (in terms of magnitude of gradient)
        momentum = np.zeros_like(a)
        # compute time limit for this run based on remaining budget
        run_deadline = time.time() + min(30.0, deadline - time.time())

        while time.time() < run_deadline:
            conv = auto_convolve(a)
            max_val = conv.max()
            # Softmax weights
            # Use parameter beta scaled relative to max to avoid overflow
            diff = conv - max_val
            exp_term = np.exp(beta * diff)
            weight = exp_term / exp_term.sum()

            # gradient
            grad = 2.0 * np.correlate(a, weight, mode='full')
            # slice to length n
            L = len(weight)
            grad = grad[L-1:L-1+n]

            # momentum update
            momentum = 0.9 * momentum - step * grad

            a_new = a + momentum

            # projection onto simplex (nonneg, sum=S)
            a_new = _project_onto_simplex(a_new, S)

            a = a_new

            # evaluate occasionally, not every iteration
            if rng.random() < 0.1:
                try:
                    score = evaluate_sequence(a.tolist())
                except Exception:
                    score = float('inf')
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] New best score {best_score:.6f} n={n}")

            # adjust parameters: increase beta gradually to focus more on max
            beta = min(beta * 1.02, 100.0)
            # step adaptation based on max_val maybe
            step = step * 0.99

        # after run, also sample final a
        try:
            final_score = evaluate_sequence(a.tolist())
        except Exception:
            final_score = float('inf')
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Final run best {best_score:.6f} n={n}")

        # early stop if best already < target
        if best_score < 1.503:
            break

    if best_seq is None:
        # fallback: uniform
        n = 1000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)

    return best_seq.tolist()

"""
Propose a non‑negative step‑function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines three ideas:

1. **Simplex projection** – the sum of the coefficients is kept fixed to
   √(2 n).  This makes the objective equal to the maximal auto‑convolution
   entry, i.e.  objective = max(conv(a,a)).  The helper `_project_onto_simplex`
   implements an Euclidean projection onto {x ≥ 0, Σx = s}.

2. **Smoothed gradient descent** – at each iteration a soft‑max over the
   convolution entries is built (parameter β).  The gradient of the weighted
   sum is a correlation of the current vector with the weight vector.
   Momentum (μ≈0.9) and an adaptive step size are used.  β is slowly increased
   so that the optimisation first sees a “blurred’’ version of the max and
   finally concentrates on the true peak.

3. **Local stochastic refinement** – after the gradient phase a short
   simulated‑annealing style hill‑climb moves a small amount of mass from one
   index to another.  Moves are accepted if they improve the maximal
   convolution or with a small temperature‑dependent probability.

Multiple random restarts with different lengths (200 … 2500) and different
initial shapes (random, triangular, sinusoidal, or the previously known best
sequence) are performed until the time budget expires.  The best sequence
found is returned.
"""

import time
import numpy as np


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _auto_convolve(a: np.ndarray) -> np.ndarray:
    """
    Auto‑convolution a * a.
    For moderate lengths a direct NumPy convolution is used,
    otherwise an FFT based convolution is employed for speed.
    """
    n = a.shape[0]
    if n <= 1500:
        return np.convolve(a, a)
    # FFT path – next power of two for efficient transform
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries with temperature 1/β."""
    # conv is a 1‑D array of length 2n‑1
    # subtract max to avoid overflow
    max_c = conv.max()
    ex = np.exp(beta * (conv - max_c))
    s = ex.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s


def _gradient_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k (a*a)_k with respect to a.
    It is 2 · corr(a, w) where corr denotes the (non‑reversed) correlation.
    """
    L = weights.size
    n = a.size
    # full correlation, then slice the central part
    corr = np.correlate(a, weights, mode="full")
    start = L - 1
    grad = 2.0 * corr[start : start + n]
    return grad
