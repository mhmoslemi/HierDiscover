def subgradient_step(a, target_sum, lr, eps_frac=0.001):
    n = a.shape[0]
    S = a.sum()
    if S == 0:
        return a.copy()
    conv = np.convolve(a, a)
    M = conv.max()
    # Determine active set indices where conv close to M
    eps = M * eps_frac
    active = np.where(conv >= M - eps)[0]
    # Compute average reversed contributions for each p.
    # For each active k, compute a_{k-p} contributions.
    rev = np.zeros_like(a)
    count = np.zeros_like(a, dtype=int)  # to sum contributions count per entry maybe not needed; average across active set same length.

    # Equivalent: rev[p] = average over active k of a_{k-p} if within bounds; else 0.
    # So rev = (1/len(active)) * sum_{k in active} shift a by k.
    for k in active:
        j = k - np.arange(n)  # j = k - p
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev = rev / len(active)

    # Gradient: (4n / S^2) * rev - (4n * M / S^3)
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new = a_new * (target_sum / sum_new)
    else:
        a_new = a.copy()
    np.clip(a_new, 0, 1000, out=a_new)
    return a_new

def _subgradient_step(a, target_sum, lr, eps_frac=0.001, max_active=5):
    n = a.shape[0]
    S = a.sum()
    conv = np.convolve(a, a)
    M = conv.max()
    eps = M * eps_frac
    active = np.where(conv >= M - eps)[0]
    if active.shape[0] > max_active:
        # pick top ones by conv value: sort descending and take max_active
        top_idx = np.argsort(-conv[active])[:max_active]
        active = active[top_idx]
    rev = np.zeros_like(a)
    for k in active:
        j = k - np.arange(n)
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active.shape[0])
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new = a_new * (target_sum / sum_new)
    else:
        a_new = a.copy()
    np.clip(a_new, 0, 1000, out=a_new)
    return a_new

"""
Search for a low‑peak step‑function.

The objective returned by ``evaluate_sequence`` equals the maximal entry
of the discrete self‑convolution (after normalising the sum to
√(2 n)).  To reduce this value we perform a *sub‑gradient descent* that
uses several near‑maximising convolution indices (an active set) to
compute a smoother descent direction.  The update is projected onto the
feasible box 0 ≤ a_i ≤ 1000 and the total sum is re‑normalised after each
step.  During the run occasional random perturbations are injected to
escape shallow minima.  The best sequence seen before the time budget
expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_with_max(a: np.ndarray):
    """
    Compute the full self‑convolution of ``a`` and return the array,
    its maximum value and the index of that maximum.
    """
    b = np.convolve(a, a)
    max_val = b.max()
    max_idx = int(b.argmax())
    return b, max_val, max_idx


# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray,
                      target_sum: float,
                      lr: float,
                      eps_frac: float = 0.001,
                      max_active: int = 5) -> np.ndarray:
    """
    One projected sub‑gradient step.

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative vector (already normalised to ``target_sum``).
    target_sum : float
        Desired ℓ₁‑norm (√(2 n)).
    lr : float
        Learning rate.
    eps_frac : float
        Fraction defining the active set: all convolution entries ≥
        (1‑eps_frac)·M are considered active.
    max_active : int
        Upper bound on the number of active indices used to form the
        sub‑gradient (averaged).

    Returns
    -------
    np.ndarray
        Updated vector, still respecting the constraints.
    """
    n = a.shape[0]
    S = a.sum()
    if S == 0.0:
        return a.copy()

    # Convolution and its maximum
    conv, M, _ = _conv_with_max(a)

    # Build the active set (indices whose value is close to the maximum)
    eps = M * eps_frac
    active = np.where(conv >= M - eps)[0]

    # Limit the active set size for speed – keep the top values
    if active.size > max_active:
        # take the indices of the largest ``max_active`` entries
        top = np.argpartition(-conv[active], max_active - 1)[:max_active]
        active = active[top]

    # Average reversed contributions over the active set:
    #   rev[p] = average_{k∈active} a_{k‑p}   (0 if out of range)
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx                # the source index for each position p
        mask = (j >= 0) & (j < n)  # stay in bounds
        rev[mask] += a[j[mask]]
    rev /= max(1, active.size)     # avoid division by zero

    # Gradient of the objective (see the analysis above)
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

    # Gradient step, projection onto the non‑negative orthant,
    # and then renormalisation to keep the ℓ₁‑norm fixed.
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)

    sum_new = a_new.sum()
    if sum_new > 0.0:
        a_new = a_new * (target_sum / sum_new)
    else:
        a_new = a.copy()           # fallback – should rarely happen

    # Respect the hard limits of ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


# ----------------------------------------------------------------------
def _perturb_step(a: np.ndarray,
                  target_sum: float,
                  sigma: float) -> np.ndarray:
    """
    Apply an additive Gaussian perturbation to ``a`` while keeping the
    sum fixed and the entries non‑negative.
    """
    a_new = a + np.random.randn(a.shape[0]) * sigma
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.0:
        a_new = a_new * (target_sum / s)
    else:
        a_new = a.copy()
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Main entry point for the search.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget (seconds).  The function returns before
        ``budget_s`` expires.
    kwargs : dict
        Unused – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best height sequence found within the time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # Keep a small safety margin for housekeeping.
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise the starting vector.
    try:
        # Use the best known construction if it exists.
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        # Fallback: a uniform vector of moderate length.
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)         # the normalisation used by the paper
    a = base.copy()
    a = a * (target_sum / a.sum())          # normalise to the fixed ℓ₁‑norm

    # Initial evaluation.
    cur_score = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    iteration = 0
    lr0 = 0.7                               # initial learning rate for GD
    decay = 5e-5                            # decay of the learning rate
    sigma0 = target_sum * 0.05              # initial noise amplitude for perturbations
    eps_frac = 0.001                        # active‑set tolerance
    max_active = 5                          # maximum size of the active set
    jitter_interval = 5000                  # how often to add a tiny jitter

    # Main optimisation loop
    while time.time() < deadline:
        # How far are we into the allotted time?
        elapsed = time.time() - start
        time_frac = elapsed / (budget_s - 5.0)

        # --------------------------------------------------------------
        # Choose the update rule.
        if time_frac < 0.75:
            # Sub‑gradient descent with an adaptive learning rate.
            lr = lr0 / (1.0 + decay * iteration)
            a_new = _subgradient_step(a, target_sum, lr,
                                      eps_frac=eps_frac,
                                      max_active=max_active)
        else:
            # Late‑stage random exploration with decreasing noise.
            sigma = sigma0 * max(0.0, 1.0 - time_frac)
            a_new = _perturb_step(a, target_sum, sigma)

        # --------------------------------------------------------------
        # Evaluate the candidate.
        try:
            new_score = evaluate_sequence(a_new.tolist())
        except Exception:   # safety net – should not happen
            new_score = float("inf")

        # --------------------------------------------------------------
        # Keep the best ever seen.
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[iter {iteration:07d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Simple acceptance rule: always accept improvements;
        # occasionally accept a worse move to escape tiny plateaus.
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if np.random.rand() < 0.001:   # tiny probability of uphill move
                accept = True

        if accept:
            a = a_new
            cur_score = new_score

        # --------------------------------------------------------------
        # Every few thousand iterations add a minuscule jitter to avoid
        # stagnation in flat regions.
        if iteration % jitter_interval == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[iter {iteration:07d}] after jitter new best = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
