    conv = np.convolve(a, a)  # length 2n-1
    # Compute softmax weights:
    maxc = np.max(beta * conv)

def proj_simplex(v):
    # project onto simplex {x >=0, sum(x) = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v):
    # v is np.ndarray
    # Implementation based on "Projection onto a Simplex"
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the rho where u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def convolve(a, b):
    size = a.shape[0] + b.shape[0] - 1
    n = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=n)
    B = np.fft.rfft(b, n=n)
    C = A * B
    conv = np.fft.irfft(C, n=n)[:size]
    return conv
```

    return a

def proj_simplex(v):
    """Project onto probability simplex (non-negative, sum=1)."""
    # Based on algorithm from Chen & Ye (2011).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All elements zero? Return uniform.
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(a, b):
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:size]
    return conv

def gradient_step(a, beta):
    """
    One gradient descent step for the smoothed max objective.
    Returns gradient (numpy array) and current smooth max value.

    max_c = np.max(conv)

    exp_vals = np.exp(beta * (conv - max_c))

propose_candidate

Search for a non‑negative step‑function whose self‑convolution has a small

smooth approximation of the max‑convolution (soft‑max with a large β).
After each optimisation a cheap pairwise‐mass‑redistribution hill‑climb is
performed.  The best sequence found (lowest ``evaluate_sequence`` value) is
returned.  The whole routine respects the 1000 s time budget.

# Helper: arcsine coefficients (the exact solution of (1‑z)^(-1/2))
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first n terms a_i = C(2i,i) / 4^i via a stable recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution (real inputs)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * b using an FFT (real‑valued)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:size]
    return conv

# One gradient step for the smooth‑max objective
# ----------------------------------------------------------------------
def _grad_step(a: np.ndarray, beta: float):
    """
    Compute gradient of the softened max of the self‑convolution.

    Returns
    -------
    grad : np.ndarray   – gradient w.r.t. a (same shape as a)
    smooth_max : float  – current value of the smooth max (optional for logging)

    n = a.shape[0]
    # a * a
    conv = _conv_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights, numerically stable
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / np.sum(exp_vals)       # shape (2n‑1), sum = 1

    # correlation a with w  →  gradient of smooth max = 2 * correlation
    rev_w = w[::-1]                        # reverse for correlation
    corr = _conv_fft(a, rev_w)             # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)       # indices that give correlation_i
    grad = 2.0 * corr[idx.astype(int)]

    # smooth max itself (useful for diagnostics)
    smooth_max = max_c + (1.0 / beta) * np.log(np.sum(exp_vals))
    return grad, smooth_max
