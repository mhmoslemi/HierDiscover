    for i in range(1, n):
        a[i] = a[i-1] * (2*(i-1) + 1) / (2.0 * i)
    a = a / a.sum()   # normalize to sum 1

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    current = a.copy()
    current_score = best_score

    start = time.time()
    iteration = 0
    # Main iterative LP based refinement
    if linprog is not None:
        while time.time() - start < budget_s - 5.0:
            iteration += 1
            conv = np.convolve(current, current)
            M = conv.max()
            # Build constraint matrix A_ub
            rows = np.arange(len(conv))[:, None]
            cols = np.arange(n)
            offset = rows - cols
            mask = (offset >= 0) & (offset < n)
            A_ub = np.zeros((len(conv), n), dtype=float)
            A_ub[mask] = current[offset[mask]]

            c = -np.ones(n)

            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,

            if not res.success:
                # cannot improve further
                break
            b = res.x
            sum_b = b.sum()
            if sum_b <= 0:
                break
            g = b / sum_b
            # Mixing step
            t = 0.3
            cand = (1 - t) * current + t * g
            # renormalize to sum 1 to avoid tiny drift
            cand = cand / cand.sum()
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[iter {iteration}] improved best_score={best_score:.6f}")

"""
propose_candidate
=================

Generate a non‑negative step‑function whose self‑convolution is as flat as
possible.  The construction starts from the analytically known “arcsine”
coefficients (the coefficients of (1‑z)^‑1/2) which give the optimal
asymptotic bound π/2 ≈ 1.5708.  To push the bound further we run a
Frank‑Wolfe‑style iteration:

* For the current profile f we compute M = max(f∗f).
* A linear program maximises Σ b_j under the linear constraints
      (f∗b)_k ≤ M   for all k
  and b_j ≥ 0.  The optimum b puts as much mass as possible on indices
  where f is small, i.e. where the convolution constraint is the weakest.
* After normalising b to have ℓ₁‑norm 1 we update
      f ← (1−t)·f + t·b   (t≈0.3) and re‑normalise.
  This mixing reduces the maximal convolution value and therefore the
  evaluation score 2·n·max(f∗f)/(∑f)².

If SciPy is unavailable a lightweight random‑hill‑climbing fallback is used.
The algorithm runs within the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper: arcsine coefficients (first n terms of (1‑z)^{-1/2})
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..n‑1 using the stable recurrence."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a

# ----------------------------------------------------------------------

    Produce a candidate sequence of non‑negative heights.

    Parameters

    budget_s : float, optional
        Time budget in seconds.
    seed : int, optional
        Random seed (used for the stochastic parts of the algorithm).

    list[float]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    # Pre‑compute the index mask that tells which entry of f participates in which
    # convolution constraint: (f∗b)_k = Σ_j f_{k‑j}·b_j.
    conv_len = 2 * n - 1
    rows = np.arange(conv_len)[:, None]          # shape (conv_len,1)
    cols = np.arange(n)                          # shape (n,)
    offset = rows - cols                         # (conv_len, n)
    valid_mask = (offset >= 0) & (offset < n)    # True where f_{k‑j} exists
    idx = offset.copy()
    idx[~valid_mask] = -1                        # -1 marks “no contribution”

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            # Linear programme: maximise Σ b_j  ⇔ minimise –Σ b_j

    final_score = evaluate_sequence(best_seq.tolist())   # pylint: disable=undefined-variable
    print(f"[propose_candidate] length={len(best_seq)}, final_score={final_score:.6f}")

    return best_seq.tolist()
