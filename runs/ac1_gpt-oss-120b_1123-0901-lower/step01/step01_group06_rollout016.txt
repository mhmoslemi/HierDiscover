"""
propose_candidate
=================

Iterative improvement of a non‑negative step function for the
self‑convolution objective.

The algorithm consists of three stages:

1. **Initialisation** – use the classic arcsine coefficients
   a_i = C(2i,i)/4^i (the “(1‑z)^(-1/2)’’ series) and normalise
   the vector to a probability distribution (sum = 1).

2. **Linear‑programming mixing** – given the current sequence `f`,
   let `M = max(conv(f,f))`.  Solve a linear programme that *maximises*
   the total mass of a new non‑negative sequence `g` under the
   constraints `conv(f,g) ≤ M` (element‑wise).  The LP solution tends to
   push mass to indices where `f` is small, therefore flattening the
   autocorrelation when it is mixed back into `f`.

   The mix `f ← (1‑t)·f + t·ĝ` (with a small step size `t`) never
   increases the maximal convolution value, and repeated mixing drives
   `M` down.  The LP is solved with ``scipy.optimize.linprog`` (HiGHS
   backend) using a sparse constraint matrix for speed.

3. **Light stochastic refinement** – after the LP phase a few random
   pairwise mass‑redistributions are tried; any improvement is kept.

The whole procedure respects the 1000 s time budget, keeps every entry
non‑negative, and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine coefficients (C(2i,i)/4^i)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """First ``n`` coefficients of the power series of (1‑z)^(-1/2)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Helper: build the sparse matrix A such that (A @ g)_k = (f * g)[k]
# ----------------------------------------------------------------------
def _make_sparse_conv_matrix(f: np.ndarray) -> "scipy.sparse.csr_matrix":
    """Return a (2n‑1)×n sparse matrix representing the convolution f * g."""
    from scipy import sparse
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    max_k = 2 * n - 1
    for j in range(n):
        # column j contributes f[i] to row i+j
        length = min(n, max_k - j)          # number of rows that fit
        rows.extend(range(j, j + length))
        cols.extend([j] * length)
        data.extend(f[:length])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(max_k, n), dtype=np.float64)
    return A


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------

    Produce a candidate step‑function (list of non‑negative heights).

    Parameters
    ----------
    budget_s : float
        Time budget in seconds.  The algorithm stops well before the limit.
    seed : int
        Random seed for the stochastic refinement phase.

    Returns
    -------
    list[float]
        The best sequence found (sum of the entries equals 1).
    """

    # Choose a problem size.

    # otherwise a moderate default (800) is used.

    # ------------------------------------------------------------------
    coeffs = _arcsine_coeffs(n).astype(np.float64)
    f = coeffs / coeffs.sum()          # sum(f) == 1.0

    max_lp_iters = 40

    from scipy import sparse

    while (time.time() - start_time) < (budget_s - 5.0) and lp_iter < max_lp_iters:
        lp_iter += 1

        # ----- compute current maximal self‑convolution value -----
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # ----- build the sparse constraint matrix (f * g) ≤ M -----
        A_sp = _make_sparse_conv_matrix(f)          # shape (2n‑1) × n
        b_ub = np.full(A_sp.shape[0], M, dtype=np.float64)

        # ----- linear programme: maximise sum(g) subject to (f * g) ≤ M -----
        c = -np.ones(n, dtype=np.float64)           # minimise -sum(g)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_sp, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success or res.x is None:
            # LP failed – abort the LP phase.
            print("[propose_candidate] LP failed at iteration", lp_iter,
                  "status:", res.message if hasattr(res, "message") else "unknown")
            break

        g_raw = res.x
        sum_g = float(g_raw.sum())
        if sum_g <= 0.0:
            # nothing useful found
            break

        # ----- create a normalised partner ĝ (sum == 1) -----
        g_hat = g_raw / sum_g    # now sum(g_hat) == 1

        # ----- convex combination -------------------------------------------------
        # We keep the total mass equal to one, so simply mix the two probability vectors.
        new_f = (1.0 - t) * f + t * g_hat

        # Numerical cleanup – ensure non‑negative and renormalise exactly.
        new_f[new_f < 0.0] = 0.0
        new_f /= new_f.sum()

            best_f = new_f.copy()
            print(f"[propose_candidate] LP iter {lp_iter}: improved score = {best_score:.6f}")

        # If the new sequence does not increase the maximal convolution,
        # we accept it (the theory guarantees non‑increase).  Otherwise we fall
        # back to the previous best.

            f = new_f
        else:
            # The mix accidentally increased the peak – reduce the step size.
            t *= 0.5
            if t < 1e-4:
                # step size too small, stop the LP phase.
                break

    # ------------------------------------------------------------------
    # 3. Light stochastic refinement (pairwise mass swaps).
    # ------------------------------------------------------------------
    max_stochastic_iters = 3000
    conv_len = 2 * n - 1
    for it in range(max_stochastic_iters):
        if (time.time() - start_time) >= (budget_s - 0.5):
            break

        # pick two distinct indices

        # move a tiny amount from i to j
        eps = min(best_f[i], 0.005) * np.random.rand() * 0.2
        if eps <= 0:
            continue
        trial = best_f.copy()
        trial[i] -= eps
        trial[j] += eps
        # re‑normalise to keep sum == 1 (tiny scaling does not affect the score)
        trial /= trial.sum()

        sc = evaluate_sequence(trial.tolist())   # pylint: disable=undefined-variable
        if sc < best_score:
            best_score = sc
            best_f = trial
            # occasionally tighten the epsilon to accelerate convergence
            if it % 200 == 0 and eps > 1e-6:
                eps *= 0.5

    # final logging
    try:
        final_score = evaluate_sequence(best_f.tolist())  # pylint: disable=undefined-variable
        print(f"[propose_candidate] final length = {n}, final_score = {final_score:.6f}")
    except Exception:
        pass

    return best_f.tolist()
