def beta_distribution(n, alpha):
    # symmetric beta with shape parameter alpha
    # generate x values on (0,1)
    i = np.arange(n, dtype=np.float64)
    # position of bin centre: (i+0.5)/n to avoid 0 and 1 exactly.
    x = (i + 0.5) / n
    # weight = x**(alpha-1) * (1-x)**(alpha-1)
    # for numerical stability, use np.exp((alpha-1)*(np.log(x)+np.log(1-x))) for alpha not 1.
    w = np.power(x, alpha-1) * np.power(1-x, alpha-1)
    total = w.sum()
    w /= total  # normalize sum=1
    return w

    n = len(x)

    conv = np.fft.irfft(fx * fx, n=size)

import time
import numpy as np

def beta_sequence(n, alpha):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid NaNs if alpha == 0
    a = alpha - 1.0
    w = np.power(x, a) * np.power(1.0 - x, a)
    # Replace any infinities (should be rare) with a large number
    # Though after normalization they become finite.
    w = np.where(np.isfinite(w), w, 0.0)
    total = w.sum()
    if total == 0.0 or not np.isfinite(total):
        w = np.ones(n, dtype=np.float64)
        total = n
    w /= total
    return w

def uniform_sequence(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def conv_max_fft(seq):
    n = seq.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fx = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    conv = conv[:2*n - 1]
    return conv.max(), conv

def score_from_max(n, max_conv):
    return float(2.0 * n * max_conv)

def evaluate_seq(seq):
    # using evaluate_sequence directly yields same result as score_from_max
    # but we prepend safe conversion.
    return evaluate_sequence(seq.tolist())

"""
Propose a step‑function (non‑negative sequence) that minimises the supplied
``evaluate_sequence``.  

The algorithm works in two phases:

1. **Parametric sweep** – a family of symmetric Beta(α,α) distributions is
   tried, optionally mixed with a small amount of uniform mass.  For each
   candidate the auto‑convolution is obtained with an FFT, its maximum is
   measured and the corresponding score ``2·n·max(conv)`` is kept.  This
   quickly finds a starting sequence that already improves on the classic
   arcsine construction.

2. **Targeted hill‑climbing** – the current sequence `x` and its exact
   convolution `conv = np.convolve(x,x)` are maintained.  In each iteration
   the index `k` where `conv` attains its maximum is selected; the pair
   `(i,j)` with the largest product `x[i]*x[j]` (i+j = k) is identified.
   A small amount of probability mass is moved from the larger of `i` or
   `j` to a random destination.  The convolution is updated in‑place using
   the exact quadratic update formula; the move is kept only if the score
   improves.  Periodically the convolution is recomputed from scratch to
   avoid drift.  The loop runs until the allotted time budget expires.

The whole procedure respects the ``budget_s`` argument (default 1000 s) and
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are defined at top level (no closures)
# ----------------------------------------------------------------------
def beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) distribution on n points.
    The grid points are (i+0.5)/n, i = 0…n‑1, to avoid the exact 0/1 singularities.
    """
    # positions in (0,1)
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = alpha - 1.0                     # exponent
    # use np.power; for negative a the values grow near the ends but stay finite
    w = np.power(x, a) * np.power(1.0 - x, a)
    # clean possible infinities (very extreme exponents)
    w = np.where(np.isfinite(w), w, 0.0)
    total = w.sum()
    if total == 0.0 or not np.isfinite(total):
        # fallback to uniform – should never happen for the ranges we use
        w = np.ones(n, dtype=np.float64)
        total = n
    w /= total
    return w


def uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution on n points (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def conv_max_fft(seq: np.ndarray) -> (float, np.ndarray):
    """
    Returns (max(conv(seq,seq)), full convolution) using an FFT.
    Complexity O(n log n) and works for any length.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    fx = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    conv = conv[: 2 * n - 1]                         # exact length
    return conv.max(), conv


def score_from_max(n: int, max_conv: float) -> float:
    """The objective value produced by ``evaluate_sequence`` (sum = 1)."""
    return float(2.0 * n * max_conv)

            dst += 1

        # Step size – decays linearly with the number of performed iterations
        frac = base_step * (1.0 - it / max_iters)
        delta = frac * x[src]
        if delta <= 1e-15:
            it += 1
            continue

        # ----------- tentative update of the convolution (in‑place) ----------
        # subtract contribution of the source
        conv[src : src + n] -= 2.0 * delta * x
        # add contribution of the destination
        conv[dst : dst + n] += 2.0 * delta * x
        # quadratic correction terms
        conv[2 * src] += delta * delta
        conv[2 * dst] += delta * delta
        conv[src + dst] -= 2.0 * delta * delta

        new_max = conv.max()
        new_score = score_from_max(n, new_max)

        if new_score < best_score:
            # Accept the move
            x[src] -= delta
            x[dst] += delta
            best_score = new_score
            best_seq = x.copy()
            print(f"[improve] it={it:06d}  score={best_score:.6f}")
        else:
            # Revert the tentative change
            conv[src : src + n] += 2.0 * delta * x
            conv[dst : dst + n] -= 2.0 * delta * x
            conv[2 * src] -= delta * delta
            conv[2 * dst] -= delta * delta
            conv[src + dst] += 2.0 * delta * delta
            # x stays unchanged

        it += 1

        # Every few thousand steps recompute the convolution from scratch
        # to suppress accumulated rounding errors.
        if it % 10_000 == 0:
            conv = np.convolve(x, x)

    # --------------------------------------------------------------
    # 4. Final evaluation (only for logging – the returned list is used)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] iterations={it:,}  best_score={best_score:.6f}  evaluator={final_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found.
    # --------------------------------------------------------------
    return best_seq.tolist()
