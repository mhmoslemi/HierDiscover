        # we want real convolution length 2*n-1
        f = np.fft.rfft(a, n*2)

        k_max = np.argmax(conv)

        rev_idx = k_max - indices
        mask = (rev_idx >=0) & (rev_idx < n)

        g[mask] = 2 * a[rev_idx[mask]]
        # diagonal fix
        diag = (2*indices == k_max)

def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math.

    # config
    n = 3428
    total_time = budget_s
    deadline = time.time() + budget_s - 0.1

    np.random.seed(seed)

    # initial shape
    p0 = 0.5
    gamma0 = 0.32
    a = (np.arange(n, dtype=float) + gamma0) ** (-p0)
    a = a / a.sum()  # normalized to sum=1

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print("Initial score:", best_score)

    # helpful function compute_conv and max
    L = 1 << ((2*n - 1).bit_length())
    def compute_conv(arr):
        fa = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        return conv

    # main loop
    it = 0
    while time.time() < deadline:
        conv = compute_conv(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # compute gradient g
        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros(n, dtype=float)
        g[mask] = 2.0 * a[rev[mask]]
        # correct diag
        diag = (2*idx == max_idx)
        if np.any(diag):
            g[diag] = a[idx[diag]]
        # highest and lowest gradient indices
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        # compute step fraction based on remaining time
        remaining = deadline - time.time()
        frac = max(0.001, 0.05 * remaining / total_time)
        eps = a[i_max] * frac
        if eps <= 0:
            break
        # transfer mass
        a[i_max] -= eps
        a[i_min] += eps
        # ensure nonnegativity (clamp)
        if a[i_max] < 0:
            a[i_min] += a[i_max]  # add back negative to i_min
            a[i_max] = 0.0
        # keep sum constant (float rounding may drift)
        # assume small drift negligible

        it += 1
        # evaluate occasionally
        if it % 200 == 0:
            # quick approximate score using max_val but note max_val might have changed after move; recompute often
            conv = compute_conv(a)
            max_val = conv.max()
            approx_score = 2.0 * n * max_val / (a.sum()**2)
            if approx_score < best_score:
                best_score = approx_score
                best_a = a.copy()

    # final evaluation
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished search – best score {final_score:.6f} after {it} iterations.")
    return best_a.tolist()

#!/usr/bin/env python
"""
Optimization of a non‑negative step function for the objective

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works with a *mass‑redistribution* heuristic that directly
minimises the quantity `max(conv(a,a))`.  Starting from a good known construction
(a power‑law  a_i = (i+γ)^(-0.5)  – the classical 1/√x shape) the code repeatedly
computes the convolution, finds the index where it attains its maximum, and then
moves a small amount of “mass’’ from the positions that contribute the most to
that peak to the positions that contribute the least.  Because the objective is
scale‑invariant the total mass is preserved (the vector is renormalised after each
step).  The move size slowly shrinks as the deadline approaches, giving a quick
coarse search first and a fine‑grained refinement later.

Only a tiny fraction of the time budget is spent calling the expensive
`evaluate_sequence`; the heavy work (FFT‑based convolution and a cheap
approximation of the score) is done thousands of times.  If a global variable
`height_sequence_1` is present it is used as the initial guess; otherwise the
power‑law seed is generated automatically.

The function returns a plain Python list that can be fed to `evaluate_sequence`.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

def _conv_fft(x: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of `x` with itself using FFT.
    `length` must be a power of two and >= 2*len(x)-1.
    Returns the first (2*len(x)-1) entries of the full convolution.
    """
    fx = rfft(x, n=length)
    conv_full = irfft(fx * fx, n=length)
    return conv_full[: 2 * len(x) - 1]


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative height sequence that (hopefully) beats the current
    best known score (≈1.571).  The routine respects a hard time limit
    (`budget_s` seconds) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # keep a small safety margin

    # ----------------------------------------------------------------------
    # 1. Problem size and initial guess
    # ----------------------------------------------------------------------
    n = 3428                     # length of the sequence (same as the reference)
    # use an existing construction if it exists, otherwise fall back to power law
    try:
        # height_sequence_1 should be a global list of floats
        seq0 = np.array(height_sequence_1, dtype=float)
        if len(seq0) != n:
            if len(seq0) > n:
                a = seq0[:n].copy()
            else:                           # pad with tiny values to reach length n
                pad = np.full(n - len(seq0), 1e-12, dtype=float)
                a = np.concatenate([seq0, pad])
        else:
            a = seq0.copy()
        # normalise – scaling does not affect the score
        a = a / a.sum()
        print("Started from global `height_sequence_1`.")
    except Exception:
        # power‑law seed: a_i = (i + γ)^(-p) with p≈0.5, γ≈0.32
        p0 = 0.5
        gamma0 = 0.32
        a = (np.arange(n, dtype=float) + gamma0) ** (-p0)
        a = a / a.sum()
        print("Started from 1/√x power‑law seed.")

    # best solution seen so far
    best_a = a.copy()
    best_score = float("inf")
    # a single evaluation of the starting point (relatively cheap)
    try:
        best_score = evaluate_sequence(best_a.tolist())
        print(f"Initial score (official evaluator) = {best_score:.6f}")
    except Exception as e:
        print("Evaluation failed on the initial sequence:", e)

    # ----------------------------------------------------------------------
    # 2. Helper objects (FFT length, etc.)
    # ----------------------------------------------------------------------
    # Length for the FFT – the next power of two not smaller than 2*n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    # 3. Main iterative improvement loop
    # ----------------------------------------------------------------------
    it = 0                     # iteration counter
    # keep track of when the last improvement happened
    last_improve_it = 0

    while time.time() < deadline:
        # ----- compute convolution and locate the current worst lag -----
        conv = _conv_fft(a, fft_len)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # ----- gradient of the max‑convolution w.r.t. the vector a -----
        # For a fixed lag k the derivative of (a*a)[k] w.r.t. a[i] is:
        #   2*a[k-i]   (if i ≠ k-i)   and   a[i] (if i = k-i)
        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]
        # correct the diagonal where i == k-i (appears only once in the sum)
        diag = (2 * idx == max_idx)
        if np.any(diag):
            g[diag] = a[diag]

        # ----- choose source (high gradient) and destination (low gradient) -----
        i_src = int(np.argmax(g))          # most detrimental index
        i_dst = int(np.argmin(g))          # most “free’’ index

        # ----- move a small amount of mass from src to dst -----------------
        # step size shrinks as we approach the deadline
        remaining = max(0.0, deadline - time.time())
        frac = max(0.001, 0.08 * remaining / budget_s)   # between 0.001 and 0.08
        eps = a[i_src] * frac
        if eps <= 0.0:
            # nothing to move – break to avoid endless loop
            break

        a[i_src] -= eps
        a[i_dst] += eps

        # renormalise to keep the L1‑norm exactly 1 (helps the cheap score)
        a /= a.sum()

        # ----- bookkeeping -------------------------------------------------
        it += 1

        # cheap score based on the current convolution (no extra FFT needed)
        cheap_score = 2.0 * n * max_val / (a.sum() ** 2)

        # store the best sequence seen so far (using cheap score is fine)
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            last_improve_it = it
            print(f"[{it:6d}] improved cheap score = {best_score:.6f}")

        # every few hundred iterations call the official evaluator (expensive)
        if it % 300 == 0:
            try:
                real_score = evaluate_sequence(a.tolist())
                if real_score < best_score:
                    best_score = real_score
                    best_a = a.copy()
                    last_improve_it = it

            except Exception as e:
                # a failure inside evaluate_sequence should never happen because
                # all numbers are non‑negative and the sum is far above 0.01,
                # but we guard against it just in case.
                print("Evaluation error during search:", e)

        # if we have gone a long time without improvement, perform a small
        # random shake‑up to try to escape flat plateaus
        if it - last_improve_it > 2000:
            # add a tiny Gaussian noise, keep non‑negativity, and renormalise
            noise = np.random.randn(n) * 0.0005
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            last_improve_it = it
            print(f"[{it:6d}] random shake‑up performed")

    # ----------------------------------------------------------------------
    # 4. Final evaluation and return
    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Search finished after {it} iterations.")
    print(f"Best sequence score (official evaluator) = {final_score:.6f}")

    # ensure a plain Python list is returned
    return best_a.tolist()
