propose_candidate
=================

A hybrid optimizer that starts from the discrete arcsine (Catalan) distribution
and repeatedly improves it by solving a *linear program* that maximises the
total mass that can be placed under the current ℓ∞‑convolution bound.
The LP provides a feasible direction `g`; a short line‑search mixes the old
distribution `f` with `g`.  After the LP phase a fast random‑hill‑climbing
phase fine‑tunes the result.

The whole search respects the time budget (default 1000 s) and always returns
a non‑negative list whose sum is 1, which makes the evaluation function
scale‑invariant.
"""

# cvxpy is optional – if unavailable we fall back to a pure hill‑climber.

    import cvxpy as cp
    _HAS_CVX = True

def _arcsine_distribution(n: int) -> np.ndarray:
    """Return the classic Catalan (discrete arcsine) coefficients of length n,
    normalized to sum to 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()
    return a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # This is the standard O(n log n) algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are negative – fall back to uniform.
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """Solve the LP  max 1ᵀb   s.t.   b ≥ 0  and  f∗b ≤ M.
    Returns the optimal b (unnormalised).  Returns ``None`` on failure."""
    n = f.shape[0]
    # Variable for the new distribution.
    b = cp.Variable(n, nonneg=True)
    # Full linear convolution f∗b (length 2n‑1).  f is a constant, b is a variable.
    conv_fb = cp.conv(cp.Constant(f), b)   # element‑wise linear in b
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [conv_fb <= M])
    try:
        # ECOS is a fast LP/SOCP solver available in the sandbox.
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:  # pragma: no cover
        return None
    if b.value is None:
        return None
    b_opt = np.asarray(b.value).reshape(-1)
    return b_opt


def _line_search_mix(f: np.ndarray, g: np.ndarray, budget: float,
                     start_score: float) -> (np.ndarray, float):
    """Try a few mixing ratios (1‑t)·f + t·g, keep the best improving one."""
    best_seq = f
    best_score = start_score
    # Candidate mixing coefficients – more weight on the new direction early on.
    candidates = [0.1, 0.2, 0.4, 0.6, 0.8]
    for t in candidates:
        h = (1 - t) * f + t * g
        h = _project_to_simplex(h)          # enforce non‑negativity & sum=1
        s = evaluate_sequence(h.tolist())
        if s < best_score:
            best_seq, best_score = h, s
    return best_seq, best_score


def _random_hill_climb(a: np.ndarray, best_score: float,
                       time_deadline: float) -> (np.ndarray, float):
    """Lightweight hill‑climbing that moves a tiny fraction of mass between
    random indices while the time budget permits."""
    n = a.shape[0]
    it = 0
    while time.time() < time_deadline:
        i, j = np.random.choice(n, size=2, replace=False)
        # move a fraction from the larger entry to the smaller one
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        # δ is a modest proportion (≈1 %) of the donor entry.
        delta = 0.01 * a[src]
        if delta <= 0:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        new_a = _project_to_simplex(new_a)
        s = evaluate_sequence(new_a.tolist())
        if s < best_score:
            a, best_score = new_a, s
            # occasional progress report
            if it % 1000 == 0:
                print(f"[hill‑climb] iter={it}, score={best_score:.6f}")
        it += 1
    return a, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises the
    supplied ``evaluate_sequence`` function.

    The algorithm:
        1. initialise with the discrete arcsine distribution,
        2. repeatedly run a linear‑programming direction step,
        3. perform a short random hill‑climb for fine‑tuning,
        4. respect the overall time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a tiny safety margin
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # 1. Choose a problem size.
    # -----------------------------------------------------------------
    # Prefer the length of a previously discovered sequence if available.
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)    # type: ignore[name-defined]
        except Exception:
            n = 2500                       # a moderate size that solves quickly
    # limit n to a reasonable size for LP solving
    n = max(200, min(4000, n))

    # -----------------------------------------------------------------
    # 2. Build the initial arcsine (Catalan) sequence.
    # -----------------------------------------------------------------
    f = _arcsine_distribution(n)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. LP‑based improvement loop.
    # -----------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline and lp_iter < 30 and _HAS_CVX:
        # Current ℓ∞ bound of the self‑convolution.
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # Solve the LP: maximise total mass while keeping f∗b ≤ M.
        b_opt = _lp_step(f, M)
        if b_opt is None:
            print("[lp] solver failed – exiting LP phase")
            break

        # Normalise the LP solution to a probability vector.
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            print("[lp] degenerate solution – exiting LP phase")
            break
        g = b_opt / sum_b

        # Mix f and g – try a few blending ratios, keep the best.
        new_f, new_score = _line_search_mix(f, g, deadline, best_score)

        if new_score < best_score - 1e-8:
            print(f"[lp] iter={lp_iter:02d}, improved score={new_score:.6f}")
            f, best_seq, best_score = new_f, new_f.copy(), new_score
        else:
            # No improvement – stop the LP phase.
            print(f"[lp] iter={lp_iter:02d}, no improvement, stopping.")
            break

        lp_iter += 1

    # -----------------------------------------------------------------
    # 4. Random hill‑climb fine‑tuning (fast, works even without cvxpy).
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        f, best_score = _random_hill_climb(f, best_score, deadline)
        best_seq = f.copy()

    # -----------------------------------------------------------------
    # 5. Final report and return.
    # -----------------------------------------------------------------
    print(f"[final] n={n}, best score={best_score:.6f}, iterations={lp_iter}")
    # Ensure the result respects the evaluator's range restrictions.
    final = np.clip(best_seq, 0.0, 1000.0).tolist()
    return final
