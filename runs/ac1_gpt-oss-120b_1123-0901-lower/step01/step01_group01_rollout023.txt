"""
Propose a step‑function (non‑negative sequence) minimizing
    2·n·max_k (a*a)[k] / (sum a)^2

The implementation builds a decreasing geometric seed, then refines it
with a projected ADAM optimiser (sub‑gradient of the active convolution
shift).  Several restarts with different decay factors are tried and the
best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
def _project_onto_simplex(v):
    """Project a vector onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the max‑convolution term
def _subgradient(seq):
    """
    Returns the sub‑gradient of the active convolution term and the index
    k where the convolution attains its maximum.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())
    grad = np.zeros_like(seq)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad[idx] = seq[k - idx]
    return grad, conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    supplied evaluation function.  The algorithm works inside the
    allocated time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) try a few problem sizes (larger sizes give more freedom)
    lengths = [2000, 3000, 4000, 5000]               # will stop early if out of time
    for n in lengths:
        if time.time() > deadline:
            break

        # ----------------------------------------------------------------
        # 2) build a few geometric seeds and keep the best one for this n
        local_best_seq = None
        local_best_score = float("inf")
        for factor in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
            if time.time() > deadline:
                break
            eps = factor / n
            r = 1.0 - eps
            if r <= 0.0:
                continue
            # geometric progression a_i = r**i, normalised to sum = 1
            powers = np.exp(np.log(r) * np.arange(n, dtype=np.float64))
            seq = powers / powers.sum()
            conv = np.convolve(seq, seq)
            score = 2.0 * n * conv.max() / (seq.sum() ** 2)
            if score < local_best_score:
                local_best_score = score
                local_best_seq = seq.copy()

        # fallback – uniform if something went wrong
        if local_best_seq is None:
            local_best_seq = np.full(n, 1.0 / n)

        # ----------------------------------------------------------------
        # 3) ADAM‑based projected sub‑gradient optimisation
        seq = local_best_seq.copy()
        m = np.zeros_like(seq)          # first moment
        v = np.zeros_like(seq)          # second moment
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        lr = 0.15                       # base learning rate
        it = 0

        while time.time() < deadline:
            grad, cur_max = _subgradient(seq)

            it += 1
            # ADAM update (minimisation, so move opposite to the gradient)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            new_seq = seq - step
            new_seq = _project_onto_simplex(new_seq)

            # compute objective on the new candidate
            conv = np.convolve(new_seq, new_seq)
            score = 2.0 * n * conv.max() / (new_seq.sum() ** 2)

            # keep track of the best we have seen for this length
            if score < local_best_score:
                local_best_score = score
                local_best_seq = new_seq.copy()
                print(f"[n={n:04d}] iter {it:05d} new local best = {score:.6f}")

            # occasional random jitter to help escape plateaus
            if it % 500 == 0:
                noise = np.random.randn(n) * 0.005
                jitter = _project_onto_simplex(new_seq + noise)
                conv_j = np.convolve(jitter, jitter)
                score_j = 2.0 * n * conv_j.max() / (jitter.sum() ** 2)
                if score_j < local_best_score:
                    new_seq = jitter
                    local_best_score = score_j
                    local_best_seq = jitter.copy()
                    print(f"[n={n:04d}] iter {it:05d} after jitter new best = {score_j:.6f}")

            seq = new_seq

            # early exit if we already beat the target comfortably
            if local_best_score < 1.45:
                break

        # ----------------------------------------------------------------
        # 4) compare with the global best and keep the improvement
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"*** New global best = {best_score:.6f} at length {n}")

        # stop if the target is reached
        if best_score < 1.503:
            break

    # ----------------------------------------------------------------------
    # final reporting
    if best_seq is None:
        # should never happen – return a trivial uniform sequence
        best_seq = np.full(1, 1.0)

    print(f"Final best objective = {best_score:.6f} (length {len(best_seq)})")
    # evaluator expects a Python list, not a NumPy array
    return best_seq.tolist()
