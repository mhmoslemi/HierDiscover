"""
Propose a candidate sequence that improves on the classic 1/√i construction.

Algorithm
---------
1. Choose a length `n` (default: the length of the previously found
   `height_sequence_1`, otherwise 2500).
2. Parameterise the step heights by a simple power law
        a_i = (i+1+δ)^{-p} · exp(-β·i)        (i = 0,…,n‑1)
   with three continuous parameters:
        p   – exponent (≈0.5…0.8)
        δ   – small shift to avoid a zero at i=0
        β   – tiny exponential cut‑off (optional)
   For a fixed `p` the ratio 2·n·max(conv)/(∑a)² is (in the continuous
   limit) `2·B(1-p,1-p)·(1-p)²`, which is minimised around `p≈0.6`,
   giving a theoretical score ≈1.35 – well below the target 1.503.
3. Perform a cheap grid search over `p` and `δ`, then a short random
   hill‑climbing of the three parameters while time allows.
4. Starting from the best parametric sequence, run a fast per‑element
   stochastic refinement (multiplicative perturbations of single
   entries) to climb out of any remaining local minima.
5. Return the best sequence found before the time budget expires.

The whole optimisation runs well within the 1000 s budget and never
produces invalid values (all entries are non‑negative, finite and far
below the 1000‑cap built into `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a power‑law sequence with optional exponential decay
# ----------------------------------------------------------------------
def generate_power_sequence(n: int, p: float, shift: float = 0.0,
                            beta: float = 0.0) -> list[float]:
    """
    a_i = (i+1+shift)^{-p} * exp(-beta * i)
    i runs from 0 to n‑1.
    All numbers are positive floats.
    """
    # i+1 guarantees a finite value for i=0 even when shift=0
    idx = np.arange(n, dtype=np.float64) + 1.0 + shift
    seq = idx ** (-p)
    if beta != 0.0:
        seq *= np.exp(-beta * np.arange(n, dtype=np.float64))
    return seq.tolist()

            best_seq = cand_seq
            best_p, best_shift, best_beta = p, shift, beta
            print(f"[rand] p={p:.4f} shift={shift:.3f} beta={beta:.6f} score={cand_score:.6f}")

    # --------------------------------------------------------------
    # 5. Fine‑grained per‑element stochastic improvement
    # --------------------------------------------------------------
    # Use the classic cheap multiplicative perturbation of single entries.
    # This phase runs until the overall budget is exhausted.
    best_seq = list(best_seq)                     # ensure mutable Python list
    best_score = float(best_score)

    t_end = time.time() + budget_s - 0.2          # tiny safety margin
    while time.time() < t_end:
        i = rng.integers(n)                      # pick a random index
        old_val = best_seq[i]

        # log‑normal perturbation: keeps the entry non‑negative
        factor = np.exp(rng.normal(0.0, 0.012))
        new_val = max(0.0, old_val * factor)

        cand = best_seq.copy()
        cand[i] = new_val
        cand_score = evaluate_sequence(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            # occasional progress report
            print(f"[improve] idx={i} new_score={cand_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
    return list(best_seq)
