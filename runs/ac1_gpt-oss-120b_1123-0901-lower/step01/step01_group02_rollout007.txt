def conv_max_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    max_val = conv.max()
    return max_val, conv
```

def compute_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    # due to floating errors, small negative values may appear; set to max(0)
    # conv = np.maximum(conv, 0)
    M = conv.max()
    # find indices where conv close to M
    mask = conv >= M - 1e-12 * np.abs(M) - 1e-12  # tolerance
    kmax_indices = np.where(mask)[0]
    return M, kmax_indices, conv

"""
Hybrid optimizer for the step‑function bound.

The score to minimise is

    E(a) = 2 * n * max_k (a*a)[k] / (∑ a_i)^2 .

The objective is homogeneous, therefore we keep the L1‑norm fixed to
S = sqrt(2*n) (the normalisation used in the reference paper).  The
algorithm combines three ingredients:

1.  **Fast convolution** – the convolution and its maximum are obtained
    with an FFT (O(n log n)) instead of the O(n²) `np.convolve`
    used by `evaluate_sequence`.  This makes it possible to run many
    thousands of internal iterations inside the time budget.

2.  **Adam‑style projected gradient descent** – the exact sub‑gradient of
    `E` (derived in the reference code) is computed from the current
    arg‑max of the convolution.  Adam’s adaptive moments give a robust
    step size, after which the vector is projected onto the non‑negative
    orthant and renormalised to keep the sum equal to `S`.

3.  **Periodic global improvement via a linear programme** – given the
    current vector `f`, we solve

        maximise   ∑ g_i
        subject to  (f * g)_k ≤ M_f   for all k,
                    g_i ≥ 0.

    where `M_f` is the current maximum convolution value.
    The solution `g` is scaled to the target sum `S` and mixed into the
    current iterate (`f ← (1‑t)·f + t·g`).  This step often yields a
    large drop in the objective and helps escape local minima.

The optimiser runs until the supplied wall‑clock budget expires,
keeping the best external score (the true `evaluate_sequence` value)
found so far.  Random jitter is added occasionally to avoid stagnation.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper: fast convolution (FFT) and max value
def _conv_max_fft(a: np.ndarray):
    """
    Return (M, k_max_indices, full_conv) where
        M  – maximum of the convolution a*a,
        k_max_indices – all indices where the maximum is attained,
        full_conv – the whole convolution array (length 2n‑1).
    """
    n = a.shape[0]
    # size for a power‑of‑two FFT (fast & accurate)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values
    conv = np.maximum(conv, 0.0)
    M = conv.max()
    # tolerance for “equal” maxima
    atol = 1e-12
    rtol = 1e-12
    mask = np.abs(conv - M) <= atol + rtol * np.abs(M)
    k_max_idx = np.where(mask)[0]
    return float(M), k_max_idx, conv


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the evaluation function
def _subgradient(a: np.ndarray, S: float, n: int):
    """
    Compute the sub‑gradient of E at a, the current sum S and length n.
    Returns (grad, M, k_max_idx).
    """
    M, k_max_idx, _ = _conv_max_fft(a)
    # Build the averaged reversed vector a_{k_max-p}
    rev = np.zeros_like(a)
    for k in k_max_idx:
        j = k - np.arange(n)                # indices of a that contribute
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev = rev / max(1, len(k_max_idx))      # average over all maximising shifts

    # Gradient derived in the reference paper
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))
    return grad, M, k_max_idx


# ----------------------------------------------------------------------
# Helper: internal (fast) evaluation of the objective (no clipping)
def _internal_score(M: float, n: int, S: float) -> float:
    """Return the value of E given max‑convolution M, length n and sum S."""
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------------------
# Helper: build the convolution matrix for the LP (sparse)
def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Return the matrix A (shape (2n‑1, n)) such that A @ g = f * g.
    Row k contains f_{k‑j} at column j whenever 0 ≤ k‑j < n.
    """
    n = f.shape[0]
    # Off‑sets 0 … n‑1 correspond to the required anti‑diagonals.
    offsets = np.arange(n, dtype=int)
    # `sp.diags` repeats the same data for each offset – exactly what we need.
    A = sp.diags(f, offsets=offsets, shape=(2 * n - 1, n), format="csc")
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence minimising `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget (seconds).  The function stops a few seconds
        before the limit to ensure a return value.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a safety margin for final I/O and cleanup
    deadline = start_time + budget_s - 4.0

    # --------------------------------------------------------------
    # Initialise a starting sequence (use the SOTA construction if present)
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.clip(a, 0.0, 1000.0)
    else:
        # fallback: uniform vector
        a = np.ones(2100, dtype=float)

    n = a.shape[0]
    S_target = math.sqrt(2.0 * n)          # fixed L1‑norm
    a = a * (S_target / a.sum())           # normalise

    # --------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.5                               # base learning rate
    lr_decay = 1e-5                         # slow decay

    # --------------------------------------------------------------
    # Record the best external score seen so far
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:                     # safety, should never happen
        best_score = float("inf")
    # internal score for quick comparisons
    M0, _, _ = _conv_max_fft(a)
    best_internal = _internal_score(M0, n, S_target)

    print(f"[init] n={n}, internal={best_internal:.6f}, external={best_score:.6f}")

    iteration = 0
    last_lp = 0
    # main optimisation loop
    while time.time() < deadline:
        iteration += 1

        # ----- Adam‑style projected gradient step ------------------
        grad, M_cur, _ = _subgradient(a, S_target, n)

        # Learning rate schedule (slow decay)
        lr = lr0 / (1.0 + lr_decay * iteration)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # Projection onto the feasible set (non‑negative, clipped)
        a_new = np.maximum(a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Renormalise to keep the sum fixed
        sum_new = a_new.sum()
        if sum_new > 0.0:
            a_new = a_new * (S_target / sum_new)
        else:
            a_new = a.copy()   # fallback – should be rare

        a = a_new

        # ----- occasional jitter to escape plateaus ---------------
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (S_target * 0.001 / n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (S_target / a.sum())

        # ----- periodic global LP improvement ----------------------
        # we perform an LP every ~10 000 iterations (or after a long pause)
        if iteration - last_lp > 10000:
            last_lp = iteration
            # Build the convolution matrix for the current a
            A = _build_conv_matrix(a)
            b_ub = np.full(2 * n - 1, M_cur, dtype=float)

            # maximise sum(g)  ⇔ minimise -sum(g)
            c = -np.ones(n, dtype=float)
            bounds = [(0.0, 1000.0) for _ in range(n)]

            # Solve with HiGHS (fast sparse LP solver)
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": 30},
            )
            if res.success:
                g0 = res.x
                sum_g0 = g0.sum()
                if sum_g0 > 0.0:
                    # Scale to the target L1‑norm
                    g = g0 * (S_target / sum_g0)
                    # Mix a little bit of the new direction
                    t = 0.25
                    a = (1.0 - t) * a + t * g
                    a = np.clip(a, 0.0, 1000.0)
                    a = a * (S_target / a.sum())
                    # Reset Adam moments because we changed the point drastically
                    m.fill(0.0)
                    v.fill(0.0)

        # ----- internal evaluation (fast) -------------------------
        if iteration % 2000 == 0 or iteration % 10000 == 0:
            M_now, _, _ = _conv_max_fft(a)
            internal_score = _internal_score(M_now, n, S_target)

            # If we improved internally, ask the true evaluator
            if internal_score < best_internal - 1e-8:
                best_internal = internal_score
                best_seq = a.copy()
                try:
                    external = evaluate_sequence(best_seq.tolist())
                except Exception:
                    external = float("inf")
                if external < best_score:
                    best_score = external
                    print(
                        f"[iter {iteration:06d}] NEW EXTERNAL BEST = {best_score:.6f}"
                    )
                else:
                    # internal improvement only
                    print(
                        f"[iter {iteration:06d}] internal improve = {internal_score:.6f}"
                    )

    # final report
    print(f"[done] iterations={iteration}, best external={best_score:.6f}")
    return best_seq.tolist()
