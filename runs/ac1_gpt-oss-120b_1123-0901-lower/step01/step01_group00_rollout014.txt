"""
Propose a step‑function (non‑negative vector) that drives the evaluation
function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

as low as possible.

The algorithm is a *soft‑max weighted gradient descent* on the
autocorrelation (a*a).  At each iteration we

1. compute the full convolution,
2. form a soft‑max distribution over the convolution entries
   (temperature β is increased slowly – a continuation scheme),
3. compute the gradient  ∇L = 2·β·∇_a Σ p_k·(a*a)[k]  where p_k are the
   soft‑max weights,
4. take a step opposite to the gradient,
5. project back onto the simplex {x_i≥0, Σx_i = √(2n)},
6. enforce symmetry (a_i = a_{n‑1‑i}) – the optimum is known to be
   symmetric for this problem.

Many independent restarts with different lengths and initialisations are
performed until the time budget runs out.  The best sequence found so far
is kept and finally returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i >= 0, sum(x_i) = s }.
    Implementation of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    For a vector `a` of length n and a weight vector `w` of length 2n‑1
    (typically a soft‑max over the convolution entries),
    returns the gradient of Σ_k w_k·(a*a)[k] w.r.t. `a`.
    """
    n = a.shape[0]
    # convolution with the reversed weight vector
    rev = w[::-1]
    full = np.convolve(rev, a)               # length = 3n‑2
    idx = 2 * n - 2 - np.arange(n)           # indices that give Σ_j w_{i+j} a_j
    grad = full[idx]
    return grad


def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial non‑negative vector of length n."""
    if kind == "prev" and "height_sequence_1" in globals():
        src = np.asarray(height_sequence_1, dtype=float)
        if src.size >= n:
            init = src[:n].copy()
        else:
            pad = np.random.rand(n - src.size) * 1e-3
            init = np.concatenate([src, pad])
    elif kind == "triangle":
        # symmetric triangular shape
        half = np.arange(1, n // 2 + 1, dtype=float)
        if n % 2 == 0:
            seq = np.concatenate([half, half[::-1]])
        else:
            seq = np.concatenate([half, half[-2::-1]])
        init = seq[:n]
    elif kind == "exp":
        scale = n / 10.0
        init = np.exp(-np.arange(n) / scale)
    else:                                   # random
        init = np.random.rand(n)
    return init


# ----------------------------------------------------------------------
# Core optimiser for a fixed length
# ----------------------------------------------------------------------


def _run_optimize(
    n: int,
    init: np.ndarray,
    global_deadline: float,
    current_best: float,
    max_run_seconds: float = 8.0,
) -> tuple[np.ndarray, float]:
    """
    Perform a soft‑max weighted gradient descent on a vector of length ``n``.
    The routine stops when ``max_run_seconds`` is exhausted or the global
    deadline is reached.  Returns the best local sequence and its score.
    """
    target_sum = np.sqrt(2.0 * n)               # scale used by the SOTA method
    a = _project_onto_simplex(init.astype(float), target_sum)

    # enforce symmetry from the start
    a = (a + a[::-1]) * 0.5
    a = _project_onto_simplex(a, target_sum)

    best_local_seq = a.copy()
    # local score computed analytically (same as evaluate_sequence for admissible a)
    conv = np.convolve(a, a)
    max_conv = conv.max()
    best_local_score = 2.0 * n * max_conv / (target_sum ** 2)

    # hyper‑parameters
    base_step = 0.7            # raw step size before scaling
    beta = 0.4                 # temperature for the soft‑max
    beta_max = 5.0
    iter_cnt = 0
    last_improve = time.time()

    run_deadline = min(global_deadline, time.time() + max_run_seconds)

    while time.time() < run_deadline:
        iter_cnt += 1

        # full convolution and its maximum
        conv = np.convolve(a, a)
        max_conv = conv.max()

        # ------- soft‑max over the convolution entries -------
        # stability: subtract the maximum before exponentiation
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        probs = exp_vals / exp_vals.sum()            # Σ probs = 1

        # ------- gradient of the smoothed max -------
        grad_raw = _gradient_from_weights(a, probs)  # derivative of Σ probs·conv
        grad = 2.0 * beta * grad_raw                 # chain‑rule factor β

        # keep the gradient symmetric
        grad = (grad + grad[::-1]) * 0.5

        # step size adaption (similar to the original code)
        max_grad = np.max(grad) if np.max(grad) > 0 else 1.0
        eta = base_step * max_conv / max_grad

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # enforce symmetry after projection
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake‑up to avoid stagnation
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = a_new + noise
            a_new = _project_onto_simplex(a_new, target_sum)
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = _project_onto_simplex(a_new, target_sum)
            base_step *= 0.9                # slightly shrink step after a shake

        a = a_new

        # compute current objective (exact, no call to evaluate_sequence)
        conv = np.convolve(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv / (target_sum ** 2)

        if score < best_local_score:
            best_local_score = score
            best_local_seq = a.copy()
            last_improve = time.time()
            print(
                f"[{time.time():.2f}]  local new best = {score:.6f} (n={n}, β={beta:.2f})"
            )

        # increase temperature slowly (continuation)
        if beta < beta_max:
            beta *= 1.003

        # if we have not improved for a while, cut the run short
        if time.time() - last_improve > 4.0:
            # no improvement for ~4 s → stop this run
            break

    return best_local_seq, best_local_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function that minimises the given evaluation function.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = float("inf")

    # possible initialisation strategies
    init_kinds = ["random", "triangle", "exp", "prev"]

    while time.time() < deadline:

        kind = np.random.choice(init_kinds)

        # run the optimiser on this restart (cap each run to ≤8 s)
