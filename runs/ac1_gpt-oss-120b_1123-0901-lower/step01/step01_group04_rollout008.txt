"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that beats the
classical arcsine construction for the supplied ``evaluate_sequence``.
The algorithm proceeds as follows

1.  Choose a length ``n`` (prefer the length of the global
    ``height_sequence_1`` if it exists).  Several start‑vectors are built:
       * the exact arcsine coefficients,
       * the same coefficients raised to a small power (to tilt the shape),
       * a few random Dirichlet vectors.
2.  For each start‑vector a lightweight “gradient‑guided hill‑climb’’ is run.
    At every iteration the self‑convolution of ``a`` is computed by FFT.
    The indices where the convolution attains its maximal value form the
    active set.  From the sub‑gradient of the max‑convolution we obtain a
    weight vector ``w`` that tells which coordinates contribute most to the
    current peak.  A small amount of mass is transferred from the coordinate
    with the largest weight to the one with the smallest weight, the move being
    accepted only if the objective improves.  The step size is gradually
    reduced; occasional random perturbations help to escape shallow plateaus.
3.  The best sequence seen over all starts is returned (as a plain Python list).

The objective being minimised is
    2·n·max(conv(a,a)) / (sum(a)²)
which is invariant under scaling – we keep the vector normalised to a
probability distribution (sum = 1).  All operations stay in the safe numeric
range, so the local evaluation (computed directly from the convolution) is
equivalent to calling the provided ``evaluate_sequence``.

The routine respects the 1000 s time budget and prints a short progress log.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    # reference: https://arxiv.org/pdf/1101.6081.pdf  (Algorithm 1)
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑d arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero → return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][ -1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length power‑of‑2."""
    n = a.shape[0]
    L = 2 * n - 1
    # next power of two >= L (FFT length)
    N = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:L]
    return conv.real


def _conv_max_and_full(a: np.ndarray):
    """Convenient wrapper returning (max(conv), full convolution)."""
    conv = _full_convolution(a)
    return conv.max(), conv


def _arcsine_raw(n: int) -> np.ndarray:
    """Un‑normalised arcsine coefficients a_k = C(2k,k)/4^k, k=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return a


def _weight_from_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution (ignoring the constant factor 2).
    For each active convolution index i we add a[i‑j] to w[j] whenever the
    pair (j,i‑j) contributes to that convolution entry.
    """
    n = a.shape[0]
    w = np.zeros(n, dtype=np.float64)
    for i in active_idx:
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        # vector of j values participating in conv[i]
        js = np.arange(lo, hi + 1, dtype=int)
        w[lo:hi + 1] += a[i - js]
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative numbers (a step‑function) that minimises
    ``evaluate_sequence`` within the allotted time.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    np.random.seed(seed)

    # ---------------------------------------------------------
    # 1.  Choose a length and a first‑guess sequence
    # ---------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)          # type: ignore[name-defined]
            # normalise the existing global vector (if any)
            base = np.array(height_sequence_1, dtype=np.float64)
            base = np.maximum(base, 0.0)
            if base.sum() < 1e-12:
                raise ValueError
            base = base / base.sum()
        except Exception:
            n = 6027
            base = None
    else:
        base = None

    # If a usable global vector was found, start from it; otherwise start
    # from the arcsine distribution.
    if base is not None:
        candidates = [base]
    else:
        # arcsine (normalised)
        raw = _arcsine_raw(n)
        a0 = raw / raw.sum()
        candidates = [a0]

    # ---------------------------------------------------------
    # 2.  Add a few “tilted’’ arcsine families  (a_k -> a_k**p)
    # ---------------------------------------------------------
    raw = _arcsine_raw(n)                 # raw (unnormalised) once
    for p in np.linspace(0.6, 1.4, 9):    # 9 values from 0.6 to 1.4
        tilted = np.power(raw, p)
        tilted /= tilted.sum()
        candidates.append(tilted)

    # ---------------------------------------------------------
    # 3.  Add a few random Dirichlet initialisations
    # ---------------------------------------------------------
    for _ in range(3):
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        candidates.append(rnd)

    # ---------------------------------------------------------
    # 4.  Main optimisation loop over all start vectors
    # ---------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for start_id, a in enumerate(candidates):
        if time.time() >= deadline:
            break

        a = a.astype(np.float64)
        n = a.shape[0]      # (re‑assert length – all candidates share it)

        # initialise step size (fraction of the source mass)
        step_frac = 0.07    # fairly aggressive at first

        # compute current score (using the same formula as evaluate_sequence)
        max_conv, conv = _conv_max_and_full(a)
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[init] start {start_id:02d}, score={cur_score:.6f}")

        it = 0
        # -----------------------------------------------------------------
        # 4a.  Guided hill‑climbing with sub‑gradient information
        # -----------------------------------------------------------------
        while time.time() < deadline and it < 8000:
            # recompute convolution and active set
            max_conv, conv = _conv_max_and_full(a)
            cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

            # indices where the convolution is (almost) maximal
            # tolerance is absolute because values are ~O(1/n)
            active = np.where(conv >= max_conv - 1e-13)[0]

            # sub‑gradient weights
            w = _weight_from_active(a, active)

            # source = biggest contributor, destination = smallest contributor
            src = int(np.argmax(w))
            dst = int(np.argmin(w))

            if src == dst or w[src] <= w[dst] + 1e-15:
                # no useful direction – stop this start vector
                break

            # move a small fraction of the mass from src to dst
            eps = step_frac * a[src]
            if eps <= 1e-15:
                break

            new_a = a.copy()
            new_a[src] -= eps
            new_a[dst] += eps

            # compute new objective
            max_conv_new, _ = _conv_max_and_full(new_a)
            new_score = 2.0 * n * max_conv_new / (new_a.sum() ** 2)

            if new_score < cur_score:
                # accept the move
                a = new_a
                if new_score < best_score:
                    best_seq = a.copy()
                    best_score = new_score
                    print(f"[improve] start {start_id:02d}, it={it:04d}, "
                          f"score={new_score:.6f}")
                # modestly increase step size – we are making progress
                step_frac = min(step_frac * 1.04, 0.3)
            else:
                # reject; shrink step size and try a random perturbation
                step_frac *= 0.7
                # random perturbation (occasionally helps escaping plateaus)
                i, j = np.random.choice(n, size=2, replace=False)
                delta = step_frac * min(a[i], a[j])
                if delta > 0:
                    tmp = a.copy()
                    if a[i] > a[j]:
                        tmp[i] -= delta
                        tmp[j] += delta
                    else:
                        tmp[j] -= delta
                        tmp[i] += delta
                    max_conv_tmp, _ = _conv_max_and_full(tmp)
                    tmp_score = 2.0 * n * max_conv_tmp / (tmp.sum() ** 2)
                    if tmp_score < cur_score:
                        a = tmp
                        if tmp_score < best_score:
                            best_seq = a.copy()
                            best_score = tmp_score
            it += 1

        # ---------------------------------------------------------
        # 4b.  A few pure random hill‑climbs as a sanity fallback
        # ---------------------------------------------------------
        # (only if we still have time left)
        while time.time() < deadline and it < 12000:
            i, j = np.random.choice(n, size=2, replace=False)
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            delta = 0.004 * min(a[src], a[dst])
            if delta <= 0:
                it += 1
                continue
            new_a = a.copy()
            new_a[src] -= delta
            new_a[dst] += delta
            max_conv_new, _ = _conv_max_and_full(new_a)
            new_score = 2.0 * n * max_conv_new / (new_a.sum() ** 2)
            if new_score < cur_score:
                a = new_a
                cur_score = new_score
                if new_score < best_score:
                    best_seq = a.copy()
                    best_score = new_score
                    print(f"[rand-improve] start {start_id:02d}, it={it:04d}, "
                          f"score={new_score:.6f}")
            it += 1

    # ---------------------------------------------------------
    # 5.  Sanity fallback – uniform distribution if everything failed
    # ---------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform vector")

    # ---------------------------------------------------------
    # 6.  Final log and return
    # ---------------------------------------------------------
    print(f"[final] length={len(best_seq)}, best score={best_score:.6f}")
    return best_seq.tolist()
