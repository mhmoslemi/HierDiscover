def project_onto_simplex(v):
    # If v already non-negative and sum = 1, return v.
    # else
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

The gradient of the smoothed objective can be written in closed form
using FFTs:

    w  = softmax(β·conv)               (probability vector, length 2n‑1)
    ∇F ≈ 2·(a ⋆ w)                     (cross‑correlation)

where “⋆” denotes linear correlation.  After each update the vector is
projected onto the Euclidean simplex (exact projection by Duchi et al.).
The algorithm runs several restarts with different initialisations
(central‑binomial, uniform, random Dirichlet, mixtures) and keeps the
best sequence seen.  Adam‑style adaptive steps and occasional tiny
random kicks help to escape plateaux.  The whole routine stops after
≈ budget_s seconds and returns the best coefficient list found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions – all at module level (no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Exact Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        # degenerate case – return uniform
        return np.full_like(x, 1.0 / x.size)
    return e / s


def init_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i, i = 0..n‑1, normalised to Σ=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_onto_simplex(a)


def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def init_random_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


def init_mixed(n: int) -> np.ndarray:
    """Convex combination of the central‑binomial and uniform vectors."""
    a_c = init_central_binomial(n)
    a_u = init_uniform(n)
    lam = np.random.rand()
    a = lam * a_c + (1.0 - lam) * a_u
    return project_onto_simplex(a)

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) achieves a low
    value of evaluate_sequence().  The routine runs for at most
    `budget_s` seconds and uses a smooth‑max gradient descent with several
    random restarts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose the length of the sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # use existing length if available
    except NameError:
        n = 8000                         # default length – reasonably large
    # FFT size must accommodate convolution of length n with length (2n‑1)
    fft_size = 1 << ((3 * n - 2).bit_length())  # ≥ 3n‑2

    # ------------------------------------------------------------------
    # Prepare a pool of initialisers.
    # ------------------------------------------------------------------
    init_fns = [
        init_central_binomial,
        init_uniform,
        init_random_dirichlet,
        init_mixed,
    ]

    best_seq = None
    best_score = float("inf")
    restart_id = 0

    # ------------------------------------------------------------------
    # Repeatedly try different restarts until time runs out.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if restart_id >= len(init_fns):
            # cycle through the initialisers again (adds stochasticity)
            restart_id = 0
        init_fn = init_fns[restart_id]
        a = init_fn(n).astype(np.float64)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t_iter = 0

        # hyper‑parameters (tuned empirically)
        lr0 = 0.6               # base learning rate
        beta = 5.0              # start temperature for soft‑max
        beta_max = 250.0        # final temperature
        beta_inc = (beta_max - beta) / 60000.0   # increase slowly

        # ------------------------------------------------------------------
        # Optimisation loop for this restart.
        # ------------------------------------------------------------------
        while time.time() < deadline:
            t_iter += 1

            # FFT of the current vector
            fa = np.fft.fft(a, fft_size)

            # ----- self‑convolution (linear) -----
            conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

            # ----- smooth approximation of the peak -----
            w = softmax(beta * conv_full)        # length 2n‑1, sums to 1
            fw = np.fft.fft(w, fft_size)

            # ----- gradient of the smoothed objective -----
            # cross‑correlation a ⋆ w via FFT: ifft(FA * conj(FW))
            grad_full = np.real(np.fft.ifft(fa * np.conj(fw)))
            # first n components correspond to ∇_a
            grad = 2.0 * grad_full[:n]

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            lr = lr0 / math.sqrt(t_iter)          # simple decay
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = project_onto_simplex(a)

            # ----- temperature schedule -----
            if beta < beta_max:
                beta = min(beta_max, beta + beta_inc)

            # ----- occasional random perturbation -----
            if t_iter % 4000 == 0:
                a = project_onto_simplex(a + np.random.randn(n) * 1e-4)

            # ----- bookkeeping: evaluate the true objective from time to time -----
            if t_iter % 150 == 0:
                # compute the exact peak for the current a
                fa_now = np.fft.fft(a, fft_size)
                conv_now = np.real(np.fft.ifft(fa_now * fa_now))[: 2 * n - 1]
                cur_score = 2.0 * n * conv_now.max()   # identical to evaluate_sequence when Σa=1
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[Restart {restart_id:02d}] iter {t_iter:06d} – new best score {best_score:.6f}"
                    )
                # early exit if we already beat the target
                if best_score < 1.503:
                    print("Target beaten – exiting early.")
                    break

            # hard iteration cap per restart – prevents endless loops if time is abundant
            if t_iter >= 80000:
                break

        # after finishing a restart, check global termination condition
        if best_score < 1.503:
            break

        restart_id += 1

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = init_uniform(n)

    # Convert to plain Python list (required by the evaluator).
    result = best_seq.tolist()

    # Print final statistics for debugging / logging.
    try:

        final_val = evaluate_sequence(result)
        print(f"Final evaluate_sequence value: {final_val:.6f}")
    except Exception as e:
        print(f"Could not evaluate final sequence: {e}")

    return result
