"""
Propose a candidate step‑function using a fast “Gaussian‑seed + Adam” search.

The evaluation E = 2·n·max(a*a)/(sum a)^2 is homogeneous;
we keep the L¹‑norm fixed to sqrt(2·n) (target_sum) so that
E equals the maximal entry of the (self)‑convolution.
A simple analytic model shows that a discrete Gaussian with a
standard deviation ≈0.38·n gives a convolution peak around 1.5.
We therefore:

1. Scan a few Gaussian widths (and a few geometric / uniform seeds)
   and keep the best of these cheap candidates.
2. Starting from the best candidate we run a projected Adam optimiser
   on the exact gradient of the max‑convolution (the same gradient used
   in the reference code).  Adam’s adaptive step‑sizes and momentum
   explore the landscape more efficiently than plain gradient descent.
3. Occasionally add a tiny random jitter and accept occasional
   non‑improving steps (annealing) to escape shallow plateaus.

The whole search runs until the given time budget (default 1000 s) is
exhausted and returns the best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helper – returns the maximum value, its index and the full array.
def _conv_max(a: np.ndarray):
    """Maximum entry of the discrete convolution a*a."""
    b = np.convolve(a, a)                 # length 2·n‑1
    M = b.max()
    k = int(b.argmax())
    return float(M), k, b


# ----------------------------------------------------------------------
# Gradient of the objective (E = max convolution) for a fixed sum.
def _grad_max_convolution(a: np.ndarray, target_sum: float, M: float, k_max: int):
    """Exact gradient of the max‑convolution objective."""
    n = a.shape[0]
    # rev[p] = a_{k_max-p} when the index is in range, else 0
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]

    # derivative of E = M under the fixed‑sum constraint
    grad = (4.0 * n / (target_sum ** 2)) * rev - (4.0 * n * M / (target_sum ** 3))
    return grad


# ----------------------------------------------------------------------
# Simple shape generators
def _gaussian_seq(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the vector."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _geometric_seq(n: int, eps: float, target_sum: float) -> np.ndarray:
    """Decreasing geometric progression a_i = (1‑eps)^i, renormalised."""
    r = 1.0 - eps
    a = r ** np.arange(n, dtype=float)
    a *= target_sum / a.sum()
    return a


def _uniform_seq(n: int, target_sum: float) -> np.ndarray:
    """Flat sequence (baseline)."""
    a = np.full(n, 1.0, dtype=float)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser (projected onto non‑negative, renormalised after each step)
def _adam_step(a, m, v, t, lr, beta1, beta2, eps_adam,
               target_sum):
    """One Adam update for the step‑function coefficients."""
    # gradient for the current a (M,k obtained outside)
    M, k_max, _ = _conv_max(a)
    grad = _grad_max_convolution(a, target_sum, M, k_max)

    # Adam moments
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    # bias‑correction
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # raw Adam step
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    # projection onto feasible set
    a_new = np.maximum(a_new, 0.0)                     # non‑negative
    s = a_new.sum()
    if s > 0.0:
        a_new *= (target_sum / s)                     # renormalise L¹‑norm
    np.clip(a_new, 0.0, 1000.0, out=a_new)            # hard caps (as in evaluate)

    return a_new, m, v


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function giving a low value of
    `evaluate_sequence`.  The routine builds several cheap seeds (Gaussian,
    geometric, uniform, random, plus the historic `height_sequence_1` if
    available) and then refines the best one with a projected Adam optimiser.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # a safety margin for the harness

    # --------------------------------------------------------------
    # Length and normalisation
    try:
        base_len = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_len = 2100
    n = base_len
    target_sum = math.sqrt(2.0 * n)            # fixed L¹‑norm that makes E = max convolution

    # --------------------------------------------------------------
    # 1) generate a handful of cheap candidates
    candidates = []

    # uniform baseline
    candidates.append(_uniform_seq(n, target_sum))

    # Gaussian seeds – scan a few width fractions of the full length
    for frac in (0.20, 0.28, 0.34, 0.376, 0.40, 0.44, 0.50):
        sigma = max(1.0, frac * n)
        candidates.append(_gaussian_seq(n, sigma, target_sum))

    # geometric seeds – scan a few epsilons (as a decreasing progression)
    for eps in (0.0012, 0.0010, 0.0009, 0.0008, 0.0007, 0.0006, 0.0005):
        candidates.append(_geometric_seq(n, eps, target_sum))

    # a few random vectors
    for _ in range(5):
        a = np.random.rand(n)
        a *= target_sum / a.sum()
        candidates.append(a)

    # historic best (if present)
    try:
        hist = np.array(height_sequence_1, dtype=float)      # type: ignore[name-defined]
        if hist.shape[0] != n:
            # resize simply by truncating / padding with zeros
            if hist.shape[0] > n:
                hist = hist[:n]
            else:
                pad = np.zeros(n - hist.shape[0], dtype=float)
                hist = np.concatenate([hist, pad])
        hist = hist * (target_sum / hist.sum())
        candidates.append(hist)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2) evaluate all cheap candidates, keep the best.
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        try:
            sc = evaluate_sequence(cand.tolist())
        except Exception:
            sc = float("inf")
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[seed {idx:02d}] initial candidate score = {sc:.6f}")

    # --------------------------------------------------------------
    # 3) Adam optimisation from the current best.
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12                     # initial learning rate (chosen experimentally)

    t = 0
    beta1_pow = 1.0
    beta2_pow = 1.0

    while time.time() < deadline:
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # learning‑rate decay (very gentle)
        lr = lr0 / (1.0 + 1e-4 * t)

        # Adam step
        a_new, m, v = _adam_step(a, m, v, t, lr,
                                 beta1, beta2, eps_adam,
                                 target_sum)

        # evaluate
        cur_score = evaluate_sequence(a_new.tolist())

        # keep the global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[iter {t:06d}] new best score = {best_score:.6f}")

        # accept the step if it improves, otherwise keep a (with a small chance to accept)
        if cur_score <= best_score or np.random.rand() < 0.008:
            a = a_new

        # every few thousand steps inject a tiny random jitter to shake out of plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * 0.003 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

    # --------------------------------------------------------------
    print(f"[finished] best score = {best_score:.6f}, length = {n}")
    return best_seq.tolist()
