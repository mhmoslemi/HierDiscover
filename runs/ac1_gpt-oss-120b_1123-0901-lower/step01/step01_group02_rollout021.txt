def conv_max(a):
    n=len(a)
    N = 1<<(2*n-1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)
    # conv length N, relevant part first 2*n-1
    return conv[:2*n-1].max(), conv[:2*n-1]
```

def conv_max_fft(a):
    n = a.shape[0]
    N = 1 << ((2*n-2).bit_length())  # next power of two >= 2n-1
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)
    conv = conv[:2*n-1]
    M = conv.max()
    kmax = conv.argmax()
    return M, kmax, conv
```

"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
* Use several candidate lengths (the original length, and a few larger
  interpolated versions).  For each length run an Adam optimiser that
  minimises the maximal convolution value M = max_k (a * a)[k].
* The gradient is the exact sub‑gradient for the max‑convolution:
      grad_i = (4·n / S²)·a_{k*‑i} – (4·n·M / S³) ,
  where k* is the (or one of the) index attaining the maximum,
  n = len(a) and S = Σ a_i (kept fixed to √(2n)).
* Adam’s moving‑average of gradients (`m`) and squared gradients (`v`)
  gives a more stable descent than plain SGD.
* After each update we project onto the non‑negative orthant and
  rescale to keep Σ a = S.
* Every few thousand iterations a tiny Gaussian jitter is added to
  escape shallow plateaus.
* The best sequence found over all runs (including the supplied global
  ``height_sequence_1`` if present) is returned.

The code respects the 1000 s wall‑clock limit, prints progress, and
requires only ``numpy`` and the already‑imported ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full) using FFT – returns max value, its index and the array
def _conv_max_fft(a: np.ndarray):
    """Return (M, k_max, full_convolution) for a real vector a."""
    n = a.shape[0]
    # next power of two >= 2*n-1  (helps FFT speed and avoids wrap‑around)
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A                     # element‑wise product = FFT of convolution
    conv = np.fft.irfft(B, n=N)  # real inverse FFT
    conv = conv[: 2 * n - 1]      # keep only the valid part
    k_max = int(conv.argmax())
    return float(conv.max()), k_max, conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution for a (possibly non‑unique) active set
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Compute the average reversed vector a_{k‑p} over all active shift indices.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    # average over the active set (helps when several shifts tie)
    rev /= max(1, active_idxs.size)
    return rev


# ----------------------------------------------------------------------
# Adam optimiser that works directly on the sub‑gradient of the objective
def _optimize_on_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
):
    """
    Run the Adam optimisation for a vector of length ``n`` starting from
    ``init_a``.  Updates ``best_seq``/``best_score`` in place.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())          # enforce the normalisation

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05            # base learning rate
    iteration = 0
    jitter_interval = 5000

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution, max, active set
        M, k_max, conv_full = _conv_max_fft(a)

        # treat all indices that are numerically equal to the max as active
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        # --------------------------------------------------------------
        # 2) sub‑gradient
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # --------------------------------------------------------------
        # 3) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple learning‑rate schedule (slow decay)
        lr = lr0 / (1.0 + 1e-4 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 4) projection onto feasible set (non‑negative + normalisation)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            # Very unlikely – fall back to uniform vector
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 5) occasional jitter to break plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 6) evaluate (with the exact formula – fast because M is known)
        cur_score = 2.0 * n * M / (target_sum ** 2)   # = M when normalised
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {iteration:6d}] new best score = {best_score:.6f}"
            )

        # stop condition: we let the outer loop enforce the time budget
    return best_seq, best_score
