def conv_max(seq):
    n = len(seq)
    L = 1 << (int(np.ceil(np.log2(2*n-1))))
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2*n-1]
    max_val = conv.max()
    argmax = conv.argmax()
    return max_val, argmax, conv

import cvxpy as cp

def improve_via_lp(f, M, t):
    n = len(f)
    g = cp.Variable(n, nonneg=True)

    constraints = []
    # For each k
    for k in range(2*n - 1):
        # Determine i range
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # Build expression sum_{i} f[i] * g[k-i]
        expr = cp.sum(cp.multiply(f[i_start:i_end+1], g[k - np.arange(i_start, i_end+1)]))
        constraints.append(expr <= M)
    # bound sum of g (to maximize)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True) # Maybe need solver selection
    g_val = g.value
    return g_val
```

import scipy.sparse as sp

def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # indices for g: j = k - i
        # For i in i_start..i_end:
        # coefficient f[i] for g[j]
        # So g index = k - i
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        rows.extend([k]*len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    T = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return T
```

def iterative_lp(f, num_iters=30, t=0.1):
    n = len(f)
    target_sum = np.sqrt(2*n)
    f = np.array(f, dtype=np.float64)
    f = f * (target_sum / f.sum())
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    for it in range(num_iters):
        # compute convolution
        conv = np.convolve(f, f)
        M = conv.max()
        # solve LP: maximize sum(g) ~ to get larger sum, but we keep sum constraint later anyway.
        # Actually we want g that maximize sum(g) under f*g <= M.
        g = solve_lp(f, M)  # returns array length n
        # rescale g to have target_sum
        g = g * (target_sum / g.sum())
        # blend
        f = (1 - t) * f + t * g
        # evaluate
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score
            best_f = f.copy()
            print(f"[iter] it={it}, score={score}")
    return best_f
```

def compute_score(seq):
    n = len(seq)
    L = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2*n-1]
    max_val = conv.max()
    sum_seq = seq.sum()
    return 2.0 * n * max_val / (sum_seq ** 2)
```

def solve_lp_best_g(f, M):

    # define conv expression
    conv_expr = cp.conv(cp.Constant(f), g)

def lp_max_sum(f, M):
    n = len(f)
    g = cp.Variable(n)
    conv = cp.conv(cp.Constant(f), g, mode='full')
    constraints = [conv <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    return np.maximum(g.value, 0.0)

"""
Simple evolutionary search for a step‐function with a low
evaluate_sequence() score.

The algorithm works in three stages:

1.  A deterministic start – the “area” discretisation of the
    continuous   f(x)=1/√x  (a_i = 2*(√(i+1)-√i)).  After normalising
    to sum 1 this shape is already close to the theoretical optimum
    (score ≈ π/2 ≈ 1.571).

2.  A cheap simulated‑annealing style optimisation that moves a small
    fraction of mass from a large entry to a distant small entry.
    Each move is accepted if it improves the score or with a
    Metropolis probability that slowly cools.

3.  An occasional linear‑programming refinement.  For the current
    sequence f the LP maximises Σ g_j under the linear constraints
    (f * g) ≤ M where M = max(f * f).  Adding a normalised version of g
    (blended with the current f) can increase the total L1‑mass while
    keeping the maximal convolution unchanged, which lowers the final
    evaluation score.

The whole routine respects the 1000 s budget, prints progress, and
finally returns the best sequence found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------

def _conv_max(seq: np.ndarray) -> float:
    """Maximum of the discrete convolution of seq with itself."""
    n = seq.shape[0]
    # length of the full convolution
    length = 2 * n - 1
    # next power of two for fast FFT
    L = 1 << ((length - 1).bit_length())
    # FFT based convolution (real‑valued)
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[:length]
    return float(conv.max())


def _score_from_max(n: int, max_conv: float) -> float:
    """Evaluation score when the L1‑sum of the sequence equals 1."""
    return 2.0 * n * max_conv   # denominator = 1² after normalisation


def _initial_area_sequence(n: int) -> np.ndarray:
    """Discrete area version of f(x)=1/√x : a_i = 2*(√(i+1)-√i)."""
    i = np.arange(n, dtype=np.float64)
    a = 2.0 * (np.sqrt(i + 1.0) - np.sqrt(i))
    a /= a.sum()                # normalise to L1‑sum = 1
    return a


def _lp_max_sum(given: np.ndarray, max_allowed: float):
    """
    Solve  max Σ g_i   subject to   g ≥ 0   and   (given * g) ≤ max_allowed.
    Returns the optimal g (as a NumPy array) or ``None`` on failure.
    The convolution constraint is linear because ``given`` is a constant.
    """
    try:
        import cvxpy as cp
    except Exception:           # cvxpy not available – skip the LP step
        return None

    n = given.shape[0]
    g = cp.Variable(n, nonneg=True)

    # cvxpy provides a linear convolution expression when one argument is constant
    conv_expr = cp.conv(cp.Constant(given), g, mode='full')
    constraints = [conv_expr <= max_allowed, g <= 1000.0]   # respect the sandbox limits
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    g_opt = np.asarray(g.value).flatten()
    # Numerical safety – enforce the bound used by the sandbox
    g_opt = np.clip(g_opt, 0.0, 1000.0)
    if g_opt.sum() == 0.0:
        return None
    return g_opt


# --------------------------------------------------------------
# Main entry point required by the runner
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) beats the
    current best score (≈ 1.5709).  The routine respects the 1000 s
    time limit, logs progress, and combines a deterministic start,
    simulated‑annealing moves and occasional LP refinements.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Choose a length.  Prefer the length of a previously found
    #     construction if it exists, otherwise fall back to a moderate
    #     size that keeps the optimisation affordable.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)        # type: ignore[name-defined]
    except Exception:
        n = 1500                          # a safe default

    # ------------------------------------------------------------------
    # 2.  Deterministic start: area version of 1/√x, normalised to sum=1.
    # ------------------------------------------------------------------
    seq = _initial_area_sequence(n)
    best_seq = seq.copy()
    max_conv = _conv_max(seq)
    best_score = _score_from_max(n, max_conv)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Parameters for the stochastic part.
    # ------------------------------------------------------------------
    start_temp = 0.02           # initial “temperature” for Metropolis moves
    temp = start_temp
    decay = 0.99995             # exponential cooling
    min_temp = 1e-5

    # Probability of launching an LP step (expensive) – keep it modest.
    lp_probability = 0.15

    # Blending factor when a new LP solution is found.
    blend_factor = 0.5

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop – stops a little before the deadline
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # keep a tiny safety margin

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Occasionally try a linear‑programming refinement.
        # --------------------------------------------------------------
        if rng.random() < lp_probability:
            # Current maximal convolution (used as the bound M)
            cur_max = _conv_max(seq)
            g_opt = _lp_max_sum(seq, cur_max)
            if g_opt is not None:
                # Normalise the LP solution to L1‑sum = 1
                g_opt /= g_opt.sum()
                # Blend a fraction of the LP solution with the current sequence.
                # Reduce the blend if the max‑convolution would increase.
                t_local = blend_factor
                while t_local > 1e-4:
                    cand = (1.0 - t_local) * seq + t_local * g_opt
                    cand /= cand.sum()          # renormalise (numerical safety)
                    cand_max = _conv_max(cand)
                    if cand_max <= cur_max * (1.0 + 1e-12):
                        seq = cand
                        cur_score = _score_from_max(n, cand_max)
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = seq.copy()
                            print(f"[lp] improved score={best_score:.6f}")
                        break
                    t_local *= 0.5
                # Continue – the annealing part will take over next iteration.
                continue

        # --------------------------------------------------------------
        # Simulated‑annealing move: transfer a small fraction of mass
        # from a large entry to a distant small entry.
        # --------------------------------------------------------------
        # pick a source index proportionally to its current mass
        i_src = rng.choice(n, p=seq)
        # pick a destination that is not too close to the source
        # (moving mass far away tends to flatten the autocorrelation)
        while True:
            i_dst = rng.integers(n)
            if i_dst != i_src and abs(i_dst - i_src) > n // 4:
                break

        # fraction of the source mass to move (between 0.1 % and 2 %)
        eps = rng.uniform(0.001, 0.02)
        delta = seq[i_src] * eps

        cand = seq.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        # (the L1‑sum stays exactly 1, no renormalisation needed)

        cand_max = _conv_max(cand)
        cand_score = _score_from_max(n, cand_max)

        # Metropolis acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # probability decays with the temperature
            prob = math.exp(-(cand_score - best_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            seq = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[anneal] new best score={best_score:.6f}")

        # cooling schedule
        temp = max(min_temp, temp * decay)

    # ------------------------------------------------------------------
    # 5.  Return the best sequence (as a plain Python list).
    # ------------------------------------------------------------------
    print(f"[final] best score achieved={best_score:.6f}")
    return best_seq.tolist()
