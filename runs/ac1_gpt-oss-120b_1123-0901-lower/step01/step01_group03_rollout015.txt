```
def project_onto_simplex(v):
    # Efficient algorithm ensures sum=1, v>=0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    rho = rho[-1] if rho.size > 0 else 0
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
def project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

- Full convolution using FFT:

```
def full_convolution(a):
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n - 1]
    return conv
```

```
def subgradient_maxconv(a, k_max):
    n = len(a)
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n, k_max + 1)
    js = k_max - np.arange(i_start, i_end)
    grad[i_start:i_end] += 2.0 * a[js]
    return grad
```

    # compute softmax weights

"""
Propose a candidate step‑function (non‑negative list of heights) that improves
the best known bound 1.570861 → lower values.

Algorithm
---------

1. Start from the already discovered sequence `height_sequence_1` if it exists,
   otherwise from the central‑binomial coefficients (the classic
   (1‑z)^‑½ series).  The vector is projected onto the probability simplex
   (Σ a_i = 1, a_i ≥ 0).

2. **Smoothed gradient descent**
   * For a chosen temperature α the objective
         f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
     is a smooth approximation of the peak M(a)=max_k (a*a)[k].
   * The gradient of f_α is
         ∇f_α = 2·(p ⋆ a)   where p_k ∝ exp(α·(a*a)[k])  (soft‑max weights)
   * The convolution (a*a) and the correlation (p ⋆ a) are evaluated by FFT,
     giving O(n log n) per iteration.
   * After each gradient step we project back onto the simplex.  The step size
     follows a diminishing schedule η_t = η₀/√t with a simple back‑tracking line‑
     search (up to three reductions).

   The temperature is increased geometrically   α ∈ {5, 10, 20, 40, 80, 160,
   320}.  Early (small α) stages give a global, smooth direction, while later
   stages concentrate on the true maximum.

3. **Local refinement**
   After the smoothed phases we perform a few thousand iterations of the
   classic projected sub‑gradient method (the exact gradient of the peak
   exists only at the maximal index).  This often squeezes a few extra
   decimal places.

4. **Random restarts & perturbations**
   – occasional Gaussian noise (amplitude 10⁻⁴ … 10⁻³) is added to escape
     plateaus,
   – at the very end a tiny grid of simple parametric families
     (power‑law, symmetric bell‑shapes) is checked; the best of those is kept
     if it improves the score.

All steps respect the time budget (default 1000 s, a small safety margin is
kept).  The best sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def correlation(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid‑mode cross‑correlation (p ⋆ a)[i] = Σ_j p[i+j]·a[j],
    length = len(p)‑len(a)+1 = len(a).
    """
    n = a.shape[0]
    size = 1 << ((len(p) + n - 1).bit_length())
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    # correlation = ifft( F(p) * conj(F(a)) )
    corr_full = np.fft.irfft(fp * np.conj(fa), size)[: len(p) - n + 1]
    return corr_full


def subgradient_maxconv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the (any) maximal index k_max.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n, k_max + 1)               # exclusive
    js = k_max - np.arange(i_start, i_end)
    grad[i_start:i_end] += 2.0 * a[js]
    return grad


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Compute the gradient of the smooth approximation
          f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
    Returns (grad, current_max_conv).
    """
    conv = full_convolution(a)                     # length 2n‑1
    # soft‑max weights (numerically stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    p = w / w.sum()
    # correlation gives Σ_j p[i+j]·a[j]
    corr = correlation(p, a)                       # length n
    grad = 2.0 * corr
    return grad, conv.max()


def evaluate_score(a: np.ndarray) -> float:
    """Local version of evaluate_sequence for a normalized (sum=1) vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------

    allocated time budget.
    """

    best_a = a.copy()
    best_score = evaluate_score(a)

    print(f"Start: n = {n}, initial metric = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Smoothed gradient descent (continuation on α)
    # --------------------------------------------------------------
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    eta0 = 0.6                               # initial step size
    iter_counter = 0

            # occasional gentle perturbation (helps breaking symmetry)
            if iter_counter % 1200 == 0:
                a = project_onto_simplex(a + np.random.randn(n) * 0.001)

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3. Sub‑gradient refinement (identical to the reference method)
    # --------------------------------------------------------------
    eta0_sub = 0.2
    while time.time() < deadline:
        conv = full_convolution(a)
        max_val = conv.max()
        k_max = int(conv.argmax())
        grad = subgradient_maxconv(a, k_max)

        eta = eta0_sub / math.sqrt(iter_counter + 1.0)
        a_new = project_onto_simplex(a - eta * grad)
        new_score = evaluate_score(a_new)

        # accept any non‑worsening step
        if new_score <= best_score + 1e-12:
            a = a_new
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                print(f"Sub‑grad iter {iter_counter:6d}, new best = {best_score:.6f}")
        else:
            # tiny random kick when stuck
            a = project_onto_simplex(a + np.random.randn(n) * 5e-5)

        iter_counter += 1

        # every few thousand steps add a broader perturbation
        if iter_counter % 5000 == 0:
            a = project_onto_simplex(a + np.random.randn(n) * 0.002)

    # --------------------------------------------------------------
    # 4. Quick sweep of a few hand‑crafted parametric families
    # --------------------------------------------------------------
    def candidate_from_weights(w):
        w = np.maximum(w, 0.0)
        if w.sum() == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w.sum()
        return w

    # power‑law families (decreasing & increasing)
    for p in (-2.0, -1.5, -1.0, -0.5, 0.0):
        idx = np.arange(n, dtype=np.float64) + 1.0
        cand = candidate_from_weights(idx ** p)                # decreasing
        score = evaluate_score(cand)
        if score < best_score - 1e-12:
            best_score = score
            best_a = cand.copy()
            print(f"Power‑law decreasing p={p:.1f} improved to {best_score:.6f}")

        cand = candidate_from_weights((n - idx + 1) ** p)       # increasing
        score = evaluate_score(cand)
        if score < best_score - 1e-12:
            best_score = score
            best_a = cand.copy()
            print(f"Power‑law increasing p={p:.1f} improved to {best_score:.6f}")

    # symmetric bell‑shape (discrete semicircle)
    t = np.arange(n, dtype=np.float64) + 0.5
    bell = np.sqrt(np.maximum(0.0, (n / 2.0) ** 2 - (t - n / 2.0) ** 2))
    cand = candidate_from_weights(bell)
    score = evaluate_score(cand)
    if score < best_score - 1e-12:
        best_score = score
        best_a = cand.copy()
        print(f"Semicircle shape improved to {best_score:.6f}")

    # sine shape (another smooth basis)
    sine = np.sin(np.pi * (t) / n)
    cand = candidate_from_weights(sine)
    score = evaluate_score(cand)
    if score < best_score - 1e-12:
        best_score = score
        best_a = cand.copy()
        print(f"Sine shape improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final report
    # --------------------------------------------------------------
    print(f"Finished after {iter_counter:,} iterations. Best metric = {best_score:.6f}")

    # (Optional) verify with the provided evaluate_sequence function
    try:
        final_check = evaluate_sequence(best_a.tolist())
        print(f"evaluate_sequence confirms metric = {final_check:.6f}")
    except Exception:
        pass

    # Return a plain Python list
    return best_a.tolist()
