def project_onto_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v):
    """Project onto the probability simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v)
    if v.sum() == 1 and np.all(v >= 0):
        return v.astype(float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

using a projected Adam optimiser on a smooth approximation of the
maximum of the self‑convolution.  The algorithm works with the
probability simplex (∑a_i = 1) and therefore the metric simplifies to
2·n·max(conv).  A soft‑max (log‑sum‑exp) with a gradually increasing
temperature provides a differentiable surrogate for the peak, which
allows the optimiser to escape the plateaus that trap the plain
sub‑gradient method.  The best sequence seen during the run is returned.

The implementation is completely self‑contained, uses only NumPy and
standard Python, and respects the 1000 s time budget.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted in decreasing order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho where u_i - (cssv_i / i) > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT (real‑valued, length 2·len(a)-1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, scale: float) -> np.ndarray:
    """
    Compute a numerically stable soft‑max over the entries of `conv`
    with temperature `scale`.  Returns a vector of the same length that
    sums to 1.
    """
    z = scale * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def _gradient_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. `a`.
    grad_i = 2 * Σ_k weights_k * a_{k-i}
    Implemented via a linear correlation (FFT based) for O(n log n).
    """
    n = a.shape[0]
    m = weights.shape[0]                     # = 2*n-1
    # Linear correlation of `weights` with `a`
    L = 1 << ((m + n - 1).bit_length())
    fw = np.fft.rfft(weights, L)
    fa = np.fft.rfft(a, L)
    # correlation = ifft( fw * conj(fa) )
    corr_full = np.fft.irfft(fw * np.conj(fa), L)[: m + n - 1]
    # valid part corresponds to lags 0 … n‑1
    grad = 2.0 * corr_full[n - 1 : m]
    return grad


def _metric(a: np.ndarray) -> float:
    """
    Exact evaluation metric (without the evaluate_sequence wrapper) for
    a vector that already satisfies Σa_i = 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current bound.  The method uses a projected Adam optimiser on a
    smooth approximation of the peak of the self‑convolution, starting
    from the best known construction (if available) or from the
    central‑binomial coefficients otherwise.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    # --------------------------------------------------------------
    # 1. Initialise the vector a
    # --------------------------------------------------------------
    try:
        # Use the previously found construction if present
        a = np.array(height_sequence_1, dtype=np.float64)
    except NameError:
        # Fallback: central‑binomial coefficients (C(2i,i)/4**i)
        n = 6000                     # a reasonably large size
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    # Normalise onto the probability simplex (∑a = 1)
    a = _project_onto_simplex(a)

    n = a.shape[0]
    # --------------------------------------------------------------
    # 2. Adam state
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # --------------------------------------------------------------
    # 3. Smoothing schedule for the soft‑max
    # --------------------------------------------------------------
    scale = 5.0            # start very smooth
    scale_max = 200.0      # final (almost true max)
    scale_inc = 0.02       # additive increase per 1000 iterations

    # --------------------------------------------------------------
    # 4. Book‑keeping
    # --------------------------------------------------------------
    best_a = a.copy()
    best_score = _metric(a)
    print(f"[init] n={n}, initial metric = {best_score:.6f}")

    iteration = 0
    last_improve_iter = 0

    # --------------------------------------------------------------
    # 5. Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ---- compute convolution and soft‑max weights ----------------
        conv = _full_convolution(a)                    # length 2n‑1
        weights = _softmax_weights(conv, scale)

        # ---- gradient of the smooth surrogate -----------------------
        grad = _gradient_softmax(a, weights)

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        # step size adapts with the smoothing parameter
        base_lr = 0.25 / (scale ** 0.5)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- tentative move + projection -----------------------------
        a_candidate = _project_onto_simplex(a - step)

        # ---- evaluate ----
        cand_score = _metric(a_candidate)

        # ---- simple back‑tracking if no improvement -----------------
        backtrack_factor = 0.5
        backtrack_cnt = 0
        while cand_score > best_score and backtrack_cnt < 5:
            base_lr *= backtrack_factor
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_candidate = _project_onto_simplex(a - step)
            cand_score = _metric(a_candidate)
            backtrack_cnt += 1

        # ---- accept if better ----------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            last_improve_iter = iteration
            print(f"[iter {iteration:6d}] metric = {best_score:.6f}, scale = {scale:.2f}")

        # update the current iterate
        a = a_candidate

        # ---- anneal the smoothness parameter slowly -----------------
        if iteration % 1000 == 0:
            scale = min(scale_max, scale + scale_inc * 1000)

        # ---- occasional random kick to escape shallow plateaus -------
        if iteration % 2500 == 0:
            noise = np.random.randn(n) * 5e-5
            a = _project_onto_simplex(a + noise)

        # ---- early termination if stagnation -------------------------
        if iteration - last_improve_iter > 200_000:
            # no improvement for a long while; stop
            break

    # --------------------------------------------------------------
    # 6. Local fine‑tuning by tiny random transfers (hill‑climbing)
    # --------------------------------------------------------------
    for _ in range(5000):
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny amount δ from i to j
        delta = (np.random.rand() - 0.5) * 1e-4
        if a[i] - delta < 0:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = _project_onto_simplex(a2)   # renormalise (very cheap)
        score2 = _metric(a2)
        if score2 < best_score:
            best_score = score2
            best_a = a2.copy()
            a = a2
            # print a terse progress line only occasionally
            if _ % 1000 == 0:
                print(f"[fine‑tune] metric = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Return the best sequence found
    # --------------------------------------------------------------
    print(f"[done] iterations={iteration}, best metric={best_score:.6f}")
    return best_a.tolist()
