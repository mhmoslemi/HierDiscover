"""
Propose a step‑function (non‑negative sequence) that beats the previous
1.5449 bound.

Algorithm
---------

* **Warm start** – use the best known construction if it is present
  (`height_sequence_1`).  Otherwise fall back to a geometric progression
  with decay `r = exp(-2/n)` (the classic starting point).

* **Simulated‑annealing phase** – add Gaussian noise to the current
  sequence, project back onto the probability simplex and accept the new
  point either if it improves the objective or with a temperature‑dependent
  probability.  The noise amplitude and temperature decay slowly.

* **Targeted mass‑shift phase** – repeatedly locate the shift `k` that
  realises the maximum entry of the self‑convolution.  Move a small
  fraction of the most “responsible’’ coefficient (the one that maximises
  `a_i * a_{k-i}`) to a location that does **not** participate in this
  shift.  This directly reduces the active convolution peak.

* **Fine‑grained sub‑gradient descent** – after the mass‑shift phase run a
  few steps of the classic projected sub‑gradient update to polish the
  solution.

All updates keep the vector on the simplex `{x ≥ 0, Σx = 1}`; the objective
`2·n·max(a*a)/(Σa)²` is invariant under scaling, so the simplex normalisation
does not affect the score.  The routine stops automatically after the
allocated time (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project a (possibly negative) vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                 # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low
    `evaluate_sequence` score.  The function respects the time budget,
    returns the best sequence found and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Warm‑start (use the previously found construction if available)
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
        seq = _project_onto_simplex(seq)           # guarantee sum = 1
        n = len(seq)
        print("[Init] Using supplied height_sequence_1 (length {})".format(n))
    else:
        # fallback: a geometric progression with decay ≈ 1‑2/n
        n = 2000
        factor = 2.0
        eps = factor / n
        r = np.exp(-eps)                           # smoother than 1‑eps
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        print("[Init] No external sequence – using geometric start, n={}".format(n))

    # keep a copy of the current best
    best_seq = seq.copy()
    # local fast objective (sum is 1 after projection, so score = 2·n·max_b)
    conv = np.convolve(best_seq, best_seq)
    best_score = float(2.0 * n * conv.max())
    print("[Init] initial score = {:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing phase (≈ 60 % of the time budget)
    sa_deadline = time.time() + 0.6 * (budget_s - 0.5)
    avg = 1.0 / n
    sigma = avg * 5.0                 # initial noise scale
    temperature = 0.05                # initial temperature
    it_sa = 0
    while time.time() < sa_deadline:
        # propose a noisy neighbour
        cand = best_seq + np.random.normal(scale=sigma, size=n)
        cand = np.maximum(cand, 0.0)                # enforce non‑negativity
        cand = _project_onto_simplex(cand)

        # fast evaluation – sum is 1, hence score = 2·n·max(conv)
        conv = np.convolve(cand, cand)
        cand_score = float(2.0 * n * conv.max())

        # acceptance rule
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / temperature):
            best_seq = cand
            best_score = cand_score
            print("[SA {:06d}] new best = {:.6f}".format(it_sa, best_score))

        # decay schedule
        sigma *= 0.9995
        temperature *= 0.9995
        it_sa += 1

    # ------------------------------------------------------------------
    # 3. Targeted mass‑shift phase (≈ 30 % of the budget)
    #    Move mass from the most harmful coefficient to a location that does
    #    not participate in the current maximal convolution shift.
    shift_deadline = time.time() + 0.3 * (budget_s - 0.5)
    seq = best_seq.copy()
    step_frac = 0.07                      # fraction of mass to move each step
    it_shift = 0
    while time.time() < shift_deadline:
        conv = np.convolve(seq, seq)
        k = int(conv.argmax())            # shift with maximal entry
        # overlapping index interval for this shift
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        # contributions a_i * a_{k-i}
        contrib = seq[idx] * seq[k - idx]
        i_local = np.argmax(contrib)               # most responsible index (local)
        i_max = idx[i_local]

        # pick a destination index outside the overlapping interval
        if hi < n - 1:
            j = n - 1
        else:
            j = 0

        # move a small fraction of the mass
        delta = step_frac * seq[i_max]
        if delta > seq[i_max]:
            delta = seq[i_max]
        if delta > 1e-12:
            seq[i_max] -= delta
            seq[j]   += delta

            # re‑normalise to avoid drift
            seq = _project_onto_simplex(seq)

            # evaluate
            conv = np.convolve(seq, seq)
            cur_score = float(2.0 * n * conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print("[Shift {:06d}] new best = {:.6f}".format(it_shift, best_score))

        # decay the step fraction slowly
        step_frac *= 0.9999
        it_shift += 1

    # ------------------------------------------------------------------
    # 4. Fine‑grained sub‑gradient descent (use remaining time)
    grad_deadline = deadline
    seq = best_seq.copy()
    step = 0.15                                 # absolute step size
    it_grad = 0
    while time.time() < grad_deadline:
        conv = np.convolve(seq, seq)
        k = int(conv.argmax())                  # active shift
        # sub‑gradient of the active quadratic term
        grad = np.zeros_like(seq)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] = seq[k - idx]                # grad_i = a_{k-i}
        # gradient step (we minimise)
        new_seq = seq - step * grad
        new_seq = _project_onto_simplex(new_seq)

        # evaluate
        cur_score = float(2.0 * n * np.convolve(new_seq, new_seq).max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            print("[GD {:06d}] new best = {:.6f}".format(it_grad, best_score))

        # prepare next iteration
        seq = new_seq
        step *= 0.9995
        it_grad += 1

    print("Final best objective = {:.6f} (length {})".format(best_score, n))
    # The evaluator expects a plain Python list
    return best_seq.tolist()
