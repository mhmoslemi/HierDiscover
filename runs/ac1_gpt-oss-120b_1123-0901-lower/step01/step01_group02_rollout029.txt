"""
Propose a candidate step‑function by exploiting the analytic form of the
self‑convolution.  For a monotone decreasing power law

    a_i ∝ (n‑i)^{p}      (i = 0 … n‑1)

the peak of the convolution can be written in closed form and, for a
proper choice of the exponent p, is far below the naïve uniform bound
(2.0).  The routine first scans a coarse grid of exponents, then refines
around the best one, and finally performs a cheap random hill‑climb to
polish the solution.  All work is carried out inside the allowed time
budget and no external files or network access are used.

The evaluation function is homogeneous, therefore the vector is always
renormalised to the “standard” total sum sqrt(2 n) – this also guarantees
that the score returned by `evaluate_sequence` equals the maximal
convolution value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` (without the security checks).
    The sequence is assumed to contain only non‑negative numbers inside
    the allowed range [0, 1000]; we explicitly apply the same clipping
    to be perfectly compatible.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    b = np.convolve(seq, seq)
    max_b = b.max()
    s = seq.sum()
    if s < 0.01:                     # same protection as the original code
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: float = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` function.

    The algorithm:
    1. Determine the problem size `n` (use the global `height_sequence_1`
       if present, otherwise fall back to 2100).
    2. Scan a family a_i ∝ (n-i)^{p} for exponents p∈[0,8] (coarse grid),
       keep the best normalised vector.
    3. Refine the exponent around the best candidate on a finer grid.
    4. Run a short random hill‑climbing phase (pairwise mass transfers)
       while the time budget permits.
    5. Return the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.shape[0]
    except Exception:
        n = 2100
    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the paper

    best_seq = None
    best_score = float("inf")
    best_p = None

    # --------------------------------------------------------------
    # 1) Coarse grid search over power‑law exponents
    coarse_ps = np.linspace(0.0, 8.0, 81)     # step 0.1
    idx_desc = np.arange(n, 0, -1, dtype=float)   # n, n-1, … , 1

    for p in coarse_ps:
        a = np.power(idx_desc, p)           # decreasing power law
        a = a * (target_sum / a.sum())      # renormalise to the fixed sum
        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            best_p = p
            print(f"[coarse] p={p:.2f}  score={best_score:.6f}")

        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2) Fine grid around the best exponent from the coarse scan
    if best_p is not None and time.time() < deadline:
        fine_ps = np.linspace(max(0.0, best_p - 0.5),
                              best_p + 0.5, 101)   # step ≈0.01
        for p in fine_ps:
            a = np.power(idx_desc, p)
            a = a * (target_sum / a.sum())
            cur = _score(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                best_p = p
                print(f"[fine]   p={p:.4f}  score={best_score:.6f}")

            if time.time() > deadline:
                break

    # --------------------------------------------------------------
    # 3) Optional: try an exponential family as a safeguard
    if time.time() < deadline:
        alphas = np.linspace(0.0, 0.02, 41)   # very small decay – keeps values >0
        idx_asc = np.arange(n, dtype=float)  # 0,…,n-1
        for alpha in alphas:
            a = np.exp(-alpha * idx_asc)
            a = a * (target_sum / a.sum())
            cur = _score(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[exp]    α={alpha:.5f}  score={best_score:.6f}")

            if time.time() > deadline:
                break

    # --------------------------------------------------------------
    # 4) Random hill‑climbing (pairwise mass transfer) to polish
    if best_seq is None:                     # should never happen
        best_seq = np.ones(n, dtype=float) * (target_sum / n)

    a = best_seq.copy()
    cur_score = best_score
    iteration = 0
    # step size slowly decays – we start with a tiny fraction of the total sum
    step0 = target_sum * 0.005               # 0.5 % of the total mass
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose a transfer of mass from i to j
        delta = (np.random.rand() - 0.5) * step0 / (1 + iteration * 1e-4)
        # ensure we stay non‑negative
        if a[i] - delta < 0.0 or a[j] + delta < 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # keep the global sum fixed (tiny numerical drift may appear)
        s = a_new.sum()
        if s > 0:
            a_new *= (target_sum / s)

        new_score = _score(a_new)
        # Accept only improving moves (simple hill‑climb)
        if new_score < cur_score:
            cur_score = new_score
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_new.copy()
                print(f"[hill {iteration:06d}] new best = {best_score:.6f}")

        iteration += 1

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
