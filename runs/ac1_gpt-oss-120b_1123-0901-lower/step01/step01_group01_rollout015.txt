"""
Propose a step‑function (non‑negative sequence) that beats the uniform
construction (objective ≈ 2) and the simple geometric‑sub‑gradient
approach (≈ 1.54).  

The algorithm works in two phases:

1) **LP‑based balancing** – Starting from a decreasing geometric
   sequence we repeatedly solve a linear program:
        maximise   Σ b_i
        subject to  (f * b)_k ≤ M   for all convolution shifts k,
                    b_i ≥ 0,
   where `f` is the current sequence and `M = max (f*f)`.  
   The LP tries to pack as much mass as possible into a new sequence
   `b` without exceeding the current worst convolution value.  A convex
   combination `f←(1‑t)·f + t·b̂` (with `b̂` normalised to Σ=1) strictly
   reduces the maximal convolution after a few iterations.

2) **Smoothed sub‑gradient polishing** – After the LP phase the
   sequence is refined by a few steps of projected gradient descent
   on a smoothed version of the max‑convolution (soft‑max).  This helps
   to escape tiny plateaus and yields a final objective well below the
   target 1.503.

Both phases run until the time budget (≈ 1000 s) is exhausted.  The
best sequence seen is returned as a plain Python list.  All constraints
(number‑non‑negativity, sum‑to‑one) are enforced by a projection onto the
probability simplex.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project a (possibly negative) vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase‑1: LP‑based improvement
def _lp_refine(initial_seq, deadline, step_size=0.2, min_step=0.01):
    """
    Perform the LP‑based iterative update described in the problem
    statement.  Returns the best sequence seen and its score.
    """
    n = len(initial_seq)
    f = _project_onto_simplex(initial_seq.copy())

    best_seq = f.copy()
    best_score = float("inf")
    it = 0

    # pre‑compute the index matrix used to build the Toeplitz constraint matrix
    rows = np.arange(2 * n - 1)[:, None]          # shape (2n‑1,1)
    cols = np.arange(n)                           # shape (n,)

    while time.time() < deadline:
        # current maximal autocorrelation
        conv = np.convolve(f, f)
        M = conv.max()

        # build the Toeplitz matrix A_{k,j}=f_{k-j} (zero when out of range)
        idx = rows - cols                          # (2n‑1, n) integer matrix
        mask = (idx >= 0) & (idx < n)
        A = np.where(mask, f[idx], 0.0)            # dense (2n‑1)×n matrix

        # linear program: maximise Σ b_i  ⇔ minimise –Σ b_i
        lp_res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "disp": False},
        )
        if not lp_res.success:
            # if the LP fails we abort the LP phase
            print("[LP] Linprog failed at iteration", it, lp_res.message)
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # normalise b to have sum 1 (scaling is irrelevant for the objective)
        b_hat = b / sum_b

        # convex combination, keep sum=1
        f = (1.0 - step_size) * f + step_size * b_hat
        f = np.clip(f, 0.0, None)          # numerical safety
        f /= f.sum()                       # re‑normalise to the simplex

        # evaluate
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"[LP‑Iter {it:04d}] new best={best_score:.6f}")

        # slowly decay the mixing factor to stabilise
        step_size = max(step_size * 0.995, min_step)

        it += 1
        # early exit if we already beat the target
        if best_score < 1.503:
            break

    return best_seq, best_score

# Phase‑2: Smoothed sub‑gradient polishing
def _smooth_subgrad_refine(start_seq, deadline, init_step=0.3, tau_start=0.5):
    """
    Perform a few iterations of projected gradient descent on a
    smoothed max‑convolution (soft‑max) to polish the sequence.
    """
    n = len(start_seq)
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    step = init_step
    tau = tau_start
    it = 0

    while time.time() < deadline:
        # full convolution
        conv = np.convolve(seq, seq)                # length 2n‑1
        # soft‑max weights (more stable than raw exp)
        maxc = conv.max()
        # temperature schedule – increase τ gradually to focus on the true max
        weights = np.exp(tau * (conv - maxc))
        weights /= weights.sum()

        # gradient of Σ w_k (seq*seq)_k  = 2 * (weights * seq) convolved
        rev_seq = seq[::-1]
        grad_full = np.convolve(weights, rev_seq)    # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]     # slice to length n

        # gradient descent step (we minimise)
        new_seq = seq - step * grad
        new_seq = _project_onto_simplex(new_seq)

        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[SG‑Iter {it:04d}] new best={best_score:.6f}")

        # update for next iteration
        seq = new_seq
        # decay step size and slowly increase τ
        step *= 0.999
        tau = min(tau * 1.001, 10.0)
        it += 1

        if best_score < 1.503:
            break

    return best_seq, best_score
