def _optimize_sequence(a, n, target_sum, deadline, best_seq, best_score):
    a = a.astype(np.float64)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = target_sum * 0.02   # step size scaling
    iteration = 0
    beta = 0.5   # start smoothing
    max_beta = 15.0
    beta_growth = 1.005  # per iteration
    # maybe use max iterations per run
    while time.time() < deadline and iteration < 3000:
        # compute convolution
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # Evaluate candidate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New global best {best_score:.6f} (n={n})")
        # compute soft weights
        max_val = max_conv
        # compute weights w = softmax(beta * conv)
        # to maintain numerical stability:
        exp_term = np.exp(beta * (conv - max_val))
        w = exp_term / exp_term.sum()
        # compute gradient: grad = 2 * conv_rev_a_w slice
        rev_a = a[::-1]
        # conv_rev_a_w = np.convolve(rev_a, w)
        grad_full = np.convolve(rev_a, w)
        grad = 2.0 * grad_full[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # update
        a_new = a - step

        # project onto simplex
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional perturbation
        if np.random.rand() < 0.01:
            a_new += np.random.randn(n) * target_sum * 0.001
            a_new = _project_onto_simplex(np.maximum(a_new,0), target_sum)

        a = a_new

        # increase beta gradually
        if beta < max_beta:
            beta *= beta_growth

        iteration += 1

    return a, best_seq, best_score

"""
Propose a non‑negative height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The method is a *smoothed‑gradient descent* on the simplex.
Key ideas
---------

1.  The optimisation domain is the simplex {a_i ≥ 0, Σ a_i = √(2 n)}.
    The projection onto this simplex is performed by the
    Wang‑Carreira‑Perpiñán algorithm (`_project_onto_simplex`).

2.  Instead of moving only against the gradient of the *single* maximal
    convolution entry, we minimise a smooth approximation of the max:
        F_β(a) = (1/β)·log Σ_k exp(β·p_k) ,  p = a * a .
    The softmax weights `w_k = exp(β·p_k)/Σ exp(β·p_k)` give a
    differentiable objective whose gradient is

        ∇F_β = 2 · conv( rev(a) , w )[n‑1 : n‑1+n]

    where `rev(a)` is `a` reversed.  For β→∞ this gradient becomes the
    sub‑gradient of the true max, while for small β it smooths the landscape
    and helps escaping shallow local minima.

3.  A simple adaptive step‑size is used:
        η = base_step·max(p) / max(|∇F_β|) .
    The step is taken opposite to the gradient and the iterate is projected
    back onto the simplex.

4.  Several independent restarts with random lengths (≈ 500‑2000) are
    performed within the time budget.  The best sequence seen so far is
    returned.  Occasionally a tiny random perturbation is added to avoid
    stagnation.

The routine respects the 1000 s budget, prints progress, and never raises
exceptions – an empty or `None` result would be penalised, so a safe uniform
fallback is provided.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    # work on a copy
    v = v.astype(np.float64, copy=True)
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the rho index
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        if max_grad == 0:
            break
        eta = base_step * max_conv / max_grad

        # 6) descent step and projection
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # 7) occasional random shake‑up (1 % chance)
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_step *= 0.9          # slightly reduce step after a shake‑up

        a = a_new

        # 8) gradually tighten the smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * beta_growth)

        iter_cnt += 1

    return best_seq, best_score
