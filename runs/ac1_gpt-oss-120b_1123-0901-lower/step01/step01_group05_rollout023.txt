"""
Propose a candidate step‑function sequence that improves the
evaluation bound.

Algorithm
---------
1.  Determine the length *n* (use the previously found sequence if
    available, otherwise fall back to 2000).
2.  Initialise with the classic inverse‑square‑root construction
    `a_i = binom(2*i,i) / 4**i`.  Its score is about π/2 ≈ 1.57.
3.  Solve a linear programme that *maximises* the ℓ¹‑norm of a new
    sequence `g` while keeping the cross‑convolution with the current
    sequence bounded by the current maximal self‑convolution `M`.
    This is the exact formulation described in the statement:
        maximise   ∑ g_i
        subject to g_i ≥ 0
                   (f * g)_k ≤ M   for all k.

    The LP has `n` variables and `2n‑1` linear constraints and can be
    solved with CVXPY/ECOS.  The resulting `g` distributes mass farther
    to the right where the coefficients of `f` are small, thereby
    increasing the total sum without inflating the convolution bound
    too much.
4.  Form a convex combination `h = (1‑t)·f + t·g` (t≈0.3).  This keeps
    the maximal self‑convolution low while inheriting the larger sum
    of `g`.  Evaluate the combination and keep the best of `{f, g, h}`.
5.  A cheap stochastic local search (multiplicative log‑normal
    perturbations of a single entry) further refines the best
    candidate until the time budget expires.
6.  Return the best sequence found.

The whole routine runs well within the 1000 s limit for n≈3‑4 k.
"""

import time
import numpy as np
import cvxpy as cp
from math import sqrt, log
import sys

def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Return a_i = binom(2*i,i)/4**i for i=0..n‑1.
    Recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) avoids big integers."""
    seq = [1.0]
    for i in range(1, n):
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n-1, n)) such that
        A @ g   =   np.convolve(f, g)
    for any vector g of length n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A.fill(0.0)
    # For each column i the coefficient f[i] appears at rows i .. i+n-1
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi
    return A

    Return a list of non‑negative floats that (hopefully) beats the
    current best bound 1.503.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # 1. length of the sequence
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)    # type: ignore[name-defined]
    except Exception:
        n = 2000

    # --------------------------------------------------------------
    # 2. Base construction – inverse‑square‑root (near optimal)
    # --------------------------------------------------------------
    f = np.array(generate_inverse_sqrt_sequence(n), dtype=np.float64)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, base score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Linear programme: maximise sum(g)  subject to  f*g ≤ M
    # --------------------------------------------------------------
    # current maximal self‑convolution of f
    conv_f = np.convolve(f, f)
    M_f = float(conv_f.max())
    print(f"[LP] current max convolution M_f={M_f:.6f}")

    # Build convolution matrix for f (dense, but still small enough)
    A = build_conv_matrix(f)

    # Set up the LP
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_f * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    except Exception as exc:
        print("[LP] solver failed:", exc)
        prob = None

    if prob is not None and g.value is not None:
        g_opt = np.maximum(0.0, np.minimum(1000.0, g.value))
        # Normalise: the evaluation function is scale invariant,
        # therefore we keep g_opt as‑is.
        cand_score = evaluate_sequence(g_opt.tolist())
        print(f"[LP] solution score={cand_score:.6f}")
        if cand_score < best_score:
            best_score = cand_score
            best_seq = g_opt.copy()
    else:
        print("[LP] No feasible solution, keep the base sequence.")

    # --------------------------------------------------------------
    # 4. Convex combination of f and the LP‑solution (if we have one)
    # --------------------------------------------------------------
    if prob is not None and g.value is not None:

        combo = (1 - t) * f + t * g_opt
        combo_score = evaluate_sequence(combo.tolist())
        print(f"[mix] convex combo score={combo_score:.6f}")
        if combo_score < best_score:

    # --------------------------------------------------------------

        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand

    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list as required by the runner
    return best_seq.tolist()
