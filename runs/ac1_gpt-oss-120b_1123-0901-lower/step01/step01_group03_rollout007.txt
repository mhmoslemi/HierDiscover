def fft_convolve(x, y):
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:len(x)+len(y)-1]
    return conv
```

Proposed candidate generator.

The algorithm searches for a non‑negative sequence a of length n that
minimises the evaluation metric

        2·n·max(conv(a,a)) / (sum(a)**2)

which, after normalising sum(a)=1, is simply 2·n·max(conv(a,a)).
We work on the probability simplex (Σa_i=1, a_i≥0) and use a
smooth‑max (soft‑max) approximation to the peak of the self‑convolution.
The gradient of this smooth objective is obtained analytically (it is a
convolution of the soft‑max weights with the reversed current vector).
An Adam‑style projected gradient descent is then performed,
gradually sharpening the soft‑max parameter so that the algorithm
behaves like sub‑gradient descent on the true peak while keeping the
updates well‑behaved.  Occasional random perturbations help to escape
plateaus.  The best sequence seen during the allotted time is returned.
"""

import numpy as np
import time
import math

# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numeric safety: renormalise (should already sum to 1)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using the FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution a*a (length 2·len(a)-1)."""
    return _fft_convolve(a, a)


def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max  φ_β(a)= (1/β)·log Σ_k exp(β·conv_k)
    w.r.t. a, where conv = a*a.
    The formula reduces to a convolution of the soft‑max weights with the
    reversed vector a.
    """
    # soft‑max weights (probability distribution over convolution indices)
    max_val = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(beta * (conv - max_val))
    p = exp_vals / exp_vals.sum()

    # reversed a
    a_rev = a[::-1]

    # full convolution of p with a_rev (length 3·n‑2)
    grad_full = _fft_convolve(p, a_rev)

    # slice the part that corresponds to the gradient entries
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad
