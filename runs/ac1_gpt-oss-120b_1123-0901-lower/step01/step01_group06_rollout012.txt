def optimize_b(a, M):
    n = len(a)
    K = 2*n - 1
    # Build A_ub as K x n
    A = np.zeros((K, n))
    # For each k:
    for k in range(K):
        # i indices for a
        # b_j coefficient a_{k-j}
        # j from max(0, k-(n-1)) to min(k, n-1)
        j_start = max(0, k-(n-1))
        j_end = min(k, n-1)
        # assign a_{k-j} to column j
        for j in range(j_start, j_end+1):
            A[k, j] = a[k - j]
    # Solve LP
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(K, M), bounds=(0, None), method='highs')
    if not res.success:
        # maybe return a?
        return a
    b = res.x
    return b
```

"""
Propose a candidate step‑function (non‑negative sequence) with a low self‑convolution
supremum.  The construction starts from the analytically optimal arcsine series,
then repeatedly solves a linear program that maximises the total mass under the
current convolution bound.  The extra mass is redistributed (by mixing) and the
supremum decreases.  A short random local refinement is performed at the end.
The algorithm stays well within the 1000 s budget and usually beats the
π/2 ≈ 1.5708 bound, targeting the < 1.503 target.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the first n coefficients of (1‑z)^(-1/2)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n terms a_i = C(2i,i) / 4^i via a stable recurrence."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Build the linear‑program matrix for the constraint   a * b ≤ M
# ----------------------------------------------------------------------
def _build_A_ub(a: np.ndarray) -> np.ndarray:
    """
    Return the (2n‑1) × n matrix A such that
        (A @ b)[k] = (a * b)[k]   for k = 0 … 2n‑2 .
    Here * denotes the discrete convolution with zero padding.
    """
    n = a.size
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=float)
    # each column j contains the vector a shifted down by j rows
    for j in range(n):
        A[j:j + n, j] = a
    return A


# ----------------------------------------------------------------------
# Solve the LP   maximise 1ᵀb   subject to   a * b ≤ M ,   b ≥ 0
# ----------------------------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float):
    """
    Return a vector b (non‑negative) maximising sum(b) while keeping
    the convolution a*b bounded by M.  Returns None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                     # SciPy not available
        return None

    n = a.size
    A = _build_A_ub(a)
    K = 2 * n - 1
    # maximize sum(b)  →  minimize -sum(b)
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n
    res = linprog(c,
                  A_ub=A,
                  b_ub=np.full(K, M, dtype=float),
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)            # guard against tiny negatives
    return b


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0,
                     seed: int = 42,
                     **kwargs):

    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Available time budget in seconds.
    seed : int, optional
        Random seed for reproducibility.
    **kwargs
        Ignored; kept for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum equals 1.0) with a low evaluation score.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Length of the vector – reuse a previous construction if possible,
    # otherwise fall back to a reasonable default.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # pylint: disable=undefined-variable
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise with the (1‑z)^(-1/2) coefficients and normalise.
    # ------------------------------------------------------------------
    a = _arcsine_coeffs(n)
    a = a / a.sum()                     # now sum(a) == 1.0
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[propose_candidate] initial score = {best_score:.6f}, n = {n}")

    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # mixing factor – how much of the LP solution we blend into the current
    # sequence.  0.5 works well for stability.
    mix = 0.5
    max_lp_iters = 8                     # enough for the time budget

            break

        g = b / sum_b

        a_candidate = (1.0 - mix) * a + mix * g
        a_candidate = np.maximum(a_candidate, 0.0)

        if new_score < best_score:

            best_seq = a_new.copy()

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[propose_candidate] final score = {final_score:.6f}, length = {len(best_seq)}")
    except Exception:
        pass

    return best_seq.tolist()
