"""
propose_candidate – Heuristic optimisation for the max‑convolution bound.

The objective for a probability vector p (sum(p)=1) is
    J(p) = 2·n·max_k (p * p)[k]               (linear convolution)

We look for a distribution with a small peak.  The search consists of:

1.  A diverse pool of analytic families (uniform, beta, exponential,
    power‑law, random Dirichlet, linear ramps).  Each is projected onto
    the probability simplex and the exact objective obtained via FFT.
    The best few are kept as seeds.

2.  A long exponentiated‑gradient (EG) phase that works directly on the
    simplex.  EG uses the exact sub‑gradient of the max‑convolution
    (uniform weight on all active shifts).  The step‑size η_t = η0/√t
    with η0≈2.0.

3.  A smooth‑max Adam phase.  The max is approximated by a soft‑max
    with temperature λ that is multiplied by 1.05 each iteration
    (λ_max≈1e15).  Adam’s momentum parameters are standard (β1=0.9,
    β2=0.999).  The parameters are tuned for faster convergence.

4.  A stochastic local‑search / simulated‑annealing loop that moves a
    random amount of mass between two coordinates (up to 2 % of the
    source weight).  Improvements are always kept; occasional
    uphill moves are accepted with a temperature that cools linearly.

5.  Random‑restart polishing: the current best vector is perturbed
    with small Gaussian noise and re‑optimised by a short Adam run.
    This is repeated until the budget expires.

6.  Finally a few cyclic shifts are tested (linear convolution is
    shift‑variant) and the best shift is returned.

All operations respect the 1000 s time budget, use only numpy (and
the supplied evaluate_sequence) and always return a non‑negative list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Optimisers ---------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 2.0,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on the simplex) using the exact
    sub‑gradient of the max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[EG it={it:07d}] new best {val:.6f}")

        # active set (all shifts within tolerance of the maximum)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)          # multiplicative update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.04,
    lam_start: float = 0.1,
    lam_factor: float = 1.05,
    lam_max: float = 1e15,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Parameters are tuned for faster temperature increase.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        # occasional learning‑rate decay to stabilise near the end
        if it % 150_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val

def _make_beta(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta(α,β) density on a uniform grid, projected onto the simplex."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def _make_exponential(rate: float, n: int) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def _make_powerlaw(p: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, p)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def _make_linear_ramp(n: int) -> np.ndarray:
    """Linearly decreasing ramp (triangular shape)."""
    a = np.linspace(1.0, 0.0, n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

    # power‑law tails
    for _ in range(20):
        p = rng.uniform(0.6, 2.5)
        cand.append(_make_powerlaw(p, n))

    # linear decreasing ramp (deterministic)
    cand.append(_make_linear_ramp(n))

    # random Dirichlet draws with varying concentration
    for _ in range(40):
        conc = rng.uniform(0.3, 3.0)
        a = rng.dirichlet(np.full(n, conc))
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (as a plain Python list) found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overshooting the deadline
    deadline = start_time + budget_s - 0.3

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0) collect a diverse set of seeds
    candidates = _initial_candidates(n)

    # ------------------------------------------------------------------
    # 1) quick screening of candidates (exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # ------------------------------------------------------------------
    # 2) main optimisation – split remaining budget
    # allocate ~60 % to EG, ~30 % to Adam, remainder to local/perturbation
    time_now = time.time()
    sub_deadline = deadline - (budget_s * 0.35)   # reserve time for later phases
    if sub_deadline > time_now:
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            eta0=2.0, max_iters=6_000_000
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[EG] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam smooth‑max fine‑tuning
    time_now = time.time()
    adam_deadline = deadline - (budget_s * 0.15)
    if adam_deadline > time_now:
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr=0.04,
            lam_start=0.2,
            lam_factor=1.07,
            lam_max=1e15,
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) stochastic local search (small moves)
    time_now = time.time()
    local_deadline = deadline - 0.8
    rng = np.random.default_rng()
    while time.time() < local_deadline:
        i, j = rng.choice(n, 2, replace=False)
        # allow a move up to a few percent of the source weight
        delta = min(best_a[i], 0.02) * rng.random()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 5) random‑restart polishing: perturb + short Adam run
    # each inner run gets at most 0.7 s (or the remaining time if lower)
    inner_time = 0.7
    while time.time() + inner_time < deadline - 0.5:
        # perturb current best with Gaussian noise (std proportional to mass)
        sigma = 0.02
        pert = rng.normal(scale=sigma, size=n)
        cand = best_a + pert
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        inner_deadline = time.time() + inner_time
        cand_opt, cand_val = _adam_optimize(
            cand, n, conv_fft_len, corr_fft_len,
            inner_deadline,
            max_iters=80_000,
            lr=0.03,
            lam_start=0.2,
            lam_factor=1.06,
            lam_max=1e14,
        )
        if cand_val < best_val:
            best_val, best_a = cand_val, cand_opt.copy()
            print(f"[Perturb‑Adam] improved to {cand_val:.6f}")

    # ------------------------------------------------------------------
    # 6) try a few cyclic shifts (linear convolution changes with a shift)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 7) final sanitisation and report
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)  # guarantee a proper probability vector
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
