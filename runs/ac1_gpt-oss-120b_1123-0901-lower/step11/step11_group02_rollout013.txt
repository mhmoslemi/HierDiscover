# Helper utilities ------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) for gradient computation.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Shape factories -------------------------------------------------------


def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2.0
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = (np.arange(n) + 0.5) / n          # mid‑points to avoid exact 0/1
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[~np.isfinite(a)] = 0.0
    s = a.sum()
    if s <= 0:
        return _make_uniform(n)
    a /= s
    return a


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_sin_power(n, p):
    """a_i ∝ sin(π·i/(n‑1)) ** p – a smooth family that interpolates
    between uniform (p≈0) and bell‑shaped (large p)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    a = np.power(a, p, where=a > 0, out=np.zeros_like(a))
    a /= a.sum()
    return a

# Optimisation primitives -----------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Returns the best vector seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                # a tad larger than the reference code
    lam = 0.08               # start temperature
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ------- soft‑max parametrisation -------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ------- evaluate surrogate -------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c          # exact objective (sum(a)=1)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] best {val:.6f}")

        # ------- smooth‑max weights -------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------- gradient of smooth‑max -------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (smooth‑max becomes sharper)
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           deadline: float,
                           max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Fine‑grained mass‑transfer local search.
    Moves a small amount of mass from the coordinate with the largest
    sub‑gradient to the one with the smallest, trying a descending geometric
    grid of step sizes until an improvement is found.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    # geometric grid of fractions (larger to smaller)
    step_grid = (0.5, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003, 0.001)

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))   # source of mass
        i_low = int(np.argmin(grad))    # destination

        if grad[i_high] <= grad[i_low] + 1e-15:
            # gradients are equal – we are at a stationary point
            break

        src_mass = a[i_high]
        improved = False

        for frac in step_grid:
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:07d}] new best {val_new:.6f}")
                break

        if not improved:
            # a small random jitter to escape possible plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.04
                a[i] -= eps
                a[j] += eps

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the 1000 s budget and prints progress.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 points gives a good balance of resolution and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous best is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # resize by truncation or zero‑padding
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a modest pool of diverse seeds
    pool = []

    # incorporate warm‑start variations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 6, n // 3, n // 2):
            pool.append(np.roll(warm, shift))

    # analytical families
    pool.append(_make_uniform(n))
    pool.append(_make_cosine(n))
    pool.append(_make_triangle(n))
    pool.append(_make_beta(n, 0.5, 0.5))       # arcsine‑like
    pool.append(_make_beta(n, 0.8, 0.8))
    pool.append(_make_beta(n, 1.2, 1.2))
    pool.append(_make_sin_power(n, 0.5))
    pool.append(_make_sin_power(n, 2.0))
    pool.append(_make_sin_power(n, 5.0))

    # a few random Dirichlet draws with widely varying concentration
    for alpha in (0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – short Adam runs for all seeds (to pick a promising start)
    print("[Phase I] quick Adam on all seeds")
    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # give each candidate only a few seconds (e.g. 2 s)
        sub_dead = min(deadline, time.time() + 2.0)
        a_tmp, val_tmp = _adam_optimize(a0, n, conv_fft_len,
                                        corr_fft_len, sub_dead,
                                        max_iters=50_000)
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Seed {idx}] improved to {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long Adam optimisation on the current best seed
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase II] long Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=800_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase II] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase III – fine‑grained mass‑transfer refinement (main effort)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] mass‑transfer local search")
        a_mt, val_mt = _mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[Phase III] mass‑transfer improved to {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – very short simulated annealing to shake any remaining local minimum
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase IV] brief annealing")
        # a tiny annealing loop (max 20 k moves)
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_a, best_val_local = a.copy(), val

        T_start = 0.05
        T_end = 1e-4
        moves = 0
        max_moves = 20_000
        while time.time() < deadline and moves < max_moves:
            T = max(T_end, T_start * (1.0 - moves / max_moves))
            i = np.random.randint(n)
            if a[i] < 1e-14:
                moves += 1
                continue
            j = np.random.randint(n)
            if i == j:
                moves += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac

            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()
            if val2 < best_val_local:
                best_val_local, best_a = val2, a2.copy()
                a = a2
                val = val2
            else:
                if np.random.rand() < math.exp(-(val2 - val) / T):
                    a = a2
                    val = val2
            moves += 1
        if best_val_local < best_val:
            best_val = best_val_local
            best_seq = best_a.copy()
            print(f"[Phase IV] annealing improved to {best_val_local:.6f}")

    # ------------------------------------------------------------------
    # Guarantee non‑empty result
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator's limits and final evaluation
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Final evaluation failed: {exc}")

    return best_seq.tolist()
