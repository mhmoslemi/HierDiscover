"""
Hybrid multi‑phase optimizer for the step‑function bound.

The routine:
  1. Starts from the previous best sequence (or a random Dirichlet draw).
  2. Runs a long Adam optimisation on a smooth‑max surrogate.
  3. Refines the result with an exponentiated‑gradient (sub‑gradient) phase.
  4. Performs a final projected‑gradient descent on the exact max‑convolution.
  5. Optionally applies a random‑phase refinement on the constant‑magnitude FFT
     manifold while time permits.
All phases respect the global time‑budget (≈ 1000 s) and the best vector found
throughout is returned as a plain list of floats.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve the current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimisation of max(conv)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.15 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Randomise FFT phases while preserving magnitude (constant‑mag manifold)."""
    mag = np.abs(np.fft.fft(a0))
    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]                       # DC unchanged

        A[1:half] = mag[1:half] * np.exp(1j * phi)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 30_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence.

    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 points give a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run stored a sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] > n:
            a0 = a0[:n]
        elif a0.shape[0] < n:
            a0 = np.pad(a0, (0, n - a0.shape[0]))
        a0 = np.maximum(a0, 0.0)
    else:
        a0 = np.random.dirichlet(np.ones(n))

    if a0.sum() < 0.01:
        a0 = np.random.dirichlet(np.ones(n))
    a0 = a0 / a0.sum()                     # work on the simplex

    best_seq = a0.copy()
    conv = _full_convolution(best_seq, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Allocate time fractions for the different phases.
    total_time = deadline - time.time()
    # 1) Adam – 30 %
    t_adam = total_time * 0.30
    # 2) Sub‑gradient – 30 %
    t_sub = total_time * 0.30
    # 3) Projected GD – 25 %
    t_gd = total_time * 0.25
    # 4) Random‑phase refinement – whatever remains (≈15 %)
    # ------------------------------------------------------------------

    # ------------------- Phase 1 – Adam -----------------------------
    if time.time() < deadline - 1.0:
        sub_dead = min(deadline, time.time() + t_adam)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=1_200_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase Adam] improved to {best_val:.6f}")

    # ------------------- Phase 2 – Sub‑gradient --------------------
    if time.time() < deadline - 1.0:
        sub_dead = min(deadline, time.time() + t_sub)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=2_000_000, eta0=0.8
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase Subgrad] improved to {best_val:.6f}")

    # ------------------- Phase 3 – Projected GD --------------------
    if time.time() < deadline - 1.0:
        sub_dead = min(deadline, time.time() + t_gd)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=800_000
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Phase GD] improved to {best_val:.6f}")

    # ------------------- Phase 4 – Random phase refinement ----------
    if time.time() < deadline - 1.0:
        sub_dead = deadline
        a_phase, val_phase = _random_phase_refine(
            best_seq, n, conv_fft_len, sub_dead, max_iters=500_000
        )
        if val_phase < best_val:
            best_val = val_phase
            best_seq = a_phase.copy()
            print(f"[Phase Random] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final safeguard – ensure a feasible vector (non‑negative, sum>0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / best_seq.sum()

    # Quick final evaluation for logging
    final_conv = _full_convolution(best_seq, conv_fft_len)
    final_val = 2.0 * n * final_conv.max()
    print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
