"""
Hybrid multi–restart optimizer.

* Build a diverse pool of feasible step‑functions (warm start,
  Dirichlet draws, exponential tails, sinusoid, sparse spikes,
  block‑wise random patterns).
* Quickly evaluate each candidate by exact convolution and keep the
  few with the lowest objective.
* For every promising start run a long Adam optimisation on the
  smooth‑max surrogate (soft‑max parametrisation, aggressive
  temperature schedule, time‑limited).
* Refine the Adam result with a projected sub‑gradient / GD phase.
* Finish with a simulated‑annealing “mass‑transfer” local search that
  moves a small fraction of probability mass between two random
  coordinates while preserving the simplex.
* The best vector found before the deadline is returned as a plain
  list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv)
    (soft‑max parametrisation).  Returns the best iterate (on the simplex)
    together with its exact objective value 2·n·max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise soft‑max logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.2                 # start temperature for smooth‑max
    lam_factor = 1.06         # grow fairly fast

    best_a = a0.copy()
    # exact value (sum(a)=1 ⇒ value = 2·n·max(conv))
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- soft‑max to obtain current point on the simplex ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution & smooth‑max weights ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of smoothed objective w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through the soft‑max (logits) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e7)

        # ---- keep best exact value seen ----
        if max_c * 2.0 * n < best_val:
            best_val = 2.0 * n * max_c
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {best_val:.6f}")

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.92

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    Operates directly on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.13 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.
    A small fraction of mass is moved between two random indices.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = val

    T_start = 0.12
    T_end = 1e-5

    for mv in range(max_moves):
        if time.time() > deadline:
            break
        # linear cooling
        T = T_end + (T_start - T_end) * (1.0 - mv / float(max_moves))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] < 1e-15:
            continue

        # move a random fraction of mass from i to j
        frac = np.random.rand() * 0.35
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        # accept if better, otherwise with Metropolis probability
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
            # occasional logging
            if mv % 25000 == 0:
                print(f"[Anneal mv={mv}] new best {best_val:.6f}")
        else:
            if np.random.rand() < np.exp(-(val2 - val) / max(T, 1e-12)):
                a = a2
                val = val2

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm‑start and its cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both sides)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law shape (triangular product)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian lumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform distribution
    cand.append(np.full(n, 1.0 / n))

    # Sparse equally‑spaced spikes
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # Random sparse spikes (handful of positions)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            cand.append(a)

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
            cand.append(a)

    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the given time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 provides a good balance of expressiveness &
    # computational cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (if a previous run stored a result)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a fairly wide pool, evaluate quickly, keep the most promising.
    pool = _initial_candidates(n, warm)

    # Simple exact evaluation (sum is 1 ⇒ objective = 2·n·max(conv))
    def _exact_val(vec: np.ndarray) -> float:
        conv = _full_convolution(vec, conv_fft_len)
        return 2.0 * n * conv.max()

    # Evaluate all candidates (cheap enough – pool size ≈ 70)
    vals = [_exact_val(p) for p in pool]
    # select the 5 lowest‑valued candidates for deep optimisation
    top_k = 5
    best_indices = np.argsort(vals)[:top_k]
    selected = [pool[i] for i in best_indices]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase I – deep optimisation of the selected starts
    for idx, a0 in enumerate(selected):
        if time.time() > deadline:
            break
        # make sure we are on the simplex
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Allocate roughly equal time slices for each candidate
        remaining = deadline - time.time()
        # leave some margin for the final refinement stages
        slice_time = max(5.0, remaining / (len(selected) - idx + 3))
        sub_dead = time.time() + slice_time

        # ---- Adam -------------------------------------------------
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=2_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Candidate {idx}] Adam improved to {val_adam:.6f}")

        # ---- Projected sub‑gradient / GD (exact max) -------------
        if time.time() < deadline - 5.0:
            # reuse whatever time is left for this candidate
            sub_dead2 = deadline - 2.0
            a_gd, val_gd = _max_conv_descent(
                best_seq, n, conv_fft_len, corr_fft_len, sub_dead2,
                max_iters=300_000,
            )
            if val_gd < best_val:
                best_val = val_gd
                best_seq = a_gd.copy()
                print(f"[Candidate {idx}] GD improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase II – global simulated annealing on the current best
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Global Anneal] Starting final annealing")
        sub_dead = deadline - 1.0
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Global Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
