    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)
    # target uniform conv array

    # Build initial population
    pop = []
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pop.append(w)
    # deterministic families
    pop.append(_make_uniform(n))
    pop.append(_make_exponential(n, gamma=0.5))
    pop.append(_make_powerlaw(n, expo=-1.2))
    pop.append(_make_triangle(n))

        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = objective_val(a, n, conv_fft_len)
        scored.append((val, a))
    scored.sort(key=lambda x: x[0])
    top_k = 5

"""
Hybrid multi‑restart optimizer for the step‑function bound.
Uses a diverse initial population, L2‑flattening (Adam), smooth‑max
Adam with an annealed temperature, exact‑max exponentiated‑gradient,
and a final projected‑gradient / simulated‑annealing polish.
The routine adapts its time allocation to the supplied budget (≈1000 s)
and returns the best sequence found.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) required for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# ----- factories -----------------------------------------------------

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)

def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a

def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a

def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a

# ----- objective -----------------------------------------------------

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective = 2·n·max(conv(a,a)).  Assumes sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * float(conv.max())

# ----- Adam – smooth max -------------------------------------------

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 200_000, lr: float = 0.05) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the softmax parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# ----- Adam – L2 flattening ---------------------------------------

def _adam_optimize_l2(a0: np.ndarray, n: int, conv_fft_len: int,
                      corr_fft_len: int, target: np.ndarray,
                      deadline: float, max_iters: int = 200_000,
                      lr: float = 0.05) -> tuple[np.ndarray, float]:
    """Adam on the L2 distance to the uniform convolution target."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val

# ----- Exponentiated‑gradient (exact max) -------------------------

def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient via exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# ----- Projected gradient descent on exact max -------------------

def _max_conv_descent(a0: np.ndarray, n: int, conv_fft_len: int,
                     corr_fft_len: int, deadline: float,
                     max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# ----- Simulated annealing -----------------------------------------

def _anneal_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                     deadline: float, max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Multi‑restart search for a low‑value step‑function.
    Uses random Dirichlet seeds, L2‑flattening, smooth‑max Adam,
    exact‑max exponentiated‑gradient and a final polishing phase.
    Returns the best sequence (list of floats) found before the deadline.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 4096                                   # problem size
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Uniform target for the L2 flattening phase
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Build an initial diverse population
    pop = []

    warm = globals().get("height_sequence_1")
    if warm is not None:

        if w.shape[0] != n:

            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pop.append(w)

    # analytic families
    pop.append(_make_uniform(n))
    pop.append(_make_exponential(n, gamma=0.5))
    pop.append(_make_powerlaw(n, expo=-1.2))
    pop.append(_make_triangle(n))

    # random Dirichlet seeds with a spread of concentrations
    for _ in range(20):
        alpha = 10 ** np.random.uniform(-2, 0.5)   # between 0.01 and ~3.2
        pop.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    # Quick evaluation of the raw population and keep the most promising
    scored = []
    for a in pop:
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
    scored.sort(key=lambda x: x[0])
    top_k = 5
    seeds = [a for _, a in scored[:top_k]]

    best_seq = None
    best_val = float("inf")

        a_l2, val_l2 = _adam_optimize_l2(a0, n, conv_fft_len, corr_fft_len,
                                         target, phase_dead,
                                         max_iters=300_000, lr=0.04)

            best_val, best_seq = val_l2, a_l2.copy()
            print(f"[Seed {idx}] L2‑Adam improved to {val_l2:.6f}")

        if time.time() < deadline - 5.0:
            phase_dead = min(deadline, time.time() + min(20.0, per_seed * 0.35))
            a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len,
                                              corr_fft_len, phase_dead,
                                              max_iters=400_000)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()

                                         corr_fft_len, deadline,
                                         max_iters=200_000)
        if val_gd < best_val:

            print(f"[Fine] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: simulated annealing

                                          deadline, max_moves=100_000)
        if val_ann < best_val:

            print(f"[Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Guarantees a non‑empty return

        best_val = _objective(best_seq, n, conv_fft_len)

"""
Hybrid multi‑restart optimiser for the step‑function bound.

The routine builds a diverse initial population (including a possible warm
start, analytic families and random Dirichlet vectors), refines the most
promising seeds with three phases (L2‑flattening, smooth‑max Adam and an
exact‑max exponentiated‑gradient), then runs a short genetic‑mixing loop,
and finally polishes the best solution with projected gradient descent
followed by a simulated‑annealing pass.  The algorithm adapts its time
allocation to the supplied budget (≈ 1000 s) and returns the best sequence
found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Analytic factories ----------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective  = 2·n·max(conv(a,a)).  Assumes Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * float(conv.max())

# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max -----------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 200_000, lr: float = 0.05) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser – L2 flattening ---------------------------------------
def _adam_optimize_l2(a0: np.ndarray, n: int, conv_fft_len: int,
                      corr_fft_len: int, target: np.ndarray,
                      deadline: float, max_iters: int = 200_000,
                      lr: float = 0.05) -> tuple[np.ndarray, float]:
    """Adam on the L2 distance to the uniform convolution target."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient --------------------------------------
def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient via exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on exact max -------------------------------
def _max_conv_descent(a0: np.ndarray, n: int, conv_fft_len: int,
                     corr_fft_len: int, deadline: float,
                     max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing ---------------------------------------------------
def _anneal_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                     deadline: float, max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main search routine ---------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    warm = globals().get("height_sequence_1")
    if warm is not None:

        if w.shape[0] != n:

            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pop.append(w)

    pop.append(_make_uniform(n))
    pop.append(_make_exponential(n, gamma=0.5))
    pop.append(_make_powerlaw(n, expo=-1.2))
    pop.append(_make_triangle(n))

    for _ in range(20):
        alpha = 10 ** np.random.uniform(-2, 0.5)   # ∈[0.01, ~3.2]
        pop.append(_make_dirichlet(n, alpha))

    # --------------------------------------------------------------
    # Quick scoring of the raw population
    scored = []
    for a in pop:
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
    scored.sort(key=lambda x: x[0])
    top_k = 5
    seeds = [a for _, a in scored[:top_k]]

        if time.time() > deadline:

        a_l2, val_l2 = _adam_optimize_l2(a0, n, conv_fft_len, corr_fft_len,
                                         target, phase_dead,

            print(f"[Seed {idx}] L2‑Adam improved to {val_l2:.6f}")

        if time.time() < deadline - 5.0:
            phase_dead = min(deadline, time.time() + min(20.0, per_seed * 0.35))
            a_adam, val_adam = _adam_optimize(best_seq, n,
                                              conv_fft_len, corr_fft_len,
                                              phase_dead, max_iters=400_000)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[Seed {idx}] Smooth‑max Adam improved to {val_adam:.6f}")

        # 3️⃣ Exact‑max exponentiated‑gradient
        if time.time() < deadline - 5.0:
            phase_dead = min(deadline, time.time() + min(20.0, per_seed * 0.35))
            a_exp, val_exp = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len,
                phase_dead, max_iters=500_000, eta0=0.8)
            if val_exp < best_val:
                best_val, best_seq = val_exp, a_exp.copy()
                print(f"[Seed {idx}] ExpGrad improved to {val_exp:.6f}")

    # --------------------------------------------------------------
    # Simple elite‑based genetic mixing while time permits
    elite = [(best_val, best_seq.copy())] if best_seq is not None else []
    for a in seeds:
        elite.append((_objective(a, n, conv_fft_len), a.copy()))
    elite.sort(key=lambda x: x[0])
    elite = elite[:5]

    while time.time() < deadline - 15.0:
        # pick two distinct parents from elite
        i1, i2 = np.random.choice(len(elite), size=2, replace=False)
        p1 = elite[i1][1]
        p2 = elite[i2][1]

        lam = np.random.rand()
        child = lam * p1 + (1.0 - lam) * p2

        eps = 1e-12
        logits = np.log(child + eps)
        logits += np.random.normal(scale=0.04, size=n)
        child = np.exp(logits)
        child = child / child.sum()

        # a few cheap exponentiated‑gradient steps (500 iters)
        a = child.copy()
        for it in range(500):
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-14)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
            eta = 0.2 / np.sqrt(it + 1.0)
            a = a * np.exp(-eta * grad)
            a = a / a.sum()

        child_val = _objective(a, n, conv_fft_len)

        if child_val < best_val:
            best_val, best_seq = child_val, a.copy()
            print(f"[Mix] New global best {child_val:.6f}")
            elite.append((child_val, a.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:5]

        if time.time() > deadline - 15.0:
            break

    # --------------------------------------------------------------
    # Final polishing phases
    if best_seq is not None and time.time() < deadline - 5.0:
        a_gd, val_gd = _max_conv_descent(best_seq, n, conv_fft_len,
                                         corr_fft_len, deadline,
                                         max_iters=200_000)
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    if best_seq is not None and time.time() < deadline - 5.0:
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len,
                                          deadline, max_moves=100_000)
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = _objective(best_seq, n, conv_fft_len)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final reporting using the supplied evaluation function
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Evaluation error: {exc}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
