```
import numpy as np, scipy.optimize as opt, scipy.sparse as sp

def lp_step(f, S):
    n = len(f)
    conv_f = _full_convolution(f, ... ) ; M = conv_f.max()
    # Build A matrix
    diag_data = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(diag_data, offsets=np.arange(n), shape=(2*n-1, n), format='csc')
    # inequality: A @ b <= M
    # also b>=0
    # objective: maximize sum(b)
    # convert to minimization: minimize -sum(b)
    c = -np.ones(n)  # minimize -sum b
    bounds = [(0, None)]*n
    # SciPy linprog only accepts dense A_ub; but it now supports sparse A_ub as well for HiGHS
    res = opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None, None
    b = res.x
    # scale b to have sum = S (same as f)
    sum_b = b.sum()
    if sum_b < 1e-12:
        b = f.copy()
        sum_b = b.sum()
    g = b * (S / sum_b)
    return g, M
```

"""
Propose a step‑function by alternating a linear‑programming (LP) improvement
step with a random‑phase refinement on the constant‑magnitude FFT manifold.

The LP step (following the “alternating LP” idea from the statement) solves:

    maximise   Σ b_j
    subject to (f * b)_k ≤ M  for all k = 0 … 2n‑2
               b_j ≥ 0

where M = max(f*f) for the current sequence f.  The solution b is scaled to
the same ℓ₁‑norm as f (S = √(2 n)) which guarantees a strictly smaller
maximum convolution value.  Repeating the LP a few times rapidly pushes the
objective down.

After the LP phase a cheap random‑phase exploration keeps the FFT magnitudes
(almost) constant while staying on the probability simplex; this often yields
a final polishing improvement.

The routine respects the 1000 s budget, never returns an empty sequence and
always returns a list of non‑negative floats.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _random_phase_refine(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Keep the FFT magnitude of a0 fixed and repeatedly randomise the phases.
    After each inverse FFT the vector is projected onto the simplex.
    The best vector (lowest exact max‑convolution) is retained.
    """
    mag = np.abs(np.fft.fft(a0))

    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]                               # DC unchanged

        # positive frequencies
        A[1:half] = mag[1:half] * np.exp(1j * phi)

        # negative frequencies (complex conjugates)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            # Nyquist frequency (real) – random sign
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))

        # project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 15_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence(...).  The routine respects the given time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    target_sum = np.sqrt(2.0 * n)            # normalisation used by the paper

    # ------------------------------------------------------------------
    # Warm start from the previous run if possible
    warm = globals().get("height_sequence_1")
    if warm is not None:
        f = np.asarray(warm, dtype=np.float64)
        if f.shape[0] > n:
            f = f[:n]
        elif f.shape[0] < n:
            f = np.pad(f, (0, n - f.shape[0]))
    else:
        # a sparse Dirichlet draw – gives a good starting point for the LP
        f = np.random.dirichlet(0.1 * np.ones(n))

    f = np.maximum(f, 0.0)
    if f.sum() < 1e-12:                      # degenerate – draw a fresh one
        f = np.random.dirichlet(np.ones(n))
    f = f / f.sum() * target_sum

    best_seq = f.copy()
    best_val = evaluate_sequence(f.tolist())
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – LP driven improvement (repeat until time runs low)
    lp_iter = 0
    # keep a few seconds for the final refinement phases
    lp_deadline = deadline - 30.0
    while time.time() < lp_deadline:
        # 1) compute current max‑convolution bound M
        conv_f = _full_convolution(f, conv_fft_len)
        M = conv_f.max()

        # 2) build the (2n‑1) × n convolution matrix A (sparse, constant diagonals)
        #    A[k, j] = f[k‑j]  (0‑based, out‑of‑range entries omitted)
        diag_data = [np.full(n, f[d]) for d in range(n)]
        offsets = np.arange(n)               # 0 … n‑1
        A = _sp.diags(diag_data, offsets=offsets,
                      shape=(2 * n - 1, n), format='csc')

        # 3) linear programme: maximise Σ b  <=> minimise -Σ b
        c = -np.ones(n)                      # objective coefficients
        b_ub = np.full(2 * n - 1, M)         # right‑hand side of the convolution constraints
        bounds = [(0.0, None)] * n           # b_j ≥ 0

        # solve
        res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                           bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP iter {lp_iter}] linprog failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= target_sum * (1.0 + 1e-12):
            # no increase in total mass – further LP steps cannot improve
            print(f"[LP iter {lp_iter}] no sum increase (sum_b={sum_b:.6f})")
            break

        # 4) scale b back to the target ℓ₁‑norm
        scale = target_sum / sum_b
        g = b * scale

        # 5) evaluate candidate g
        val_g = evaluate_sequence(g.tolist())
        if val_g < best_val - 1e-9:
            best_val = val_g
            best_seq = g.copy()
            print(f"[LP iter {lp_iter}] improved to {val_g:.6f}")

        # 6) for the next round we move partially towards g (convex combination)
        #    this stabilises the iteration and keeps all vectors feasible.
        t = 0.5
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0.0)
        f = f / f.sum() * target_sum

        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase II – random‑phase refinement on the constant‑magnitude FFT manifold
    # (a lightweight final polishing step)
    if time.time() < deadline - 5.0:
        try:
            a_rand, val_rand = _random_phase_refine(
                best_seq, n, conv_fft_len, deadline, max_iters=200_000
            )
            if val_rand < best_val:
                best_val = val_rand
                best_seq = a_rand.copy()
                print(f"[PhaseRandom] improved to {val_rand:.6f}")
        except Exception as exc:
            print("[PhaseRandom] exception:", exc)

    # ------------------------------------------------------------------
    # Final safety – ensure every entry respects the evaluator's limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
