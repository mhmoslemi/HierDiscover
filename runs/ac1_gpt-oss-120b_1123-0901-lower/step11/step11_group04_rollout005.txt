def adam_l2_flatten(a0, n, conv_len, corr_len, target, deadline,
                    max_iters=500_000, lr=0.05):
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # convert to a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # conv
        conv = _full_convolution(a, conv_len)
        # compute val
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2 {it:06d}] new best {val:.6f}")
        # gradient L2
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_len)
        # ADAM step for logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        logits -= step
        it += 1
    # Final compute best_val to ensure correct
    conv = _full_convolution(best_a, conv_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val

def adam_smooth_max(a0, n, conv_len, corr_len, deadline,
                    max_iters=400_000, lr=0.05):
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1,beta2, eps_adam = 0.9,0.999,1e-8
    lam = 0.1
    lam_factor = 1.02
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[SmoothAdam {it:06d}] new best {val:.6f}")
        # smooth-max weights
        # compute w = softmax(lam*(conv - max_c))
        # In case lam large leads to overflow, use stable method
        shifted = lam * (conv - max_c)
        # subtract max of shifted for numerical stability
        w_exp = np.exp(shifted - shifted.max())
        w = w_exp / w_exp.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)
        # chain rule via logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log**2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step
        lam = min(lam * lam_factor, 1e7)
        it += 1
    conv = _full_convolution(best_a, conv_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val

def expgrad_subgradient(a0, n, conv_len, corr_len, deadline,
                        max_iters=800_000, eta0=0.8):

    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()

            best_val = val
            best_a = a.copy()

                print(f"[ExpGrad {it:06d}] new best {val:.6f}")
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)

        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    best_val = 2.0 * n * conv.max()
    return best_a, best_val

def anneal_opt(a0, n, conv_len, deadline, max_moves=80000):
    a = a0.copy()
    conv = _full_convolution(a, conv_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()
    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.4
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:

"""
Hybrid multi‑phase optimiser for the step‑function objective.

The algorithm works on the probability simplex (Σa = 1, a_i ≥ 0) and
alternates the following phases:

1.  L2‑flattening: minimise the squared distance between the
    self‑convolution a∗a and the uniform target (the theoretical lower
    bound) using Adam on a soft‑max parametrisation.
2.  Smooth‑max Adam: minimise a differentiable log‑sum‑exp surrogate
    of max(a∗a) while the “temperature” is raised gradually.
3.  Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
4.  Projected gradient descent on the exact max (optional fine‑tune).
5.  Simulated‑annealing local search for a final polish.

The total wall‑time is bounded by ``budget_s`` seconds; each phase only
runs while time remains.  A small pool of diverse initial vectors is
generated (warm start, Dirichlet draws, exponential, power‑law, triangle,
uniform).  After the pool is exhausted a few jittered variants of the
current best are explored to escape local minima.

All helpers are top‑level functions – no closures – and the final
sequence is returned as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories --------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------
def _adam_l2_flatten(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on L2 distance ‖a∗a – target‖₂.
    Returns the best vector encountered and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- convert logits → a (simplex) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & objective -----
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # ----- gradient of L2 loss -----
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_len)

        # ----- chain rule through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- optional learning‑rate decay -----
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    # recompute final objective for the best vector
    conv = _full_convolution(best_a, conv_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max ‖max_λ(a∗a)‖,
    where the smooth max is the log‑sum‑exp surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[SmoothAdam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (stable) -----
        shifted = lam * (conv - max_c)
        w_exp = np.exp(shifted - shifted.max())
        w = w_exp / w_exp.sum()

        # ----- gradient of smoothed objective -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # ----- chain rule through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- increase temperature -----
        lam = min(lam * lam_factor, 1e7)

        # ----- modest learning‑rate decay -----
        if it % 150_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    conv = _full_convolution(best_a, conv_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _expgrad_subgradient(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    conv = _full_convolution(best_a, conv_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _max_conv_gd(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent directly on the exact max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    conv = _full_convolution(best_a, conv_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    conv = _full_convolution(best_a, conv_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine --------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ budget_s seconds. Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size (feel free to raise if time permits)
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best) if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())

    # Dirichlet draws with various concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))

    # Analytic families
    pool.append(_make_uniform(n))
    pool.append(_make_exponential(n, gamma=0.5))
    pool.append(_make_powerlaw(n, expo=-1.2))
    pool.append(_make_triangle(n))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Uniform target for L2 flattening (the lower bound distribution)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to run the whole optimisation pipeline on a given start vector
    def _run_pipeline(a_start: np.ndarray, remaining_deadline: float) -> tuple[np.ndarray, float]:
        """Runs L2‑flatten → smooth‑max → exp‑grad → GD → anneal."""
        # Ensure feasible start
        a_start = np.maximum(a_start, 0.0)
        if a_start.sum() < 0.01:
            a_start = np.random.dirichlet(np.ones(n))
        a_start = a_start / a_start.sum()

        # Phase 1 – L2 flattening
        phase_dead = min(remaining_deadline, time.time() + 20.0)
        a_l2, val_l2 = _adam_l2_flatten(
            a_start,
            n,
            conv_len,
            corr_len,
            target,
            phase_dead,
            max_iters=300_000,
            lr=0.05,
        )
        cur_best_a, cur_best_val = a_l2, val_l2

        # Phase 2 – smooth‑max Adam
        if time.time() < deadline - 5:
            phase_dead = min(remaining_deadline, time.time() + 15.0)
            a_smooth, val_smooth = _adam_smooth_max(
                cur_best_a,
                n,
                conv_len,
                corr_len,
                phase_dead,
                max_iters=300_000,
                lr=0.05,
            )
            if val_smooth < cur_best_val:
                cur_best_a, cur_best_val = a_smooth, val_smooth

        # Phase 3 – exponentiated‑gradient sub‑gradient
        if time.time() < deadline - 5:
            a_exp, val_exp = _expgrad_subgradient(
                cur_best_a,
                n,
                conv_len,
                corr_len,
                deadline,
                max_iters=600_000,
                eta0=0.8,
            )
            if val_exp < cur_best_val:
                cur_best_a, cur_best_val = a_exp, val_exp

        # Phase 4 – fine‑grained projected GD
        if time.time() < deadline - 5:
            a_gd, val_gd = _max_conv_gd(
                cur_best_a,
                n,
                conv_len,
                corr_len,
                deadline,
                max_iters=200_000,
            )
            if val_gd < cur_best_val:
                cur_best_a, cur_best_val = a_gd, val_gd

        # Phase 5 – simulated‑annealing refinement
        if time.time() < deadline - 5:
            a_ann, val_ann = _anneal_optimize(
                cur_best_a,
                n,
                conv_len,
                deadline,
                max_moves=80_000,
            )
            if val_ann < cur_best_val:
                cur_best_a, cur_best_val = a_ann, val_ann

        return cur_best_a, cur_best_val

    # ------------------------------------------------------------------
    # Primary optimisation over the initial pool
    while pool and time.time() < deadline:
        a0 = pool.pop(0)
        a_opt, val_opt = _run_pipeline(a0, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Pool] improved best to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # If time permits, explore jittered variants of the current best
    jitter_round = 0
    while time.time() < deadline - 10 and best_seq is not None and jitter_round < 5:
        jitter_round += 1
        sigma = 0.05 * (1.0 - jitter_round / 5.0)  # gradually shrink
        a_jitter = best_seq + sigma * np.random.randn(n)
        a_jitter = np.maximum(a_jitter, 0.0)
        if a_jitter.sum() < 0.01:
            a_jitter = np.random.dirichlet(np.ones(n))
        a_jitter = a_jitter / a_jitter.sum()
        a_opt, val_opt = _run_pipeline(a_jitter, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Jitter {jitter_round}] improved best to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a uniform distribution in the extremely unlikely case
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = float("inf")

    # Clip to the allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation for reporting
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
