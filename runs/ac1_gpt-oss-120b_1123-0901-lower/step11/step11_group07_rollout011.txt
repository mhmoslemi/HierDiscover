    rev_a = a[::-1]

    fr = np.fft.rfft(rev_a, fft_len)

def _project_simplex(v):
    # Euclidean projection onto simplex
    n = v.shape[0]

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Search for a non‑negative step function minimizing
    2·n·max(conv(a,a)) / (sum(a)^2)

The optimizer uses three stages:
1. Random sampling from several parametric families (beta, Dirichlet,
   exponential, Gaussian, power‑law, spikes, triangular, sine, cosine,
   Laplacian, uniform).  The best sample is kept.
2. A long exponentiated‑gradient refinement that works directly on the
   exact max‑convolution (sub‑gradient on the active set).  The iteration
   count is limited only by the remaining time budget.
3. A cheap random‑move local search (mass transfer between two positions)
   that can still squeeze a few percent out of the result.

All helpers are top‑level, the code respects a 1000 s budget and always
returns a feasible (non‑negative, sum ≥ 0.01) sequence.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    # keep only the genuine linear part
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Sub‑gradient optimiser (exponentiated‑gradient on exact max)

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient optimisation using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active‑set mask (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing learning rate
        eta = eta0 / math.sqrt(it + 1.0)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap stochastic local search (mass transfer between two indices)

def _local_search_random(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_steps: int = 200_000,
    step_scale: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Random neighbour search: repeatedly move a small fraction of mass
    from a random source index to a random target index.
    """
    best_a = a.copy()
    conv_len = 2 * n - 1
    conv_fft_len = _next_pow2(conv_len)
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_steps:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best_a[i] < 1e-15:
            it += 1
            continue

        # move a random fraction of the source mass
        frac = np.random.rand() * step_scale
        delta = best_a[i] * frac

        a2 = best_a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2
            print(f"[Local it={it:07d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The algorithm mixes random sampling, a long exponentiated‑gradient
    descent on the exact max‑convolution, and a cheap stochastic local
    search.  It always returns a feasible sequence before the budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0            # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (4096 gives a good trade‑off)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Warm start from a previously found construction if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            val = evaluate_sequence(warm.tolist())
            if val < best_val:
                best_val = val
                best_seq = warm.copy()
                print(f"[Warm] initial value {val:.6f}")

    # ------------------------------------------------------------------
    # Random pool of parametric families
    def gen_uniform():
        return np.full(n, 1.0 / n, dtype=np.float64)

    def gen_dirichlet():
        alpha = 10 ** np.random.uniform(-3, 2)           # log‑uniform 1e‑3 … 100
        a = np.random.dirichlet(alpha * np.ones(n))
        return a.astype(np.float64)

    def gen_beta():
        a = np.random.uniform(0.1, 5.0)
        b = np.random.uniform(0.1, 5.0)
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        eps = 1e-12
        x = np.clip(x, eps, 1.0 - eps)
        vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
        if vals.sum() == 0.0:
            return np.full(n, 1.0 / n, dtype=np.float64)
        return vals / vals.sum()

    def gen_exponential():
        gamma = np.random.uniform(0.05, 2.0)
        i = np.arange(n, dtype=np.float64)
        left = np.exp(-gamma * i)
        right = left[::-1]
        a = left + right
        a /= a.sum()
        return a.astype(np.float64)

    def gen_gaussian():
        sigma = np.random.uniform(0.01, 0.4)
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((i - (n - 1) / 2.0) / (sigma * n)) ** 2)
        a /= a.sum()
        return a.astype(np.float64)

    def gen_powerlaw():
        expo = np.random.choice([-0.5, -1.0, -1.5, -2.0, -2.5, -3.0])
        i = np.arange(1, n + 1, dtype=np.float64)
        a = i ** expo
        a = np.where(a > 0, a, 0.0)
        if a.sum() == 0.0:
            return np.full(n, 1.0 / n, dtype=np.float64)
        a /= a.sum()
        return a.astype(np.float64)

    def gen_spikes():
        k = np.random.choice([8, 16, 32, 64, 128, 256])
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        a /= a.sum()
        return a

    def gen_half_sine():
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 0.5) / n)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        return a.astype(np.float64)

    def gen_cosine():
        i = np.arange(n, dtype=np.float64)
        a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        return a.astype(np.float64)

    def gen_laplacian():
        gamma = np.random.uniform(0.01, 0.2)
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        a = np.exp(-gamma * np.abs(i - centre))
        a /= a.sum()
        return a.astype(np.float64)

    def gen_triangle():
        i = np.arange(1, n + 1, dtype=np.float64)
        a = i * (n - i + 1)
        a /= a.sum()
        return a.astype(np.float64)

    factories = [
        gen_uniform,
        gen_dirichlet,
        gen_beta,
        gen_exponential,
        gen_gaussian,
        gen_powerlaw,
        gen_spikes,
        gen_half_sine,
        gen_cosine,
        gen_laplacian,
        gen_triangle,
    ]

    # Sample a few hundred candidates (time‑bounded)
    pool_iters = 0
    while time.time() < deadline - 30.0 and pool_iters < 800:
        factory = np.random.choice(factories)
        a = factory()
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            continue
        a = a / a.sum()
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Pool] new best {val:.6f} (factory {factory.__name__})")
        pool_iters += 1

    # ------------------------------------------------------------------
    # Main refinement – exponentiated‑gradient on exact max‑convolution
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 20.0:
            sub_dead = time.time() + time_left * 0.65   # 65 % of remaining time
            print(f"[Refine] exp‑grad start, {time_left:.1f}s left")
            best_seq, best_val = _subgradient_optimize_expgrad(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_dead,
                max_iters=2_500_000,
                eta0=0.7,
            )
            print(f"[Refine] after exp‑grad best {best_val:.6f}")

        # ------------------------------------------------------------------
        # Final cheap local random search
        if time.time() < deadline - 5.0:
            print("[Local] random neighbour search")
            best_seq, best_val = _local_search_random(
                best_seq,
                n,
                deadline,
                max_steps=250_000,
                step_scale=0.15,
            )
            print(f"[Local] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"[Fallback] uniform value {best_val:.6f}")

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    total_time = time.time() - start_time
    print(f"Finished ({total_time:.1f}s). Best value {best_val:.6f} (len {len(best_seq)})")
    return best_seq.tolist()
