import time
import numpy as np
from scipy import special as _sp  # only for the Kaiser window


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep the valid linear part
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for the sub‑gradient of max(conv).
    Returns a vector of length ``len(a)``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is 1) – 2·n·max(conv)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * a.shape[0] * conv.max()


def _exponential_gradient_step(a: np.ndarray,
                               conv_fft_len: int,
                               corr_fft_len: int,
                               eta: float) -> tuple[np.ndarray, float]:
    """
    One exponentiated‑gradient (mirror‑descent) step on the simplex.
    Returns the new vector and its objective value.
    """
    # convolution and active set
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    active = np.isclose(conv, max_c, atol=1e-13)
    w = np.zeros_like(conv)
    if active.any():
        w[active] = 1.0 / active.sum()

    # sub‑gradient of the *max* (scaled by 2)
    grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # exponentiated‑gradient update (stay strictly positive)
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s == 0 or np.isnan(s):
        # numerical fallback – small uniform noise
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s

    val = 2.0 * a.shape[0] * max_c
    return a_new, val


# ----------------------------------------------------------------------
def _generate_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a heterogeneous set of start vectors."""
    pool = []

    # Warm start (if any)
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w.copy())
            # a few shifts – they often produce a different local basin
            for sh in (n // 4, n // 2, 3 * n // 4):
                pool.append(np.roll(w, sh))

    # ------------------- analytic windows ---------------------------
    i = np.arange(n, dtype=np.float64)

    # Hann (raised‑cosine) – zero at both ends
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    pool.append(hann)

    # Hamming
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    hamming /= hamming.sum()
    pool.append(hamming)

    # Blackman
    blackman = (0.42
                - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
                + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    blackman /= blackman.sum()
    pool.append(blackman)

    # Gaussian families (σ relative to n)
    for sigma in (0.12, 0.18, 0.25, 0.35):
        gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
        gauss /= gauss.sum()
        pool.append(gauss)

    # Kaiser windows (β controls side‑lobe level)
    for beta in (2.0, 4.0, 6.0, 8.0):
        z = (2.0 * i) / (n - 1) - 1.0
        kaiser = _sp.i0(beta * np.sqrt(1 - z * z))
        kaiser /= kaiser.sum()
        pool.append(kaiser)

    # Half‑wave sinusoid (already used but keep as baseline)
    half_sine = np.sin(np.pi * (i + 0.5) / n)
    half_sine = np.maximum(half_sine, 0.0)
    half_sine /= half_sine.sum()
    pool.append(half_sine)

    # Exponential tails (both sides)
    for gamma in (0.15, 0.30, 0.60):
        left = np.exp(-gamma * i)
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law (inverse‑parabolic) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.6, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Random Dirichlet draws – several concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        pool.append(a)

    # Random sparse patterns (a handful of equally‑spaced spikes)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(a)

    # Random piece‑wise constant blocks (coarse granularity)
    for blocks in (8, 16, 32, 64):
        block_len = max(1, n // blocks)
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len: (b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
            pool.append(a)

    return pool


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Find a non‑negative step‑function with a low evaluation score.
    The routine respects a wall‑clock budget (≈ 1000 s) and always
    returns the best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # problem size – 4096 works well for the time budget
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # warm start (previous run) if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        # we accept the global variable only if it looks reasonable
        try:
            warm = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm = None

    # --------------------------------------------------------------
    # Phase I – broad exploration of analytic families
    pool = _generate_initial_pool(n, warm)

    best_seq = None
    best_val = float("inf")
    print(f"[Init] evaluating {len(pool)} starter vectors ...")

    for idx, cand in enumerate(pool):
        if time.time() > deadline:
            break
        # safety: enforce non‑negativity and normalisation
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 1e-12:
            cand = np.full(n, 1.0 / n)
        else:
            cand = cand / s

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Init] new best #{idx} → {val:.6f}")

    # --------------------------------------------------------------
    # Phase II – long‑run exponentiated‑gradient refinement
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # make sure we start from a strictly positive vector (=> safe log/exp)
    eps = 1e-12
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    # algorithm parameters
    eta0 = 0.6               # initial learning rate for EG
    it = 0
    print("[Refine] start long projected‑gradient refinement")
    while time.time() < deadline:
        eta = eta0 / np.sqrt(it + 1.0)
        best_seq, cur_val = _exponential_gradient_step(
            best_seq, conv_fft_len, corr_fft_len, eta
        )
        # keep the global best (objective value is 2·n·max(conv))
        if cur_val < best_val:
            best_val = cur_val
            print(f"[Refine] it={it:07d} improved → {cur_val:.6f}")
        it += 1
        # occasional tiny random perturbation to escape plateaus
        if it % 5000 == 0 and (time.time() + 2.0) < deadline:
            noise = np.random.randn(n) * 2e-4
            pert = best_seq + noise
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            # one quick EG step on the perturbed point
            pert, pert_val = _exponential_gradient_step(
                pert, conv_fft_len, corr_fft_len, eta0 / np.sqrt(it + 1.0)
            )
            if pert_val < best_val:
                best_seq = pert
                best_val = pert_val
                print(f"[Perturb] it={it:07d} new best → {pert_val:.6f}")

    # --------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final clipping to satisfy the evaluator’s hard limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[Done] final best value ≈ {best_val:.6f} (length {n})")
    return best_seq.tolist()
