def _fourier_adam_opt(alpha0: np.ndarray, beta0: np.ndarray, n: int, K: int,
                      conv_fft_len: int, corr_fft_len: int,
                      deadline: float, cos_mat: np.ndarray, sin_mat: np.ndarray,
                      max_iters: int = 400_000):
    # Initialize parameters
    alpha = alpha0.copy()
    beta = beta0.copy()
    m_alpha = np.zeros_like(alpha)
    v_alpha = np.zeros_like(alpha)
    m_beta = np.zeros_like(beta)
    v_beta = np.zeros_like(beta)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.2
    lam_factor = 1.025

    best_a = None
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # construct sequence a
        # log_a = alpha @ cos_mat + beta @ sin_mat
        log_a = np.dot(alpha, cos_mat) + np.dot(beta, sin_mat)
        # stable softmax
        log_a_shift = log_a - np.max(log_a)
        a = np.exp(log_a_shift)
        sum_a = a.sum()
        if sum_a == 0 or np.isnan(sum_a):
            a = np.random.dirichlet(np.ones(n))
            sum_a = a.sum()
        a = a / sum_a

        # conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Fourier Adam it={it:07d}] new best {val:.6f}")

        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)  # shape (n,)

        # gradient w.r.t Fourier parameters
        grad_alpha = np.dot(grad_log, cos_mat.T)  # shape (K,)
        grad_beta = np.dot(grad_log, sin_mat.T)   # shape (K,)

        # Adam update for alpha
        m_alpha = beta1 * m_alpha + (1 - beta1) * grad_alpha
        v_alpha = beta2 * v_alpha + (1 - beta2) * (grad_alpha ** 2)
        m_hat_alpha = m_alpha / (1 - beta1 ** (it + 1))
        v_hat_alpha = v_alpha / (1 - beta2 ** (it + 1))
        step_alpha = lr * m_hat_alpha / (np.sqrt(v_hat_alpha) + eps_adam)
        alpha -= step_alpha

        # Adam update for beta
        m_beta = beta1 * m_beta + (1 - beta1) * grad_beta
        v_beta = beta2 * v_beta + (1 - beta2) * (grad_beta ** 2)
        m_hat_beta = m_beta / (1 - beta1 ** (it + 1))
        v_hat_beta = v_beta / (1 - beta2 ** (it + 1))
        step_beta = lr * m_hat_beta / (np.sqrt(v_hat_beta) + eps_adam)
        beta -= step_beta

        lam = min(lam * lam_factor, 1e7)

        # occasional learning rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a if best_a is not None else a, best_val

"""
Propose a low‑value step‑function.

The search consists of two complementary parts:

1. **Low‑dimensional Fourier optimisation** –  
   The sequence is represented as a soft‑max of a truncated Fourier
   series (only the first K cosine/sine modes of the log‑density are
   kept).  Adam is applied to the K‑dimensional parameter vector using a
   smooth‑max surrogate of the max‑convolution.  Because the number of
   variables is tiny (≈ 2·K) we can afford many iterations and obtain a
   high‑quality shape that is difficult to reach with full‑dimensional
   optimisation.

2. **Fine‑tuning on the full vector** –  
   The best sequence from the Fourier stage is fed to the previously
   successful pipeline (spectral flattening → projected gradient descent
   → simulated annealing → short perturb‑and‑refine cycles).  This
   combination keeps the strengths of the original hybrid optimiser while
   exploring a different region of the search space.

All phases respect the global time budget (≈ 1000 s) and the routine
always returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for sub‑gradient / gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _fourier_adam_opt(alpha0: np.ndarray,
                      beta0: np.ndarray,
                      n: int,
                      K: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      cos_mat: np.ndarray,
                      sin_mat: np.ndarray,
                      max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a low‑dimensional Fourier parametrisation of the log‑density.
    The vector a is obtained as the soft‑max of log_a = Σ α_k cos + Σ β_k sin,
    and a is normalised to unit sum.
    The loss is the smooth‑max surrogate of 2·n·max(conv(a,a)).
    """
    # initialise Adam state
    alpha = alpha0.copy()
    beta = beta0.copy()
    m_alpha = np.zeros_like(alpha)
    v_alpha = np.zeros_like(alpha)
    m_beta = np.zeros_like(beta)
    v_beta = np.zeros_like(beta)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.2
    lam_factor = 1.025

    best_a = None
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- construct the full‑dimensional sequence from the Fourier coeffs ----
        log_a = np.dot(alpha, cos_mat) + np.dot(beta, sin_mat)      # shape (n,)
        # stable soft‑max
        log_a_shift = log_a - np.max(log_a)
        a = np.exp(log_a_shift)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # ---- convolution and smooth‑max weight ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Fourier Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights (temperature = 1/lam)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (sub‑gradient of the smooth‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (grad w.r.t. log‑a)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)          # shape (n,)

        # gradient w.r.t. Fourier parameters
        grad_alpha = np.dot(grad_log, cos_mat.T)   # (K,)
        grad_beta = np.dot(grad_log, sin_mat.T)    # (K,)

        # ---- Adam update for α ----
        m_alpha = beta1 * m_alpha + (1.0 - beta1) * grad_alpha
        v_alpha = beta2 * v_alpha + (1.0 - beta2) * (grad_alpha ** 2)
        m_hat_alpha = m_alpha / (1.0 - beta1 ** (it + 1))
        v_hat_alpha = v_alpha / (1.0 - beta2 ** (it + 1))
        step_alpha = lr * m_hat_alpha / (np.sqrt(v_hat_alpha) + eps_adam)
        alpha -= step_alpha

        # ---- Adam update for β ----
        m_beta = beta1 * m_beta + (1.0 - beta1) * grad_beta
        v_beta = beta2 * v_beta + (1.0 - beta2) * (grad_beta ** 2)
        m_hat_beta = m_beta / (1.0 - beta1 ** (it + 1))
        v_hat_beta = v_beta / (1.0 - beta2 ** (it + 1))
        step_beta = lr * m_hat_beta / (np.sqrt(v_hat_beta) + eps_adam)
        beta -= step_beta

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    # fallback – if nothing was stored (unlikely) return the last a
    if best_a is None:
        best_a = a
        best_val = 2.0 * n * np.max(_full_convolution(best_a, conv_fft_len))
    return best_a, best_val

def _spectral_flatten_optimize(a0: np.ndarray,
                               n: int,
                               conv_fft_len: int,
                               deadline: float,
                               max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Iteratively force the FFT magnitude to be (almost) constant."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")

        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s
        it += 1
    return best_a, best_val


def _max_conv_descent(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")
        it += 1
    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1
    return best_a, best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    for gamma in (0.15, 0.3, 0.6):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.6, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.45):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))

    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        cand.append(a)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (48, 96, 192):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (length ≈ 4096) with a small
    evaluation score.  The routine runs several phases:
      1. Fourier‑based Adam optimisation (low‑dimensional representation).
      2. Spectral flattening.
      3. Projected gradient descent on the exact max‑convolution.
      4. Simulated annealing.
      5. Short perturb‑and‑refine cycles.
    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Phase 0 – Fourier Adam optimisation
    K = 20                                 # number of cosine/sine modes
    if warm is not None:
        log_warm = np.log(np.maximum(warm, 1e-12))
        fft = np.fft.rfft(log_warm)
        max_k = min(K, fft.shape[0] - 1)
        alpha0 = np.zeros(K)
        beta0 = np.zeros(K)
        for k in range(1, max_k + 1):
            c = fft[k] / n
            alpha0[k - 1] = 2.0 * np.real(c)
            beta0[k - 1] = -2.0 * np.imag(c)
    else:
        alpha0 = np.random.randn(K) * 0.1
        beta0 = np.random.randn(K) * 0.1

    # pre‑compute trig basis
    i = np.arange(n, dtype=np.float64)
    k_vec = np.arange(1, K + 1, dtype=np.float64)[:, None]
    cos_mat = np.cos(2.0 * np.pi * k_vec * i / n)   # (K, n)
    sin_mat = np.sin(2.0 * np.pi * k_vec * i / n)   # (K, n)

    print("[Fourier] Starting low‑dimensional Adam optimisation")
    a_fourier, val_fourier = _fourier_adam_opt(alpha0, beta0, n, K,
                                                conv_fft_len, corr_fft_len,
                                                deadline,
                                                cos_mat, sin_mat,
                                                max_iters=600_000)
    best_seq = a_fourier.copy()
    best_val = val_fourier
    print(f"[Fourier] finished with value {val_fourier:.6f}")

    # --------------------------------------------------------------
    # Phase I – deep optimisation of a handful of diverse starters (optional)
    # We keep a small pool of extra candidates; if time permits we run the
    # original hybrid optimiser on them.
    if time.time() < deadline - 30.0:
        pool = _initial_candidates(n, warm) + _spike_candidates(n)
        max_candidates = 6
        for idx, a0 in enumerate(pool):
            if time.time() > deadline or idx >= max_candidates:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 0.01:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()
            sub_dead = min(deadline, time.time() + 15.0)

            # Re‑use the previously defined Adam optimiser on the full vector
            # (light version – fewer iterations) to polish the candidate.
            a_adam, val_adam = _fourier_adam_opt(
                np.random.randn(K) * 0.05, np.random.randn(K) * 0.05,
                n, K, conv_fft_len, corr_fft_len, sub_dead,
                cos_mat, sin_mat, max_iters=200_000)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Explore {idx}] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase II – intensive spectral flattening
    if best_seq is not None and time.time() < deadline - 10.0:
        print("[Spectral] Starting intensive spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_iters=800_000)
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # --------------------------------------------------------------
    # Phase III – projected gradient descent on exact max‑conv
    if best_seq is not None and time.time() < deadline - 12.0:
        print("[GD] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=400_000)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase IV – simulated annealing refinement
    if best_seq is not None and time.time() < deadline - 10.0:
        print("[Anneal] Starting simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=250_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase V – short perturb‑and‑refine cycles
    if best_seq is not None and time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        cycles = max(1, int(remaining // 5))
        print(f"[Perturb] Running {cycles} short refinement cycles")
        for c in range(cycles):
            noise = np.random.randn(n) * 0.001
            a_noise = best_seq + noise
            a_noise = np.maximum(a_noise, 0.0)
            a_noise = _project_simplex(a_noise)

            sub_dead = min(deadline, time.time() + 4.5)
            a_ref, val_ref = _max_conv_descent(
                a_noise, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=30_000)
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Perturb] cycle {c} improved to {val_ref:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
