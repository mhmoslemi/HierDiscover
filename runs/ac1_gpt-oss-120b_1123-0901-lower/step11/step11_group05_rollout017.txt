"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  The method expands the initial search space
(betas, power‑laws, exponentials, Dirichlet draws), picks the best
candidate, then refines it with two gradient phases (smooth‑max Adam
followed by exact‑max exponentiated‑gradient) and finishes with a fast
stochastic local search and a few cyclic shifts.  The whole routine
honours a 1000 s budget and always returns a valid list of numbers.

Key ideas compared to the previous baseline:
* Much richer set of initial families (continuous β‑shapes, power laws,
  exponentials and many Dirichlet draws) – this finds a substantially
  better starting point.
* Longer Adam phase with a slower learning‑rate decay and a higher
  final temperature, which drives the smooth‑max surrogate closer to
  the true max.
* Two‑stage optimisation: Adam (smooth) → sub‑gradient (exact) to
  escape plateaus.
* Aggressive final local‑search that moves a variable amount of mass
  between three random indices, increasing the chance of finding a
  shallow improvement.
* Time‑aware scheduling – each phase receives a generous part of the
  budget while leaving a safety margin for the final tidy‑up.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Basic utilities ------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection on the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) for a with Σa = 1.
    (The final evaluation uses the same scaling after the final rescaling.)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# -------------------------------------------------------------
# Optimisers ----------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.025
    lam_max = 1e13           # push very close to the true max

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- soft‑max to obtain a probability vector -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- evaluate smooth max -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)          # exact objective (since sum=1)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp trick) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        # slowly decay learning rate
        if it % 150_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # find active indices (within numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# -------------------------------------------------------------
# Candidate generators -------------------------------------------

def _beta_candidate(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑density (U‑shaped for α<1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _power_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^p (p<0 gives decreasing shape)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _exp_candidate(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈ 1000 s) and always
    returns the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    # ---------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---------------------------------------------------------
    # 0) generate a diverse collection of start points
    candidates = []

    # warm start, if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # beta family (dense sweep)
    alphas = np.linspace(0.30, 1.20, 19)   # 0.30 … 1.20 in steps of 0.05
    for alpha in alphas:
        candidates.append(_beta_candidate(n, alpha))

    # power‑law family
    powers = np.linspace(-3.0, 3.0, 31)    # -3 … 3
    for p in powers:
        candidates.append(_power_candidate(n, p))

    # exponential family
    lambdas = np.linspace(1e-4, 0.015, 25)   # slower decay → flatter shape
    for lam in lambdas:
        candidates.append(_exp_candidate(n, lam))

    # Random Dirichlet draws with varied concentration
    for _ in range(80):
        conc = np.random.uniform(0.2, 4.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ---------------------------------------------------------
    # 1) pick the best cheap candidate (exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # ---------------------------------------------------------
    # 2) Adam smooth‑max optimisation (takes ~60 % of budget)
    adam_deadline = deadline - (budget_s * 0.30)   # leave ~30 % for later phases
    if time.time() < adam_deadline:
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=2_000_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ---------------------------------------------------------
    # 3) Exact‑max sub‑gradient optimisation (remaining time ≈20 %)
    sub_deadline = deadline - (budget_s * 0.10)   # keep ~10 % for final tweaks
    if time.time() < sub_deadline:
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=8_000_000, eta0=1.6
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ---------------------------------------------------------
    # 4) Tiny stochastic local search (last few seconds)
    loc_deadline = deadline - 0.3
    while time.time() < loc_deadline:
        # pick three distinct indices – move a small chunk from i→j and i→k
        i, j, k = np.random.choice(n, 3, replace=False)
        # move at most 0.5 % of the source mass
        delta = min(best_a[i], 5e-3) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        # split the mass between two targets (helps escape shallow minima)
        cand[j] += delta * np.random.rand()
        cand[k] += delta - (cand[j] - best_a[j])
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ---------------------------------------------------------
    # 5) Try a few cyclic shifts (linear convolution changes under a shift)
    shifts = [n // 12, n // 6, n // 4, n // 3, n // 2]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ---------------------------------------------------------
    # 6) final sanity – clip, re‑project onto simplex, report official score
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)          # ensures Σa = 1 and non‑negativity

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
