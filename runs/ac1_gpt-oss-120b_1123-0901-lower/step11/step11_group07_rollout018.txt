"""
Propose a step‑function (non‑negative sequence) with a low 
evaluate_sequence() score.

The algorithm works entirely on the probability simplex (∑a_i = 1),
which does not affect the scale‑invariant objective.
It uses three complementary phases:

1. Exponentiated‑gradient (sub‑gradient) descent on the exact 
   max‑convolution (the same routine used in the reference code).  
   This quickly drives the peak down and yields a good feasible point.

2. A linear‑programming “rebalancing’’ step derived from the classic
   Frank‑Wolfe / LP improvement described in the problem statement.
   For the current distribution f we solve  

        maximise   Σ b_j
        subject to  (f * b)[k] ≤ M   for all k,
                    b_j ≥ 0,

   where M = max(f * f).  The solution b puts as much mass as possible
   on indices where f is small without increasing the peak.  We then
   mix f and the normalised b (f←(1‑t)f + t·b) to obtain a strictly
   lower peak.  A few such LP steps give a noticeable boost.

3. A final exponentiated‑gradient polishing phase.

All phases respect the 1000 s budget, always return a non‑negative
sequence, and use only NumPy / SciPy (no file or network I/O).
"""

import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT for speed)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)­‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exponentiated‑gradient (exact max) optimiser

def _expgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      max_iters: int = 2_000_000,
                      eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient optimiser (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()          # sum(a)=1 ⇒ denominator = 1

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set = positions attaining the max (sub‑gradient)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) (see analysis)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        eta = eta0 / math.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent with entropy)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise to the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming improvement step (Frank‑Wolfe style)

def _lp_improve_step(f: np.ndarray,
                     M: float,
                     mix: float,
                     n: int) -> np.ndarray:
    """
    Solve  maximise Σ b_j
          subject to   (f * b)[k] ≤ M   for all k,
                      b_j ≥ 0
    and return the mixed vector  (1‑mix)·f + mix·(b/Σb).
    """
    # Build the dense convolution matrix A where A @ b = f * b
    # A is Toeplitz with first column = [f, 0,…,0] and first row = [f0,0,…,0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = scipy.linalg.toeplitz(c, r)          # shape (2*n‑1, n)

    # Linear program: maximise Σb  ⇔ minimise -Σb
    c_obj = -np.ones(n, dtype=np.float64)    # minimise -1ᵀb
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Bounds: b_j ≥ 0
    bounds = [(0.0, None) for _ in range(n)]

    res = scipy.optimize.linprog(c_obj,
                                 A_ub=A,
                                 b_ub=b_ub,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True, 'time_limit': 20.0})

    if not res.success:
        # If the LP fails for any reason we simply return the input vector.
        print("  LP step failed:", res.message)
        return f

    b = res.x
    sum_b = b.sum()
    if sum_b < 1e-12:
        return f

    g = b / sum_b                     # normalise to unit sum
    new_f = (1.0 - mix) * f + mix * g
    new_f = np.maximum(new_f, 0.0)
    total = new_f.sum()
    if total < 1e-12:
        new_f = np.full_like(new_f, 1.0 / n)
    else:
        new_f = new_f / total
    return new_f

                               sub_dead, max_iters=1_500_000, eta0=0.9)
    if val < best_val:
        best_val, best_seq = val, a.copy()
        print(f"[Phase I] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – linear‑programming mixing steps
    # Allocate a modest slice of the remaining budget for 3‑5 LP calls.
    lp_mix = 0.25                # convex‑combination weight
    max_lp_iters = 5
    lp_iter = 0
    while (time.time() < deadline - 30.0) and (lp_iter < max_lp_iters):
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # Perform one LP re‑balancing step
        a = _lp_improve_step(a, M, mix=lp_mix, n=n)

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[LP {lp_iter+1}] new best {val:.6f}")

        # gradually reduce mixing to stay near the current point
        lp_mix = max(0.05, lp_mix * 0.8)
        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase III – final exponentiated‑gradient polishing (full remaining time)
    final_dead = deadline
    a, val = _expgrad_optimize(a, n, conv_fft_len, corr_fft_len,
                               final_dead, max_iters=2_000_000, eta0=0.7)
    if val < best_val:
        best_val, best_seq = val, a.copy()
        print(f"[Phase III] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Ensure a legal return (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Guard against pathological near‑zero sums (should never happen)
    if best_seq.sum() < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
