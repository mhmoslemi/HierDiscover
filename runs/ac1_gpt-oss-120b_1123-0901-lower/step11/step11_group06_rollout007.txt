def _spectral_step(a, conv_fft_len):
    energy = np.dot(a,a)
    A = np.fft.fft(a)
    n = a.shape[0]
    if n>1:
        rem_energy = max(0.0, n*energy - (np.abs(A[0])**2))
        const_mag = np.sqrt(rem_energy/(n-1))
    else:
        const_mag = np.sqrt(energy)
    A_new = np.empty_like(A)
    A_new[0] = A[0]
    if n>1:
        A_new[1:] = const_mag * np.exp(1j*np.angle(A[1:]))
    a_new = np.real(np.fft.ifft(A_new))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12 or np.isnan(s):
        a_new = np.random.dirichlet(np.ones(n))
        s = a_new.sum()
    a_new = a_new / s
    return a_new

"""
Hybrid random‑restart + long sub‑gradient optimizer.

Steps
1. Build a diverse pool of starter vectors (Dirichlet draws, spikes,
   exponential tails, half‑wave sinusoids, and a few “spectrally‑flattened’’
   random draws).  All vectors are normalised to unit sum.
2. Quickly evaluate each starter (exact max‑convolution) and keep the
   best few (default 10).
3. For each of those keep‑candidates run a long exponentiated‑gradient
   sub‑gradient descent (the exact max‑convolution is used, the active
   set receives uniform weight).  The run time per candidate is capped,
   the remaining budget is spent on the current overall best.
4. When time permits, a final “spectral‑flattening’’ phase (constant
   magnitude of the FFT, projection onto the simplex) is applied,
   followed by a short projected‑gradient descent on the exact max‑conv.
5. The best sequence found before the deadline is returned.  All entries
   are clipped to the evaluator’s allowed range (≥0, ≤1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for sub‑gradient / gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.2                # initial temperature for smooth‑max
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to get a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the current max)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the FFT magnitude of ``a`` to be (almost) constant
    while staying on the probability simplex.  After each inverse FFT the
    vector is projected back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")

        # ---- spectral flatten step ----
        A = np.fft.fft(a)                       # full complex FFT (size n)
        energy = np.dot(a, a)                   # ||a||_2^2

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                         # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _spectral_step(a: np.ndarray) -> np.ndarray:
    """One iteration of the constant‑magnitude‐FFT projection."""
    n = a.shape[0]
    energy = np.dot(a, a)
    A = np.fft.fft(a)
    if n > 1:
        rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
        const_mag = np.sqrt(rem_energy / (n - 1))
    else:
        const_mag = np.sqrt(energy)
    A_new = np.empty_like(A)
    A_new[0] = A[0]
    if n > 1:
        A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))
    a_new = np.real(np.fft.ifft(A_new))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12 or np.isnan(s):
        a_new = np.random.dirichlet(np.ones(n))
        s = a_new.sum()
    return a_new / s


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # Warm start (if available) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws – a spread of concentration parameters
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails
    for gamma in (0.1, 0.2, 0.4):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law families (inverse)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.6, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.45):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        cand.append(a)

    # Add a handful of spectrally‑flattened random draws
    for _ in range(12):
        a = np.random.dirichlet(np.ones(n))
        a = _spectral_step(a)               # one flatten iteration
        cand.append(a)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    # Random sparse patterns with a few hundred spikes
    for k in (48, 96, 192):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # Length of the vector – 4096 gives a good trade‑off between quality
    # and per‑iteration cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # Quick evaluation of all candidates (exact max‑convolution)
    quick = []
    for a in pool:
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        quick.append((val, a))

    quick.sort(key=lambda x: x[0])
    top_k = min(12, len(quick))
    top_candidates = [a for _, a in quick[:top_k]]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – short sub‑gradient runs on the top starters
    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # give each candidate up to ~12 s (or less if deadline is near)
        sub_dead = min(deadline, time.time() + 12.0)
        a_opt, val_opt = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=5_000_000,
            eta0=1.2,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Explore {idx}] Subgrad improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase II – intensive sub‑gradient refinement on the currently best
    if best_seq is not None and time.time() < deadline - 8.0:
        print("[Refine] Long sub‑gradient refinement")
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=20_000_000,
            eta0=1.0,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase III – intensive spectral flattening (if time permits)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Spectral] Intensive flattening")
        a_opt, val_opt = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_iters=2_000_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Spectral] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – short projected gradient descent on exact max‑conv
    if best_seq is not None and time.time() < deadline - 3.0:
        print("[GD] Final projected gradient descent")
        a_opt, val_opt = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=500_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[GD] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase V – fallback: guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final score according to the provided evaluator (ensures consistency)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:            # safety net – should never happen
        final_score = float("inf")
    print(f"Final best value {final_score:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
