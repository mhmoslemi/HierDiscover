5.  Refine the basin with a limited L‑BFGS‑B run on a smooth‑max surrogate.
    The surrogate  φ(a)=max_λ a∗a is approximated by

8.  Clip, project back onto the simplex, report the official score and return
    the best list.

The routine respects the given time budget (≈ 1000 s) and always returns a
valid sequence.  Hyper‑parameters (learning‑rates, λ for the smooth‑max,
etc.) have been tuned to obtain a modest improvement over the reference
implementation (target ≈ 1.503).  The code needs only NumPy and (optionally)

import time

# FFT‑based convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)−1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)−1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact value of 2·n·max(a∗a) assuming Σa=1.
    (evaluate_sequence returns the same value after its own normalisation.)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

    each step.  Returns the best vector seen (and its value).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, up to floating error)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / math.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# L‑BFGS‑B refinement on a smooth‑max surrogate
try:
    from scipy.optimize import minimize
    SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    SCIPY_AVAILABLE = False


class _SmoothMaxObj:
    """
    Callable that returns (value, gradient) of the smooth‑max surrogate.
    Parameters are the logarithmic (soft‑max) coordinates:  a_i = exp(y_i)/Σexp(y).
    The surrogate is
        S_λ(a) = max(a∗a) + log(∑exp(λ·(a∗a−max))) / λ .
    Gradient w.r.t. y follows the chain rule described in the analysis.
    """
    def __init__(self, n: int, conv_fft_len: int, corr_fft_len: int, lam: float):
        self.n = n
        self.conv_fft_len = conv_fft_len
        self.corr_fft_len = corr_fft_len
        self.lam = lam
        self.eps = 1e-15

    def __call__(self, y: np.ndarray) -> Tuple[float, np.ndarray]:
        # soft‑max parametrisation
        y_shift = y - np.max(y)                # numerical stability
        a = np.exp(y_shift)
        a = a / a.sum()

        conv = _full_convolution(a, self.conv_fft_len)
        max_c = conv.max()

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(self.lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0

        # smooth‑max value: max_c + log(w_sum)/λ
        smooth = max_c + np.log(w_sum) / self.lam
        val = 2.0 * self.n * smooth

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, self.corr_fft_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)           # ∂L/∂y

        return val, grad_y


def _lbfgs_callback(xk: np.ndarray, deadline: float) -> bool:   # pragma: no cover
    """Terminate L‑BFGS when the global deadline is exceeded."""
    return time.time() > deadline


def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 250,
    lam: float = 1_000.0,
) -> Tuple[np.ndarray, float]:
    """
    Run a short L‑BFGS‑B optimisation on the smooth‑max surrogate.
    Returns the best vector seen (or the input if SciPy is unavailable or the
    optimisation aborts early).
    """
    if not SCIPY_AVAILABLE:
        return a0, _objective(a0, n, conv_fft_len)

    # initialise in log‑probability space
    y0 = np.log(a0 + 1e-15)

    obj = _SmoothMaxObj(n, conv_fft_len, corr_fft_len, lam)

    try:
        res = minimize(
            fun=obj,
            x0=y0,
            method="L-BFGS-B",
            jac=True,
            callback=_lbfgs_callback,
            options={"maxiter": max_iter, "disp": False},
            args=(),
        )
    except Exception:                     # pragma: no cover
        return a0, _objective(a0, n, conv_fft_len)

    a_opt = np.exp(res.x - np.max(res.x))
    a_opt = a_opt / a_opt.sum()
    val_opt = _objective(a_opt, n, conv_fft_len)

    if val_opt < _objective(a0, n, conv_fft_len):
        return a_opt, val_opt
    else:
        return a0, _objective(a0, n, conv_fft_len)

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the supplied time budget (≈ 1000 s) and always
    returns a valid list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3    # small safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---------------------------------------------------------
    # 0) generate a pool of diverse initial vectors
    candidates = []

    # warm start, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform distribution
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric beta‑shaped densities (U‑shaped for α<1)
    alphas = [0.55, 0.60, 0.65, 0.70, 0.75, 0.80, 0.85, 0.90, 0.95]
    x = (np.arange(n) + 0.5) / n
    for alpha in alphas:
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # random Dirichlet draws with varied concentration
    for _ in range(30):
        conc = np.random.uniform(0.3, 3.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # anti‑symmetric seeds – try to decorrelate mirrored entries
    for _ in range(10):
        base = np.random.dirichlet(np.full(n, 1.0))
        mask = np.random.rand(n // 2) > 0.5
        a = base.copy()
        a[: n // 2][mask] *= 1.5
        a[n - 1 - np.arange(n // 2)][mask] *= 0.5
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        candidates.append(a)

    # ---------------------------------------------------------
    # 1) pick the best cheap candidate (exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # ---------------------------------------------------------
    # 2) long sub‑gradient optimisation (exponentiated‑gradient)
    sub_deadline = deadline - 15.0                     # leave room for polishing
    if sub_deadline > time.time():
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=10_000_000, eta0=1.5
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Subgrad] improved to {val_opt:.6f}")

    # ---------------------------------------------------------
    # 3) smooth‑max polishing with L‑BFGS‑B (if SciPy is available)
    lbf_deadline = deadline - 5.0
    if lbf_deadline > time.time():
        a_lbf, val_lbf = _lbfgs_optimize(
            best_a, n, conv_fft_len, corr_fft_len, lbf_deadline,
            max_iter=300, lam=1_000.0
        )
        if val_lbf < best_val:
            best_val, best_a = val_lbf, a_lbf.copy()
            print(f"[L‑BFGS] improved to {val_lbf:.6f}")

    # ---------------------------------------------------------
    # 4) tiny stochastic local search for the last few seconds
    local_deadline = deadline - 0.5
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        delta = min(best_a[i], 5e-4) * np.random.rand()   # move at most 5e‑4
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ---------------------------------------------------------
    # 5) try a few cyclic shifts (convolution is shift‑variant)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ---------------------------------------------------------
    # 6) final sanitisation and return
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
