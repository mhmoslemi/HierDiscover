def _lp_optimize(f, n, time_allow):
    # solve min t s.t. A @ b <= t, sum(b)=1, b>=0
    # Use SciPy linprog
    # Build matrix A (2n-1, n)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # vectorized fill:
    # for each i column:
    #   rows i:i+n => f
    for i in range(n):
        A[i:i+n, i] = f
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0  # minimize t
    # Append column -1 for variable t to A_ub: A_ub = [A, -1s]
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=np.float64)])
    b_ub = np.zeros(rows, dtype=np.float64)
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0, None)] * n + [(0, None)]
    # call linprog
    options = {'disp': False, 'time_limit': time_allow if time_allow < 300 else 300}
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options=options)
    except Exception as e:
        print(f"[LP] exception {e}")
        return None, None
    if not res.success:
        print(f"[LP] not successful: {res.message}")
        return None, None
    x = res.x
    b = x[:n]
    t = x[-1]
    # ensure sum close to 1
    s = b.sum()
    if s > 0:
        b = b / s
    else:
        return None, None
    return b, t

1. Build a small pool of diverse initial vectors (uniform, arcsine,
   Dirichlet draws, simple shapes).  Each vector is normalised to sum = 1.

2. Run a short Adam optimiser on a smooth‑max surrogate of the
   convolution peak for every candidate.  Keep the best vector found.

3. Refine that best vector with projected gradient descent on the *exact*
   max‑convolution, followed by the fine‑grained “mass‑transfer” local
   search that repeatedly moves a tiny amount of mass from the coordinate
   with the largest gradient to the one with the smallest gradient.

4. **Linear‑programming refinement on a reduced dimension.**
   The current vector `f` (size `n_small`) is used to solve

        min   t
        s.t.  (f * b)[k] ≤ t   for all k
              Σ b_i = 1 ,  b_i ≥ 0

   This LP (solved with `scipy.optimize.linprog`) gives a new distribution
   `b` whose peak of the *cross*‑convolution with `f` is as low as possible.
   The solution is up‑sampled to the original size, normalised and fed back
   into the fine‑grained mass‑transfer optimiser.

5. One final short mass‑transfer run on the full‑size vector.

All steps respect the overall 1000 s budget.  The returned list is
clipped to the evaluator’s allowed range (≥0, ≤1000) and the exact score
is printed before returning.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Factories for initial shapes

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric (or asymmetric) Beta pdf on a uniform grid of mid‑points."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers (same building blocks as the previous state‑of‑the‑art)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
):
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
):
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        improved = False
        src_mass = a[i_high]
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement on a reduced dimension

def _lp_optimize(f: np.ndarray, n: int, time_allow: float):
    """
    Solve   min t   s.t.   (f * b)[k] ≤ t  (∀k),
            Σ b_i = 1 ,  b_i ≥ 0.
    Returns the optimal b (length n) and the optimal peak value t.
    """
    # Build dense convolution matrix A where A @ b = f * b
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f  # each column is a shifted copy of f

    # Linear programme data
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0                     # minimise t
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=np.float64)])
    b_ub = np.zeros(rows, dtype=np.float64)           # A b - t ≤ 0  ⇔  A b ≤ t
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # linprog options: give a modest time limit for safety
    options = {"presolve": True, "time_limit": max(0.0, time_allow)}
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options=options)
    except Exception as e:
        print(f"[LP] exception: {e}")
        return None, None

    if not res.success:
        print(f"[LP] failed: {res.message}")
        return None, None

    x = res.x
    b = x[:n]
    t = x[-1]
    # Normalise – numerical noise can move the sum slightly away from 1
    s = b.sum()
    if s <= 0:
        return None, None
    b = b / s
    return b, t


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ 1000 s.  Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    n_big = 4096
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous construction exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_big:
            warm = warm[:n_big]
        elif warm.shape[0] < n_big:
            warm = np.pad(warm, (0, n_big - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a modest pool of diverse initial vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())
    pool.extend([
        _make_uniform(n_big),
        _make_beta(n_big, 0.5, 0.5),   # arcsine
        _make_beta(n_big, 0.8, 0.8),
        _make_beta(n_big, 2.0, 2.0),
        _make_triangle(n_big),
        _make_cosine(n_big),
    ])
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
        pool.append(_make_dirichlet(n_big, alpha))

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – quick Adam runs on a few starters
    max_candidates = min(12, len(pool))
    for idx in range(max_candidates):
        if time.time() > deadline:
            break
        a0 = np.maximum(pool[idx], 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n_big))
        a0 = a0 / a0.sum()

        sub_dead = min(deadline, time.time() + 12.0)    # ~12 s per start
        a_adam, val_adam = _adam_optimize(
            a0, n_big, conv_fft_len_big, corr_fft_len_big,
            sub_dead, max_iters=250_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – projected gradient descent on the exact max‑convolution
    if best_seq is not None and time.time() < deadline - 20:
        sub_dead = min(deadline, time.time() + 30.0)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n_big,
            conv_fft_len_big, corr_fft_len_big,
            sub_dead, max_iters=300_000)
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase III – mass‑transfer fine‑grained local search
    if best_seq is not None and time.time() < deadline - 15:
        sub_dead = min(deadline, time.time() + 25.0)
        a_mt, val_mt = _mass_transfer_optimize(
            best_seq, n_big,
            conv_fft_len_big, corr_fft_len_big,
            sub_dead, max_iters=200_000)
        if val_mt < best_val:
            best_val, best_seq = val_mt, a_mt.copy()
            print(f"[MassTransfer] improved best to {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – Linear‑programming refinement on a reduced grid
    if best_seq is not None and time.time() < deadline - 30:
        # Reduce to n_small = 1024 for a cheap LP
        n_small = 1024
        factor = n_big // n_small
        f_small = best_seq.reshape(n_small, factor).mean(axis=1)
        f_small = np.maximum(f_small, 0.0)
        f_small /= f_small.sum()

        lp_iter = 0
        while time.time() < deadline - 15 and lp_iter < 6:
            remaining = deadline - time.time()
            b_small, t_val = _lp_optimize(f_small, n_small, remaining)
            if b_small is None:
                break
            # Upsample to full size for a fair comparison
            x_small = np.linspace(0, 1, n_small, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_up = np.interp(x_big, x_small, b_small)
            a_up = np.clip(a_up, 0.0, 1000.0)
            a_up /= a_up.sum()

            conv = _full_convolution(a_up, conv_fft_len_big)
            val_up = 2.0 * n_big * conv.max()
            print(f"[LP iter {lp_iter}] peak={t_val:.6e}  up‑sampled val={val_up:.6f}")

            if val_up + 1e-12 < best_val:
                best_val, best_seq = val_up, a_up.copy()
                print(f"[LP] new best after up‑sampling {val_up:.6f}")

            # Prepare next LP round
            f_small = b_small
            lp_iter += 1

    # ------------------------------------------------------------------
    # Phase V – final short mass‑transfer polishing (if time permits)
    if best_seq is not None and time.time() < deadline - 5:
        sub_dead = min(deadline, time.time() + 15.0)
        a_final, val_final = _mass_transfer_optimize(
            best_seq, n_big,
            conv_fft_len_big, corr_fft_len_big,
            sub_dead, max_iters=150_000)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f"[Final MT] improved best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n_big)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (for logging)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Final evaluation failed: {exc}")

    return best_seq.tolist()
