"""
Hybrid optimisation with variance regularisation.

The objective 2·n·max(conv) (sum = 1) is minimised by a smooth‑max Adam
step followed by a long exact‑max exponentiated‑gradient refinement.
A variance term (∑(conv−avg)²) is added to the smooth‑max loss;
its gradient pushes the whole convolution towards a flat shape,
making the max drop faster.
The algorithm works on the probability simplex via a soft‑max
parameterisation (logits).  The whole budget (≈1000 s) is used for one
deep run starting from the best available warm start (or a half‑sine
fallback).  Progress is printed and the best sequence found is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple parametric factories (all return a normalised probability vector)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_half_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_laplacian(n, gamma):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-gamma * np.abs(i - center))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main optimiser

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    The routine runs a single long Adam optimisation (smooth‑max + variance
    regularisation) followed by an exact‑max exponentiated‑gradient polish.
    """
    np.random.seed(seed)
    # small safety margin for the host environment
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4096 offers a good speed/quality trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a warm start if available; otherwise fall back to
    # a half‑sine which is known to be a decent starter.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    if warm is None:
        warm = _make_half_sine(n)

    # ------------------------------------------------------------------
    # Adam optimiser on a combined loss:
    #   L = smooth_max(conv) + λ·variance(conv)
    eps = 1e-15
    logits = np.log(warm + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5                # temperature for smooth‑max
    lam_factor = 1.07

    # variance regularisation weight (starts strong and decays to 0)
    var_weight_start = 0.5

    best_a = warm.copy()
    # analytic initial value (since sum(best_a)=1)
    best_val = 2.0 * n * (_full_convolution(best_a, conv_fft_len).max())

    it = 0
    start_time = time.time()
    while time.time() < deadline:
        # ----------- soft‑max -------------------------------------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()                # a ≥ 0, Σa = 1

        # ----------- convolution --------------------------------------------
        conv = _full_convolution(a, conv_fft_len)  # length 2·n‑1
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Iter {it:07d}] new best {cur_val:.6f}")

        # ----------- smooth‑max weights --------------------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_max = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient of variance (∑(conv−avg)²) part.
        # avg = 1/(2n‑1) because Σconv = 1, so we can just use conv itself.
        grad_var = 4.0 * _cross_correlation(conv, a, corr_fft_len)

        # linearly decay variance weight towards the end of the budget
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / (deadline - start_time))
        var_weight = var_weight_start * remaining_frac

        # combined gradient
        grad = grad_max + var_weight * grad_var

        # ----- chain rule through the soft‑max (log‑it) ----------------------
        avg_grad = np.dot(a, grad)
        grad_log = a * (grad - avg_grad)

        # ----- Adam update --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        it += 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----------------------------------------
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 200_000 == 0:
            lr *= 0.9

    # ------------------------------------------------------------------
    # Final polish: exact‑max exponentiated‑gradient (sub‑gradient) run
    # Uses only the active set where conv attains its maximum.
    if time.time() < deadline - 5.0:
        a = best_a.copy()
        it_sub = 0
        while time.time() < deadline:
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            cur_val = 2.0 * n * max_c
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[ExpGrad {it_sub:07d}] new best {cur_val:.6f}")

            # active constraints (indices where the max is attained)
            active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
            if not np.any(active):
                break
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # step size decays like 1/√t (standard for sub‑gradient)
            eta = 0.6 / np.sqrt(it_sub + 1.0)

            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s == 0 or np.isnan(s):
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s

            it_sub += 1

    # ------------------------------------------------------------------
    # Return the best result found (clipped to the safe range)
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
