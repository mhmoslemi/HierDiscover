def propose_candidate(seed=0, budget_s=1000):

        s = a.sum()
        if s < 0.01:
            # tiny sum, set to random uniform to avoid inf
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        maxc = conv.max()
        return 2.0 * n * maxc, a, maxc

    best_seq = None

    # store candidate as np array

    warm = globals().get('height_sequence_1')

"""
Hybrid multi‑stage optimizer for the step‑function attack.

The algorithm builds a diverse pool of seed vectors, evaluates them
quickly, then spends the remaining budget repeatedly refining the
current best sequence with a cascade of optimisers:
    * Adam on a smooth‑max surrogate of the peak of the convolution,
    * exact max‑convolution sub‑gradient (exponentiated‑gradient),
    * spectral flattening (constant FFT magnitude),
    * a small linear‑programming pass that reduces the peak,
    * projected gradient descent on the true max‑convolution,
    * simulated‑annealing polish,
    * finally a short random‑perturbation search.

All updates keep the vector on the probability simplex, so the
objective 2·n·max(conv) (the original evaluation after normalisation)
is invariant under scaling.  The routine stops
shortly before the deadline and returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 10,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence f we minimise M subject to (f★b)_k ≤ M,
    and then mix the solution b back into f.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # Pick a modest active set of constraints (the peak positions + a few spread ones)
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # minimise M  <=>  minimise the extra variable M
        c = np.concatenate([np.zeros(n), [1.0]])               # last variable is M
        bounds = [(0, None)] * n + [(0, None)]
        b_ub = M * np.ones(len(active))

        # Append M to constraints matrix: A * b + (-1) * M ≤ 0
        # i.e. A b - M ≤ 0  →  [A | -1]
        A_ext = scipy.sparse.hstack([A, -np.ones((len(active), 1))])

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A_ext,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x[:n]
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # Mix the new direction with the current iterate
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Entry point – returns a non‑negative step function (list of floats)."""

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem dimensions
    n = 4096                                 # length of the vector
    conv_fft_len = _next_pow2(2 * n - 1)      # FFT size for convolution
    corr_fft_len = _next_pow2(3 * n - 2)      # FFT size for cross‑correlation

    # --------------------------------------------------------------
    # Helper to evaluate a candidate (value = 2·n·max(conv) after normalisation)
    def _value_of(a_raw: np.ndarray):
        a = np.maximum(a_raw, 0.0)
        s = a.sum()
        if s < 0.01:                         # avoid the “sum too small” guard
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max(), a

    # --------------------------------------------------------------
    # Warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        val, warm = _value_of(warm)
        best_seq, best_val = warm, val
        print(f"[Warm] value {val:.6f}")
    else:
        best_seq = None
        best_val = float("inf")

    # --------------------------------------------------------------
    # Build an initial diverse pool (cheap to evaluate)
    pool = []

    # deterministic families
    pool.append(np.full(n, 1.0 / n))                     # uniform
    x = np.arange(n, dtype=np.float64)
    pool.append(np.sin(np.pi * (x + 0.5) / n).clip(0) / np.sum(np.sin(np.pi * (x + 0.5) / n)))  # sine bump
    # exponential fall‑off
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0):
        a = np.exp(-gamma * x)
        pool.append(a / a.sum())
    # symmetric triangular
    tri = np.concatenate((np.arange(1, n // 2 + 1), np.arange(n // 2, 0, -1)))
    pool.append(tri / tri.sum())
    # power‑law tails
    for alpha in (-2.0, -1.5, -1.0, -0.5):
        a = (x + 1) ** alpha
        pool.append(a / a.sum())

    # random Dirichlet draws with various concentration
    for alpha in (0.1, 0.5, 1.0, 2.0, 5.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        pool.append(a)

    # a few sparse spike candidates
    for k in (16, 32, 64, 128, 256):
        a = np.zeros(n)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(a)

    # Evaluate the pool and keep the best seed
    for cand in pool:
        if time.time() > deadline:
            break
        val, a_norm = _value_of(cand)
        if val < best_val:
            best_val, best_seq = val, a_norm
            print(f"[Init] new best {val:.6f}")

    # --------------------------------------------------------------
    # Helper wrappers that respect the remaining time
    def _run_adam(a, t):
        sub_dead = min(deadline, time.time() + t)
        return _adam_optimize(a, n, conv_fft_len, corr_fft_len, sub_dead,
                              max_iters=800_000)

    def _run_subgrad(a, t):
        sub_dead = min(deadline, time.time() + t)
        return _subgradient_optimize_expgrad(a, n, conv_fft_len, corr_fft_len,
                                            sub_dead, max_iters=800_000,
                                            eta0=0.8)

    def _run_spectral(a, t):
        sub_dead = min(deadline, time.time() + t)
        return _spectral_flatten_optimize(a, n, conv_fft_len, sub_dead,
                                          max_iters=500_000)

    def _run_lp(a, t):
        sub_dead = min(deadline, time.time() + t)
        return _lp_improve(a, n, conv_fft_len, sub_dead,
                           max_iter=12, tau=0.5)

    def _run_gd(a, t):
        sub_dead = min(deadline, time.time() + t)
        return _max_conv_descent(a, n, conv_fft_len, corr_fft_len,
                                 sub_dead, max_iters=250_000)

    def _run_anneal(a, t):
        sub_dead = min(deadline, time.time() + t)
        return _anneal_optimize(a, n, conv_fft_len, sub_dead,
                                max_moves=120_000)

    # --------------------------------------------------------------
    # Main refinement loop – repeatedly apply the suite of optimisers
    while time.time() < deadline - 0.5:
        remaining = deadline - time.time()
        if remaining < 15.0:
            break
        # split the remaining time into equal slices for each stage
        slice_t = remaining * 0.18

        a1, v1 = _run_adam(best_seq, slice_t)
        if v1 < best_val:
            best_val, best_seq = v1, a1
            print(f"[Adam refine] improved to {v1:.6f}")

        a2, v2 = _run_subgrad(best_seq, slice_t)
        if v2 < best_val:
            best_val, best_seq = v2, a2
            print(f"[Subgrad refine] improved to {v2:.6f}")

        a3, v3 = _run_spectral(best_seq, slice_t)
        if v3 < best_val:
            best_val, best_seq = v3, a3
            print(f"[Spectral refine] improved to {v3:.6f}")

        a4, v4 = _run_lp(best_seq, slice_t)
        if v4 < best_val:
            best_val, best_seq = v4, a4
            print(f"[LP refine] improved to {v4:.6f}")

        a5, v5 = _run_gd(best_seq, slice_t)
        if v5 < best_val:
            best_val, best_seq = v5, a5
            print(f"[GD refine] improved to {v5:.6f}")

        a6, v6 = _run_anneal(best_seq, slice_t)
        if v6 < best_val:
            best_val, best_seq = v6, a6
            print(f"[Anneal refine] improved to {v6:.6f}")

    # --------------------------------------------------------------
    # Final tiny random‑perturbation search (quick “noise” hill‑climb)
    while time.time() < deadline - 0.2:
        eps = 5e-4
        pert = best_seq + eps * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        s = pert.sum()
        if s < 1e-12:
            continue
        pert = pert / s
        val, _ = _value_of(pert)
        if val < best_val:
            best_val, best_seq = val, pert
            print(f"[Noise] improved to {val:.6f}")

    # --------------------------------------------------------------
    # Guarantee a non‑empty, bounded return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
