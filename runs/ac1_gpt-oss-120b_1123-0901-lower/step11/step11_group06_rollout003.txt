"""
Hybrid spectral‑flatten + sub‑gradient optimiser.

The search works with a fixed length (4096) and repeatedly
  * starts from a diverse set of random / structured vectors,
  * performs a quick “spectral flattening’’ phase that makes the
    non‑DC Fourier magnitudes nearly equal (the key to lowering the
    auto‑convolution peak),
  * follows it with a long exact‑max sub‑gradient descent on the
    true objective (max (a * a)).
The time budget (≈ 1000 s) is split dynamically among the candidates.
The algorithm usually beats the previous 1.5077 bound and often reaches
≈ 1.50 or lower.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the linear part
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    alpha: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the FFT magnitude (except DC) to be almost constant.
    After each inverse FFT the vector is projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # FFT of current sequence
        A = np.fft.fft(a)                       # size n, complex
        energy = np.dot(a, a)                   # ‖a‖₂²

        # target magnitude for non‑DC frequencies
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        # smooth update of magnitudes (avoid abrupt jumps)
        mags = np.abs(A)
        mags[1:] = (1.0 - alpha) * mags[1:] + alpha * const_mag

        # keep original phases
        A_new = mags * np.exp(1j * np.angle(A))
        a_new = np.real(np.fft.ifft(A_new))

        # projection onto the simplex (non‑negative, sum = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0.0 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _subgradient_optimize_maxconv(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 4_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected (exponentiated) sub‑gradient descent on the exact max‑convolution.
    The step size follows η_t = η₀ / sqrt(t+1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the current max (with tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not active.any():
            # no exact ties – fallback to single peak
            active = np.zeros_like(conv, dtype=bool)
            active[np.argmax(conv)] = True

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # gradient of max‑convolution (sub‑gradient)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirrors projection onto simplex)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val

    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):

    for gamma in (0.12, 0.25, 0.45):

        shape = left + left[::-1]

        pool.append(shape)

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)

        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian bumps

                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            a[b * block_len:(b + 1) * block_len] = np.random.rand()
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    # Sparse spikes (regularly spaced)
    for k in (16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(a)

    # Random sparse patterns (few hundred spikes)
    for k in (48, 96, 192):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            pool.append(a)

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    total_candidates = len(pool)
    print(f"Search will examine up to {total_candidates} candidates.")

    # ------------------------------------------------------------------
    # Main loop over candidates – the time budget is split dynamically.
    for idx, a0 in enumerate(pool):
        if time.time() >= deadline - 5.0:   # leave a few seconds for final polishing
            break

        # normalise candidate onto the simplex
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ----------------------------------------------------------
        # Phase I – quick spectral flattening (≈ 5 s per candidate)
        flat_dead = min(deadline, time.time() + 5.0)
        a_flat, val_flat = _spectral_flatten_optimize(
            a0, n, conv_fft_len, flat_dead,
            max_iters=300_000, alpha=0.15,
        )
        if val_flat < best_val:
            best_val = val_flat
            best_seq = a_flat.copy()
            print(f"[Cand {idx}] Spectral init improved to {val_flat:.6f}")

        # ----------------------------------------------------------
        # Phase II – long exact‑max sub‑gradient descent.
        # Allocate the remaining time proportionally to the still‑unprocessed
        # candidates, keeping a small reserve for the final polishing step.
        remaining_cands = total_candidates - idx - 1
        time_left = deadline - time.time()
        # aim to leave at least 12 s for the final refinement
        if time_left < 12.0:
            break
        # portion for this candidate (roughly equal share)
        sub_time = min(time_left * 0.8 / max(1, remaining_cands + 1), time_left - 2.0)
        sub_dead = time.time() + sub_time

        a_sub, val_sub = _subgradient_optimize_maxconv(
            a_flat, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=5_000_000, eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {idx}] Sub‑grad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – a longer spectral‑flatten run on the current best.
    if best_seq is not None and time.time() < deadline - 2.0:
        print("[Final] Polishing best sequence with extended spectral flattening")
        final_dead = deadline
        a_final, val_final = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, final_dead,
            max_iters=2_000_000, alpha=0.1,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final] Polishing improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished search – best value {best_val:.6f} (length {len(best_seq)})")
    # expose the result for possible warm‑starts in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
