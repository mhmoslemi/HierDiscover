"""
Propose a step‑function with a lower evaluation score.

The algorithm works on the probability simplex (∑a_i = 1) – the score is
scale invariant, so normalisation does not change the result.  It proceeds
in four stages:

1. **Diverse seeding** – generate a large pool of candidate distributions
   (warm start, Dirichlet draws, Beta families with very small shape
   parameters, exponential tails, sparse spikes, random mixtures).  The
   best few according to the exact max‑convolution are kept.

2. **Continuation L‑BFGS** – starting from each seed, minimise a smooth
   approximation of the max‑convolution (log‑sum‑exp) using a
   quasi‑Newton (L‑BFGS‑B) optimiser on the soft‑max parametrisation
   `a = softmax(logits)`.  The smoothing parameter λ is increased
   geometrically (λ = 20 → 200 → 2000) so the optimiser first sees a
   smooth landscape and then focuses on the true L∞ objective.

3. **Exact sub‑gradient polishing** – an exponentiated‑gradient descent
   on the *exact* max‑convolution (active–set subgradient) is run for a
   long horizon to tighten the bound further.

4. **Simulated‑annealing polish** – a cheap local‑search that swaps a
   small fraction of mass between two coordinates is performed as a final
   polish.

All operations are performed with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s budget and always returns a non‑negative
sequence.  Larger n gives more flexibility – we use n = 8192 (FFT size
2 ⁿ⁺¹) which still fits comfortably in the time budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = rfft(w, fft_len)
    fr = rfft(rev_a, fft_len)
    corr = irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (x≥0, Σx=1)."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories

def _make_dirichlet(n, alpha):
    return np.random.dirichlet(alpha * np.ones(n))


def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    a = left + left[::-1]
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Optimisation helpers

def _max_conv_val(a, conv_fft_len):
    """Exact evaluation 2·n·max(conv) for a simplex‑normalised a."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * a.shape[0] * conv.max()


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    a = a / a.sum()

    best_a = a.copy()
    best_val = _max_conv_val(a, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing local search."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1
    return best_a, best_val


def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lam: float = 200.0,
    max_iter: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Quasi‑Newton minimisation (L‑BFGS‑B) of the smoothed max convolution.

    The optimisation variable is `logits` with `a = softmax(logits)`.  The
    smooth objective is 2·n·(1/λ)·log∑exp(λ·conv).  The gradient is obtained
    analytically via a weighted cross‑correlation.
    """
    try:
        from scipy.optimize import minimize
    except Exception:                     # pragma: no cover
        # If SciPy is missing we simply return the unmodified sequence.
        return a0, _max_conv_val(a0, conv_fft_len)

    eps = 1e-12
    logits0 = np.log(a0 + eps)

    def obj_and_grad(logits):
        # soft‑max to obtain a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)

        # smooth max via log‑sum‑exp
        lse = np.log(np.sum(np.exp(lam * conv))) / lam
        loss = 2.0 * n * lse

        # gradient of loss w.r.t. a
        w = np.exp(lam * conv)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        return loss, grad_log

    res = minimize(
        obj_and_grad,
        logits0,
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12},
    )
    logits_opt = res.x
    mx = logits_opt.max()
    exp_log = np.exp(logits_opt - mx)
    a_opt = exp_log / exp_log.sum()
    val_opt = _max_conv_val(a_opt, conv_fft_len)
    return a_opt, val_opt


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) minimising the
    provided `evaluate_sequence`.  The routine respects the given time
    budget (≈ budget_s seconds) and always returns a sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives more flexibility; 8192 fits in the budget
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Gather a rich pool of initial candidates
    pool = []

    # warm start if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.size > n:
            w = w[:n]
        elif w.size < n:
            w = np.pad(w, (0, n - w.size), "constant")
        w = np.clip(w, 0.0, 1000.0)
        w = w / max(w.sum(), 1e-12)
        pool.append(w.copy())

    # Uniform baseline
    pool.append(_make_uniform(n))

    # Dirichlet draws – wide concentration range
    for alpha in (0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))

    # Beta families – include very small shape parameters
    for _ in range(30):
        a = 10 ** np.random.uniform(-3, 0)   # 0.001 … 1
        b = 10 ** np.random.uniform(-3, 0)
        pool.append(_make_beta(n, a, b))

    # Exponential tails – a few decay rates
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_make_exponential(n, gamma))

    # Sparse spikes – varying sparsity
    for k in (32, 64, 128, 256, 512):
        pool.append(_make_spikes(n, k))

    # Random mixtures of Dirichlet draws
    for _ in range(12):
        a1 = np.random.dirichlet(np.ones(n))
        a2 = np.random.dirichlet(np.ones(n))
        w = np.random.rand()
        mix = w * a1 + (1 - w) * a2
        mix /= mix.sum()
        pool.append(mix)

    # ------------------------------------------------------------------
    # Quick evaluation of the raw pool, keep the best few
    evaluated = []
    for cand in pool:
        if time.time() > deadline - 25.0:            # keep reserve for refinement
            break
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        evaluated.append((val, cand.copy()))
    evaluated.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in evaluated[:5]]   # keep 5 most promising starts

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # Helper to possibly update the global best
    def _maybe_update(a, val):
        nonlocal best_seq, best_val
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Update] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Optimise each promising seed
    for idx, a0 in enumerate(top_candidates):
        if time.time() > deadline - 10.0:
            break
        print(f"[Seed {idx}] starting optimisation")

        a = np.maximum(a0, 0.0)
        a /= max(a.sum(), 1e-12)

        # ---- Continuation L‑BFGS ----
        for lam in (20.0, 200.0, 2000.0):
            if time.time() > deadline - 8.0:
                break
            a, val = _lbfgs_optimize(
                a,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                lam=lam,
                max_iter=300,
            )
            _maybe_update(a, val)

        # ---- Exact sub‑gradient (exponentiated‑gradient) ----
        if time.time() < deadline - 5.0:
            a, val = _subgradient_optimize_expgrad(
                a,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=600_000,
                eta0=0.8,
            )
            _maybe_update(a, val)

        # ---- Simulated‑annealing polish ----
        if time.time() < deadline - 2.0:
            a, val = _anneal_optimize(
                a,
                n,
                conv_fft_len,
                deadline,
                max_moves=150_000,
            )
            _maybe_update(a, val)

    # ------------------------------------------------------------------
    # If nothing survived (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = _max_conv_val(best_seq, conv_fft_len)

    # Final sanity – enforce evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
```

This implementation builds a diverse initial pool (including very sharp Beta distributions), selects the most promising seeds, and refines them using a continuation L‑BFGS optimiser on a smooth approximation of the max‑convolution, followed by an exact sub‑gradient step and a brief simulated‑annealing polish.  
All arithmetic respects the non‑negative constraint and projects onto the probability simplex, guaranteeing that `evaluate_sequence` never returns `inf`. The routine respects the 1000 s budget and returns a valid sequence.
