"""
Hybrid optimizer for the non‑negative step‑function problem (improved).

Goal:
    minimise 2·n·max(conv(a,a)) / (sum(a)^2)   with a_i ≥ 0.
The formulation is scale‑invariant, therefore we keep ∑a_i = 1
(i.e. we optimise on the probability simplex).

Approach
--------
1.  Initialise a diverse set of candidates (uniform, Dirichlet,
    exponential tails, mixtures of Gaussians, sparse spikes, …) and
    keep the best one according to the *exact* objective.

2.  Continuation with L‑BFGS on a smooth‑max surrogate.
    The surrogate is  s_λ(a) = max_k a*a[k]  replaced by
    (1/λ)·log∑_k exp(λ·a*a[k]).
    λ is increased geometrically (0.1 → 1e5) and a few L‑BFGS
    iterations are performed at each λ.  Gradients are obtained
    via FFT‑based cross‑correlation.

3.  Long exponentiated‑gradient (mirror‑descent) on the *exact*
    max‑convolution.  A diminishing step size  η_t = η0/√t  is used.
    The current best vector is stored throughout.

4.  Final short simulated‑annealing (SA) pass that swaps a tiny
    fraction of mass between random coordinates; SA helps to escape
    tiny plateaus left by the sub‑gradient phase.

All phases constantly check the remaining time and stop before the
deadline, guaranteeing that a non‑empty sequence is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (result length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradient of conv."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → Δⁿ (probability simplex)."""
    m = logits.max()
    exp = np.exp(logits - m)
    return exp / exp.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _exact_objective(a: np.ndarray, n: int) -> float:
    """Exact objective value for a normalized (sum=1) vector."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _smooth_max_and_grad(logits: np.ndarray,
                         lam: float,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int):
    """
    Return (smooth‑max value, gradient w.r.t. logits) for the
    surrogate  (1/λ)·log∑ exp(λ·conv) .
    """
    a = _softmax(logits)

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # log‑sum‑exp in a numerically stable way
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    smooth_max = max_c + np.log(w_sum) / lam   # = (1/λ)·log∑exp(λ·conv)

    # gradient of smooth‑max w.r.t. a
    w_norm = w / w_sum
    grad_a = 2.0 * _cross_correlation(w_norm, a, corr_fft_len)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)

    # final objective (scale 2·n)
    val = 2.0 * n * smooth_max
    return val, grad_log


def _lbfgs_smooth_opt(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      lambdas=(0.1, 0.5, 1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 1e4)):
    """
    Continuation over λ with a few L‑BFGS iterations at each stage.
    Returns the best vector (on the simplex) and its exact objective.
    """
    eps = 1e-12
    logits = np.log(a0 + eps)        # initialise in log‑space

    best_a = a0.copy()
    best_val = _exact_objective(best_a, n)

    for lam in lambdas:
        if time.time() >= deadline - 2.0:
            break

        def fun(logits_vec):
            val, grad = _smooth_max_and_grad(logits_vec, lam, n,
                                             conv_fft_len, corr_fft_len)
            return val, grad

        # limit total L‑BFGS steps to a modest number (max 200 iters)
        opts = {"maxiter": 200, "disp": False}
        res = minimize(fun, logits, method="L-BFGS-B", jac=True,
                       options=opts)

        # Retrieve the candidate from the optimiser
        a_candidate = _softmax(res.x)
        val_exact = _exact_objective(a_candidate, n)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a_candidate.copy()
            print(f"[L‑BFGS λ={lam:.1e}] improved to {val_exact:.6f}")

        # use the optimiser's final logits as starting point for next λ
        logits = res.x

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Runs until the deadline and keeps the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    a /= max(a.sum(), 1e-12)          # ensure normalised

    best_a = a.copy()
    best_val = _exact_objective(a, n)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active indices (where conv reaches its maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # re‑project onto simplex (protect against under‑flow)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Very short simulated‑annealing refinement.
    Moves a random fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start, T_end = 0.08, 1e-4
    for it in range(max_moves):
        if time.time() >= deadline:
            break
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

    return best_a, best_val

def _make_initial_pool(n: int,
                       warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest but diverse pool of start vectors.
    Includes several parametric families and a few random Gaussian mixtures.
    """
    pool = []

    # warm start (if supplied)
    if warm is not None:
        pool.append(warm.copy())
        # rotations of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law family (triangular product)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Gaussian mixture (randomly many components)
    for _ in range(12):
        k = np.random.randint(2, 6)                # number of components
        mus = np.random.rand(k) * (n - 1)          # means in [0,n)
        sigs = np.random.rand(k) * (n / 6.0) + 1.0  # std dev, never too small
        weights = np.random.dirichlet(np.ones(k))
        x = np.arange(n, dtype=np.float64)
        mixture = np.zeros(n, dtype=np.float64)
        for w, mu, sigma in zip(weights, mus, sigs):
            mixture += w * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        mixture = np.maximum(mixture, 0.0)
        if mixture.sum() > 0:
            mixture /= mixture.sum()
            pool.append(mixture)

    # Sparse spikes (≈√n non‑zero entries)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low objective.
    Runs for at most `budget_s` seconds and always returns the best vector.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096                                      # length of the vector
    conv_fft_len = _next_pow2(2 * n - 1)          # 8192
    corr_fft_len = _next_pow2(3 * n - 2)          # enough for cross‑corr

    # ------------------------------------------------------------------
    # Warm start: try to reuse a previous solution if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build initial pool and pick the best by the exact objective
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _exact_objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() > deadline:
            break
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:02d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – L‑BFGS continuation on smooth‑max surrogate
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase I] L‑BFGS on smooth‑max started")
        sub_dead = deadline - 5.0
        a_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead
        )
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"[Phase I] L‑BFGS improved to {val_lbfgs:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient (sub‑gradient) optimisation
    if best_seq is not None and time.time() < deadline - 8:
        print("[Phase II] Sub‑gradient optimisation started")
        sub_dead = deadline - 7.0
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, eta0=0.7
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase III – final simulated‑annealing refinement (short)
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase III] Simulated annealing final pass")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=70_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase III] SA improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _exact_objective(best_seq, n)

    # Clip to the validator’s allowed range (just in case)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
