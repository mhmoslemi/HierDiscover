def max_conv(a):
    n = len(a)
    fft_len = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return conv.max()
```

"""
Hybrid search for a low‑max‑convolution step function.

The algorithm works in four phases:
1. Random exploration of several analytic families (beta, Gaussian, cosine‑pow,
   Hann‑pow, half‑sine, uniform, Dirichlet).  Each shape is normalised to sum=1
   and its exact objective 2·n·max(conv) is evaluated via FFT.  The best shape
   is retained.
2. L2‑flattening Adam optimisation that minimises the L2 distance between the
   convolution and the uniform target (the theoretical lower bound).
3. Smooth‑max Adam optimiser (log‑sum‑exp surrogate) to directly reduce the
   max‑convolution.
4. Projected gradient descent on the exact max‑convolution for final fine‑tuning.

All phases respect the overall time budget (default 1000 s) and the final
sequence is returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories ------------------------------------------------------
def _make_beta(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta PDF on n points (mid‑point rule), symmetric when a==b."""
    x = (np.arange(n) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        pdf = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    pdf = np.where(np.isnan(pdf) | np.isinf(pdf), 0.0, pdf)
    pdf = np.maximum(pdf, 0.0)
    # avoid a completely zero vector
    if pdf.sum() < 1e-12:
        pdf = np.full(n, 1.0 / n)
    return pdf


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    pdf = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    if pdf.sum() < 1e-12:
        pdf = np.full(n, 1.0 / n)
    return pdf


def _make_half_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    pdf = np.sin(np.pi * (i + 0.5) / n)
    pdf = np.maximum(pdf, 0.0)
    return pdf


def _make_cosine_pow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    pdf = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    pdf = np.maximum(pdf, 0.0)
    return pdf


def _make_hann_pow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    pdf = (0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))) ** p
    pdf = np.maximum(pdf, 0.0)
    return pdf


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n))


def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max -----------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of (conv-target)^2  w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # evaluate current objective (max‑conv)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on exact max (existing) -------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# Main search routine --------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ budget_s seconds. Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously discovered sequence (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – random analytic families
    print("[Phase 1] Random shape exploration")
    shape_types = [
        "beta", "gaussian", "cosine_pow", "hann_pow",
        "half_sine", "uniform", "dirichlet"
    ]
    # allocate about 30 % of the total budget to this phase
    phase1_dead = min(deadline - 30.0, start_time + budget_s * 0.30)

    while time.time() < phase1_dead:
        shape = np.random.choice(shape_types)
        if shape == "beta":
            a = np.exp(np.random.uniform(-1.5, 2.5))   # ≈0.22 … 12.2
            b = np.exp(np.random.uniform(-1.5, 2.5))
            cand = _make_beta(n, a, b)
        elif shape == "gaussian":
            sigma = np.random.uniform(n / 30.0, n / 2.0)
            cand = _make_gaussian(n, sigma)
        elif shape == "cosine_pow":
            p = np.random.uniform(0.2, 6.0)
            cand = _make_cosine_pow(n, p)
        elif shape == "hann_pow":
            p = np.random.uniform(0.2, 6.0)
            cand = _make_hann_pow(n, p)
        elif shape == "half_sine":
            cand = _make_half_sine(n)
        elif shape == "uniform":
            cand = _make_uniform(n)
        else:   # dirichlet
            alpha = np.exp(np.random.uniform(-3.0, 3.0))
            cand = _make_dirichlet(n, alpha)

        # ensure a valid probability vector
        tot = cand.sum()
        if tot < 1e-12:
            continue
        cand = cand / tot

        # evaluate objective via FFT (fast)
        mx = _full_convolution(cand, conv_fft_len).max()
        val = 2.0 * n * mx
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Shape] new best {val:.6f} type={shape}")

    # incorporate warm start if it's better
    if warm is not None:
        mx = _full_convolution(warm, conv_fft_len).max()
        warm_val = 2.0 * n * mx
        if warm_val < best_val:
            best_val = warm_val
            best_seq = warm.copy()
            print(f"[Warm] incorporated warm start with value {warm_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – L2‑flattening Adam (conv → uniform)
    if best_seq is not None and time.time() < deadline - 60.0:
        print("[Phase 2] L2‑flattening Adam")
        phase2_dead = min(deadline - 30.0, time.time() + 0.35 * (deadline - time.time()))
        target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
        a_l2, val_l2 = _adam_optimize_l2(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            target,
            phase2_dead,
            max_iters=400_000,
            lr=0.05,
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[L2Adam] improved to {val_l2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – smooth‑max Adam (direct max reduction)
    if best_seq is not None and time.time() < deadline - 30.0:
        print("[Phase 3] Smooth‑max Adam")
        phase3_dead = deadline - 15.0
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase3_dead,
            max_iters=600_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected gradient descent on exact max
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase 4] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation for reporting (uses the provided function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
