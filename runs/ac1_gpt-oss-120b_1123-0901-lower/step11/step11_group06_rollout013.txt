"""
Hybrid active‑set linear‑programming optimizer.

The routine starts from the previous best sequence (if available) and
repeatedly solves a tiny linear program that maximises the total mass of
a candidate vector `g` under the *active* max‑convolution constraints
```
    (f * g)[t] ≤ M   for all t where (f * f)[t] ≈ M .
```
`g` therefore concentrates mass where `f` is small.  By mixing a small
fraction of `g` into the current iterate the maximal convolution value
strictly decreases.  The loop runs as long as time permits; the LP has
only a handful of rows (typically ≤ 30) so each iteration is cheap.
A short projected‑gradient refinement is performed at the end.

The algorithm stays completely in the feasible region of the evaluator:
all vectors are non‑negative, normalised to unit sum (the objective is
scale‑invariant) and finally clipped to the allowed range.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _build_constraint_matrix(active_idxs: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (|active| × n) such that (A @ g)[k] = (a * g)[t_k]
    for each active index t_k.
    """
    n = a.shape[0]
    rows = active_idxs.shape[0]
    A = np.zeros((rows, n), dtype=np.float64)

    for r, k in enumerate(active_idxs):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # loop over i where a_i contributes to (a * g)[k]
        # coefficient of g_{k-i} is a_i
        for i in range(i_min, i_max + 1):
            j = k - i
            A[r, j] = a[i]
    return A


def _lp_step(a: np.ndarray, M: float, conv_fft_len: int,
            max_active: int = 60) -> tuple[np.ndarray, float]:
    """
    Perform one LP‑based improvement step.
    Returns the new vector (or the original if no improvement) and the new max.
    """
    n = a.shape[0]
    # ------------------------------------------------------------------
    # 1. Identify the (near‑)active indices of the current max‑convolution.
    conv = _full_convolution(a, conv_fft_len)
    M = conv.max()
    # tolerance – a few *machine eps* relative to M.
    tol = max(1e-12, M * 1e-9)
    active = np.where(conv >= M - tol)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=int)

    # keep the active set reasonably small (speed‑up LP)
    if active.size > max_active:
        # keep the most extreme ones (the true maximum and a few neighbours)
        # random subsample works well enough.
        active = np.random.choice(active, size=max_active, replace=False)

    # ------------------------------------------------------------------
    # 2. Build the tiny LP (maximize Σ g_j  subject to A g ≤ M, g ≥ 0).
    A_ub = _build_constraint_matrix(active, a)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g  ⇔ maximise Σ g
    bounds = [(0.0, None)] * n

    # solve – we explicitly give a tiny time limit per solve to stay inside budget
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 0.5})

    if not res.success:
        # LP failed (rare) – fall back to the original vector
        return a, M

    g0 = np.maximum(res.x, 0.0)
    s = g0.sum()
    if s < 1e-14:
        return a, M

    # ------------------------------------------------------------------
    # 3. Mix a little bit of g into a.
    g = g0 / s
    # mixing factor – start relatively aggressive and decay quickly.
    # The factor is capped from above (0.3) and from below (1e‑4).
    t = 0.18
    new_a = (1.0 - t) * a + t * g
    new_a = _project_simplex(new_a)

    # ------------------------------------------------------------------
    # 4. Compute the new maximal convolution.
    new_M = _full_convolution(new_a, conv_fft_len).max()
    # If the mix was not beneficial, try a smaller step size.
    if new_M >= M:
        # try halving the step a few times
        for _ in range(4):
            t *= 0.5
            if t < 1e-4:
                break
            new_a = (1.0 - t) * a + t * g
            new_a = _project_simplex(new_a)
            new_M = _full_convolution(new_a, conv_fft_len).max()
            if new_M < M:
                break
    # accept only if we really improved
    if new_M < M:
        return new_a, new_M
    return a, M


def _projected_subgrad(a: np.ndarray, conv_fft_len: int,
                       max_iters: int = 300_000,
                       lr0: float = 0.12) -> np.ndarray:
    """
    Classic projected sub‑gradient descent on the exact max‑convolution
    (used as a final polish after the LP phase).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    a_cur = a.copy()

    for it in range(max_iters):
        conv = _full_convolution(a_cur, conv_fft_len)
        M = conv.max()
        active = np.isclose(conv, M, atol=1e-13, rtol=0.0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        # gradient of the max‑conv w.r.t. a
        grad = 2.0 * np.cross(w, a_cur)  # not a true cross‑correlation – use FFT version instead
        # We'll compute the exact gradient via FFT for correctness
        # (the line above is a placeholder – we replace it after the loop)
        # Use FFT‑based cross‑correlation (mirrored as in the original code)
        rev_a = a_cur[::-1]
        fw = np.fft.rfft(w, conv_fft_len)
        fr = np.fft.rfft(rev_a, conv_fft_len)
        corr = np.fft.irfft(fw * fr, conv_fft_len)
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        step = lr0 / np.sqrt(it + 1.0)
        a_cur = a_cur - step * grad
        a_cur = np.maximum(a_cur, 0.0)
        a_cur = _project_simplex(a_cur)

        val = 2.0 * n * _full_convolution(a_cur, conv_fft_len).max()
        if val < best_val:
            best_val = val
            best_a = a_cur.copy()
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises   2·n·max(conv(a,a)) / (∑a)².
    The method uses an active‑set linear programming refinement loop
    followed by a short projected sub‑gradient polish.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # a tiny safety margin

    # length of the step function – 4096 gives a good trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # warm start : previous best if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.shape[0] > n:
            a = a[:n]
        elif a.shape[0] < n:
            a = np.pad(a, (0, n - a.shape[0]))
        a = np.maximum(a, 1e-12)               # avoid exact zeros
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # keep the best solution seen so far
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    print(f"[Init] value {best_val:.6f}")

    it = 0
    # ------------------------------------------------------------------
    # Phase I – active‑set LP optimisation (main workhorse)
    while time.time() < deadline and it < 200_000:
        a, M = _lp_step(a, None, conv_fft_len)
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP it={it:06d}] new best {best_val:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # Phase II – short projected sub‑gradient descent for fine tuning
    if time.time() < deadline - 5.0:
        print("[GD] final projected sub‑gradient polish")
        a_gd = _projected_subgrad(best_a, conv_fft_len,
                                  max_iters=80_000, lr0=0.08)
        val_gd = 2.0 * n * _full_convolution(a_gd, conv_fft_len).max()
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[GD] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # safety fallback
    if best_a is None or best_a.size == 0:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # clip to allowed range for the evaluator
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
