"""
Hybrid optimiser with an LP‐flattening phase.

The original pipeline flattened the convolution with an L2‐Adam phase and then
performed smooth‑max Adam, exponentiated‑gradient and projected‑gradient steps.
Here we insert a short linear‑programming (LP) phase that directly enforces
the current maximum convolution bound.  For a fixed sequence f,
the constraints  (f∗b)[k] ≤ M  are linear in b, so we can maximise
∑b under those constraints, renormalise the result and blend it with f.
Repeated LP updates tend to reduce the peak of f∗f while keeping the total
mass fixed.

The algorithm works in three stages:

1.  Small‑scale (n≈1024) L2‑Adam flattening to obtain a decent starting shape.
2.  One to a few LP mixing iterations (fast, cheap, often improve the bound).
3.  Upsample to the final length (4096) and finish with the proven Adam /
    exponentiated‑gradient refinements.

All steps respect the 1000 s time budget and fall back to a uniform sequence
if something goes wrong.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple factories -------------------------------------------------------
def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max (log‑sum‑exp surrogate) -------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# Adam optimiser – L2 flattening (target = uniform convolution) -------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of (conv‑target)^2 w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # evaluate current objective (max‑conv)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val

# Sub‑gradient optimiser – exponentiated‑gradient -----------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Projected gradient descent on exact max -------------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP helper – build Toeplitz convolution matrix (sparse) ----------------
def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A (shape (2n-1, n)) such that A @ b = np.convolve(f, b, mode='full')
    for any non‑negative vector b.  The matrix is extremely sparse and stored in CSR format.
    """
    n = f.shape[0]
    m = 2 * n - 1
    # Each offset i (0 ≤ i < n) contributes a diagonal of length min(n, m‑i)
    data = []
    offsets = []
    for i in range(n):
        length = min(n, m - i)          # number of rows that contain this diagonal
        data.append(np.full(length, f[i]))
        offsets.append(i)               # row = col + i
    A = sparse.diags(data, offsets, shape=(m, n), format="csr")
    return A

# Main search routine ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a small‑scale L2‑Adam start, a few LP mixing steps, then up‑samples
    to the full size and finishes with Adam / exponentiated‑gradient refinements.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Parameters
    final_n = 4096            # target length used for the final evaluation
    n_small = 1024            # size used for the cheap LP phase

    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # --------------------------------------------------------------
    # Warm start handling (global variable may contain a previous best)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_small:
            warm = warm[:n_small]
        elif warm.shape[0] < n_small:
            warm = np.pad(warm, (0, n_small - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Initial candidate (either warm or a fresh Dirichlet)
    if warm is not None:
        a0_small = warm.copy()
    else:
        a0_small = _make_dirichlet(n_small, alpha=0.2)

    # --------------------------------------------------------------
    # Stage 1 – L2‑Adam flattening (target = uniform convolution)
    target_small = np.full(2 * n_small - 1, 1.0 / (2 * n_small - 1), dtype=np.float64)
    phase_dead = min(deadline, time.time() + 15.0)   # about 15 s for the flattening
    a_small, val_small = _adam_optimize_l2(
        a0_small,
        n_small,
        conv_fft_len_small,
        corr_fft_len_small,
        target_small,
        phase_dead,
        max_iters=400_000,
        lr=0.07,
    )
    best_a_small = a_small.copy()
    best_val_small = 2.0 * n_small * np.max(_full_convolution(best_a_small, conv_fft_len_small))
    print(f"[Stage 1] L2‑Adam finished best={best_val_small:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – LP mixing iterations (very cheap, each < 0.5 s)
    while time.time() < deadline - 10.0:
        conv = _full_convolution(best_a_small, conv_fft_len_small)
        M = conv.max()

        # Build the convolution matrix (sparse)
        A = _build_conv_matrix(best_a_small)

        # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n_small)                     # minimise -∑b
        b_ub = np.full(2 * n_small - 1, M)
        bounds = [(0, None)] * n_small

        # Solve with HiGHS (fast for sparse problems)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True, "time_limit": 5.0})

        if not res.success:
            print("[LP] Linprog failed – exiting LP phase")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b < 1e-12:
            print("[LP] Solution has negligible mass – stop")
            break

        # Normalise to a probability vector
        g = b_opt / sum_b

        # Simple line‑search on mixing coefficient
        improved = False
        for t in (0.5, 0.3, 0.1, 0.05, 0.02):
            cand = (1 - t) * best_a_small + t * g
            conv_cand = _full_convolution(cand, conv_fft_len_small)
            val_cand = 2.0 * n_small * conv_cand.max()
            if val_cand < best_val_small:
                best_a_small = cand
                best_val_small = val_cand
                improved = True
                print(f"[LP] mix t={t:.2f} improved to {val_cand:.6f}")
                break
        if not improved:
            # No progress → stop the LP loop
            break

    # --------------------------------------------------------------
    # Stage 3 – Upsample to final length (linear interpolation)
    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, final_n)
    a_big = np.interp(x_big, x_small, best_a_small)
    a_big = np.maximum(a_big, 0.0)
    # Normalise for numerical stability (not required by the objective)
    if a_big.sum() > 1e-12:
        a_big = a_big / a_big.sum()
    else:
        a_big = _make_dirichlet(final_n, alpha=0.2)

    # --------------------------------------------------------------
    # Stage 4 – Smooth‑max Adam fine‑tuning on the full size
    conv_fft_len_big = _next_pow2(2 * final_n - 1)
    corr_fft_len_big = _next_pow2(3 * final_n - 2)
    phase_dead = min(deadline, time.time() + 40.0)   # ~40 s for Adam
    a_big, val_big = _adam_optimize(
        a_big,
        final_n,
        conv_fft_len_big,
        corr_fft_len_big,
        phase_dead,
        max_iters=600_000,
    )
    best_seq = a_big.copy()
    best_val = val_big
    print(f"[Stage 4] Adam finished best={best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – Exponentiated‑gradient sub‑gradient refinement
    if time.time() < deadline - 5.0:
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            final_n,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=400_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Stage 5] ExpGrad improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – Projected gradient descent (exact max)
    if time.time() < deadline - 5.0:
        a_gd, val_gd = _max_conv_descent(
            best_seq,
            final_n,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Stage 6] GD improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Stage 7 – Final simulated‑annealing polish
    if time.time() < deadline - 5.0:
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            final_n,
            conv_fft_len_big,
            deadline,
            max_moves=80_000,
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Stage 7] Anneal improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Guarantee a non‑empty return (fallback to uniform)
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(final_n, 1.0 / final_n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final reporting using the supplied evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Evaluation error: {exc}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
