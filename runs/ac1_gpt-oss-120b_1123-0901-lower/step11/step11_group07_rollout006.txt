def solve_lp_max_sum(f, M):
    n = len(f)
    offsets = np.arange(n)
    data = np.tile(f, (n,1))  # shape (n, n)
    A_ub = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'time_limit': time_limit})
    if not res.success:
        return None
    g = res.x
    # ensure non-negative and sum > 0
    return g
```

Hybrid optimizer with LP refinement + exponentiated gradient polishing.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _project_simplex(v):
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _eval_val(a, n, conv_fft_len):
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

def _solve_lp_max_sum(f, M, max_time=None):
    """Solve max sum(g) s.t. f*g <= M (linear constraints). Returns g or None."""
    import scipy.sparse
    import scipy.optimize
    n = f.shape[0]
    # Build sparse matrix where each diagonal d (0..n-1) contains f on that diagonal.
    offsets = np.arange(n)
    # data for each diagonal: the same vector f
    data = np.tile(f, (n, 1))  # shape (n, n)
    A_ub = scipy.sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)  # maximize sum(g) -> minimize -sum(g)
    bounds = [(0, None)] * n

    # Use high-performance HiGHS method; it respects a 'time_limit' option
    options = {}
    if max_time is not None:
        options['time_limit'] = max_time

    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                 bounds=bounds, method='highs',
                                 options=options)

    if not res.success:
        return None
    g = res.x
    # clip tiny negatives due to numerical errors
    g = np.maximum(g, 0.0)
    return g

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step function minimizing the evaluation metric.

    The algorithm starts from a warm start (if available), refines it by a
    small linear‑programming based “active‑set” step that reduces the
    max‑convolution, and finally polishes the result with an exact
    exponentiated‑gradient sub‑gradient descent.  The whole search respects
    the 1000 s budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # problem size – 4096 points give a good speed/quality trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start – try to reuse a previously discovered good sequence
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Initialise f – either the warm start or a fresh random Dirichlet
    if warm is not None:
        f = warm.copy()
    else:
        f = np.random.dirichlet(np.ones(n)).astype(np.float64)

    # Remember the best solution seen so far.
    best_f = f.copy()
    best_val = _eval_val(best_f, n, conv_fft_len)
    print(f"[Init] initial value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – LP‑based active‑set refinement
    # Perform a handful (≤ 6) of LP+mix steps while time permits.
    lp_iters = 0
    while time.time() < deadline - 30.0 and lp_iters < 6:
        M = _full_convolution(f, conv_fft_len).max()
        # allocate a small slice for the LP solve
        time_left = deadline - time.time()
        lp_time = min(8.0, time_left * 0.2)  # give at most 8 s per LP
        g = _solve_lp_max_sum(f, M, max_time=lp_time)
        if g is None:
            print("[LP] failed to solve linear program")
            break
        sum_g = g.sum()
        if sum_g < 1e-12:
            print("[LP] trivial solution returned")
            break
        g = g / sum_g

        # mixing coefficient – start aggressive, then damp.
        t = 0.5 if lp_iters == 0 else 0.3
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = _project_simplex(f_new)

        val = _eval_val(f_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            f = f_new
            print(f"[LP it={lp_iters}] improved to {val:.6f}")
        else:
            # If no improvement, try a smaller mixing factor
            improved = False
            for t_small in (0.2, 0.1, 0.05):
                f_tmp = (1 - t_small) * f + t_small * g
                f_tmp = np.maximum(f_tmp, 0.0)
                f_tmp = _project_simplex(f_tmp)
                val_tmp = _eval_val(f_tmp, n, conv_fft_len)
                if val_tmp < best_val:
                    best_val = val_tmp
                    best_f = f_tmp.copy()
                    f = f_tmp
                    improved = True
                    print(f"[LP small t={t_small}] improved to {val_tmp:.6f}")
                    break
            if not improved:
                print(f"[LP it={lp_iters}] no improvement, terminating LP phase")
                break
        lp_iters += 1

    # ------------------------------------------------------------------
    # Phase II – Exponentiated‑gradient sub‑gradient polishing
    # Run until the deadline is reached (or for a safe upper bound).
    # Implementation follows the _subgradient_optimize_expgrad logic.
    a = best_f.copy()
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _eval_val(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline - 2.0:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # Identify active indices where conv == max_c (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # Exact gradient of max_c w.r.t a: 2 * (w ★ a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Exponentiated‑gradient update with diminishing step size
        eta = 0.9 / np.sqrt(it + 1.0)   # same schedule as previous code
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (convert to Python list)
    best_seq = best_a if best_val <= _eval_val(best_f, n, conv_fft_len) else best_f
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
