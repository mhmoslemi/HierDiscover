"""
Hybrid non‑negative step‑function optimizer (≈ 1 000 s budget).

Goal
----
Minimise  2·n·max(conv(a,a)) / (sum(a)²)  with a_i ≥ 0.
The objective is invariant to scaling → we keep the vector on the simplex
 Σ a_i = 1.  The lower bound for large n is ≈ 1 (the average of the
 convolution).  Our strategy is a long adaptive Adam phase on a smooth‑max
 surrogate, followed by a pure sub‑gradient exponentiated‑gradient phase,
 an aggressive spectral‑flattening step and a short simulated‑annealing
 polish.  All phases run until the global deadline and keep the best
 vector seen.

Key ideas compared with the previous version
--------------------------------------------
* Use a larger problem size (n = 8192) – more degrees of freedom give a
  lower theoretical optimum.
* Faster “smooth‑max” temperature schedule (λ grows slowly) and a
  larger Adam learning‑rate to explore more aggressively.
* Sub‑gradient phase uses a simple diminishing step‑size and projects
  onto the simplex with the Euclidean projector (instead of exponentiated‑gradient),
  which often yields sharper descent.
* Spectral flattening enforces *exact* constant magnitude of the FFT
  (|FFT| = const) on every iteration, not just an approximate one.
* A final “local‑swap” annealing that moves a random fraction of mass
  between two coordinates (many moves, fast temperature decay).

The code is fully self‑contained, uses only NumPy and the provided
`evaluate_sequence` for the final scoring, and respects the 1000 s
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions ----------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for gradient of max(conv).

    Returns an array of length len(a) containing (w ★ a)[i] = Σ_j w[i+j] a[j].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)

# Optimisation phases --------------------------------------------------
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                n: int,
                conv_fft_len: int,
                corr_fft_len: int,
                deadline: float,
                max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of max(conv).

    Returns the best vector encountered (simplex‑projected) and its exact
    objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation (a = softmax(logits))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr = 0.08                         # a bit larger than before
    lam = 0.03                        # start temperature
    lam_factor = 1.012                # slower temperature rise

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # exact objective for bookkeeping (still cheap)
        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {best_val:.6f}")

        # smooth‑max (soft‑max over the convolution entries)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.92

        it += 1
        # cheap deadline check every 1000 iters
        if it % 1000 == 0 and time.time() >= deadline:
            break

    # final exact evaluation of best point
    best_val = _objective(best_a, n, conv_fft_len)
    return best_a, best_val


def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       conv_fft_len: int,
                       corr_fft_len: int,
                       deadline: float,
                       eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected (Euclidean) gradient descent on the exact max‑convolution.
    A diminishing step size η = η0 / sqrt(t) is used.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a = a / a_sum

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Subgrad it={it:07d}] new best {best_val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()  # uniform over active constraints

        # sub‑gradient of max(conv) w.r.t a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # gradient step and simplex projection
        a = a - eta * grad
        a = _project_simplex(a)

        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


def _spectral_flatten_phase(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           deadline: float,
                           max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Enforce (near) constant magnitude of FFT(a) while staying on the simplex.
    The iterate is projected onto the simplex after each inverse FFT.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    # target magnitude is sqrt( n * ||a||_2^2 )  (energy spread equally over all frequencies)
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        cur_val = _objective(a, n, conv_fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Spectral it={it:06d}] new best {best_val:.6f}")

        # ----- spectral flattening step -----
        A = np.fft.fft(a)
        # keep DC term (sum) untouched, redistribute remaining energy uniformly
        energy = np.dot(a, a)                      # ||a||_2^2
        if n > 1:
            # total energy in non‑DC frequencies = n*energy - |A[0]|^2
            rem = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # enforce non‑negativity and simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


def _anneal_phase(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  deadline: float,
                  max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style local search.
    Moves a random fraction of mass between two coordinates.
    """
    a = a0.copy()
    cur_val = _objective(a, n, conv_fft_len)
    best_val = cur_val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-5

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of mass from i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            cur_val = val2
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - cur_val) / T):
                a = a2
                cur_val = val2

        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to generate a diverse pool of initial vectors -----------------
# ----------------------------------------------------------------------


def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Return a modest but diverse pool of probability vectors."""
    pool = []

    # warm start (if supplied)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with many concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential shapes (single‑sided and mirrored)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law “triangular product”
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian spikes
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.15, 0.3):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Sparse spike candidates (≈√n spikes)
    k_vals = (int(np.sqrt(n) * 0.8), int(np.sqrt(n)), int(np.sqrt(n) * 1.2))
    for k in k_vals:
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    n = 8192                    # length of the vector (more degrees of freedom)
    conv_fft_len = _next_pow2(2 * n - 1)      # for convolution (≈ 16384)
    corr_fft_len = _next_pow2(3 * n - 2)      # for cross‑correlation (≈ 32768)

    # ------------------------------------------------------------------
    # Warm start from a previous run, if present -----------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Initialise a pool of diverse candidates and pick the best ----------
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:03d} val={val:.6f}")
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase I – Adam optimisation (smooth‑max) -------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase I] Adam optimisation started")
        sub_dead = deadline - 5.0
        a_adam, val_adam = _adam_phase(best_seq, n,
                                       conv_fft_len, corr_fft_len,
                                       sub_dead, max_iters=800_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – Projected sub‑gradient descent -------------------------
    if best_seq is not None and time.time() < deadline - 7:
        print("[Phase II] Sub‑gradient optimisation started")
        sub_dead = deadline - 7.0
        a_sub, val_sub = _subgradient_phase(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            sub_dead, eta0=0.9)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase III – aggressive spectral flattening ------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] Spectral flattening started")
        a_spec, val_spec = _spectral_flatten_phase(best_seq, n,
                                                   conv_fft_len, deadline,
                                                   max_iters=400_000)
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Phase III] spectral improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – final simulated‑annealing refinement --------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase IV] Simulated annealing final pass")
        a_ann, val_ann = _anneal_phase(best_seq, n,
                                       conv_fft_len, deadline,
                                       max_moves=150_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the validator’s allowed numeric range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
