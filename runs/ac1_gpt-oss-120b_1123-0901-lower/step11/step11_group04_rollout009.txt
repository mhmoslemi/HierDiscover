"""
Hybrid optimisation for the step‑function problem.

The routine works in three stages:

1. **Broad exploration** – generate many analytic families (uniform,
   exponential, power‑law, Gaussian, Hann, random‑phase, Dirichlet,
   triangular) and sweep a small set of parameters.  Each candidate is
   normalised to Σaᵢ=1 and evaluated with a cheap FFT‑based objective
   2·n·max(conv(a,a)).  The best among them seeds the refinements.

2. **Refinement** – the best seed is polished by:
   * an L2‑flattening Adam phase (minimise ‖conv−uniform‖₂),
   * a smooth‑max Adam phase (optimise a differentiable surrogate of the
     max‑convolution),
   * an exact‑max exponentiated‑gradient sub‑gradient phase,
   * a projected‑gradient descent on the true max‑convolution,
   * a short simulated‑annealing local search.

   All phases respect a global time budget and keep the vector on the
   probability simplex (non‑negative, Σ=1).

3. **Return** – the best vector found (as a plain list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def _make_hann(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    if power != 1.0:
        a = a ** power
    a /= a.sum()
    return a


def _make_random_phase(n: int) -> np.ndarray:
    half = n // 2 + 1
    phi = np.random.uniform(0.0, 2.0 * np.pi, size=half)
    phi[0] = 0.0               # zero‑frequency must be real positive
    X = np.exp(1j * phi)
    a = np.fft.irfft(X, n=n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum a = 1 → 2·n·max(conv))."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam – L2 flattening ---------------------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_len)
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_len)

        # exact objective for bookkeeping
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val

# Adam – smooth‑max surrogate -------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# Exponentiated‑gradient (exact‑max sub‑gradient) -----------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Projected gradient descent on the exact max ---------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# Simulated annealing (lightweight) ------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0

    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            if np.random.rand() < np.exp(-(val2 - best_val) / T):
                a = a2

        it += 1

    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The algorithm combines a fast parametric sweep with several gradient‑based
    refinements (L2‑flattening Adam → smooth‑max Adam → exponentiated‑gradient
    → projected GD → annealing).  It respects the time budget and returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n gives a tighter theoretical bound but also
    #   larger FFTs; 4096 is a good trade‑off for the allotted time)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start if a previous construction is available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of cheap candidates
    pool = []

    if warm is not None:
        pool.append(warm.copy())

    # Dirichlet draws with a range of concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))

    # Analytic families (some with a small parameter sweep later)
    pool.append(_make_uniform(n))
    pool.append(_make_triangle(n))

    # --- Parameter sweeps ------------------------------------------------
    # Exponential decay
    for gamma in np.logspace(-4, 0, 12):
        pool.append(_make_exponential(n, gamma))

    # Power‑law
    for expo in np.linspace(-2.0, -0.3, 12):
        pool.append(_make_powerlaw(n, expo))

    # Gaussian (sigma as a fraction of n)
    for sigma_frac in np.linspace(0.02, 0.25, 8):
        sigma = sigma_frac * n
        pool.append(_make_gaussian(n, sigma))

    # Hann windows with optional exponent
    for power in (0.5, 1.0, 2.0):
        pool.append(_make_hann(n, power))

    # Random‑phase constructions
    for _ in range(5):
        pool.append(_make_random_phase(n))

    # ------------------------------------------------------------------
    # Quick evaluation of the pool – keep the best few for heavy work
    best_val = float("inf")
    best_seq = None
    top_candidates = []   # store (value, vector) for later refinement

    print("[Stage 0] Evaluating cheap candidates …")
    for idx, seq in enumerate(pool):
        seq = np.maximum(seq, 0.0)
        if seq.sum() < 0.01:
            seq = np.random.dirichlet(np.ones(n))
        seq = seq / seq.sum()
        val = _objective(seq, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        # keep up to 5 promising candidates
        if len(top_candidates) < 5:
            top_candidates.append((val, seq.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif val < top_candidates[-1][0]:
            top_candidates[-1] = (val, seq.copy())
            top_candidates.sort(key=lambda x: x[0])

        if idx % 20 == 0:
            print(f"  candidate {idx:3d} → {val:.6f}")

    print(f"[Stage 0] Best cheap value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Uniform target for the L2‑flattening phase
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Deep optimisation – run through the top candidates until time runs out
    for rank, (init_val, a_start) in enumerate(top_candidates):
        if time.time() > deadline - 5:
            break
        print(f"[Stage 1] Refining candidate #{rank+1} (init {init_val:.6f})")

        a0 = a_start.copy()

        # ----- L2‑flattening Adam -----
        phase_dead = min(deadline, time.time() + 25.0)
        a_l2, val_l2 = _adam_optimize_l2(
            a0, n, conv_fft_len, corr_fft_len, target, phase_dead,
            max_iters=300_000, lr=0.07
        )
        if val_l2 < best_val:
            best_val, best_seq = val_l2, a_l2.copy()
            print(f"   L2‑Adam improved to {val_l2:.6f}")

        # ----- Smooth‑max Adam (fine‑tune) -----
        if time.time() < deadline - 5:
            phase_dead = min(deadline, time.time() + 30.0)
            a_adam, val_adam = _adam_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, phase_dead,
                max_iters=400_000
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"   smooth‑max Adam improved to {val_adam:.6f}")

        # ----- Exact‑max exponentiated‑gradient -----
        if time.time() < deadline - 5:
            a_exp, val_exp = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=800_000, eta0=0.9
            )
            if val_exp < best_val:
                best_val, best_seq = val_exp, a_exp.copy()
                print(f"   ExpGrad improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Global projected gradient descent (if time permits)
    if best_seq is not None and time.time() < deadline - 15:
        print("[Stage 2] Projected gradient descent on exact max‑conv")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"   GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing refinement
    if best_seq is not None and time.time() < deadline - 15:
        print("[Stage 3] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"   Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Simple blend of the two best candidates (optional cheap boost)
    if len(top_candidates) >= 2 and best_seq is not None:
        sec_seq = top_candidates[1][1]
        for t in np.linspace(0.0, 1.0, 21):
            blend = (1.0 - t) * best_seq + t * sec_seq
            blend = np.maximum(blend, 0.0)
            blend = blend / blend.sum()
            val = _objective(blend, n, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, blend.copy()
                print(f"   Blend t={t:.2f} gave new best {val:.6f}")

    # ------------------------------------------------------------------
    # Ensure we return a non‑empty list
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's admissible range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final external evaluation (for reporting)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
