"""
Hybrid optimizer for the step‑function (self‑convolution peak) problem.

The algorithm proceeds in several stages, each respecting a global time budget
(≈1000 s).  The main ingredients are:

1.   A diverse pool of start vectors (including an arcsine‐type shape,
     warm‑start if available, random Dirichlet draws and sparse “spike’’
     vectors).  The best of these is selected after a quick evaluation.

2.   Adam optimisation on a smooth‑max surrogate of the peak of the
     self‑convolution.  The surrogate becomes tighter during optimisation
     (the temperature λ grows geometrically).

3.   Exact sub‑gradient descent using an exponentiated‑gradient update
     (works directly on the true max‑convolution).

4.   Spectral‑flattening: the magnitude of the discrete Fourier transform of
     the candidate is forced to be (almost) constant while staying on the
     probability simplex.

5.   Linear‑programming refinements as described in the statement – a few
     iterations of the LP (maximise Σb subject to f★b ≤ M) are mixed back
     into the current iterate.

6.   Projected gradient descent on the exact max‑convolution (simple Euclidean
     projection onto the simplex after each step).

7.   Simulated‑annealing polishing.

8.   A tiny “power‑transform’’ local search (a⁽ᵖ⁾, p≈1) and very short random
     perturbations finish the run.

All iterates are non‑negative and normalised (Σa = 1); the evaluation function
is homogeneous, so normalisation does not affect the score.
The routine always returns the best sequence found before the deadline.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.2          # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to get probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            # keep DC term, flatten all other magnitudes
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 8,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence f we maximise Σb subject to (f★b)_k ≤ M,
    where M = max_k (f★f)_k.  The solution b is mixed back into f.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active positions of the peak (with tiny tolerance)
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        # add a few well‑spread indices to avoid degenerate LPs
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # ----- build sparse constraint matrix (f★b)_k ≤ M -----
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # ----- LP: maximise Σb  <=> minimise -Σb -----
        c = -np.ones(n)                      # objective
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # mixing
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Generate a step‑function (list of non‑negative heights) with low evaluation."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Length – a larger n gives more freedom at a modest extra cost.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- build initial pool -------------------------------------------------
    pool = []

    # warm start if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] > n:
            warm_arr = warm_arr[:n]
        elif warm_arr.shape[0] < n:
            warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]), "constant")
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() >= 0.01:
            warm_arr = warm_arr / warm_arr.sum()
            pool.append(warm_arr)

    # an arcsine‑type shape (often good for anti‑concentration)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = arcsine / arcsine.sum()
    pool.append(arcsine)

    # random Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        pool.append(a)

    # sparse spike candidates (handy for escaping plateaus)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        pool.append(a)

    # quick scan of pool to pick a promising start
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        if time.time() > deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Phase I – Adam optimisation (≈20 s)
    if time.time() < deadline - 20:
        sub_dead = min(deadline, time.time() + 20.0)
        best_seq, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, best_seq.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # Phase II – sub‑gradient (exponentiated‑gradient) (≈30 s)
    if time.time() < deadline - 15:
        sub_dead = min(deadline, time.time() + 30.0)
        best_seq, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=400_000, eta0=0.6,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, best_seq.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # Phase III – spectral flattening (≈30 s)
    if time.time() < deadline - 15:
        limit = deadline - 5
        best_seq, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, limit, max_iters=200_000
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, best_seq.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # Phase IV – linear‑programming refinement (few iterations)
    if time.time() < deadline - 15:
        best_seq, val_lp = _lp_improve(
            best_seq, n, conv_fft_len, deadline, max_iter=8, tau=0.5
        )
        if val_lp < best_val:
            best_val, best_seq = val_lp, best_seq.copy()
            print(f"[LP] improved to {val_lp:.6f}")

    # Phase V – projected gradient descent (fine‑grained)
    if time.time() < deadline - 10:
        best_seq, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=150_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, best_seq.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # Phase VI – simulated annealing polish
    if time.time() < deadline - 5:
        best_seq, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=80_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, best_seq.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # Phase VII – power‑transform local search (very cheap)
    if time.time() < deadline - 2:
        for p in (0.9, 0.95, 0.98, 1.0, 1.02, 1.05):
            a = best_seq ** p
            a = a / a.sum()
            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Power {p:.2f}] improved to {val:.6f}")

    # Phase VIII – tiny random perturbations (last seconds)
    if time.time() < deadline - 0.5:
        trials = 0
        while time.time() < deadline and trials < 3000:
            a = best_seq.copy()
            i = np.random.randint(n)
            if a[i] < 1e-12:
                trials += 1
                continue
            j = np.random.randint(n)
            eps = np.random.rand() * 0.008 * a[i]   # move ≤ 0.8 % of i‑mass
            a[i] -= eps
            a[j] += eps
            a = _project_simplex(a)
            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Perturb] improved to {val:.6f}")
            trials += 1

    # ----------------------------------------------------------------------
    # final safety net
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
