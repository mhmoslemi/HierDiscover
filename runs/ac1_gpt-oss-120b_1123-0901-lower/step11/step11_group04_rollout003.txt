Hybrid large‑scale optimizer for the step‑function problem.

The routine builds a fairly large candidate (default n=16384) and
optimises it in three stages:

3. **Exact‑max sub‑gradient** – Exponentiated‑gradient updates use the
   true max‑convolution sub‑gradient to polish the solution.

All stages respect the global time budget (≈ 1000 s).  Progress and
intermediate objective values are printed.  The final sequence is
returned as a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT / convolution helpers -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple factories for starting points ------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_Ushape(n: int) -> np.ndarray:
    """Symmetric “U‑shaped” distribution – large at the ends, small in the centre."""
    i = np.arange(n, dtype=np.float64)
    left = 1.0 / (i + 1.0)
    right = 1.0 / (n - i)
    a = left + right
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -------------------------------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the L2 distance to the uniform convolution target.
    Returns the best vector encountered and its exact objective value
    (2·n·max(conv)).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                      # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of ∥conv‑target∥² w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # exact objective (max‑convolution)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2 Adam it={it:06d}] new best {val:.6f}")

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # optional learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp smooth approximation of max(conv).
    Returns the best vector encountered and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Smooth Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a large problem size (n≈16000) and a three‑stage optimiser.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – we take a fairly large n to get a better theoretical bound.
    n = 16384
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Uniform target for L2‑flattening (each entry = 1/(2n‑1))
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Build a small diverse pool of initial vectors.
    pool = []

    # Warm‑start from a previous solution (if it exists) – up‑sample / pad.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] < n:
            pad_before = (n - warm.shape[0]) // 2
            pad_after = n - warm.shape[0] - pad_before
            warm = np.pad(warm, (pad_before, pad_after))
        else:
            warm = warm[:n]
        warm = np.clip(warm, 0, None)
        s = warm.sum()
        if s >= 0.01:
            warm = warm / s
            pool.append(warm)

    # Analytic families
    pool.append(_make_uniform(n))
    pool.append(_make_exponential(n, gamma=0.5))
    pool.append(_make_powerlaw(n, expo=-0.8))
    pool.append(_make_Ushape(n))

    # A few random Dirichlets (different concentrations)
    for alpha in (0.01, 0.1, 0.5):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optimisation over the pool – each candidate gets a share of the budget.
    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break

        # Normalise to a proper probability vector
        a0 = np.clip(a0, 0, None)
        s0 = a0.sum()
        if s0 < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
            s0 = a0.sum()
        a0 = a0 / s0

        # ---------- Phase 1 : L2‑flattening (large chunk of remaining time) ----------
        time_left = deadline - time.time()
        # keep at least 30 s for the later phases
        phase_dead = time.time() + max(5.0, 0.5 * time_left)
        a_l2, val_l2 = _adam_optimize_l2(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            target,
            phase_dead,
            max_iters=600_000,
            lr=0.05,
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Candidate {idx}] L2‑Adam improved to {val_l2:.6f}")

        # ---------- Phase 2 : smooth‑max Adam (use whatever time remains) ----------
        if time.time() < deadline - 5:
            phase_dead2 = deadline  # use the rest of the budget
            a_smooth, val_smooth = _adam_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                phase_dead2,
                max_iters=800_000,
            )
            if val_smooth < best_val:
                best_val = val_smooth
                best_seq = a_smooth.copy()
                print(f"[Candidate {idx}] Smooth‑max Adam improved to {val_smooth:.6f}")

        # ---------- Phase 3 : exact‑max sub‑gradient (exp‑grad) ----------
        if time.time() < deadline - 5:
            a_exp, val_exp = _subgradient_optimize_expgrad(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=600_000,
                eta0=0.8,
            )
            if val_exp < best_val:
                best_val = val_exp
                best_seq = a_exp.copy()
                print(f"[Candidate {idx}] ExpGrad improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # If nothing was produced (unlikely), fall back to a uniform vector.
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = float("inf")

    # ------------------------------------------------------------------
    # Clip to the allowed value range and report final value.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
