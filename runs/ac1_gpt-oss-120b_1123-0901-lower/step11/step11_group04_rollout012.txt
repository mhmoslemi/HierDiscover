def _local_search(a, n, conv_fft_len, deadline):
    best_a = a.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_max = conv.max()
    while time.time() < deadline:
        # pick i, j
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # pick delta fraction
        delta = best_a[i] * np.random.rand() * 0.2  # up to 20% of a[i]
        if delta < 1e-12:
            continue
        a2 = best_a.copy()
        a2[i] -= delta
        a2[j] += delta
        # due to numerical errors may cause negative; ensure non-negative
        a2[i] = max(a2[i], 0.0)
        # Normalize to sum=1 (optional as scaling cancels)
        # Actually may not need normalization but keep to maintain fairness.
        a2 /= a2.sum()
        conv2 = _full_convolution(a2, conv_fft_len)
        max2 = conv2.max()
        if max2 < best_max:
            best_max = max2
            best_a = a2
            # print improvement
            print(f"[Local] improved max to {best_max:.8f}")
        # else maybe accept with probability? Not needed.
    return best_a, best_max

def _subgradient_descent_max(a, n, fft_len, corr_fft_len, deadline, max_iters=200_000):

"""
Hybrid optimiser for the step‑function evaluation problem.

The algorithm combines several complementary optimisation phases:

1. **Warm‑start pool** – includes the previous best sequence (if any) and a
   collection of analytically‑generated distributions (uniform, exponential,
   power‑law, triangular) as well as Dirichlet samples with varied
   concentration.

2. **L2‑flattening (Adam)** – minimises the L2‑distance between the
   auto‑convolution `a*a` and the uniform target (the theoretical lower bound
   for the max‑convolution).  This phase quickly brings the convolution close
   to flat.

3. **Smooth‑max (Adam)** – a log‑sum‑exp surrogate of the max‑convolution is
   minimised; the temperature `lam` is increased gradually so that the
   optimisation eventually concentrates on the true maximum.

4. **Random‑seed exploration** – while time permits, new Dirichlet seeds are
   generated and a short L2‑Adam run is performed; this diversifies the search
   and often uncovers better basins.

5. **Exact‑max sub‑gradient (exponentiated‑gradient)** – fine‑tunes the
   solution with the true max‑convolution; the step‑size decays as `1/√t`.

6. **Projected gradient descent on the exact max** – a simple GD on the
   max‑convolution, projected onto the probability simplex.

7. **Simulated‑annealing refinement** – a long annealing phase with many
   pairwise mass moves; the temperature decays slowly to allow escaping shallow
   local minima.

All phases respect the global time budget (default 1000 s).  The final
sequence is clipped to `[0,1000]` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories ----------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser – L2 flattening -----------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the L2 distance between conv(a,a) and the uniform target.
    Returns the best vector encountered and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to get a in the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of (conv‑target)^2 w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # exact objective (max‑conv)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # learning‑rate decay every 50 k iterations
        if it % 50_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max (log‑sum‑exp surrogate) --------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.05                # start temperature
    lam_factor = 1.05         # accelerate towards true max

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[SmoothAdam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # decay learning‑rate slowly
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser – exponentiated‑gradient ------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on exact max ---------------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement -----------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Long‑run simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    T_start = 0.3
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        # pick two distinct indices
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of a[i] to a[j]
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        # maintain simplex (renormalise – scaling cancels in the objective)
        a2 = np.maximum(a2, 0.0)
        a2_sum = a2.sum()
        if a2_sum < 1e-12:
            it += 1
            continue
        a2 /= a2_sum

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            if np.random.rand() < np.exp(-(val2 - best_val) / T):
                a = a2

        it += 1
        # occasional progress report
        if it % 250_000 == 0:
            print(f"[Anneal] it={it:07d}, best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    A time budget of `budget_s` seconds is respected.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4096 gives a good speed / quality trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    print(f"[Info] optimisation started – n={n}")

    # ------------------------------------------------------------------
    # Warm start (previous best) if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] != n:
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]), mode="constant")
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() > 1e-6:
            warm_arr = warm_arr / warm_arr.sum()
            pool = [warm_arr]
            print("[Info] using warm start")
        else:
            pool = []
    else:
        pool = []

    # ------------------------------------------------------------------
    # Analytic and Dirichlet families for the initial pool
    pool.extend([
        _make_dirichlet(n, alpha=0.02),
        _make_dirichlet(n, alpha=0.05),
        _make_dirichlet(n, alpha=0.2),
        _make_dirichlet(n, alpha=0.7),
        _make_uniform(n),
        _make_exponential(n, gamma=0.6),
        _make_powerlaw(n, expo=-1.1),
        _make_triangle(n),
    ])

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # uniform target for L2 flattening
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 1: deterministic pool (short L2 + short smooth‑max)
    for idx, a0 in enumerate(pool):
        if time.time() > deadline - 180:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-8:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # L2‑flattening (≈12 s)
        phase_dead = min(deadline, time.time() + 12.0)
        a_l2, val_l2 = _adam_optimize_l2(
            a0, n, conv_fft_len, corr_fft_len, target, phase_dead,
            max_iters=250_000, lr=0.07,
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Pool {idx}] L2‑Adam improved to {val_l2:.6f}")

        # smooth‑max fine‑tune (≈8 s)
        if time.time() < deadline - 150:
            phase_dead2 = min(deadline, time.time() + 8.0)
            a_smax, val_smax = _adam_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, phase_dead2,
                max_iters=400_000,
            )
            if val_smax < best_val:
                best_val = val_smax
                best_seq = a_smax.copy()
                print(f"[Pool {idx}] Smooth‑max Adam improved to {val_smax:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: random Dirichlet scouting while time permits
    scout_idx = 0
    while time.time() < deadline - 120:
        # random concentration in [0.01, 5]
        alpha = 10 ** np.random.uniform(-2, 0.7)
        a0 = _make_dirichlet(n, alpha)

        # very short L2 run (≈5 s)
        phase_dead = min(deadline, time.time() + 5.0)
        a_l2, val_l2 = _adam_optimize_l2(
            a0, n, conv_fft_len, corr_fft_len, target, phase_dead,
            max_iters=100_000, lr=0.07,
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Scout {scout_idx}] L2‑Adam improved to {val_l2:.6f}")
        scout_idx += 1

    # ------------------------------------------------------------------
    # Phase 3: extended smooth‑max optimisation on the current best
    if best_seq is not None and time.time() < deadline - 80:
        print("[Refine] Extended smooth‑max Adam")
        phase_dead = deadline - 80  # reserve time for later phases
        a_smax, val_smax = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, phase_dead,
            max_iters=2_000_000,
        )
        if val_smax < best_val:
            best_val = val_smax
            best_seq = a_smax.copy()
            print(f"[Refine] Smooth‑max Adam improved to {val_smax:.6f}")

    # ------------------------------------------------------------------
    # Phase 4: sub‑gradient exponentiated‑gradient
    if best_seq is not None and time.time() < deadline - 50:
        print("[Refine] Exponentiated‑gradient sub‑gradient")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=800_000, eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Refine] ExpGrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 5: projected gradient descent on exact max
    if best_seq is not None and time.time() < deadline - 30:
        print("[Refine] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=500_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Refine] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase 6: long simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 10:
        print("[Refine] Long simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len,
            deadline, max_moves=2_000_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Refine] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = float("inf")

    # Clip to allowed range and final evaluation
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
