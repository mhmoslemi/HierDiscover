def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = len(a)-1
    return corr[start:start+len(a)]

def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)
```

def shape_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    s = np.exp(-0.5 * ((i - center) / (sigma * n))**2)

"""
Hybrid optimizer for the self‑convolution maximal‑value problem.

The routine works in several stages:
1.   Small parametric sweep (sine‑power, beta‑power, Gaussian, exponential tails
     and Dirichlet draws) to obtain a good starting distribution.
2.   Adam optimisation on a smooth‑max surrogate of the max‑convolution.
3.   Exponentiated‑gradient sub‑gradient descent on the exact max.
4.   Projected gradient descent (exact max‑convolution).
5.   Spectral‑flattening (force FFT magnitude to be (almost) constant).
6.   Simulated‑annealing local refinement.

All iterations respect the global time budget (≈ 1000 s) and the best
sequence found so far is always returned.  The code uses only NumPy
(and the already‑imported `evaluate_sequence` for the final scoring)
and runs on a single thread.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for sub‑gradient / gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.2                # initial temperature for smooth‑max
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to get current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # decay learning rate a bit every 100 k iters
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the current max
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Force the FFT magnitude of the vector to be (almost) constant while keeping
    the total mass (DC term) unchanged, projecting back onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")

        # ---- spectral flattening step ----
        A = np.fft.fft(a)                       # full complex FFT (size n)
        energy = np.dot(a, a)                   # ||a||_2^2

        # Preserve DC (sum = 1) and spread remaining energy uniformly.
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                         # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45   # up‑to‑45 % of the mass at i
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Parametric shape generators -----------------------------------------

def _shape_sine_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    s = np.sin(np.pi * i / n) ** p
    s = np.maximum(s, 0.0)
    s_sum = s.sum()
    if s_sum <= 0:
        return np.full(n, 1.0 / n)
    return s / s_sum


def _shape_beta_power(n: int, g: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    left = i
    right = n + 0.5 - i
    s = (left * right) ** g
    s = np.maximum(s, 0.0)
    s_sum = s.sum()
    if s_sum <= 0:
        return np.full(n, 1.0 / n)
    return s / s_sum


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    s = np.exp(-0.5 * ((i - centre) / (sigma * n)) ** 2)
    s = np.maximum(s, 0.0)
    s_sum = s.sum()
    if s_sum <= 0:
        return np.full(n, 1.0 / n)
    return s / s_sum


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = np.exp(-gamma * (n - 1 - i))
    s = left + right
    s = np.maximum(s, 0.0)
    s_sum = s.sum()
    if s_sum <= 0:
        return np.full(n, 1.0 / n)
    return s / s_sum


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # Length of the vector – 8192 offers more freedom than 4096,
    # while FFT‑based convolutions stay cheap.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (identical to evaluate_sequence for
    # non‑clipped, non‑negative vectors with sum > 0).
    def _eval(a: np.ndarray) -> float:
        s = a.sum()
        if s < 0.01:
            return np.inf
        conv = _full_convolution(a, conv_fft_len)
        maxb = conv.max()
        return float(2.0 * n * maxb / (s * s))

    best_seq: np.ndarray | None = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1. Small parametric sweep – cheap but often finds a very good start.
    print("[Sweep] exploring parametric families")
    # sine^p
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        a = _shape_sine_power(n, p)
        val = _eval(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] sine^p={p:.2f} → {val:.6f}")

    # beta‑power (product power) – symmetric “U‑shaped” family.
    for g in (-1.5, -1.0, -0.5, 0.0, 0.5, 1.0, 1.5):
        a = _shape_beta_power(n, g)
        val = _eval(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] beta_pow={g:.2f} → {val:.6f}")

    # Gaussian bumps (single centred peak)
    for sigma in (0.05, 0.08, 0.12, 0.2, 0.3):
        a = _shape_gaussian(n, sigma)
        val = _eval(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] gauss_sigma={sigma:.2f} → {val:.6f}")

    # Symmetric exponential tails
    for gamma in (0.1, 0.2, 0.4, 0.6, 0.8):
        a = _shape_exponential(n, gamma)
        val = _eval(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] expo_gamma={gamma:.2f} → {val:.6f}")

    # Random Dirichlet draws (different concentrations)
    for alpha in (0.1, 0.3, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        val = _eval(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] Dirichlet_alpha={alpha:.2f} → {val:.6f}")

    # Ensure feasibility
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = best_seq / best_seq.sum()

    # ------------------------------------------------------------------
    # 2. Adam optimisation on a smooth‑max surrogate.
    if time.time() < deadline:
        print("[Phase] Adam optimisation (smooth‑max)")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=800_000,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase] Adam improved to {val_adam:.6f}")

    # 3. Exponentiated‑gradient sub‑gradient on the exact max.
    if time.time() < deadline:
        print("[Phase] Sub‑gradient (exp‑grad) optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=800_000, eta0=1.0,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase] Sub‑gradient improved to {val_sub:.6f}")

    # 4. Projected gradient descent on the exact max‑convolution.
    if time.time() < deadline:
        print("[Phase] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=400_000,
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Phase] GD improved to {val_gd:.6f}")

    # 5. Spectral flattening – forces the spectrum to be (almost) flat.
    if time.time() < deadline:
        print("[Phase] Spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_iters=800_000,
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Phase] Spectral improved to {val_spec:.6f}")

    # 6. Simulated‑annealing refinement.
    if time.time() < deadline:
        print("[Phase] Simulated annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=300_000,
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final safety: clip to allowed range and report.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
