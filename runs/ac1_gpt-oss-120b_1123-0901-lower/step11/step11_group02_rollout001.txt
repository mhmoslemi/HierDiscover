"""
Propose a step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

under non‑negativity and sum>0.  The program works in two stages:

1. **Coarse search (n=4096).**  
   A rich pool of analytically‑parameterised shapes (Beta,
   exponential, Gaussian, power‑law, spikes, etc.) together with a warm‑start
   (if available) is generated.  Each candidate is normalised to sum = 1 and
   its objective (computed via an FFT convolution) is recorded.  The best
   few candidates are kept.

2. **Fine optimisation (n=8192).**  
   The best coarse candidate is interpolated to a finer grid.  A long
   exponentiated‑gradient (sub‑gradient) phase minimises a smooth surrogate
   of the maximum convolution, followed by a *mass‑transfer* local search
   that moves a tiny amount of mass from the coordinate with the largest
   sub‑gradient to the one with the smallest.  Finally a brief simulated‑
   annealing pass is performed.  All steps respect the 1000 s budget and
   keep the iterate on the probability simplex.

The final vector is clipped to the evaluator’s allowed range and returned
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Candidate factories (all output sums to 1)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric or asymmetric Beta pdf on mid‑points of a uniform grid."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a

# Optimisers (sub‑gradient + mass‑transfer + simulated annealing)

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient minimisation of the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of mass from the coordinate with the largest
    sub‑gradient to the one with the smallest, accepting only strict
    improvements.  Simple but very effective for fine‑grained polishing.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        # stop if gradients are essentially equal
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        src_mass = a[i_high]
        improved = False
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break

        if not improved:
            # occasional random jitter to escape plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Brief simulated‑annealing pass on the exact objective."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

        print("[Fine] Exponentiated‑gradient optimisation start")
        a_exp, val_exp = _subgradient_optimize_expgrad(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            sub_dead, max_iters=800_000, eta0=0.65,
        )
        if val_exp < _score_big(a_big):
            a_big = a_exp
            print(f"[Fine] ExpGrad finished with best {val_exp:.6f}")
        else:
            print("[Fine] ExpGrad made no improvement")
    else:
        print("[Fine] Skipping ExpGrad due to time")

    # ------------------------------------------------------------------
    # 4) Mass‑transfer fine‑grained local search (the most effective polish)

    if time.time() < deadline - 6:
        print("[MassTransfer] Fine‑grained polishing start")
        mt_dead = deadline - 5.0   # leave a few seconds for annealing & finalisation
        a_mt, val_mt = _mass_transfer_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            mt_dead, max_iters=300_000,
        )
        if val_mt < _score_big(a_big):
            a_big = a_mt
            print(f"[MassTransfer] new best {val_mt:.6f}")
        else:
            print("[MassTransfer] no improvement")
    else:
        print("[MassTransfer] Skipped (time)")

    # ------------------------------------------------------------------
    # 5) Very short simulated‑annealing pass (helps escape possible plateaus)

    if time.time() < deadline - 2:
        print("[Anneal] short simulated‑annealing")
        ann_dead = deadline - 1.0
        a_ann, val_ann = _anneal_optimize(
            a_big, n_big, conv_fft_len_big, ann_dead, max_moves=80_000,
        )
        if val_ann < _score_big(a_big):
            a_big = a_ann
            print(f"[Anneal] improved to {val_ann:.6f}")
        else:
            print("[Anneal] no improvement")
    else:
        print("[Anneal] skipped (time)")

    # ------------------------------------------------------------------
    # Final clipping to the evaluator’s allowed range
    best_seq = np.clip(a_big, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Final exact evaluation (for logging)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
