"""
Hybrid optimisation for the step‑function attack.

The routine works on the probability simplex (∑a_i = 1) – the evaluation
function is scale‑invariant, so this does not affect the score.
We use a longer sequence (n = 8192) which gives the optimiser more
degrees of freedom.  A diverse pool of seed vectors is generated
(uniform, Dirichlet draws, exponential / Gaussian families, a
warm‑start if one exists, and a few random‑spike patterns).  Each seed
is first optimised with Adam on a smooth‑max surrogate of the
self‑convolution, then refined with an exact‑max exponentiated‑gradient
(sub‑gradient) phase.  The best sequence is polished with a long
exp‑gradient run until the time budget (≈1000 s) expires.  All
operations stay on the simplex, respect the non‑negativity / range
limits of ``evaluate_sequence`` and finally return a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories (simple parametric families)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, expo)
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_random_spikes(n, k):
    """k random locations, equal weight."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_half_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_laplacian(n, gamma):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-gamma * np.abs(i - center))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5                # starting temperature
    lam_factor = 1.04

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max (guarantees non‑negative, Σ = 1)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c               # denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 150_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient optimisation."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)       # multiplicative update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a longer sequence (n=8192) and a two‑phase hybrid optimiser
    (Adam + exponentiated‑gradient).  Returns the best sequence found
    before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 points give a larger feasible space than before
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)      # = 16384
    corr_fft_len = _next_pow2(3 * n - 2)      # = 32768

    # ------------------------------------------------------------------
    # Warm start (if a previous run produced a good sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # Upsample to the new length by simple repetition
        if warm.shape[0] < n:
            repeat = int(np.ceil(n / warm.shape[0]))
            warm = np.tile(warm, repeat)[:n]
        else:
            warm = warm[:n]
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of initial vectors (kept modest to fit the time budget)
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 8, n // 4, n // 2):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws – wide range of concentration
    for alpha in (0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential tails
    for gamma in (0.05, 0.1, 0.2, 0.4, 0.8, 1.2):
        pool.append(_make_exponential(n, gamma))

    # Gaussian families (different widths)
    for sigma in (0.02, 0.05, 0.1, 0.2):
        pool.append(_make_gaussian(n, sigma))

    # Triangle, half‑sine, cosine
    pool.append(_make_triangle(n))
    pool.append(_make_half_sine(n))
    pool.append(_make_cosine(n))

    # Beta‑distribution shapes
    for a, b in ((0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (0.2, 2.0)):
        pool.append(_make_beta(n, a, b))

    # Laplacian tails
    for gamma in (0.02, 0.05, 0.1):
        pool.append(_make_laplacian(n, gamma))

    # Power‑law tails (negative exponents)
    for expo in (-0.5, -1.0, -1.5, -2.0):
        pool.append(_make_powerlaw(n, expo))

    # Sparse random spikes (different sparsities)
    for k in (64, 128, 256, 512):
        pool.append(_make_random_spikes(n, k))

    # A handful of completely random Dirichlet draws (different alphas)
    for _ in range(8):
        alpha = 10 ** np.random.uniform(-3, 2)   # between 1e‑3 and 100
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    total_candidates = len(pool)
    cand_idx = 0

    # ------------------------------------------------------------------
    # Phase I – optimise a number of promising seeds
    while time.time() < deadline and cand_idx < total_candidates:
        # Keep a small reserve for the final polishing phase
        if deadline - time.time() < 120.0:
            break

        a0 = pool[cand_idx]
        cand_idx += 1

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # allocate a time slice for this candidate (≈ 30‑40 s)
        remaining = deadline - time.time()
        slice_time = min(38.0, remaining * 0.25)
        sub_dead = time.time() + slice_time

        # ---- Adam optimisation (smooth‑max) ----
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=800_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cand_idx-1}] Adam improved to {val_adam:.6f}")

        # ---- Exponentiated‑gradient sub‑gradient (exact max) ----
        if time.time() < deadline - 30.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=1_500_000, eta0=0.85,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Cand {cand_idx-1}] ExpGrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – final deep polishing on the best candidate
    if best_seq is not None and time.time() < deadline - 20.0:
        print("[Final polish] Long exponentiated‑gradient run")
        a_final, val_final = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=4_000_000, eta0=0.7,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Ensure a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # expose for a possible next run
    globals()["height_sequence_2"] = best_seq.copy()
    return best_seq.tolist()
