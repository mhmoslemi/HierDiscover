"""
Hybrid optimizer for the step‑function attack (enhanced).

The algorithm builds a diverse pool of initial probability vectors,
performs a fast L2‑flattening phase (minimise variance of the
self‑convolution), then a smooth‑max Adam phase, an exact
sub‑gradient phase, an aggressive LP‑based “mass‑balancing’’ step
with a line‑search for the mixing ratio, short projected‑gradient
descent, and a final simulated‑annealing polish.
All iterates stay non‑negative and are normalised to sum = 1,
so the homogeneous evaluation function is unchanged.

The new ingredients over the previous state‑of‑the‑art are:
* an L2‑flattening surrogate (conv → constant) that drives the whole
  convolution pattern towards uniformity before attacking the peak;
* a larger active set in the LP improvement (all indices within
  5 % of the current maximum);
* a simple line‑search to pick the best mixing factor τ when mixing
  the current vector with the LP‑solution;
* a broader pool of parametrised candidates (arcsine, power‑law,
  exponential tails, sparse spikes) and a larger number of
  “top‑K’’ starts (8 instead of 5).

The routine respects the 1000 s budget and always returns the best
sequence found.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _obj_value(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a★a) (sum(a) is assumed to be 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# 1.  L2‑flattening optimizer (Adam on ∑(conv−mean)²)
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the L2‑distance of the self‑convolution
    to its global mean (i.e. flattens the whole spectrum).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    # the target mean of the convolution (sum(a)=1) is 1/(2n‑1)
    target_mean = 1.0 / (2 * n - 1)

    best_a = a0.copy()
    best_val = _obj_value(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- current a from soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution and L2 loss -----
        conv = _full_convolution(a, conv_fft_len)
        w = conv - target_mean               # gradient weight vector (sums to zero)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- record best (according to the true objective) -----
        val = _obj_value(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2 Adam it={it:06d}] new best {val:.6f}")

        # ----- learning‑rate decay -----
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# 2.  Smooth‑max Adam optimiser (identical to the one from the reference)
def _adam_optimize_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)            # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _obj_value(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Max Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# 3.  Exact sub‑gradient (exponentiated‑gradient) optimiser
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _obj_value(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# 4.  Projected gradient descent on the exact max‑convolution
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# 5.  Simulated‑annealing polishing
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val, best_a, a, conv = val2, a2.copy(), a2, conv2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, conv = a2, conv2

        it += 1

    return best_a, best_val

# 6.  Aggressive LP improvement with line‑search for τ
def _lp_improve_enhanced(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 12,
    tau_init: float = 0.5,
    rel_tol: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    Uses a broader active set (all indices within rel_tol·M) and
    a line‑search over mixing ratios τ ∈ {0.1,0.25,0.5,0.75}.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _obj_value(a, n, conv_fft_len)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # Active constraints: indices whose value is close to the current peak.
        eps = rel_tol * M
        active = np.where(conv_a >= M - eps)[0]
        # add a few well‑spread guard indices (helps numerical stability)
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # ---- build sparse constraint matrix A (|active| × n) ----
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # ---- linear program: maximise Σ b  <=> minimise -Σ b ----
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={
                "presolve": True,
                "time_limit": max(0.0, deadline - time.time())
            },
        )

        if not lp_res.success:
            print(f"[LP {it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # ---- line‑search for mixing factor τ ----
        best_local_val = best_val
        best_local_a = best_a
        for tau in (0.1, 0.25, 0.5, 0.75):
            a_new = (1.0 - tau) * a + tau * b_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            val_new = _obj_value(a_new, n, conv_fft_len)
            if val_new < best_local_val:
                best_local_val = val_new
                best_local_a = a_new.copy()
        if best_local_val < best_val - 1e-9:
            best_val, best_a = best_local_val, best_local_a.copy()
            a = best_local_a
            print(f"[LP {it}] improved to {best_val:.6f}")
        else:
            # if no improvement, shrink τ for future iterations
            tau_init *= 0.5
            if tau_init < 1e-4:
                break

    return best_a, best_val

# 7.  Candidate generation (diverse families)
def _shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like shape a_i ∝ 1/√( (i+½)(n‑i‑½) )."""
    i = np.arange(n) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    a = a / a.sum()
    return a


def _shape_power(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i∝(i+1)^p·(n‑i)^p (p<0 gives edge‑heavy)."""
    i = np.arange(n)
    a = (i + 1) ** p * (n - i) ** p
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential tails."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    a = left + right
    a = a / a.sum()
    return a


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates: equally‑spaced spikes and random spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """A modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with different concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Symmetric exponential tails
    for gamma in (0.2, 0.5, 1.0, 2.0):
        cand.append(_shape_exponential(n, gamma))

    # Power‑law edge‑heavy shapes
    for p in np.linspace(-2.0, -0.5, 5):
        cand.append(_shape_power(n, p))

    # Gaussian bumps at the ends
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        a = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
            + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        a = a / a.sum()
        cand.append(a)

    # Pure sine wave (positive half)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    cand.append(a)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Discrete arcsine
    cand.append(_shape_arcsine(n))

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Main entry point – returns a non‑negative step function."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size (trade‑off between expressiveness and speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous construction is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] > n:
            warm_arr = warm_arr[:n]
        elif warm_arr.shape[0] < n:
            warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() >= 0.01:
            warm_arr = warm_arr / warm_arr.sum()
        else:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Build a rich pool of candidates
    pool = _initial_candidates(n, warm_arr) + _spike_candidates(n)

    # some extra random Dirichlet draws
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    # random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    # ------------------------------------------------------------------
    # Evaluate all candidates quickly and keep the best few
    evals = []
    for a in pool:
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _obj_value(a, n, conv_fft_len)
        evals.append((val, a))
    evals.sort(key=lambda x: x[0])
    top_k = 8
    best_seq = None
    best_val = float("inf")
    for i in range(min(top_k, len(evals))):
        val, a = evals[i]
        if val < best_val:
            best_val, best_seq = val, a.copy()
    print(f"[Init] best of pool: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – L2 flattening (fast Adam)
    if time.time() < deadline - 30:
        a_l2, val_l2 = _adam_optimize_l2(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=180_000
        )
        if val_l2 < best_val:
            best_val, best_seq = val_l2, a_l2.copy()
            print(f"[Phase I – L2] improved to {val_l2:.6f}")

    # ------------------------------------------------------------------
    # Phase II – smooth‑max Adam (focus on the peak)
    if time.time() < deadline - 30:
        a_max, val_max = _adam_optimize_max(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=240_000
        )
        if val_max < best_val:
            best_val, best_seq = val_max, a_max.copy()
            print(f"[Phase II – Max Adam] improved to {val_max:.6f}")

    # ------------------------------------------------------------------
    # Phase III – exact sub‑gradient (exponentiated‑gradient)
    if time.time() < deadline - 20:
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=260_000, eta0=0.8
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase III – Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – LP‑based refinement (aggressive)
    if time.time() < deadline - 15:
        a_lp, val_lp = _lp_improve_enhanced(
            best_seq, n, conv_fft_len,
            deadline, max_iter=12, tau_init=0.5, rel_tol=0.05
        )
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[Phase IV – LP] improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase V – projected gradient descent on the exact max‑conv.
    if time.time() < deadline - 10:
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=120_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Phase V – GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase VI – final simulated‑annealing polish
    if time.time() < deadline - 5:
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len,
            deadline, max_moves=60_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase VI – Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
