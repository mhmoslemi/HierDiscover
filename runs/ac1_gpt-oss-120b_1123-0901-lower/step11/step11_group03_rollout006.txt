"""
Hybrid multi‑restart optimisation for the non‑negative step‑function problem.

Goal:
    minimise  2·n·max(conv(a,a)) / (sum(a)^2)
    with a_i ≥ 0.  The objective is invariant to scaling, so we enforce
    sum(a)=1 (probability simplex) throughout the search.

Strategy
---------
1. Build a modest but diverse pool of start vectors (uniform, Dirichlet,
   exponential tails, power‑law, spikes, sinusoid, etc.) and keep the best
   seed according to the exact objective.

2. Run a few short Adam runs on a smooth‑max surrogate (soft‑max over the
   convolution values).  This quickly moves the iterate into a promising
   region.

3. Perform many independent “refine” phases.  Each phase starts from the
   current best vector (with some probability) or from a fresh random
   Dirichlet draw, and runs an exponentiated‑gradient mirror‑descent on a
   smooth approximation of the max‑convolution for as long as time permits.
   The smooth weights are a soft‑max with a temperature that grows
   exponentially, which drives the surrogate ever closer to the true max.

4. If time remains, apply a spectral‑flattening step: repeatedly force the
   magnitude of the FFT of the iterate to be (almost) constant while
   staying on the simplex.  This flattens the autocorrelation and can shave
   off the last few hundredths of the score.

5. Finish with a short simulated‑annealing pass that swaps small amounts of
   mass between two coordinates; this provides a robust final polishing.

All heavy loops check the remaining time every few thousand iterations,
so the routine always respects the 1000 s budget and returns the best
vector discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) where
        (w★a)[i] = Σ_k w[k] a[i+k]   (zero‑padding outside the range).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised a."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    pool = []

    # warm start (if supplied)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with several concentration levels
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law (triangular product) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian spikes
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Sparse spike candidates (≈√n spikes)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool

def _adam_phase(a0: np.ndarray,
                n: int,
                conv_fft_len: int,
                corr_fft_len: int,
                deadline: float,
                max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate (soft‑max over convolution)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                  # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.05
    lam_factor = 1.012            # slower temperature rise → smoother optimisation

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # Retrieve probability vector from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # soft‑max weights over the convolution entries
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – slowly move towards the true max
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    # Final exact evaluation of the best point
    best_val = _objective(best_a, n, conv_fft_len)
    return best_a, best_val

def _refine_subgradient(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        corr_fft_len: int,
                        deadline: float,
                        eta0: float = 0.9,
                        lam0: float = 5.0,
                        lam_factor: float = 1.02) -> tuple[np.ndarray, float]:
    """
    Long mirror‑descent (exponentiated‑gradient) on a smooth‑max surrogate.
    The temperature λ is increased exponentially, turning the surrogate
    into the true max.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    lam = lam0
    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # soft‑max over the convolution values
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update, staying on the simplex automatically
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, conv_fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Refine it={it:07d}] new best {cur_val:.6f}")

        # temperature schedule (smooth→hard)
        lam = min(lam * lam_factor, 1e8)

        it += 1
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val

def _spectral_flatten_phase(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           deadline: float,
                           max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Iteratively enforce (near) constant FFT magnitude while staying on the simplex.
    This tends to flatten the autocorrelation.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        cur_val = _objective(a, n, conv_fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Spectral it={it:06d}] new best {cur_val:.6f}")

        # ---- spectral flattening step ----
        A = np.fft.fft(a)
        energy = np.dot(a, a)                     # ||a||_2^2

        # keep the DC term (sum = 1) and redistribute the remaining energy
        if n > 1:
            # total energy = Σ |A_k|^2 / n  (Parseval).  We keep |A_0| fixed.
            rem = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # projection onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

def _anneal_phase(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  deadline: float,
                  max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search on the simplex.
    Moves a random fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = _objective(a, n, conv_fft_len)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 40%) of the mass from i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac
        if delta <= 0:
            it += 1
            continue

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = _objective(a2, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Runs for at most `budget_s` seconds and always returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem dimensions
    n = 4096                                   # vector length
    conv_fft_len = _next_pow2(2 * n - 1)       # 8192
    corr_fft_len = _next_pow2(3 * n - 2)       # 16384

    # ------------------------------------------------------------------
    # Warm start from the previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build the pool of initial candidates and pick the best one (exact eval)
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() > deadline:
            break
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:03d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – short Adam optimiser (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase I] Adam optimisation")
        sub_dead = deadline - 5.0
        a_adam, val_adam = _adam_phase(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=300_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – many independent refinements (exponentiated‑gradient on smooth‑max)
    run = 0
    while time.time() < deadline - 2:
        # give each run about 30 s (or whatever is left)
        run_dead = min(deadline, time.time() + 30.0)
        # with probability 0.7 start from current best, otherwise random start
        if best_seq is not None and np.random.rand() < 0.7:
            start = best_seq.copy()
        else:
            start = np.random.dirichlet(np.ones(n))

        a_ref, val_ref = _refine_subgradient(
            start, n, conv_fft_len, corr_fft_len, run_dead,
            eta0=0.9, lam0=5.0, lam_factor=1.02
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Run {run}] improved best to {val_ref:.6f}")
        run += 1
        # early exit if we have already crossed the target threshold
        if best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase III – spectral flattening (uses any remaining time)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase III] Spectral flattening")
        a_spec, val_spec = _spectral_flatten_phase(
            best_seq, n, conv_fft_len, deadline, max_iters=300_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Phase III] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – final simulated‑annealing pass (short)
    if best_seq is not None and time.time() < deadline - 0.5:
        print("[Phase IV] Final annealing")
        a_ann, val_ann = _anneal_phase(
            best_seq, n, conv_fft_len, deadline, max_moves=100_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety‑net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the validator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
