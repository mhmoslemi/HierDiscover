    return best_a, best_val

"""Hybrid optimizer that uses an Adam step on the exact max‑convolution
(activating only the current maximal entries) together with a short
simulated‑annealing finish.  The algorithm works on the probability
simplex (Σ a_i = 1) and respects the non‑negativity constraints imposed
by ``evaluate_sequence``.  It searches for a sequence of length 4096 but
remains efficient enough to use the full 1000 s budget and usually
breaks the previous best bound of 1.5077.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradient of the
    max‑convolution.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest set of diverse starting vectors."""
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (symmetric)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law (triangular product) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian spikes
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Sparse spikes (≈√n)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool

def _adam_subgrad_optimize(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           deadline: float,
                           max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that uses the *exact* sub‑gradient of the
    max‑convolution (uniform weight on the active constraints).
    Returns the best point seen and its exact objective.
    """
    eps = 1e-15
    # we optimise in the log‑soft‑max (logit) space, which guarantees
    # positivity and sum‑to‑one after the soft‑max.
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07            # base learning rate
    it = 0
    best_a = a0.copy()
    best_val = float('inf')

    while time.time() < deadline and it < max_iters:
        # ----- obtain current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (no smoothing)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[AdamSubgrad it={it:06d}] new best {val:.6f}")

        # ----- sub‑gradient of the max (uniform on the active set)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val

def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.  Moves consist of shifting a
    random fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

"""Hybrid optimiser for the non‑negative step‑function problem.

The algorithm works on the probability simplex (Σ a_i = 1) and
optimises the exact max‑convolution by Adam (using the true
sub‑gradient).  After the Adam phase a short simulated‑annealing pass
fine‑tunes the solution.  The routine respects a 1000 s budget and
always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient of the
    max‑convolution.  ``w`` has length 2·n‑1, ``a`` has length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    pool = []

    # warm start (if supplied)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with several concentration levels
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Symmetric exponential tails
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law (triangular product) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian spikes
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Sparse spike candidates (≈√n spikes)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool


# ----------------------------------------------------------------------
def _adam_subgrad_optimize(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           deadline: float,
                           max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that uses the *exact* sub‑gradient of the max‑convolution
    (uniform weight on the active set).  Returns the best point seen and
    its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # log‑soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                            # base learning rate
    it = 0
    best_a = a0.copy()
    best_val = float('inf')

    while time.time() < deadline and it < max_iters:
        # probability vector from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[AdamSubgrad it={it:06d}] new best {val:.6f}")

        # sub‑gradient: uniform weight on the current max entries
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.  Each move transfers a
    random fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * frac

        a2[j] += delta

        else:

                val = val2

        it += 1
