"""
Propose a low‑score step‑function.

The evaluation metric is
    2·n·max(conv(a,a)) / (sum(a)²)   with a_i ≥ 0 .
Because the ratio is scale‑invariant we work on the probability simplex
∑a_i = 1.  For a vector of length n the theoretical lower bound is
≈2·n/(2·n‑1) ≃ 1, so values around 1.5 are still far from optimal.

Algorithm
---------
1. **Rich initial pool** – several analytic windows (Hann, Hamming,
   Blackman, Kaiser, Gaussian, exponential tails, power‑law, half‑wave
   sinusoid) together with a few random Dirichlet draws.  All candidates
   are projected onto the simplex and the exact objective (2·n·max(conv))
   is evaluated; the best candidate is kept.

2. **Long exponentiated‑gradient (mirror‑descent) phase** – we minimise the
   exact L∞‑norm of the convolution using the sub‑gradient
   `grad = 2·cross_correlation(w,a)` where `w` is a uniform distribution
   over the currently active lag(s).  The update
   `a ← a·exp(‑η·grad)` keeps the iterate non‑negative; after each step we
   renormalise onto the simplex.  The step size η decays as η₀/√t.
   The whole remaining time budget (≈ 1000 s) is spent here, continuously
   remembering the best vector seen.

3. **Optional simulated‑annealing polishing** – if a few seconds remain a
   cheap local‑search moves a random fraction of mass between two
   coordinates; Metropolis acceptance with a linear cooling schedule
   refines the result.

All heavy loops poll the wall‑clock after every 1 000 iterations, so the
routine always returns before the 1000 s deadline.  The final vector is
clipped to the validator’s allowed range and returned as a plain Python
list.
"""

import time
import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _eval_val(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective value 2·n·max(conv(a,a)) (sum(a)=1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          fft_len: int,
                          deadline: float,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:

    Exponentiated‑gradient (mirror descent) minimising the exact max‑convolution.
    Returns the best vector seen and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _eval_val(a, n, fft_len)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set (indices attaining the current maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        # step size – diminishing √t schedule
        eta = eta0 / np.sqrt(it + 1.0)
        # limit exponent to avoid overflow
        step = np.clip(eta * grad, -50.0, 50.0)

        a = a * np.exp(-step)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float,
                     moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Plain simulated‑annealing on the simplex.  Moves a random fraction of mass
    from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur = 2.0 * n * conv.max()
    best_val = cur
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    for it in range(moves):
        if time.time() >= deadline:
            break
        T = max(T_end, T_start * (1.0 - it / float(moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            cur = val2
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - cur) / T):
                a = a2
                cur = val2

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Runs for at most `budget_s` seconds and always returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # Problem size – using a smaller n speeds up the search and still
    # allows us to beat the target 1.503.
    n = 2048
    fft_len = _next_pow2(2 * n - 1)               # 4096

    # ------------------------------------------------------------------
    # 1) Build a diverse set of analytic windows and random seeds.
    pool = []

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # half‑wave sinusoid (already good baseline)
    pool.append(np.sin(np.pi * (np.arange(n) + 0.5) / n))

    # Hann window
    pool.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # Hamming window
    pool.append(0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # Blackman window
    pool.append(
        0.42
        - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1))
    )

    # Kaiser windows for several beta values
    for beta in (2.0, 4.0, 6.0, 8.0):
        i = np.arange(n, dtype=np.float64)
        arg = (2.0 * i / (n - 1) - 1.0)
        shape = i0(np.pi * beta * np.sqrt(1.0 - arg ** 2)) / i0(np.pi * beta)
        pool.append(shape)

    # Gaussian windows with a few σ values
    for sigma in (0.12, 0.18, 0.24):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
        pool.append(shape)

    # Exponential tails (both sides)
    for gamma in (0.15, 0.30, 0.60):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        pool.append(shape)

    # Power‑law (triangular product) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        pool.append(shape)

    # Random Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Sparse spike candidates (≈√n spikes)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        pool.append(a)

    # ------------------------------------------------------------------
    # 2) Evaluate the pool and keep the best start point.
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _eval_val(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:03d} → val={val:.6f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3) Long exponentiated‑gradient (sub‑gradient) optimisation.
    if best_seq is not None and time.time() < deadline - 2.0:
        print("[Phase I] Sub‑gradient optimisation started")
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, fft_len, deadline, eta0=0.8
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase I] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional tiny annealing pass (only if a few seconds are left).
    if best_seq is not None and time.time() < deadline - 0.5:
        print("[Phase II] Simulated‑annealing polish")
        a_ann, val_ann = _anneal_optimize(best_seq, n, fft_len, deadline)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase II] polished best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # 5) Safety – guarantee a non‑empty return.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Clip to the validator’s allowed numeric range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
