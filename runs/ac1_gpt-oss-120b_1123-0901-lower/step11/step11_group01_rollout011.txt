def _lp_minmax_cuts(a, n, conv_fft_len, deadline, max_cuts=30, tolerance=1e-12):
    # compute initial conv
    conv = _full_convolution(a, conv_fft_len)
    M = conv.max()
    # initial active_set: indices where conv near M
    active_set = list(np.where(conv >= M - tolerance)[0])
    # also add 0, 2n-2 maybe
    active_set += [0, 2*n-2]
    active_set = sorted(set(active_set))
    best_M = M
    best_b = a.copy()  # fallback

    while time.time() < deadline and len(active_set) <= max_cuts + 10:
        # Build LP
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active_set):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # j = i_start..i_end ; j corresponds to b index
            j_vals = np.arange(i_start, i_end+1, dtype=int)
            # coefficients a_i where i = k - j
            coeffs = a[k - j_vals]
            rows.extend([row_idx] * len(j_vals))
            cols.extend(j_vals.tolist())
            data.extend(coeffs.tolist())
        # Add column for M variable with -1 coefficient
        # shape rows x (n+1)
        # We'll create sparse matrix with shape (len(active_set), n+1)
        M_col = np.full(len(active_set), -1.0)
        # Append M column to data
        # Build as csr matrix
        # Prepare row indices repeated length of each row for M?
        # Actually easiest: create a sparse matrix for b variables, then later hstack with M column
        A_b = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_set), n))
        # Append M column:
        # Create column as dense column vector; use scipy.sparse.hstack
        M_col_sparse = scipy.sparse.csr_matrix(M_col).reshape(-1, 1)
        A_ub = scipy.sparse.hstack([A_b, M_col_sparse])
        # Equality: sum(b) = 1
        A_eq = scipy.sparse.csr_matrix(np.ones((1, n+1)))
        A_eq[0, -1] = 0  # M variable has coefficient 0

        c = np.zeros(n+1)
        c[-1] = 1.0  # minimize M

        bounds = [(0, None)] * n + [(0, None)]

        # b_ub = M * 1 for each active; but M is variable; we have <=M => row inequality is sum_i a_i * b_{k-i} - M <= 0 => rhs = 0
        b_ub = np.zeros(len(active_set))

        # Use linprog with high-s accuracy
        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=np.array([1.0]),
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())}
        )
        if not lp_res.success:
            print("[LP minmax] linprog failed:", lp_res.message)
            break

        x = lp_res.x
        b = x[:n]
        M_opt = x[-1]

        # Compute conv_ab for feasibility check
        conv_ab = np.convolve(a, b)[:2*n-1]
        max_ab = conv_ab.max()
        # If feasible (max_ab <= M_opt + tolerance), we have solution
        if max_ab <= M_opt + 1e-12:
            best_M = M_opt
            best_b = b.copy()
            # optional break if improvement small? break loop
            break

        # else, add violator index to active_set
        viol_idx = int(np.argmax(conv_ab))
        if viol_idx not in active_set:
            active_set.append(viol_idx)
        # else maybe add second violator as well to accelerate
        # Also possibly add others near viol_idx (neighbors) to stabilize

        # Continue loop

    return best_b, best_M

"""
Propose a step‐function sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search uses a *cutting‑plane linear program* that directly
minimises the peak of the convolution (instead of the previous
“maximise Σb’’ formulation).  Starting from a warm or random
distribution the LP iteratively adds the most violated convolution
constraint, solves a small LP (few tens of rows) and updates the
distribution.  A few lightweight sub‑gradient refinements are then
applied as a polish.  The whole routine respects the 1000 s budget
and always returns a non‑negative sequence.

The main steps are:

1. initialise a candidate on the probability simplex,
2. repeatedly call `_lp_minmax_cuts` to lower the maximal convolution,
3. optionally perform a few exponentiated‑gradient sub‑gradient steps,
4. keep the best sequence seen so far and return it.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _lp_minmax_cuts(a: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    deadline: float,
                    max_cuts: int = 40,
                    eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Cutting‑plane LP that finds b (∑b=1, b≥0) minimising the
    peak M of the convolution a★b :

        minimise   M
        subject to a★b ≤ M   (for every shift)
                  ∑b = 1
                  b ≥ 0 , M ≥ 0

    The LP starts with the currently active constraints (those attaining
    the peak of a★a) and adds the most violated constraint until feasibility
    is reached or `max_cuts` constraints have been added.
    """
    # initial convolution of a with itself
    conv_a = _full_convolution(a, conv_fft_len)
    M_cur = conv_a.max()
    # active set = indices where the peak is (with a small tolerance)
    active = set(np.where(conv_a >= M_cur - eps)[0].tolist())
    # a few well‑spread indices to keep the LP well‑posed
    extra = {0, len(conv_a) // 2, len(conv_a) - 1}
    active.update(extra)

    # main cutting‑plane loop
    for _ in range(max_cuts):
        if time.time() > deadline:
            break

        act_list = sorted(active)
        m = len(act_list)

        # Build sparse matrix for the b‑variables (size m × n)
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(act_list):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            j = np.arange(i_start, i_end + 1, dtype=int)               # b‑indices
            coeff = a[k - j]                                            # a‑coefficients
            rows.extend([row_idx] * len(j))
            cols.extend(j.tolist())
            data.extend(coeff.tolist())
        A_b = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))

        # Column for M (coefficient -1 in every inequality)
        Mcol = scipy.sparse.csr_matrix((-np.ones((m, 1))), shape=(m, 1))
        A_ub = scipy.sparse.hstack([A_b, Mcol])

        # Equality: sum(b) = 1  (M has coefficient 0)
        A_eq = np.ones((1, n + 1))
        A_eq[0, -1] = 0.0

        # Objective: minimise M
        c = np.zeros(n + 1)
        c[-1] = 1.0

        # Bounds: b_i ≥ 0,  M ≥ 0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        # Right‑hand side for inequalities is zero (a★b - M ≤ 0)
        b_ub = np.zeros(m)

        # Solve the LP
        res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=np.array([1.0]),
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())}
        )

        if not res.success:
            # LP failed – give up and return current a
            print("[LP‑cut] linprog failed:", res.message)
            return a, M_cur

        x = res.x
        b = x[:n]
        M_opt = max(x[-1], 0.0)          # enforce non‑negativity numerically

        # Verify feasibility on the full set of constraints
        conv_ab = np.convolve(a, b)[: 2 * n - 1]
        max_ab = conv_ab.max()
        if max_ab <= M_opt + eps:
            # feasible – we have reduced the peak
            return b, M_opt

        # otherwise add the most violated index to the active set and repeat
        viol_idx = int(np.argmax(conv_ab))
        active.add(viol_idx)

    # If we exit the loop without full feasibility, return the last feasible b
    # (it still improves the objective compared with the input a).
    return b, M_opt


def _lp_refine(a0: np.ndarray,
               n: int,
               conv_fft_len: int,
               deadline: float,
               max_outer: int = 6,
               max_cuts: int = 40) -> np.ndarray:
    """
    Apply a few rounds of the cutting‑plane LP to lower the convolution peak.
    """
    a = a0.copy()
    for it in range(max_outer):
        if time.time() > deadline:
            break
        b, M_new = _lp_minmax_cuts(a, n, conv_fft_len, deadline, max_cuts)
        # compare with the current peak
        cur_max = _full_convolution(a, conv_fft_len).max()
        if M_new < cur_max - 1e-14:
            a = b
        else:
            # no further improvement – stop early
            break
    return a


def _subgradient_expgrad(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_iters: int = 80_000,
                         eta0: float = 0.7) -> np.ndarray:
    """
    A short exponentiated‑gradient sub‑gradient phase (exact max‑convolution).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # sub‑gradient (uniform over all active indices)
        active = np.isclose(conv, max_c, atol=1e-14)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        # cross‑correlation of w with a (exact gradient of max_c)
        rev_a = a[::-1]
        fw = np.fft.rfft(w, _next_pow2(len(w) + len(a) - 1))
        fr = np.fft.rfft(rev_a, _next_pow2(len(w) + len(a) - 1))
        corr = np.fft.irfft(fw * fr)[:len(a)]
        grad = 2.0 * corr

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Return a non‑negative sequence (list of floats) that minimizes the
    given evaluation function.  The method uses a cutting‑plane LP
    (direct min‑max of the convolution) followed by a brief
    exponentiated‑gradient polishing step.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    # --------------------------------------------------------------
    # Length of the vector – a compromise between expressiveness and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Warm start (if a previous construction is available).
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a small pool of starters.
    starters = []
    if warm is not None:
        starters.append(warm.copy())
    # a few Dirichlet draws with different concentration parameters
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        starters.append(a)

    # a couple of deterministic “spike’’ patterns
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.linspace(0, n - 1, k).astype(int)
        a[pos] = 1.0 / k
        starters.append(a)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Main optimisation loop over the starters.
    for start_idx, a0 in enumerate(starters):
        if time.time() > deadline:
            break

        a = np.maximum(a0, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()

        # ---------- 1️⃣ Cutting‑plane LP refinement ----------
        sub_dead = min(deadline, time.time() + 25.0)   # budget per start
        a_lp = _lp_refine(a, n, conv_fft_len, sub_dead,
                          max_outer=5, max_cuts=35)

        # ---------- 2️⃣ Optional short sub‑gradient polish ----------
        if time.time() < deadline - 2.0:
            a_poly = _subgradient_expgrad(a_lp, n, conv_fft_len,
                                          deadline, max_iters=50_000)
        else:
            a_poly = a_lp

        # ---------- 3️⃣ Evaluate ----------
        val = evaluate_sequence(a_poly.tolist())
        if val < best_val:
            best_val, best_seq = val, a_poly.copy()
            print(f"[Candidate {start_idx}] new best {val:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
