"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full linear convolution.  The strategy is:

1. Build a large but cheap pool of parametric candidates
   (Beta, Gaussian, exponential, power‑law, sinusoid‑powers,
   mixtures, spikes, uniform, triangle, cosine).

2. Evaluate each candidate with a fast FFT‑based convolution and keep the
   best one.

3. Refine the best candidate with a deterministic “mass‑transfer”
   local search that repeatedly moves a fraction of mass from the index
   with the largest gradient (w.r.t. the active max‑convolution) to the
   index with the smallest gradient.  The move is accepted only if it
   strictly lowers the exact objective.  Randomised moves are added
   occasionally to escape plateaus.

4. If time permits a short projected‑gradient phase is run.

5. The final vector is clipped to the evaluator’s allowed range and
   returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -----------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using real FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories --------------------------------------------------

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric (or asymmetric) Beta pdf sampled on a uniform grid."""
    i = (np.arange(n) + 0.5) / n               # points in (0,1)
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


def _make_sinpow(n, p):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_cospow(n, p):
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_two_gaussians(n, sigma1=0.08, sigma2=0.08, shift=0.25, w1=0.5):
    """Mixture of two symmetric Gaussians placed symmetrically."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    g1 = np.exp(-0.5 * ((x - shift) / sigma1) ** 2)
    g2 = np.exp(-0.5 * ((x - (1 - shift)) / sigma2) ** 2)
    a = w1 * g1 + (1 - w1) * g2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective evaluation (fast, without the safety wrapper)

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective used by the optimiser (no clipping)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Improved mass‑transfer local optimiser ------------------------------

def _improved_mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic + stochastic mass‑transfer descent.
    Moves a fraction of mass from a coordinate with large gradient to one
    with small gradient, accepting only strict improvements.  Randomised
    selections and occasional jitter help escape plateaus.
    """
    a = a0.copy()
    cur_sum = a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13)
        # weight vector for the active (max) convolution indices
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        # gradient of max-convolution w.r.t. each a_i
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # pick source (high grad) and destination (low grad)
        if np.random.rand() < 0.2:
            # stochastic choice among the top/bottom 5
            high_idx = np.argpartition(grad, -5)[-5:]
            low_idx = np.argpartition(grad, 5)[:5]
            i_high = int(np.random.choice(high_idx))
            i_low = int(np.random.choice(low_idx))
        else:
            i_high = int(np.argmax(grad))
            i_low = int(np.argmin(grad))

        if i_high == i_low or a[i_high] < 1e-15:
            it += 1
            continue

        src_mass = a[i_high]
        improved = False

        # try a descending list of fractions
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03,
                     0.015, 0.007, 0.003, 0.001):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            # exact objective (sum unchanged, so denominator constant)
            val_new = _objective(a_new, n, conv_fft_len)
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[ImpMass it={it:06d}] new best {best_val:.6f}")
                break

        if not improved:
            # occasional random jitter to escape stagnation
            if np.random.rand() < 0.05:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j and a[i] > 1e-12:
                    eps = a[i] * 0.05
                    a[i] -= eps
                    a[j] += eps

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine --------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑evaluation step‑function within the time budget.
    Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------- problem size ---------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------- warm start ------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------- build candidate pool -------------------------
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet samples
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        cand.append(_make_dirichlet(n, alpha))

    # Exponential family
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0, 4.0):
        cand.append(_make_exponential(n, gamma))

    # Gaussian family
    for sigma in (0.04, 0.08, 0.12, 0.2, 0.4, 0.6):
        cand.append(_make_gaussian(n, sigma))

    # Power‑law (negative exponents give decreasing tails)
    for expo in (-3.0, -2.0, -1.5, -1.0, -0.7, -0.5):
        cand.append(_make_powerlaw(n, expo))

    # Simple shapes
    cand.append(_make_triangle(n))
    cand.append(_make_uniform(n))
    cand.append(_make_cosine(n))

    # Sinusoid power families
    for p in (0.4, 0.6, 0.8, 1.0, 1.5, 2.0, 3.0):
        cand.append(_make_sinpow(n, p))
        cand.append(_make_cospow(n, p))

    # Beta distributions (symmetric)
    for alpha in np.logspace(-0.3, 0.8, 12):   # ~0.5 … 6.3
        cand.append(_make_beta(n, alpha, alpha))

    # Asymmetric Beta (skewed)
    for a, b in [(0.3, 2.0), (0.2, 1.5), (1.5, 0.3), (2.0, 0.3)]:
        cand.append(_make_beta(n, a, b))

    # Two‑Gaussian mixtures
    for sigma1, sigma2, shift, w1 in [
        (0.08, 0.08, 0.25, 0.5),
        (0.06, 0.10, 0.20, 0.6),
        (0.10, 0.10, 0.30, 0.5),
    ]:
        cand.append(_make_two_gaussians(n, sigma1, sigma2, shift, w1))

    # Random Dirichlet draws (extra diversity)
    for _ in range(6):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        cand.append(_make_dirichlet(n, alpha))

    # ---------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # -------------------------- quick scan -----------------------------
    print("[Scan] evaluating %d candidates" % len(cand))
    for idx, a in enumerate(cand):
        if time.time() > deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a /= a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Scan #{idx}] new best {val:.6f}")

    # -------------------------- refinement -----------------------------
    # 1) mass‑transfer optimisation (most work)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Refine] improved mass‑transfer")
        a_mt, val_mt = _improved_mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[Refine] improved best to {val_mt:.6f}")

    # 2) a brief projected gradient descent as a safety net
    def _proj_grad_descent(a0, iters=120_000):
        a = a0.copy()
        best = a.copy()
        bestv = _objective(a, n, conv_fft_len)
        for it in range(iters):
            if time.time() > deadline:
                break
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-13)
            w = np.zeros_like(conv)
            if np.any(active):
                w[active] = 1.0 / active.sum()
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
            step = 0.12 / np.sqrt(it + 1.0)
            a = a - step * grad
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            val = 2.0 * n * max_c / (a.sum() ** 2)
            if val < bestv:
                bestv = val
                best = a.copy()
                print(f"[GD it={it:06d}] new best {val:.6f}")
        return best, bestv

    if best_seq is not None and time.time() < deadline - 3:
        print("[Refine] short projected gradient")
        a_gd, val_gd = _proj_grad_descent(best_seq, iters=80_000)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Refine] GD improved best to {val_gd:.6f}")

    # ---------------------------------------------------------------
    # Final safety: ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator's allowed interval
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (optional, for logging)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation error: {e}")

    return best_seq.tolist()
