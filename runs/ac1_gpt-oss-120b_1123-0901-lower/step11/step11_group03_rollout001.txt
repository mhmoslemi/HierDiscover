"""
Hybrid non‑negative step‑function optimizer.

Approach
--------
1. **Diverse initial seeds** – draw Dirichlet vectors with a wide range of
   concentration parameters, sort them in descending order (so that the
   reverse of the vector is in ascending order) and apply a random cyclic
   shift.  This pairing of large with small values minimises the central
   convolution term `a·rev(a)`.

2. **Quick evaluation** – keep the probability simplex (∑a = 1) throughout.
   The objective is `2·n·max(conv(a,a))`; we evaluate it directly with an
   FFT‑based convolution for speed.

3. **Optional gradient phase** – a short exponentiated‑gradient (mirror‑descent)
   optimisation is performed if time permits.  It often improves the
   initial candidates.

4. **Permutation‑based simulated annealing** – the best vector so far is
   refined by repeatedly swapping two entries (which keeps the simplex
   constraints automatically).  Swaps are accepted according to a cooling
   schedule; the current best and the overall best are tracked.

The routine respects the given time budget (default 1000 s) and always
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective_fast(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a vector that already sums to 1."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          fft_len: int,
                          deadline: float,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best vector encountered and its objective.
    """
    # start from a0 (already normalised)
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective_fast(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # exact convolution
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c

        # active set (indices where conv equals max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # gradient of max conv w.r.t. a (see derivation in original code)
        # cross‑correlation of w with a
        rev_a = a[::-1]
        fw = np.fft.rfft(w, fft_len)
        fr = np.fft.rfft(rev_a, fft_len)
        grad = 2.0 * np.fft.irfft(fw * fr, fft_len)[:n]

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (maintains positivity)
        a = a * np.exp(-eta * grad)

        # project back to simplex
        a = _project_simplex(a)

        # keep best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Subgrad] iter {it:07d} new best {best_val:.6f}")

        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑score non‑negative step‑function.
    A mixture of Dirichlet initialisation, a short mirror‑descent stage and
    a permutation‑based simulated‑annealing refinement is used.
    The function always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    n = 4096
    fft_len = _next_pow2(2 * n - 1)   # length for a*a convolution (8192)

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of starting vectors
    pool = []

    # Warm‑start from previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.shape[0] > n:
            a = a[:n]
        elif a.shape[0] < n:
            a = np.pad(a, (0, n - a.shape[0]))
        a = np.maximum(a, 0.0)
        if a.sum() >= 0.01:
            a = a / a.sum()
            pool.append(a.copy())

    # Dirichlet draws with a wide range of concentration parameters
    alphas = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    samples_per_alpha = 12
    for alpha in alphas:
        for _ in range(samples_per_alpha):
            a = np.random.dirichlet(alpha * np.ones(n))
            # sort descending → reverse will be ascending → minimise centre term
            a = np.sort(a)[::-1]
            # random cyclic shift to diversify the shape
            shift = np.random.randint(n)
            if shift:
                a = np.roll(a, shift)
            pool.append(a)

    # a few deterministic shapes (sinusoid, exponential tails, etc.)
    i = np.arange(n, dtype=np.float64)

    # half‑wave sinusoid
    s = np.sin(np.pi * (i + 0.5) / n)
    s = np.maximum(s, 0.0)
    pool.append(s / s.sum())

    # exponential tail
    for gamma in (0.2, 0.5, 1.0):
        left = np.exp(-gamma * i)
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate the pool and retain the best vector
    best_a = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        if time.time() >= deadline:
            break
        # ensure simplex and numerical safety
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective_fast(a, fft_len)   # exact objective (sum=1)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:03d} objective {val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Optional mirror‑descent (sub‑gradient) refinement
    if best_a is not None and time.time() < deadline - 8:
        print("[Phase I] Mirror‑descent refinement")
        sub_dead = deadline - 5.0
        a_sub, val_sub = _subgradient_optimize(best_a, n, fft_len, sub_dead, eta0=0.9)
        if val_sub < best_val:
            best_val = val_sub
            best_a = a_sub.copy()
            print(f"[Phase I] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Permutation‑based simulated annealing (swap moves)
    if best_a is not None and time.time() < deadline - 1:
        print("[Phase II] Swap‑based simulated annealing")
        a = best_a.copy()
        cur_val = _objective_fast(a, fft_len)
        best_seq = a.copy()
        best_seq_val = cur_val

        # Annealing schedule parameters
        T_start = 0.08
        T_end = 1e-4
        total_swaps = 0

        while time.time() < deadline:
            # linear cooling based on elapsed fraction
            elapsed = time.time() - (deadline - budget_s + 0.5)
            frac = min(elapsed / (budget_s - 0.5), 1.0)
            T = T_end + (T_start - T_end) * (1.0 - frac)

            # propose a random swap of two distinct indices
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            a[i], a[j] = a[j], a[i]

            new_val = _objective_fast(a, fft_len)

            # accept if better, otherwise accept with Metropolis probability
            if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_seq_val:
                    best_seq_val = new_val
                    best_seq = a.copy()
                    print(f"[Swap] iter {total_swaps:07d} new best {best_seq_val:.6f}")
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]

            total_swaps += 1
            if total_swaps % 200000 == 0:
                print(f"[Swap] {total_swaps} iterations, cur={cur_val:.6f}, best={best_seq_val:.6f}")

        best_a = best_seq
        best_val = best_seq_val

    # ------------------------------------------------------------------
    # 5️⃣  Final normalisation and safety checks
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s < 1e-12:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_a = best_a / s

    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
