"""
propose_candidate
-----------------
Search for a low‑value step‑function (non‑negative, sum‑normalised) that
minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

(the sum‑normalisation makes the objective scale‑invariant).

The strategy is a multi‑phase optimiser that works on a larger problem
size (n = 8192) than the previous baseline.  A good known solution for
n = 4096 is up‑sampled and then refined by several complementary
optimisers:

1. **L2‑flattening Adam** – minimise the L2 distance between the
   autocorrelation and the ideal uniform target (lower bound).  This
   quickly reduces the peak of the convolution.

2. **Smooth‑max Adam** – a log‑sum‑exp surrogate of the max‑convolution
   provides a smooth gradient; the temperature is annealed to focus
   on the true maximum.

3. **Exponentiated‑gradient sub‑gradient** – works directly on the exact
   max‑convolution, exploiting the simplex geometry.

4. **Projected gradient descent** – fine‑tunes the solution on the exact
   objective.

5. **Simulated‑annealing** – a cheap local‑search that can escape tiny
   basins.

All phases respect the global time budget (default 1000 s) and the
constraints (non‑negative, sum ≥ 0.01).  The best sequence found is
returned as a plain Python list.

The implementation uses only NumPy (no external solvers) and runs
comfortably within the time limit on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) = convolution(w, rev(a)).
    Returns an array of length len(a) (the part needed for the gradient).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Candidate factories ---------------------------------------------------

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _upscale(seq, new_n):
    """Linear interpolation of a non‑negative sequence to a larger length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    up_sum = up.sum()
    if up_sum < 1e-12:
        up = _make_uniform(new_n)
    else:
        up /= up_sum
    return up

def _adam_optimize_l2(a0, n, conv_fft_len, corr_fft_len, target,
                      deadline, max_iters=200_000, lr=0.05):
    """
    Adam optimiser that minimises the L2 distance between conv(a,a) and
    a uniform target.  Returns the best vector encountered and its exact
    objective value (max‑convolution based).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                    # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to enforce simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        # L2 loss
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # exact objective
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if (it + 1) % 50000 == 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _adam_optimize_smoothmax(a0, n, conv_fft_len, corr_fft_len,
                             deadline, max_iters=200_000,
                             lr=0.05, lam0=0.1, lam_factor=1.02):
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate of the
    maximum convolution entry.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam0
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[SmoothAdam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if (it + 1) % 50000 == 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0, n, conv_fft_len, corr_fft_len,
                                 deadline, max_iters=300_000,
                                 eta0=1.0):
    """
    Exact‑max sub‑gradient optimiser using exponentiated gradient
    (mirrored onto the probability simplex).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # identify active constraints (max entries)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0, n, conv_fft_len, corr_fft_len,
                     deadline, max_iters=150_000):
    """
    Plain projected gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(a0, n, conv_fft_len, deadline, max_moves=80_000):
    """
    Simulated‑annealing style local search on the simplex.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    global time budget and returns the best sequence found as a Python
    list.
    """
    np.random.seed(seed)
    # leave a small safety margin for the final evaluation
    deadline = time.time() + budget_s - 0.5

    # Problem size (larger than the baseline to approach the theoretical bound)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start: up‑sample the previous best if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = _upscale(warm, n)
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    pool = []

    if warm is not None:
        pool.append(warm.copy())

    # Dirichlet draws with varying concentration (encourage sparsity / spread)
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))

    # Analytic families
    pool.append(_make_uniform(n))
    pool.append(_make_exponential(n, gamma=0.4))
    pool.append(_make_powerlaw(n, expo=-1.1))
    pool.append(_make_triangle(n))

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # uniform target for L2 flattening (the ideal lower‑bound convolution)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Optimisation over each candidate (time‑aware)
    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        print(f"[Candidate {idx}] starting optimisation (n={n})")

        if val_l2 < best_val:

    except Exception as exc:

    return best_seq.tolist()
