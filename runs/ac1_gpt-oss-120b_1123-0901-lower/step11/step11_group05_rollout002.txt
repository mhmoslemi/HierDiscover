The following implementation searches for a non‑negative step‑function that minimizes  

\[
\text{score}= \frac{2 n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear convolution of the vector with itself.  
The algorithm consists of several complementary phases:

1. **Diverse warm‑starts** – many simple families (uniform, Beta‑shapes,
   Dirichlet draws, exponential draws and the previous best solution) are
   generated and normalised to the probability simplex.  
   The five best among them are kept as seeds.

2. **Multi‑restart sub‑gradient optimisation** – from each seed a short
   exponentiated‑gradient (sub‑gradient) run is performed.  The runs are cheap
   (≈ 20 k iterations) and are repeated many times until the bulk of the time‑budget
   is exhausted.  The best vector found so far is always retained.

3. **Fine‑tuning with Adam on a smooth‑max surrogate** – the current global best
   is improved with Adam optimisation of the soft‑max approximation of the
   max‑convolution.  Temperature (`lam`) is increased gradually so that the
   surrogate approaches the true max.

4. **Projected gradient descent (PGD)** – a small deterministic phase that
   moves opposite to the sub‑gradient of the exact max‑convolution, with an
   adaptive step size and projection onto the simplex.

5. **Simulated‑annealing local search** – tiny mass transfers between two
   coordinates are accepted either if they improve the score or with a
   probability depending on a slowly decreasing temperature.

6. **Cyclic shift exploitation** – because a linear convolution is not shift‑invariant,
   a few rolls of the final vector are tested and the best shift is kept.

All phases respect the total time budget (≈ 1000 s).  
The code uses only `numpy` and the provided `evaluate_sequence` function,
and runs on a single process (it easily fits into the two‑CPU limit).

```python
"""
Hybrid multi‑restart optimisation for the max‑convolution bound.

The routine builds many diverse initial vectors, runs a large number of
short exponentiated‑gradient sub‑gradient optimisations, refines the best
candidate with Adam on a smooth‑max surrogate, applies a deterministic
projected‑gradient descent, performs a simulated‑annealing local search and
finally tries a handful of cyclic shifts.  The whole search respects a
1000 s time budget and always returns a non‑negative list suitable for
evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2 * n * max(a*a)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimiser for the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # multiplicative EG step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate of max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e12           # push very close to the true max

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic projected gradient descent on the exact max‑convolution.
    Uses an adaptive step size (increase on improvement, halve otherwise).
    """
    a = _project_simplex(a0.copy())
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    step = 0.1

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        cand = a - step * grad
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)

        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step *= 1.05   # slightly accelerate
        else:
            step *= 0.5    # backtrack

        if step < 1e-12:
            break
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    a valid list of heights.
    """
    np.random.seed(seed)
    # A small safety margin ensures we never exceed the deadline.
    deadline = time.time() + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0) Build a large pool of diverse initial candidates
    candidates = []

    # warm start (if a previous solution is bound to the name)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-8:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / s
        candidates.append(a)

    # uniform distribution
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric beta‑shaped densities (U‑shaped for α<1, bell‑shaped for α>1)
    x = (np.arange(n) + 0.5) / n
    for alpha in np.linspace(0.3, 2.0, 12):
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # random Dirichlet draws with varied concentration
    for conc in np.concatenate([
            np.linspace(0.05, 0.5, 5),
            np.linspace(0.5, 3.0, 10),
            np.linspace(3.0, 8.0, 5)]):
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # exponential draws (heavy‑tailed) – also useful as spiky starts
    for _ in range(15):
        a = np.random.exponential(scale=1.0, size=n)
        a = a / a.sum()
        candidates.append(a)

    # ------------------------------------------------------------------
    # 1) Evaluate all candidates and keep the best few (simplex‑normalised)
    best_a = None
    best_val = np.inf
    top_k = 5
    top = []                       # list of (value, vector)

    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] new global best {val:.6f}")
        # maintain top‑k list
        if len(top) < top_k:
            top.append((val, a.copy()))
            top.sort(key=lambda t: t[0])
        elif val < top[-1][0]:
            top[-1] = (val, a.copy())
            top.sort(key=lambda t: t[0])

    # ------------------------------------------------------------------
    # 2) Multi‑restart sub‑gradient optimisation (the heavy part)
    # Allocate ~85 % of the remaining time to many short runs.
    time_remaining = deadline - time.time()
    sub_deadline = time.time() + 0.85 * time_remaining
    max_iters_per_run = 30_000          # cheap runs – many restarts

    run_id = 0
    while time.time() < sub_deadline:
        # choose a start vector: 60 % from current top‑k, otherwise fresh Dirichlet
        if np.random.rand() < 0.6:
            _, start = top[np.random.randint(len(top))]
        else:
            conc = np.random.uniform(0.05, 5.0)
            start = np.random.dirichlet(np.full(n, conc))

        a_opt, val_opt = _subgradient_optimize_expgrad(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=sub_deadline,
            max_iters=max_iters_per_run,
            eta0=1.2 * (0.5 + np.random.rand())   # randomise step‑size a bit
        )

        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Run {run_id}] improved global best to {best_val:.6f}")

            # keep top‑k up‑to‑date
            if len(top) < top_k:
                top.append((val_opt, a_opt.copy()))
                top.sort(key=lambda t: t[0])
            elif val_opt < top[-1][0]:
                top[-1] = (val_opt, a_opt.copy())
                top.sort(key=lambda t: t[0])

        run_id += 1

    # ------------------------------------------------------------------
    # 3) Adam fine‑tuning on the smooth‑max surrogate (still a large chunk of time)
    adam_deadline = deadline - 4.0
    if adam_deadline > time.time():
        a_adam, val_adam = _adam_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=adam_deadline,
            max_iters=400_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Deterministic projected‑gradient descent (helps to get out of flat plateaus)
    pgd_deadline = deadline - 3.0
    if pgd_deadline > time.time():
        a_pgd, val_pgd = _pgd_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=pgd_deadline,
            max_iters=80_000
        )
        if val_pgd < best_val:
            best_val, best_a = val_pgd, a_pgd.copy()
            print(f"[PGD] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Tiny stochastic local search with simulated annealing
    local_deadline = deadline - 0.5
    # Linear temperature schedule from T_start to T_end
    T_start, T_end = 2e-2, 5e-6
    while time.time() < local_deadline:
        # progress fraction (0 → 1)
        prog = (time.time() - (deadline - budget_s)) / budget_s
        T = T_start * (1 - prog) + T_end * prog

        # pick source and destination indices
        i, j = np.random.choice(n, 2, replace=False)
        delta = min(best_a[i], 1e-4) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        val = _objective(cand, n, conv_fft_len)
        if val < best_val or np.random.rand() < np.exp(-(val - best_val) / T):
            best_val, best_a = val, cand.copy()
            # occasional logging
            if np.random.rand() < 0.001:
                print(f"[Anneal] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Try a few cyclic shifts (the linear convolution changes under a shift)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2,
              5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) Final sanitisation and reporting
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)          # ensure exact simplex normalisation

    # official evaluation (mainly for the log)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
```
