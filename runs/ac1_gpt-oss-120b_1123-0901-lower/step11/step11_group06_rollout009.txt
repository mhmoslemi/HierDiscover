"""
Propose a non‑negative step function with a low evaluation score.

Algorithm
---------
1. **Small‑scale exploration** (n≈1024) – a cheap L2‑flattening of the
   self‑convolution towards a constant target (the average value 1/(2n‑1)).
   Gradient = 4·cross_correlation(conv‑avg, a).  After each update the
   vector is projected onto the probability simplex.

2. **Upsampling** – the best small vector is interpolated to a larger size
   (n≈4096) and normalised.

3. **Large‑scale L2‑flattening** – the same gradient descent is run for a
   longer period, now directly reducing the peak of the convolution.

4. **Fine‑tuning** – an exponentiated‑gradient sub‑gradient method
   (exact max‑convolution) is applied to push the remaining
   max‑convolution down.

All phases respect the global time budget (≈1000 s).  The routine always
returns the best sequence seen so far; if no valid vector is found a
uniform vector is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length ``len(a)``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and its shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both ends)
    for gamma in (0.15, 0.3, 0.6):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law with interior peaks
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.6, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps at the edges
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.45):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
            cand.append(a)

    return cand


def _upsample(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from length ``len(a_small)`` to ``n_large``."""
    x_small = np.linspace(0.0, 1.0, len(a_small))
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    s = a_large.sum()
    if s == 0 or np.isnan(s):
        a_large = np.random.dirichlet(np.ones(n_large))
        s = a_large.sum()
    return a_large / s

    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lr_start: float = 0.3,
    lr_end: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the L2 distance between ``a*a`` and the constant
    target 1/(2n‑1).  The update uses the exact gradient
    ``grad = 4·cross_correlation(conv−mean, a)``.
    Returns the best vector (by the true evaluation function) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    target = 1.0 / (2 * n - 1)
    avg_vec = np.full(2 * n - 1, target)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)          # length 2n‑1
        max_b = conv.max()
        val = 2.0 * n * max_b
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2 it={it:07d}] new best {val:.6f}")

        # gradient of ½‖conv−target‖²
        r = conv - avg_vec
        grad = 4.0 * _cross_correlation(r, a, corr_fft_len)

        # learning‑rate schedule (linear decay)
        lr = lr_start * (1.0 - it / max_iters) + lr_end * (it / max_iters)

        # normalised step to avoid exploding updates
        gn = np.linalg.norm(grad)
        if gn > 0:
            a = a - lr * (grad / gn)
        else:
            a = a + lr * (np.random.rand(n) - 0.5) * 0.01

        # non‑negativity & simplex projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation using exponentiated‑gradient updates.
    This is identical to the routine used in the reference implementation.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The method uses a two‑stage multiscale L2‑flattening followed by a
    sub‑gradient refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Small‑scale stage (n≈1024) – fast exploration
    n_small = 1024
    conv_fft_len_s = _next_pow2(2 * n_small - 1)
    corr_fft_len_s = _next_pow2(3 * n_small - 2)

    # warm start from the global variable if present
    warm_small = None
    warm_big = globals().get("height_sequence_1")
    if warm_big is not None:
        warm_big = np.asarray(warm_big, dtype=np.float64)
        # down‑sample to the small size
        warm_small = _upsample(warm_big, n_small)

    pool_small = _initial_candidates(n_small, warm_small)

    best_a_small = None
    best_val = float("inf")
    max_cand_small = 8

    for idx, a0 in enumerate(pool_small):
        if time.time() > deadline or idx >= max_cand_small:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n_small))
        a0 = a0 / a0.sum()

        sub_dead = min(deadline, time.time() + 12.0)  # ~12 s per candidate
        a_opt, val_opt = _l2_flatten_optimize(
            a0,
            n_small,
            conv_fft_len_s,
            corr_fft_len_s,
            sub_dead,
            max_iters=300_000,
            lr_start=0.35,
            lr_end=0.02,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a_small = a_opt.copy()
            print(f"[Small {idx}] L2 flatten improved to {val_opt:.6f}")

    if best_a_small is None:
        best_a_small = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # Upsample to the large problem size (n≈4096)
    n_large = 4096
    conv_fft_len_l = _next_pow2(2 * n_large - 1)
    corr_fft_len_l = _next_pow2(3 * n_large - 2)

    best_a_large = _upsample(best_a_small, n_large)

    # ------------------------------------------------------------------
    # Large‑scale L2 flattening (most of the time budget)
    if time.time() < deadline - 40.0:
        print("[Large] Starting long L2‑flattening phase")
        sub_dead = min(deadline, time.time() + 260.0)   # up to ~4 min
        best_a_large, best_val = _l2_flatten_optimize(
            best_a_large,
            n_large,
            conv_fft_len_l,
            corr_fft_len_l,
            sub_dead,
            max_iters=800_000,
            lr_start=0.25,
            lr_end=0.01,
        )
        print(f"[Large] L2 flatten best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final exact‑max sub‑gradient refinement
    if time.time() < deadline - 5.0:
        print("[Subgrad] Final exponentiated‑gradient refinement")
        best_a_large, best_val = _subgradient_optimize_expgrad(
            best_a_large,
            n_large,
            conv_fft_len_l,
            corr_fft_len_l,
            deadline,
            max_iters=600_000,
            eta0=0.9,
        )
        print(f"[Subgrad] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a valid return
    if best_a_large is None or best_a_large.sum() < 1e-12:
        best_a_large = np.full(n_large, 1.0 / n_large)

    best_a_large = np.clip(best_a_large, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_a_large)})")
    return best_a_large.tolist()
