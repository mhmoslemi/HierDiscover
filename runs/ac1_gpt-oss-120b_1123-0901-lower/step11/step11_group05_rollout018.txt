"""
Propose a low‑score step‑function.

The routine builds a large set of analytical candidate shapes
(beta, exponential, power‑law, linear‑decay, etc.), evaluates each
exactly (via FFT convolution) and selects the best one.
From that starter it runs two different sub‑gradient phases
(additive and exponentiated‑gradient), a short Adam fine‑tune on a
smooth‑max surrogate and a tiny stochastic local search.  All phases
are timed so that the 1000 s budget is never exceeded.  The final
vector is projected onto the probability simplex, clipped to the
allowed range and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value of 2·n·max(a*a) for a normalised (sum=1) vector."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Candidate generators ---------------------------------------------------
def _candidate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _candidate_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(alpha,alpha) on the grid."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _candidate_exponential(n: int, delta: float) -> np.ndarray:
    """
    a_i ∝ (1-e^{-delta}) * e^{-delta*i}, then normalised.
    delta > 0.
    """
    r = np.exp(-delta)
    # geometric series truncated at n
    geom = r ** np.arange(n, dtype=np.float64)
    a = (1.0 - r) * geom
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _candidate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(n, dtype=np.float64) + 1.0) ** (-p)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _candidate_linear_decay(n: int) -> np.ndarray:
    """a_i ∝ (n-i)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _candidate_quadratic_decay(n: int) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=np.float64) ** 2)
    return a / a.sum()


def _candidate_mixed_exponential(n: int, delta1: float, delta2: float, w: float) -> np.ndarray:
    """Convex combination w*exp(delta1)+(1-w)*exp(delta2)."""
    e1 = _candidate_exponential(n, delta1)
    e2 = _candidate_exponential(n, delta2)
    a = w * e1 + (1.0 - w) * e2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a

def _subgradient_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative update (soft‑max parametrisation)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_additive(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Additive projected sub‑gradient descent.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[AddGrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a - eta * grad_a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.04
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform
    candidates.append(_candidate_uniform(n))

    # symmetric beta densities (including very 'U' shaped)
    for alpha in np.concatenate(
        (np.arange(0.05, 0.51, 0.05), np.arange(0.55, 1.01, 0.05))
    ):
        candidates.append(_candidate_beta(n, float(alpha)))

    # exponential families (delta = -ln(r))
    deltas = np.concatenate(
        (np.linspace(1.0 / n, 0.004, 15), np.linspace(0.004, 0.02, 15))
    )
    for delta in deltas:
        candidates.append(_candidate_exponential(n, float(delta)))

    # power‑law tails
    for p in np.arange(0.05, 2.05, 0.10):
        candidates.append(_candidate_powerlaw(n, float(p)))

    # simple decays
    candidates.append(_candidate_linear_decay(n))
    candidates.append(_candidate_quadratic_decay(n))

    # mixed exponentials
    for w in (0.3, 0.5, 0.7):
        for d1, d2 in ((0.001, 0.003), (0.002, 0.01), (0.004, 0.015)):
            candidates.append(_candidate_mixed_exponential(n, d1, d2, w))

    # random Dirichlet draws (varied concentration)
    for _ in range(30):
        conc = np.random.uniform(0.3, 3.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ------------------------------------------------------------------
    # 1) pick the best cheap candidate (exact eval)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] candidate objective {val:.6f}")

    # ------------------------------------------------------------------
    # 2) additive sub‑gradient phase (explorative)
    sub_deadline = deadline - 20.0   # keep time for Adam & local tweaks
    if sub_deadline > time.time():
        a_sub, val_sub = _subgradient_additive(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=10_000_000, eta0=0.8
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Additive] improved to {val_sub:.6f}")

    # 3) exponentiated‑gradient sub‑gradient phase
    sub_deadline2 = deadline - 12.0
    if sub_deadline2 > time.time():
        a_sub2, val_sub2 = _subgradient_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline2,
            max_iters=5_000_000, eta0=1.5
        )
        if val_sub2 < best_val:
            best_val, best_a = val_sub2, a_sub2.copy()
            print(f"[ExpGrad] improved to {val_sub2:.6f}")

    # ------------------------------------------------------------------
    # 4) Adam fine‑tuning on the smooth‑max surrogate
    adam_deadline = deadline - 6.0
    if adam_deadline > time.time():
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=500_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 5) tiny stochastic local search (last few seconds)
    local_deadline = deadline - 0.4
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny random chunk, keep positivity
        delta = min(best_a[i], 5e-4) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)          # stay on the simplex
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 6) try a few cyclic shifts (convolution is shift‑sensitive)
    shifts = [n // 8, n // 4, n // 2, 3 * n // 4]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 7) final sanitisation and return
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)

    # official evaluation (mainly for logging)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
```

This implementation enriches the initial pool of candidates, adds an **additive sub‑gradient** phase (different dynamics from the original exponentiated‑gradient), allocates more iterations to both sub‑gradient stages, speeds up the Adam schedule, and finishes with a brief stochastic local‑search and a few cyclic shifts. All steps respect the 1000 s budget and return a valid non‑negative list suitable for `evaluate_sequence`.
