    mag_const = np.mean(mags[1:])   # some positive.

"""
Hybrid optimiser for the step‑function (max‑convolution) attack.

The algorithm works in two scales:
1.  A cheap optimisation on a small problem (n≈1024) to discover a good
    “shape’’ of the distribution.  This stage uses many random starts
    (including phase‑only Fourier candidates) and a short Adam/sub‑gradient
    descent.
2.  The shape is up‑sampled to the target length (n=4096) and refined with
    a deeper optimiser:
      – Adam on a smooth‑max surrogate,
      – exact sub‑gradient (exponentiated‑gradient),
      – spectral‑flattening (forcing all non‑DC FFT magnitudes to be equal),
      – a linear‑programming mixing step,
      – projected gradient descent on the true max‑convolution,
      – a greedy local‑search / simulated‑annealing polish.
All iterates stay on the probability simplex (∑aᵢ=1, aᵢ≥0) – the
evaluation function is homogeneous, so scaling does not matter.
The routine respects a total time budget (default 1000 s) and always
returns the best sequence found before the deadline.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate
        φ(a)=logsumexp(λ·conv(a)) / λ  .
    The sequence a is kept on the simplex via a soft‑max parametrisation.
    Returns the best a encountered and its value 2·n·max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.01
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (softmax over λ·(conv‑max))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # schedule λ → ∞
        lam = min(lam * lam_factor, 1e7)

        # slowly decay learning‑rate
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 350_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best a and its value 2·n·max(conv).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively enforce constant magnitude for all non‑DC Fourier coefficients
    while staying on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # FFT of current a
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        # compute the target constant magnitude for the non‑DC part
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC untouched
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 8,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For current f we solve   max Σ b_j   s.t.  (f★b)_k ≤ M,
    where M = max_k (f★f)_k.  The solution b is normalised and mixed with f.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active constraints (indices attaining the peak)
        eps = 1e-12
        active_idx = np.where(conv_a >= M - eps)[0]
        # add a few well‑spread positions to keep the LP well‑posed
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active_idx, extra)))

        rows, cols, data = [], [], []
        for row, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        c = -np.ones(n)                         # maximise Σ b_j
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        # normalise
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # convex mixture
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            # reduce mixing if no progress
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 180_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of deterministic starts plus a few geometric shapes.
    """
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # a few circular shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # symmetric exponential / Laplace shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # power‑law shapes (i·(n‑i+1))^α
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # narrow Gaussian bumps at both ends
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # sine‑shaped bump
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _random_phase_candidates(n: int, conv_fft_len: int, count: int = 8) -> list[np.ndarray]:
    """
    Generate candidates by keeping a constant magnitude spectrum and randomising the phase.
    """
    cand = []
    for _ in range(count):
        # start from a random Gaussian vector
        x = np.random.randn(n)
        X = np.fft.rfft(x, conv_fft_len)

        # keep the DC component (its magnitude encodes the total mass)
        dc = X[0]

        # target constant magnitude for all other frequencies
        mag_const = np.mean(np.abs(X[1:])) if X.shape[0] > 1 else 0.0
        phase = np.angle(X[1:])

        X_new = np.empty_like(X)
        X_new[0] = dc
        if X.shape[0] > 1:
            X_new[1:] = mag_const * np.exp(1j * phase)

        a = np.fft.irfft(X_new, conv_fft_len)[:n]
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12 or np.isnan(a).any():
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a.sum()
        cand.append(a)
    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Returns a non‑negative step function (list of floats) with a low
    evaluation value under `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Lengths: a cheap global search at small scale, then refinement at
    # the target scale.
    n_small = 1024
    n_big = 4096

    def run_optimisation_on_scale(n, warm_seq):
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # Build candidates
        pool = _initial_candidates(n, warm_seq) + _spike_candidates(n) + \
               _random_phase_candidates(n, conv_fft_len, count=6)

        # add a few pure Dirichlet draws
        for alpha in (0.1, 0.5, 1.0, 2.0):
            a = np.random.dirichlet(alpha * np.ones(n))
            pool.append(a)

        best_seq: np.ndarray | None = None
        best_val = float("inf")
        max_candidates = 10

        # Phase I – Adam + sub‑gradient on a handful of promising starts
        for idx, a0 in enumerate(pool):
            if time.time() > deadline or idx >= max_candidates:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 0.01:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # allocate about 15 s for Adam per candidate (or whatever is left)
            sub_dead = min(deadline, time.time() + 15.0)
            a_adam, val_adam = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=300_000
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

            if time.time() < deadline - 5:
                a_sub, val_sub = _subgradient_optimize(
                    best_seq, n, conv_fft_len, corr_fft_len, deadline,
                    max_iters=300_000, eta0=0.9
                )
                if val_sub < best_val:
                    best_val, best_seq = val_sub, a_sub.copy()
                    print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

        # Phase II – spectral flattening
        if best_seq is not None and time.time() < deadline - 5:
            print("[Spectral] Flattening (scale=", n, ")")
            a_spec, val_spec = _spectral_flatten_optimize(
                best_seq, n, conv_fft_len, deadline,
                max_iters=180_000
            )
            if val_spec < best_val:
                best_val, best_seq = val_spec, a_spec.copy()
                print(f"[Spectral] improved to {val_spec:.6f}")

        # Phase III – LP mixing (few iterations)
        if best_seq is not None and time.time() < deadline - 10:
            print("[LP] Refinement (scale=", n, ")")
            a_lp, val_lp = _lp_improve(
                best_seq, n, conv_fft_len, deadline,
                max_iter=6, tau=0.5
            )
            if val_lp < best_val:
                best_val, best_seq = val_lp, a_lp.copy()
                print(f"[LP] improved to {val_lp:.6f}")

        return best_seq, best_val, conv_fft_len, corr_fft_len

    # ------------------------------------------------------------------
    # Stage 1 – cheap search on the small scale
    warm_small = None
    if 'height_sequence_1' in globals():
        warm_small = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if warm_small.shape[0] > n_small:
            warm_small = warm_small[:n_small]
        elif warm_small.shape[0] < n_small:
            warm_small = np.pad(warm_small, (0, n_small - warm_small.shape[0]))
        warm_small = np.maximum(warm_small, 0.0)
        if warm_small.sum() >= 0.01:
            warm_small = warm_small / warm_small.sum()
        else:
            warm_small = None

    best_small, val_small, _, _ = run_optimisation_on_scale(n_small, warm_small)
    print(f"[Stage 1] best value at n={n_small}: {val_small:.6f}")

    # ------------------------------------------------------------------
    # Upsample to the large scale
    if best_small is None:
        best_big_start = np.full(n_big, 1.0 / n_big)
    else:
        x_small = np.arange(n_small)
        x_big = np.linspace(0, n_small - 1, n_big)
        best_big_start = np.interp(x_big, x_small, best_small)
        best_big_start = np.maximum(best_big_start, 0.0)
        if best_big_start.sum() < 0.01:
            best_big_start = np.random.dirichlet(np.ones(n_big))
        else:
            best_big_start = best_big_start / best_big_start.sum()

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on the target length
    best_big, val_big, conv_fft_len_big, corr_fft_len_big = run_optimisation_on_scale(
        n_big, best_big_start)

    # Phase IV – projected gradient descent on the exact max‑conv
    if best_big is not None and time.time() < deadline - 8:
        print("[GD] Projected gradient descent (large)")
        a_gd, val_gd = _max_conv_descent(
            best_big, n_big, conv_fft_len_big,
            corr_fft_len_big, deadline, max_iters=150_000
        )
        if val_gd < val_big:
            val_big, best_big = val_gd, a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # Phase V – final simulated‑annealing polish
    if best_big is not None and time.time() < deadline - 5:
        print("[Anneal] Final simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_big, n_big, conv_fft_len_big, deadline,
            max_moves=80_000
        )
        if val_ann < val_big:
            val_big, best_big = val_ann, a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Guard against an empty result.
    if best_big is None:
        best_big = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # Clip to evaluator’s allowed range, just for safety.
    best_big = np.clip(best_big, 0.0, 1000.0)

    print(f"Final best value {val_big:.6f} (length {len(best_big)})")
    return best_big.tolist()
