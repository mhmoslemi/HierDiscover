The goal is to minimise  

\[
\text{obj}(a)=\frac{2\,n\;\max\_{k}(a*a)[k]}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the linear convolution of a non‑negative vector `a`.  
The objective is scale–invariant, therefore we keep the vectors normalised
(`∑a_i = 1`) during optimisation and rescale only at the very end.

The programme follows a **multi‑phase hybrid optimiser**:

1. **Diversified initial pool** – many different shapes (uniform, Dirichlet,
   exponential, linear, power‑law, sinusoid, spike‑patterns, …).  
   Each candidate is normalised and its exact objective is computed quickly
   with an FFT‑based convolution.

2. **Selection** – the ten best candidates (smallest objective) are taken
   forward.

3. **Deep optimisation of the selected candidates**  
   * *Adam* on a smooth‑max surrogate of the maximum convolution,  
   * *Exponentiated‑gradient sub‑gradient* on the exact max,  
   * *Projected gradient descent* when time permits.  
   The best‑found vector is kept globally.

4. **Spectral flattening** – repeatedly force the FFT magnitude to be
   (almost) constant while staying on the probability simplex.

5. **Random‑phase refinement** – keep the spectrum magnitude fixed, randomise
   the phases, project back onto the simplex and keep any improvement.

6. **Fine‑tuned projected gradient descent**.

7. **Long simulated‑annealing walk** – small mass transfers between two
   coordinates are accepted with a decreasing temperature schedule.

All phases respect the global deadline (≈ 1000 s).  The algorithm works with
`n = 8192` (larger than the previous `4096`) which gives the optimiser more
degrees of freedom while still being fast enough with FFT‑based convolutions.

The final vector is clipped to `[0, 1000]` (the limits enforced by
`evaluate_sequence`) and returned as a plain Python list.

```python
"""
Hybrid multi‑phase optimiser for the step‑function objective.

- builds a large diverse pool of start vectors (uniform, Dirichlet,
  exponential, linear, power‑law, sinusoid, spikes, piece‑wise blocks),
- evaluates them quickly with FFT‑based convolution,
- keeps the best few and performs deep optimisation on each:
    * Adam on a smooth‑max surrogate,
    * exponentiated‑gradient sub‑gradient on the exact max,
    * projected gradient descent,
- follows with spectral‑flattening, random‑phase refinement,
  a final projected‑gradient pass and a long simulated‑annealing walk.
All work is done under the given time budget and the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve the current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                # sum(a)=1 → denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 350_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Force FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        A = np.fft.fft(a)
        energy = np.dot(a, a)               # ||a||_2^2
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s
        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Keep the FFT magnitude fixed and randomise the phases."""
    mag = np.abs(np.fft.fft(a0))

    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]                       # DC unchanged

        A[1:half] = mag[1:half] * np.exp(1j * phi)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 15_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Simple exponential families (decreasing)
    for gamma in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        shape = np.exp(-gamma * np.arange(n))
        shape /= shape.sum()
        cand.append(shape)

    # Mixture of uniform and exponential (controls the end‑point size)
    for gamma in (0.001, 0.003, 0.005):
        exp_shape = np.exp(-gamma * np.arange(n))
        exp_shape /= exp_shape.sum()
        for lam in (0.2, 0.5, 0.8):
            mixed = (1.0 - lam) * np.full(n, 1.0 / n) + lam * exp_shape
            mixed /= mixed.sum()
            cand.append(mixed)

    # Linear decreasing (a “ramp”)
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    ramp /= ramp.sum()
    cand.append(ramp)

    # Power‑law decreasing
    idx = np.arange(1, n + 1, dtype=np.float64)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = (n - idx + 1) ** (-alpha)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the time budget (≈ 1000 s)
    and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Use a larger length – more degrees of freedom give a better bound.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously found solution (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build the initial candidate pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # a few extra random Dirichlet shuffles
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    # random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    # ------------------------------------------------------------------
    # Quick evaluation of every candidate (FFT‑based, cheap)
    scored = []
    for a in pool:
        if time.time() > deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c      # sum(a)=1
        scored.append((val, a))

    scored.sort(key=lambda x: x[0])
    top_k = min(8, len(scored))
    selected = [a for _, a in scored[:top_k]]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – deep optimisation of the most promising starts
    for idx, a0 in enumerate(selected):
        if time.time() > deadline:
            break
        a0 = a0.copy()
        a0 = a0 / a0.sum()
        # --- Adam pass (≈ 20 s per candidate) ---
        sub_dead = min(deadline, time.time() + 20.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=400_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # --- Sub‑gradient (exp‑grad) phase ---
        if time.time() < deadline - 5.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=400_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – spectral flattening
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Spectral] Starting spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline, max_iters=250_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase III – random‑phase refinement
    if best_seq is not None and time.time() < deadline - 10.0:
        print("[PhaseRandom] Random‑phase refinement")
        a_rand, val_rand = _random_phase_refine(
            best_seq, n, conv_fft_len, deadline, max_iters=400_000
        )
        if val_rand < best_val:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[PhaseRandom] improved to {val_rand:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – final projected gradient descent
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[GD] Final projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase V – lightweight simulated‑annealing walk (more moves)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Anneal] Light simulated‑annealing refinement")
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val_ann = val
        best_a_ann = a.copy()

        T_start = 0.07
        T_end = 1e-4
        moves = 0
        max_moves = 200_000
        while time.time() < deadline and moves < max_moves:
            T = max(T_end, T_start * (1.0 - moves / float(max_moves)))
            i = np.random.randint(n)
            if a[i] < 1e-14:
                moves += 1
                continue
            j = np.random.randint(n)
            if i == j:
                moves += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()
            if val2 < best_val_ann:
                best_val_ann = val2
                best_a_ann = a2.copy()
                a = a2
                val = val2
            else:
                if np.random.rand() < np.exp(-(val2 - val) / T):
                    a = a2
                    val = val2
            moves += 1

        if best_val_ann < best_val:
            best_val = best_val_ann
            best_seq = best_a_ann.copy()
            print(f"[Anneal] improved best to {best_val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
