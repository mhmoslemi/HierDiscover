"""
Hybrid optimizer for the max‑convolution bound.

The routine builds a diverse pool of start vectors (uniform,
Dirichlet draws, spikes, exponentials, Gaussians, sinusoid, etc.),
optimises each promising candidate with projected gradient descent on the
exact max‑convolution, then refines the current best by
* spectral flattening (forcing the FFT magnitude to be (almost) constant),
* random‑phase exploration on the constant‑magnitude manifold,
* a light simulated‑annealing local search.
All phases respect the global time budget (≈ 1000 s) and keep the vector on
the probability simplex (non‑negative, unit sum).  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) = n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are the same (e.g. zero vector) – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective_val(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) for a (assumed to sum to 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _gradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected (sub‑)gradient descent on the exact max‑convolution.
    Returns the best iterate seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_val(a, n, conv_fft_len)

    # simple diminishing step size
    step0 = 0.3

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # weight vector w – uniform on the active set of max entries
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not np.any(active):
            # numerical safety – fall back to uniform weighting
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # evaluate exact objective
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD] it {it:06d}  new best {val:.6f}")

    return best_a, best_val


def _spectral_flatten(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the FFT magnitude of a to be (almost) constant while
    preserving the DC component.  After each inverse FFT the vector is
    projected back onto the simplex.  The best iterate (lowest exact
    objective) is kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_val(a, n, conv_fft_len)

    half = n // 2
    it = 0
    while it < max_iters and time.time() < deadline:
        A = np.fft.fft(a)
        energy = np.dot(a, a)               # ||a||_2^2

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        # constant magnitude for all remaining frequencies
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))
            # Nyquist frequency (real) – random sign to keep real output
            if n % 2 == 0:
                sign = -1.0 if np.random.rand() < 0.5 else 1.0
                A_new[half] = sign * const_mag

        a = np.real(np.fft.ifft(A_new))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        val = _objective_val(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral] it {it:06d}  new best {val:.6f}")

        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 700_000,
) -> tuple[np.ndarray, float]:
    """
    Keep the FFT magnitude of a0 fixed and repeatedly randomise the phases.
    After each inverse FFT the vector is projected onto the simplex.
    The best vector (lowest exact objective) is retained.
    """
    mag = np.abs(np.fft.fft(a0))

    best_a = a0.copy()
    best_val = _objective_val(best_a, n, conv_fft_len)

    half = n // 2
    it = 0
    while it < max_iters and time.time() < deadline:
        # random phases for positive frequencies 1..half-1
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]                     # DC unchanged

        # positive frequencies
        A[1:half] = mag[1:half] * np.exp(1j * phi)
        # negative frequencies (complex conjugates)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        # Nyquist (real) frequency for even n
        if n % 2 == 0:
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        val = _objective_val(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[PhaseRandom] it {it:06d}  new best {val:.6f}")

        it += 1

    return best_a, best_val


def _annealing_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex:
    repeatedly transfer a random fraction of mass from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    # temperature schedule
    T_start = 0.07
    T_end = 1e-4

    for move in range(max_moves):
        if time.time() > deadline:
            break
        T = max(T_end, T_start * (1.0 - move / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move a random fraction (up to 40 %) from i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - best_val) / T):
                a = a2

    return best_a, best_val

# Candidate generation

def _initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with different concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential‑type decays (symmetric)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law (triangular) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Sparse spike patterns
    for k in (16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    # Randomly placed sparse spikes
    for k in (32, 64, 128):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            cand.append(a)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the given time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Fixed length – 4096 works well for the trade‑off between quality
    # and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run stored ``height_sequence_1``)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors
    pool = _initial_pool(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 6                     # number of candidates given a deep GD pass

    # ------------------------------------------------------------------
    # Phase I – deep gradient descent on a handful of promising starts
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Allocate a time slice (≈ 12 s) for this candidate
        sub_dead = min(deadline - 5.0, time.time() + 12.0)

        a_opt, val_opt = _gradient_descent(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=400_000
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Explore {idx}] GD improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase II – spectral flattening (constant‑magnitude FFT)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Spectral] Starting spectral flattening")
        a_spec, val_spec = _spectral_flatten(
            best_seq, n, conv_fft_len, deadline, max_iters=250_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase III – random‑phase refinement on the constant‑magnitude manifold
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[PhaseRandom] Random‑phase refinement")
        a_rand, val_rand = _random_phase_refine(
            best_seq, n, conv_fft_len, deadline, max_iters=600_000
        )
        if val_rand < best_val:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[PhaseRandom] improved to {val_rand:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – final simulated‑annealing fine‑tuning
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Anneal] Simulated‑annealing refinement")
        a_ann, val_ann = _annealing_local_search(
            best_seq, n, conv_fft_len, deadline, max_moves=300_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
