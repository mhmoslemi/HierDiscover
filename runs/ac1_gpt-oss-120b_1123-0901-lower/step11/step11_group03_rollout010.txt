"""
Hybrid random‑search + first‑order optimisation for the non‑negative step‑function
problem.

Goal: minimise 2·n·max(conv(a,a)) / (sum(a)**2) with a_i ≥ 0.
The objective is homogeneous, so we keep sum(a)=1 (simplex) throughout.
The algorithm consists of three stages:

1. **Broad random exploration** – generate many candidates (exponential,
   Dirichlet‑like or sparse spikes), evaluate the exact objective with a fast
   FFT‑based convolution and keep the best one.  This quickly discovers diverse
   low‑peak shapes.

2. **Refinement with Adam on a smooth‑max surrogate** – start from the best
   random candidate and run Adam optimisation on a soft‑max approximation of
   max(conv).  The parameter λ controlling the smoothness is slowly increased,
   so the optimisation gradually focuses on the true max.

3. **Exact sub‑gradient (exponentiated‑gradient) phase** – switch to the true
   max‑convolution and perform mirror‑descent on the simplex with a diminishing
   step size.  This phase gives the final fine‑tuning.

If time permits a short simulated‑annealing pass is applied as a last polish.
The whole routine respects the 1000 s budget and always returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Returns the best simplex vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation (keeps Σa=1)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06                         # base learning rate (a bit smaller)
    lam = 0.05                        # starting temperature
    lam_factor = 1.012                # very slow increase → smoother final stage

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- obtain current simplex vector -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- evaluate exact objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e8)

        # ----- occasional learning‑rate decay -----
        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Runs until the deadline and keeps the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # ----- active set for the current max -----
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- diminishing step size (mirror‑descent) -----
        eta = eta0 / np.sqrt(it + 1.0)

        # ----- exponentiated‑gradient update -----
        a = a * np.exp(-eta * grad_a)

        # ----- renormalise onto the simplex -----
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search on the simplex.
    Moves a random fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    Runs for at most `budget_s` seconds and always returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096                               # length of the vector
    conv_fft_len = _next_pow2(2 * n - 1)   # 8192
    corr_fft_len = _next_pow2(3 * n - 2)   # sufficient for cross‑corr

    # --------------------------------------------------------------
    # Warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Random‑search phase
    best_seq = None
    best_obj = float("inf")

    if warm is not None:
        conv = _full_convolution(warm, conv_fft_len)
        best_obj = 2.0 * n * conv.max()
        best_seq = warm.copy()
        print(f"[Init] warm start objective {best_obj:.6f}")

    it = 0
    while time.time() < deadline:
        # Choose a generator at random
        r = np.random.rand()
        if r < 0.55:
            # Exponential‑type full‑support vector
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum()
        elif r < 0.80:
            # Sparse spike vector (k roughly geometric)
            k = int(2 ** np.random.uniform(3, 7))  # 8 … 128 spikes
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a = np.zeros(n)
            a[pos] = masses
        else:
            # Power‑law / decaying tail
            gamma = np.random.uniform(0.2, 2.0)
            a = np.exp(-gamma * np.arange(n, dtype=np.float64))
            a = a / a.sum()

        # Optional tiny low‑pass smoothing (10 % of candidates)
        if np.random.rand() < 0.07:
            A = np.fft.rfft(a)
            keep = int(0.25 * len(A))          # keep low‑freq part only
            A[keep:] = 0
            a = np.real(np.fft.irfft(A, n))
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                a = np.random.dirichlet(np.ones(n))
            else:
                a = a / a.sum()

        # Exact objective
        conv = _full_convolution(a, conv_fft_len)
        obj = 2.0 * n * conv.max()

        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"[Random {it:06d}] new best {obj:.6f}")

        it += 1
        if it % 5000 == 0 and time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase I – Adam optimisation on a smooth‑max surrogate
    if best_seq is not None and time.time() < deadline - 6:
        print("[Phase I] Adam optimisation started")
        sub_dead = deadline - 5.0
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=600_000
        )
        if val_adam < best_obj:
            best_obj, best_seq = val_adam, a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase II – Sub‑gradient (exponentiated‑gradient) optimisation
    if best_seq is not None and time.time() < deadline - 4:
        print("[Phase II] Sub‑gradient optimisation started")
        sub_dead = deadline - 3.0
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, eta0=0.7
        )
        if val_sub < best_obj:
            best_obj, best_seq = val_sub, a_sub.copy()
            print(f"[Phase II] Sub‑gradient improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase III – optional simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase III] Simulated annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000
        )
        if val_ann < best_obj:
            best_obj, best_seq = val_ann, a_ann.copy()
            print(f"[Phase III] anneal improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_obj = 2.0 * n * (_full_convolution(best_seq, conv_fft_len).max())

    # Clip to the validator’s allowed range (no effect for our normalised vectors)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_obj:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
