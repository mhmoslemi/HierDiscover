"""
propose_candidate – long‑run single‑candidate optimiser

The function builds (or re‑uses) a non‑negative step function and improves it
by a sequence of deterministic optimisation phases:

1. **Adam on a smooth‑max surrogate** – a soft‑max (log‑sum‑exp) approximation
   of the peak of the autocorrelation is minimised.  The iteration count is
   limited only by the remaining time budget, allowing many hundred‑thousand
   updates.

2. **Exact‑max sub‑gradient (exponentiated‑gradient)** – after the Adam
   phase the true max‑convolution is used; the update is performed on the
   probability simplex by an exponentiated‑gradient step.

3. **Spectral flattening** – the FFT magnitude of the current vector is forced
   to be (almost) constant while staying on the simplex, a trick that reduces
   the height of the autocorrelation peak.

4. **Linear‑programming refinement** – we solve a small LP that maximises the
   L1‑norm of a new vector `b` under the constraint (f★b) ≤ M, where `M` is the
   current peak.  The feasible `b` is mixed with the current iterate.

5. **Final simulated‑annealing polish** – local moves on the simplex are
   accepted according to a cooling schedule.

All phases respect the 1000 s global deadline.  The best vector found
throughout the run is returned as a plain Python list of floats.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) for gradient computation.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # slightly larger for faster early progress
    lam = 0.2                         # start with a gentle smoothing
    lam_factor = 1.018                # grow a bit slower than the previous code

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- soft‑max re‑parameterisation -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & smooth‑max -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c          # equal to evaluate_sequence because Σa = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over the convolution values)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smoothed objective
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase smoothness
        lam = min(lam * lam_factor, 1e7)

        # modest learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_200_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # identify all indices attaining the current maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-15, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)    # diminishing step size

        a = a * np.exp(-eta * grad_a)    # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Force FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            # keep DC component unchanged, spread the remaining energy uniformly
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 6,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence a we maximise Σb subject to (a★b) ≤ M,
    where M = max (a★a).  The result b is normalised and mixed with a.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # pick a modest active set: all near‑max entries + a few well‑spaced ones
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # ----- build sparse Toeplitz constraint matrix for the active set -----
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # ----- linear program: maximise Σb  <=> minimise -Σb -----
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            break
        b_norm = b / sum_b

        # mix the new feasible vector with the current iterate
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            # no improvement – shrink the mixing factor
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a fraction of mass from i to j
        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.
    A single, long‑running optimisation is performed on a step function.
    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    # ------------------------------------------------------------------
    # Choose length.  If a warm start exists we keep its length, otherwise use 4096.
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        warm = np.asarray(warm, dtype=np.float64)
    else:
        warm = None

    if warm is not None and warm.size > 0:
        n = int(warm.shape[0])
        # ensure validity
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        n = 4096
        warm = None

    # If the warm vector is too small (e.g. < 2048), we enlarge n for more freedom.
    if n < 2048:
        n = 4096

    # ------------------------------------------------------------------
    # Prepare FFT lengths
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise the starting point
    if warm is not None:
        a0 = warm.copy()
    else:
        # a random Dirichlet start; using alpha=0.7 gives a mildly heavy‑tailed distribution
        a0 = np.random.dirichlet(0.7 * np.ones(n))
        a0 = a0 / a0.sum()

    best_seq = a0.copy()
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (≈ 45 % of time)
    remain = deadline - time.time()
    if remain > 5.0:
        sub_dead = min(deadline, time.time() + 0.45 * remain)
        print("[Phase 1] Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=5_000_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 1] Improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact‑max sub‑gradient (≈ 30 % of time)
    remain = deadline - time.time()
    if remain > 5.0:
        sub_dead = min(deadline, time.time() + 0.30 * remain)
        print("[Phase 2] Sub‑gradient (exp‑grad) optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=3_000_000, eta0=1.0,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase 2] Improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Spectral flattening (≈ 10 % of time)
    remain = deadline - time.time()
    if remain > 4.0:
        sub_dead = min(deadline, time.time() + 0.10 * remain)
        print("[Phase 3] Spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_iters=300_000
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Phase 3] Improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Linear‑programming refinement (≈ 10 % of time)
    remain = deadline - time.time()
    if remain > 3.0:
        sub_dead = min(deadline, time.time() + 0.10 * remain)
        print("[Phase 4] LP refinement")
        a_lp, val_lp = _lp_improve(best_seq, n, conv_fft_len, sub_dead, max_iter=7, tau=0.5)
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[Phase 4] Improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final annealing polish (≈ remaining time)
    remain = deadline - time.time()
    if remain > 2.0:
        sub_dead = deadline
        print("[Phase 5] Final simulated annealing")
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len, sub_dead, max_moves=120_000)
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase 5] Improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range and return as list
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        # Final sanity check (optional – not required for the return value)
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        # If evaluate_sequence is not available for some reason, just report the internal best.
        print(f"Final internal best value {best_val:.6f}")

    return best_seq.tolist()
