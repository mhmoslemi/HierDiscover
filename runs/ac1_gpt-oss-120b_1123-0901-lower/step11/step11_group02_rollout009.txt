"""
Hybrid optimiser – intensified local mass‑transfer with exact line‑search.

The algorithm
---------------
1. Build a diverse pool of candidate step‑functions (uniform, exponential,
   Gaussian, power‑law, Beta, spikes, etc.) and, if available, a warm‑start
   vector from the previous run.

2. For each candidate (a small number, e.g. 8) run a short Adam optimiser on
   a smooth‑max surrogate of the self‑convolution.  The best vector obtained
   (normalised to unit L1‑mass) is kept.

3. Starting from that best vector, perform a *fine‑grained mass‑transfer*
   local search:
   • Compute the exact sub‑gradient of the current max‑convolution.
   • Choose the source coordinate with the largest gradient and the target
     with the smallest gradient.
   • The effect of moving an amount ε from source → target on the entire
     convolution can be expressed analytically:
        conv′ = conv + 2·ε·Δ_lin + ε²·Δ_quad
     where Δ_lin and Δ_quad are simple shift‑vectors that can be built in
     O(n) time.
   • The objective (max(conv′)) is a convex function of ε on the interval
     [0, a_source].  We minimise it with a cheap ternary‑search (≈15
     evaluations) to obtain the *optimal* ε for this pair.
   • Apply the move if it strictly reduces the exact objective,
     update the convolution vector by the analytical formula and repeat.
   This exact line‑search yields considerably larger improvements than the
   former fixed‑fraction scheme.

4. If time permits, run a short projected‑gradient descent on the exact
   max‑convolution and a brief simulated‑annealing pass.

5. Clip the final vector to the evaluator’s allowed range, evaluate the
   exact score, and return the list of floats.

The routine respects the 1000 s budget, prints intermediate bounds and
guarantees a non‑empty return.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric or asymmetric Beta pdf on a uniform grid of mid‑points."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.2                # start a bit higher for faster smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _mass_transfer_optimize_exact(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Fine‑grained mass‑transfer optimisation.
    Uses the exact analytical effect of moving ε mass from a source index
    (largest gradient) to a target index (smallest gradient) and minimises
    the resulting max‑convolution via a short ternary search for the optimal ε.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)          # length 2n‑1
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        src_mass = a[i_high]
        if src_mass < 1e-14:
            break

        # Build linear and quadratic update terms
        delta_lin = np.zeros(2 * n - 1, dtype=np.float64)
        # a_{k-i_low}
        delta_lin[i_low: i_low + n] += a
        # -a_{k-i_high}
        delta_lin[i_high: i_high + n] -= a

        delta_quad = np.zeros(2 * n - 1, dtype=np.float64)
        idx_2low = 2 * i_low
        idx_2high = 2 * i_high
        idx_sum = i_low + i_high
        if idx_2low < delta_quad.size:
            delta_quad[idx_2low] = 1.0
        if idx_2high < delta_quad.size:
            delta_quad[idx_2high] = 1.0
        if idx_sum < delta_quad.size:
            delta_quad[idx_sum] -= 2.0

        # Define the objective as a function of ε (max of updated conv)
        def max_conv_at_eps(eps: float) -> float:
            return np.max(conv + 2.0 * eps * delta_lin + eps * eps * delta_quad)

        # Ternary search on a convex 1‑D function
        lo, hi = 0.0, src_mass
        # current value (eps = 0) is max_c
        for _ in range(15):                     # ≈ 1e‑5 relative precision
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = max_conv_at_eps(m1)
            f2 = max_conv_at_eps(m2)
            if f1 > f2:
                lo = m1
            else:
                hi = m2
        eps_opt = (lo + hi) / 2.0
        new_max = max_conv_at_eps(eps_opt)
        new_obj = 2.0 * n * new_max

        if new_obj < best_obj - 1e-12:
            # accept move
            a[i_high] -= eps_opt
            a[i_low] += eps_opt
            # update convolution exactly (avoid re‑FFT)
            conv = conv + 2.0 * eps_opt * delta_lin + eps_opt * eps_opt * delta_quad
            best_obj = new_obj
            best_a = a.copy()
            print(f"[MassTransferExact it={it:06d}] new best {best_obj:.6f}")
        else:
            # No improvement – try a tiny random nudge to escape possible plateaux
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps_rand = a[i] * 0.04
                a[i] -= eps_rand
                a[j] += eps_rand
                conv = _full_convolution(a, conv_fft_len)   # cheap reset
        it += 1

    return best_a, best_obj


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Short simulated‑annealing refinement.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2

        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ 1000 s.  Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final eval

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a rich pool of diverse initial vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential families
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0, 4.0):
        pool.append(_make_exponential(n, gamma))

    # Gaussian families
    for sigma in (0.04, 0.1, 0.2, 0.4, 0.8):
        pool.append(_make_gaussian(n, sigma))

    # Power‑law families
    for expo in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw(n, expo))

    # Simple shapes
    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))
    pool.append(_make_cosine(n))

    # Sparse spike patterns
    for k in (32, 64, 128, 256):
        pool.append(_make_spikes(n, k))

    # Beta distributions (including arcsine)
    for alpha, beta in [(0.5, 0.5), (0.8, 0.8), (1.0, 1.0),
                        (2.0, 2.0), (3.0, 3.0), (5.0, 5.0), (10.0, 10.0)]:
        pool.append(_make_beta(n, alpha, beta))

    # A few random Dirichlet draws
    for _ in range(6):
        alpha = 10 ** np.random.uniform(-3, 1)
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 8   # fewer, longer optimisation per start

    # ------------------------------------------------------------------
    # Phase I – short Adam runs on diverse starts
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(deadline, time.time() + 20.0)   # ~20 s per Adam run
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=250_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – exact mass‑transfer fine‑grained local search
    if best_seq is not None and time.time() < deadline - 12:
        print("[MassTransferExact] Starting refined local search")
        a_mt, val_mt = _mass_transfer_optimize_exact(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[MassTransferExact] improved best to {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Phase III – projected gradient descent (few seconds)
    if best_seq is not None and time.time() < deadline - 8:
        print("[GD] Short projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=100_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved best to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simulated annealing (optional, time‑bounded)
    if best_seq is not None and time.time() < deadline - 4:
        print("[Anneal] Final annealing pass")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase V – up‑sample to a finer grid if time permits
    if best_seq is not None and time.time() + 30 < deadline:
        try:
            print("[Upsample] Trying finer grid (8192 points)")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 12.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=150_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] refined best {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (using the provided evaluate_sequence)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
