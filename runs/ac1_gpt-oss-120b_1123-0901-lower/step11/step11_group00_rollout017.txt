def _local_swap_refine(a0, conv0, deadline, max_iters=2_000_000):
    a = a0.copy()
    n = len(a)
    conv = conv0.copy()
    cur_max = conv.max()
    cur_max_idx = conv.argmax()
    best_a = a.copy()
    best_val = 2.0*n*cur_max
    best_conv_max = cur_max
    it = 0
    # temperature schedule:
    T_start = 0.1
    T_end = 1e-6
    while time.time() < deadline and it < max_iters:
        # compute temperature
        T = T_start * (1 - it / max_iters) + T_end * (it / max_iters)
        # pick i,j distinct
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            # skip if zero mass
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # delta: random proportion up to some fraction of a[i]
        frac = np.random.rand() * 0.2  # up to 20%
        delta = a[i] * frac
        if delta <= 0:
            continue
        # compute new max after update using O(1)
        ai = a[i]
        aj = a[j]
        # update conv entries
        idx1 = 2*i
        idx2 = 2*j
        idx3 = i + j
        # store old conv values for potential revert
        old1 = conv[idx1]
        old2 = conv[idx2]
        old3 = conv[idx3]
        # compute changes
        conv[idx1] = old1 - 2*delta*ai + delta*delta
        conv[idx2] = old2 + 2*delta*aj + delta*delta
        conv[idx3] = old3 + 2*delta*(ai - aj) - 2*delta*delta
        # update a
        a[i] = ai - delta
        a[j] = aj + delta
        # compute new max
        new_max = cur_max
        # if any changed index equals cur_max_idx then recompute
        if cur_max_idx in (idx1, idx2, idx3):
            new_max = conv.max()
            new_max_idx = conv.argmax()
        else:
            # check if any changed value exceeds current max
            # compute max among three
            cand = max(conv[idx1], conv[idx2], conv[idx3])
            if cand > new_max:
                new_max = cand
                new_max_idx = conv.argmax() # not necessary if bigger
            else:
                new_max_idx = cur_max_idx
        # compute objective
        val = 2.0 * n * new_max
        # acceptance
        accept = False
        if val < best_val - 1e-12:
            accept = True
        else:
            # accept with prob
            if np.random.rand() < np.exp(-(val - best_val) / max(T, 1e-12)):
                accept = True
        if accept:
            cur_max = new_max
            cur_max_idx = new_max_idx
            if val < best_val:
                best_val = val
                best_a = a.copy()
                best_conv_max = new_max
                # maybe print progress occasionally
                if it % 50000 == 0:
                    print(f"[LocalRefine it={it}] new best {best_val:.6f}")
        else:
            # revert changes
            a[i] = ai
            a[j] = aj
            conv[idx1] = old1
            conv[idx2] = old2
            conv[idx3] = old3
        it += 1
    return best_a, best_val
```

"""
Hybrid search focused on the exponentiated‑gradient sub‑gradient method.

We construct a diverse set of initial step‑functions (uniform, Dirichlet,
beta‑distribution shapes, half‑sine, and a warm start if available) and
run short sub‑gradient refinements on each of them.  The best candidate
then receives a long sub‑gradient run (the bulk of the time‑budget).  To
polish the result we finish with a projected‑gradient descent pass and a
random‑phase refinement on the (almost) constant‑magnitude Fourier
manifold.  All iterates stay on the probability simplex (non‑negative,
Σa=1), so the evaluation value is simply 2·n·max(conv).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10**9,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation of
    2·n·max(conv) with the simplex constraint.
    Returns the best vector seen (as a probability vector) and its exact
    objective value (2·n·max(conv), i.e. the value returned by evaluate_sequence
    when the sum of the vector equals 1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=0.0, rtol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of the max‑convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing stepsize
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirrored on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=0.0, rtol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """Keep the FFT magnitude (including DC) fixed and repeatedly randomise the phases."""
    mag = np.abs(np.fft.fft(a0))
    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    half = n // 2
    it = 0

    while time.time() < deadline and it < max_iters:
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]
        A[1:half] = mag[1:half] * np.exp(1j * phi)
        A[-(half - 1):] = np.conj(A[1:half][::-1])
        if n % 2 == 0:
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val - 1e-12:
            best_val = val
            a_best = a.copy()
            if it % 15000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start and simple shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws (different concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Beta‑distribution shapes (a,b>0)
    for a, b in ((0.5, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)):
        x = (np.arange(n) + 0.5) / n
        shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Exponential envelopes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law family (indices product)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian peaks at the ends
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        g = np.exp(-0.5 * (i / (sigma * n)) ** 2) + np.exp(
            -0.5 * ((i - (n - 1)) / (sigma * n)) ** 2
        )
        g /= g.sum()
        cand.append(g)

    # Half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand
