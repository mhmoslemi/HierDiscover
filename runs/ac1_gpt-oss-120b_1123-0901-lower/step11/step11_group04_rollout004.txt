def propose_candidate(seed=0, budget_s=1000):
    # set seeds
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # constants
    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)
    target = np.full(2*n-1, 1.0/(2*n-1), dtype=np.float64)

    best_seq = None
    best_val = float('inf')
    
    # define candidate generator functions list
    candidates = [] # list of vectors
    
    # Add warm start if exists
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] != n:
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        s = warm_arr.sum()
        if s >= 0.01:
            warm_arr = warm_arr / s
        else:
            warm_arr = None
        if warm_arr is not None:
            candidates.append(warm_arr)

    # Uniform
    candidates.append(_make_uniform(n))

    # Dirichlet with various alpha
    for alpha in (0.01,0.05,0.2,0.5,1.0):
        candidates.append(_make_dirichlet(n, alpha))

    # Exponential family parameter sweep
    for gamma in (0.1,0.2,0.5,0.8,1.0,1.5,2.0):
        candidates.append(_make_exponential(n, gamma))

    # Powerlaw sweep
    for expo in (-3.0,-2.0,-1.5,-1.2,-1.0,-0.8,-0.5):
        candidates.append(_make_powerlaw(n, expo))

    # Triangle
    candidates.append(_make_triangle(n))

    # Cosine power sweep
    for p in (0.5,1.0,2.0,4.0,8.0):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i+0.5) / n) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)

    # Mixed exponential+uniform
    for alpha in (0.3,0.5,0.7):
        for gamma in (0.2,0.5,1.0):
            exp = _make_exponential(n, gamma)
            mix = alpha * _make_uniform(n) + (1-alpha) * exp
            mix /= mix.sum()
            candidates.append(mix)
    
    # Add random perturbations of existing candidates later
    # Main optimization loop
    for idx, a0 in enumerate(candidates):
        if time.time() > deadline:
            break
        # ensure valid
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        # Phase 1: L2 Adam (flatten)
        phase_time = min(12.0, deadline - time.time() - 15) # keep buffer
        if phase_time <= 0:
            break
        phase_dead = time.time() + phase_time
        a_l2, val_l2 = _adam_optimize_l2(a0, n, conv_fft_len, corr_fft_len, target,
                                         phase_dead, max_iters=300_000, lr=0.07)
        # update best
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Candidate {idx}] L2 Adam val {val_l2:.6f}")
        # Phase 2: Smooth Max Adam
        if time.time() < deadline - 10:
            phase_time = min(8.0, deadline - time.time() - 10)
            phase_dead = time.time() + phase_time
            a_smooth, val_smooth = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len,
                                                  phase_dead, max_iters=250_000)
            if val_smooth < best_val:
                best_val = val_smooth
                best_seq = a_smooth.copy()
                print(f"[Candidate {idx}] Smooth Adam val {val_smooth:.6f}")
        # Phase 3: Exponentiated Gradient
        if time.time() < deadline - 8:
            phase_time = min(6.0, deadline - time.time() - 8)
            phase_dead = time.time() + phase_time
            a_exp, val_exp = _subgradient_optimize_expgrad(best_seq, n, conv_fft_len,
                                                          corr_fft_len, phase_dead,
                                                          max_iters=400_000, eta0=0.9)
            if val_exp < best_val:
                best_val = val_exp
                best_seq = a_exp.copy()
                print(f"[Candidate {idx}] ExpGrad val {val_exp:.6f}")
    
    # After candidate loop, perform final fine tuning
    if best_seq is not None and time.time() < deadline - 12:
        print("[Fine] Performing projected GD")
        a_gd, val_gd = _max_conv_descent(best_seq, n, conv_fft_len, corr_fft_len, deadline,
                                         max_iters=400_000)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    if best_seq is not None and time.time() < deadline - 10:
        print("[Fine] Simulated annealing")
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len, deadline,
                                          max_moves=150_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # Guarantee return
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = float('inf')
    # Clip and final evaluate
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
    return best_seq.tolist()

"""
Hybrid multi‑start optimiser for the step‑function evaluation problem.

The routine builds a diverse pool of candidate probability vectors:
  - warm start (if a previous solution exists)
  - uniform, exponential, power‑law, triangular, cosine‑power families
  - mixtures of exponential and uniform
  - random Dirichlet draws with several concentrations
  - sparse random supports (very few non‑zero entries)

Each candidate is first flattened by L2‑Adam (tries to make the
self‑convolution close to the uniform target).  The result is then
fine‑tuned with a smooth‑max Adam optimiser, followed by an exact‑max
sub‑gradient phase (exponentiated‑gradient).  The best vector found is
finally polished with projected gradient descent and a short simulated‑
annealing refinement.  The whole search respects the given time budget
(~1000 s) and always returns a non‑negative normalised sequence.

The objective that is minimised is
    2·n·max(conv(a,a)) / (sum(a)^2)  (the evaluate_sequence() function).
All intermediate vectors are kept on the probability simplex (sum = 1)
so the denominator can be ignored during optimisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Factory functions for diverse initial vectors
def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_cosine_power(n, p):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_mixed_exp_uniform(n, alpha, gamma):
    exp = _make_exponential(n, gamma)
    uni = _make_uniform(n)
    a = alpha * uni + (1 - alpha) * exp
    a /= a.sum()
    return a


def _make_sparse_random(n, k):
    """Create a probability vector with exactly k equal non‑zero entries."""
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max (as in the reference code)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser – L2‑flattening phase
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of (conv‑target)^2 w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # evaluate current real objective (max‑conv)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser – exponentiated‑gradient (exact‑max)
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the exact max‑convolution
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main search routine
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget, explores many analytic families,
    performs L2‑flattening, smooth‑max Adam, exact‑max sub‑gradient,
    and finishes with projected GD + simulated annealing.
    Returns the best sequence found (as a list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size (feel free to increase if time permits)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Uniform target for the L2‑flattening stage
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Gather a pool of diverse starting vectors
    pool = []

    # Warm start if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w.copy())

    # Analytic families
    pool.append(_make_uniform(n))
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))
    for gamma in (0.1, 0.2, 0.5, 0.8, 1.0, 1.5, 2.0):
        pool.append(_make_exponential(n, gamma))
    for expo in (-3.0, -2.0, -1.5, -1.2, -1.0, -0.8, -0.5):
        pool.append(_make_powerlaw(n, expo))
    pool.append(_make_triangle(n))
    for p in (0.5, 1.0, 2.0, 4.0, 8.0):
        pool.append(_make_cosine_power(n, p))
    for alpha in (0.3, 0.5, 0.7):
        for gamma in (0.2, 0.5, 1.0):
            pool.append(_make_mixed_exp_uniform(n, alpha, gamma))

    # Random sparse supports (helps escaping local minima)
    for _ in range(8):
        k = np.random.randint(80, 200)  # number of spikes
        pool.append(_make_sparse_random(n, k))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optimisation loop over candidates
    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ------- Phase 1: L2‑flattening (≈12 s) ------------
        remaining = deadline - time.time()
        phase_time = min(12.0, remaining - 20.0)  # keep buffer for later phases
        if phase_time <= 0:
            break
        phase_dead = time.time() + phase_time
        a_l2, val_l2 = _adam_optimize_l2(
            a0, n, conv_fft_len, corr_fft_len, target,
            phase_dead, max_iters=300_000, lr=0.07
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Candidate {idx}] L2‑Adam improved to {val_l2:.6f}")

        # ------- Phase 2: smooth‑max Adam (≈8 s) ----------
        if time.time() < deadline - 12:
            remaining = deadline - time.time()
            phase_time = min(8.0, remaining - 12.0)
            if phase_time > 0:
                phase_dead = time.time() + phase_time
                a_smooth, val_smooth = _adam_optimize(
                    best_seq, n, conv_fft_len, corr_fft_len,
                    phase_dead, max_iters=250_000
                )
                if val_smooth < best_val:
                    best_val = val_smooth
                    best_seq = a_smooth.copy()
                    print(f"[Candidate {idx}] smooth‑max Adam improved to {val_smooth:.6f}")

        # ------- Phase 3: exact‑max sub‑gradient (≈6 s) ---
        if time.time() < deadline - 8:
            remaining = deadline - time.time()
            phase_time = min(6.0, remaining - 8.0)
            if phase_time > 0:
                phase_dead = time.time() + phase_time
                a_exp, val_exp = _subgradient_optimize_expgrad(
                    best_seq, n, conv_fft_len, corr_fft_len,
                    phase_dead, max_iters=400_000, eta0=0.9
                )
                if val_exp < best_val:
                    best_val = val_exp
                    best_seq = a_exp.copy()
                    print(f"[Candidate {idx}] ExpGrad improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Global fine‑tuning with projected GD
    if best_seq is not None and time.time() < deadline - 15:
        print("[Fine] Projected gradient descent on exact max‑conv")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=400_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 8:
        print("[Fine] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=150_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Fine] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = float("inf")

    # Clip to allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation (for reporting only)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
