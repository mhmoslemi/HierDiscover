"""
Hybrid multi‑stage optimizer for the step‑function attack.

The routine explores a diverse set of initial candidates and then
iteratively refines the current best sequence using a mix of
  * Adam on a smooth‑max surrogate of the autocorrelation peak,
  * Exact sub‑gradient (exponentiated‑gradient) on the true max,
  * Spectral‑flattening (forcing the FFT magnitude to be (almost) constant),
  * Linear‑programming improvement (the LP‑step described in the statement),
  * Projected gradient descent on the exact max‑convolution,
  * Simulated‑annealing polish.

All phases are run repeatedly while time permits, each receiving a
time‑slice that shrinks as the overall budget (default 1000 s) is
consumed.  The algorithm works on the probability simplex
(∑a_i = 1, a_i ≥ 0); the evaluation function is homogeneous, so this
normalisation does not affect the final score.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(convolution)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                 # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad iter={it:06d}] new best {val:.6f}")

        # set of near‑max indices (relaxed tolerance helps far from exact ties)
        eps = max(1e-12, 1e-8 * max_c)
        active = conv >= max_c - eps
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0: np.ndarray, n: int,
                     conv_fft_len: int, corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # use a slightly relaxed active set
        eps = max(1e-12, 1e-8 * max_c)
        active = conv >= max_c - eps
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.15 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD iter={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45   # move up to 45 % of a[i]
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(a0: np.ndarray, n: int,
                               conv_fft_len: int,
                               deadline: float,
                               max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral iter={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        # Keep the DC component, flatten the rest
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _lp_improve(a0: np.ndarray, n: int,
                conv_fft_len: int, deadline: float,
                max_iter: int = 8, tau: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence f we maximise Σb subject to (f★b)_k ≤ M,
    where M = max_k (f★f)_k.  The solution b is normalised and mixed with f.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active set (indices where conv is close to M)
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # build sparse constraint matrix for (f★b)_k ≤ M
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # Linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – returns a non‑negative step function (list of floats)
    that (hopefully) achieves a low value of `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start, if a previous solution is available.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build an initial pool of diverse candidates.
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # Add a handful of fully random Dirichlet draws.
    for _ in range(20):
        pool.append(np.random.dirichlet(np.ones(n)))

    # Random piecewise‑constant blocks (helps explore low‑frequency shapes).
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 10    # number of start vectors we invest heavy time in

    # ------------------------------------------------------------------
    # Phase I – deep optimisation of a few promising starts.
    for idx, a0 in enumerate(pool[:max_candidates]):
        if time.time() > deadline:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # allocate ~30 s for the whole cascade on this start
        sub_dead = min(deadline, time.time() + 30.0)

        a_cur = a0
        # 1) Adam (smooth‑max)
        a_cur, val_cur = _adam_optimize(
            a_cur, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=250_000)
        # 2) Exact sub‑gradient (exponentiated‑gradient)
        if time.time() < deadline - 5:
            a_cur, val_cur = _subgradient_optimize_expgrad(
                a_cur, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=250_000, eta0=1.0)
        # 3) Spectral flattening
        if time.time() < deadline - 5:
            a_cur, val_cur = _spectral_flatten_optimize(
                a_cur, n, conv_fft_len, deadline, max_iters=120_000)
        # 4) LP improvement (more iterations than before)
        if time.time() < deadline - 5:
            a_cur, val_cur = _lp_improve(
                a_cur, n, conv_fft_len, deadline, max_iter=12, tau=0.5)
        # 5) Projected gradient descent on the exact max‑conv
        if time.time() < deadline - 5:
            a_cur, val_cur = _max_conv_descent(
                a_cur, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=150_000)
        # 6) Final annealing polish
        if time.time() < deadline - 5:
            a_cur, val_cur = _anneal_optimize(
                a_cur, n, conv_fft_len, deadline, max_moves=40_000)

        # Evaluate with the supplied (slow) evaluator to guard against
        # any accidental overflow/NaN handling.
        try:
            v = evaluate_sequence(a_cur.tolist())
        except Exception as e:      # should never happen, but be robust
            print(f"[Candidate {idx}] evaluation error: {e}")
            v = float("inf")

        if v < best_val:
            best_val, best_seq = v, a_cur.copy()
            print(f"[Candidate {idx}] NEW OVERALL BEST {v:.6f}")

    # ------------------------------------------------------------------
    # Phase II – exhaustive refinement of the current best.
    # Re‑run short cycles of the six phases while time permits.
    while best_seq is not None and time.time() < deadline - 5:
        remaining = deadline - time.time()
        # give each sub‑phase a modest slice of the remaining time
        slice_frac = 0.12
        sub_dead = time.time() + remaining * slice_frac

        # Adam
        a_tmp, val_tmp = _adam_optimize(best_seq, n, conv_fft_len,
                                        corr_fft_len, sub_dead,
                                        max_iters=150_000)
        if val_tmp < best_val:
            best_val, best_seq = val_tmp, a_tmp.copy()
            print(f"[Refine] Adam improved to {val_tmp:.6f}")

        # Sub‑gradient
        if time.time() < deadline - 5:
            a_tmp, val_tmp = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=150_000, eta0=0.8)
            if val_tmp < best_val:
                best_val, best_seq = val_tmp, a_tmp.copy()
                print(f"[Refine] Subgrad improved to {val_tmp:.6f}")

        # Spectral flattening
        if time.time() < deadline - 5:
            a_tmp, val_tmp = _spectral_flatten_optimize(
                best_seq, n, conv_fft_len, deadline, max_iters=80_000)
            if val_tmp < best_val:
                best_val, best_seq = val_tmp, a_tmp.copy()
                print(f"[Refine] Spectral improved to {val_tmp:.6f}")

        # LP refinement
        if time.time() < deadline - 5:
            a_tmp, val_tmp = _lp_improve(
                best_seq, n, conv_fft_len, deadline,
                max_iter=8, tau=0.4)
            if val_tmp < best_val:
                best_val, best_seq = val_tmp, a_tmp.copy()
                print(f"[Refine] LP improved to {val_tmp:.6f}")

        # Gradient descent
        if time.time() < deadline - 5:
            a_tmp, val_tmp = _max_conv_descent(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=120_000)
            if val_tmp < best_val:
                best_val, best_seq = val_tmp, a_tmp.copy()
                print(f"[Refine] GD improved to {val_tmp:.6f}")

        # Annealing
        if time.time() < deadline - 5:
            a_tmp, val_tmp = _anneal_optimize(
                best_seq, n, conv_fft_len, deadline, max_moves=30_000)
            if val_tmp < best_val:
                best_val, best_seq = val_tmp, a_tmp.copy()
                print(f"[Refine] Anneal improved to {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty, valid return.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print("[Fallback] returning uniform distribution")

    # Clip to the evaluator’s allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
