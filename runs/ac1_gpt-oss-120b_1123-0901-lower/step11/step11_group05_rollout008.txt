"""
Hybrid multi‑restart optimiser for the max‑convolution bound.

The routine builds a diverse set of seed vectors (uniform, beta‑shaped,
exponential‑decay, Dirichlet draws and a possible warm‑start).  After a
cheap evaluation the best seed is taken as the initial point.

The main optimisation consists of repeated exponentiated‑gradient (EG)
sub‑gradient phases started from random perturbations of the current
best vector.  The EG update
    a ← a * exp(–η·(∇ – <a,∇>))
preserves non‑negativity and the simplex constraint after the
normalisation step.  Multiple restarts explore different basins and
allow the limited time budget (≈1000 s) to be spent where it matters.

When the remaining time becomes small a short Adam optimiser on a
smooth‑max surrogate is run, followed by a few deterministic mixing
steps (with the uniform and mirrored vectors) and a tiny stochastic
local‑search/annealing phase.  All candidates are projected onto the
probability simplex {x ≥ 0, Σx = 1} before evaluation.

The final sequence is clipped to the allowed range, re‑projected onto
the simplex and returned as a plain Python list.  Throughout the
run progress and intermediate bounds are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolution)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradient / Adam.
    Both arguments are real arrays; w has length 2·len(a)‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = 1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation of max‑convolution.
    Stops when the time deadline is reached.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[EG it={it:07d}] new best {val:.6f}")

        # active set (indices where conv attains the maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (mirrored‑descent with KL geometry)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max surrogate.
    The temperature λ is increased geometrically up to a very large
    value so that the surrogate converges to the true max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # start temperature
    lam_factor = 1.03
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    a valid sequence (list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    # problem size – 4096 works well and matches the reference code
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0) build a pool of diverse seeds
    candidates = []

    # warm start if present in the global namespace
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform distribution
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric Beta‑shaped (U‑shaped for α<1)
    alphas = np.arange(0.2, 1.01, 0.1)   # explore more extreme shapes
    x = (np.arange(n) + 0.5) / n
    for alpha in alphas:
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # exponential‑decay family (spikes at both ends)
    taus = [0.0005, 0.001, 0.002, 0.004, 0.008]
    idx = np.arange(n)
    for tau in taus:
        a = np.exp(-tau * idx) + np.exp(-tau * (n - 1 - idx))
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        candidates.append(a)

    # random Dirichlet draws (different concentration levels)
    for _ in range(20):
        conc = np.random.uniform(0.3, 5.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # --------------------------------------------------------------
    # 1) quick evaluation of the seeds, keep the best
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] seed objective {val:.6f}")

    # --------------------------------------------------------------
    # 2) multi‑restart exponentiated‑gradient phase
    # Reserve a few seconds for later fine‑tuning.
    while time.time() < deadline - 30.0:
        # pick a random seed (sometimes perturb the current best)
        if np.random.rand() < 0.4:
            a0 = best_a.copy()
            # add a small random perturbation before the next run
            perturb = np.random.randn(n) * 0.005
            a0 = np.clip(a0 + perturb, 0.0, None)
            a0 = _project_simplex(a0)
        else:
            a0 = np.random.choice(candidates)

        # allocate a modest slice of remaining time for this run
        run_deadline = min(deadline - 5.0, time.time() + 12.0)

        a_opt, val_opt = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, run_deadline,
            max_iters=3_000_000, eta0=1.8
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Restart] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 3) short Adam fine‑tuning on the smooth‑max surrogate
    adam_deadline = deadline - 8.0
    if adam_deadline > time.time():
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=250_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) deterministic mixing with the uniform and mirrored vectors
    uniform_a = np.full(n, 1.0 / n, dtype=np.float64)
    rev_a = best_a[::-1]
    for t in np.linspace(0.0, 0.2, 21):
        for base in (uniform_a, rev_a):
            cand = (1.0 - t) * best_a + t * base
            cand = _project_simplex(cand)
            val = _objective(cand, n, conv_fft_len)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Mix t={t:.3f}] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 5) tiny stochastic local‑search / annealing for the final seconds
    temp = 0.8
    local_deadline = deadline - 0.5
    while time.time() < local_deadline:
        # pick a pair with bias: try to move mass away from indices that
        # have large gradient (i.e. contribute to the active max)
        i, j = np.random.choice(n, 2, replace=False)
        # The amount of mass moved is a small fraction of the donor entry
        delta = min(best_a[i], 5e-4) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        val = _objective(cand, n, conv_fft_len)
        # Simulated‑annealing acceptance
        if val < best_val or np.random.rand() < np.exp(-(val - best_val) / temp):
            best_val, best_a = val, cand.copy()
            if val < best_val * 0.9999:
                print(f"[Anneal] improved to {val:.6f}")
        # geometric cooling
        temp = max(0.01, temp * 0.99995)

    # --------------------------------------------------------------
    # 6) final sanity checks and return
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)

    # official evaluation (mainly for logging)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
