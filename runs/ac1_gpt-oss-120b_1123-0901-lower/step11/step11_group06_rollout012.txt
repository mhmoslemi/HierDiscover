def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 8192  # increase size
    conv_fft_len = _next_pow2(2 * n - 1)
    # prepare initial list of candidates: various shapes
    candidates = []
    # Warm start if existing
    if 'height_sequence_1' in globals():
        warm = np.asarray(height_sequence_1, dtype=float)
        if warm.shape[0] != n:
            # reshape by interpolation or truncation/padding
            warm = np.copy(warm[:n]) if warm.shape[0] > n else np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
            candidates.append(warm)
    # Add some Dirichlet draws
    for alpha in (0.1, 0.5, 1.0, 2.0, 5.0):
        candidates.append(np.random.dirichlet(alpha * np.ones(n)))
    # exponential tails left-heavy
    for gamma in (0.001, 0.003, 0.01):
        shape = np.exp(-gamma * np.arange(n))
        shape /= shape.sum()
        candidates.append(shape)
    # monotonic decreasing (power law)
    idx = np.arange(1, n + 1, dtype=float)
    shape = 1.0 / idx**0.5
    shape /= shape.sum()
    candidates.append(shape)

    best_seq = None
    best_val = float('inf')

    # For each candidate run spectral flattening for a slice of time
    for i, a0 in enumerate(candidates):
        if time.time() > deadline:
            break
        # allocate budget per candidate (by remaining time / remaining candidates)
        remaining = deadline - time.time()
        remaining_cands = len(candidates) - i
        sub_dead = time.time() + remaining / remaining_cands
        
        a_opt, val_opt = _spectral_flatten_optimize_improved(a0, n, conv_fft_len, sub_dead, max_iters=5_000_000)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt
            print(f"[Candidate {i}] improved to {val_opt:.6f}")

    # After spectral stage, apply short gradient or annealing refinement if time left
    if best_seq is not None and time.time() < deadline - 5.0:
        # gradient descent (exact max) to fine tune
        a_gd, val_gd = _max_conv_descent(best_seq, n, conv_fft_len, conv_fft_len, deadline, max_iters=300_000)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd
            print(f"[GD refine] improved to {val_gd:.6f}")

    # fallback uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

def _spectral_flatten_optimize_improved(a0, n, fft_len, deadline, max_iters=500000):
    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    L = fft_len
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral imp it={it}] new best {val:.6f}")

        # FFT
        A = np.fft.rfft(a, L)
        # total energy multiplied by L (Parseval)
        energy = np.dot(a, a)   # sum_i a_i^2
        S = np.sum(a)           # =1 after projection
        # compute target magnitude for non-DC components using L
        # total magnitude squared = L * energy
        remaining_energy = max(0.0, L * energy - (abs(A[0])**2))
        const_mag = np.sqrt(remaining_energy / (L/2 - 1)) if L > 2 else 0.0
        # Note: rfft length = L//2 +1 = L_half+1. There are L_half-1 non-DC frequencies (excluding DC and Nyquist if L even)
        # For even L, rfft includes Nyquist term at index L/2, which corresponds to frequency pi, treat same as other non-DC.
        # We'll set magnitude for all indices 1..len(A)-1 to const_mag.
        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if len(A) > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.fft.irfft(A_new, L)[:n]  # take first n entries (since padded zeros beyond n)
        # projection onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new
        it += 1
    return best_a, best_val

Careful about mapping between rfft length and non-DC rfft entries: For real FFT length L (e.g., 8192), rfft returns L//2+1 complex numbers: frequencies 0..L/2 inclusive. The DC is index 0, Nyquist at index L/2 (real). The number of non-DC freq components is (L//2) (including Nyquist) for even L. So we can set magnitude constant for indices 1..len(A)-1.

Thus const_mag = sqrt(remaining_energy / (len(A)-1)). Because sum of squared magnitudes of all amplitude components must equal L*energy. So we use:

num_non_dc = len(A) - 1

"""
Propose a low‑score step‑function sequence.

The core idea:
*   The objective = 2·n·max(conv(a,a)) for any a with sum(a)=1.
    Therefore we only need to minimise the peak of the linear
    convolution a⋆a.
*   If the magnitude of all non‑DC Fourier coefficients of the
    zero‑padded vector is forced to be (almost) equal, the resulting
    convolution becomes almost flat (its values are close to the
    average 1/(2n‑1)).  This “spectral‑flattening’’ is the main
    optimisation tool.
*   The previous implementation used the original vector length (n)
    when redistributing the energy among the frequencies; here we use
    the true FFT length (the zero‑padded length) which respects Parseval
    exactly and gives a stronger flattening effect.
*   We start from several diverse seeds (warm start, Dirichlet draws,
    exponential tails, power‑law decay) and run an alternating‑projection
    loop (FFT → magnitude equalisation → inverse FFT → simplex projection)
    for each seed for a time‑slice of the total budget.
*   The best sequence is then polished with a few steps of exact‑max
    projected gradient descent.
*   All operations stay on the probability simplex, so scaling does not
    affect the score.  The final vector is clipped only to satisfy the
    evaluator’s range limits.

The routine respects the 1000 s budget and always returns the best
sequence found (or a uniform fallback).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a⋆a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Alternating‑projection spectral flattening.

    *   Forward FFT of the zero‑padded vector (length = fft_len).
    *   Keep the DC term, force all other magnitudes to the same value
        that preserves the total energy (Parseval’s identity with the
        *actual* FFT length).
    *   Inverse FFT, keep only the first n entries, project onto the
        probability simplex.
    *   Periodically evaluate the exact objective and keep the best
        iterate.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0
    L = fft_len                     # total FFT length (including zero‑padding)

    while time.time() < deadline and it < max_iters:
        # ----- evaluate current candidate -----
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                # sum(a)=1 ⇒ denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")

        # ----- spectral equalisation -----
        # full complex FFT (size L)
        A = np.fft.fft(a, L)

        energy = float(np.dot(a, a))         # Σ a_i²
        # Parseval: Σ |A_k|² = L * energy
        remaining_energy = max(0.0, L * energy - (abs(A[0]) ** 2))
        num_non_dc = L - 1
        const_mag = np.sqrt(remaining_energy / num_non_dc) if num_non_dc > 0 else 0.0

        A_new = np.empty_like(A)
        A_new[0] = A[0]                       # keep DC (total mass)
        A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        # back to time domain, keep only the first n entries (rest are padding zeros)
        a_new = np.real(np.fft.ifft(A_new))[:n]

        # simplex projection (non‑negative + unit sum)
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        # gradient of the max‑convolution w.r.t. a
        # (cross‑correlation of w with a in the linear sense)
        # use FFT‑based correlation
        rev_a = a[::-1]
        fw = np.fft.rfft(w, conv_fft_len)
        fr = np.fft.rfft(rev_a, conv_fft_len)
        grad = 2.0 * np.fft.irfft(fw * fr, conv_fft_len)[:n]

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The time budget (~1000 s) is split into several short phases:
    *   Build a small diverse pool of seeds (warm start, Dirichlet draws,
        exponential tails, power‑law decay).
    *   Run the improved spectral‑flattening routine on each seed for a
        time slice.
    *   Refine the best result with a short projected‑gradient descent.
    *   Return the best sequence found (clipped to the evaluator limits).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Length of the vector – 8192 gives a finer lower bound (≈1.00006)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)      # zero‑pad length for exact convolutions

    # ------------------------------------------------------------------
    # Warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0.0:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Construct a modest but diverse pool of starting vectors
    pool = []

    if warm is not None:
        pool.append(warm)

    # Dirichlet draws with several concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails – left‑handed (asymmetric)
    for gamma in (0.001, 0.003, 0.01):
        shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law decay (inverse sqrt)
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = 1.0 / np.sqrt(idx)
    shape /= shape.sum()
    pool.append(shape)

    # Random permutation of a Dirichlet to add a bit of noise
    shape = np.random.dirichlet(np.ones(n))
    np.random.shuffle(shape)
    pool.append(shape)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – spectral flattening on several seeds
    max_candidates = min(10, len(pool))   # limit to keep each candidate sizable

    for idx, a0 in enumerate(pool[:max_candidates]):
        if time.time() > deadline:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Allocate roughly equal remaining time for the remaining candidates
        remain = deadline - time.time()
        remain_per_cand = remain / (max_candidates - idx)
        sub_deadline = time.time() + min(30.0, remain_per_cand)   # at most ~30 s per seed

        a_opt, val_opt = _spectral_flatten_optimize(
            a0, n, conv_fft_len, sub_deadline,
            max_iters=2_000_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Seed {idx}] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase II – short projected‑gradient descent refinement
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Refine] Projected gradient descent")
        a_ref, val_ref = _max_conv_descent(
            best_seq, n, conv_fft_len, deadline,
            max_iters=300_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (guarantees a valid return)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range [0, 1000]
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
