"""
Propose a step‑function (non‑negative vector) with very low
evaluate_sequence() score.

Algorithm
---------
1. Initialise a candidate (warm‑start if a global `height_sequence_1`
   exists, otherwise start from a few parametric families).
2. Sweep a few one‑parameter families (exponential, power‑law,
   beta, Gaussian) on a dense grid; keep the best vector seen.
3. Refine the current best with a simple projected sub‑gradient descent
   on the exact L∞‑norm of the self‑convolution.  The step size decays
   as 1/√iteration and the iterate is projected onto the probability
   simplex after each step.
4. Polish the result with a fast simulated‑annealing local search that
   moves a tiny amount of mass between two random coordinates.
5. Return the best vector (as a plain list) before the 1000 s budget
   expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (result length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Parametric families (all return a vector with sum = 1)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_half_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a

    a = best_seq.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()

    T_start = 0.08
    T_end = 1e-5
    moves = 0
    max_moves = int(5e5)   # will be limited by remaining time automatically
    while time.time() < deadline - 0.5 and moves < max_moves:
        T = max(T_end, T_start * (1.0 - moves / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue

        # move a random fraction of the mass at i towards j
        frac = np.random.rand() * 0.3
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < cur_val or np.random.rand() < np.exp(-(val2 - cur_val) / T):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_seq = a2.copy()
                print(f"[SA move={moves}] new best {val2:.6f}")

        moves += 1

    # ------------------------------------------------------------------
    # Final safety post‑processing
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:                     # avoid the degenerate case
        best_seq[:] = 1.0 / len(best_seq)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
