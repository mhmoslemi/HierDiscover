def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective value for a (no clipping or scaling)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):

def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):

    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):

    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_cosine(n):

def _make_arcsine(n):
    """Discrete arcsine (Beta(0.5,0.5)) – 1/√(x(1‑x))."""

def _make_powerlaw_symmetric(n, alpha):
    """Symmetric power‑law a_i ∝ i^{-α} + (n‑i)^{-α}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha) + (n - i + 1) ** (-alpha)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric (or asymmetric) Beta pdf on the mid‑points of a uniform grid."""
    i = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a

# Optimisers (adapted from previous state‑of‑the‑art) ------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max parametrisation
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # sum(a)=1 after soft‑max
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if grad[i_high] <= grad[i_low] + 1e-15:
            break  # near‑stationary

        src_mass = a[i_high]
        improved = False
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # small random nudge to try to escape plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine builds a rich pool of analytic shapes, keeps the most
    promising candidates, refines them with Adam + exponentiated‑gradient
    + fine‑grained mass‑transfer, and finally returns the best vector.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # -------------------- warm start handling -------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # -------------------- build candidate pool ------------------------
    pool = []

    # warm‑start and its cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet family
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential family
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0, 4.0):
        pool.append(_make_exponential(n, gamma))

    # Gaussian family
    for sigma in (0.04, 0.1, 0.2, 0.4, 0.8):
        pool.append(_make_gaussian(n, sigma))

    # Classic simple shapes
    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))
    pool.append(_make_cosine(n))

    # New shapes -------------------------------------------------------
    pool.append(_make_inverted_cosine(n))
    pool.append(_make_arcsine(n))
    for expo in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw_symmetric(n, expo))

    # Symmetric Beta sweep
    beta_vals = np.linspace(0.2, 5.0, 12)
    for a_alpha in beta_vals:
        for b_beta in beta_vals:
            pool.append(_make_beta(n, a_alpha, b_beta))

    # Sparse spikes
    for k in (32, 64, 128, 256):
        pool.append(_make_spikes(n, k))

    # Random Dirichlet draws
    for _ in range(8):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # -------------------- fast evaluation of all candidates ----------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    fast_times = []

    print("[Stage I] Evaluating a large pool of candidates (fast) …")
    for idx, a in enumerate(pool):
        if time.time() > deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            continue
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Pool] new overall best {val:.6f} from candidate {idx}")
        # keep the best 12 seeds for deeper work
        fast_times.append((val, a.copy()))

    # keep the 12 lowest‑value candidates
    fast_times.sort(key=lambda t: t[0])
    seed_vectors = [vec for _, vec in fast_times[:12]]
    print(f"[Stage I] selected {len(seed_vectors)} seeds for refinement")

    # -------------------- deep optimisation of seeds -----------------
    for seed_idx, a0 in enumerate(seed_vectors):
        if time.time() > deadline:
            break
        print(f"[Seed {seed_idx}] start deep optimisation")
        a0 = a0 / a0.sum()

        # time slice for this seed (~budget / remaining seeds)
        sub_dead = min(deadline, time.time() + max(5.0, (deadline - time.time()) / (len(seed_vectors) - seed_idx)))

        # 1) Adam on smooth‑max surrogate
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Seed {seed_idx}] Adam improved to {val_adam:.6f}")

        # 2) Exponentiated‑gradient sub‑gradient phase
        if time.time() < deadline - 5:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=500_000, eta0=0.7
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Seed {seed_idx}] ExpGrad improved to {val_sub:.6f}")

        # 3) Mass‑transfer fine‑grained local search
        if time.time() < deadline - 2:
            a_mt, val_mt = _mass_transfer_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=150_000
            )
            if val_mt < best_val:
                best_val = val_mt
                best_seq = a_mt.copy()
                print(f"[Seed {seed_idx}] MassTransfer improved to {val_mt:.6f}")

    # -------------------- final long mass‑transfer refinement -------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Final] extensive mass‑transfer refinement")
        a_final, val_final = _mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final] improved best to {val_final:.6f}")

    # -------------------- ensure non‑empty return --------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (just for logging)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
