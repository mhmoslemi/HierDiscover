def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    n = 4096

    corr_fft_len = _next_pow2(3*n-2)

    best_seq = None
    best_val = float('inf')
    
    # Derived functions: fast_eval(a) using FFT conv.
    
    def fast_val(a):
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()
    
    # Phase: random spectral flattening search
    while time.time() < deadline - 30.0:
        # generate random Dirichlet
        alpha = np.random.uniform(0.001, 10)  # random concentration
        a = np.random.dirichlet(np.ones(n) * alpha)  # shape maybe
        # optionally simple exponential shape:
        if np.random.rand() < 0.3:
            gamma = np.random.uniform(0.05, 2.0)
            i = np.arange(n)
            left = np.exp(-gamma*i)
            right = left[::-1]
            a = left+right
            a = a/a.sum()
        # apply spectral flattening a few iterations
        iters = np.random.randint(30, 101)   # random number of iterations
        for _ in range(iters):
            # compute DFT
            A = np.fft.rfft(a)
            mags = np.abs(A)

            mags = np.where(mags==0, 1e-12, mags)
            energy = np.sum(mags**2)   # same as sum of squares in time domain * n

def propose_candidate(seed=0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes the evaluation score.
    The algorithm combines random sampling, a simple spectral flattening
    heuristic, and a final refinement using Adam and exponentiated‑gradient.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    n = 4096
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    best_seq = None
    best_val = float('inf')

    # load warm start if provided
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] != n:
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.clip(warm_arr, 0.0, 1000.0)
        total = warm_arr.sum()
        if total >= 0.01:
            warm_arr = warm_arr / total
            best_seq = warm_arr.copy()
            best_val = _fast_eval(best_seq, n, conv_fft_len)
            print(f"[Warm] initial value {best_val:.6f}")

    # Phase I: random sampling + spectral flattening
    while time.time() < deadline - 200.0:
        # generate random base distribution
        # Choose a family probabilistically
        r = np.random.rand()
        if r < 0.4:
            # Dirichlet with random concentration
            alpha = 10 ** np.random.uniform(-3, 2)  # 1e-3 to 100
            a = np.random.dirichlet(np.full(n, alpha))
        elif r < 0.7:
            # exponential symmetric
            gamma = np.random.uniform(0.05, 2.0)
            i = np.arange(n, dtype=np.float64)
            left = np.exp(-gamma * i)
            right = left[::-1]
            a = left + right
            a /= a.sum()
        else:
            # Gaussian mixture with random number of components
            K = np.random.randint(2, 5)
            mus = np.random.uniform(0, n-1, K)
            sigmas = np.random.uniform(0.01, 0.2, K) * n
            weights = np.random.dirichlet(np.ones(K))
            a = np.zeros(n, dtype=np.float64)
            idx = np.arange(n, dtype=np.float64)
            for k in range(K):
                a += weights[k] * np.exp(-0.5 * ((idx - mus[k]) / sigmas[k]) ** 2)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum

        # Optional random roll to break symmetry
        if np.random.rand() < 0.2:
            shift = np.random.randint(n)
            a = np.roll(a, shift)

        # Apply spectral flattening for a few iterations
        proj_iters = np.random.randint(30, 81)
        for _ in range(proj_iters):
            # forward FFT
            A = np.fft.rfft(a)
            mags = np.abs(A)
            # avoid division by zero
            mags = np.where(mags == 0, 1e-12, mags)
            # preserve total energy
            energy = np.sum(mags ** 2)
            target_mag = np.sqrt(energy / len(mags))
            A = A / mags * target_mag
            a = np.fft.irfft(A, n=n)
            # enforce non‑negativity and simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

        val = _fast_eval(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

        # early exit if we already beat target
        if best_val < 1.503:
            break

    # Phase II – refined optimisation with Adam (smooth‑max)
    if best_seq is not None and time.time() < deadline - 60.0:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.6
        a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len,
                                          sub_deadline, max_iters=800_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # Phase III – final exp‑gradient polishing
    if best_seq is not None and time.time() < deadline - 10.0:
        a_final, val_final = _expgrad_optimize(best_seq, n, conv_fft_len, corr_fft_len,
                                               deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[ExpGrad] final improved to {val_final:.6f}")

    # fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _fast_eval(best_seq, n, conv_fft_len)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (len {len(best_seq)})")
    return best_seq.tolist()
```

"""
Hybrid random‑search + spectral‑flattening + gradient polishing.

1. Generate many candidate step‑functions from a few simple parametric families:
   Dirichlet draws, symmetric exponentials, random Gaussian mixtures and
   symmetric power‑law shapes.  Each candidate is projected onto the simplex.
2. Apply a cheap “spectral flattening’’ iteration:  replace the magnitude of the
   DFT by the (energy‑preserving) constant magnitude, inverse‑FFT, then enforce
   non‑negativity and re‑normalise.  A few (30‑80) such iterations tend to
   reduce the L∞‑norm of the linear convolution.
3. Keep the best candidate seen so far (objective ≈ 2·n·max(conv)).
4. With the remaining time run a longer Adam optimiser on the smooth‑max
   surrogate of the maximum convolution, followed by exponentiated‑gradient
   sub‑gradient polishing.
5. Return the best sequence (non‑negative, sum ≥ 0.01); it is automatically
   scale‑invariant for evaluate_sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _fast_val(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a simplex‑normalised a."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for random seeding)

def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha)).astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian_mixture(n: int, K: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    mus = np.random.uniform(0, n - 1, K)
    sigmas = np.random.uniform(0.01, 0.2, K) * n
    weights = np.random.dirichlet(np.ones(K))
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _make_powerlaw_sym(n: int, expo: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^expo + (n‑i)^expo."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo + (n - i + 1) ** expo
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5           # start temperature for smooth‑max
    lam_factor = 1.02   # slower increase => more stable gradients

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to get a in the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _expgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimisation of the exact max‑conv."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # multiplicative update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation score.
    The algorithm mixes cheap random sampling + a spectral‑flattening heuristic,
    followed by a longer Adam optimiser on a smooth‑max surrogate and a final
    exponentiated‑gradient polishing phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # problem size – 4096 gives a good balance between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Warm start (if a previous run left a good sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] != n:
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.clip(warm_arr, 0.0, 1000.0)
        if warm_arr.sum() >= 0.01:
            warm_arr = warm_arr / warm_arr.sum()
            best_seq = warm_arr.copy()
            best_val = _fast_val(best_seq, n, conv_fft_len)
            print(f"[Warm] initial evaluation {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – random candidates + cheap spectral flattening
    while time.time() < deadline - 200.0:
        # choose a family at random
        r = np.random.rand()
        if r < 0.4:
            # Dirichlet with random concentration
            alpha = 10 ** np.random.uniform(-3, 2)   # 1e‑3 … 1e2
            a = _make_dirichlet(n, alpha)
        elif r < 0.7:
            # Symmetric exponential tail
            gamma = np.random.uniform(0.05, 2.0)
            a = _make_exponential(n, gamma)
        elif r < 0.9:
            # Random Gaussian mixture (2‑4 components)
            K = np.random.randint(2, 5)
            a = _make_gaussian_mixture(n, K)
        else:
            # Symmetric power‑law (exponent in [‑3, 2])
            expo = np.random.uniform(-3.0, 2.0)
            a = _make_powerlaw_sym(n, expo)

        # occasional random cyclic shift (breaks symmetry)
        if np.random.rand() < 0.2:
            shift = np.random.randint(n)
            a = np.roll(a, shift)

        # ---- spectral flattening (few iterations) ----
        proj_iters = np.random.randint(30, 81)  # 30 … 80
        for _ in range(proj_iters):
            # forward DFT
            A = np.fft.rfft(a)
            mags = np.abs(A)
            mags = np.where(mags == 0, 1e-12, mags)          # avoid div‑by‑0
            energy = np.sum(mags ** 2)                     # = n·∑a_i²
            target_mag = np.sqrt(energy / len(mags))       # equal‑energy target
            A = A / mags * target_mag
            a = np.fft.irfft(A, n=n)                       # back to time domain
            a = np.maximum(a, 0.0)                         # non‑negativity
            s = a.sum()
            if s < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

        val = _fast_val(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

        # early exit if we have already passed the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase II – Adam optimisation on smooth‑max (longer run)
    if best_seq is not None and time.time() < deadline - 60.0:
        remaining = deadline - time.time()
        # allocate roughly 60 % of the remaining budget to Adam
        sub_deadline = time.time() + remaining * 0.6
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=1_200_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase III – final exponentiated‑gradient polishing
    if best_seq is not None and time.time() < deadline - 10.0:
        a_final, val_final = _expgrad_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=2_500_000, eta0=0.7,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[ExpGrad] final improvement {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _fast_val(best_seq, n, conv_fft_len)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
