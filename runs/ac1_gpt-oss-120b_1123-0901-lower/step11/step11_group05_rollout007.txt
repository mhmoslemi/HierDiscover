"""
Optimising a non‑negative step‑function for the “max‑convolution” objective.

Algorithm (≈ 1000 s budget)

1. **Diverse warm‑starts** – a collection of candidates:
   * the previously discovered sequence (if present)
   * the uniform vector
   * a dense grid of symmetric beta densities (α∈[0.1,2] with step 0.1)
   * 200 random Dirichlet draws with concentrations ranging from 0.1 to 5.
   The best of these (with exact objective) seeds the optimiser.

2. **Smooth‑max Adam** – minimise a differentiable surrogate
   `L = log‑sum‑exp(λ·conv)`.  The surrogate approaches
   `max(conv)` as λ grows.  Adam is run on the logits
   (log‑softmax parametrisation) while λ is increased geometrically.
   The Adam loop is stopped when the time budget for the stage
   is exhausted.

3. **Exponential‑gradient sub‑gradient phase** – a classic
   exponentiated‑gradient update on the true max (no smoothing) is
   performed for a few seconds to escape shallow plateaus.

4. **Stochastic local moves** – tiny mass transfers between random
   coordinates, re‑projected onto the simplex, are accepted only if
   they improve the exact objective.

5. **Cyclic shifts** – the linear convolution changes under a shift;
   a handful of shifts are tried and the best is kept.

All steps respect the global time limit.  The routine always returns a
valid non‑negative sequence (projected onto the simplex) and prints
intermediate objective values for monitoring.
"""

import time
import numpy as np


# --------------------------- basic utilities ---------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    Returns the full convolution (not wrapped).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x≥0, Σx=1}.  Implements the classic O(n) algorithm.
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact evaluation used during the search.
    `a` is assumed to be on the simplex (∑a=1).  Returns 2·n·max(conv).
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# -------------------------- optimisation steps ---------------------------

def _adam_smooth_max(y: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 300_000) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    `y` are logits (log‑softmax parameters).  The algorithm
    multiplies the temperature λ by 1.5 every few hundred
    iterations, up to a very large value.
    """
    eps = 1e-15
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.5
    lam_max = 1e12

    best_y = y.copy()
    best_val = np.inf
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max parametrisation
        mx = y.max()
        exp_y = np.exp(y - mx)
        a = exp_y / exp_y.sum()

        # convolution & smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        y -= step

        # keep the best (exact) objective seen so far
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_y = y.copy()
            print(f"[Adam] it={it:06d} λ={lam:.1e} val={cur_val:.6f}")

        # temperature schedule
        lam = min(lam * lam_factor, lam_max)

        # modest learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    # return the best point (logits) converted to a probability vector
    mx = best_y.max()
    exp_y = np.exp(best_y - mx)
    a_best = exp_y / exp_y.sum()
    return a_best


def _expgrad_subgrad(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 1.2) -> np.ndarray:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    A few seconds of this phase are useful for escaping shallow minima.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = float(2.0 * n * max_c)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad] it={it:07d} val={cur_val:.6f}")

        # active set (ties are rare, atol is generous)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)       # multiplicative update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a


# --------------------------- main entry point --------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈ 1000 s) and always
    returns the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # little safety margin
    deadline = time.time() + budget_s - 0.3

    # length of the candidate (the same as in the reference implementation)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----------------------- 0) build initial candidates -----------------------
    candidates = []

    # warm start if a previous construction is available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric beta family – try a dense grid of α values
    xs = (np.arange(n) + 0.5) / n
    for alpha in np.arange(0.10, 2.01, 0.10):
        a = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, alpha - 1.0)
        a = np.where(np.isnan(a) | np.isinf(a), 0.0, a)  # safety
        if a.sum() > 0:
            a = a / a.sum()
            candidates.append(a)

    # a collection of Dirichlet draws with varied concentration
    for _ in range(200):
        conc = np.random.uniform(0.1, 5.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ----------------------- 1) pick the best cheap candidate -----------------
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Init] objective {val:.6f}")

    # ----------------------- 2) smooth‑max Adam optimisation -------------------
    adam_deadline = deadline - 8.0
    if adam_deadline > time.time():
        a_adam = _adam_smooth_max(
            y=np.log(best_a + 1e-15),  # initialise logits
            n=n,
            conv_fft_len=conv_fft_len,
            corr_fft_len=corr_fft_len,
            deadline=adam_deadline,
            max_iters=400_000,
        )
        val_adam = _objective(a_adam, n, conv_fft_len)
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam final] improved to {val_adam:.6f}")

    # ----------------------- 3) short exact‑max sub‑gradient phase -------------
    sub_deadline = deadline - 4.0
    if sub_deadline > time.time():
        a_sub = _expgrad_subgrad(
            a0=best_a,
            n=n,
            conv_fft_len=conv_fft_len,
            corr_fft_len=corr_fft_len,
            deadline=sub_deadline,
            max_iters=1_500_000,
            eta0=1.5,
        )
        val_sub = _objective(a_sub, n, conv_fft_len)
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Subgrad final] improved to {val_sub:.6f}")

    # ----------------------- 4) stochastic local search (last seconds) -----------
    local_deadline = deadline - 0.5
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny random chunk (max 1e‑3 of the source mass)
        delta = min(best_a[i], 1e-3) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ----------------------- 5) try a few cyclic shifts ------------------------
    shifts = [0, n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ----------------------- 6) final sanity checks & return ------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)               # ensure exact simplex
    final_val = evaluate_sequence(best_a.tolist())   # official evaluation
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
