"""
Hybrid optimizer for the step‑function attack (extended to larger size).

The algorithm proceeds in several stages while respecting a 1000 s budget:
1.  Obtain a warm start from the previously discovered construction
    (`height_sequence_1`).  If needed it is up‑sampled to the target length.
2.  Build a tiny, diverse pool of starts (warm, random Dirichlet,
    exponential decay, linear decay) and run a short Adam optimisation on
    each to pick the most promising few.
3.  On the selected candidates run a deep optimisation pipeline:
    * Adam on a smooth‑max surrogate of the peak convolution,
    * Exact‑max sub‑gradient phase via exponentiated‑gradient,
    * “Spectral flattening’’ – force the FFT magnitude to be almost constant,
    * Linear‑programming refinement (the LP described in the statement),
    * Projected gradient descent on the true max‑convolution,
    * Final simulated‑annealing polishing.
All iterates stay non‑negative and are projected onto the simplex
(∑a_i = 1); the objective is homogeneous, so this does not affect the score.
The routine always returns the best sequence found before the deadline.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:

    eps = 1e-15
    logits = np.log(a0 + eps)       # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 during optimisation
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            # keep DC component, distribute remaining energy uniformly
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 6,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement: maximise Σb under (a★b)_k ≤ M,
    where M = max_k (a★a)_k.  The solution b is normalised and mixed with a.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active constraints (indices where conv attains its peak)
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        # add a few spread positions to guarantee feasibility
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # Build sparse constraint matrix for active set
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # Linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={
                "presolve": True,
                "time_limit": max(0.0, deadline - time.time()),
                "msg_level": 0,
            },
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # mix with current iterate
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val

def _upsample_to_length(arr: np.ndarray, target_len: int) -> np.ndarray:
    """Upsample by linear interpolation to the target length."""
    if arr.shape[0] == target_len:
        return arr.copy()
    old_x = np.linspace(0, 1, arr.shape[0])
    new_x = np.linspace(0, 1, target_len)
    up = np.interp(new_x, old_x, arr)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s < 1e-12:
        up = np.full(target_len, 1.0 / target_len)
    else:
        up = up / s
    return up


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Generate a step‑function sequence with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a longer length – 8192 gives a better asymptotic bound.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start: try to reuse a previously discovered construction.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = _upsample_to_length(warm, n)
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a tiny diverse pool of starting vectors.
    pool = []

    if warm is not None:
        pool.append(warm)

    # random Dirichlet draws
    for alpha in (0.3, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        pool.append(a)

    # exponential decay (decreasing)
    for gamma in (0.001, 0.005, 0.01):
        a = np.exp(-gamma * np.arange(n, dtype=np.float64))
        a = a / a.sum()
        pool.append(a)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n)
    lin = lin / lin.sum()
    pool.append(lin)

    # ------------------------------------------------------------------
    # Phase I – short Adam on each start to pick the most promising ones.
    best_seq = None
    best_val = float("inf")
    max_initial = min(4, len(pool))

    for idx, a0 in enumerate(pool[:max_initial]):
        if time.time() > deadline:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # allocate a modest budget (≈15 s) for this quick Adam run
        sub_dead = min(deadline, time.time() + 15.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=150_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Init {idx}] Adam gave {val_adam:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded (unlikely), fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * (1.0 / n)   # =2.0

    # ------------------------------------------------------------------
    # Phase II – deep optimisation on the current best sequence.
    # Allocate the remaining time among the stages proportionally.
    while time.time() < deadline:
        remaining = deadline - time.time()
        # Adam (smooth‑max)
        sub_dead = time.time() + min(120.0, remaining * 0.35)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=400_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Deep Adam] improved to {val_adam:.6f}")

        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # Sub‑gradient (exponentiated‑gradient)
        sub_dead = time.time() + min(150.0, remaining * 0.30)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=400_000, eta0=0.8,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # Spectral flattening
        sub_dead = time.time() + min(80.0, remaining * 0.20)
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_iters=250_000
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # Linear‑programming refinement
        sub_dead = time.time() + min(30.0, remaining * 0.15)
        a_lp, val_lp = _lp_improve(
            best_seq, n, conv_fft_len, sub_dead, max_iter=5, tau=0.5
        )
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[LP] improved to {val_lp:.6f}")

        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # Projected gradient descent on exact max‑conv
        sub_dead = time.time() + min(30.0, remaining * 0.15)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=200_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # Final simulated‑annealing polish
        sub_dead = time.time() + min(30.0, remaining * 0.15)
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_moves=60_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

        # Loop once more only if still time left; otherwise break.
        break

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
