"""
Hybrid optimiser with extended run‑time and aggressive fine‑grained refinement.

The algorithm
--------------
1.  Initialise a candidate distribution `a` (warm start if the global
    variable ``height_sequence_1`` exists, otherwise a Dirichlet draw).
    The vector is always normalised to sum = 1 and kept on the probability
    simplex.

2.  **Adam on a smooth‑max surrogate** – a long run (≈ 300 s) of Adam
    optimisation on the log‑sum‑exp approximation of the peak of the
    self‑convolution.  The temperature is increased gradually, learning
    rate is annealed and the best iterate is memorised.

3.  **Exponentiated‑gradient sub‑gradient phase** – a shorter run (≈ 150 s)
    that directly minimises the exact max‑convolution using an active‑set
    sub‑gradient and multiplicative updates.

4.  **Projected gradient descent** – a few hundred thousand exact
    sub‑gradient steps (≈ 120 s) on the true max‑convolution with projection
    onto the simplex.

5.  **Mass‑transfer local search** – a deterministic greedy scheme
    (≈ 200 s) that repeatedly moves a small amount of mass from the coordinate
    with the largest gradient (w.r.t. the active constraints) to the one with
    the smallest gradient, accepting only strict improvements.

6.  **Simulated annealing** – a fast annealing pass (≈ 80 s) to shake
    the solution out of possible plateaus.

7.  **Upsampling (optional)** – if time permits, the distribution is
    interpolated to 8192 points and a brief Adam run is performed.

The whole routine respects the 1000 s budget, prints intermediate
objective values and finally returns the best sequence found as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric (or asymmetric) Beta pdf on a uniform grid of mid‑points."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    """Simple cosine‑bell shape."""
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.12
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.035

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve a valid probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # schedule updates
        lam = min(lam * lam_factor, 1e7)
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 1_500_000,
                                 eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0: np.ndarray, n: int,
                     conv_fft_len: int, corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected (sub‑)gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.15 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(a0: np.ndarray, n: int,
                           conv_fft_len: int, corr_fft_len: int,
                           deadline: float,
                           max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer: move a small fraction of mass from the coordinate
    with the largest gradient (w.r.t. the current active constraints) to the
    one with the smallest gradient, accepting only strict improvements.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # Stop if gradients are effectively equal (stationary point)
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        src_mass = a[i_high]
        improved = False
        # try decreasing fractions of the source mass
        for frac in (0.5, 0.35, 0.2, 0.1, 0.05, 0.02, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:07d}] new best {val_new:.6f}")
                break
        if not improved:
            # try a cheap random jitter to possibly escape a plateau
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.04
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 250_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    """
    np.random.seed(seed)

    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size – 4096 is a good trade‑off between resolution and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Initialise first candidate
    if warm is not None:
        a = warm.copy()
        print("[Init] Using provided warm start")
    else:
        # a fairly smooth Dirichlet draw as a generic start
        a = _make_dirichlet(n, alpha=0.5)
        print("[Init] No warm start – using Dirichlet(0.5)")

    # Ensure valid simplex start
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # Keep best solution seen so far
    best_seq = a.copy()
    # Compute initial exact score (fast via evaluate_sequence)
    try:
        best_val = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_val = float("inf")
    print(f"[Init] Initial score {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase I – Adam optimisation (long run)
    sub_dead = min(deadline, time.time() + 300.0)   # ≈5 min
    a_adam, val_adam = _adam_optimize(a, n, conv_fft_len,
                                      corr_fft_len, sub_dead)
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase II – Exponentiated‑gradient sub‑gradient
    if time.time() < deadline - 30:
        sub_dead = min(deadline, time.time() + 150.0)
        a_exp, val_exp = _subgradient_optimize_expgrad(best_seq, n,
                                                       conv_fft_len,
                                                       corr_fft_len,
                                                       sub_dead)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[Phase II] ExpGrad improved to {val_exp:.6f}")

    # --------------------------------------------------------------
    # Phase III – Projected gradient descent (exact max)
    if time.time() < deadline - 30:
        sub_dead = min(deadline, time.time() + 120.0)
        a_gd, val_gd = _max_conv_descent(best_seq, n,
                                         conv_fft_len, corr_fft_len,
                                         sub_dead)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Phase III] GD improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase IV – Mass‑transfer fine‑grained local search
    if time.time() < deadline - 30:
        sub_dead = min(deadline, time.time() + 180.0)
        a_mt, val_mt = _mass_transfer_optimize(best_seq, n,
                                               conv_fft_len, corr_fft_len,
                                               sub_dead)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[Phase IV] MassTransfer improved to {val_mt:.6f}")

    # --------------------------------------------------------------
    # Phase V – Simulated annealing refinement
    if time.time() < deadline - 30:
        sub_dead = min(deadline, time.time() + 80.0)
        a_ann, val_ann = _anneal_optimize(best_seq, n,
                                          conv_fft_len, sub_dead)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase V] Anneal improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase VI – Optional up‑sampling to a finer grid
    if time.time() + 30 < deadline:
        try:
            print("[Phase VI] Upsampling to 8192 points")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # linear interpolation of the current best vector
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 30.0)
            a_big, val_big = _adam_optimize(a_big, n_big,
                                            conv_fft_len_big,
                                            corr_fft_len_big,
                                            sub_dead_big,
                                            max_iters=250_000)
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Phase VI] Upsample Adam improved to {val_big:.6f}")
        except Exception as e:
            print(f"[Phase VI] Upsample failed: {e}")

    # --------------------------------------------------------------
    # Final safety: ensure non‑empty return and clip range
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (for reporting)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
