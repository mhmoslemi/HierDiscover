"""
Hybrid non‑convex optimizer for the autocorrelation flattening problem.

The goal is to minimise
    2·n·max(conv(a,a))     (with a ≥ 0, Σa = 1)
where `conv` is the linear convolution of a vector with itself.
The program works in four stages:

1.  Build a diverse pool of random start vectors (Dirichlet draws,
    spikes, sinusoid, piece‑wise blocks, warm‑start if available).

2.  For each start vector run a *spectral‑flattening* (Gerchberg‑Saxton)
    iteration that forces the FFT magnitude to be (almost) constant while
    staying on the probability simplex.

3.  Refine the result with an *exponentiated‑gradient sub‑gradient*
    method that directly minimises the exact max‑convolution.

4.  Perform a short *random‑phase* search on the constant‑magnitude
    manifold and a final projected‑gradient descent to polish the
    solution.

The whole routine respects a global time budget (≈ 1000 s) and always
returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _spectral_flatten(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> np.ndarray:
    """
    Alternating projection between the simplex and the set of vectors
    whose non‑DC FFT magnitude is constant.
    Returns the best iterate (lowest exact max‑convolution) found.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = np.inf

    while time.time() < deadline and max_iters > 0:
        # ----- evaluate ----- #
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # ----- projection to constant magnitude ----- #
        A = np.fft.fft(a)
        energy = np.dot(a, a)                     # ‖a‖_2²
        # preserve DC (the sum) and distribute the remaining energy equally
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                 # DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a = np.real(np.fft.ifft(A_new))

        # ----- projection onto simplex ----- #
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        max_iters -= 1

    return best_a


def _subgradient_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.9,
) -> np.ndarray:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Returns the best iterate seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = np.inf
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set (indices where conv reaches its maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> np.ndarray:
    """
    Keep the FFT magnitude of a0 fixed and repeatedly randomise the phases.
    After each inverse FFT the vector is projected onto the simplex.
    Returns the best vector found.
    """
    mag = np.abs(np.fft.fft(a0))
    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    half = n // 2
    it = 0
    while time.time() < deadline and it < max_iters:
        phi = np.random.random(half - 1) * 2.0 * np.pi

        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]

        # positive frequencies
        A[1:half] = mag[1:half] * np.exp(1j * phi)
        # negative frequencies (conjugate symmetry)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            # Nyquist component (real)
            A[half] = (1.0 if np.random.rand() < 0.5 else -1.0) * mag[half]

        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            a_best = a.copy()

        it += 1

    return a_best


def _projected_gradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> np.ndarray:
    """
    Simple projected gradient descent on the exact max‑convolution.
    Uses a diminishing step size  α / sqrt(t) and Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        it += 1

    return best_a

def _make_initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest, diverse pool of start vectors."""
    cand = []

    # warm start, plus a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential spikes on both ends
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law shaped vectors
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Sparse “spike” families
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    # Random sparse spikes
    for k in (32, 64, 128):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            cand.append(a)

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        cand.append(a)

    # A few extra random Dirichlet shuffles
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        cand.append(a)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the given time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.

    deadline = time.time() + budget_s - 0.5   # small safety margin

    n = 8192                               # doubling the previous size
    conv_fft_len = _next_pow2(2 * n - 1)    # for a*a
    corr_fft_len = _next_pow2(3 * n - 2)    # for cross‑correlation

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build candidate pool
    pool = _make_initial_candidates(n, warm)

    best_seq = None
    best_val = np.inf
    candidates_evaluated = 0

    # ------------------------------------------------------------------
    # Phase I – quick screening (short sub‑gradient runs)
    # Keep the top few promising candidates for deeper work.
    top_candidates = []
    max_screen = 12                     # number of vectors we fully optimise later

    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # fast sub‑gradient for a few thousand steps (cheap screening)
        short_dead = min(deadline, time.time() + 4.0)    # about 4 s per candidate
        a_tmp = _subgradient_expgrad(

            max_iters=40_000, eta0=0.8
        )
        conv = _full_convolution(a_tmp, conv_fft_len)
        val = 2.0 * n * conv.max()

        if val < best_val:
            best_val = val
            best_seq = a_tmp.copy()
            print(f"[Screen {idx}] new overall best {val:.6f}")

        # keep for deeper processing
        top_candidates.append((val, a_tmp))
        candidates_evaluated += 1

    # sort and keep only the best few
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = [vec for (_, vec) in top_candidates[:max_screen]]

    print(f"Screening finished: {candidates_evaluated} candidates, "
          f"keeping {len(top_candidates)} for deep optimisation.")

    # ------------------------------------------------------------------
    # Phase II – deep optimisation of the selected candidates
    for idx, a_start in enumerate(top_candidates):
        if time.time() > deadline:

        # 1) Spectral flattening (Gerchberg‑Saxton)
        sub_dead = min(deadline, time.time() + 12.0)
        a_flat = _spectral_flatten(a_start, n, conv_fft_len, sub_dead,
                                   max_iters=80_000)

        # 2) Precise sub‑gradient descent
        sub_dead = min(deadline, time.time() + 30.0)
        a_sub = _subgradient_expgrad(
            a_flat, n, conv_fft_len, corr_fft_len, sub_dead,

        conv = _full_convolution(a_sub, conv_fft_len)

    # ------------------------------------------------------------------

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
