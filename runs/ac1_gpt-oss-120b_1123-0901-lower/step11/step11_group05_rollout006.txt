def subgradient_opt(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters, eta0=2.0):
    a = a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)

"""
Propose a high‑quality step‑function by a hybrid search.

Algorithm
---------
1.  Initialise a diverse pool of candidates (warm start, uniform,
    several beta‑shaped densities, exponential decays and random
    Dirichlet draws).  Keep the best after an exact evaluation.

2.  Spend the majority of the time budget on many short
    exponentiated‑gradient (EG) sub‑gradient runs with random restarts.
    Each run projects onto the probability simplex and stops when the
    allocated time expires.  The best EG result is retained globally.

3.  Fine‑tune the current best with Adam applied to a smooth‑max
    surrogate of the convolution maximum.  The temperature (``lam``)
    is increased steadily so that the surrogate approaches the true
    maximum.

4.  Perform a lightweight simulated‑annealing style local search for
    the last few seconds: repeatedly move a tiny amount of mass from
    one coordinate to another and accept the move according to a
    decreasing temperature schedule.

5.  Try a handful of cyclic shifts (the linear convolution changes) and
    keep the shift with the smallest objective.

6.  Clip, re‑project onto the simplex and return the list.  The function
    respects the supplied time budget (≈ 1000 s) and always returns a
    valid non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimiser.
    Stops when the wall‑clock ``deadline`` is reached or after ``max_iters``.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set of binding inequalities (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)       # EG update
        s = a.sum()
        if s > 0 and not np.isnan(s):
            a /= s
        else:
            a = np.random.dirichlet(np.ones(n))
            a /= a.sum()

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate of
    ``max(conv)``.  The temperature ``lam`` is increased gradually so that
    the surrogate converges to the true maximum.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # start temperature
    lam_factor = 1.02
    lam_max = 1e12

    best_a = np.exp(logits - np.max(logits))
    best_a /= best_a.sum()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        # slowly decay learning rate
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _local_search(
    a_start: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic local‑search (annealing style).  Repeatedly moves a
    small random amount of mass from one index to another and accepts
    the move according to a temperature schedule that cools linearly.
    """
    best_a = a_start.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    cur_a = best_a.copy()
    cur_val = best_val

    t0 = time.time()
    # simple linear cooling: T = T0 * (1 - elapsed / total)
    T0 = 0.02

    while time.time() < deadline:
        elapsed = time.time() - t0
        total = deadline - t0
        T = max(T0 * (1 - elapsed / total), 1e-6)

        i, j = np.random.choice(n, 2, replace=False)
        if cur_a[i] <= 0:
            continue
        # transfer up to 1 % of the source mass
        delta = min(cur_a[i] * 0.01, cur_a[i]) * np.random.rand()
        if delta <= 0:
            continue

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        val = _objective(cand, n, conv_fft_len)

        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / T):
            cur_a = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[Local SA] improved to {val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` bound.  The routine respects the time budget
    (≈ 1000 s) and always returns a valid list of floats.
    """
    np.random.seed(seed)
    # reserve a small safety margin
    deadline = time.time() + budget_s - 0.3

    # --------------------------------------------------------------
    # 0) problem size (large enough to give a good bound, still fast)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) generate an initial diverse pool of candidates
    candidates = []

    # warm start (if the notebook already produced one)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-6:
            a = np.random.dirichlet(np.ones(n))
        a /= a.sum()
        candidates.append(a)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # beta‑shaped families (U‑shaped for α<1)
    alphas = [0.55, 0.60, 0.65, 0.70, 0.75, 0.80, 0.85, 0.90, 0.95]
    x = (np.arange(n) + 0.5) / n
    for alpha in alphas:
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            a /= a.sum()
            candidates.append(a)

    # exponential decays
    for lam in [0.3, 0.5, 0.7, 0.9]:
        a = np.exp(-lam * np.arange(n))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)

    # random Dirichlet draws with varied concentration
    for _ in range(30):
        conc = np.random.uniform(0.5, 3.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # --------------------------------------------------------------
    # 2) pick the best cheap candidate (exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # --------------------------------------------------------------
    # 3) main search – many short EG sub‑gradient runs with random restarts
    # allocate ~70 % of the total time to this phase
    subphase_deadline = deadline - (budget_s * 0.30)
    while time.time() < subphase_deadline:
        # random start (Dirichlet) – very cheap to generate
        a0 = np.random.dirichlet(np.ones(n))
        # each restart gets at most 12 s (or whatever is left)
        run_deadline = min(subphase_deadline, time.time() + 12.0)
        a_opt, val_opt = _subgradient_optimize_expgrad(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            run_deadline,
            max_iters=3_000_000,
            eta0=2.0,
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Subgrad restart] new best {val_opt:.6f}")

    # --------------------------------------------------------------
    # 4) short Adam fine‑tuning (smooth‑max surrogate)
    adam_deadline = deadline - (budget_s * 0.10)
    if time.time() < adam_deadline:
        a_adam, val_adam = _adam_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam fine] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 5) simulated‑annealing style local search for the final seconds
    local_deadline = deadline - 0.5
    if time.time() < local_deadline:
        a_loc, val_loc = _local_search(best_a, n, conv_fft_len, local_deadline)
        if val_loc < best_val:
            best_val, best_a = val_loc, a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 6) try a few cyclic shifts (the linear convolution changes)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 7) final sanitisation
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)           # guarantees Σ=1 & non‑negativity

    # Report official evaluation (mostly for logging)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
