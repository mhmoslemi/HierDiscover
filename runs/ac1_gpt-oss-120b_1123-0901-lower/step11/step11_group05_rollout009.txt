def subgradient_opt(a0, n, fft_len, end_time):
    a = np.copy(a0)
    best_a = a.copy()
    best_val = obj(a)
    it = 0
    while time.time()<end_time:
        conv = _fft_convolve(a, fft_len)
        max_val = conv.max()
        # compute active indices where conv approx = max
        active = np.isclose(conv, max_val, atol=1e-12)
        # compute weight w: uniform on active indices
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0/active.sum()
        # gradient = 2 * cross_correlation(w, a)  (like earlier)
        grad = 2 * _cross_correlation(w, a, corr_fft_len)  # we need to compute corr_fft_len maybe same as fft_len*2?
        # Step size schedule: eta = 0.5 / sqrt(it+1)
        eta = 0.5 / np.sqrt(it+1)
        a = a - eta * grad
        # project onto simplex
        a = _project_simplex(a)
        # evaluate and update best
        val = obj(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        it += 1
    return best_a, best_val
```

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1

"""
Gradient+Evolution search for low max‐convolution step function.

- Generates several initial candidates (uniform, beta‑shaped, random Dirichlet,
  edge spikes, warm‑start if available) for each chosen length.
- Runs a fast exponentiated‑gradient sub‑gradient phase (exact max‑convolution
  sub‑gradient) until the allocated time expires.
- Refines the best point with an Adam optimizer on a smooth‑max surrogate.
- Finishes with a short stochastic local‑search and a few cyclic shifts.
- Tries several lengths (1024 … 8192) and returns the best sequence overall.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact evaluation 2·n·max(a*a) for a with sum(a)=1.
    Returns the quantity that the official `evaluate_sequence` would compute
    after the same normalisation.
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimiser (exact max‑convolution).
    Stops when the current time exceeds `deadline`.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Uses a soft‑max parametrisation a = softmax(logits).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.2                # start temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 150_000 == 0 and it > 0:

Gradient‑plus‑evolution search for a low‑max‑convolution step function.

The routine:
1.  Tries several lengths (1024 … 8192) – larger lengths give more freedom.
2.  For each length builds a diverse pool of initial vectors:
    * warm start (if a previous sequence is in `height_sequence_1`);
    * uniform distribution;
    * symmetric Beta densities with a wide range of shape parameters;
    * random Dirichlet draws (varied concentration);
    * simple two‑spike patterns and a few multi‑spike random patterns.
3.  Picks the best cheap candidate (exact objective `2·n·max(conv)`).
4.  Runs an exponentiated‑gradient sub‑gradient phase (exact max‑convolution
    sub‑gradient) until the allocated time expires.
5.  Refines the best point with a short Adam optimisation on a smooth‑max
    surrogate of the max‑convolution.
6.  Finishes with a tiny stochastic local‑search and a handful of cyclic
    shifts (linear convolution depends on the shift).
7.  The best sequence over all lengths is projected onto the probability
    simplex, clipped to the allowed range and finally evaluated with the
    official `evaluate_sequence`.
The whole procedure respects the supplied ~1000 s budget and always returns
a valid non‑negative list of floats.

# Basic linear‑algebra / FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) for a with Σa = 1.
    This is exactly the quantity `evaluate_sequence` would compute after the
    built‑in normalisation and clipping (our vectors stay non‑negative and tiny,
    so clipping does not change them).
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimiser (exact max‑convolution).
    Stops when `time.time() >= deadline`.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:

    a = a / a.sum()

    best_val = _objective(a, n, conv_fft_len)

    while time.time() < deadline and it < max_iters:

        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):

            best_len = n
            print(f"[Overall] new best length {n} with value {val_n:.6f}")

    # fallback – should never happen, but keep a safe guard
    if best_global_a is None:
        best_len = 1024
        best_global_a = np.full(best_len, 1.0 / best_len)

    # final sanity: clip, re‑project, evaluate with the official function
    best_global_a = np.clip(best_global_a, 0.0, 1000.0)
    best_global_a = _project_simplex(best_global_a)

    final_val = evaluate_sequence(best_global_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_global_a)})")
    return best_global_a.tolist()
