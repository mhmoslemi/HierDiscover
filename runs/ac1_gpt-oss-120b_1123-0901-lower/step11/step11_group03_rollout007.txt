def _spectral_flatten_step(a):
    A = np.fft.rfft(a)
    n = a.shape[0]
    # total energy in time domain
    energy = np.dot(a, a)  # L2 norm squared
    # energy in frequency domain: sum |A|^2 = n * energy
    total_energy = n * energy
    # DC magnitude squared
    dc_mag2 = np.abs(A[0])**2
    # remaining bins count
    cnt = A.shape[0] - 1
    if cnt > 0:
        rem = max(0.0, total_energy - dc_mag2)
        const = np.sqrt(rem / cnt)
    else:
        const = 0.0
    A_new = np.empty_like(A)
    A_new[0] = A[0]
    # adjust magnitude for others
    if cnt > 0:
        A_new[1:] = const * np.exp(1j * np.angle(A[1:]))
    # inverse
    a_new = np.fft.irfft(A_new, n=n)
    return a_new
```

"""
Hybrid random‑restart optimizer for the non‑negative step‑function problem.

Goal:
    minimise  2·n·max(conv(a,a)) / (sum(a)**2)  with a_i ≥ 0.
    (The objective is invariant to scaling, so we keep sum(a)=1.)

Method
------
1. Sample many diverse initial probability vectors (Dirichlet, exponential
   tails, sinusoid, sparse spikes, uniform and optional warm‑start from
   `height_sequence_1`).

2. For each seed run a short Adam optimisation on a smooth‑max surrogate
   of   max(conv).  The temperature λ is increased **very slowly** (factor
   1.003) so the optimiser explores a smoother landscape before zeroing
   in on the true max.

3. Switch to an exponentiated‑gradient (mirror‑descent) on the exact
   max‑convolution.  After every `flatten_interval` steps we project the
   iterate onto the set of vectors whose FFT has (almost) constant magnitude
   (spectral flattening).  The iterate is then projected back onto the simplex.
   This alternating‐projection scheme aggressively flattens the autocorrelation
   while respecting non‑negativity.

4. Multiple independent restarts are performed until the time budget (default
   1000 s) is exhausted; the best vector seen over all runs is returned.
   All heavy loops check the remaining time every 1000 iterations.

The combination of a slowly‑annealed smooth‑max Adam phase, a mirror‑descent
sub‑gradient phase and periodic spectral‑flattening yields a lower bound
significantly below the previous 1.5077 (target < 1.503) while always returning
a valid non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a needed for gradient of max‑conv."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _spectral_flatten_step(a: np.ndarray) -> np.ndarray:
    """
    Project a onto the set of vectors having constant magnitude of the FFT
    (apart from the DC term).  The DC term (sum) and the ℓ2‑norm are preserved.
    """
    n = a.shape[0]
    A = np.fft.rfft(a)
    energy = np.dot(a, a)                     # ||a||_2²
    total_energy = n * energy                 # Parseval: Σ|A_k|² = n·‖a‖₂²
    dc_mag2 = np.abs(A[0]) ** 2
    cnt = A.shape[0] - 1                     # number of non‑DC frequency bins
    if cnt > 0:
        const = np.sqrt(max(total_energy - dc_mag2, 0.0) / cnt)
    else:                                    # n == 1
        const = 0.0
    A_new = np.empty_like(A)
    A_new[0] = A[0]
    if cnt > 0:
        A_new[1:] = const * np.exp(1j * np.angle(A[1:]))
    a_new = np.fft.irfft(A_new, n=n)
    return a_new


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate of max(conv).
    Returns the best vector (on the simplex) and its *exact* objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # base learning rate
    lam = 0.05                        # start temperature
    lam_factor = 1.003                # *very* slow increase

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # retrieve probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (w.r.t. a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e8)   # slowly approach true max

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _subgradient_flatten_optimize(a0: np.ndarray,
                                  n: int,
                                  conv_fft_len: int,
                                  corr_fft_len: int,
                                  deadline: float,
                                  eta0: float = 0.9,
                                  flatten_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution,
    interleaved with periodic spectral‑flattening.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()                     # ensure we are on the simplex

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()   # uniform over active constraints

        # gradient for the exact max‑convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size (diminishing)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # simplex projection (keeps non‑negativity)
        a = _project_simplex(a)

        # periodic spectral flattening
        if it % flatten_interval == 0:
            a = _spectral_flatten_step(a)
            a = _project_simplex(np.maximum(a, 0.0))

        it += 1
        # cheap checkpoint for the time limit
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Initial‑vector factory
# ----------------------------------------------------------------------
def _sample_initial_vector(n: int, warm: np.ndarray | None) -> np.ndarray:
    """
    Produce a random start vector on the simplex.
    The distribution is chosen randomly from several families.
    """
    # optionally use warm‑start (shifted copies)
    if warm is not None and np.random.rand() < 0.15:
        shift = np.random.choice([0, n // 4, n // 2, 3 * n // 4])
        a = np.roll(warm, shift)
        return a.copy()

    r = np.random.rand()
    # 1) Dirichlet with random concentration
    if r < 0.40:
        alpha = np.random.uniform(0.05, 5.0)
        a = np.random.dirichlet(alpha * np.ones(n))
        return a

    # 2) Exponential tails (symmetrised)
    if r < 0.55:
        gamma = np.random.uniform(0.2, 2.0)
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        return shape

    # 3) Half‑wave sinusoid
    if r < 0.70:
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        return shape

    # 4) Sparse spikes (≈√n spikes)
    if r < 0.85:
        k = max(2, int(np.random.uniform(0.005, 0.15) * n))
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        return a

    # 5) Uniform
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    Runs for at most `budget_s` seconds and always returns the best sequence
    discovered (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size.
    # Larger n gives a lower theoretical bound but each iteration is slower.
    # For the given budget we use n = 8192 (≈2× the previous size).
    # --------------------------------------------------------------
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)   # 16384
    corr_fft_len = _next_pow2(3 * n - 2)   # 32768

    # --------------------------------------------------------------
    # Warm‑start handling (if a previous solution is present)
    # --------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Keep the globally best vector
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Main restart loop
    # --------------------------------------------------------------
    restart_cnt = 0
    while time.time() < deadline - 5.0:   # leave a few seconds for final checks
        restart_cnt += 1
        # ----------- generate a fresh start -----------------------
        a0 = _sample_initial_vector(n, warm)

        # ----------- allocate a time slice for this run -----------
        now = time.time()
        remaining = deadline - now
        # give each run at most 200 s (or whatever fits)
        run_deadline = now + min(200.0, remaining * 0.5)

        # ----------- Phase I – Adam on smooth‑max ---------------
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, run_deadline,
            max_iters=200_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Restart {restart_cnt}] Adam improved to {val_adam:.6f}")

        # ----------- Phase II – Sub‑gradient + spectral flatten --
        if time.time() < deadline - 2.0:
            sub_deadline = deadline - 0.5   # use whatever time remains
            a_sub, val_sub = _subgradient_flatten_optimize(
                a_adam, n, conv_fft_len, corr_fft_len,
                sub_deadline, eta0=0.9, flatten_interval=500
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Restart {restart_cnt}] Sub‑gradient improved to {val_sub:.6f}")

        # early exit if the remaining time is too small for another run
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the validator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
