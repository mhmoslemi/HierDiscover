"""
Hybrid L‑BFGS + exponential‑gradient optimizer for the max‑convolution bound.

Strategy
--------
1. Build a diverse pool of start vectors (Dirichlet draws, spikes, sinusoids,
   exponential tails, etc.) and keep the best few according to a fast
   FFT‑based evaluation of 2·n·max(conv).

2. Refine each of the top candidates with a sequence of L‑BFGS runs on a smooth‑max
   surrogate fα(a)=max(conv)+1/α·log∑exp(α·(conv−max(conv))).  The surrogate
   becomes the true max as α grows, but small α gives a smooth landscape.
   The optimizer works on the unconstrained logits of a soft‑max
   parametrisation – this enforces non‑negativity and the unit‑sum constraint.

3. Take the best result and perform a long exact‑max sub‑gradient phase
   (exponentiated‑gradient updates) to polish the solution.

4. Optional: symmetry‑enforce step (mirror the sequence) and a few
   perturb‑and‑refine cycles.

The whole routine respects the 1000 s budget, prints progress, and always
returns a non‑empty list of non‑negative heights.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradient of Σ w·(a*a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest and diverse pool of start vectors."""
    cand = []

    # warm start, if supplied
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet with several concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both sides)
    for gamma in (0.15, 0.3, 0.6):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.6, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.45):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        cand.append(a)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)

    for k in (48, 96, 192):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def _lbfgs_fun(logits: np.ndarray,
               n: int,
               conv_fft_len: int,
               corr_fft_len: int,
               alpha: float):
    """
    Smooth‑max surrogate objective and its gradient w.r.t. logits.
    Returns (f, grad) where f = 2·n·smooth_max(conv) and grad
    is ∂f/∂logits.
    """
    # soft‑max parametrisation (a ≥ 0, Σa = 1)
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    conv = _full_convolution(a, conv_fft_len)

    # smooth‑max: max_c + (1/α)·log∑exp(α·(conv-max_c))
    max_c = conv.max()
    exps = np.exp(alpha * (conv - max_c))
    exps_sum = exps.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(exps_sum)

    f = 2.0 * n * smooth_max

    # weights for the gradient (soft‑max over the α‑scaled conv)
    w = exps / exps_sum

    # gradient w.r.t. a (chain rule through the quadratic form a*a)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # gradient w.r.t. logits (soft‑max Jacobian)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)

    return f, 2.0 * n * grad_log


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a cascade of L‑BFGS optimisations with increasing smoothness.
    Returns the best a and its exact evaluation (2·n·max(conv)).
    """
    # start from logits of a0
    logits = np.log(a0 + 1e-15)        # avoid log(0)

    # schedule of α – larger α → tighter approximation of the true max
    alphas = [5.0, 20.0, 100.0, 500.0, 2000.0]

    best_a = a0.copy()
    best_val = 2.0 * n * _full_convolution(best_a, conv_fft_len).max()

    for alpha in alphas:
        if time.time() > deadline:
            break

        args = (n, conv_fft_len, corr_fft_len, alpha)
        res = minimize(_lbfgs_fun,
                       logits,
                       args=args,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'ftol': 1e-12,
                                'disp': False})

        logits = res.x                       # warm‑start next stage

        a = np.exp(logits - np.max(logits))
        a /= a.sum()

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS α={alpha:.0f}] new best {val:.6f}")

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 800_000,
                                 eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient refinement."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # EG step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Subgrad final] improved to {val_ref:.6f}")

    # --------------------------------------------------------------
    # Phase III – optional symmetry enforcement (often reduces the peak).
    if time.time() < deadline - 2.0:
        sym = 0.5 * (best_seq + best_seq[::-1])
        sym = np.maximum(sym, 0.0)
        s = sym.sum()
        if s > 1e-12:
            sym /= s
            conv = _full_convolution(sym, conv_fft_len)
            val_sym = 2.0 * n * conv.max()
            if val_sym < best_val:
                best_val = val_sym
                best_seq = sym.copy()
                print(f"[Symmetrise] improved to {val_sym:.6f}")

    # --------------------------------------------------------------
    # Phase IV – short perturb‑and‑refine cycles if time permits.
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        cycles = max(1, int(remaining // 5))
        print(f"[Perturb] running {cycles} short cycles")
        for c in range(cycles):
            noise = np.random.randn(n) * 0.001
            a_noise = best_seq + noise
            a_noise = np.maximum(a_noise, 0.0)
            a_noise = _project_simplex(a_noise)

            # Brief GD (30 k iters) on the noisy start
            sub_dead = min(deadline, time.time() + 4.5)
            a_ref, val_ref = _subgradient_optimize_expgrad(
                a_noise, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=30_000, eta0=0.9
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Perturb] cycle {c} improved to {val_ref:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
