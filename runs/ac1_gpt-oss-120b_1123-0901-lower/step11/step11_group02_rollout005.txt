"""
Hybrid hyper‑parameter sweep + Adam + mass‑transfer refinement.

The algorithm
---------------
1. Build a rich pool of candidate step‑functions (Beta, inverse‑power‑law,
   Dirichlet draws, Gaussian, exponential, spikes, uniform, cosine, …).
2. Normalise each candidate to ℓ₁‑norm 1 and evaluate the exact objective
   2·n·max(conv) (the denominator is 1 after normalisation).  Keep the
   K ≈ 15 best starts.
3. For every promising start run a short Adam optimiser on a smooth‑max
   surrogate of the self‑convolution.  The parameters are stored in a
   soft‑max (logits) representation, which guarantees the simplex
   constraint.  The best vector among all starts is remembered.
4. Run a dedicated mass‑transfer local search on the best vector:
   repeatedly move a small amount of mass from the coordinate with the
   largest sub‑gradient (w.r.t. the current maximal convolution) to the one
   with the smallest.  Only strict improvements are accepted; a few
   random perturbations are added to escape plateaus.
5. If time permits, perform a brief simulated‑annealing refinement and,
   optionally, up‑sample the solution to a finer grid (8192 points) and run
   another Adam phase.
6. Clip the final vector to the evaluator’s allowed range and return it
   as a plain list.

The routine respects the 1000 s time budget, prints intermediate scores
and always returns a non‑empty sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Candidate factories

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Beta pdf (possibly asymmetric) on a uniform grid of mid‑points."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_beta_sym(n, a):
    """Symmetric Beta (alpha=beta)."""
    return _make_beta(n, a, a)


def _make_cosine(n):
    """Simple cosine‑bell shape."""
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


def _make_inverse_powerlaw(n, p):
    """
    U‑shaped “inverse power‑law” shape:
    a_i ∝ ((i+1)*(n-i))^{-p}.
    """
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 1) * (n - i)) ** (-p)
    vals[np.isnan(vals) | np.isinf(vals)] = 0.0
    vals /= vals.sum()
    return vals

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Returns the best vector seen and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weighting
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Fine‑grained mass‑transfer local search.
    Moves a fraction of mass from the coordinate with the largest gradient
    to the one with the smallest, accepting only strict improvements.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # Stationary point check
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        src_mass = a[i_high]
        improved = False
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # occasional random perturbation to escape flat regions
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.
    Randomly moves a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (plain Python list) before the time budget
    expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous construction is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Pool of diverse candidates
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Basic families
    pool.append(_make_uniform(n))

    # Symmetric Beta (U‑shaped) – sweep a ∈ [0.15,0.9]
    for a in np.linspace(0.15, 0.9, 16):
        pool.append(_make_beta_sym(n, a))

    # Asymmetric Beta (to allow one‑sided concentration)
    for a, b in [(0.2, 0.8), (0.3, 0.7), (0.4, 0.6),
                 (0.6, 0.4), (0.7, 0.3), (0.8, 0.2)]:
        pool.append(_make_beta(n, a, b))

    # Inverse power‑law (U‑shape) – p ∈ [0.1,2.0]
    for p in np.linspace(0.1, 2.0, 20):
        pool.append(_make_inverse_powerlaw(n, p))

    # Dirichlet draws – wide range of concentration
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_make_dirichlet(n, alpha))

    # Gaussian families – various widths
    for sigma in (0.03, 0.07, 0.12, 0.25, 0.45):
        pool.append(_make_gaussian(n, sigma))

    # Exponential families – left/right decay rates
    for gamma in (0.08, 0.2, 0.5, 1.0, 2.0):
        pool.append(_make_exponential(n, gamma))

    # Simple shapes
    pool.append(_make_triangle(n))
    pool.append(_make_cosine(n))

    # Sparse spike patterns
    for k in (64, 128, 256, 512):
        pool.append(_make_spikes(n, k))

    # A few completely random Dirichlet draws
    for _ in range(5):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # Normalise all candidates (ensure ℓ₁ = 1)
    norm_pool = []
    for a in pool:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        norm_pool.append(a)
    pool = norm_pool

    # ------------------------------------------------------------------
    # Fast exact scoring (exact max‑convolution, no extra denominator)
    def _fast_score(a):
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()

    scores = [( _fast_score(a), a) for a in pool]
    scores.sort(key=lambda x: x[0])

    top_k = 15
    top = scores[:top_k]
    print(f"[Init] Best of initial pool: {top[0][0]:.6f}")

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – short Adam runs on the most promising starts
    for rank, (init_val, a0) in enumerate(top):
        if time.time() > deadline:
            break
        print(f"[Phase I] Candidate #{rank} – init {init_val:.6f}")

        sub_dead = min(deadline, time.time() + 22.0)   # ~20‑25 s per Adam run
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"   >>> Adam improved global best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – mass‑transfer refinement on the current best vector
    if best_seq is not None and time.time() < deadline - 6:
        print("[Phase II] Mass‑transfer refinement")
        a_mt, val_mt = _mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=150_000,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"   >>> Mass‑transfer improved best to {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Phase III – optional simulated annealing (short)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] Simulated‑annealing pass")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"   >>> Annealing improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – up‑sampling to a finer grid if time permits
    if best_seq is not None and time.time() + 30 < deadline:
        try:
            print("[Phase IV] Upsampling to 8192 points")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 12.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=150_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"   >>> Upsampled Adam improved best to {val_big:.6f}")
        except Exception as e:
            print(f"[Phase IV] Upsample failed: {e}")

    # ------------------------------------------------------------------
    # Fallback – ensure a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)

    # Clip to evaluator limits and final verification
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[Result] Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"[Result] Final evaluation error: {exc}")

    return best_seq.tolist()
