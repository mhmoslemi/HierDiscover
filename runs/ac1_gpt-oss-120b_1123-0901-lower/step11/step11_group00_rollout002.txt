"""
Hybrid global‑local optimizer for the self‑convolution max‑norm.

The procedure builds a diversified initial pool (uniform, Dirichlet,
exponential, sinusoidal, spikes, random blocks etc.), runs a deep
Adam + exponentiated‑gradient optimisation on a few promising starts,
flattens the FFT magnitude, randomises the phases, performs a projected
gradient descent and finally a cheap hill‑climbing (pairwise mass‑transfer)
finetuning.  The optimiser runs for the whole 1000 s budget, always
returning the best sequence seen.  All iterates stay on the probability
simplex (non‑negative, sum‑to‑one), so the evaluation reduces to
`2·n·max(conv)` which is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a required for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0 or np.isnan(ws):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 450_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient on the exact max‑conv."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.11 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Force FFT magnitude to be (almost) constant while staying on simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 350_000,
) -> tuple[np.ndarray, float]:
    """Keep FFT magnitude fixed and repeatedly randomise phases."""
    mag = np.abs(np.fft.fft(a0))

    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]

        A[1:half] = mag[1:half] * np.exp(1j * phi)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 20_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val


def _hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    """Very cheap local refinement: transfer a random fraction of mass
    from one index to another; accept if the objective improves or with a
    simple annealing probability."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_val = cur_val
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        # pick source with non‑trivial mass
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 20 %) of source mass to target
        frac = np.random.rand() * 0.2
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        # compute new objective
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        # accept if better; otherwise probabilistic annealing
        if val2 < cur_val:
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[HillClimb it={it:06d}] new best {val2:.6f}")
        else:
            # simple temperature schedule
            T = max(1e-4, 0.1 * (1.0 - it / float(max_iters)))
            if np.random.rand() < np.exp(-(val2 - cur_val) / T):
                a = a2
                cur_val = val2

        if best_val <= target:
            # early exit – we already beat the target
            break

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modestly sized but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws (different concentration parameters)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential edges
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law profiles
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform distribution
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the given time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 works well for the budget and gives a fine search
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build diverse pool of starts
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # add a few extra random permutations / block structures
    for _ in range(6):
        a = np.random.dirichlet(np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 5  # number of deep‑optimisation runs

    # ------------------------------------------------------------------
    # Phase I – deep optimisation on a few promising starts
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ---- Adam ----
        sub_dead = min(deadline, time.time() + 25.0)  # ~25 s per candidate
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=600_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # ---- Sub‑gradient (exp‑grad) ----
        if time.time() < deadline - 6.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=500_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – spectral flattening
    if best_seq is not None and time.time() < deadline - 7.0:
        print("[Spectral] Starting spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline, max_iters=300_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase III – random‑phase refinement (constant magnitude manifold)
    if best_seq is not None and time.time() < deadline - 10.0:
        print("[PhaseRandom] Random‑phase refinement")
        a_rand, val_rand = _random_phase_refine(
            best_seq, n, conv_fft_len, deadline, max_iters=400_000
        )
        if val_rand < best_val:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[PhaseRandom] improved to {val_rand:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – projected gradient descent (exact max)
    if best_seq is not None and time.time() < deadline - 6.0:
        print("[GD] Final projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase V – cheap hill‑climbing local search
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[HillClimb] Local mass‑transfer refinement")
        a_hc, val_hc = _hill_climb(
            best_seq, n, conv_fft_len, deadline,
            max_iters=600_000, target=1.503
        )
        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"[HillClimb] improved to {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
