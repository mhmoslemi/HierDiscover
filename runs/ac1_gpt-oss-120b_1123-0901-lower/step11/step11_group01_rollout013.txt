"""
Hybrid multistart optimisation for the step‑function attack.

The algorithm proceeds in two phases.

Phase 1 – rapid exploration:
    * Build a modest but diverse pool of start vectors (Dirichlet draws,
      sinusoidal, triangular, piece‑wise constant, spike patterns and a
      warm‑start if one is available).
    * Run a short Adam optimisation on each candidate (soft‑max
      parametrisation + smooth‑max surrogate of the max‑convolution).  The
      best candidate (lowest evaluation) is kept.

Phase 2 – deep refinement of the best start (uses the remaining time budget):
    1. Exponentiated‑gradient sub‑gradient optimisation (exact max‑conv).
    2. Two cycles of
         – spectral‑flattening (force the FFT magnitude to be almost constant)
         – LP‑improvement (solve the linear programme that maximises Σb
           under the active peak constraints and mix the result).
    3. Projected gradient descent on the exact max‑convolution.
    4. A final simulated‑annealing polish.

All iterates stay non‑negative and are projected onto the probability simplex,
so the evaluation is invariant to scaling.  The routine respects the global
time‑budget (default 1000 s) and always returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 20,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence f we maximise Σb subject to (f★b)_k ≤ M,
    where M = max_k (f★f)_k.
    The solution b is normalised and mixed with f.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active set (indices attaining the peak)
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        # add a few well‑spread positions to make the LP well‑posed
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # ---- build sparse constraint matrix for the active set ----
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # ---- linear programme: maximise sum(b)  <=> minimise -sum(b) ----
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        conv_ab = np.convolve(a, b)[: 2 * n - 1]
        if np.any(conv_ab > M + 1e-9):
            print(f"[LP it={it}] minor violation after LP (ignored)")

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Return a non‑negative step‑function sequence (list of floats)."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off; larger sizes are possible
    # but increase per‑iteration cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start handling (if a previous construction is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Phase 1 – generate a diverse pool and run short Adam runs
    pool = []

    # include warm‑start and a few cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # random Dirichlet draws with a variety of concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            a = np.random.dirichlet(alpha * np.ones(n))
            pool.append(a)

    # sinusoidal, triangular and piece‑wise constant shapes
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    xx = np.arange(1, n + 1, dtype=np.float64)
    prod = xx * (n - xx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 4):
        s = np.where(prod > 0, prod ** alpha, 0.0)
        if s.sum() > 0:
            s /= s.sum()
            pool.append(s)

    # a few block‑constant patterns
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            a[b * block_len:(b + 1) * block_len] = np.random.rand()
        a /= a.sum()
        pool.append(a)

    # spike‑type candidates (few equally spaced spikes)
    for k in (16, 32, 64):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(a)

    # --------------------------------------------------------------
    # Quick exploration – keep the best candidate seen
    best_seq = None
    best_val = float("inf")
    max_quick = min(30, len(pool))          # limit to keep within time

    for idx, a0 in enumerate(pool[:max_quick]):
        if time.time() >= deadline:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(deadline, time.time() + 5.0)   # 5 s per short Adam run
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Quick {idx}] Adam gave new best {val_adam:.6f}")

    if best_seq is None:
        # fallback to uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * (1.0 / n)

    # --------------------------------------------------------------
    # Phase 2 – deep refinement on the incumbent solution
    # Allocate the remaining budget among the following stages.
    # The fractions are heuristic but sum to roughly the leftover time.
    remaining = deadline - time.time()
    if remaining <= 5.0:
        # not enough time left – just return what we have
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        print(f"Final best value {best_val:.6f}")
        return best_seq.tolist()

    # 1) Sub‑gradient phase (≈30 % of remaining time)
    sub_dead = min(deadline, time.time() + remaining * 0.30)
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=600_000, eta0=0.8
    )
    if val_sub < best_val:
        best_val, best_seq = val_sub, a_sub.copy()
        print(f"[Refine] Sub‑gradient improved to {val_sub:.6f}")

    # 2) Two cycles of (spectral → LP → spectral) refinements
    for cycle in range(2):
        if time.time() >= deadline:
            break
        # spectral flattening (≈15 % per call)
        spec_dead = min(deadline, time.time() + remaining * 0.15 / 2)
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, spec_dead, max_iters=500_000
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Cycle {cycle}] Spectral improved to {val_spec:.6f}")

        if time.time() >= deadline:
            break
        # LP improvement (≈10 % per call)
        lp_dead = min(deadline, time.time() + remaining * 0.10 / 2)
        a_lp, val_lp = _lp_improve(
            best_seq, n, conv_fft_len, lp_dead, max_iter=25, tau=0.5
        )
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[Cycle {cycle}] LP improved to {val_lp:.6f}")

    # 3) Projected gradient descent (≈15 % of remaining time)
    if time.time() < deadline:
        gd_dead = min(deadline, time.time() + remaining * 0.15)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, gd_dead, max_iters=250_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Refine] GD improved to {val_gd:.6f}")

    # 4) Final annealing polish with whatever time is left
    if time.time() < deadline:
        anneal_dead = deadline
        # estimate moves: roughly 30 k moves per second on typical hardware
        est_moves = int(max(0, (deadline - time.time()) * 30_000))
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len, anneal_dead,
                                         max_moves=max(10_000, est_moves))
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Final] Anneal improved best to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Ensure the result respects the evaluator's clipping bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
