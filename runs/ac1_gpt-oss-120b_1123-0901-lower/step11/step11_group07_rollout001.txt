"""
Hybrid random‑search + large‑scale first‑order optimisation.

- Uses a longer sequence (n=8192) – larger n gives a finer discretisation of the
  optimal continuous shape and (empirically) lowers the value of the evaluation
  function.
- Phase 1: very fast random sampling of several parametric families (Dirichlet,
  Beta, Power‑law, exponential, Gaussian, Laplacian) on the simplex.  The best
  candidate is kept.
- Phase 2: a long Adam optimiser on a smooth‑max surrogate of the convolution
  peak.  The surrogate temperature λ is increased slowly (geometric schedule)
  so that the optimiser can first move in a smooth landscape and later focus on
  the true max.
- Phase 3: an exact‑max exponentiated‑gradient (sub‑gradient) refinement.
- Phase 4: a short simulated‑annealing polish.
- If a previous run produced a sequence in the global variable
  ``height_sequence_1`` it is up‑sampled to the target length and used as a warm
  start.

All operations respect the 1000 s budget and the returned sequence consists of
non‑negative floats (the evaluator will clip to the allowed range automatically).
"""

import time

# Helper utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (output length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _make_uniform(n):

def _make_dirichlet(n, alpha):

def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a

    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()

def _eval_fast(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(conv)/sum(a)^2 using the fast FFT convolution."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate.
    The sequence is kept on the simplex via a soft‑max parametrisation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.2                 # start temperature for smooth‑max
    lam_factor = 1.03         # very gradual increase

    best_a = a0.copy()
    best_val = _eval_fast(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & smooth‑max weight -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # smooth max using log‑sum‑exp with temperature λ
        w_raw = np.exp(lam * (conv - max_c))          # ≤1
        w_sum = w_raw.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # ----- gradient w.r.t. a (smooth‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- bookkeeping -----
        val = 2.0 * n * max_c          # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation with exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _eval_fast(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing on the simplex (pairwise mass swaps).
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-5

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val or np.random.rand() < math.exp(-(val2 - val) / T):
            a = a2
            val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()

        it += 1

    return best_a, best_val

            a = np.maximum(a, 0.0)
            a /= a.sum()
        elif fam == "exponential":
            gamma = 10 ** np.random.uniform(-2, 0)   # 0.01 … 1.0
            i = np.arange(n, dtype=np.float64)
            left = np.exp(-gamma * i)
            right = left[::-1]
            a = left + right
            a /= a.sum()
        elif fam == "gaussian":
            sigma = np.random.uniform(0.01, 0.25)
            i = np.arange(n, dtype=np.float64)
            a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
            a /= a.sum()
        elif fam == "laplacian":
            gamma = 10 ** np.random.uniform(-2, 0)   # 0.01 … 1.0
            i = np.arange(n, dtype=np.float64)
            centre = (n - 1) / 2.0
            a = np.exp(-gamma * np.abs(i - centre))
            a /= a.sum()
        elif fam == "half_sine":
            a = _make_half_sine(n)
        elif fam == "cosine":
            a = _make_cosine(n)
        elif fam == "triangle":
            a = _make_triangle(n)
        elif fam == "uniform":
            a = _make_uniform(n)
        else:
            continue

        val = _eval_fast(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # --------------------------------------------------------------
    # Ensure we have a starting point
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _eval_fast(best_seq, n, conv_fft_len)

    # --------------------------------------------------------------
    # Phase II – Adam optimisation on smooth‑max
    adam_deadline = deadline - 120.0   # keep ~2 min for final phases
    a_adam, val_adam = _adam_optimize(
        best_seq, n, conv_fft_len, corr_fft_len, adam_deadline,
        max_iters=1_200_000,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase III – exact‑max exponentiated‑gradient refinement
    a_exp, val_exp = _subgradient_optimize_expgrad(
        best_seq, n, conv_fft_len, corr_fft_len, deadline,
        max_iters=2_000_000, eta0=0.7,
    )
    if val_exp < best_val:
        best_val = val_exp
        best_seq = a_exp.copy()
        print(f"[ExpGrad] improved to {val_exp:.6f}")

    # --------------------------------------------------------------
    # Phase IV – short simulated‑annealing polish (if time permits)
    if time.time() < deadline - 5.0:
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Final housekeeping
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
