import time
import numpy as np
import scipy.optimize
import scipy.sparse

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max approximation of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max parametrisation
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val

def _expgrad_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                      corr_fft_len: int, deadline: float,
                      max_iters: int = 800_000,
                      eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient."""
    a = np.maximum(a0, 0.0)
    a = a / max(a.sum(), 1e-12)

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it}] new best {val:.6f}")

        # active set of the exact maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _max_conv_descent(a0: np.ndarray, n: int, conv_fft_len: int,
                     corr_fft_len: int, deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Projected GD on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it}] new best {val:.6f}")

        it += 1
    return best_a, best_val

def _lp_improve(a0: np.ndarray, n: int, conv_fft_len: int,
                deadline: float, max_iter: int = 30, tau: float = 0.5) -> tuple[np.ndarray, float]:
    """Repeated linear‑programming improvement."""
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        # add some spread positions to keep the LP well‑posed
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        c = -np.ones(n)                # maximise Σb ↔ minimise -Σb
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())}
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            tau *= 0.5             # shrink mixing factor if no progress
            if tau < 1e-4:
                break
    return best_a, best_val

def _hill_climb(a0: np.ndarray, n: int, conv_fft_len: int,
                deadline: float, max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """Simple random hill climbing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val
    move = 0
    while time.time() < deadline and move < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            move += 1
            continue
        j = np.random.randint(n)
        if i == j:
            move += 1
            continue
        # shift a small fraction of mass from i to j
        frac = np.random.rand() * 0.2   # up to 20 %
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # re‑project onto simplex (mass is preserved automatically)
        conv2 = _full_convolution(a2, conv_fft_len)
        max2 = conv2.max()
        if max2 < cur_max - 1e-12:
            a = a2
            cur_max = max2
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill it={move}] new best {best_val:.6f}")
        else:
            # occasional uphill move (simulated annealing style)
            T = max(1e-4, 0.05 * (1.0 - move / float(max_moves)))
            if np.random.rand() < np.exp(-(max2 - cur_max) / T):
                a = a2
                cur_max = max2
        move += 1
    return best_a, best_val

    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _lp_improve(a0: np.ndarray, n: int, conv_fft_len: int,
                deadline: float, max_iter: int = 20, tau: float = 0.5) -> tuple[np.ndarray, float]:
    """Repeated linear‑programming improvement (mass‑inflation)."""
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        # add a few spread positions to keep LP well‑posed
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        c = -np.ones(n)                # maximise Σb  <=> minimise –Σb
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())}
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val


def _hill_climb(a0: np.ndarray, n: int, conv_fft_len: int,
                deadline: float, max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """Random hill‑climbing (with a tiny simulated‑annealing component)."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val
    move = 0
    while time.time() < deadline and move < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            move += 1
            continue
        j = np.random.randint(n)
        if i == j:
            move += 1
            continue

        # move a small fraction of mass from i to j
        frac = np.random.rand() * 0.2          # up to 20 %
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        max2 = conv2.max()
        if max2 < cur_max - 1e-12:
            a = a2
            cur_max = max2
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill it={move}] new best {best_val:.6f}")
        else:
            # occasional uphill move (very low temperature)
            T = max(1e-4, 0.05 * (1.0 - move / float(max_moves)))
            if np.random.rand() < np.exp(-(max2 - cur_max) / T):
                a = a2
                cur_max = max2

        move += 1
    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Produce a step‑function (list of non‑negative heights) that minimizes
    `evaluate_sequence`.  The algorithm starts from the best known
    construction (``height_sequence_1`` if present) and applies a series of
    refinements (Adam, exponentiated‑gradient, LP, projected GD, hill‑climb)
    while respecting the overall time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Initialise from the best known sequence if it is available.
    warm = globals().get("height_sequence_1")
    target_n = 4096
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] > target_n:
            a0 = a0[:target_n]
        elif a0.shape[0] < target_n:
            a0 = np.pad(a0, (0, target_n - a0.shape[0]))
    else:
        # Simple sinusoidal start – a classic good baseline.
        a0 = np.sin(np.pi * (np.arange(target_n) + 0.5) / target_n)
        a0 = np.maximum(a0, 0.0)

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / max(a0.sum(), 1e-12)   # normalise to the probability simplex
    n = target_n

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    conv = _full_convolution(a0, conv_fft_len)
    best_seq = a0.copy()
    best_val = 2.0 * n * conv.max()
    print(f"[Init] start value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam on a smooth‑max surrogate (≈30 % of total budget)
    remaining = deadline - time.time()
    sub_dead = time.time() + remaining * 0.30
    best_seq, val = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_dead)
    if val < best_val:
        best_val, best_seq = val, best_seq.copy()
        print(f"[Stage1] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Exact sub‑gradient (exponentiated‑gradient) phase (≈30 %)
    remaining = deadline - time.time()
    sub_dead = time.time() + remaining * 0.30
    best_seq, val = _expgrad_optimize(best_seq, n, conv_fft_len, corr_fft_len,
                                      sub_dead, max_iters=800_000, eta0=0.8)
    if val < best_val:
        best_val, best_seq = val, best_seq.copy()
        print(f"[Stage2] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Repeated LP‑mass‑inflation (≈15 %)
    remaining = deadline - time.time()
    sub_dead = time.time() + remaining * 0.15
    best_seq, val = _lp_improve(best_seq, n, conv_fft_len, sub_dead,
                                max_iter=20, tau=0.5)
    if val < best_val:
        best_val, best_seq = val, best_seq.copy()
        print(f"[Stage3] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Projected GD on the exact max‑convolution (≈15 %)
    remaining = deadline - time.time()
    sub_dead = time.time() + remaining * 0.15
    best_seq, val = _max_conv_descent(best_seq, n, conv_fft_len,
                                      corr_fft_len, sub_dead, max_iters=300_000)
    if val < best_val:
        best_val, best_seq = val, best_seq.copy()
        print(f"[Stage4] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Final hill‑climbing polish (use whatever time is left)
    if time.time() < deadline - 1.0:
        best_seq, val = _hill_climb(best_seq, n, conv_fft_len,
                                    deadline, max_moves=300_000)
        if val < best_val:
            best_val, best_seq = val, best_seq.copy()
            print(f"[Stage5] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Defensive fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed interval.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
