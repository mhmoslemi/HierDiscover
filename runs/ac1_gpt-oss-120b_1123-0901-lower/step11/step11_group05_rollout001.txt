"""
Hybrid optimisation for the self‑convolution L∞ problem.

The algorithm builds a large pool of diverse probability vectors (uniform,
beta‑shaped, arcsine, Dirichlet, triangular and convex mixtures).  Each
candidate is projected onto the simplex and evaluated with the exact
objective 2·n·max(conv).  The best candidate is then refined by three
phases:

1. Exponentiated‑gradient sub‑gradient descent (exact max‑convolution
   gradient) – long, time‑controlled, large step‑size decay.
2. Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) with a
   temperature that is exponentially increased so that the surrogate
   converges to the true max.
3. Simulated‑annealing local search that moves a small amount of probability
   mass between two coordinates and accepts uphill moves with a temperature
   schedule.

After the local search the solution is cyclically shifted a few times,
projected back onto the simplex, clipped to the allowed range and finally
reported via the official ``evaluate_sequence`` function.

The routine respects the supplied time budget (≈1000 s) and always returns
a valid non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradient / Adam.
    Both arguments are real arrays; w has length 2·len(a)‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    # Fast O(n) algorithm from Duchi et al.
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact evaluation of 2·n·max(a*a) for a with sum(a)=1.
    (Matches the scaling used by ``evaluate_sequence``.)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Optimisers
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Runs until the supplied deadline.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices achieving the maximal convolution value)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (works on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of ``max(conv)``."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # initial temperature for smooth‑max
    lam_factor = 1.03
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain current a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 150_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(alpha,beta) density on a regular grid."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = _make_uniform(n)
    return a


def _make_arcsine(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine law p(x)=1/(π√(x(1‑x)))."""
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a /= a.sum()
    return a


def _make_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the middle)."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    tri = np.concatenate([inc, dec])
    tri /= tri.sum()
    return tri


def _mix_vectors(v1: np.ndarray, v2: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w·v1 + (1‑w)·v2, projected onto the simplex."""
    mixed = w * v1 + (1.0 - w) * v2
    mixed = np.maximum(mixed, 0.0)
    return _project_simplex(mixed)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    a valid sequence (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n reduces discretisation error.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0) Build a diverse pool of initial candidates
    candidates = []

    # warm start (if available from the previous run)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        candidates.append(a)

    # uniform
    candidates.append(_make_uniform(n))

    # a wide family of Beta‑shaped densities (including very U‑shaped)
    for _ in range(30):
        # draw exponents in log‑space to explore tiny values as well
        alpha = 10 ** np.random.uniform(-2.0, 0.5)   # [0.01, 3.16]
        beta = 10 ** np.random.uniform(-2.0, 0.5)
        candidates.append(_make_beta(n, alpha, beta))

    # arcsine law – known to flatten the sum distribution
    candidates.append(_make_arcsine(n))

    # triangular shape (just for completeness)
    candidates.append(_make_triangular(n))

    # random Dirichlet draws with low concentration (spiky)
    for _ in range(20):
        conc = np.random.uniform(0.05, 0.5)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # convex mixtures of the two best‑known shapes (uniform & arcsine)
    uniform = _make_uniform(n)
    arcsine = _make_arcsine(n)
    for w in np.linspace(0.0, 1.0, 7):
        candidates.append(_mix_vectors(arcsine, uniform, w))

    # ------------------------------------------------------------------
    # 1) Pick the best cheap candidate (exact evaluation)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Sub‑gradient optimisation (exact max‑convolution gradient)
    sub_deadline = deadline - 9.0
    if sub_deadline > time.time():
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            eta0=1.7,
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Subgrad] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam fine‑tuning on a smooth‑max surrogate
    adam_deadline = deadline - 4.0
    if adam_deadline > time.time():
        a_adam, val_adam = _adam_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=350_000,
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing local search (last few seconds)
    local_deadline = deadline - 0.6
    it_local = 0
    # temperature schedule (geometric decay)
    T_start, T_end = 5e-4, 5e-7
    while time.time() < local_deadline:
        # geometric interpolation based on iteration count
        frac = it_local / 1_500_000.0
        T = T_start * (T_end / T_start) ** max(min(frac, 1.0), 0.0)

        # propose moving a tiny chunk from i → j
        i, j = np.random.choice(n, 2, replace=False)
        max_move = best_a[i] * 0.12          # up to 12 % of source mass
        if max_move <= 0:
            continue
        delta = max_move * np.random.rand()
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        val = _objective(cand, n, conv_fft_len)
        delta_obj = val - best_val
        accept = False
        if delta_obj < 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-delta_obj / max(T, 1e-12)):
                accept = True

        if accept:
            best_a = cand
            best_val = val
            if it_local % 200_000 == 0:
                print(f"[SA it={it_local}] new best {val:.6f}")

        it_local += 1

    # ------------------------------------------------------------------
    # 5) Try a handful of cyclic shifts (convolution depends on the shift)
    for sh in [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 6) Final sanity check, clipping and reporting
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)  # ensures sum exactly 1

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
