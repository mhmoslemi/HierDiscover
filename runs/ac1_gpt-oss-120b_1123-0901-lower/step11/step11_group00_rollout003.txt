"""
Hybrid multi‑scale random search + adaptive Adam + spectral flattening.

The optimiser works in several phases:

1. **Random exploration** – for a few different sequence lengths
   (2048, 4096, 8192) we draw many diverse candidates
   (Dirichlet, Beta‑shaped, exponential tails, spikes, Gaussians,
   sinusoid).  The best ~10 candidates (according to the exact
   2·n·max(conv)/(∑a)² value) are retained.

2. **Local refinement** – every retained candidate is refined with
   a smooth‑max Adam optimiser, an exponentiated‑gradient sub‑gradient
   step, a constant‑FFT‑magnitude (spectral flattening) pass and a
   random‑phase refinement on the same magnitude manifold.

3. **Final projected gradient descent** – a short projected‑gradient
   descent cleans up any remaining irregularities.

All phases respect the global 1000 s budget and the function always
returns the best sequence discovered.  The implementation uses only
NumPy and pure‑Python code; it does not rely on external solvers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    The result is truncated to the exact length.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    w is a weight vector of the same length as the full convolution.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 350_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.08                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve probability vector (sum = 1)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c           # denominator = 1 because sum(a)=1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact maximal convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Enforce (almost) constant FFT magnitude while staying on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ---- spectral flattening step ----
        A = np.fft.fft(a)                     # full complex FFT
        energy = np.dot(a, a)                  # ‖a‖₂²

        # Preserve DC component (A[0]) and distribute remaining energy evenly.
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Keep the magnitude of a0 fixed and repeatedly randomise the phases.
    After each inverse FFT the vector is projected onto the simplex.
    """
    mag = np.abs(np.fft.fft(a0))

    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        # random phases for positive frequencies 1..half-1
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]                       # DC unchanged

        # positive frequencies
        A[1:half] = mag[1:half] * np.exp(1j * phi)
        # negative frequencies (complex conjugates)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            # Nyquist frequency (real) – random sign
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 15_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val

    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            cand.append(w.copy())
            # shifted versions
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand.append(np.roll(w, shift))

    # Dirichlet draws with a range of concentrations
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Beta‑type (polynomial) shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # Exponential tails (left/right symmetric)
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape = shape / shape.sum()
        cand.append(shape)

    # Sparse spike patterns (equally spaced)
    for k in (8, 16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # Random sparse spikes
    for k in (32, 64, 128):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            cand.append(a)

    # Double‑Gaussian (mirrored edge bumps)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.15, 0.25):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape = shape / shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid (already used in the reference)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    cand.append(shape)

    # Uniform distribution – a safe baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the supplied time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Use any warm start made available by previous runs.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 0 – random sampling across several lengths.
    # Keep the overall best ~15 candidates.
    global_top = []          # elements are (value, vector, n)
    max_global_kept = 15

    # Lengths to examine (larger lengths offer finer granularity but cost more)
    lengths = [2048, 4096, 8192]

    # Rough allocation: spend about 30 % of the budget on random sampling.
    sampling_deadline = time.time() + (deadline - time.time()) * 0.30

    for n in lengths:
        if time.time() > deadline:
            break
        conv_len = 2 * n - 1
        fft_len = _next_pow2(conv_len)

        # Build a pool of cheap candidates for this length.
        pool = _candidate_pool(n, warm)

        # Evaluate all pool members (they are already normalised to sum=1).
        for a in pool:
            if time.time() > sampling_deadline:
                break
            conv = _full_convolution(a, fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c   # because sum(a)=1
            # Insert into the sorted global list.
            if len(global_top) < max_global_kept:
                global_top.append((val, a.copy(), n))
                global_top.sort(key=lambda x: x[0])
            elif val < global_top[-1][0]:
                global_top[-1] = (val, a.copy(), n)
                global_top.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase I – deep refinement of the most promising candidates.
    # Give each of the top candidates a few seconds for Adam + sub‑grad.
    for idx, (cur_val, a0, n) in enumerate(global_top):
        if time.time() > deadline - 5.0:
            break

        print(f"[Refine {idx}] start length={n}, current val={cur_val:.6f}")

        # Normalise to the simplex (sum = 1) – required by the optimiser.
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # Allocate a small time slice for Adam optimisation.
        sub_dead = min(deadline, time.time() + 15.0)   # ~15 s per candidate

        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=450_000
        )
        if val_adam < cur_val:
            cur_val = val_adam
            a0 = a_adam
            print(f"[Refine {idx}] Adam improved to {val_adam:.6f}")

        # Exponentiated‑gradient sub‑gradient pass.
        if time.time() < deadline - 5.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                a0, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=350_000, eta0=0.9
            )
            if val_sub < cur_val:
                cur_val = val_sub
                a0 = a_sub
                print(f"[Refine {idx}] Sub‑grad improved to {val_sub:.6f}")

        # Replace the entry in global_top with the refined version.
        global_top[idx] = (cur_val, a0.copy(), n)

    # ------------------------------------------------------------------
    # Phase II – pick the absolute best entry and perform heavy post‑processing.
    best_val = float("inf")
    best_seq = None

    # Identify the best among refined candidates.
    for val, a, n in global_top:
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            best_n = n

    if best_seq is None:               # extremely unlikely, but guard anyway
        best_n = 4096
        best_seq = np.full(best_n, 1.0 / best_n)

    # Spectral flattening
    if time.time() < deadline - 5.0:
        print("[Spectral] start")
        conv_fft_len = _next_pow2(2 * best_n - 1)
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, best_n, conv_fft_len, deadline, max_iters=180_000
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # Random‑phase refinement on the constant‑magnitude manifold
    if time.time() < deadline - 5.0:
        print("[PhaseRandom] start")
        conv_fft_len = _next_pow2(2 * best_n - 1)
        a_rand, val_rand = _random_phase_refine(
            best_seq, best_n, conv_fft_len, deadline, max_iters=250_000
        )
        if val_rand < best_val:
            best_val, best_seq = val_rand, a_rand.copy()
            print(f"[PhaseRandom] improved to {val_rand:.6f}")

    # Final projected gradient descent fine‑tuning
    if time.time() < deadline - 3.0:
        print("[GD] final fine‑tuning")
        conv_fft_len = _next_pow2(2 * best_n - 1)
        corr_fft_len = _next_pow2(3 * best_n - 2)
        a_gd, val_gd = _max_conv_descent(
            best_seq, best_n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Final validation via the provided evaluate_sequence (ensures clipping etc.)
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    # Clip to the evaluator's accepted range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
