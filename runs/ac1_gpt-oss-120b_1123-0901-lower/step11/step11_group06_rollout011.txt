def propose_candidate(seed=0, budget_s=1000):
    import numpy, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3
    n=4096; conv_fft_len = _next_pow2(2*n-1); corr_fft_len = _next_pow2(3*n-2)

    # get warm start if present
    best_seq = None; best_val = float('inf')
    warm = globals().get('height_sequence_1')
    if warm is not None...
        Compute its val via evaluate_sequence; update best.
    
    # Stage 0: quick uniform and some candidates (maybe keep warm).
    # Stage 1: random support search
    # Determine time left; allocate portion for this stage: e.g., first 0.4 fraction of total time.
    stage_deadline = time.time() + min(300, deadline - time.time() - 200) # allocate some.
    support_candidates = []
    max_support_candidates = 5000
    while time.time() < stage_deadline and len(support_candidates) < max_support_candidates:
        m = np.random.randint(80, 150)
        supp = np.random.choice(n, size=m, replace=False)
        # compute max count using outer sum
        sums = np.add.outer(supp, supp).ravel()
        cnts = np.bincount(sums, minlength=2*n-1)
        max_cnt = cnts.max()
        val_est = 2.0 * n * (max_cnt / (m*m))
        # store if promising
        support_candidates.append((val_est, supp, m, max_cnt))
    # sort by val_est
    support_candidates.sort(key=lambda x: x[0])
    # keep top_k
    top_k = 20
    top = support_candidates[:top_k]

    # Stage 2: refine top supports
    for idx, (val_est, supp, m, max_cnt) in enumerate(top):
        if time.time() > deadline: break
        # initialize a0 uniform on support
        a0 = np.zeros(n, dtype=np.float64)
        a0[supp] = 1.0/m
        # optional random small perturb to avoid symmetric
        # subgradient optimize with time budget
        sub_dead = min(deadline, time.time() + 30.0)  # allocate up to 30s per candidate
        a_opt, val_opt = _subgradient_optimize_expgrad(a0, n, conv_fft_len, corr_fft_len,
                                                       sub_dead, max_iters=600_000, eta0=0.9)
        # compute true val
        val_true = 2.0 * n * _full_convolution(a_opt, conv_fft_len).max()
        if val_true < best_val:
            best_val = val_true; best_seq = a_opt.copy()
            print(f"New global best {best_val:.6f} from support candidate {idx}")

    # Stage 3: intensive refinement on best_seq
    if best_seq is not None and time.time() < deadline - 30:
        # maybe run spectral flattening for some iterations
        a_spec, val_spec = _spectral_flatten_optimize(best_seq, n, conv_fft_len,
                                                      deadline, max_iters=1_000_000)
        if val_spec < best_val:

"""
Hybrid sparse‑support search + exponentiated‑gradient refinement.

1.  Randomly draw many “spike’’ supports (sets of indices) of various
    cardinalities.  For each support we compute a cheap upper bound:
    the convolution peak for equal masses = max_cnt / m².
    The 10‑15 supports with the smallest bound are kept.

2.  On each retained support we run a fast exponentiated‑gradient
    sub‑gradient optimiser that respects the support mask
    (zero outside the chosen positions) and projects onto the simplex.
    The best vector found so far is stored.

3.  The best vector is then refined by the three classic phases
    (spectral flattening → projected GD → simulated annealing)
    used in the original optimiser.

All phases respect the global 1000 s deadline.  The routine always
returns a non‑negative sequence (length 4096) and prints intermediate
objective values so you can monitor progress.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _subgradient_optimize_on_support(
    a0: np.ndarray,
    mask: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation that keeps the
    vector zero outside ``mask`` (the support).  Returns the best
    iterate found together with its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = mask.astype(float)
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set of indices attaining the current max
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step

        # enforce the support mask and re‑normalise onto the simplex
        a[~mask] = 0.0
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = mask.astype(float)
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the FFT magnitude of the vector to be (almost) constant
    while preserving the DC component (the total mass).  After each inverse
    FFT the vector is projected back onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.full(n, 1.0 / n)
            s = 1.0
        a = a_new / s
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (length 4096) with a low
    evaluation score.  The method:

    * draws many random sparse supports,
    * keeps the most promising according to a cheap equal‑mass bound,
    * refines each kept support with an exponentiated‑gradient optimiser
      that respects the support,
    * finally applies spectral flattening → projected GD → annealing
      on the best vector found.

    The routine respects the global time budget and always returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3      # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start, if available
    best_seq = None
    best_val = float("inf")
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            val = 2.0 * n * _full_convolution(w, conv_fft_len).max()
            best_seq = w.copy()
            best_val = val
            print(f"[Warm] initial value {val:.6f}")

    # ------------------------------------------------------------------
    # ------- Phase 0 : quick random support exploration ----------------
    max_trials = 2000                     # how many random supports to test
    support_pool = []
    support_deadline = min(deadline - 100, time.time() + 200)  # spend ≤200 s

    while time.time() < support_deadline and len(support_pool) < max_trials:
        # pick a support size (the range is deliberately wide)
        m = np.random.randint(80, 351)           # 80 … 350 spikes
        supp = np.random.choice(n, size=m, replace=False)

        # count ordered sums i+j for this support (fast O(m²) method)
        sums = np.add.outer(supp, supp).ravel()
        cnts = np.bincount(sums, minlength=2 * n - 1)
        max_cnt = cnts.max()

        # cheap bound assuming equal mass on the support
        val_est = 2.0 * n * (max_cnt / (m * m))
        support_pool.append((val_est, supp, m, max_cnt))

    # keep the most promising few supports
    support_pool.sort(key=lambda x: x[0])
    top_supports = support_pool[:15]   # keep at most 15 candidates

    # ------------------------------------------------------------------
    # ------- Phase 1 : support‑restricted sub‑gradient optimisation ----
    for idx, (val_est, supp, m, max_cnt) in enumerate(top_supports):
        if time.time() > deadline:
            break

        # initialise with uniform mass on the support
        a0 = np.zeros(n, dtype=np.float64)
        a0[supp] = 1.0 / m
        mask = np.zeros(n, dtype=bool)
        mask[supp] = True

        sub_dead = min(deadline, time.time() + 30.0)   # ≤30 s per support
        a_opt, val_opt = _subgradient_optimize_on_support(
            a0, mask, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=800_000, eta0=0.9,
        )

        # compute the true objective (no rounding errors)
        val_real = 2.0 * n * _full_convolution(a_opt, conv_fft_len).max()
        if val_real < best_val:
            best_val = val_real
            best_seq = a_opt.copy()
            print(f"[Support {idx}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # ------- Phase 2 : intensive refinement of the current best -------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("[Spectral] intensive flattening started")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline, max_iters=500_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {best_val:.6f}")

        if time.time() < deadline - 20.0:
            print("[GD] projected gradient descent")
            a_gd, val_gd = _max_conv_descent(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=200_000,
            )
            if val_gd < best_val:
                best_val = val_gd
                best_seq = a_gd.copy()
                print(f"[GD] improved to {best_val:.6f}")

        if time.time() < deadline - 15.0:
            print("[Anneal] simulated‑annealing refinement")
            a_ann, val_ann = _anneal_optimize(
                best_seq, n, conv_fft_len, deadline, max_moves=150_000,
            )
            if val_ann < best_val:
                best_val = val_ann
                best_seq = a_ann.copy()
                print(f"[Anneal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # ------- Phase 3 : cheap perturb‑and‑refine cycles (if time left) --
    if best_seq is not None and time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        cycles = max(1, int(remaining // 5.0))
        print(f"[Perturb] running {cycles} short refinement cycles")
        for c in range(cycles):
            noise = np.random.randn(n) * 0.001
            a_noise = best_seq + noise
            a_noise = np.maximum(a_noise, 0.0)
            a_noise = _project_simplex(a_noise)

            sub_dead = min(deadline, time.time() + 4.5)
            a_ref, val_ref = _max_conv_descent(
                a_noise, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=30_000,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Perturb] cycle {c} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * _full_convolution(best_seq, conv_fft_len).max()
        print("Fallback to uniform sequence")

    # Clip to the evaluator’s allowed range and return as a list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
