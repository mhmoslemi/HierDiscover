"""
Propose a step‑function minimizing  2·n·max(a*a) / (sum(a)^2).

Algorithm
---------
1. Initialise a probability vector `f` of moderate size (n≈1024) using a
   symmetric Beta distribution (arcsine shape) or a warm‑start if provided.

2. Repeatedly solve a linear programme:
      maximise   Σ g_i
      subject to  (f * g)_k ≤ M   for all k        (M = max(f*f))
                  g_i ≥ 0
   The LP moves mass towards directions that can increase the total mass
   while keeping the current peak of the self‑convolution unchanged.
   The solution `g` is rescaled to unit sum and mixed with `f`
   (convex combination with step size t).  If the peak decreases,
   the new mixture is accepted; otherwise the step size is halved.
   This “exchange” step is cheap (≈O(n²) per iteration) and drives the
   peak down quickly.

3. After a few seconds of LP refinements we up‑sample the best small‑scale
   vector to the canonical size n=4096 (linear interpolation) and renormalise.

4. A fine‑grained mass‑transfer local search (exact evaluation of the
   max‑convolution) is run on the full‑size vector.  At each iteration the
   gradient of the active peak is computed; a small amount of mass is moved
   from the coordinate with the largest gradient to the one with the smallest.
   Adaptive step sizes and occasional random perturbations avoid stalls.

5. The final vector is clipped to the evaluator’s range and returned.

The whole routine respects the 1000 s budget, prints intermediate objective
values and falls back to a uniform distribution if SciPy is unavailable.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a⋆a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta pdf on a uniform grid of mid‑points."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a


def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from the coordinate with the largest
    gradient (of the current max‑convolution) to the one with the smallest
    gradient, accepting strictly improving moves.
    Returns the best vector and its exact objective value.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        src_mass = a[i_high]
        improved = False
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # random jitter to escape a plateau
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1
    return best_a, best_val

# Main entry point
# -------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (list of floats).  The algorithm combines
    a small‑scale linear‑program exchange step with a fine‑grained mass‑
    transfer refinement on the full grid.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ---------------------------------------------------------
    # 1. Small‑scale initialisation (n ≈ 1024)
    # ---------------------------------------------------------
    n_small = 1024
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_small:
            warm = warm[:n_small]
        else:
            warm = np.pad(warm, (0, n_small - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            f = warm / warm.sum()
        else:
            f = _make_beta(n_small, 0.5, 0.5)
    else:
        f = _make_beta(n_small, 0.5, 0.5)      # arcsine shape

    best_f = f.copy()
    best_val = 2.0 * n_small * np.convolve(f, f).max()
    print(f"[Init] objective {best_val:.6f}")

    # ---------------------------------------------------------
    # 2. LP‑exchange refinements (if SciPy is available)
    # ---------------------------------------------------------
    try:
        from scipy.optimize import linprog
        # We cap the number of LP iterations to keep within the budget.
        lp_iters = 0
        no_improve = 0
        t_step = 0.3               # initial mixing coefficient

        while time.time() < deadline - 30 and lp_iters < 60 and no_improve < 10:
            conv = np.convolve(f, f)
            M = conv.max()
            rows = 2 * n_small - 1

            # Build the Toeplitz matrix that maps g → f*g.
            # First column = f padded with zeros, first row = [f[0], 0, ..., 0].
            col = np.concatenate([f, np.zeros(n_small - 1, dtype=np.float64)])
            row = np.concatenate([f[:1], np.zeros(n_small - 1, dtype=np.float64)])
            try:
                # SciPy may have a fast toeplitz implementation.
                from scipy.linalg import toeplitz
                A = toeplitz(col, row)
            except Exception:
                # Fallback to a manual (still reasonably fast) construction.
                A = np.zeros((rows, n_small), dtype=np.float64)
                for k in range(rows):
                    j_start = max(0, k - (n_small - 1))
                    j_end = min(k, n_small - 1)
                    cols = slice(j_start, j_end + 1)
                    A[k, cols] = f[k - np.arange(j_start, j_end + 1)]

            # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
            c = -np.ones(n_small, dtype=np.float64)
            b_ub = np.full(rows, M, dtype=np.float64)
            bounds = [(0.0, None) for _ in range(n_small)]

            # Give the solver a modest time limit (remaining budget minus a safety margin)
            lp_time_left = max(0.0, deadline - time.time() - 5.0)
            options = {"presolve": True}
            # Newer versions of SciPy accept a time_limit option.
            if "time_limit" in linprog.__code__.co_varnames:
                options["time_limit"] = lp_time_left

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options=options)

            if not res.success:
                print("[LP] optimisation failed, breaking.")
                break

            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 < 1e-12:
                print("[LP] trivial solution, breaking.")
                break

            # Scale to unit sum (the same normalisation as f)
            g = g0 / sum_g0

            # Convex combination step
            t = t_step
            improved = False
            while t > 1e-5:
                f_new = (1.0 - t) * f + t * g
                conv_new = np.convolve(f_new, f_new)
                M_new = conv_new.max()
                if M_new < M - 1e-12:
                    f = f_new
                    improved = True
                    break
                t *= 0.5  # reduce step size and try again

            if improved:
                val_new = 2.0 * n_small * M_new
                if val_new < best_val:
                    best_val = val_new
                    best_f = f.copy()
                    print(f"[LP it={lp_iters:03d}] improved to {val_new:.6f}")
                no_improve = 0
                # slowly increase step size for next iteration
                t_step = min(t_step * 1.1, 0.8)
            else:
                # no improvement – shrink step size and count stagnation
                t_step = max(t_step * 0.7, 0.05)
                no_improve += 1

            lp_iters += 1

    except Exception as exc:
        # SciPy not available or some unexpected error – fall back to the small‑scale f.
        print(f"[LP] Skipped (exception: {exc})")

    # ---------------------------------------------------------
    # 3. Upsample to the target size (n = 4096)
    # ---------------------------------------------------------
    n_big = 4096
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    f_big = np.interp(x_big, x_small, best_f)
    f_big = np.clip(f_big, 0.0, 1000.0)
    f_big = f_big / f_big.sum()   # re‑normalise to unit sum

    # ---------------------------------------------------------
    # 4. Fine‑grained mass‑transfer refinement on the full size
    # ---------------------------------------------------------
    if time.time() < deadline - 8:
        conv_fft_len = _next_pow2(2 * n_big - 1)
        corr_fft_len = _next_pow2(3 * n_big - 2)
        a_mt, val_mt = _mass_transfer_optimize(
            f_big, n_big, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[MassTransfer final] improved to {val_mt:.6f}")
        else:
            best_seq = f_big.copy()
    else:
        best_seq = f_big.copy()

    # ---------------------------------------------------------
    # 5. Final safety clipping & evaluation
    # ---------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
