    def fun(v):
        a = np.exp(v - np.max(v))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        if np.any(active):

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

"""
Hybrid optimizer for the step‑function problem.

The search combines several complementary phases:
  • Randomly generated Dirichlet / analytic seeds.
  • L2‑flattening Adam (minimise the L2 distance of the auto‑convolution
    to the uniform target).
  • Smooth‑max Adam (directly minimises a differentiable surrogate of the
    max‑convolution) with a slowly increasing temperature.
  • Optional L‑BFGS‑B refinement (exact max‑convolution with sub‑gradient).
  • Exponentiated‑gradient sub‑gradient descent on the true max.
  • Light hill‑climbing local search.
Symmetrisation of the candidate after each phase reduces the search space
and empirically improves the peak‑to‑average ratio.  The routine runs
as many independent restarts as time permits (≈ 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, length 2·len(a)‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple constructors ---------------------------------------------------

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _make_sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the L2 distance ‖a*a - target‖₂.
    Returns the best vector encountered and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of ‖conv‑target‖² w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # current exact objective (max‑conv)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2 Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        it += 1

    return best_a, best_val


def _adam_optimize_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    lr: float = 0.04,
    lam_start: float = 0.2,
    lam_factor: float = 1.01,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv(a,a)).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Smooth‑max Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.96

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient optimisation via exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


class _LbfgsHelper:
    """Helper class for L‑BFGS‑B: stores problem dimensions and computes loss & grad."""
    def __init__(self, n: int, conv_fft_len: int, corr_fft_len: int):
        self.n = n
        self.conv_fft_len = conv_fft_len
        self.corr_fft_len = corr_fft_len

    def __call__(self, v: np.ndarray) -> tuple[float, np.ndarray]:
        # soft‑max to obtain probabilities
        mx = v.max()
        exp_v = np.exp(v - mx)
        a = exp_v / exp_v.sum()

        conv = _full_convolution(a, self.conv_fft_len)
        max_c = conv.max()
        loss = 2.0 * self.n * max_c

        # sub‑gradient for the max (average over all active indices)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, self.corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)          # chain rule through soft‑max
        return loss, grad_log


def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 150,
) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B refinement using the exact max (sub‑gradient)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return a0, 2.0 * n * _full_convolution(a0, conv_fft_len).max()

    helper = _LbfgsHelper(n, conv_fft_len, corr_fft_len)
    v0 = np.log(a0 + 1e-15)

    # limit execution time – SciPy does not accept a deadline, so we bound
    # the iteration count and break if we are out of overall time.
    opts = {"maxiter": max_iter, "gtol": 1e-12, "disp": False}
    res = minimize(helper, v0, jac=True, method="L-BFGS-B", options=opts)

    a_opt = np.exp(res.x - np.max(res.x))
    a_opt = a_opt / a_opt.sum()
    val_opt = 2.0 * n * _full_convolution(a_opt, conv_fft_len).max()
    return a_opt, val_opt


def _hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climbing: shift a random fraction of mass from a
    randomly chosen index i to another index j.  Accept only improving moves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 30 %) of a[i] to position j
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ budget_s seconds. Returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the well‑tested length 4096 (larger does not
    # change the objective scaling but would increase per‑iteration cost).
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best) if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of seeds (analytic + random Dirichlet)
    pool = []
    if warm is not None:
        pool.append(warm.copy())

    # analytic families
    pool.append(_make_uniform(n))
    pool.append(_make_hann(n))
    pool.append(_make_sine_window(n))
    pool.append(_make_exponential(n, gamma=0.6))
    pool.append(_make_powerlaw(n, expo=-1.0))
    pool.append(_make_triangle(n))

    # a few random Dirichlets with varying concentration
    for _ in range(8):
        alpha = 10 ** np.random.uniform(-3, 1)   # log‑uniform between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Uniform target for the L2 flattening phase
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to symmetrise (a = (a + rev(a))/2) and renormalise
    def _symmetrize(v: np.ndarray) -> np.ndarray:
        v = (v + v[::-1]) / 2.0
        s = v.sum()
        if s > 0:
            v /= s
        return v

    # ------------------------------------------------------------------
    # Main optimisation loop over candidate seeds
    for idx, a0 in enumerate(pool):
        if time.time() > deadline - 30:   # keep a safety buffer for final refinements
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Phase 1 – L2 flattening (20 s per candidate or time‑limited)
        phase_dead = min(deadline, time.time() + 20.0)
        a_l2, val_l2 = _adam_optimize_l2(
            a0, n, conv_fft_len, corr_fft_len, target, phase_dead,
            max_iters=300_000, lr=0.08,
        )
        a_l2 = _symmetrize(a_l2)
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Candidate {idx}] L2‑Adam improved to {val_l2:.6f}")

        # Phase 2 – smooth‑max Adam (15 s)
        if time.time() < deadline - 15:
            phase_dead = min(deadline, time.time() + 15.0)
            a_smt, val_smt = _adam_optimize_max(
                best_seq, n, conv_fft_len, corr_fft_len,
                phase_dead, max_iters=400_000, lr=0.04,
                lam_start=0.2, lam_factor=1.01,
            )
            a_smt = _symmetrize(a_smt)
            if val_smt < best_val:
                best_val = val_smt
                best_seq = a_smt.copy()
                print(f"[Candidate {idx}] Smooth‑max Adam improved to {val_smt:.6f}")

        # Phase 3 – optional L‑BFGS-B refinement (fast, ~5 s)
        if time.time() < deadline - 8:
            a_lb, val_lb = _lbfgs_optimize(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, max_iter=150,
            )
            a_lb = _symmetrize(a_lb)
            if val_lb < best_val:
                best_val = val_lb
                best_seq = a_lb.copy()
                print(f"[Candidate {idx}] L‑BFGS improved to {val_lb:.6f}")

        # Phase 4 – exponentiated‑gradient sub‑gradient (10 s)
        if time.time() < deadline - 10:
            a_exp, val_exp = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, max_iters=400_000, eta0=0.8,
            )
            a_exp = _symmetrize(a_exp)
            if val_exp < best_val:
                best_val = val_exp
                best_seq = a_exp.copy()
                print(f"[Candidate {idx}] ExpGrad improved to {val_exp:.6f}")

        # Phase 5 – light hill‑climbing (if time permits)
        if time.time() < deadline - 5:
            a_hc, val_hc = _hill_climb(
                best_seq, n, conv_fft_len,
                deadline, max_moves=25_000,
            )
            a_hc = _symmetrize(a_hc)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"[Candidate {idx}] Hill‑climb improved to {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Global final refinement with a longer smooth‑max Adam run (if time left)
    if best_seq is not None and time.time() < deadline - 10:
        print("[Final] Long smooth‑max Adam refinement")
        final_dead = deadline
        a_final, val_final = _adam_optimize_max(
            best_seq, n, conv_fft_len, corr_fft_len,
            final_dead, max_iters=600_000, lr=0.03,
            lam_start=0.1, lam_factor=1.005,
        )
        a_final = _symmetrize(a_final)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final] improved best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = float("inf")

    # Clip to allowed range, enforce positivity (already satisfied)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation for reporting (uses the provided function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
