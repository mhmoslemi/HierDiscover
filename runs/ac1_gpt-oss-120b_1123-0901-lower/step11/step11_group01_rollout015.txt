"""
Hybrid optimizer for the step‑function attack (new variant).

The search pipeline (≈ 1000 s budget):

1.  Warm‑start from the previously discovered construction if available,
    otherwise a random Dirichlet draw.
2.  Aggressive “spectral‑flattening’’ – repeatedly enforce an (almost)
    constant FFT magnitude while staying on the probability simplex.
3.  Long Adam optimisation on a smooth‑max surrogate of the max‑convolution.
4.  Exact‑max sub‑gradient phase (exponentiated‑gradient).
5.  Linear‑programming improvement with a larger number of iterations.
6.  Gradient‑guided hill‑climbing / simulated‑annealing on the exact
    max‑convolution (moves mass from the most “responsible’’ index to the
    least responsible one).
7.  Return the best sequence found before the deadline.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 700_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a, conv_fft_len)
        energy = np.dot(a, a)

        if n > 1:
            # keep the DC term, flatten the rest to constant magnitude
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 15,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence f we maximise Σb subject to (f★b)_k ≤ M,
    where M = max_k (f★f)_k.  The solution b is normalised and mixed with f.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active constraints (indices where convolution is close to M)
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        # add a few spread positions to stabilise the LP
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # ---- build sparse constraint matrix for the active set ----
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                    shape=(len(active), n))

        # ---- linear program: maximise Σb  <=> minimise –Σb ----
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # mix with current iterate
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            # shrink mixing factor if no improvement
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill climbing / simulated annealing.
    Moves a small amount of mass from the index with highest sub‑gradient
    (most responsible for the current peak) to the index with lowest.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    best_a = a.copy()
    best_max = max_c

    start = time.time()
    T_start = 0.12
    T_end = 1e-5

    it = 0
    while time.time() < deadline:
        # temperature schedule (linear)
        frac = (time.time() - start) / (deadline - start)
        T = max(T_end, T_start * (1.0 - frac))

        # compute current max and active set
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = conv >= max_c - 1e-12
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        # sub‑gradient of the smooth max
        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)

        i = np.argmax(grad)  # donor (most increase)
        j = np.argmin(grad)  # receiver (most decrease)
        if i == j:
            # fallback to random distinct pair
            j = (i + np.random.randint(1, n)) % n

        # move a modest fraction of mass
        delta = a[i] * 0.12
        if delta < 1e-14:
            # if donor empty, pick another random donor
            i = np.random.randint(n)
            delta = a[i] * 0.12
            if delta < 1e-14:
                # random restart when stuck
                a = np.random.dirichlet(np.ones(n))
                continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s > 0:
            a2 /= s
        else:
            a2 = np.full(n, 1.0 / n)

        conv2 = _full_convolution(a2, conv_fft_len)
        max_c2 = conv2.max()

        # Metropolis acceptance
        if max_c2 < max_c or np.random.rand() < np.exp(-(max_c2 - max_c) / (T + 1e-14)):
            a = a2
            max_c = max_c2
            if max_c2 < best_max:
                best_max = max_c2
                best_a = a2.copy()

        it += 1
        if it % 200_000 == 0:
            print(f"[HillClimb it={it:06d}] best max {best_max:.6f}")

    return best_a, best_max

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Generate a non‑negative step function with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size (the current best solution used n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # warm start (previous best if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # initialise a start vector
    if warm is not None:
        a = warm.copy()
    else:
        a = np.random.dirichlet(np.ones(n))
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # ------------------------------------------------------------------
    # Phase 1 – spectral flattening (aggressive)
    a, _ = _spectral_flatten_optimize(
        a, n, conv_fft_len, deadline, max_iters=500_000
    )

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on smooth‑max surrogate
    sub_dead = min(deadline, time.time() + 180.0)   # up to 3 min if time permits
    a, _ = _adam_optimize(
        a, n, conv_fft_len, conv_fft_len, sub_dead, max_iters=600_000
    )

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient phase (exponentiated gradient)
    if time.time() < deadline - 30:
        a, _ = _subgradient_optimize_expgrad(
            a, n, conv_fft_len, conv_fft_len, deadline,
            max_iters=700_000, eta0=0.8
        )

    # ------------------------------------------------------------------
    # Phase 4 – linear‑programming refinement (more iterations)
    if time.time() < deadline - 30:
        a, _ = _lp_improve(
            a, n, conv_fft_len, deadline, max_iter=15, tau=0.5
        )

    # ------------------------------------------------------------------
    # Phase 5 – gradient‑guided hill climbing / simulated annealing
    if time.time() < deadline - 5:
        a, _ = _hill_climb_optimize(
            a, n, conv_fft_len, deadline
        )

    # ------------------------------------------------------------------
    # final safety clipping (evaluate_sequence applies the same clamps)
    a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # quick sanity check with the official evaluator (optional)
    try:
        # evaluate_sequence is assumed to be provided in the environment
        from __main__ import evaluate_sequence  # type: ignore
        val = evaluate_sequence(a.tolist())
        print(f"Final evaluated value {val:.6f} (length {len(a)})")
    except Exception:
        pass

    return a.tolist()
