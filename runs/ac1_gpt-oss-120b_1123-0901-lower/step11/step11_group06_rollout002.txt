def _spectral_flatten_linear_opt(a0, n, M, deadline, max_iters=500_000):
    a = a0.copy()
    best_a = a.copy()
    # compute best value using evaluate_sequence for correct metric? The evaluate includes max convolution but we can compute our own quickly: val = 2 * n * np.max(np.convolve(a,a)) / (np.sum(a)**2). Since we keep sum=1, value = 2*n*max(conv). Use our own.

    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute linear conv via FFT
        conv = _full_convolution(a, M) # conv length 2n-1 <= M
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[SpecLin it={it}] new best {val:.6f}")

        # spectral flatten using zero-padded FFT
        A = np.fft.rfft(a, M)
        # compute sum of squares
        sum_sq = np.dot(a,a)
        total_energy = M * sum_sq
        mag0 = np.abs(A[0])
        # compute constant magnitude for rest
        if M > 1:
            const_mag = np.sqrt(max(0.0, (total_energy - mag0**2) / (M - 1)))
        else:
            const_mag = 0.0
        
        # set new spectrum
        A_new = np.empty_like(A)
        A_new[0] = A[0]  # preserve DC (phase)
        # Use original phases for rest
        A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))
        # Inverse transform
        a_new = np.fft.irfft(A_new, M)
        # Truncate to length n
        a_new = a_new[:n]
        # ensure nonneg, renorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

"""
Search for a low‑value non‑negative step function (sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

where conv is the linear (full) convolution.
The algorithm consists of:

1.  Build a modest pool of diversified initial vectors (warm start,
    Dirichlet draws, spikes, sinusoid, uniform, etc.) and keep the best.
2.  Apply a *spectral‑flattening* iteration that works on the **zero‑padded**
    FFT of the vector (length M ≥ 2·n‑1).  By forcing all non‑DC magnitudes
    of the FFT to be equal we make the linear auto‑convolution as flat as
    possible while staying on the probability simplex.
3.  Refine the result with an exact‑max sub‑gradient method based on
    exponentiated‑gradient updates (the same routine used in the reference
    implementation).
4.  If time permits, perform a few short perturb‑and‑refine cycles.

All steps respect the time budget (≈ 1000 s) and the sequence is kept
non‑negative and normalised to unit sum (the objective is scale‑invariant).
The best sequence found before the deadline is returned as a Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·len(a)-1 valid samples
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for sub‑gradient / gradient.
    Implements the linear correlation of a with w (both real vectors).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates made of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    # random sparse patterns with a few hundred spikes
    for k in (48, 96, 192):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # Warm start (if available) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with several concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both sides)
    for gamma in (0.15, 0.3, 0.6):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law (inverse) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.6, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.45):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        cand.append(a)

    # A few extra Dirichlet draws with different alphas (shuffled)
    for alpha in (0.1, 0.3, 0.7, 1.5):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        cand.append(a)

    # Add spike patterns
    cand.extend(_spike_candidates(n))

    return cand

# Fast objective (identical to evaluate_sequence for normalised vectors)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)^2).  Assumes a is non‑negative."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Spectral flattening on the zero‑padded FFT (linear convolution version)
# ----------------------------------------------------------------------
def _spectral_flatten_linear_opt(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the magnitude of the zero‑padded FFT (length fft_len)
    to be constant (except the DC term) while staying on the simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # current convolution and objective
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)  # sum is 1 but keep formula
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Spectral it={it:07d}] new best {val:.6f}")

        # ----- spectral flatten step (zero‑padded) -----
        # FFT of zero‑padded a
        A = np.fft.rfft(a, fft_len)

        # Energy bookkeeping (Parseval)
        sum_sq = np.dot(a, a)
        total_energy = fft_len * sum_sq
        mag0 = np.abs(A[0])                # = sum(a)
        # constant magnitude for all non‑DC frequencies
        if fft_len > 1:
            const_mag = np.sqrt(max(0.0, (total_energy - mag0 ** 2) / (fft_len - 1)))
        else:
            const_mag = 0.0

        # Build new spectrum: keep DC, replace others by const_mag with original phase
        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # DC unchanged
        if fft_len > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        # Inverse FFT, truncate to length n
        a_new = np.fft.irfft(A_new, fft_len)[:n]

        # Project back onto the probability simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient optimisation (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient refinement on the true max‑convolution objective."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the current max
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val

        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, None)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build the candidate pool
    # --------------------------------------------------------------
    pool = _initial_candidates(n, warm)

    # --------------------------------------------------------------
    # Phase 0 – evaluate the pool and keep the best so far
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        if time.time() > deadline:
            break
        a = np.clip(a, 0.0, None)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] new best {val:.6f}")

    # --------------------------------------------------------------
    # Phase I – aggressive spectral flattening (zero‑padded FFT)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("[Phase I] Spectral flatten (zero‑padded FFT)")
        a_spec, val_spec = _spectral_flatten_linear_opt(
            best_seq,
            n,
            conv_fft_len,
            deadline - 10.0,
            max_iters=800_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Phase I] improved to {val_spec:.6f}")

    # --------------------------------------------------------------
    # Phase II – exact‑max sub‑gradient refinement (exponentiated‑gradient)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        print("[Phase II] Subgradient (exp‑grad) refinement")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline - 5.0,
            max_iters=600_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase III – short perturb‑and‑refine cycles if any time left
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        cycles = max(1, int(remaining // 5))
        print(f"[Phase III] Running {cycles} perturb‑refine cycles")
        for c in range(cycles):
            # Tiny Gaussian perturbation + projection
            noise = np.random.randn(n) * 0.001
            a_noise = best_seq + noise
            a_noise = np.maximum(a_noise, 0.0)
            a_noise = _project_simplex(a_noise)

            # A brief sub‑gradient refinement (≈ 30 k iters)
            sub_dead = min(deadline, time.time() + 4.5)
            a_ref, val_ref = _subgradient_optimize_expgrad(
                a_noise,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_dead,
                max_iters=30_000,
                eta0=0.7,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Phase III] cycle {c} improved to {val_ref:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final report (use the original evaluate_sequence for correctness)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
    except Exception:          # safety – evaluate_sequence should be available
        final_val = _objective(best_seq)

    print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
