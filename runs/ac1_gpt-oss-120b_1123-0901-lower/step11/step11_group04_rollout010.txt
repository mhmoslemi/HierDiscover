def solve_lp_max_sum(a_vec, M):
    n = len(a_vec)
    import scipy.sparse as sp
    import scipy.optimize as opt
    offsets = np.arange(n)
    # put a_vec[i] for each diagonal offset i
    diag_data = [np.full(n, a_vec[i], dtype=np.float64) for i in range(n)]
    A = sp.diags(diag_data, offsets, shape=(2*n-1, n), format='csr')
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximize sum(g)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        sum_g = g.sum()
        if sum_g > 0:
            g = g / sum_g
        return g
    else:
        return None

"""
Improved optimiser for the “step‑function” challenge.

Key ideas
---------
* Initialise from a family of *U‑shaped* distributions 
  a_i ∝ ((i+0.5)*(n-i-0.5))**beta (beta < 0).  
  A fine grid over beta is scanned quickly and the best
  candidate becomes the starting point.
* Spend the majority of the time budget on a *single* promising
  candidate – the previous implementation split the budget over
  many unrelated starts.
* Use the same three‑phase cascade that proved effective before
  (L2‑flattening Adam → smooth‑max Adam → exact‑max exponentiated‑gradient
  sub‑gradient), but with longer, time‑driven phases so that each
  optimiser can run many more iterations.
* All heavy work (FFT‑based convolutions, gradients, projections)
  lives in top‑level helper functions – no closures or lambdas.
* The routine finally evaluates the best sequence with the provided
  ``evaluate_sequence`` and returns it as a Python list.

The net effect is a tighter final bound (target ≈ 1.503) while staying
well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT/conv utilities ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the linear part
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max (max‑conv surrogate) --------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # start temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- soft‑max reconstruction ---------------------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact max (objective) ----------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # Σa = 1 → denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights --------------------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max (chain rule through convolution)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through soft‑max -----------------------------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # simple learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser – L2 flattening (target = uniform) -------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam minimising the L2 distance between conv(a,a) and a uniform target.
    Returns the best vector visited and the exact max‑conv value of that vector.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max re‑parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of ½‖conv−target‖² w.r.t. a (the ½ cancels the 2)
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # exact max‑conv value (objective we ultimately care about)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # gradual learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val

# Sub‑gradient optimiser – exponentiated gradient (exact max) ---------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active constraints (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Candidate factories --------------------------------------------------
def _make_beta_family(n: int, beta: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ ((i+0.5)*(n-i-0.5))**beta, beta<0."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 0.5) * (n - i - 0.5)
    a = np.power(base, beta, where=base > 0)
    # avoid NaNs for extremely negative beta at the centre
    a[~np.isfinite(a)] = 0.0
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, expo, where=i > 0)
    a[a < 0] = 0.0
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    a = left + left[::-1]
    a /= a.sum()
    return a

# Main search routine -------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine concentrates the budget on a single, well‑chosen start
    (beta‑family) and runs long Adam / exp‑gradient phases.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 2048 offers a good trade‑off between granularity
    # and speed (the bound does not depend on n except for the trivial
    # 2·n factor).
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) QUICK SCAN OVER THE β‑FAMILY (U‑shaped) --------------------
    i = np.arange(n, dtype=np.float64)
    betas = np.linspace(-1.5, -0.05, 30)   # 30 values → ≈ 0.1 s total
    best_start = None
    best_val = float("inf")
    for beta in betas:
        a = _make_beta_family(n, beta)
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_start = a.copy()
    print(f"[Init] best β‑candidate value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) OPTIONAL backup candidates (uniform, a few power‑laws)
    #    Only kept if they happen to beat the β‑scan (unlikely).
    cand_pool = [
        _make_uniform(n),
        _make_powerlaw(n, -0.2),
        _make_powerlaw(n, -0.7),
        _make_exponential(n, gamma=0.5),
    ]
    for a0 in cand_pool:
        a0 = a0 / a0.sum()
        conv = _full_convolution(a0, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_start = a0.copy()
            print(f"[Init] backup candidate improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 3) MAIN optimisation pipeline – time‑driven phases
    a_cur = best_start.copy()
    best_seq = a_cur.copy()
    best_val = float("inf")

    # Helper to keep the best sequence seen so far
    def _maybe_update_best(a_vec):
        nonlocal best_seq, best_val
        conv = _full_convolution(a_vec, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a_vec.copy()
            print(f"[Update] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L2‑flattening Adam  (majority of the time)
    phase_dead = min(deadline, time.time() + (budget_s * 0.55))
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
    a_cur, _ = _adam_optimize_l2(
        a_cur,
        n,
        conv_fft_len,
        corr_fft_len,
        target,
        deadline=phase_dead,
        max_iters=400_000,
        lr=0.07,
    )
    _maybe_update_best(a_cur)

    # ------------------------------------------------------------------
    # Phase B – smooth‑max Adam (refine the exact max)
    if time.time() < deadline - 10:
        phase_dead = min(deadline, time.time() + (budget_s * 0.30))
        a_cur, _ = _adam_optimize(
            a_cur,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=phase_dead,
            max_iters=500_000,
        )
        _maybe_update_best(a_cur)

    # ------------------------------------------------------------------
    # Phase C – exact‑max sub‑gradient (exponentiated gradient)
    if time.time() < deadline - 5:
        a_cur, _ = _subgradient_optimize_expgrad(
            a_cur,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=deadline,
            max_iters=600_000,
            eta0=0.8,
        )
        _maybe_update_best(a_cur)

    # ------------------------------------------------------------------
    # Guarantees – fall back to uniform if nothing was found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to allowed range and final evaluation (uses supplied function)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Evaluation error: {exc}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
