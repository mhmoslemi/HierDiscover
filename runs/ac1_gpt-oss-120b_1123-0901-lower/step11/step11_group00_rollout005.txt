def lp_optimize(n, deadline):
    # Initialize f = uniform with sum sqrt(2*n)
    f = np.full(n, np.sqrt(2*n) / n)
    best_f = f.copy()
    conv = np.convolve(f, f)
    best_M = conv.max()
    it = 0
    while time.time() < deadline - margin and it < max_iter:
        # Solve LP to maximize sum(g)
        M = best_M
        # build conv matrix A
        # compute 2n-1 * n dense matrix
        # use np.zeros
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f
        # Solve LP
        c = -np.ones(n)
        b = M * np.ones(2*n-1)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            # fallback random
            g = np.random.dirichlet(np.ones(n))
            sum_g = 1.0
        scale = np.sqrt(2*n) / sum_g
        g_scaled = g * scale
        # step size t (line search)
        t = 1.0
        # ensure monotonic M decrease
        while t > 1e-4:
            f_new = (1 - t) * f + t * g_scaled
            conv_new = np.convolve(f_new, f_new)
            M_new = conv_new.max()
            if M_new < best_M - 1e-12:
                f = f_new
                best_M = M_new
                best_f = f.copy()
                break
            t *= 0.5
        it += 1
        # maybe print
        print(f"[LP it={it}] M={best_M:.9f}")
        # also break if M not improving enough
        if it > 20 and (prev_M - best_M) < 1e-8:
            break
        prev_M = best_M
    return best_f, best_M
```

Hybrid LP‑gradient optimizer for the step‑function bound.

The algorithm works in two phases:

1. **Alternating linear programme (Frank‑Wolfe style).**
   Starting from a normalized distribution (sum=1) we repeatedly:

   * solve   maximise  Σ g_i   s.t.   (f∗g)_k ≤ M  (∀k) , g_i ≥ 0;
   * scale g to keep Σ g_i = 1;
   * move f ← (1‑t) f + t g with a back‑tracking line‑search
   guaranteeing a strict decrease of M.
   The LP is solved with `scipy.optimize.linprog` using a sparse
   convolution matrix; only a few dozen iterations are needed to
   approach the theoretical optimum (M ≈ 0.75·1/n).

2. **Projected‑gradient fine‑tuning.**
   The best vector from phase 1 is refined with a sub‑gradient descent
   on the exact max‑convolution (using FFT for speed) while staying
   on the probability simplex.  This removes the remaining slack and
   typically pushes the bound from ≈ 1.507 → < 1.503.

The whole routine respects the supplied time budget (≈ 1000 s) and
always returns a non‑empty list of non‑negative numbers.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as spopt

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _lp_optimize(f: np.ndarray,
                 conv_fft_len: int,
                 deadline: float,
                 max_iter: int = 25) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style alternating LP.
    Keeps Σ f_i = 1.  Returns the best vector and its current max‑conv.
    """
    n = len(f)
    # fixed sparse pattern of the convolution matrix (independent of f)
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n)          # rows i … i+n‑1
        cols[pos:pos + n] = i
        pos += n
    shape = (2 * n - 1, n)

    best_f = f.copy()
    conv = _full_convolution(f, conv_fft_len)
    best_M = conv.max()
    prev_M = best_M + 1.0
    it = 0

    while time.time() < deadline - 0.5 and it < max_iter:
        # Build the sparse convolution matrix with the current f
        data = np.tile(f, n)                # each column contains f
        A = sp.csr_matrix((data, (rows, cols)), shape=shape)

        # Linear programme: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n)
        b = np.full(2 * n - 1, best_M)     # (f ∗ g)_k ≤ M
        res = spopt.linprog(c, A_ub=A, b_ub=b,
                            bounds=(0, None), method='highs')
        if not res.success:
            # If the solver fails we abort the LP phase.
            print("[LP] linprog failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            # fallback to a random Dirichlet vector
            g = np.random.dirichlet(np.ones(n))
            sum_g = g.sum()

        # Scale g back to the simplex (Σ g_i = 1)
        g = g / sum_g

        # --- line search to guarantee a strict decrease of M ---
        t = 1.0
        while t > 1e-4:
            f_new = (1.0 - t) * f + t * g
            conv_new = _full_convolution(f_new, conv_fft_len)
            M_new = conv_new.max()
            if M_new < best_M - 1e-12:
                f = f_new
                best_M = M_new
                best_f = f.copy()
                break
            t *= 0.5

        it += 1
        print(f"[LP it={it:02d}] max‑conv = {best_M:.9f}")
        # stop if improvement becomes negligible
        if abs(prev_M - best_M) < 1e-12:
            break
        prev_M = best_M

    return best_f, best_M


def _gd_refine(a: np.ndarray,
               n: int,
               conv_fft_len: int,
               corr_fft_len: int,
               deadline: float,
               max_iters: int = 180_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Keeps the iterate on the probability simplex (Σ a_i = 1).
    """
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()          # this equals the evaluator output
    it = 0

    while time.time() < deadline - 0.5 and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max (conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # evaluate
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation value.
    The routine first runs an alternating LP/Frank‑Wolfe phase and then
    refines the best vector with a projected sub‑gradient descent.
    The final vector is returned as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a length. 4096 gives a good trade‑off between quality and speed.
    # The algorithm works with any n; here we use 2048 to keep the LP cheap.
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start: if a previous construction exists, use it (normalised).
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() < 1e-12:
            f = np.full(n, 1.0 / n)
        else:
            f = w / w.sum()
    else:
        # plain uniform start (sum = 1)
        f = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Phase 1 – alternating LP (Frank‑Wolfe)
    f, max_conv = _lp_optimize(f, conv_fft_len, deadline, max_iter=30)
    print(f"[Phase 1] finished with max‑conv = {max_conv:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – projected gradient descent fine‑tuning
    if time.time() < deadline - 2.0:
        f, best_val = _gd_refine(f, n, conv_fft_len, corr_fft_len, deadline,
                                 max_iters=200_000)
        print(f"[Phase 2] final bound = {best_val:.6f}")
    else:
        # we still want a value for printing
        best_val = 2.0 * n * _full_convolution(f, conv_fft_len).max()
        print(f"[Phase 2] (skipped) bound ≈ {best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanity checks and return
    f = np.maximum(f, 0.0)                 # just in case of tiny negatives
    # The evaluator is invariant under global scaling, but we keep the
    # vector on the simplex for safety.
    s = f.sum()
    if s > 0:
        f = f / s                          # Σ f_i = 1
    else:
        f = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {len(f)} with estimated value {best_val:.6f}")
    return f.tolist()
