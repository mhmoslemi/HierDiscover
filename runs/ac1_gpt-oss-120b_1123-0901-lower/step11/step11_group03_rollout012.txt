"""
Hybrid optimizer for the non‑negative step‑function problem.

The objective
    2·n·max(conv(a,a)) / (sum(a)^2)
is invariant under scaling, therefore all iterates stay on the
probability simplex Σ a_i = 1.

Algorithm
---------
1. Build a modest but diverse pool of initial probability vectors.
2. Pick the best seed (exact objective) and run a smooth‑max Adam
   optimisation on the soft‑max surrogate of the max‑convolution.
3. Run a long exponentiated‑gradient (mirror‑descent) sub‑gradient
   optimisation on the exact max‑convolution.
4. Run a second short Adam phase (refinement) after the sub‑gradient.
5. Finish with a cheap greedy local‑search that repeatedly moves a tiny
   amount of mass from the most responsible coordinates of the current
   peak to the least responsible ones.
6. Return the best sequence ever seen (always non‑negative and bounded).

All heavy loops check the remaining time every few thousand iterations
so the routine always respects the 1000 s budget.
"""

import time
import numpy as np
from math import log, exp

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for gradients:
       (w ★ a)[i] = Σ_j w_{i+j} a_j.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.07,
                   lam_start: float = 0.05,
                   lam_factor: float = 1.015) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Returns the best probability vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    # exact value (sum is 1)
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # --- retrieve current probability vector ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- exact convolution (needed for weight computation) ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {best_val:.6f}")

        # --- smooth‑max weights (soft‑max over λ·(conv−max_c)) ---
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient w.r.t. a of the surrogate ---
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- chain rule through the soft‑max parametrisation ---
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # --- temperature schedule – move towards the true max ---
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) optimisation on the exact
    max‑convolution.  Runs until the deadline and keeps the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {cur_val:.6f}")

        # active set (indices attaining the current maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update, staying on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against under‑flow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _greedy_refine(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap local‑search: repeatedly move a small fraction of mass
    from the coordinates that most contribute to the current maximum
    convolution entry to those that contribute the least.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline - 0.1:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # pick one of the (possibly several) maximal indices
        active_idx = np.flatnonzero(conv >= max_c * (1 - 1e-12))
        k = int(active_idx[0])

        # gradient of max_c w.r.t. a (ignoring the factor 2)
        # due to symmetry: grad_i = a_{k-i} if 0<=k-i<n else 0
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i runs from lo..hi, j = k-i is guaranteed in [0,n-1]
        idx_i = np.arange(lo, hi + 1, dtype=int)
        idx_j = k - idx_i
        grad[idx_i] = a[idx_j]

        # indices with largest / smallest gradient magnitude
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        if a[i_high] <= 1e-14:
            # nothing to move – stop
            break

        # move a small fraction of mass (10 % of the donor, capped)
        delta = min(a[i_high] * 0.10, 0.01)
        if delta < 1e-12:
            break

        a2 = a.copy()
        a2[i_high] -= delta
        a2[i_low] += delta

        # compute new objective (exact)
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        # accept if improvement; otherwise keep old a
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # occasional uphill moves (simulated‑annealing flavour)
            # temperature decays linearly with iteration count
            T = max(1e-4, 0.01 * (1.0 - it / 50000.0))
            if np.random.rand() < np.exp(-(val2 - best_val) / T):
                a = a2

        it += 1
        if it % 5000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    pool = []

    # warm start (if supplied)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both sides)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law (triangular product) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian spikes
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # A few sparse spike candidates (≈√n spikes)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    Runs for at most `budget_s` seconds and always returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096                               # length of the vector
    conv_fft_len = _next_pow2(2 * n - 1)   # 8192
    corr_fft_len = _next_pow2(3 * n - 2)   # safe length for cross‑correlation

    # ------------------------------------------------------------------
    # Warm start from a previously found construction (if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build pool and keep the best seeding vector (exact objective)
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() > deadline:
            break
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:03d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – Adam (smooth‑max) optimisation
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase I] Adam optimisation (coarse) started")
        sub_dead = deadline - 5.0
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            sub_dead,
            max_iters=350_000, lr=0.07,
            lam_start=0.05, lam_factor=1.015
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long sub‑gradient (exponentiated‑gradient) optimisation
    if best_seq is not None and time.time() < deadline - 15:
        print("[Phase II] Sub‑gradient optimisation started")
        sub_dead = deadline - 15.0
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            sub_dead, eta0=0.6
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase III – second Adam refinement (focus on exact max)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] Adam refinement (fine) started")
        sub_dead = deadline - 5.0
        a_refine, val_refine = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            sub_dead,
            max_iters=150_000, lr=0.04,
            lam_start=5.0, lam_factor=1.025   # start already sharp
        )
        if val_refine < best_val:
            best_val = val_refine
            best_seq = a_refine.copy()
            print(f"[Phase III] fine‑tuned to {val_refine:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – greedy local‑search refinement
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase IV] Greedy local‑search refinement")
        a_greedy, val_greedy = _greedy_refine(
            best_seq, n, conv_fft_len, deadline
        )
        if val_greedy < best_val:
            best_val = val_greedy
            best_seq = a_greedy.copy()
            print(f"[Phase IV] improved to {val_greedy:.6f}")

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the validator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
