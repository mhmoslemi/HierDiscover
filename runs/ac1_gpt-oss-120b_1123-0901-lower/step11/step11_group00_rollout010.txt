def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n=4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    # Warm start
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm = np.array(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # pad or trim
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
    else:
        warm = None

    # Create pool of candidates:
    pool = []
    if warm is not None:
        pool.append(warm)
        pool.append(np.roll(warm, n//3))
        pool.append(np.roll(warm, 2*n//3))

    # Uniform
    pool.append(np.full(n, 1.0/n))

    # Random Dirichlet(s)
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            a = np.random.dirichlet(alpha * np.ones(n))
            pool.append(a)

    # Spike patterns:
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n)
        pos = np.round(np.linspace(0, n-1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(a)

    # Additional random piecewise:
    for blocks in (8, 16):
        block_len=n//blocks
        a=np.zeros(n)
        for b in range(blocks):
            mass=np.random.rand()
            a[b*block_len:(b+1)*block_len]=mass
        if a.sum()>0:
            a=a/a.sum()
        pool.append(a)

    # Evaluate each candidate quickly to find top few
    best_val = float('inf')
    best_seq = None
    top_candidates = []
    for a in pool:
        a = np.maximum(a,0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
        top_candidates.append((val, a))
    top_candidates.sort(key=lambda x: x[0])
    # Keep top K
    K = 4
    top_candidates = top_candidates[:K]

    # Phase I: Deep Adam+Subgrad
    for idx, (init_val, a0) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"[Phase I start candidate {idx} initial val {init_val:.6f}]")
        sub_dead = min(deadline, time.time() + (budget_s*0.2))  # allocate ~20% of total per candidate? but dynamic.

        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len, corr_fft_len,
                                          sub_dead, max_iters=600_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I Adam] improved to {val_adam:.6f}")

        # Subgradient
        if time.time() < deadline - 5.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=500_000, eta0=0.9)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase I Subgrad] improved to {val_sub:.6f}")

    # Phase II: Spectral flattening
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase II] spectral flatten")
        a_spec, val_spec = _spectral_flatten_optimize(best_seq, n,
                                                      conv_fft_len, deadline,
                                                      max_iters=300_000)
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Phase II] improved to {val_spec:.6f}")

    # Phase III: Random phase refinement
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase III] random phase refinement")
        a_rand, val_rand = _random_phase_refine(best_seq, n, conv_fft_len,
                                                deadline, max_iters=500_000)
        if val_rand < best_val:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[Phase III] improved to {val_rand:.6f}")

    # Phase IV: Fine-grained projected gradient descent
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase IV] final projected GD")
        a_gd, val_gd = _max_conv_descent(best_seq, n, conv_fft_len,
                                          corr_fft_len, deadline,
                                          max_iters=300_000)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Phase IV] improved to {val_gd:.6f}")

    # Phase V: Simulated annealing local refinement
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase V] simulated annealing")
        a_ann, val_ann = _simulated_annealing_refine(best_seq, n, conv_fft_len,

        if val_ann < best_val:

    # Store globally for warm start

    return best_seq.tolist()

"""
Hybrid global‑local optimizer for the step‑function evaluation problem.

The routine builds a diverse pool of start vectors (warm start, uniform,
Dirichlet draws, spike patterns, block shapes).  The top few candidates are
deep‑optimised with an Adam optimiser on a smooth‑max surrogate of the
convolution maximum, followed by an exponentiated‑gradient sub‑gradient
method.  The current best solution is further refined by
* spectral‑flattening (constant FFT magnitude),
* extensive random‑phase exploration on the constant‑magnitude manifold,
* a short projected‑gradient descent, and finally
* a lightweight simulated‑annealing pass.

All phases respect the global time‑budget (≈ 1000 s) and the best
non‑negative sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)        # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09                       # a little larger than before
    lam = 0.1                       # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 60_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient method with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.13 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 350_000,
) -> tuple[np.ndarray, float]:
    """Force FFT magnitude to be (almost) constant whilst staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ---- spectral flattening step ----
        A = np.fft.fft(a)               # full complex FFT (length n)
        energy = np.dot(a, a)           # ||a||_2^2

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                 # preserve DC term
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s
        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 700_000,
) -> tuple[np.ndarray, float]:
    """
    Keep the FFT magnitude of a0 fixed and repeatedly randomise the phases.
    After each inverse FFT the vector is projected onto the simplex.
    The best vector (lowest exact max‑convolution) is retained.
    """
    mag = np.abs(np.fft.fft(a0))

    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        phi = np.random.random(half - 1) * 2.0 * np.pi

        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]

        A[1:half] = mag[1:half] * np.exp(1j * phi)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 20_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val


def _simulated_annealing_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Lightweight SA that moves a fraction of mass between two random indices."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-4

    moves = 0
    while time.time() < deadline and moves < max_moves:
        T = max(T_end, T_start * (1.0 - moves / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        moves += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the given time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Length – 4096 is a good compromise.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of start vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            a = np.random.dirichlet(alpha * np.ones(n))
            pool.append(a)

    # Spike patterns (equally spaced)
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        pool.append(a)

    # Random sparse spikes
    for k in (32, 64, 128):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            pool.append(a)

    # Piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    # ------------------------------------------------------------------
    # Keep the top few candidates after a quick evaluation
    quick_vals = []
    for a in pool:
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        quick_vals.append((val, a))
    quick_vals.sort(key=lambda x: x[0])
    top_k = 4
    top_candidates = quick_vals[:top_k]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – deep Adam + sub‑gradient optimisation on the best seeds
    for idx, (init_val, a0) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"[Phase I] candidate {idx} start value {init_val:.6f}")

        # Allocate up to 20 % of the remaining budget for the Adam pass
        sub_dead = min(deadline, time.time() + (budget_s * 0.2))
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=800_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

        # Sub‑gradient phase using the (potentially) improved vector
        if time.time() < deadline - 5.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, max_iters=600_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase I] Sub‑grad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – spectral flattening
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase II] Spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_iters=350_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Phase II] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase III – intensive random‑phase refinement
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase III] Random‑phase refinement")
        a_rand, val_rand = _random_phase_refine(
            best_seq, n, conv_fft_len, deadline,
            max_iters=700_000
        )
        if val_rand < best_val:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[Phase III] improved to {val_rand:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – final projected gradient descent fine‑tuning
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase IV] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=250_000
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Phase IV] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase V – lightweight simulated‑annealing local search
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase V] Simulated annealing")
        a_ann, val_ann = _simulated_annealing_refine(
            best_seq, n, conv_fft_len, deadline,
            max_moves=120_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase V] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * _full_convolution(best_seq, conv_fft_len).max()

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")

    # Store globally for possible warm‑start next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
