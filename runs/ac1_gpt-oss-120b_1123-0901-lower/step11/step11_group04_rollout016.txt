def improve_a(a, max_iters, deadline, t_initial=0.5):
    n = len(a)
    best_val = evaluate_sequence(a.tolist())
    best_a = a.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, next_pow2(2*n-1)) # conv length 2n-1
        M = conv.max()
        # optional: if M <= best_M - tolerance, update best
        val = 2.0 * n * M / (np.sum(a) ** 2) # sum a maybe 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Iter {it}] new best {val}")
        # Determine active indices
        eps = 1e-12 * M # relative tolerance
        active = np.where(conv >= M - eps)[0]
        # Possibly limit active size to maybe 20: keep first few largest.
        if len(active) > 20:
            # pick those with highest conv (maybe top 20)
            # Actually conv is the same for active (close to M)
            active = active[:20] # but they are all same; maybe random subset
        # Build A matrix
        rows = []
        for i in active:
            # compute row_i
            idx = np.arange(n)
            diff = i - idx
            mask = (diff >= 0) & (diff < n)
            row = np.zeros(n, dtype=float)
            row[mask] = a[diff[mask]]
            rows.append(row)
        A = np.vstack(rows) if rows else np.empty((0,n))
        # solve LP: maximize sum(b) subject to A @ b <= M, b >= 0
        # set objective c = -np.ones(n) (minimize -sum)
        c = -np.ones(n)
        # bounds = [(0, None)] * n
        # constraints: A_ub = A, b_ub = np.full(len(active), M)
        # Use SciPy linprog
        res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=[(0, None)] * n, method='highs')
        if not res.success:
            # fallback: no improvement
            break
        b = res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            # can't improve
            break
        # normalize b to sum 1
        b = b / sum_b
        # combine
        t = t_initial / np.sqrt(it + 1) # decreasing
        a = (1 - t) * a + t * b
        # enforce sum and positivity
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full(n, 1.0 / n)
    return best_a, best_val

"""
Hybrid optimiser with an additional linear‑programming refinement.

The search proceeds as follows (all steps respect the simplex
{a_i ≥ 0, Σ a_i = 1}):

1.  Build a diverse pool of seed vectors (uniform, Dirichlet, exponential,
    power‑law, triangular, optional warm‑start).
2.  For each seed
    a) run an L2‑flattening Adam optimiser that drives the
       autocorrelation a*a towards the uniform target,
    b) run a cheap LP‑refinement that solves
          maximise   Σ b_i
          s.t.       (a ★ b)_k ≤ M  for the currently active lag‑indices k
                     b_i ≥ 0
       (M = max(a*a)).  The solution is a sparse vector that pushes mass
       into positions where a is small; a convex combination with a
       (step size decreasing as 1/√it) strictly reduces the max‑convolution.
    c) finish with a smooth‑max Adam pass (log‑sum‑exp surrogate) for fine‑tuning.
3.  Keep the best sequence seen over all restarts.
4.  As a final safety net run a few steps of projected GD and simulated
    annealing (as in the previous baseline) – these are cheap compared to the
    main phases.

All heavy work (FFT‑based convolutions) is O(n log n) and the LPs are tiny
(≤ 5 constraints), so many iterations fit easily into the 1000 s budget.
The algorithm is deliberately deterministic apart from the random seeds,
making it suitable for repeated runs.

The returned sequence is clipped to the allowed range and guaranteed to be
non‑empty.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT helpers ----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic factories ----------------------------------------------------
def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser – L2 flattening ---------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of (conv‑target)^2 w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2‑Adam it={it:06d}] new best {val:.6f}")

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val

# Adam optimiser – smooth‑max -----------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.2          # start temperature (larger = smoother)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Smooth‑Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP refinement ---------------------------------------------------------
def _lp_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 4000,
    min_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny LP that maximises Σb subject to the active
    convolution constraints (a ★ b ≤ M). A convex combination with a slowly
    decaying step size drives the max‑convolution down.
    """
    a = np.maximum(a0, min_eps)
    a = a / a.sum()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # current objective
        val = 2.0 * n * M
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP‑refine it={it:05d}] new best {val:.6f}")

        # active set – indices whose value is within 1e‑12·M of the maximum
        eps = max(1e-12, M * 1e-9)
        active = np.where(conv >= M - eps)[0]
        # keep only a few (≤5) active rows to keep LP tiny
        if len(active) > 5:
            active = np.random.choice(active, 5, replace=False)

        # build A matrix: each row k is a shifted copy of a (reversed shift)
        rows = []
        idx = np.arange(n)
        for k in active:
            diff = k - idx
            mask = (diff >= 0) & (diff < n)
            row = np.zeros(n, dtype=np.float64)
            row[mask] = a[diff[mask]]
            rows.append(row)
        A = np.vstack(rows) if rows else np.empty((0, n))

        # Solve LP: maximise Σb  ⇔ minimise -Σb
        if A.shape[0] > 0:
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0.0, None)] * n
            res = linprog(
                c,
                A_ub=A,
                b_ub=np.full(A.shape[0], M, dtype=np.float64),
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
            if res.success:
                b = res.x
            else:
                # fall back to uniform direction if the LP fails
                b = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            b = np.full(n, 1.0 / n, dtype=np.float64)

        sum_b = b.sum()
        if sum_b <= 0:
            b = np.full(n, 1.0 / n, dtype=np.float64)
            sum_b = 1.0
        b = b / sum_b

        # convex combination – step size shrinks like 1/√it
        t = 0.5 / np.sqrt(it + 1)
        a = (1.0 - t) * a + t * b
        a = np.maximum(a, min_eps)
        a = a / a.sum()

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine combines an L2‑flattening Adam phase, a tiny‑LP refinement,
    a smooth‑max Adam finetune and a final GD/annealing fallback.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # problem size – larger n gives more freedom but costs more per iteration
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous best is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a pool of diverse initial vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())

    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))
    pool.append(_make_uniform(n))
    pool.append(_make_exponential(n, gamma=0.5))
    pool.append(_make_powerlaw(n, expo=-1.2))
    pool.append(_make_triangle(n))

    best_seq = None
    best_val = float("inf")

    # uniform target for the L2‑flattening phase
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Iterate over the seed pool
    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ---- Phase 1: L2‑flattening Adam ------------------------------
        phase_dead = min(deadline, time.time() + 20.0)

        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Candidate {idx}] L2‑Adam improved to {val_l2:.6f}")

        # ---- Phase 2: LP refinement ----------------------------------
        if time.time() < deadline - 5:
            phase_dead = min(deadline, time.time() + 15.0)
            a_lp, val_lp = _lp_refine(
                best_seq, n, conv_fft_len, phase_dead, max_iters=4000
            )
            if val_lp < best_val:
                best_val = val_lp
                best_seq = a_lp.copy()
                print(f"[Candidate {idx}] LP‑refine improved to {val_lp:.6f}")

        # ---- Phase 3: smooth‑max Adam fine‑tuning --------------------
        if time.time() < deadline - 5:
            phase_dead = min(deadline, time.time() + 12.0)
            a_adam, val_adam = _adam_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, phase_dead, max_iters=250_000
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Candidate {idx}] Smooth‑Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Global projected‑gradient cleanup (cheap, fast)
    if best_seq is not None and time.time() < deadline - 10:
        print("[Fine] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=150_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Optional simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 10:
        print("[Anneal] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=100_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty solution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = float("inf")

    # Clip to the allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation (uses the provided reference implementation)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best value (internal) {best_val:.6f}")

    return best_seq.tolist()

def _max_conv_descent(
    a0: np.ndarray,

    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Simple projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")
        it += 1
    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val
