"""
Propose a low‑value step‑function using a multi‑stage optimisation:

1. Initialise a diverse pool (warm start, Dirichlet draws, analytic families).
2. Keep the few best candidates after a cheap evaluation.
3. L2‑flattening Adam – minimise the squared distance to the uniform
   convolution (the theoretical lower bound).  This gives a very flat
   starting point.
4. Smooth‑max Adam – minimise a log‑sum‑exp surrogate of the max‑convolution
   to further reduce peaks.
5. L‑BFGS (SciPy) on the exact objective 2·n·max(conv) using a soft‑max
   parametrisation; this quasi‑Newton step often jumps past the Adam
   plateau.
6. Exponentiated‑gradient sub‑gradient descent for the exact max‑convolution
   until the time budget expires.
7. A short simulated‑annealing sweep as a final polish.

All steps respect the non‑negativity and simplex constraints.
The routine returns the best sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser – L2 flattening -----------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the L2 distance between conv(a,a) and a uniform target.
    Returns the best vector encountered and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of (conv‑target)^2 w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        # exact objective (max‑conv)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val

            print(f"[SmoothAdam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser – exponentiated‑gradient ------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing ----------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Simple annealing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the exact objective ----------------------------
def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Quasi‑Newton (L‑BFGS‑B) optimisation of 2·n·max(conv) via soft‑max."""
    eps = 1e-15
    logits0 = np.log(a0 + eps)

    # objective (exact max‑conv)
    def fun(logits):
        mx = logits.max()
        exp_shift = np.exp(logits - mx)
        a = exp_shift / exp_shift.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        return 2.0 * n * max_c

    # gradient w.r.t. logits (chain rule through soft‑max)
    def jac(logits):
        mx = logits.max()
        exp_shift = np.exp(logits - mx)
        a = exp_shift / exp_shift.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.zeros_like(conv)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w[active] = 1.0 / active.sum()
        # grad of max(conv) w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len) * (2.0 * n)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        return grad_log.astype(np.float64)

    # run L‑BFGS‑B – it respects no bounds (soft‑max guarantees feasibility)
    res = minimize(fun=fun,
                   x0=logits0,
                   jac=jac,
                   method='L-BFGS-B',
                   options={'maxiter': max_iter, 'disp': False})

    logits_opt = res.x
    mx = logits_opt.max()
    exp_shift = np.exp(logits_opt - mx)
    a_opt = exp_shift / exp_shift.sum()
    val = res.fun
    return a_opt, val

# Main search routine ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a combination of L2‑flattening Adam, smooth‑max Adam,
    L‑BFGS‑B refinement, exponentiated‑gradient sub‑gradient and a final
    simulated‑annealing sweep.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – you may increase it if time allows
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous sequence is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a small yet diverse pool of initial vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())

    # random Dirichlet draws with varying concentration
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))

    # simple analytic families with a few parameter choices
    pool.append(_make_uniform(n))
    pool.append(_make_exponential(n, gamma=0.5))
    pool.append(_make_exponential(n, gamma=1.0))
    pool.append(_make_powerlaw(n, expo=-1.2))
    pool.append(_make_powerlaw(n, expo=-0.8))
    pool.append(_make_triangle(n))

    # ------------------------------------------------------------------
    # Quick evaluation and keep the top few starters
    evals = []
    for a in pool:
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        evals.append((val, a))
    evals.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in evals[:5]]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – L2‑flattening Adam (short but effective)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
    for idx, a0 in enumerate(top_candidates):
        if time.time() > deadline:
            break
        phase_dead = min(deadline, time.time() + 15.0)   # 15 s per candidate
        a_l2, val_l2 = _adam_optimize_l2(a0, n, conv_fft_len,
                                         corr_fft_len, target,
                                         phase_dead,
                                         max_iters=200_000, lr=0.05)
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Candidate {idx}] L2‑Adam improved to {val_l2:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max Adam (fine‑tune the max directly)
    if best_seq is not None and time.time() < deadline - 20:
        phase_dead = min(deadline, time.time() + 20.0)
        a_smooth, val_smooth = _adam_optimize(best_seq, n,
                                              conv_fft_len, corr_fft_len,
                                              phase_dead,
                                              max_iters=400_000)
        if val_smooth < best_val:
            best_val = val_smooth
            best_seq = a_smooth.copy()
            print(f"[SmoothAdam] improved to {val_smooth:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – L‑BFGS‑B refinement on the exact max‑convolution
    if best_seq is not None and time.time() < deadline - 15:
        # allocate a few seconds; L‑BFGS converges very quickly
        a_lbfgs, val_lbfgs = _lbfgs_optimize(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            deadline,
                                            max_iter=2500)
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"[L‑BFGS] improved to {val_lbfgs:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient sub‑gradient descent (runs until budget end)
    if best_seq is not None and time.time() < deadline - 5:
        a_exp, val_exp = _subgradient_optimize_expgrad(best_seq, n,
                                                       conv_fft_len,
                                                       corr_fft_len,
                                                       deadline,
                                                       max_iters=800_000,
                                                       eta0=0.8)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[ExpGrad final] improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – short simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 5:
        a_ann, val_ann = _anneal_optimize(best_seq, n,
                                          conv_fft_len,
                                          deadline,
                                          max_moves=150_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal final] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation using the provided function (for reporting)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final value reported by evaluate_sequence: {final_val:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value: {best_val:.6f}")

    return best_seq.tolist()
