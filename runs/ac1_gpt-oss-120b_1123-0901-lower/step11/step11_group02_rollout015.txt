    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) used for exact gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories – all return a probability vector (sum = 1)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a

# Optimisers (adapted from the previous state‑of‑the‑art)

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=250_000, lr=0.06):
    """Adam on a smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max re‑parameterisation (ensures sum=1)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)
        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _expgrad_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                      max_iters=600_000, eta0=0.9):
    """Exact‑max exponentiated‑gradient sub‑gradient phase."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0, n, conv_fft_len, corr_fft_len, deadline,
                     max_iters=200_000):
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


def _mass_transfer_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                           max_iters=150_000):
    """
    Tiny local moves: shift a little mass from the coordinate with the
    largest gradient (w.r.t. the current max‑convolution) to the one with
    the smallest gradient.  Accept only strict improvements.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        improved = False
        src_mass = a[i_high]
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # random tweak to escape plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1
    return best_a, best_val


def _anneal_optimize(a0, n, conv_fft_len, deadline,
                    max_moves=120_000):
    """Short simulated‑annealing pass on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1
    return best_a, best_val

# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  Returns the best sequence found within the
    allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the size used in the reference implementation
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # 1) Random sampling phase – quick scan for promising starters
    pool = []

    # start with warm‑start if available
    if warm is not None:
        pool.append(warm.copy())

    # a bunch of diverse Dirichlet draws
    for _ in range(1000):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # a few handcrafted families (they are cheap, we keep them)
    pool.append(_make_uniform(n))
    pool.append(_make_triangle(n))
    pool.append(_make_cosine(n))
    for gamma in (0.2, 0.5, 1.0, 2.0):
        pool.append(_make_exponential(n, gamma))
    for sigma in (0.1, 0.25, 0.5):
        pool.append(_make_gaussian(n, sigma))
    for expo in (-2.0, -1.5, -1.0):
        pool.append(_make_powerlaw(n, expo))
    for k in (32, 64, 128):
        pool.append(_make_spikes(n, k))
    for a, b in [(0.5, 0.5), (0.8, 0.8), (2.0, 2.0), (5.0, 5.0)]:
        pool.append(_make_beta(n, a, b))

    # Keep only the 30 most promising (by cheap FFT‑based max)
    def cheap_score(vec):
        conv = _full_convolution(vec, conv_fft_len)
        return 2.0 * n * conv.max()

    # Evaluate pool quickly (avoid calling evaluate_sequence)
    scored = []
    for vec in pool:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() < 0.01:
            continue
        vec = vec / vec.sum()
        score = cheap_score(vec)
        scored.append((score, vec))
    scored.sort(key=lambda t: t[0])
    elite = [vec for _, vec in scored[:5]]   # keep 5 elites

    # ------------------------------------------------------------------
    # 2) Deep optimisation of each elite vector
    best_seq = None
    best_val = float("inf")
    elite_idx = 0
    for a0 in elite:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising elite #{elite_idx} ===")
        elite_idx += 1

        # allocate a slice of remaining time (≈ 0.2·budget for each elite)
        slice_dead = time.time() + min(30.0, (deadline - time.time()) * 0.2)

        # Adam phase
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, slice_dead,
            max_iters=300_000, lr=0.06
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Elite] Adam improved to {val_adam:.6f}")

        # Exponentiated‑gradient phase
        if time.time() < deadline - 5:
            a_sub, val_sub = _expgrad_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=400_000, eta0=0.7
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Elite] ExpGrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3) Final fine‑grained refinement on the current best
    if best_seq is not None and time.time() < deadline - 12:
        print("\n[Fine] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    if best_seq is not None and time.time() < deadline - 8:
        print("\n[Fine] Mass‑transfer local search")
        a_mt, val_mt = _mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[Fine] Mass‑Transfer improved to {val_mt:.6f}")

    if best_seq is not None and time.time() < deadline - 6:
        print("\n[Fine] Simulated‑annealing pass")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=150_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Fine] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional up‑sampling (if generous time left)
    if best_seq is not None and time.time() + 25 < deadline:
        try:
            print("\n[Upsample] trying a finer grid (8192 points)")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 15.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=120_000, lr=0.07
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] refined best {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # ------------------------------------------------------------------
    # 5) Guarantee a non‑empty return and final evaluation
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation using the provided function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
