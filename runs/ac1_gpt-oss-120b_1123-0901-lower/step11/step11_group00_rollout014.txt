def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # choose list of lengths
    lengths = [512, 1024, 2048, 4096]
    best_seq = None
    best_val = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break
        conv_fft_len = _next_pow2(2*n - 1)
        # init candidates
        cand = _initial_candidates(n, None)   # but adapt _initial_candidates to take n and maybe warm
        # maybe also include previous best resized
        if best_seq is not None:
            # resize previous best to n via interpolation
            old = np.array(best_seq)
            # simple linear interpolation to n
            if old.shape[0] != n:
                idx_new = np.linspace(0, old.shape[0]-1, n)
                old_interp = np.interp(idx_new, np.arange(old.shape[0]), old)
                old_interp = np.maximum(old_interp, 0)
                old_interp /= old_interp.sum() if old_interp.sum()>0 else 1
                cand.append(old_interp)
        # cand limit maybe 30

        # run deep optimization for first few candidates
        max_candidates = 5
        for i, a0 in enumerate(cand):
            if time.time() >= deadline:
                break
            if i >= max_candidates:
                break
            # ensure simplex
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # run spectral flattening heavily
            a_spec, val_spec = _spectral_flatten_optimize(a0, n, conv_fft_len, deadline, max_iters=5000)
            # run random-phase refinement heavily (more iterations)
            a_phase, val_phase = _random_phase_refine(a_spec, n, conv_fft_len, deadline, max_iters=200_000)
            # run Adam for some iterations
            sub_dead = min(deadline, time.time() + 30.0)
            a_adam, val_adam = _adam_optimize(a_phase, n, conv_fft_len, conv_fft_len, sub_dead, max_iters=600_000)
            # evaluate final
            val = evaluate_sequence(a_adam.tolist())
            if val < best_val:
                best_val = val
                best_seq = a_adam.copy()
                print(f"[Best n={n} cand={i}] value {val}")

        # if time left, run quick random-phase sampling directly on best_seq of this n maybe
        if best_seq is not None and time.time() < deadline:
            # generate random-phase many times
            a_cur = best_seq if len(best_seq) == n else np.interp(np.linspace(0, len(best_seq)-1, n), np.arange(len(best_seq)), best_seq)
            a_cur = np.maximum(a_cur,0); a_cur/=a_cur.sum()
            a_rand, val_rand = _random_phase_refine(a_cur, n, conv_fft_len, deadline, max_iters=200_000)
            val = evaluate_sequence(a_rand.tolist())
            if val < best_val:
                best_val = val
                best_seq = a_rand.copy()
                print(f"[Final n={n}] improved to {val}")

    # fallback create uniform
    if best_seq is None:
        n=512
        best_seq = np.full(n, 1.0/n)

    # convert to list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Overall best {best_val:.6f}, length {len(best_seq)}")
    return best_seq.tolist()

    f = f_initial.copy()
    f = np.maximum(f,0)
    if f.sum() == 0:
        f = np.random.dirichlet(np.ones(n))
    f = f / f.sum()
    best_f = f.copy()
    best_val = _objective_val(f,n)
    for i in range(max_iters):
        if time.time() >= deadline:
            break
        M = _max_conv(f, n)  # conv max
        # compute conv matrix
        rows = 2*n-1
        C = np.zeros((rows, n))
        for col in range(n):
            C[col:col+n, col] = f
        # linear program
        # b lower bounds zero
        # maximize sum(b) => minimize -sum(b)
        # constraints C @ b <= M
        res = linprog(c=-np.ones(n), A_ub=C, b_ub=np.full(rows, M), bounds=[(0, None)]*n, method='highs', options={'presolve': True})

            break

        sum_b = b.sum()

        val = _objective_val(f,n)

            best_val = val
            best_f = f.copy()

        # Early stop if improvement small
    return best_f, best_val

"""
Hybrid optimizer for step‑function autocorrelation.
Combines a linear‑programmable “max‑convolution” reduction (small lengths),
spectral‑flattening (constant‑magnitude FFT) and extensive random‑phase
search, followed by a smooth‑max Adam fine‑tuning.  The routine respects a
global deadline (≈ 1000 s) and always returns the best simplex‑constrained
vector found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# --------------------- utility helpers --------------------- #
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Linear convolution a*a computed by FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (a on the simplex)."""
    conv = _conv_fft(a)
    return float(conv.max())


def _objective_val(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence (sum(a) assumed >0)."""
    n = a.shape[0]
    return 2.0 * n * _max_convolution(a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# --------------------- optimisation primitives --------------------- #
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate of max(conv).  Returns the best iterate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective_val(a0)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve the current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of max‑convolution w.r.t. a (cross‑correlation)
        # compute correlation w ★ a
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = a.shape[0] - 1
        grad_a = 2.0 * corr[start: start + a.shape[0]]

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Alternating projection onto (i) fixed‑energy constant‑magnitude FFT
    (except the DC term) and (ii) the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_val(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        # evaluate current objective
        val = _objective_val(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # -------- spectral flattening step --------
        A = np.fft.fft(a)
        energy = np.dot(a, a)                     # ||a||_2²

        # keep DC term, equalise the rest
        if n > 1:
            # total energy in non‑DC frequencies = n*energy - |A0|²
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                 # preserve DC (real)
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # projection onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Keep the magnitude of a0 fixed, generate many random phase patterns,
    project each inverse‑FFT back onto the simplex and keep the best.
    """
    mag = np.abs(np.fft.fft(a0))

    a_best = a0.copy()
    best_val = _objective_val(a0)
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        # random phases for the positive frequencies
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]                     # DC unchanged

        # positive frequencies
        A[1:half] = mag[1:half] * np.exp(1j * phi)
        # negative frequencies (conjugate symmetry)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            # Nyquist frequency (real) – random sign
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))

        # project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = _objective_val(a)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 10_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val


def _lp_iterative_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_outer_iters: int = 15,
) -> tuple[np.ndarray, float]:
    """
    Small‑scale linear‑programming based reduction of the maximal
    aperiodic autocorrelation.  Only used for n ≤ 1024 because the LP
    matrix has size (2n‑1)×n.
    """
    f = a0.copy()
    f = np.maximum(f, 0.0)
    if f.sum() < 1e-12:
        f = np.random.dirichlet(np.ones(n))
    f = f / f.sum()

    best_f = f.copy()
    best_val = _objective_val(f)

    rows = 2 * n - 1

    for it in range(max_outer_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution(f)

        # Build the (2n‑1)×n convolution matrix C such that C @ b = conv(f,b)
        C = np.zeros((rows, n), dtype=np.float64)
        for col in range(n):
            C[col:col + n, col] = f

        # Linear programme: maximise 1ᵀb  ⇔ minimise –1ᵀb
        res = linprog(
            c=-np.ones(n),
            A_ub=C,
            b_ub=np.full(rows, M),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP it={it}] linprog failed: {res.message}")
            break

        b = res.x
        sb = b.sum()
        if sb < 1e-12:
            break
        b = b / sb

        # Simple convex combination
        t = 0.2
        f = (1.0 - t) * f + t * b
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        val = _objective_val(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP n={n} it={it}] new best {val:.6f}")

        # early stopping if improvement negligible
        if abs(val - best_val) < 1e-12:
            break

    return best_f, best_val

"""
Hybrid optimizer for the step‑function autocorrelation problem.

The routine works under a global time budget (≈ 1000 s) and always returns a
non‑negative vector on the probability simplex that minimises
evaluate_sequence(...).  The search proceeds through the following stages:

1. **Small‑scale LP reduction** – for n ≤ 1024 a linear programme
   (maximise Σ b_j  subject to  b≥0 and  f★b ≤ max(f★f)) shrinks the peak
   of the autocorrelation.  This is fast and often gives a strong first‑
   order improvement.

2. **Spectral flattening** – alternating projections force the FFT magnitude
   (except the DC component) to become constant.  Constant magnitude implies
   a flat autocorrelation, which is exactly what we need.

3. **Random‑phase refinement** – keeping the magnitude fixed, many random
   phase patterns are generated, the inverse FFT is projected back onto the
   simplex, and the best candidate is retained.  This step dominates the
   run‑time and is very effective at reducing the maximal convolution entry.

4. **Smooth‑max Adam fine‑tuning** – a smooth approximation of the max‑operator
   is minimised with Adam on the simplex, polishing the solution obtained
   from the previous stages.

The algorithm explores several sequence lengths (1024, 2048, 4096, 512, 256)
so that the best trade‑off between flexibility (large n) and speed (small n)
is automatically selected.  If a previous construction `height_sequence_1`
is available it is used as a warm start (resampled to the current length).

All helper functions are top‑level; no closures or lambdas are used.

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv). Returns the best
    iterate (on the simplex) and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective_val(a0)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve the current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of max‑convolution w.r.t. a (cross‑correlation)
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = a.shape[0] - 1
        grad_a = 2.0 * corr[start: start + a.shape[0]]

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Alternating projection onto (i) constant‑magnitude FFT (except DC) and
    (ii) the probability simplex.  The best iterate is kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_val(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        val = _objective_val(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ----- enforce constant FFT magnitude (except DC) -----
        A = np.fft.fft(a)
        energy = np.dot(a, a)                     # ||a||₂²

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                 # preserve DC (real)
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # ----- project onto simplex -----
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _random_phase_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Keep the magnitude of a0 fixed, generate many random phase patterns,
    project each inverse‑FFT back onto the simplex and keep the best.
    """
    mag = np.abs(np.fft.fft(a0))

    a_best = a0.copy()
    best_val = _objective_val(a0)
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iters:
        # random phases for the positive frequencies (excluding DC)
        phi = np.random.random(half - 1) * 2.0 * np.pi
        A = np.empty(n, dtype=np.complex128)
        A[0] = mag[0]                     # DC unchanged

        # positive frequencies
        A[1:half] = mag[1:half] * np.exp(1j * phi)
        # negative frequencies (conjugate symmetry)
        A[-(half - 1):] = np.conj(A[1:half][::-1])

        if n % 2 == 0:
            # Nyquist frequency (real) – random sign
            sign = -1.0 if np.random.rand() < 0.5 else 1.0
            A[half] = sign * mag[half]

        a = np.real(np.fft.ifft(A))

        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = _objective_val(a)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            if it % 10_000 == 0:
                print(f"[PhaseRandom it={it}] new best {val:.6f}")

        it += 1

    return a_best, best_val


def _lp_iterative_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_outer_iters: int = 12,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming based reduction of the maximal aperiodic autocorrelation.
    Only used for n ≤ 1024 because the LP matrix has size (2n‑1)×n.
    """
    if linprog is None:                      # pragma: no cover
        raise RuntimeError("scipy.optimize.linprog is required for the LP stage")

    f = np.maximum(a0, 0.0)
    if f.sum() < 1e-12:
        f = np.random.dirichlet(np.ones(n))
    f = f / f.sum()

    best_f = f.copy()
    best_val = _objective_val(f)

    rows = 2 * n - 1

    for it in range(max_outer_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution(f)

        # Build the convolution matrix C such that (C @ b) = conv(f, b)
        C = np.zeros((rows, n), dtype=np.float64)
        for col in range(n):
            C[col:col + n, col] = f

        # Linear programme: maximise Σ b_j  ⇔ minimise –Σ b_j
        res = linprog(
            c=-np.ones(n),
            A_ub=C,
            b_ub=np.full(rows, M),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP it={it}] linprog failed: {res.message}")
            break

        b = res.x
        sb = b.sum()
        if sb < 1e-12:
            break
        b = b / sb

        # convex combination with a modest step size
        t = 0.20
        f = (1.0 - t) * f + t * b
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        val = _objective_val(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP n={n} it={it}] new best {val:.6f}")

        # tiny improvement check – early exit
        if abs(val - best_val) < 1e-12:
            break

    return best_f, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine respects the given time budget
    (≈ 1000 s) and always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # Lengths to explore – small lengths make the LP cheap,
    # larger lengths give more freedom for shaping the spectrum.
    lengths = [1024, 2048, 4096, 512, 256]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # -----------------------------------------------------------------
        # Build a modest, diverse pool of start vectors.
        cand = []

        # Warm start from the previous run, if available.
        warm = globals().get("height_sequence_1")
        if warm is not None:
            w = np.asarray(warm, dtype=np.float64)
            if w.shape[0] != n:
                idx = np.linspace(0, w.shape[0] - 1, n)
                w = np.interp(idx, np.arange(w.shape[0]), w)
            w = np.maximum(w, 0.0)
            if w.sum() > 0:
                w = w / w.sum()
                cand.append(w)

        # Random Dirichlet draws.
        for _ in range(3):
            cand.append(np.random.dirichlet(np.ones(n)))

        # Symmetric exponential decay.
        gamma = 0.5
        lin = np.exp(-gamma * np.arange(n))
        shape = lin + lin[::-1]
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

        # Half‑sinusoid (positive half‑wave).
        sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        sin_shape = np.maximum(sin_shape, 0.0)
        sin_shape /= sin_shape.sum()
        cand.append(sin_shape)

        # Uniform distribution (baseline).
        cand.append(np.full(n, 1.0 / n))

        # -----------------------------------------------------------------
        # Process each candidate – time allocated per candidate is limited.
        for idx, a0 in enumerate(cand):
            if time.time() >= deadline:
                break

            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # ----- 1. LP‑based reduction (only for modest lengths) -----
            if n <= 1024 and linprog is not None:
                sub_dead = min(deadline, time.time() + 20.0)
                a_lp, val_lp = _lp_iterative_optimize(a0, n, sub_dead, max_outer_iters=10)
                if val_lp < best_val:
                    best_val = val_lp
                    best_seq = a_lp.copy()
                    print(f"[n={n} cand={idx}] LP improved to {val_lp:.6f}")
                a_cur = a_lp
            else:
                a_cur = a0

            # ----- 2. Spectral flattening -----
            if time.time() < deadline - 5.0:
                sub_dead = min(deadline, time.time() + 15.0)
                a_spec, val_spec = _spectral_flatten_optimize(
                    a_cur, n, conv_fft_len, sub_dead, max_iters=8_000
                )
                if val_spec < best_val:
                    best_val = val_spec
                    best_seq = a_spec.copy()
                    print(f"[n={n} cand={idx}] Spectral improved to {val_spec:.6f}")
                a_cur = a_spec

            # ----- 3. Random‑phase refinement (heavy sampling) -----
            if time.time() < deadline - 8.0:
                sub_dead = min(deadline, time.time() + 30.0)
                a_rand, val_rand = _random_phase_refine(
                    a_cur, n, conv_fft_len, sub_dead, max_iters=250_000
                )
                if val_rand < best_val:
                    best_val = val_rand
                    best_seq = a_rand.copy()
                    print(f"[n={n} cand={idx}] Phase‑random improved to {val_rand:.6f}")
                a_cur = a_rand

            # ----- 4. Final Adam fine‑tuning (smooth‑max) -----
            if time.time() < deadline - 5.0:
                sub_dead = min(deadline, time.time() + 12.0)
                a_adam, val_adam = _adam_optimize(
                    a_cur, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
                )
                if val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={idx}] Adam final improved to {val_adam:.6f}")

    # -----------------------------------------------------------------
    # Fallback – guarantee a non‑empty return.
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Overall best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
