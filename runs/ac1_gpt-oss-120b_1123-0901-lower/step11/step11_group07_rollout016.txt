def lp_max_g(f, M):
    n = len(f)
    # Build A (2n-1, n)
    rows = 2*n-1

def downsample(seq, n):
    L = len(seq)
    factor = L // n
    if factor * n == L:

    else:
        # Use linear interpolation for remainder.
        return np.interp(np.linspace(0, L-1, n), np.arange(L), seq)

"""
Hybrid LP + exponentiated‑gradient optimiser.

* Work at a moderate size (n≈1024) so that many optimisation steps fit
  into the 1000 s budget.
* Repeatedly solve a tiny linear programme:
      maximise Σ g_j   subject to   (f★g)[k] ≤ M   (for all shifts k)
  where M = max(f★f).  The LP pushes mass into “quiet’’ regions of the
  current convolution.
* Mix the LP‑solution with the current distribution using a decreasing
  step‑size t, accepting the mix only if it strictly lowers the peak.
* After the LP‑phase, run a long exponentiated‑gradient (exact‑max)
  refinement on the remaining time.
* All vectors are kept on the probability simplex (Σa_i = 1) – the
  evaluation function is scale‑invariant.
* The final sequence is clipped to the evaluator’s range and returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities – no closures, all top‑level as required

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the exact max‑gradient.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _downsample_to(arr: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D array to a new length.
    """
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n)
    return np.interp(x_new, x_old, arr)


# ----------------------------------------------------------------------
# Linear‑programming step – maximise ℓ₁‑norm under the current peak bound

def _lp_maximize_l1(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t.   (f★g)[k] ≤ M  for all k,   g ≥ 0 .
    Returns the optimal g (not yet normalised) or None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the (rows × n) convolution matrix A where A[k, j] = f_{k‑j}
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f

    # Upper‑bound constraints A·g ≤ M
    b_ub = np.full(rows, M, dtype=np.float64)

    # Objective – maximise Σg  ⇔ minimise –Σg
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # Use the high‑performance HiGHS simplex solver
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    """

    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
