def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    n = 4096
    fft_len = _next_pow2(2*n-1)
    best_seq = None
    best_val = float('inf')
    now = time.time()
    deadline = now + budget_s - 0.5

    # include warm start from previous global variable
    warm = globals().get('height_sequence_1')
    if warm is not None:
         seq = np.asarray(warm, dtype=np.float64)
         if seq.shape[0] != n:
              if seq.shape[0] > n: seq = seq[:n]
              else: seq = np.pad(seq, (0,n-seq.shape[0]))
         seq = np.clip(seq,0,1000)
         seq = seq / seq.sum()
         val = _fast_value(seq)
         best_seq, best_val = seq.copy(), val

    # Define helper to evaluate
    def eval_seq(a):
        # a is normalized to sum 1
        conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)
        max_c = conv[:2*n-1].max()
        return 2.0 * n * max_c

    # candidate families functions as defined earlier (including param)
    # We'll generate many candidates in loop until deadline approaching.
    while time.time() < deadline - 0.5:
        # randomly choose family
        r = np.random.rand()
        if r < 0.2:
            # decreasing exponential (left)
            gamma = np.random.uniform(0.001, 5.0)  # but using left only
            i = np.arange(n, dtype=np.float64)
            a = np.exp(-gamma * i)
        elif r < 0.4:
            # power law decreasing
            expo = -np.random.uniform(0.1, 3.0)  # negative exponent
            i = np.arange(1, n+1, dtype=np.float64)
            a = i**expo
        elif r < 0.6:
            # beta distribution decreasing (alpha < 1, beta > 1)
            alpha = np.random.uniform(0.2, 0.9)
            beta = np.random.uniform(1.1, 5.0)
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            a = np.power(x, alpha-1) * np.power(1-x, beta-1)
        elif r < 0.8:
            # linear decreasing
            a = (n - np.arange(n, dtype=np.float64))
        else:
            # random Dirichlet with random concentration
            alpha = 10**np.random.uniform(-3, 2)
            a = np.random.dirichlet(alpha * np.ones(n))

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        val = eval_seq(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # After random search, perform refinement via exponentiated gradient on best_seq
    # Use subgradient iterations
    if best_seq is not None:
        a = best_seq.copy()
        # compute convolution length
        conv_fft_len = fft_len
        corr_fft_len = _next_pow2(3*n-2)
        # re-use earlier function for subgrad? Let's implement simplified loop
        m = time.time()
        while time.time() < deadline - 0.5 and (time.time() - m) < 200.0:  # up to 200 sec
            # conv
            conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)
            conv = conv[:2*n-1]
            max_c = conv.max()
            val = 2.0 * n * max_c
            # record best
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Refine] improved to {val:.6f}")
            # find active indices (lag(s) achieving max)
            active = np.isclose(conv, max_c, atol=1e-14)
            if not np.any(active):
                break
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
            # gradient: 2 * cross-correlation of w with a
            # Use _cross_correlation defined earlier
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
            # learning rate schedule
            it = (time.time() - m) * 1000  # rough iteration count
            eta = 0.9 / np.sqrt(it+1)
            a = a * np.exp(-eta * grad)
            a_sum = a.sum()
            if a_sum <= 0:
                a = np.random.dirichlet(np.ones(n))
                a_sum = a.sum()
            a = a / a_sum

    # final output
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
        best_val = eval_seq(best_seq)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {n})")
    return best_seq.tolist()

"""
Hybrid search for the step‑function with the smallest evaluation score.

The algorithm consists of three phases:

1. **Diverse random sampling** – generate many candidates from simple
   parametric families (exponential, power‑law, beta, linear, mixtures,
   Dirichlet).  Each candidate is normalised to sum = 1 and evaluated with a
   fast FFT‑based convolution (the true objective equals 2·n·max(conv)
   when the sum is 1).  The best candidate seen so far is kept.

2. **Sub‑gradient refinement** – starting from the current best, apply an
   exponentiated‑gradient (mirror‑descent) update using the exact sub‑gradient
   of the max‑convolution.  The update keeps the vector on the probability
   simplex and runs until the time budget is almost exhausted.

3. **Local simulated‑annealing polish** – a final hill‑climbing stage
   randomly moves a small fraction of mass from one index to another.
   Improvements are kept, otherwise moves are accepted with a temperature
   schedule.

All steps respect the 1000 s time limit and always return a non‑negative
sequence (the evaluator also clips to [0,1000]).

The implementation avoids closures, uses only NumPy and the provided
`evaluate_sequence` for the final check, and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, corr_fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, corr_fft_len)
    fr = np.fft.rfft(rev_a, corr_fft_len)
    corr = np.fft.irfft(fw * fr, corr_fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories (no closures)

def _make_exponential_decreasing(n, gamma):
    """a_i ∝ exp(-γ·i)  (monotonically decreasing, not reflected)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * i)
    a = a / a.sum()
    return a


def _make_powerlaw_decreasing(n, p, shift):
    """a_i ∝ 1 / (i+shift)^p   with p>0, shift>0."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(i + shift, p)
    a = a / a.sum()
    return a


def _make_linear_decreasing(n):
    """a_i ∝ (n-i)   (simple linear decay)."""
    i = np.arange(n, dtype=np.float64)
    a = n - i
    a = a / a.sum()
    return a


def _make_beta_decreasing(n, alpha, beta):
    """Beta distribution on [0,1] with α<1, β>1 → decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def _make_dirichlet(n, conc):
    """Random Dirichlet draw with concentration `conc`."""
    a = np.random.dirichlet(conc * np.ones(n))
    return a.astype(np.float64)


def _make_mixture(n, a1, a2, lam):
    """Convex combination lam·a1 + (1-lam)·a2."""
    a = lam * a1 + (1.0 - lam) * a2
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Fast objective (used during the search)

def _fast_eval(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(conv) assuming sum(a)=1 (scale‑invariant)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Sub‑gradient (exponentiated‑gradient) optimiser

def _expgrad_refine(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    deadline: float,
                    max_iters: int = 1_000_000,
                    eta0: float = 0.9) -> np.ndarray:
    """Exact‑max sub‑gradient optimisation via exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _fast_eval(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # weight vector for the active max‑lag(s)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max(conv) = 2·(w ★ a)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing learning rate
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step (mirror descent on the simplex)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing polish (local moves)

def _anneal_polish(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  deadline: float,
                  max_moves: int = 300_000) -> np.ndarray:
    """Simple simulated‑annealing temperature sweep on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4   # up to 40 % of a[i] moved
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (default 1000 s) and

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # small safety margin

    # Problem size – a power‑of‑two length gives fast FFTs.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previous run (if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Keep track of the best sequence seen so far
    best_seq = None
    best_val = float("inf")

    if warm is not None:
        val = _fast_eval(warm, n, conv_fft_len)
        best_seq = warm.copy()
        best_val = val
        print(f"[Init] warm start value {val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – random sampling of parametric families
    families = ["exp", "pow", "beta", "lin", "mix", "dir"]
    # continue sampling while we still have at least ~60 s left for refinements
    while time.time() < deadline - 60.0:
        choice = np.random.choice(families)
        if choice == "exp":
            gamma = 10 ** np.random.uniform(-2, 1)          # 0.01 … 10
            cand = _make_exponential_decreasing(n, gamma)
        elif choice == "pow":
            p = np.random.uniform(0.3, 2.5)                # exponent >0

        elif choice == "beta":
            alpha = np.random.uniform(0.2, 0.9)            # α<1  → decreasing

        elif choice == "lin":
            cand = _make_linear_decreasing(n)
        elif choice == "mix":
            # mix two random Dirichlet draws

            a2 = _make_dirichlet(n, 10 ** np.random.uniform(-2, 1))
            lam = np.random.rand()
            cand = _make_mixture(n, a1, a2, lam)
        else:  # "dir"
            conc = 10 ** np.random.uniform(-3, 2)         # 1e‑3 … 100
            cand = _make_dirichlet(n, conc)

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        val = _fast_eval(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random] new best {val:.6f}")

    # ------------------------------------------------------------------

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report final value using the official evaluator (optional, for logging)

    return best_seq.tolist()
