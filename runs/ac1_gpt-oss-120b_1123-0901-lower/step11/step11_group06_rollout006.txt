"""
Hybrid random‑restart + multi‑phase optimizer for the max‑convolution
objective.  The algorithm works on a probability simplex (non‑negative,
sum = 1) and aims to minimise

    2 · n · max(a * a) / (∑a)²  =  2·n·max(a * a)   (since we normalise ∑a=1).

The search proceeds as follows:

1.  Build a large, diverse pool of starter vectors (Dirichlet draws,
    exponential tails, double‑Gaussian bumps, spike patterns and
    random‑phase “flat‑spectrum’’ vectors).  All are projected onto the
    simplex and the 15 best (according to a cheap FFT‑based evaluation)
    are kept.

2.  Starting from the current best, run a long Adam optimisation on a
    smooth‑max surrogate of the max‑convolution.

3.  Refine with an exact‑max exponentiated‑gradient sub‑gradient phase.

4.  Apply an intensive spectral‑flattening phase that enforces (almost)
    constant Fourier magnitude while staying on the simplex.

5.  Perform projected gradient descent directly on the exact max‑convolution.

6.  Run a simulated‑annealing local search.

7.  If time permits, execute several short “perturb‑and‑refine’’ cycles.

All phases respect the global deadline (≈ 1000 s).  The routine always
returns the best sequence found – a list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for sub‑gradient / gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _normalize(v: np.ndarray) -> np.ndarray:
    """Force non‑negativity and normalise to unit sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12 or np.isnan(s):
        v = np.full_like(v, 1.0 / v.shape[0])
        s = 1.0
    return v / s


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.2                # initial temperature for smooth‑max
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to get current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the FFT magnitude (except DC) to be (almost) constant,
    projecting back onto the probability simplex after each inverse FFT.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")

        A = np.fft.fft(a)                       # full complex FFT (size n)
        energy = np.dot(a, a)                   # ‖a‖₂²

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                         # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    for gamma in (0.15, 0.3, 0.6):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.6, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.45):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))

    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        cand.append(a)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (48, 96, 192):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _random_phase_candidates(n: int, count: int = 30) -> list[np.ndarray]:
    """Generate vectors whose DFT magnitude is (almost) constant (random phases)."""
    out = []
    for _ in range(count):
        # Create a Hermitian‑symmetric spectrum with constant magnitude.
        mag = np.sqrt(n)                    # arbitrary magnitude, later renorm.
        phases = np.random.uniform(0, 2 * np.pi, n // 2 - 1)
        A = np.empty(n, dtype=complex)
        A[0] = mag                          # DC term (real)
        A[1:n // 2] = mag * np.exp(1j * phases)
        if n % 2 == 0:
            A[n // 2] = mag * np.exp(1j * np.random.uniform(0, 2 * np.pi))
            A[n // 2 + 1:] = np.conj(A[1:n // 2][::-1])
        else:
            A[n // 2 + 1:] = np.conj(A[1:n // 2][::-1])
        a = np.real(np.fft.ifft(A))
        a = np.maximum(a, 0.0)
        a = _normalize(a)
        out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    # Vector length – 4096 is a good trade‑off.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a large, diverse pool of starter vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)
    # Additional Dirichlet draws with random concentration
    for _ in range(30):
        conc = 10 ** np.random.uniform(-3, 1)   # between 0.001 and 10
        a = np.random.dirichlet(conc * np.ones(n))
        pool.append(a)
    # Random‑phase “flat‑spectrum’’ candidates
    pool.extend(_random_phase_candidates(n, count=30))

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool, keep the best few (15)
    candidates = []
    for a in pool:
        a = _normalize(a)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c          # same as evaluate_sequence for sum=1
        candidates.append((val, a))

    candidates.sort(key=lambda x: x[0])
    top_k = 15
    top_candidates = candidates[:top_k]

    best_seq = top_candidates[0][1].copy()
    best_val = top_candidates[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – Adam optimisation (≈ 40 % of remaining budget)
    remaining = deadline - time.time()
    if remaining > 10.0:
        adam_deadline = min(deadline, time.time() + remaining * 0.4)
        best_seq, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=2_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            print(f"[Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – Exact‑max exponentiated‑gradient sub‑gradient
    remaining = deadline - time.time()
    if remaining > 8.0:
        sub_deadline = deadline
        best_seq, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=1_200_000, eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            print(f"[Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase III – Intensive spectral flattening
    remaining = deadline - time.time()
    if remaining > 12.0:
        print("[Spectral] Starting intensive spectral flattening")
        best_seq, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_iters=3_000_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            print(f"[Spectral] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – Projected gradient descent on exact max‑conv
    remaining = deadline - time.time()
    if remaining > 12.0:
        print("[GD] Projected gradient descent")
        best_seq, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=800_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            print(f"[GD] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase V – Simulated‑annealing refinement
    remaining = deadline - time.time()
    if remaining > 12.0:
        print("[Anneal] Starting simulated‑annealing refinement")
        best_seq, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=800_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            print(f"[Anneal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase VI – Short perturb‑and‑refine cycles (if time permits)
    remaining = deadline - time.time()
    if remaining > 5.0:
        cycles = max(1, int(remaining // 5))
        print(f"[Perturb] Running {cycles} short refinement cycles")
        for c in range(cycles):
            noise = np.random.randn(n) * 0.001
            a_noise = best_seq + noise
            a_noise = np.maximum(a_noise, 0.0)
            a_noise = _project_simplex(a_noise)

            sub_dead = min(deadline, time.time() + 4.5)
            a_ref, val_ref = _max_conv_descent(
                a_noise, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=30_000,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Perturb] cycle {c} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
