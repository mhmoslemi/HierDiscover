def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    warm = globals().get("height_sequence_1")
    # process warm ...

    pool = _initial_candidates(n, warm) + _spike_candidates(n)
    # add some random perturbed warm
    if warm is not None:
        for _ in range(3):
            pert = warm + 0.01*np.random.randn(n)
            pert = np.maximum(pert, 0)
            pert /= max(pert.sum(), 1e-12)
            pool.append(pert)

    best_seq = None
    best_val = float('inf')
    max_candidates = 6

    # Stage: candidate optimization
    for idx, a0 in enumerate(pool[:max_candidates]):
        if time.time() > deadline:
            break
        # Normalize a0
        a0 = np.maximum(a0,0)
        if a0.sum()<0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        # Adam
        sub_dead = min(deadline, time.time() + 15.0)
        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=500_000)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Explore {idx}] Adam best {val_adam:.6f}")
        # subgrad if time left
        if time.time() < deadline - 5:
            sub_dead2 = min(deadline, time.time() + 15.0)
            a_sub, val_sub = _subgradient_optimize_expgrad(best_seq, n, conv_fft_len, corr_fft_len, sub_dead2, max_iters=400_000, eta0=1.0)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Explore {idx}] Subgrad best {val_sub:.6f}")

    # Now iterative refinement loop
    while time.time() < deadline - 5:
        time_left = deadline - time.time()
        # choose a stage based on remaining time and maybe cycle variable
        # We'll schedule each stage for approx 30-60 sec
        # Use spectral flatten
        stage_time = min(45.0, time_left)
        a_spec, val_spec = _spectral_flatten_optimize(best_seq, n, conv_fft_len, time.time() + stage_time, max_iters=200_000)
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")
        # LP improvement
        if time.time() < deadline - 5:
            stage_time = min(60.0, deadline - time.time())
            a_lp, val_lp = _lp_improve(best_seq, n, conv_fft_len, time.time() + stage_time, max_iter=20, tau=0.5)
            if val_lp < best_val:
                best_val, best_seq = val_lp, a_lp.copy()
                print(f"[LP] improved to {val_lp:.6f}")

        # Gradient descent
        if time.time() < deadline - 5:
            stage_time = min(40.0, deadline - time.time())
            a_gd, val_gd = _max_conv_descent(best_seq, n, conv_fft_len, corr_fft_len, time.time() + stage_time, max_iters=150_000)
            if val_gd < best_val:
                best_val, best_seq = val_gd, a_gd.copy()
                print(f"[GD] improved to {val_gd:.6f}")

        # Annealing
        if time.time() < deadline - 5:
            stage_time = min(40.0, deadline - time.time())
            a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len, time.time() + stage_time, max_moves=80_000)
            if val_ann < best_val:
                best_val, best_seq = val_ann, a_ann.copy()

"""
Hybrid multi‑stage optimizer for the step‑function attack.

The algorithm builds a diverse pool of start vectors, runs a short
Adam optimisation on a smooth‑max surrogate, follows with an exact
sub‑gradient (exponentiated‑gradient) phase, then repeatedly applies
spectral flattening, a linear‑programming refinement, projected‑gradient
descent and simulated‑annealing.  The stages are run in a loop until
the 1000 s budget is exhausted, allowing many refinement cycles and
much longer optimisation than the previous baseline.  The goal is to
push the evaluation value below the target 1.503.

All vectors are kept non‑negative and normalised (sum = 1); the
evaluation function is homogeneous, so this does not affect the score.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Returns the best sequence found and its (unnormalised) objective
    value 2·n·max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Iteratively force the FFT magnitude to be (almost) constant."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # include the warm start if present
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with different concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Symmetric exponential shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Polynomial shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Gaussian bumps at both ends
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # A single sine bump
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _lp_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 20,
    tau: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence a we maximise Σb subject to (a★b)_k ≤ M,
    where M = max_k (a★a)_k. The solution b is normalised and mixed with a.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # Active constraints (indices attaining the peak, with tiny slack)
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        # Add a few well‑spread positions to keep the LP non‑degenerate
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        # Build sparse constraint matrix A (rows = active constraints, cols = variables b)
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        # Linear programme: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # Mix with controlled step size
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce mixing factor if no improvement
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Return a (non‑negative) step‑function sequence with low evaluation value."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Length of the vector – a trade‑off between expressiveness and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous best is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of start vectors.
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # Some extra random Dirichlet draws.
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    # Random piece‑wise constant blocks.
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 6   # a handful of promising starts

    # ------------------------------------------------------------------
    # Phase I – quick Adam + sub‑gradient refinements on a few starts.
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Adam optimisation (≈15 s per candidate)
        sub_dead = min(deadline, time.time() + 15.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=500_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # Sub‑gradient phase (≈12 s if time permits)
        if time.time() < deadline - 5:
            sub_dead2 = min(deadline, time.time() + 12.0)
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, sub_dead2,
                max_iters=400_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – iterative refinement loop (spectral → LP → GD → anneal)
    # The loop repeats as long as there is time left; each stage gets a
    # modest time slice so that many cycles can be performed.
    cycle = 0
    while time.time() < deadline - 5:
        remaining = deadline - time.time()
        if remaining < 30.0:   # not enough time for a full cycle
            break
        print(f"[Cycle {cycle}] remaining time {remaining:.1f}s")
        # 1. Spectral flattening
        stage_time = min(40.0, remaining / 4.0)
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, time.time() + stage_time, max_iters=200_000
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Cycle {cycle}] Spectral improved to {val_spec:.6f}")

        # 2. Linear‑programming refinement (more iterations than before)
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        stage_time = min(50.0, remaining / 3.0)
        a_lp, val_lp = _lp_improve(
            best_seq, n, conv_fft_len, time.time() + stage_time,
            max_iter=20, tau=0.5
        )
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[Cycle {cycle}] LP improved to {val_lp:.6f}")

        # 3. Projected gradient descent on exact max‑conv
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        stage_time = min(45.0, remaining / 2.0)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, time.time() + stage_time,
            max_iters=150_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Cycle {cycle}] GD improved to {val_gd:.6f}")

        # 4. Simulated annealing polish
        remaining = deadline - time.time()
        if remaining < 15.0:
            break
        stage_time = min(30.0, remaining)
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, time.time() + stage_time,
            max_moves=80_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Cycle {cycle}] Anneal improved to {val_ann:.6f}")

        cycle += 1

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
