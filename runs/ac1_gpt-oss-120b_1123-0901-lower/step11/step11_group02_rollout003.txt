   a = softmax_logits(logits)
   conv = _full_convolution(a, conv_fft_len)
   max_c = conv.max()
   # compute softmax weights w = exp(lam*(conv - max_c))
   # to improve numerical stability, we can subtract max_c (or max(conv)) cause larger than others.
   # But we need w that sums to 1.
   exp_arg = lam * (conv - max_c)
   # avoid overflow: use np.exp with float64, values may be up to lam*max_c; lam may be up to 100, max_c maybe ~1/n ~ 0.0003; product ~0.03 -> safe.
   w = np.exp(exp_arg)
   w_sum = w.sum()
   if w_sum == 0: w = np.full_like(conv, 1.0/conv.size)
   else: w /= w_sum
   smooth_max = max_c + np.log(w_sum)/lam   # because log-sum-exp = max_c + log(sum(exp(lam*(conv - max_c)))/lam
   # alternatively compute directly: smooth_max = np.log(np.exp(lam*conv).sum())/lam
   # but using stable form.
   obj = 2 * n * smooth_max

   # gradient w.r.t a: grad_a = 2 * cross_corr(w, a)
   grad_a = 2 * _cross_correlation(w, a, corr_fft_len)

   # gradient w.r.t logits: grad_logit = a * (grad_a - np.dot(a, grad_a))
   avg = np.dot(a, grad_a)
   grad_logit = a * (grad_a - avg)

   return obj, grad_logit

"""
Propose a low‑score step‑function via multi‑start quasi‑Newton optimisation.

Algorithm
---------
1. Build a large, diverse pool of candidate vectors (Dirichlet draws,
   exponential / Gaussian / power‑law shapes, simple spikes, beta distributions,
   uniform / triangular / cosine).  All candidates are normalised to sum = 1.

2. For each candidate (as many as time permits) run a continuation of
   smooth‑max L‑BFGS‑B optimisations:
        – the exact objective is 2·n·max(a*a) (max of the self‑convolution).
        – we replace max by a smooth log‑sum‑exp surrogate with a
          temperature λ.  Convex gradients are obtained analytically.
        – a short L‑BFGS‑B is performed for a sequence of increasing λ
          (0.5 → 2 → 5 → 10 → 20 → 50 → 100).  The variables are the
          logits of a soft‑max parametrisation, guaranteeing non‑negativity
          and the simplex constraint.

3. The result of (2) is refined by the fine‑grained “mass‑transfer’’ local
   search already used in the previous state‑of‑the‑art code: repeatedly
   move a tiny amount of mass from the coordinate with the largest gradient
   (w.r.t. the exact max‑convolution) to the one with the smallest gradient,
   accepting only strictly improving moves.

4. Keep the best vector over all candidates.  Finally clip the vector to the
   evaluator’s admissible range and return it.

The routine respects the 1000 s budget, prints intermediate best scores,
and works entirely with NumPy, SciPy and the supplied ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic FFT utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Candidate factories ----------------------------------------------------
def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max objective (log‑sum‑exp) ------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _smoothmax_objective(logits: np.ndarray, n: int,
                         conv_fft_len: int, corr_fft_len: int,
                         lam: float):
    """Return (obj, grad) where obj = 2·n·smooth_max(conv) and grad w.r.t logits."""
    # soft‑max parametrisation (ensures a≥0, sum=1)
    a = _softmax(logits)

    # exact linear convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # smooth max via log‑sum‑exp (stable)
    max_c = conv.max()
    # w_i ∝ exp(lam * (conv_i - max_c))
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
        smooth_max = max_c
    else:
        w /= w_sum
        smooth_max = max_c + np.log(w_sum) / lam

    obj = 2.0 * n * smooth_max

    # gradient w.r.t a (exact max‑conv gradient with weights w)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_logit = a * (grad_a - avg)

    return obj, grad_logit


class _SmoothMaxObj:
    """Callable object required by scipy.optimize.minimize."""
    def __init__(self, n, conv_fft_len, corr_fft_len, lam):
        self.n = n
        self.conv_fft_len = conv_fft_len
        self.corr_fft_len = corr_fft_len
        self.lam = lam

    def __call__(self, logits):
        obj, grad = _smoothmax_objective(
            logits, self.n, self.conv_fft_len, self.corr_fft_len, self.lam
        )
        return obj, grad

# Fine‑grained mass‑transfer local search -------------------------------
def _mass_transfer_optimize(a0: np.ndarray, n: int,
                           conv_fft_len: int, corr_fft_len: int,
                           deadline: float, max_iters: int = 120_000):
    """
    Repeatedly move a small amount of mass from the coordinate with the
    largest gradient (w.r.t. the exact max‑convolution) to the one with the
    smallest gradient, accepting strictly improving moves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if grad[i_high] <= grad[i_low] + 1e-15:
            break  # near‑stationary

        src_mass = a[i_high]
        improved = False
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # random jitter to escape possible plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1
    return best_a, best_val

    # ------------------------------------------------------------------
    # Problem size (tunable).  4096 gives a good trade‑off between speed
    # and expressive power; we stay with it unless the budget is huge.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of initial vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with various alphas
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential families
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0, 4.0):
        pool.append(_make_exponential(n, gamma))

    # Gaussian families
    for sigma in (0.04, 0.1, 0.2, 0.4, 0.8):
        pool.append(_make_gaussian(n, sigma))

    # Power‑law (negative exponents produce heavy tails)
    for expo in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw(n, expo))

    # Simple shapes
    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))
    pool.append(_make_cosine(n))

    # Sparse spike patterns
    for k in (32, 64, 128, 256):
        pool.append(_make_spikes(n, k))

    # Beta distributions (including the arcsine shape)
    for a, b in [(0.5, 0.5), (0.8, 0.8), (1.0, 1.0),
                 (2.0, 2.0), (3.0, 3.0), (5.0, 5.0), (10.0, 10.0)]:
        pool.append(_make_beta(n, a, b))

    # A few completely random Dirichlet draws with random concentration
    for _ in range(6):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    candidate_idx = 0
    lam_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]

    # Process candidates one‑by‑one while time permits
    while time.time() < deadline and candidate_idx < len(pool):
        a0 = np.maximum(pool[candidate_idx], 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Initialise logits for soft‑max parametrisation
        logits = np.log(a0 + 1e-15)

        # Continuation over the λ‑schedule
        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            obj = _SmoothMaxObj(n, conv_fft_len, corr_fft_len, lam)
            # modest iteration budget – enough to make progress but cheap
            max_iter = 180
            res = minimize(obj, logits, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter, 'disp': False})
            logits = res.x

        # Convert back to probabilities
        a = _softmax(logits)

        # Exact objective (no smoothing)
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()

        # Optional fine‑grained mass‑transfer refinement (if time left)
        if time.time() < deadline - 5.0:
            a, val = _mass_transfer_optimize(
                a, n, conv_fft_len, corr_fft_len, deadline, max_iters=80_000
            )

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Candidate {candidate_idx}] new best {val:.6f}")

        candidate_idx += 1

    # ------------------------------------------------------------------
    # Final polish on the overall best vector (if any time left)
    if best_seq is not None and time.time() < deadline - 3.0:
        best_seq, best_val = _mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=150_000
        )
        print(f"[Final polish] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator's allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Exact evaluation for logging purposes
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
