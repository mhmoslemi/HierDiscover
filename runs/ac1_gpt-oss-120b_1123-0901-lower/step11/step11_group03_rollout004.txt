def _lp_improve(f, n, conv_len, deadline):
    """Perform constraint‑generation LP steps to reduce max‑convolution."""
    f = f.copy()
    best_f = f.copy()
    conv = _full_convolution(f, conv_len)
    best_val = 2.0 * n * conv.max()
    # store best objective
    while time.time() < deadline:
        # recompute conv, M
        conv = _full_convolution(f, conv_len)
        M = conv.max()
        # identify active indices (within 1e-12 of M)
        # we can use tolerance relative.
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        # optionally include a few near maxima
        if len(active) == 0:
            break
        # constraint generation loop
        # start with active set
        active_set = set(active.tolist())
        # Use while to add violations
        while True:
            # build sparse A_ub
            rows = []
            cols = []
            data = []
            for idx, k in enumerate(active_set):
                # k is conv index
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                # values for j from i_start..i_end: f[k-j]
                # length = i_end - i_start + 1
                length = i_end - i_start + 1
                rows.extend([idx] * length)
                cols.extend(range(i_start, i_end + 1))
                # vector of f[k - j]:
                # compute f_slice = f[k - np.arange(i_start,i_end+1)]
                # Could compute via list comprehension or array.
                slice_vals = f[k - np.arange(i_start, i_end + 1)]
                data.extend(slice_vals.tolist())
            if len(rows) == 0:
                break
            A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_set), n))
            b_ub = np.full(A.shape[0], M)
            # solve LP
            c = -np.ones(n)  # maximize sum b -> minimize -sum
            bounds = [(0, None)] * n
            # set high precision options
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                # cannot solve, break
                break
            b_opt = res.x
            # ensure b_opt nonnegative
            if np.any(b_opt < -1e-12):
                b_opt = np.maximum(b_opt, 0.0)
            # compute conv_f_b to see if any violation
            conv_fb = np.convolve(f, b_opt)
            viol = np.where(conv_fb > M + 1e-12)[0]
            if len(viol) == 0:
                # feasible solution
                break
            # add violating indices to active_set
            active_set.update(viol.tolist())
            # loop again
        # after feasible b_opt found
        sum_b = b_opt.sum()
        if sum_b < 1e-12:
            # reinitialize
            b_opt = np.random.dirichlet(np.ones(n))
            sum_b = b_opt.sum()
        b_norm = b_opt / sum_b
        # evaluate new objective
        conv_new = _full_convolution(b_norm, conv_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val:
            best_val = val_new
            best_f = b_norm.copy()
            f = b_norm.copy()
            print(f"[LP] improved to {val_new:.6f}")
        else:
            # no improvement, break loop
            break
        # check time budget: loop may repeat for few cycles
        if time.time() >= deadline - 0.1:
            break
    return best_f, best_val

"""
Hybrid optimiser for the non‑negative step‑function problem (improved variant).

The algorithm consists of three main stages:

1. **Stochastic exponentiated‑gradient (sub‑gradient) optimisation**  
   Works on the exact max‑convolution using a mirror‑descent update.  
   This quickly brings the objective from ~2 down to ~1.51.

2. **Constraint‑generation linear‑programming (LP) refinement**  
   For the current vector `f` we solve a LP that *maximises* the total mass
   of a new vector `b` under the linear constraints  
   `conv(f, b)[k] ≤ M` where `M` is the current maximum convolution value.
   By normalising `b` we obtain a strictly lower bound on the objective.
   Violated constraints are added iteratively (cut‑generation) until a
   feasible `b` is found.  This phase is responsible for the extra
   0.004 improvement needed to beat the previous best.

3. **Short simulated‑annealing polishing** – random mass transfers on the
   simplex for a few ten‑thousand moves.

The whole search runs for at most `budget_s` seconds and always returns the
best sequence discovered.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_convolution(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of two vectors via FFT."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Both arguments have length n, the result has length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set for the current maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size (mirror‑descent on simplex)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _lp_improve(f0: np.ndarray,
                n: int,
                fft_len: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Constraint‑generation LP that tries to increase total mass
    under the bound `conv(f, b) ≤ M`.  After each feasible solve
    the vector is normalised and kept if it improves the objective.
    """
    f = f0.copy()
    best_f = f.copy()
    conv = _full_convolution(f, fft_len)
    best_val = 2.0 * n * conv.max()
    iteration = 0

    while time.time() < deadline:
        conv = _full_convolution(f, fft_len)
        M = conv.max()

        # start with the true maximising indices
        active_set = set(np.where(conv >= M - 1e-12)[0].tolist())
        if not active_set:
            break

        # --- constraint generation loop ---------------------------------
        while True:
            act_list = sorted(active_set)
            rows, cols, data = [], [], []
            for row_idx, k in enumerate(act_list):
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                # slice of f that participates in this convolution entry
                idx_range = np.arange(i_start, i_end + 1, dtype=int)
                vals = f[k - idx_range]      # f[k‑j] for each j
                rows.extend([row_idx] * len(idx_range))
                cols.extend(idx_range.tolist())
                data.extend(vals.tolist())

            if not rows:      # safety
                break

            A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                        shape=(len(act_list), n))
            b_ub = np.full(A.shape[0], M, dtype=np.float64)

            # maximise sum(b)  → minimise -sum(b)
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0, None)] * n
            res = scipy.optimize.linprog(c,
                                         A_ub=A,
                                         b_ub=b_ub,
                                         bounds=bounds,
                                         method='highs',
                                         options={'presolve': True})
            if not res.success:
                # fall back to the current vector
                break

            b_opt = res.x
            b_opt = np.maximum(b_opt, 0.0)    # numerical safety

            # check for newly violated constraints
            conv_fb = _cross_convolution(f, b_opt, fft_len)
            viol = np.where(conv_fb > M + 1e-12)[0]
            if len(viol) == 0:
                # feasible solution found
                break
            active_set.update(viol.tolist())

        # normalise the feasible solution
        sum_b = b_opt.sum()
        if sum_b < 1e-12:
            b_opt = np.random.dirichlet(np.ones(n))
            sum_b = b_opt.sum()
        b_norm = b_opt / sum_b

        # evaluate the new point
        conv_new = _full_convolution(b_norm, fft_len)
        val_new = 2.0 * n * conv_new.max()

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_f = b_norm.copy()
            f = b_norm.copy()
            iteration += 1
            print(f"[LP it={iteration:03d}] improved to {val_new:.6f}")
        else:
            # no improvement – stop the outer LP loop
            break

        # tiny safety – abort if out of time
        if time.time() >= deadline - 0.1:
            break

    return best_f, best_val


# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float,
                     max_moves: int = 80000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.
    Moves a random fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point for the optimiser.
    Returns a list of non‑negative heights (length n) that minimises the
    evaluation function within the allotted budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Initialise a start vector (use warm start if present)
    if warm is None:
        warm = np.random.dirichlet(np.ones(n))
    best_seq = warm.copy()
    best_val = 2.0 * n * _full_convolution(best_seq, conv_fft_len).max()
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – long sub‑gradient optimisation (covers ~70 % of budget)
    sub_dead = deadline - 10.0               # keep ~10 s for the LP phase
    if sub_dead > time.time():
        print("[Phase I] Sub‑gradient optimisation")
        best_seq, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, eta0=0.8
        )
        if val_sub < best_val:
            best_val = val_sub
            print(f"[Phase I] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – LP refinement (active‑set generation)
    lp_dead = deadline - 2.0
    if lp_dead > time.time():
        print("[Phase II] LP refinement")
        best_seq, val_lp = _lp_improve(best_seq, n, conv_fft_len, lp_dead)
        if val_lp < best_val:
            best_val = val_lp
            print(f"[Phase II] improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase III – short simulated annealing polish (remaining time)
    if time.time() < deadline:
        print("[Phase III] Simulated annealing")
        best_seq, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=60000
        )
        if val_ann < best_val:
            best_val = val_ann
            print(f"[Phase III] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure validity for the external validator
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
