"""
Hybrid optimiser with a refined mass‑transfer line‑search.

Key ideas
---------
1. Build a diverse pool of 4096‑point candidate step‑functions
   (warm‑start, Dirichlet draws, exponential / Gaussian / Beta / “U”‑shape,
   uniform, triangular, sparse spikes).

2. For each start run a fast Adam optimiser on a smooth‑max surrogate of the
   self‑convolution peak, followed by an exponentiated‑gradient sub‑gradient
   phase.  Only a handful of starts are fully optimised – the rest are cheap.

3. After a best vector is found, run a *projected exact‑max gradient descent*
   to polish it.

4. **Refined mass‑transfer optimiser** – instead of using a few hand‑picked
   fractions, we compute the exact effect of moving ε mass from the coordinate
   with the largest sub‑gradient to the one with the smallest, and perform a
   short ternary search (8 iterations) to pick the ε that yields the lowest
   exact objective.  The convolution update is done analytically (O(n) per
   evaluation) so many such moves fit easily in the budget.

5. A short simulated‑annealing pass for additional stochastic escape.

6. If time permits, interpolate the best 4096‑vector to a finer 8192 grid
   and repeat steps 4–5 for a final squeeze.

The whole routine respects a 1000 s budget, prints progress, and finally
returns the best non‑negative sequence (list of floats) it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric (or asymmetric) Beta pdf on a uniform grid of mid‑points."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    """Simple cosine‑bell shape."""
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


def _make_Ushape(n, expo):
    """
    Symmetric U‑shaped distribution: a[i] ∝ ((i+1)*(n-i))**expo,
    where expo < 0 concentrates mass at the ends.
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** expo
    a = np.where(a > 0, a, 0.0)
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=300_000):
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0, n, conv_fft_len, corr_fft_len, deadline,
                                 max_iters=500_000, eta0=0.7):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val


def _max_conv_descent(a0, n, conv_fft_len, corr_fft_len, deadline,
                     max_iters=200_000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_a = a.copy()
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# Refined mass‑transfer optimiser (with line‑search)

def _mass_transfer_optimize2(a0, n, conv_fft_len, corr_fft_len, deadline,
                            max_iters=200_000):
    """
    Move mass from the coordinate with the largest sub‑gradient to the one
    with the smallest, choosing the step size ε via a short ternary search.
    Convolution updates are performed analytically in O(n) time.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    conv_len = 2 * n - 1

    it = 0
    while time.time() < deadline and it < max_iters:
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # If gradients are (almost) equal we are at a stationary point
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        src_mass = a[i_high]
        if src_mass < 1e-12:
            # pick a random non‑zero source if the extreme is empty
            nz = np.where(a > 1e-12)[0]
            if nz.size == 0:
                break
            i_high = int(np.random.choice(nz))
            src_mass = a[i_high]

        # Pre‑compute the linear and quadratic effect of moving ε mass
        shift_low = np.zeros(conv_len)
        shift_high = np.zeros(conv_len)
        shift_low[i_low:i_low + n] = a
        shift_high[i_high:i_high + n] = a

        lin0 = 2.0 * (shift_low - shift_high)               # linear term * ε
        quad0 = np.zeros(conv_len)                          # quadratic term * ε²
        quad0[2 * i_low] += 1.0
        quad0[2 * i_high] += 1.0
        quad0[i_low + i_high] -= 2.0

        # ternary search on ε ∈ [0, src_mass]
        lo_eps, hi_eps = 0.0, src_mass
        best_eps = 0.0
        best_obj = best_val

        for _ in range(8):
            eps1 = lo_eps + (hi_eps - lo_eps) * 0.33
            eps2 = lo_eps + (hi_eps - lo_eps) * 0.66

            conv1 = conv + eps1 * lin0 + (eps1 * eps1) * quad0
            val1 = 2.0 * n * conv1.max()
            conv2 = conv + eps2 * lin0 + (eps2 * eps2) * quad0
            val2 = 2.0 * n * conv2.max()

            if val1 < best_obj:
                best_obj = val1
                best_eps = eps1
            if val2 < best_obj:
                best_obj = val2
                best_eps = eps2

            if val1 < val2:
                hi_eps = eps2
            else:
                lo_eps = eps1

        # also test the endpoint (full transfer)
        conv_hi = conv + hi_eps * lin0 + (hi_eps * hi_eps) * quad0
        val_hi = 2.0 * n * conv_hi.max()
        if val_hi < best_obj:
            best_obj = val_hi
            best_eps = hi_eps

        # Apply if we improved
        if best_obj < best_val - 1e-12:
            eps = best_eps
            a[i_high] -= eps
            a[i_low] += eps
            conv = conv + eps * lin0 + (eps * eps) * quad0
            best_val = best_obj
            best_a = a.copy()
            print(f"[MassTransfer2 it={it:06d}] new best {best_val:.6f}")
        else:
            # occasional random perturbation to escape plateaus
            if it % 500 == 0:
                i, j = np.random.randint(n, size=2)
                if a[i] > 1e-12:
                    eps = a[i] * 0.05
                    a[i] -= eps
                    a[j] += eps
                    conv = _full_convolution(a, conv_fft_len)  # recompute
                    cur_val = 2.0 * n * conv.max()
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        print(f"[MassTransfer2 random it={it}] new best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Simulated annealing (unchanged)

def _anneal_optimize(a0, n, conv_fft_len, deadline, max_moves=120_000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ 1000 s.  Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a pool of diverse initial vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential families
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0, 4.0):
        pool.append(_make_exponential(n, gamma))

    # Gaussian families
    for sigma in (0.04, 0.1, 0.2, 0.4, 0.8):
        pool.append(_make_gaussian(n, sigma))

    # Power‑law (decreasing tails)
    for expo in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw(n, expo))

    # Symmetric “U”‑shapes with strong endpoints
    for expo in (-0.3, -0.5, -0.8, -1.0):
        pool.append(_make_Ushape(n, expo))

    # Simple shapes
    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))
    pool.append(_make_cosine(n))

    # Sparse spike patterns
    for k in (32, 64, 128, 256):
        pool.append(_make_spikes(n, k))

    # Beta distributions (including arcsine)
    for alpha, beta in [(0.5, 0.5), (0.8, 0.8), (1.0, 1.0),
                        (2.0, 2.0), (3.0, 3.0), (5.0, 5.0), (10.0, 10.0)]:
        pool.append(_make_beta(n, alpha, beta))

    # Random Dirichlet draws with varied concentration
    for _ in range(6):
        alpha = 10 ** np.random.uniform(-3, 1)  # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    max_candidates = 12   # number of starts that get the full deep optimisation

    # ------------------------------------------------------------------
    # Phase I – deep optimisation of a few promising starts
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(deadline, time.time() + 27.0)

        # 1) Adam
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # 2) Exponentiated‑gradient sub‑gradient (if time permits)
        if time.time() < deadline - 8:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=500_000, eta0=0.7,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] ExpGrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – projected gradient descent on the exact max‑conv
    if best_seq is not None and time.time() < deadline - 12:
        print("[Fine] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase III – refined mass‑transfer optimisation (line‑search)
    if best_seq is not None and time.time() < deadline - 6:
        print("[MassTransfer2] Refined local search")
        a_mt, val_mt = _mass_transfer_optimize2(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[MassTransfer2] improved best to {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simulated annealing
    if best_seq is not None and time.time() < deadline - 5:
        print("[Anneal] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase V – optional up‑sampling to a finer grid (8192 points) if time permits
    if best_seq is not None and time.time() + 30 < deadline:
        try:
            print("[Upsample] Refining on a finer grid (8192 points)")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # Linear interpolation to the larger grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            # A short Adam run on the finer grid
            sub_dead_big = min(deadline, time.time() + 12.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=150_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] Adam improved best to {val_big:.6f}")

            # Finish with a refined mass‑transfer on the fine grid (fewer iters)
            if time.time() < deadline - 5:
                a_big, val_big = _mass_transfer_optimize2(
                    best_seq, n_big, conv_fft_len_big, corr_fft_len_big,
                    deadline, max_iters=80_000,
                )
                if val_big < best_val:
                    best_val = val_big
                    best_seq = a_big.copy()
                    print(f"[Upsample] Mass‑transfer improved best to {val_big:.6f}")

        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Enforce evaluator's range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (optional, for logging)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
