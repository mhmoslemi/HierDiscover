"""
Propose a high‑quality step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

using a fast analytic “mass‑transfer” local search.
The routine
1. builds a diverse pool of simple analytic shapes;
2. keeps the best one (by exact evaluation of the objective);
3. repeatedly moves a small amount of mass from the coordinate with the
   largest gradient of the active convolution peak to the one with the
   smallest gradient.  For each pair the maximal admissible step size is
   computed analytically, so every accepted move reduces the exact
   max‑convolution.
4. When time permits the result is interpolated to a finer grid (8192
   points) and the same optimiser is applied again.
The whole procedure respects the 1000 s budget and returns a plain
Python list ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions -----------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient of the
    max‑convolution. Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories ---------------------------------------------------

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential_left(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) (purely decreasing, not symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * i)
    a /= a.sum()
    return a


def _make_exponential_symmetric(n: int, gamma: float) -> np.ndarray:
    """symmetric sum of two exponentials (the shape used before)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_spikes(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Analytic mass‑transfer optimiser ---------------------------------------

def _mass_transfer_optimize_analytic(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    top_k: int = 10,
) -> tuple[np.ndarray, float]:
    """
    Analytic mass‑transfer local search.
    Repeatedly selects a source index i_high (largest gradient) and a sink
    index i_low (smallest gradient) and moves the maximal admissible amount
    of mass from i_high to i_low.  The maximal step is obtained analytically
    from the current convolution values so that the new max‑convolution never
    exceeds the old one.  Each accepted move strictly reduces the exact
    objective.  Returns the best vector found and its exact score.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        # gradient of the max‑conv w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # candidate source / sink indices
        high_inds = np.argsort(-grad)[:top_k]
        low_inds = np.argsort(grad)[:top_k]

        improved = False

        for i_high in high_inds:
            src_mass = a[i_high]
            if src_mass < 1e-15:
                continue
            for i_low in low_inds:
                if i_low == i_high:
                    continue

                # d = shift(i_low) - shift(i_high)   (length 2n‑1)
                d = np.zeros(2 * n - 1, dtype=np.float64)
                d[i_low:i_low + n] = a
                d[i_high:i_high + n] -= a

                # maximal admissible epsilon (so that no other entry surpasses max_c)
                pos = d > 1e-16
                if np.any(pos):
                    eps_pos = ((max_c - conv[pos]) / (2.0 * d[pos])).min()
                else:
                    eps_pos = src_mass  # nothing increases the max

                eps = min(src_mass, eps_pos)
                if eps <= 1e-15:
                    continue

                # exact new max after the move (no FFT needed)
                new_max = np.max(conv + 2.0 * eps * d)
                if new_max < max_c - 1e-12:
                    # accept the move
                    a[i_high] -= eps
                    a[i_low] += eps
                    best_val = 2.0 * n * new_max
                    best_a = a.copy()
                    improved = True
                    print(f"[AnalyticMT it={it:06d}] new best {best_val:.6f}")
                    break
            if improved:
                break

        if not improved:
            # simple random perturbation to escape possible plateau
            i = np.random.randint(n)
            if a[i] > 1e-12:
                j = np.random.randint(n)
                if i != j:
                    eps = a[i] * 0.05
                    a[i] -= eps
                    a[j] += eps
        it += 1

    return best_a, best_val

# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """

    """
    np.random.seed(seed)
    start_time = time.time()

            warm = warm[:n]
        elif warm.shape[0] < n:

            warm = None
