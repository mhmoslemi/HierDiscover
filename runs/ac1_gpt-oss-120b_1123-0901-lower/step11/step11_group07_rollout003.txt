"""
Hybrid optimiser with deeper refinement.

The optimiser works on the probability simplex (Σ a_i = 1) – the
evaluation is scale‐invariant, so this does not affect the score.
We start from a warm solution (if any), or from a diverse set of
parametric families (uniform, beta, Gaussian, exponential, etc.).
The best seed is then refined in three stages:

1.  Adam on a smooth‑max surrogate of the peak of the autocorrelation.
2.  Exact‑max exponentiated‑gradient sub‑gradient descent (Polyak step).
3.  A fast simulated‑annealing style local search that moves tiny
    amounts of mass between coordinates.

All stages respect the global 1000 s budget and always keep the
sequence non‑negative.  The final sequence is returned as a plain
list of Python floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_half_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_laplacian(n, gamma):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-gamma * np.abs(i - center))
    a /= a.sum()
    return a


def _make_mixture_gaussian(n, mu1, sigma1, mu2, sigma2, w1):
    """Two‑Gaussian mixture (weights w1 and 1‑w1)."""
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
    g2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    a = w1 * g1 + (1.0 - w1) * g2
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

# Optimisers

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   lr0: float = 0.07,
                   lam0: float = 0.5,
                   lam_factor: float = 1.04,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam0
    best_a = a0.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max re‑parameterisation
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {cur_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # modest learning‑rate decay
        if it % 200_000 == 0 and it > 0:
            lr0 *= 0.8

        it += 1

    return best_a, best_val


def _expgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      eta0: float = 0.9,
                      max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on exact max‑conv."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {cur_val:.6f}")

        # active set for the exact max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_search(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  deadline: float,
                  move_budget: int = 200_000) -> tuple[np.ndarray, float]:
    """Fast simulated‑annealing / hill‑climb on the exact max‑conv."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    # linear cooling schedule
    T_start = 0.08
    T_end = 1e-6

    moves = 0
    while time.time() < deadline and moves < move_budget:
        T = max(T_end, T_start * (1.0 - moves / move_budget))

        # pick two distinct indices, ensure source has enough mass
        i = np.random.randint(n)
        if a[i] < 1e-14:
            moves += 1
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 30 %) of the source mass to the target
        frac = np.random.rand() * 0.3
        delta = a[i] * frac
        if delta <= 0.0:
            moves += 1
            continue

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        cur_val = 2.0 * n * conv2.max()

        if cur_val < best_val:
            best_val = cur_val
            best_a = a2.copy()
            a = a2
        else:
            # occasional uphill moves (simulated annealing)
            if np.random.rand() < np.exp(-(cur_val - best_val) / T):
                a = a2

        moves += 1

    return best_a, best_val

# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ budget_s seconds (default 1000 s).  Returns the best
    sequence (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.9  # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 points give a good speed/quality trade‑off.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (if a previous run produced a good sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Initialise a diverse pool of candidates – cheap evaluation only.
    pool = []

    # start with warm if available
    if warm is not None:
        pool.append(warm.copy())

    # basic families
    pool.append(_make_uniform(n))
    pool.append(_make_half_sine(n))
    pool.append(_make_cosine(n))
    pool.append(_make_laplacian(n, 0.02))

    # beta family sweep (log‑uniform grid)
    alphas = np.logspace(-1.5, 0.5, 12)   # ~0.03 … 3.2
    betas  = np.logspace(-1.5, 0.5, 12)
    for a in alphas:
        for b in betas:
            pool.append(_make_beta(n, a, b))

    # gaussian / exponential sweeps
    for sigma in (0.04, 0.07, 0.1, 0.15, 0.25):
        pool.append(_make_gaussian(n, sigma))
    for gamma in (0.07, 0.12, 0.2, 0.35, 0.6):
        pool.append(_make_exponential(n, gamma))

    # random Dirichlet draws
    for _ in range(30):
        alpha = 10 ** np.random.uniform(-3, 2)   # 1e‑3 … 1e2
        pool.append(_make_dirichlet(n, alpha))

    # random two‑Gaussian mixtures
    for _ in range(20):
        mu1 = np.random.uniform(0, n - 1)
        mu2 = np.random.uniform(0, n - 1)
        sigma1 = np.random.uniform(0.5, 0.25 * n)
        sigma2 = np.random.uniform(0.5, 0.25 * n)
        w1 = np.random.rand()
        pool.append(_make_mixture_gaussian(n, mu1, sigma1, mu2, sigma2, w1))

    # --------------------------------------------------------------
    # Pick the initial best from the cheap pool
    best_seq = None
    best_val = float('inf')
    for idx, a in enumerate(pool):
        # ensure feasibility
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        # inexpensive evaluation (no extra clamping)
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Init {idx}] new best {cur_val:.6f}")

    # --------------------------------------------------------------
    # Phase I – Adam (smooth‑max) on the current best
    if best_seq is not None and time.time() < deadline - 200.0:
        print("[Phase I] Adam smooth‑max")
        sub_dead = deadline - 180.0   # reserve ~3 min for later phases
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            lr0=0.07, lam0=0.5, lam_factor=1.04,
            max_iters=4_000_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase II – Exact‑max exponentiated‑gradient descent
    if best_seq is not None and time.time() < deadline - 120.0:
        print("[Phase II] Exponentiated‑gradient")
        sub_dead = deadline - 100.0
        a_exp, val_exp = _expgrad_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            eta0=0.9, max_iters=6_000_000
        )
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[Phase II] improved to {val_exp:.6f}")

    # --------------------------------------------------------------
    # Phase III – Fast local search (simulated annealing)
    if best_seq is not None and time.time() < deadline - 30.0:
        print("[Phase III] Local perturbation")
        sub_dead = deadline - 5.0
        a_loc, val_loc = _local_search(
            best_seq, n, conv_fft_len, sub_dead,
            move_budget=250_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase III] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Final sanity – ensure non‑empty return and clip to allowed range
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value (approx) {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
