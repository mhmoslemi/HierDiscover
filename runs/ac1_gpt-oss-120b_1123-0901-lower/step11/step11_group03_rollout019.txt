def projected_subgradient_optimize(a0, n, deadline):
    a = a0 / a.sum()
    best_a = a; best_val=evaluate(a)
    it=0
    while time.time()<deadline:
        conv = full_conv(a)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        # compute gradient = 2 * crosscorr(w,a)
        w = np.zeros_like(conv)
        w[active] = 1.0/active.sum()
        grad = 2.0 * cross_correlation(w, a, corr_len)
        # Step
        eta = 0.1 / np.sqrt(it+1) # may tune
        a_new = a - eta * grad
        # enforce non-neg and simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a_new = a_new / s

        # Evaluate and accept if better; else keep old (or probabilistic)
        val = evaluate(a_new)
        if val < best_val:
            best_val=val; best_a=a_new.copy()
        # optional acceptance based on improvement property: always move to new a (subgradient) even if worse? Typically subgradient descent doesn't use acceptance.
        a = a_new
        it+=1

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose length n based on remaining time: if budget > 600 seconds use n=8192 else 4096.
    n = 8192 if budget_s > 400 else 4096

    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    # generate pool of candidates
    pool = []
    # uniform
    pool.append(np.full(n, 1.0/n))
    # Dirichlet draws
    for alpha in (0.05,0.2,0.5,1.0,2.0,5.0):
        pool.append(np.random.dirichlet(alpha*np.ones(n)))
    # exponential decays
    for gamma in (0.001, 0.005,0.01,0.02,0.05):
        shape = np.exp(-gamma * np.arange(n))
        shape /= shape.sum()
        pool.append(shape)
    # random-phase samples
    for _ in range(20):
        # generate random phases, constant magnitude
        phases = np.random.uniform(0, 2*np.pi, size=conv_fft_len//2+1)
        # we will construct via irfft
        A = np.empty(conv_fft_len//2+1, dtype=np.complex128)
        A[0] = 1.0  # DC component sum=1
        # set random phases with equal magnitude 1
        A[1:] = np.exp(1j * phases[1:])
        # length n for irfft
        a = np.fft.irfft(A, n)
        a = np.maximum(a,0.0)
        s = a.sum()
        if s>0:
            a/=s
        else:
            a = np.full(n,1.0/n)
        pool.append(a)

    best_seq = None
    best_val = float('inf')
    for a in pool:
        val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() > deadline:
            break

    # If time left, apply Adam optimization on best_seq
    if time.time() < deadline - 5:
        a_opt, val_opt = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len,
                                         deadline - 5, max_iters=500_000)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()

    # Continue with subgradient/exponential gradient longer
    if time.time() < deadline - 5:
        a_opt2, val_opt2 = _subgradient_optimize(best_seq, n, conv_fft_len, corr_fft_len,
                                                 deadline - 5, eta0=0.5)
        if val_opt2 < best_val:
            best_val = val_opt2
            best_seq = a_opt2.copy()
    # final clip
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best objective {best_val:.6f}")
    return best_seq.tolist()

"""
Hybrid search for a low‑value non‑negative step function.

The evaluator computes
    2·n·max(conv(a,a)) / (sum(a)**2)

with n = len(a) and a_i ≥ 0.  Because the objective is scale‑invariant we keep
the simplex constraint Σ a_i = 1 throughout the search.

Main ideas
----------
1.   A diverse *seed* pool:
     • uniform vector,
     • Dirichlet draws (several concentration parameters),
     • simple exponential decays,
     • a few random‑phase Fourier constructions (constant spectrum, random phase).
2.   Pick the best seed (exact objective, i.e. 2·n·max(conv)).
3.   Long *mirror‑descent* (exponentiated‑gradient) on the exact max‑convolution.
     The gradient of the active (max) constraints is computed via FFT‑based
     cross‑correlation.  The step size η decays as 1/√t.
4.   Optionally a short Adam pass on a smooth‑max surrogate (temperature λ
     slowly increased) to clean up any remaining irregularities.
5.   All phases respect the global time budget (default 1000 s) and the best
     vector found is always returned.

The implementation works for any length; we use a larger length (8192) when the
time budget permits, because a larger search space allows a lower theoretical
minimum.  The code is completely self‑contained – only NumPy is required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.05
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e8)

        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    # final exact evaluation
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val

def _random_phase_candidate(n: int, conv_fft_len: int) -> np.ndarray:
    """Generate a candidate by assigning random phases to a constant spectrum."""
    # rfft length for a real vector of length n
    rfft_len = n // 2 + 1
    phases = np.random.uniform(0.0, 2.0 * np.pi, size=rfft_len)
    A = np.empty(rfft_len, dtype=np.complex128)
    A[0] = 1.0                     # DC term = sum = 1
    # keep Nyquist term (if present) real to stay on the real sub‑space
    if n % 2 == 0:
        # last element is its own conjugate, keep it real
        A[-1] = np.cos(phases[-1]) + 0j
        phases[1:-1] = phases[1:-1]
        A[1:-1] = np.exp(1j * phases[1:-1])
    else:
        A[1:] = np.exp(1j * phases[1:])

    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _make_initial_pool(n: int) -> list[np.ndarray]:
    """Create a small pool of diverse start vectors."""
    pool = []

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Simple exponentials (both directions)
    for gamma in (0.001, 0.005, 0.01, 0.02, 0.05):
        shape = np.exp(-gamma * np.arange(n))
        shape /= shape.sum()
        pool.append(shape)

    # Random‑phase Fourier candidates
    for _ in range(30):
        pool.append(_random_phase_candidate(n, None))

    return pool
