def _solve_lp_max_sum(f, M):
    n = len(f)
    # build A_ub: shape 2n-1 x n
    data = []
    rows = []
    cols = []
    for i in range(n):
        # rows indices: i .. i + n - 1
        rows_i = np.arange(i, i+n)
        cols_i = np.full(n, i)
        data_i = np.full(n, f[i])
        rows.append(rows_i)
        cols.append(cols_i)
        data.append(data_i)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ g <= M
    b = np.full(2*n-1, M)
    # objective: minimize -sum(g) => maximize sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

"""
Hybrid optimizer using linear‑programming based mass‑expansion.
[0] Down‑sample warm start to a modest size.
[1] Repeatedly solve a small LP that maximizes the ℓ₁‑norm of a
    candidate while keeping the max‑convolution ≤ the current bound.
    The candidate is rescaled to the target ℓ₁‑norm and mixed into the
    current iterate.
[2] Upsample the resulting vector to the full size and finish with the
    previously‑successful spectral‑flattening / sub‑gradient phases.
The routine respects the given budget and always returns the best
sequence discovered.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto the probability simplex.
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val

def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")
        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)
        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s
        it += 1
    return best_a, best_val

def _build_conv_ub_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Construct A (2n‑1 × n) with A[k,i] = f[i] if i ≤ k ≤ i+n‑1 else 0."""
    n = f.shape[0]
    diags = [f[: n - shift] for shift in range(n)]
    offsets = list(range(n))
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return A

    a_full = np.clip(a_full, 0.0, 1000.0)

    return a_full.tolist()

Hybrid optimiser for the max‑convolution objective.

The algorithm works in three stages:

1. **Small‑scale LP refinement** – the current sequence f (down‑sampled to
   a modest size, e.g. 2048) is kept fixed while we solve a linear program
   that maximises the ℓ₁‑norm of a new candidate g under the same
   max‑convolution bound M = max(f*f).  The resulting g is rescaled to the
   target ℓ₁‑norm and mixed into f.  Repeating this step shrinks M
   monotonically.

2. **Upscaling & final polishing** – the improved small‑scale vector is
   interpolated to the full resolution (4096), normalised, and then
   refined with the spectral‑flattening routine that was already very
   effective in the reference implementation.  A short projected‑gradient
   descent is performed at the end if time permits.

The routine respects the given time budget and always returns the best
sequence discovered before the deadline.

    return np.maximum(v - theta, 0.0)


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """Iteratively flatten the FFT magnitude (except the DC term)."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {val:.6f}")

        A = np.fft.fft(a)
        energy = np.dot(a, a)
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s
        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")
        it += 1
    return best_a, best_val


def _build_conv_ub_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the matrix A (shape (2n‑1, n)) such that  A @ g  =  (f ★ g)
    (linear convolution of f with g).  Entry A[k,i] = f[i] iff i ≤ k ≤ i+n‑1,
    otherwise 0.
    """
    n = f.shape[0]
    diags = [f[: n - shift] for shift in range(n)]
    offsets = list(range(n))
    return sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣ LP‑based mass‑expansion on a modest problem size
    n_small = 2048
    target_sum_small = np.sqrt(2.0 * n_small)
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # warm start (optional) – down‑sample the previously best long vector
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64).ravel()
        x_src = np.linspace(0, warm_arr.size - 1, warm_arr.size)
        x_dst = np.linspace(0, warm_arr.size - 1, n_small)
        f = np.interp(x_dst, x_src, warm_arr)
    else:
        f = np.full(n_small, 1.0 / n_small, dtype=np.float64)

    f = np.maximum(f, 0.0)
    if f.sum() < 0.01:
        f = np.random.dirichlet(np.ones(n_small))
    f = f * (target_sum_small / f.sum())

    best_f = f.copy()
    best_val = float("inf")
    mix_factor = 0.25                # how strongly we blend the new LP solution
    eps = 1e-8
    lp_iter = 0

    # iterate the LP refinement while we still have ample budget left
    while time.time() < deadline - 30.0 and lp_iter < 20:
        # current bound
        conv = np.convolve(f, f)
        M = conv.max()
        val = 2.0 * n_small * M / (f.sum() ** 2)   # = M after normalisation
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LPIter {lp_iter:02d}] current best {val:.6f}")

        # Build constraints A @ g ≤ M  (with a tiny epsilon to keep LP bounded)
        f_eps = np.maximum(f, eps)
        f_eps = f_eps * (target_sum_small / f_eps.sum())
        A_ub = _build_conv_ub_matrix(f_eps)
        b_ub = np.full(2 * n_small - 1, M, dtype=np.float64)
        c = -np.ones(n_small)                       # maximise Σ g

        # solve a modest linear program
        res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n_small,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print("[LP] linprog failed – terminating LP phase")
            break

        g = res.x
        if g.sum() < 1e-12 or np.isnan(g).any():
            g = np.random.dirichlet(np.ones(n_small))
        g = g * (target_sum_small / g.sum())         # enforce the ℓ₁‑norm

        # blend the new candidate into the current iterate
        f = (1.0 - mix_factor) * f + mix_factor * g
        f = np.maximum(f, 0.0)
        f = f * (target_sum_small / f.sum())

        lp_iter += 1

    # ------------------------------------------------------------------
    # 2️⃣ Optional sub‑gradient fine‑tuning on the small problem
    if time.time() < deadline - 10.0:
        a_sub, val_sub = _subgradient_optimize_expgrad(
            f,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            deadline,
            max_iters=400_000,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_f = a_sub.copy()
            print(f"[Subgrad small] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Upscale to the full resolution (4096) for the final phases
    n_full = 4096
    target_sum_full = np.sqrt(2.0 * n_full)
    # simple linear interpolation
    x_small = np.linspace(0, n_small - 1, n_small)
    x_full = np.linspace(0, n_small - 1, n_full)
    a_full = np.interp(x_full, x_small, best_f)
    a_full = np.maximum(a_full, 0.0)
    a_full = a_full * (target_sum_full / a_full.sum())

    # ------------------------------------------------------------------
    # 4️⃣ Intense spectral‑flattening on the full‑size vector
    conv_fft_len_full = _next_pow2(2 * n_full - 1)
    if time.time() < deadline - 5.0:
        a_spec, val_spec = _spectral_flatten_optimize(
            a_full,
            n_full,
            conv_fft_len_full,
            deadline,
            max_iters=1_000_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            a_full = a_spec.copy()
            print(f"[Spectral full] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Very short projected‑gradient descent (exact max‑conv) as a last polish
    corr_fft_len_full = _next_pow2(3 * n_full - 2)
    if time.time() < deadline - 2.0:
        a_gd, val_gd = _max_conv_descent(
            a_full,
            n_full,
            conv_fft_len_full,
            corr_fft_len_full,
            deadline,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            a_full = a_gd.copy()
            print(f"[GD final] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Return the best sequence (guaranteed non‑empty)
    if a_full is None:
        a_full = np.full(n_full, 1.0 / n_full, dtype=np.float64)

    a_full = np.clip(a_full, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(a_full)})")
    return a_full.tolist()
