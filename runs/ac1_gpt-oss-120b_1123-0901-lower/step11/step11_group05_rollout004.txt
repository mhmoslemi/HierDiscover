def _subgradient_additive(a0, n, conv_len, corr_len, deadline):
    a = a0.copy()
    a = np.maximum(a, 0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, conv_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 1000 == 0:
                print(...)
        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_len)
        # step size
        eta = eta0 / np.sqrt(it + 1.0)  # maybe adjust

        # projection onto simplex
        a = _project_simplex(a)
        it += 1
    return best_a, best_val

"""
Hybrid random sampling + projected additive sub‑gradient optimizer.

The programme works in three stages:

1️⃣  Randomly draws many Dirichlet‑type candidates (with a wide range of
    concentration parameters) and keeps the best according to the exact
    objective 2·n·max(a*a) (the value returned by ``_objective``).

2️⃣  Starting from that best candidate we run a long *additive* sub‑gradient
    descent on the exact max‑convolution objective.
    At each step we:
        – compute the convolution,
        – form the sub‑gradient w.r.t. the active (max‑tight) indices,
        – take an Euclidean (additive) step,
        – project back onto the probability simplex (guaranteeing non‑negativity
          and Σa=1).
    The step size follows the classic η_t = η₀/√t schedule.
    This optimizer is able to drive many coefficients to zero, something
    multiplicative updates struggle with, and therefore discovers sparser,
    lower‑max solutions.

3️⃣  A short stochastic local‑search (mass transfer between two coordinates)
    followed by trying a few cyclic shifts of the final vector (the linear
    convolution is shift‑variant).  The best sequence found is clipped,
    re‑projected onto the simplex and finally evaluated with the official
    ``evaluate_sequence`` before being returned.

The whole routine respects a 1000 s time budget, prints progress information
and always returns a valid non‑negative list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (sum(a)=1 assumed)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(a*a) for a with Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Additive projected sub‑gradient optimizer
def _additive_subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 0.5,
    atol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses additive (Euclidean) updates and projects onto the simplex after each step.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and current objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = float(2.0 * n * max_c)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[AddSubGrad it={it:07d}] new best {cur_val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=atol, rtol=0)
        if not np.any(active):
            # should not happen, but protect against numerical issues
            active = np.ones_like(conv, dtype=bool)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of the max‑convolution w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # additive update + simplex projection
        a = a - eta * grad
        a = _project_simplex(a)

        # occasional safety‑reset if projection failed (nan)
        if np.isnan(a).any():
            a = np.random.dirichlet(np.ones(n))

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given time budget (≈ 1000 s) and always returns
    a valid list of heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0) generate a pool of diverse initial candidates
    candidates = []

    # warm start, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a_sum
        candidates.append(a)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # beta‑shaped densities (including more aggressive U‑shapes)
    alphas = [0.3, 0.4, 0.5, 0.55, 0.6, 0.7, 0.8, 0.9]
    x = (np.arange(n) + 0.5) / n
    for alpha in alphas:
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # Dirichlet draws with very low concentration (encourages sparsity)
    for _ in range(30):
        conc = 10 ** np.random.uniform(-2.0, 0.5)   # from 0.01 to ~3.16
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ------------------------------------------------------------------
    # 1) pick the best cheap candidate (exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # ------------------------------------------------------------------
    # 2) random sampling phase (quickly try many fresh Dirichlet vectors)
    sample_deadline = deadline - 300.0          # reserve ~5 min for gradient phase + polishing
    while time.time() < sample_deadline:
        conc = 10 ** np.random.uniform(-2.0, 0.5)
        a = np.random.dirichlet(np.full(n, conc))
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[RandomSample] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 3) long additive sub‑gradient optimisation
    sub_deadline = deadline - 20.0               # leave a few seconds for final tweaks
    if sub_deadline > time.time():
        a_opt, val_opt = _additive_subgradient_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=10_000_000,
            eta0=0.7,
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[AddSubGrad] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) tiny stochastic local search (mass transfer)
    local_deadline = deadline - 1.0
    max_delta = 0.01  # maximum chunk that can be moved in one step
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move a small random chunk from i to j
        delta = min(best_a[i], max_delta) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[LocalMove] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 5) try a few cyclic shifts (convolution is shift‑variant)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 6) final sanity checks
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)   # guarantee exact simplex normalisation

    # final official evaluation (mainly for logging)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
