def apply_mass_transfer(a, conv, i_src, i_dst, eps):
    # conv: array length 2n-1
    # a: current vector
    # update conv for movement of eps mass from src to dst.
    # compute conv_new = conv - 2*eps * a shifted by src + 2*eps * a shifted by dst - 2*eps**2 at i_src + i_dst
    conv2 = conv.copy()
    # shift contributions: use slicing
    # For src: conv2[src:src+n] -= 2*eps * a
    conv2[i_src : i_src + len(a)] -= 2.0 * eps * a
    conv2[i_dst : i_dst + len(a)] += 2.0 * eps * a
    # Quadratic term
    idx = i_src + i_dst
    if idx < len(conv2):
        conv2[idx] -= 2.0 * eps * eps
    # Update a vector
    a2 = a.copy()
    a2[i_src] -= eps
    a2[i_dst] += eps
    return a2, conv2

"""
Propose a low‑value step‑function for the given evaluator.

The algorithm
----------------
1. Build a fairly rich library of candidate probability vectors.
   • Warm‑start (if ``height_sequence_1`` exists).
   • Uniform, simple triangle, and many Beta‑distribution shapes
     (both symmetric and asymmetric, with parameters sweeping the
     whole (0.01, 1] range).
   • Linear mixtures of an arc‑sine (Beta(0.5,0.5)) and the uniform
     distribution.

   All candidates are normalised to sum = 1 and their exact objective
   ``2·n·max(conv)`` is evaluated (the evaluator’s scaling cancels).

2. Keep the best candidate and start a *mass‑transfer* local optimiser.
   The objective is ``F(a)=2·n·max(a*a)`` (for a normalised vector).
   For a move that transfers a small amount ε from index *i* to *j*
   the convolution can be updated exactly in O(n) time
   (no FFT per move).  The optimiser repeatedly:
        – computes the active set (indices attaining the current max)
        – evaluates the sub‑gradient
        – moves mass from the highest‑gradient coordinate to the lowest‑gradient
          coordinate, trying a short list of ε‑fractions.
        – if no improvement is possible, performs a small random move.
   Periodically the full convolution is recomputed from scratch and the
   vector is re‑normalised to avoid drift.

3. The loop runs until the time budget (≈ 1000 s) is exhausted.
   The best vector found is returned as a plain Python list.

The whole routine respects the budget, prints progress and the final
exact score (using ``evaluate_sequence``).  No external files or
network access are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple factories for candidate shapes

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta pdf evaluated on a uniform grid."""
    # mid‑points to avoid exact 0/1
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Incremental convolution update for a tiny mass transfer

def _apply_mass_transfer(a: np.ndarray,
                         conv: np.ndarray,
                         src: int,
                         dst: int,
                         eps: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (a_new, conv_new) after moving `eps` mass from `src` to `dst`.
    `conv` is the current full convolution of `a`.
    Both updates are performed in O(n).
    """
    n = a.shape[0]
    # copy objects (the copies are cheap – size < 2·n)
    a_new = a.copy()
    conv_new = conv.copy()

    # linear part:   conv ← conv - 2·eps·shift(src) + 2·eps·shift(dst)
    # shift(src) = a[k‑src] (zero‑padded)
    conv_new[src: src + n] -= 2.0 * eps * a
    conv_new[dst: dst + n] += 2.0 * eps * a

    # quadratic term only at index src+dst
    idx = src + dst
    if idx < conv_new.size:
        conv_new[idx] -= 2.0 * eps * eps

    # modify the probability vector
    a_new[src] -= eps
    a_new[dst] += eps
    return a_new, conv_new


# ----------------------------------------------------------------------
# Main optimisation routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the evaluator.
    Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size – 4096 is a good compromise between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1️⃣  Build a pool of diverse candidate vectors
    pool = []

    # warm start (if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            w = w / w.sum()
            pool.append(w)

    # uniform and simple shapes
    pool.append(_make_uniform(n))

    # symmetric Beta (arc‑sine and neighbours)
    for a in (0.5, 0.4, 0.3, 0.2, 0.1):
        pool.append(_make_beta(n, a, a))

    # asymmetric Beta families – useful for “one‑sided” mass
    asym_params = [(0.2, 0.8), (0.1, 0.9), (0.05, 0.95),
                   (0.8, 0.2), (0.9, 0.1), (0.3, 0.7), (0.7, 0.3)]
    for a, b in asym_params:
        pool.append(_make_beta(n, a, b))

    # random Beta draws (log‑uniform in (0.01, 1])
    for _ in range(250):
        a = 10 ** np.random.uniform(-2, 0)   # between 0.01 and 1
        b = 10 ** np.random.uniform(-2, 0)
        pool.append(_make_beta(n, a, b))

    # mixtures of arc‑sine and uniform (helps fine‑tuning)
    arcsine = _make_beta(n, 0.5, 0.5)
    uniform = _make_uniform(n)
    for w in (0.1, 0.3, 0.5, 0.7, 0.9):
        mix = w * arcsine + (1 - w) * uniform
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        pool.append(mix)

    # ------------------------------------------------------------------
    # 2️⃣  Pick the best raw candidate (exact objective)
    best_seq = None
    best_val = float("inf")

    for idx, a in enumerate(pool):
        if time.time() > deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()                     # normalise to a probability vector
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()          # exact objective (sum = 1 ⇒ denominator = 1)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init] candidate {idx:03d} improved to {val:.6f}")

    # fallback – should never happen
    if best_seq is None:
        best_seq = _make_uniform(n)

    # ------------------------------------------------------------------
    # 3️⃣  Mass‑transfer refinement (direct, cheap updates)
    a = best_seq.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[Refine start] best initial value {best_val:.6f}")

    it = 0
    recompute_every = 5000   # re‑normalise & recompute from scratch periodically

    while time.time() < deadline:
        # --------------------------------------------------------------
        # compute active set and sub‑gradient (exact for the max‑conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --------------------------------------------------------------
        # directed move: from highest to lowest gradient index
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        grad_gap = grad[i_hi] - grad[i_lo]

        moved = False
        if grad_gap > 1e-12:                     # we have a sensible descent direction
            src, dst = i_hi, i_lo
            src_mass = a[src]
            # try a few step‑sizes (larger first)
            for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003, 0.001):
                eps = src_mass * frac
                if eps <= 0:
                    continue
                a_new, conv_new = _apply_mass_transfer(a, conv, src, dst, eps)
                val_new = 2.0 * n * conv_new.max()
                if val_new < best_val - 1e-12:          # strict improvement
                    a, conv = a_new, conv_new
                    best_val = val_new
                    best_seq = a_new.copy()
                    moved = True
                    print(f"[Refine it={it:06d}] improved to {val_new:.6f}")
                    break

        # --------------------------------------------------------------
        # If directed move failed, try a tiny random perturbation
        if not moved:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-14:
                eps = a[i] * 0.02          # 2 % of source mass
                a_new, conv_new = _apply_mass_transfer(a, conv, i, j, eps)
                val_new = 2.0 * n * conv_new.max()
                if val_new < best_val - 1e-12:
                    a, conv = a_new, conv_new
                    best_val = val_new
                    best_seq = a_new.copy()
                    moved = True
                    print(f"[Random it={it:06d}] improved to {val_new:.6f}")

        it += 1

        # --------------------------------------------------------------
        # Periodic full recomputation to avoid drift
        if it % recompute_every == 0:
            a /= a.sum()                     # renormalise (tiny numerical drift)
            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Recompute it={it}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Post‑process and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # The evaluator is scale‑invariant; we keep the sum at 1 for clarity.
    if best_seq.sum() < 0.01:
        best_seq = _make_uniform(n)
    else:
        best_seq = best_seq / best_seq.sum()

    # Final exact check (helps the competition harness to see the score)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
