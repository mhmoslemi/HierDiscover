def _lp_refine(a, n, conv_fft_len, deadline, max_refine_iters=10):
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    # initial M
    while time.time() < deadline and max_refine_iters > 0:
        max_refine_iters -= 1
        conv = _full_convolution(best_a, conv_fft_len)
        M = conv.max()
        # active indices
        eps_active = 1e-12  # absolute tolerance
        active = np.where(conv >= M - eps_active)[0]
        # optionally also include near maximum
        if len(active) < 5:
            # include indices within fraction
            rel_tol = 0.001  # 0.1% of M
            active = np.where(conv >= M * (1 - rel_tol))[0]
        # Build constraint matrix
        K = len(active)
        if K == 0:
            break
        A = np.zeros((K, n))
        for idx_row, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # indices i = i_start..i_end
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_indices = k - i_range
            A[idx_row, j_indices] = best_a[i_range]
        # solve LP: maximize sum(g)
        c = -np.ones(n)  # minimize -sum(g)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # fallback: break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # try mixing with several t values
        improved = False
        for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
            cand = (1 - t) * best_a + t * g_norm
            cand = _project_simplex(cand)
            val = _objective(cand, n, conv_fft_len)
            if val < best_val - 1e-9:
                best_val, best_a = val, cand.copy()
                improved = True
                # print
                print(f"[LP refine t={t}] improved to {val:.6f}")
                break
        if not improved:
            break
    return best_a, best_val

"""
Hybrid optimizer for the max‑convolution bound.

Strategy
--------
1.  Build a diverse pool of initial probability vectors (uniform,
    beta‑shapes, power‑law, triangular and a few random Dirichlet draws,
    plus the optional warm‑start `height_sequence_1`).
2.  Pick the best one w.r.t. the exact objective
    `2·n·max(conv)`.
3.  Run a long exponentiated‑gradient sub‑gradient phase (as in the
    reference code) – this quickly drives the maximum of the convolution
    down.
4.  Refine with a few hundred Adam iterations on a smooth‑max surrogate
    (log‑sum‑exp) – this sharpens the improvement.
5.  **LP‑tightening step** – for the current vector `f`
    * compute the set of (near‑)active convolution indices,
    * solve a small linear program
          max   Σ g_i
          s.t.  (f ★ g)_k ≤ M   for all active k,
                g_i ≥ 0,
      where `M` is the present maximum.
      The LP is solved with `scipy.optimize.linprog`; only a handful of
      constraints are present, so it is cheap.
    * Mix the solution `g` with `f` (line‑search over a few mixing
      coefficients) and keep the best mixture.
    * Repeat a few times while time permits.
6.  Final stochastic local search (tiny mass transfers) and a few cyclic
    shifts – these cheap moves often escape shallow minima.
7.  Return the best sequence found (projected onto the probability simplex
    and clipped to the allowed range).

The whole routine respects the provided time budget (≈ 1000 s) and
always produces a valid non‑negative list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection -----------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective --------------------------------------------------------

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact evaluation 2·n·max(a*a) for a with sum(a)=1.
    (The official `evaluate_sequence` uses the same scaling.)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Sub‑gradient (exponentiated‑gradient) ----------------------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using EG updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            # sporadic logging
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # build the weight vector that is 1 on the active set
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # EG step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam on smooth‑max -----------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP tightening (active‑set) --------------------------------------------

def _lp_refine(a: np.ndarray, n: int, conv_fft_len: int,
               deadline: float, max_iters: int = 10) -> tuple[np.ndarray, float]:
    """
    Small linear‑programming refinement.
    For the current vector a (sum≈1) we solve
        max Σ g_i
        s.t. (a ★ g)_k ≤ M   for k in an active set,
             g_i ≥ 0
    where M = max(a*a).  The optimal g is mixed back into a with a
    line‑search over a few mixing coefficients.
    """
    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(best_a, conv_fft_len)
        M = conv.max()

        # --- active set -------------------------------------------------
        # indices where the convolution is within a tiny tolerance of the max
        atol = 1e-12
        active = np.where(conv >= M - atol)[0]

        # If active set is too small we broaden it a bit
        if len(active) < 3:
            rel_tol = 0.001  # 0.1 % of the peak
            active = np.where(conv >= M * (1 - rel_tol))[0]

        K = len(active)
        if K == 0:
            break

        # Build the (K × n) constraint matrix A such that A @ g = (a★g)_k
        A = np.zeros((K, n))
        for row, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_range
            A[row, j_idx] = best_a[i_range]

        # --- linear program ---------------------------------------------
        c = -np.ones(n)                     # maximise Σ g_i  → minimise -Σ g_i
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            # LP failed – give up on further refinements
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g

        # --- line‑search mixing -----------------------------------------
        improved = False
        for t in (0.1, 0.3, 0.5, 0.7, 0.9):
            cand = (1.0 - t) * best_a + t * g_norm
            cand = _project_simplex(cand)
            val = _objective(cand, n, conv_fft_len)
            if val < best_val - 1e-9:
                best_val, best_a = val, cand.copy()
                improved = True
                print(f"[LP refine it={it} t={t}] improved to {val:.6f}")
                break
        if not improved:
            # no improvement → stop the LP loop
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # keep a small safety margin
    deadline = time.time() + budget_s - 0.3

    n = 4096                                   # length of the vector
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0)  Build a diverse set of start vectors
    candidates = []

    # warm start if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric beta families (α in [0.4,1.6])
    alphas = np.concatenate(
        (np.linspace(0.4, 0.99, 6), np.linspace(1.01, 1.6, 6))
    )
    x = (np.arange(n) + 0.5) / n
    for alpha in alphas:
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # triangular (pyramid) shape – symmetric peak in the middle
    tri = np.minimum(x, 1.0 - x) * 2.0
    tri = np.maximum(tri, 0.0)
    tri = tri / tri.sum()
    candidates.append(tri)

    # power‑law shapes (exponent p ∈ [0.5,1.5])
    for p in [0.5, 0.8, 1.2, 1.5]:
        a = np.power(x, -p) + np.power(1.0 - x, -p)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        candidates.append(a)

    # a batch of Dirichlet draws with varied concentration
    for _ in range(20):
        conc = np.random.uniform(0.2, 3.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ------------------------------------------------------------------
    # 1)  Pick the best cheap candidate (exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # ------------------------------------------------------------------
    # 2)  Heavy sub‑gradient optimisation (exponentiated gradient)
    sub_deadline = deadline - 8.0
    if sub_deadline > time.time():
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=5_000_000, eta0=1.5
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Subgrad] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3)  Adam fine‑tuning on a smooth‑max surrogate
    adam_deadline = deadline - 5.0
    if adam_deadline > time.time():
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=300_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4)  Linear‑programming tightening (active‑set)
    lp_deadline = deadline - 2.0
    if lp_deadline > time.time():
        a_lp, val_lp = _lp_refine(
            best_a, n, conv_fft_len, lp_deadline, max_iters=12
        )
        if val_lp < best_val:
            best_val, best_a = val_lp, a_lp.copy()
            print(f"[LP] improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # 5)  Tiny stochastic local search (mass transfer moves)
    local_deadline = deadline - 0.5
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny chunk (at most 1 % of a[i])
        delta = min(best_a[i], 1e-3) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 6)  Try a few cyclic shifts (convolution changes under shift)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 7)  Final sanity checks & return
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
