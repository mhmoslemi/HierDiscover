"""
Hybrid random‑rearrangement + long annealing optimizer.

Goal: minimise  2·n·max(conv(a,a)) / (sum(a)²)  with a_i ≥ 0.
The objective is invariant to scaling, therefore we keep the vector on the
probability simplex (Σa_i = 1) throughout the search.

Strategy
--------
1. Build a diverse initial pool of probability vectors:
   * the previous best vector (if available) and its circular shifts,
   * Dirichlet draws with many concentration parameters,
   * exponential tails, power‑law shapes and a half‑sinusoid,
   * for every non‑uniform shape also an “alternating‑extremes’’ arrangement
     (largest weight left, second largest right, …) which tends to decorrelate
     a with its reversal.

2. Choose the best seed by the exact objective.

3. Intensive simulated‑annealing on the simplex.
   - At each iteration a random fraction of mass is moved from one index to
     another (the total mass stays 1).
   - Acceptance follows the Metropolis rule with a linearly decreasing
     temperature.
   - The current best vector is remembered.
   - The loop runs until the time budget is exhausted (≈1000 s).

4. A short Adam optimisation on a smooth‑max surrogate is performed with the
   remaining time – this refines the solution found by the annealer.

The routine always returns the best vector found before the deadline,
so it is safe against time‑outs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact value of the evaluation function (sum(a) is 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * a.shape[0] * conv.max()


def _alternate_arrangement(v: np.ndarray) -> np.ndarray:
    """
    Place the sorted entries of v alternating from the leftmost and rightmost
    positions (largest left, second largest right, …).  This often decorrelates
    the vector with its reversal, lowering the central convolution peak.
    """
    n = v.shape[0]
    order = np.argsort(v)[::-1]          # descending indices
    out = np.empty_like(v)
    left, right = 0, n - 1
    toggle = True
    for idx in order:
        if toggle:
            out[left] = v[idx]
            left += 1
        else:
            out[right] = v[idx]
            right -= 1
        toggle = not toggle
    return out


def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest yet diverse pool of starting vectors."""
    pool = []

    # --- warm start ----------------------------------------------------
    if warm is not None:
        pool.append(warm.copy())
        # a few circular shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # --- Dirichlet draws -----------------------------------------------
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # --- exponential tails ---------------------------------------------
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # --- power‑law (triangular product) --------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # --- half‑wave sinusoid --------------------------------------------
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    pool.append(sinus)

    # --- uniform --------------------------------------------------------
    pool.append(np.full(n, 1.0 / n))

    # --- alternate‑extreme versions of non‑uniform seeds ----------------
    extra = []
    for v in list(pool):
        if not np.allclose(v, np.full_like(v, 1.0 / n)):
            extra.append(_alternate_arrangement(v))
    pool.extend(extra)

    return pool


def _anneal_optimize(a: np.ndarray,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Long simulated‑annealing on the simplex.  Moves a random fraction of the
    mass from one coordinate to another; the total mass stays 1.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_val = _objective(a, conv_fft_len)
    best_val = cur_val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-5
    move = 0
    report_every = 30_000

    while time.time() < deadline and move < max_moves:
        # linearly decreasing temperature
        T = max(T_end, T_start * (1.0 - move / max_moves))

        i = rng.integers(0, n)
        if a[i] < 1e-14:               # nothing to move from this index
            move += 1
            continue

        # pick a distinct destination
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass
        frac = rng.random() * 0.4
        delta = a[i] * frac

        # apply move in‑place
        a[i] -= delta
        a[j] += delta

        new_val = _objective(a, conv_fft_len)

        # store the best ever seen
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
            if move % report_every == 0:
                print(f"[Anneal it={move:07d}] new best {new_val:.6f}")

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur_val = new_val
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        move += 1

        if move % 5_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    The surrogate temperature λ is increased slowly so that the surrogate
    approaches the true max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.04
    lam_factor = 1.012

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over λ·(conv−max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        # cross‑correlation (a ★ w) == (w ★ a) for real vectors
        rev_a = a[::-1]
        fw = np.fft.rfft(w, conv_fft_len)
        fr = np.fft.rfft(rev_a, conv_fft_len)
        corr = np.fft.irfft(fw * fr, conv_fft_len)
        start = a.shape[0] - 1
        grad_a = 2.0 * corr[start:start + a.shape[0]]

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – slowly move towards the true max
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point used by the competition harness.
    Returns a list of non‑negative heights (the step‑function) with the
    lowest evaluation score found within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)   # 8192

    # ------------------------------------------------------------------
    # Warm start from a previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a pool of initial candidates and keep the best by exact eval
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:03d} val={val:.6f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Phase I – long simulated‑annealing on the simplex (the heaviest phase)
    if best_seq is not None and time.time() < deadline - 2.0:
        print("[Phase I] intensive annealing started")
        a_ann, val_ann = _anneal_optimize(best_seq, conv_fft_len, deadline,
                                          max_moves=800_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase I] annealing improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase II – Adam optimisation on a smooth‑max surrogate (short refinement)
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[Phase II] Adam refinement started")
        sub_dead = deadline - 0.5
        a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len,
                                          sub_dead, max_iters=250_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase II] Adam refined to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
