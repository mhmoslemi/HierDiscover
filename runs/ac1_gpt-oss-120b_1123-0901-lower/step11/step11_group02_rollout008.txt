"""
Multi‑start non‑negative step‑function optimiser.

The routine builds a diverse pool of random and structured candidates,
runs a cheap Adam optimisation on each of them for a few seconds and
keeps the best.  The best candidate is then refined with a longer
pipeline:

    • Adam on the smooth‑max surrogate (many more iterations)
    • Exact max‑convolution sub‑gradient (exponentiated‑gradient)
    • Projected gradient descent on the true max‑convolution
    • Very fine‑grained mass‑transfer local search
    • A short simulated‑annealing pass

The whole search respects a 1000 s wall‑clock budget.  All vectors are
kept on the probability simplex (∑aᵢ=1), because the evaluation function
is scale‑invariant; the final list is clipped to the evaluator’s allowed
range before returning.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic FFT utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories (all return a probability vector)

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_log_normal(n: int, sigma: float) -> np.ndarray:
    z = np.random.normal(0.0, sigma, size=n)
    a = np.exp(z)
    a /= a.sum()
    return a


def _make_harmonic(n: int) -> np.ndarray:
    a = 1.0 / (np.arange(1, n + 1, dtype=np.float64))
    a /= a.sum()
    return a


def _make_piecewise_constant(n: int, segs: int) -> np.ndarray:
    """Random piece‑wise constant on `segs` random intervals."""
    a = np.zeros(n, dtype=np.float64)
    cuts = sorted(np.random.choice(np.arange(1, n), size=segs - 1, replace=False))
    cuts = [0] + cuts + [n]
    for lo, hi in zip(cuts[:-1], cuts[1:]):
        val = np.random.rand()
        a[lo:hi] = val
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers (nearly identical to the reference implementation but
# with a few tuned defaults)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> Tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.7,
) -> Tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-13)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> Tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> Tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        improved = False
        src_mass = a[i_high]
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break

        if not improved:
            # occasional random shake‑up
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> Tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to compute the objective quickly (same as evaluate_sequence for sum=1)

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function (length 4096) that minimises
    ``evaluate_sequence``.  Uses a multi‑start Adam phase followed by a
    deep refinement pipeline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final eval

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if the platform supplied a previous solution)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase I – many cheap Adam runs on random starts
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # allocate ~60 % of the total budget to the exploratory stage
    explore_deadline = start_time + 0.60 * (deadline - start_time)

    # small helper to produce a random candidate from a diversified set
    factory_list = [
        (lambda: _make_uniform(n), ()),
        (lambda: _make_triangle(n), ()),
        (lambda: _make_cosine(n), ()),
        (lambda: _make_dirichlet(n, 0.01), ()),
        (lambda: _make_dirichlet(n, 0.1), ()),
        (lambda: _make_dirichlet(n, 0.5), ()),
        (lambda: _make_dirichlet(n, 1.0), ()),
        (lambda: _make_dirichlet(n, 2.0), ()),
        (lambda: _make_dirichlet(n, 5.0), ()),
        (lambda: _make_dirichlet(n, 10.0), ()),
        (lambda: _make_exponential(n, 0.2), ()),
        (lambda: _make_exponential(n, 0.6), ()),
        (lambda: _make_exponential(n, 1.5), ()),
        (lambda: _make_gaussian(n, 0.05), ()),
        (lambda: _make_gaussian(n, 0.15), ()),
        (lambda: _make_gaussian(n, 0.3), ()),
        (lambda: _make_powerlaw(n, -0.5), ()),
        (lambda: _make_powerlaw(n, -1.0), ()),
        (lambda: _make_powerlaw(n, -2.0), ()),
        (lambda: _make_spikes(n, 64), ()),
        (lambda: _make_spikes(n, 128), ()),
        (lambda: _make_spikes(n, 256), ()),
        (lambda: _make_beta(n, 0.5, 0.5), ()),
        (lambda: _make_beta(n, 1.0, 1.0), ()),
        (lambda: _make_beta(n, 2.0, 2.0), ()),
        (lambda: _make_log_normal(n, 0.5), ()),
        (lambda: _make_log_normal(n, 1.0), ()),
        (lambda: _make_harmonic(n), ()),
        (lambda: _make_piecewise_constant(n, 4), ()),
        (lambda: _make_piecewise_constant(n, 8), ()),
    ]

    # optionally add shifted warm start vectors
    if warm is not None:
        factory_list.append((lambda: warm.copy(), ()))
        for shift in (n // 4, n // 2, 3 * n // 4):
            factory_list.append((lambda s=shift: np.roll(warm, s), ()))

    # run exploratory loop
    while time.time() < explore_deadline:
        # pick a random factory
        fn, _ = factory_list[np.random.randint(len(factory_list))]
        a0 = fn()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # give each trial at most ~5 s (or whatever remains)
        sub_dead = min(explore_deadline, time.time() + 5.0)
        a_opt, val_opt = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=250_000
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Explore] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – deep refinement of the current best vector
    if best_seq is None:
        best_seq = _make_uniform(n)

    # allocate the rest of the budget for a systematic refinement
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # 1) Adam (smooth‑max) – longer run
        sub_dead = min(deadline, time.time() + min(30.0, remaining * 0.35))
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=600_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Refine] Adam improved to {best_val:.6f}")

        remaining = deadline - time.time()
        if remaining < 4.0:
            break

        # 2) Exact max‑convolution sub‑gradient (exp‑grad)
        sub_dead = min(deadline, time.time() + min(20.0, remaining * 0.30))
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=500_000, eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Refine] ExpGrad improved to {best_val:.6f}")

        remaining = deadline - time.time()
        if remaining < 4.0:
            break

        # 3) Projected gradient descent on exact max‑conv
        sub_dead = min(deadline, time.time() + min(15.0, remaining * 0.25))
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Refine] GD improved to {best_val:.6f}")

        remaining = deadline - time.time()
        if remaining < 4.0:
            break

        # 4) Fine‑grained mass‑transfer step
        sub_dead = min(deadline, time.time() + min(10.0, remaining * 0.20))
        a_mt, val_mt = _mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=150_000,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[Refine] MassTransfer improved to {best_val:.6f}")

        remaining = deadline - time.time()
        if remaining < 4.0:
            break

        # 5) Very short simulated‑annealing polishing
        sub_dead = min(deadline, time.time() + min(5.0, remaining * 0.10))
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_moves=50_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Refine] Anneal improved to {best_val:.6f}")

        # after a full cycle break to re‑enter while and check time left
        # (this loop will typically run only once or twice because we
        # consume the remaining budget quickly)
        break

    # ------------------------------------------------------------------
    # Final safety steps (ensure non‑empty return)
    if best_seq is None:
        best_seq = _make_uniform(n)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (for logging only)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
