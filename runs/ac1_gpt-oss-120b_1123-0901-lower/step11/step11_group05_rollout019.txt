"""
Propose a low‑evaluation step‑function.

The routine builds a large set of candidate shapes (uniform,
beta‑densities with α<1 and α>1, power‑law, exponential, sine‑shaped
and many Dirichlet samples), picks the few best by the exact
objective and then refines them with two complementary gradient
phases:

1. Exponentiated‑gradient sub‑gradient descent (exact max‑convolution
   sub‑gradient).  Several top candidates are treated independently;
   this gives a cheap multi‑restart exploration.
2. Adam optimisation on a smooth‑max surrogate (the log‑sum‑exp of
   all convolution entries).  A very high temperature is reached,
   effectively optimising the true max.

Finally a short stochastic local search (random mass moves with
simulated‑annealing acceptance) and a few cyclic shifts are used to
hop out of shallow minima.  All steps respect the supplied time budget
(~1000 s) and the returned list always satisfies the non‑negative,
finite constraints required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution helpers (FFT based, re‑used everywhere)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value of 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Optimisers -----------------------------------------------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑conv."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active constraints (within machine epsilon)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)   # mirror‑descent (KL‑prox)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

def _beta_shape(n, alpha):

    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform
    candidates.append(_uniform(n))

    # beta families (both U‑shaped and bell‑shaped)
    for alpha in [0.4, 0.55, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 3.0]:
        candidates.append(_beta_shape(n, alpha))

    # power‑law tails
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        candidates.append(_powerlaw_shape(n, p))

    # exponential decay
    for gamma in [0.001, 0.003, 0.006, 0.01, 0.02]:
        candidates.append(_exponential_shape(n, gamma))

    # sine‑shaped (zero at the ends)
    candidates.append(_sine_shape(n))

    # random Dirichlet draws with varying concentration
    for _ in range(40):
        conc = np.random.uniform(0.5, 5.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ------------------------------------------------------------------
    # 1) quick exact evaluation of all seeds
    best_a = None
    best_val = np.inf
    for idx, a in enumerate(candidates):
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init candidate {idx}] objective {val:.6f}")

    # ------------------------------------------------------------------
    # 2) multi‑restart sub‑gradient optimisation (≈ 70 % of time)
    sub_deadline = deadline - 0.35 * budget_s          # leave room for final phases
    # keep the 5 best seeds for independent runs
    top_inds = np.argsort([_objective(c, n, conv_fft_len) for c in candidates])[:5]
    for run, ind in enumerate(top_inds):
        if time.time() >= sub_deadline:
            break
        run_deadline = min(sub_deadline, time.time() + (budget_s * 0.12))
        print(f"[Subgrad restart {run}] starting from candidate {ind}")
        a_opt, val_opt = _subgradient_optimize_expgrad(
            candidates[ind], n, conv_fft_len, corr_fft_len,
            run_deadline, max_iters=5_000_000, eta0=2.0
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"  >>> improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) short Adam fine‑tuning (≈ 20 % of remaining time)
    adam_deadline = deadline - 0.15 * budget_s
    if adam_deadline > time.time():
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=300_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam fine] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) tiny stochastic local search with simulated annealing (last seconds)
    local_deadline = deadline - 0.05 * budget_s
    temperature = 1e-3
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # propose moving a fraction of the donor mass
        delta = min(best_a[i], np.random.rand() * 0.02)  # up to 2 % of donor
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            temperature = 1e-3  # reset temperature on improvement
            print(f"[Local move] improved to {val:.6f}")
        else:
            # simulated‑annealing acceptance
            prob = np.exp(-(val - best_val) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                best_a = cand
            temperature *= 0.9999  # cool slowly

    # ------------------------------------------------------------------
    # 5) try a few cyclic shifts (linear convolution changes under a shift)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 6) final sanitisation and reporting
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)          # ensure true simplex projection

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
