"""
Hybrid optimizer for the step‑function bound (target ≈ 1.503).

The pipeline:
1.  Warm‐start from the previously discovered best sequence (global
    ``height_sequence_1``) or a random Dirichlet vector.
2.  Adam optimisation on a smooth‐max surrogate of the peak of the
    self‑convolution, followed by an exact sub‑gradient phase with
    exponentiated‑gradient updates.
3.  “Spectral flattening” – force the FFT magnitude of the candidate to
    be (almost) constant while staying on the probability simplex.
4.  **Full‑LP refinement** – maximise Σb subject to the *full*
    convolution constraints (a★b)ₖ ≤ M (M = current peak).  After each
    LP solve the most violating constraints are added (active‑set
    method).  The resulting b is normalised and mixed with a using a
    line‑search that finds the optimal mixing factor t (instead of a
    fixed τ).  This step dramatically pushes the peak down.
5.  Projected gradient descent on the exact max‑convolution.
6.  Final simulated‑annealing polish.

All iterates stay non‑negative and are normalised (∑a = 1); the
evaluation function is homogeneous, so this does not affect the score.
The routine respects a 1000 s wall‑clock budget and always returns the
best sequence found.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _full_convolution_general(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*b using FFT."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth‑max surrogate of the peak convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # spectral flattening step
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _lp_improve_full(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 8,
    tau_start: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Full linear‑programming refinement.

    For the current sequence a we maximise Σb subject to
        (a ★ b)_k ≤ M   for all k,
    where M = max_k (a ★ a)_k.
    The LP is solved on an active set of constraints; after each solve
    we add any newly violated constraints.  The resulting b is normalised
    and mixed with a using a small 1‑D line‑search that finds the optimal
    mixing factor t∈[0,1] (instead of a fixed τ).  The process iterates
    a few times and yields a strict reduction of the peak.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()
    tau = tau_start

    for outer in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()
        eps = 1e-12

        # initialise active set with peaks and a few well‑spread indices
        active = set(np.where(conv_a >= M - eps)[0].tolist())
        extra = [0, len(conv_a) // 2, len(conv_a) - 1]
        active.update(extra)

        # ------ active‑set LP loop ------
        inner_loop = 0
        while True:
            if time.time() > deadline or inner_loop > 5:
                break
            active_list = sorted(active)
            rows, cols, data = [], [], []
            for row_idx, k in enumerate(active_list):
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                i = np.arange(i_start, i_end + 1, dtype=int)
                j = k - i
                rows.extend([row_idx] * len(i))
                cols.extend(j.tolist())
                data.extend(a[i].tolist())
            A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                        shape=(len(active_list), n))

            c = -np.ones(n)                 # maximise sum(b)  ↔ minimise −∑b
            bounds = [(0, None)] * n
            b_ub = M * np.ones(len(active_list))

            lp_res = scipy.optimize.linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
            if not lp_res.success:
                print(f"[LPfull it={outer}] linprog failed: {lp_res.message}")
                break

            b = lp_res.x
            sum_b = b.sum()
            if sum_b <= 1e-12:
                break
            b_norm = b / sum_b

            # check every constraint (full convolution) for violation
            conv_ab = _full_convolution_general(a, b_norm, conv_fft_len)
            viol = np.where(conv_ab > M + 1e-12)[0]
            if viol.size == 0:
                # feasible – stop the inner loop
                break
            # add new violating indices to the active set
            active.update(viol.tolist())
            inner_loop += 1

        # ---- mix a with the feasible b ----
        # pre‑compute needed convolutions for line‑search
        conv_b = _full_convolution(b_norm, conv_fft_len)
        conv_ab = _full_convolution_general(a, b_norm, conv_fft_len)

        # 1‑D line‑search for the best mixing factor t∈[0,1]
        lo, hi = 0.0, 1.0
        for _ in range(30):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            mix1 = (1 - t1) ** 2 * conv_a + t1 ** 2 * conv_b + 2 * t1 * (1 - t1) * conv_ab
            mix2 = (1 - t2) ** 2 * conv_a + t2 ** 2 * conv_b + 2 * t2 * (1 - t2) * conv_ab
            if mix1.max() < mix2.max():
                hi = t2
            else:
                lo = t1
        t_best = (lo + hi) / 2.0

        # build the new candidate
        a_new = (1 - t_best) * a + t_best * b_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            a_new = np.full_like(a_new, 1.0 / n)

        # evaluate
        conv_new = (1 - t_best) ** 2 * conv_a + t_best ** 2 * conv_b + 2 * t_best * (1 - t_best) * conv_ab
        val_new = 2.0 * n * conv_new.max()

        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            print(f"[LPfull it={outer}] improved to {val_new:.6f} (t={t_best:.4f})")
        else:
            # no improvement – shrink future mixing factor
            tau *= 0.5
            if tau < 1e-4:
                break
            # still move a a little towards b to escape potential stagnation
            a = a_new

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Main entry point – returns a non‑negative step function (list of floats)."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ---- problem dimensions ----
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- warm start (previous best if present) ----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        elif warm.shape[0] > n:
            warm = warm[:n]
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ---- initial candidate ----
    if warm is not None:
        a_curr = warm.copy()
    else:
        a_curr = np.random.dirichlet(np.ones(n))

    best_seq = a_curr.copy()
    best_val = 2.0 * n * _full_convolution(best_seq, conv_fft_len).max()
    print(f"[Init] value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – Adam + sub‑gradient + spectral flattening
    phase1_dead = min(deadline, time.time() + budget_s * 0.30)

    # Adam
    if time.time() < phase1_dead:
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, phase1_dead, max_iters=400_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase I – Adam] improved to {val_adam:.6f}")

    # Sub‑gradient (exponentiated gradient)
    if time.time() < phase1_dead:
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, phase1_dead,
            max_iters=400_000, eta0=0.8
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase I – Subgrad] improved to {val_sub:.6f}")

    # Spectral flattening
    if time.time() < phase1_dead:
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, phase1_dead, max_iters=180_000
        )
        if val_spec < best_val:
            best_val, best_seq = val_spec, a_spec.copy()
            print(f"[Phase I – Spectral] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase II – Full LP refinement with line‑search mixing
    if time.time() < deadline - 20:
        print("[Phase II – LPfull] start")
        a_lp, val_lp = _lp_improve_full(
            best_seq, n, conv_fft_len, deadline, max_iter=8, tau_start=0.5
        )
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[Phase II – LPfull] improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase III – Projected gradient descent (exact max)
    if time.time() < deadline - 10:
        print("[Phase III – GD] start")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=150_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Phase III – GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – Final simulated‑annealing polish
    if time.time() < deadline - 2:
        print("[Phase IV – Anneal] start")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=80_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase IV – Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s permitted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
