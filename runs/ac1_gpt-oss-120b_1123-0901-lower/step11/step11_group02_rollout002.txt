"""
Hybrid coarse‑to‑fine optimizer for the step‑function evaluation problem.

Algorithm
---------
1. **Coarse phase (n≈1024)** – many cheap random starts.
   * Generate a random candidate (Dirichlet, Beta or sparse spikes).
   * Run a short exponentiated‑gradient sub‑gradient optimisation on the
     exact max‑convolution objective (≈5 s per start).
   * Keep the best coarse shape.

2. **Upsampling** – interpolate the best coarse vector to the target length
   (n=4096) and renormalise to unit sum.

3. **Fine phase (n=4096)** – longer deterministic refinements.
   * Adam optimisation on a smooth‑max surrogate (gradually increasing
     temperature) – up to a few 10 s.
   * Exponentiated‑gradient sub‑gradient optimisation – up to a few 10 s.
   * Targeted mass‑transfer local search – moves mass from the coordinate
     with the largest gradient to the one with the smallest, accepting only
     strict objective improvements.
   * Optional short simulated‑annealing (if time permits).

4. Clip the final vector to the evaluator’s range and return it as a plain
   Python list.  Intermediate progress and objective values are printed for
   monitoring.

The implementation respects the 1000 s budget, uses only NumPy and the
provided ``evaluate_sequence`` function, and works without any file or
network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility primitives

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)**2)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Candidate factories

def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta density on a uniform grid of cell centres."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a


def _make_spikes(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a = a / a.sum()
    return a

# Optimisers

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(self‑conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                    # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max (ensures a stays on simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _expgrad_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                      corr_fft_len: int, deadline: float,
                      max_iters: int = 200_000,
                      eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-13)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                           corr_fft_len: int, deadline: float,
                           max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Fine‑grained mass‑transfer local search:
    move a fraction of mass from the coordinate with the largest gradient
    to the one with the smallest gradient, accepting only strict improvements.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if grad[i_high] <= grad[i_low] + 1e-15:
            break   # near stationary

        src_mass = a[i_high]
        improved = False
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # random perturb to escape plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the provided evaluation function.  A coarse‑to‑fine strategy is used:
    many cheap random starts at n≈1024, up‑sampling the best, followed by
    deeper deterministic refinements at n=4096.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Dimensions
    coarse_n = 1024
    fine_n = 4096

    # FFT lengths for both resolutions
    conv_fft_len_coarse = _next_pow2(2 * coarse_n - 1)
    corr_fft_len_coarse = _next_pow2(3 * coarse_n - 2)

    conv_fft_len_fine = _next_pow2(2 * fine_n - 1)
    corr_fft_len_fine = _next_pow2(3 * fine_n - 2)

    # Warm‑start handling (if previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > coarse_n:
            warm = warm[:coarse_n]
        elif warm.shape[0] < coarse_n:
            warm = np.pad(warm, (0, coarse_n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Coarse random‑search phase
    coarse_deadline = min(deadline, start_time + 0.6 * budget_s)  # ~60 % time
    best_coarse_a = None
    best_coarse_val = float("inf")
    it_coarse = 0

        r = np.random.rand()
        if r < 0.45:
            # Dirichlet with random concentration
            alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10

        elif r < 0.80:
            # Beta with random (α,β) on log‑scale

            a0 = _make_spikes(coarse_n, k)

        # Optional warm‑start injection

            a0 = (a0 + warm) / 2.0

        # ----- short exp‑gradient optimisation -----
        sub_dead = min(coarse_deadline, time.time() + 4.5)   # ~4.5 s per start
        a_opt, val_opt = _expgrad_optimize(
            a0, coarse_n, conv_fft_len_coarse, corr_fft_len_coarse,
            sub_dead, max_iters=80_000, eta0=0.9
        )
        if val_opt < best_coarse_val:
            best_coarse_val = val_opt
            best_coarse_a = a_opt.copy()
            print(f"[Coarse #{it_coarse}] new best {val_opt:.6f}")

        it_coarse += 1

    if best_coarse_a is None:
        best_coarse_a = np.full(coarse_n, 1.0 / coarse_n, dtype=np.float64)
        best_coarse_val = _objective(best_coarse_a, coarse_n)

    # ------------------------------------------------------------------

    fine_a = np.interp(x_fine, x_coarse, best_coarse_a)
    fine_a = np.maximum(fine_a, 0.0)
    fine_a = fine_a / fine_a.sum()

    best_seq = fine_a.copy()
    best_val = _objective(best_seq, fine_n)
    print(f"[After upsample] coarse best val={best_coarse_val:.6f} → fine start val={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fine deterministic refinement phase
    # Allocate remaining time among the three fine optimisers
    time_left = deadline - time.time()
    # 1) Adam optimisation (≈25 % of remaining)
    if time_left > 30:
        sub_dead = min(deadline, time.time() + time_left * 0.25)
        a_adam, val_adam = _adam_optimize(
            best_seq, fine_n, conv_fft_len_fine, corr_fft_len_fine,
            sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Fine Adam] improved to {val_adam:.6f}")

    # 2) Exponentiated‑gradient sub‑gradient (≈30 % of remaining)
    time_left = deadline - time.time()
    if time_left > 30:
        sub_dead = min(deadline, time.time() + time_left * 0.30)
        a_exp, val_exp = _expgrad_optimize(
            best_seq, fine_n, conv_fft_len_fine, corr_fft_len_fine,
            sub_dead, max_iters=400_000, eta0=0.7
        )
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[Fine ExpGrad] improved to {val_exp:.6f}")

    # 3) Mass‑transfer local search (≈20 % of remaining)
    time_left = deadline - time.time()
    if time_left > 20:
        sub_dead = min(deadline, time.time() + time_left * 0.20)
        a_mt, val_mt = _mass_transfer_optimize(
            best_seq, fine_n, conv_fft_len_fine, corr_fft_len_fine,
            sub_dead, max_iters=200_000
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[Fine MassTransfer] improved to {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Optional very short simulated annealing if any time remains
    time_left = deadline - time.time()
    if time_left > 10:
        # simple annealing moving small mass fractions
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len_fine)
        val = 2.0 * fine_n * conv.max()
        best_val_ann = val
        best_a_ann = a.copy()
        T_start = 0.06
        T_end = 6e-5
        moves = int(time_left * 1000)  # heuristic number of moves
        for it in range(moves):
            T = max(T_end, T_start * (1.0 - it / moves))
            i = np.random.randint(fine_n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(fine_n)
            if i == j:
                continue
            frac = np.random.rand() * 0.44
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution(a2, conv_fft_len_fine)
            val2 = 2.0 * fine_n * conv2.max()
            if val2 < best_val_ann:
                best_val_ann = val2
                best_a_ann = a2.copy()
                a = a2
                val = val2
            else:
                if np.random.rand() < np.exp(-(val2 - val) / T):
                    a = a2
                    val = val2
        if best_val_ann < best_val:
            best_val = best_val_ann
            best_seq = best_a_ann.copy()
            print(f"[Anneal] improved to {best_val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final clipping and evaluation
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
