"""
Propose a low‑score step‑function.

The score is  2·n·max(conv(a,a)) / (sum(a)^2)  with a_i ≥ 0.
Because the formula is scale‑invariant we keep the sequence on the
probability simplex (∑a_i = 1).  The goal is to make the linear
convolution a*a as flat as possible – its average value is 1/(2n‑1),
so the theoretical optimum is 2n/(2n‑1) ≃ 1.  We use a hybrid
continuous optimiser:

1. A pool of diverse initial vectors (uniform, Dirichlet, exponential,
   Gaussian, half‑sine …) on a relatively large size n=8192.
2. Adam optimisation on a smooth‑max surrogate of max(conv).  The
   surrogate uses a temperature λ that grows during optimisation.
3. Exponentiated‑gradient sub‑gradient optimiser that works on the
   exact max (active constraints only).
4. An L2‑descent phase that drives the whole convolution towards its
   average value (flattening the whole profile).
5. A short simulated‑annealing polish.

All phases respect a global 1000 s budget and always return a
non‑negative list.  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of initial candidates

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    a = left + left[::-1]
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_half_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_laplacian(n, gamma):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-gamma * np.abs(i - center))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=600_000):
    """Adam on a log‑sum‑exp (smooth‑max) surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5                # starting temperature
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to enforce simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _expgrad_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                     max_iters=1_500_000, eta0=0.9):
    """Exponentiated‑gradient optimiser on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _l2_flatten_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                         max_iters=300_000, lr0=0.2):
    """Projected gradient descent on the L2 distance to a flat convolution."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    avg_target = 1.0 / (2 * n - 1)          # average of a*a when sum(a)=1
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        diff = conv - avg_target
        grad = 2.0 * _cross_correlation(diff, a, corr_fft_len)

        step = lr0 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.clip(a, 0.0, None)
        a = _project_simplex(a)

        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2 it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(a0, n, conv_fft_len, deadline,
                    max_moves=200_000):
    """Simulated‑annealing polish on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence (list of floats) it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Problem size – larger than in the reference code to improve the theoretical bound
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run produced a good sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Assemble a modest but diverse pool of initial vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # a few shifted copies of the warm start
        for shift in (n // 8, n // 4, n // 2):
            pool.append(np.roll(warm, shift))

    pool.append(_make_uniform(n))

    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        pool.append(_make_dirichlet(n, alpha))

    for gamma in (0.05, 0.15, 0.3, 0.6, 1.0):
        pool.append(_make_exponential(n, gamma))

    for sigma in (0.04, 0.1, 0.2):
        pool.append(_make_gaussian(n, sigma))

    pool.append(_make_half_sine(n))
    pool.append(_make_cosine(n))

    for a, b in ((0.5, 0.5), (0.5, 1.5), (1.5, 0.5), (2.0, 2.0)):
        pool.append(_make_beta(n, a, b))

    for gamma in (0.01, 0.03, 0.07):
        pool.append(_make_laplacian(n, gamma))

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    cand_idx = 0
    total_candidates = len(pool)

    # ------------------------------------------------------------------
    # Phase I – Adam optimisation on a few promising starts
    while time.time() < deadline and cand_idx < total_candidates:
        # Reserve ~80 s for the final refinement phases
        if deadline - time.time() < 80.0:
            break

        a0 = pool[cand_idx]
        cand_idx += 1

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # allocate a slice for this candidate
        remaining = deadline - time.time()
        slice_time = min(45.0, remaining * 0.35)
        sub_dead = time.time() + slice_time

        # ---- Adam on smooth‑max -----------------
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=800_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cand_idx-1}] Adam improved to {val_adam:.6f}")

        # ---- Exponentiated‑gradient (exact max) -----
        if time.time() < deadline - 30.0:
            a_exp, val_exp = _expgrad_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=1_200_000, eta0=0.9
            )
            if val_exp < best_val:
                best_val = val_exp
                best_seq = a_exp.copy()
                print(f"[Cand {cand_idx-1}] ExpGrad improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase II – L2 flattening (global conv flattening)
    if best_seq is not None and time.time() < deadline - 40.0:
        print("[Phase II] L2 flattening")
        a_l2, val_l2 = _l2_flatten_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=500_000, lr0=0.15
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Phase II] L2 improved to {val_l2:.6f}")

    # ------------------------------------------------------------------
    # Phase III – Projected gradient descent on the exact max (fine‑tune)
    if best_seq is not None and time.time() < deadline - 30.0:
        print("[Phase III] Projected GD fine‑tuning")
        a_gd, val_gd = _expgrad_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=800_000, eta0=0.7
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Phase III] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – Simulated‑annealing polish
    if best_seq is not None and time.time() < deadline - 25.0:
        print("[Phase IV] Simulated‑annealing polish")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=250_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety – ensure we return something
    if best_seq is None:
        best_seq = _make_uniform(n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
