def _make_double_gaussian(n, sigma, offset_frac):
    i = np.arange(n, dtype=np.float64)
    center1 = offset_frac * (n-1)

def _adam_optimize_smart(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=2_000_000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr0 = 0.04
    lam = 0.02
    lam_factor = 1.02
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # smooth max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(convolution(a, a)) / (sum(a)^2)

The optimizer works in four stages:

1. **Initial pool** – a diverse collection of parametric families
   (uniform, Gaussian, exponential, power‑law, double‑Gaussian, etc.).
   The best member of the pool (according to the official
   ``evaluate_sequence``) is kept as a warm‑start.

2. **Adam on a smooth‑max surrogate** – the objective ``max(conv)``
   is replaced by a soft‑max with a slowly rising temperature (``lam``).
   The variables are represented by logits of a soft‑max, guaranteeing
   non‑negativity and unit sum.  Adam with a decreasing learning‑rate
   is run for a large number of iterations (subject to the time budget).

3. **Exact‑max exponentiated‑gradient refinement** – after the smooth‑max
   stage the temperature is already huge, i.e. the gradient is the exact
   sub‑gradient of the ``max``.  An exponentiated‑gradient (mirror descent
   on the simplex) with a ``1/√t`` step‑size shrinks the maximal
   convolution further.

4. **Simulated‑annealing polish** – a short local‐search that moves a
   random fraction of mass from one index to another.  The temperature
   is annealed logarithmically; moves that improve the objective are
   always accepted.

The whole routine respects the 1000 s budget, prints progress,
and always returns a list of floats (length 4096) that satisfies the
security checks of ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple families (used to seed the optimiser)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_double_gaussian(n, sigma, offset_frac):
    """Two equal Gaussians symmetrically placed around the centre."""
    i = np.arange(n, dtype=np.float64)
    c1 = offset_frac * (n - 1)
    c2 = (1 - offset_frac) * (n - 1)
    a = np.exp(-0.5 * ((i - c1) / (sigma * n)) ** 2)
    a += np.exp(-0.5 * ((i - c2) / (sigma * n)) ** 2)
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a


def _make_cosine_power(n, exponent):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** exponent
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_sine_power(n, exponent):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** exponent
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_half_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_laplacian(n, gamma):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-gamma * np.abs(i - center))
    a /= a.sum()
    return a


def _make_spikes(n, k):
    """k equally‑spaced spikes of equal weight."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers

def _adam_optimize_smart(a0: np.ndarray, n: int, conv_fft_len: int,
                         corr_fft_len: int, deadline: float,
                         max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of the max‑convolution.
    The temperature ``lam`` grows slowly (factor 1.02) so that the
    optimisation stays “smooth’’ for many iterations.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr0 = 0.04                               # base learning rate
    lam = 0.02                               # start temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                   # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate decay ~ 1/√t
        lr = lr0 / np.sqrt(it + 1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e7)

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation with the exact max‑convolution.
    ``eta0`` is the initial step‑size; it decays as 1/√t.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (exact max)
        active = np.isclose(conv, max_c, atol=1e-15, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against degeneracy
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    deadline: float, max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search on the simplex.
    Moves transfer a random fraction of mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The search proceeds through an initial pool, a long Adam stage,
    a precise exponentiated‑gradient stage and a final short annealing
    polish.  The function always returns a list (length 4096) whose sum
    is positive, satisfying the evaluator’s constraints.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # problem size – 4096 points give a good speed/quality trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)   # 8192
    corr_fft_len = _next_pow2(3 * n - 2)   # 16384 (ample)

    # ------------------------------------------------------------------
    # Build a diverse initial pool (including the previous best)
    pool = []

    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
        if w is not None:
            pool.append(w)

    # parametric families
    pool.append(_make_uniform(n))
    pool.append(_make_half_sine(n))
    pool.append(_make_cosine(n))
    pool.append(_make_triangle(n))

    # Gaussian sweep
    for sigma in np.linspace(0.02, 0.5, 24):
        pool.append(_make_gaussian(n, sigma))

    # Exponential sweep
    for gamma in np.linspace(0.05, 2.0, 20):
        pool.append(_make_exponential(n, gamma))

    # Power‑law sweep (negative exponents)
    for expo in np.linspace(-0.2, -3.0, 15):
        pool.append(_make_powerlaw(n, expo))

    # Double‑Gaussian mixtures
    for offset in (0.2, 0.3, 0.4):
        pool.append(_make_double_gaussian(n, sigma=0.05, offset_frac=offset))

    # Cosine‑power family
    for expnt in (0.5, 1.0, 1.5, 2.0):
        pool.append(_make_cosine_power(n, expnt))

    # Sine‑power family
    for expnt in (0.5, 1.0, 1.5, 2.0):
        pool.append(_make_sine_power(n, expnt))

    # Beta‑distribution shapes
    for alpha, beta in (
        (0.5, 0.5), (0.5, 1.0), (1.0, 0.5), (2.0, 2.0),
        (0.2, 2.0), (2.0, 0.2), (0.8, 0.8), (5.0, 1.0)
    ):
        pool.append(_make_beta(n, alpha, beta))

    # Laplacian tails
    for gamma in (0.01, 0.02, 0.05, 0.1):
        pool.append(_make_laplacian(n, gamma))

    # Sparse spikes of various cardinalities
    for k in (16, 32, 64, 128, 256):
        pool.append(_make_spikes(n, k))

    # Random Dirichlet draws
    for _ in range(30):
        alpha = 10 ** np.random.uniform(-3, 2)   # 1e‑3 … 1e2
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    # Pick the best element of the pool (cheap evaluation)
    best_seq = None
    best_val = float('inf')
    for a in pool:
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            continue
        a = a / a.sum()
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"[Init] best value from pool {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – long Adam optimisation (smooth‑max)
    phase_i_end = time.time() + 0.45 * (deadline - time.time())
    if best_seq is not None and time.time() < phase_i_end:
        print("[Phase I] Adam optimisation start")
        a_adam, val_adam = _adam_optimize_smart(
            best_seq, n, conv_fft_len, corr_fft_len, phase_i_end,
            max_iters=2_500_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – exponentiated‑gradient refinement (exact max)
    phase_ii_end = time.time() + 0.85 * (deadline - time.time())
    if best_seq is not None and time.time() < phase_ii_end:
        print("[Phase II] Exponentiated‑gradient refinement start")
        a_exp, val_exp = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, phase_ii_end,
            max_iters=5_000_000, eta0=0.8
        )
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[Phase II] Exp‑grad improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase III – short simulated‑annealing polish
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase III] Simulated‑annealing polish")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=350_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase III] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final safety: ensure we return a non‑empty, valid sequence
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
