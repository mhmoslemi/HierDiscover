def random_phase_candidate(n, energy):
   # DC is sum = 1 -> A0 = 1
   M = sqrt((n*energy - 1)/ (n-1))
   # generate random phases for k=1..n-1

   A[0] = 1.0
   # fill half
   half = n//2
   if n % 2 == 0:
       # even: Nyquist frequency index half is real
       A[1:half] = M * np.exp(1j*phi[:half-1])

def random_phase_sampling(a0, n, deadline, max_iters=500000):
    # compute target energy from a0
    energy = np.dot(a0, a0)
    # compute magnitude M (if energy less than 1/n? then rem_energy negative => set M=0)
    if n*energy <= 1.0:
        M = 0.0
    else:
        M = math.sqrt((n * energy - 1.0) / (n - 1))
    best = a0.copy()
    best_val = score(a0)
    it = 0
    while time.time() < deadline and it < max_iters:
        # generate random phases (only half due to hermitian symmetry)
        phi = np.random.rand(n-1) * 2 * np.pi
        # generate complex spectrum
        # We need to fill appropriate indices to enforce conjugate symmetry
        A = np.empty(n, dtype=np.complex128)
        A[0] = 1.0
        if n % 2 == 0:
            half = n//2
            # indices 1..half-1
            A[1:half] = M * np.exp(1j * phi[:half-1])
            # Nyquist frequency at index half is real: set to random sign +/- M
            A[half] = M * (1 if np.random.rand()<0.5 else -1)  # real
            # fill rest
            A[half+1:] = np.conj(A[1:half][::-1])
        else:
            half = n//2

        s = a_candidate.sum()
        if s < 1e-12:

        a_candidate /= s
        val = score(a_candidate)
        if val < best_val:
            best_val = val
            best = a_candidate.copy()
            print(f"[RandomPhase it={it}] new best {val:.6f}")
        it += 1
    return best, best_val

"""
Hybrid optimizer that mixes projected gradient descent, an intensive
random‑Fourier‑phase search and simulated‑annealing refinements.
It respects a ~1000 s budget and always returns the best non‑negative
step‑function it discovers (the sequence is finally clipped to
[0, 1000] to satisfy the evaluator).

import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT of length ``fft_len``.
    Returns the first 2*len(a)-1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for sub‑gradient / gradient.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

    Projected gradient descent on the exact max‑convolution.

    best_val = 2.0 * n * conv.max()

            best_a = a.copy()
        it += 1

    return best_a, best_val


def _compute_score(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value for a (expects Σa = 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _random_phase_candidate(M_base: float, n: int) -> np.ndarray | None:
    """
    Build a vector from a random Fourier phase pattern.
    The DC component is fixed to 1 (so the sum of the vector equals 1).
    All non‑DC magnitudes are equal to M_base (optionally scaled).
    """
    # Occasionally jitter the magnitude a little
    scale = 1.0
    if np.random.rand() < 0.3:
        scale = np.random.uniform(0.5, 1.5)
    M = M_base * scale

    phi = np.random.rand(n - 1) * 2.0 * np.pi
    A = np.empty(n, dtype=np.complex128)
    A[0] = 1.0
    if n % 2 == 0:
        half = n // 2
        if half > 1:
            A[1:half] = M * np.exp(1j * phi[:half - 1])
        # Nyquist frequency must be real
        A[half] = M * (1.0 if np.random.rand() < 0.5 else -1.0)
        A[half + 1:] = np.conj(A[1:half][::-1])
    else:
        half = n // 2
        A[1:half + 1] = M * np.exp(1j * phi[:half])
        A[half + 1:] = np.conj(A[1:half + 1][::-1])

    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return None
    a /= s
    return a


def _random_phase_search(
    base_a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exhaustive exploration of random phase vectors.
    Keeps the energy (L2 norm) of ``base_a`` as a reference.
    """
    energy = float(np.dot(base_a, base_a))
    if n * energy <= 1.0:
        M_base = 0.0
    else:
        M_base = math.sqrt((n * energy - 1.0) / (n - 1))

    best = base_a.copy()
    best_val = _compute_score(best, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        a_cand = _random_phase_candidate(M_base, n)
        if a_cand is None:
            it += 1
            continue
        val = _compute_score(a_cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best = a_cand.copy()
            print(f"[RandomPhase it={it}] new best {val:.6f}")
        it += 1
    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    if warm is None:
        warm = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = warm.copy()
    best_val = _compute_score(best_seq, n, conv_fft_len)
    print(f"Initial score {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – a quick projected‑gradient descent from the warm start
    if time.time() < deadline - 300:
        sub_dead = min(deadline - 250, time.time() + 100.0)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=400_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random‑phase exploration (the core of this submission)
    if time.time() < deadline - 150:
        rp_dead = deadline - 150
        a_rp, val_rp = _random_phase_search(
            best_seq, n, conv_fft_len, rp_dead, max_iters=5_000_000
        )
        if val_rp < best_val:
            best_val, best_seq = val_rp, a_rp.copy()
            print(f"[RandomPhase] improved to {val_rp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – a short gradient polish after the phase‑2 search
    if time.time() < deadline - 80:
        gd_dead = min(deadline - 50, time.time() + 60.0)
        a_gd2, val_gd2 = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, gd_dead, max_iters=200_000
        )
        if val_gd2 < best_val:
            best_val, best_seq = val_gd2, a_gd2.copy()
            print(f"[GD‑post] improved to {val_gd2:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing refinement
    if time.time() < deadline - 30:
        ann_dead = deadline - 30
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, ann_dead, max_moves=400_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final short perturb‑and‑refine cycles (few seconds each)
    while time.time() < deadline - 5:
        noise = np.random.randn(n) * 0.0008
        a_noise = best_seq + noise
        a_noise = np.maximum(a_noise, 0.0)
        a_noise = _project_simplex(a_noise)
        sub_dead = min(deadline, time.time() + 4.5)
        a_ref, val_ref = _max_conv_descent(
            a_noise, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=30_000
        )
        if val_ref < best_val:
            best_val, best_seq = val_ref, a_ref.copy()
            print(f"[Perturb] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Final safety clamp
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
