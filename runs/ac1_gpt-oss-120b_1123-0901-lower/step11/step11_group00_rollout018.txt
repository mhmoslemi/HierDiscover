"""
Propose a near‑optimal non‑negative step‑function sequence.

The algorithm starts from a warm initial vector (the previously found
`height_sequence_1` if available, otherwise a uniform distribution), then
applies a sequence of continuous‑optimisation stages:

1. Adam on a smooth‑max approximation of the max‑convolution.
2. Exponentiated‑gradient (sub‑gradient) optimisation of the exact max‑convolution.
3. Projected gradient descent with a simple back‑tracking line search.
4. Spectral flattening (forcing the FFT magnitude to be (almost) constant).
5. Local phase‑perturbation on the constant‑magnitude manifold.
6. A lightweight simulated‑annealing move‑swap phase for final fine‑tuning.

Each stage runs only while time permits (total budget ≈ 1000 s) and
keeps the best vector seen so far.  The returned list is guaranteed to be
non‑negative, sum‑normalised and clipped to the evaluator’s allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex {x≥0, Σx=1}. """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c               # sum(a)=1 by construction
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 350_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient optimisation of the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    step = 0.12

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # simple back‑track: keep the step if it improves
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.02, 0.3)   # gently increase step when successful
            if it % 5_000 == 0:
                print(f"[GD it={it:06d}] new best {val_new:.6f}")
        else:
            step *= 0.5                     # halve step on failure

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ---- spectral flattening step ----
        A = np.fft.fft(a)
        energy = np.dot(a, a)           # ||a||_2^2
        if n > 1:
            # keep DC component unchanged, make all other mags equal
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                # preserve DC (sum = 1)
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _phase_local_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Local phase‑perturbation: keep the magnitude fixed (from a0) and
    repeatedly add a small Gaussian perturbation to the phases,
    gradually decreasing the perturbation amplitude.
    """
    mag = np.abs(np.fft.fft(a0))
    # initial phases (real valued)
    phi = np.angle(np.fft.fft(a0))

    a_best = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    start = time.time()
    half = n // 2

    while time.time() < deadline and (time.time() - start) < max_iters * 0:  # dummy loop condition
        # linearly decay sigma from π/2 to 1e-3 over the allowed time
        elapsed = time.time() - start
        total_allowed = deadline - start
        frac = min(1.0, elapsed / total_allowed)
        sigma = (np.pi / 2) * (1.0 - frac) + 1e-3 * frac

        # perturb the phases (except DC)
        delta_phi = sigma * np.random.randn(half - 1)
        new_phi = phi.copy()
        new_phi[1:half] += delta_phi
        # enforce Hermitian symmetry for real IFFT
        new_phi[-(half - 1):] = -new_phi[1:half][::-1]

        if n % 2 == 0:
            # Nyquist frequency (real): random sign flip
            if np.random.rand() < 0.5:
                new_phi[half] = -new_phi[half]

        A = mag * np.exp(1j * new_phi)
        a = np.real(np.fft.ifft(A))

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            a_best = a.copy()
            phi = new_phi  # keep the improved phases as a new centre
            if int(elapsed * 10) % 5000 == 0:
                print(f"[PhaseLocal it={int(elapsed*1000)}] new best {val:.6f}")

        # protection against infinite loops: stop after a fixed number of tries
        if elapsed > total_allowed:
            break

    return a_best, best_val


def _anneal_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Light simulated‑annealing refinement moving small amounts of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-4
    moves = 0

    while time.time() < deadline and moves < max_moves:
        T = max(T_end, T_start * (1.0 - moves / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue
        # transfer a random fraction of mass from i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        moves += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096                                     # a good trade‑off
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously discovered sequence, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Initialise a pool of diverse start vectors (small pool – we run long
    # optimisation on each, then keep the best)
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # a few Dirichlet draws with different concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))
    # a sinusoid‑like start
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    sinusoid /= sinusoid.sum()
    pool.append(sinusoid)

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 5                 # we will fully optimise a handful of starts

    # ------------------------------------------------------------------
    # Stage 1 – deep optimisation of a few promising starts
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ---- Adam (smooth‑max) ----
        sub_dead = min(deadline, time.time() + 20.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # ---- Sub‑gradient (exponentiated‑gradient) ----
        if time.time() < deadline - 5.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, max_iters=400_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected gradient descent fine‑tuning
    if best_seq is not None and time.time() < deadline - 5.0:
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=200_000
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – spectral flattening (constant‑magnitude FFT)
    if best_seq is not None and time.time() < deadline - 5.0:
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline, max_iters=180_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – local phase perturbation on the constant‑magnitude manifold
    if best_seq is not None and time.time() < deadline - 5.0:
        a_phase, val_phase = _phase_local_refine(
            best_seq, n, conv_fft_len, deadline, max_iters=250_000
        )
        if val_phase < best_val:
            best_val = val_phase
            best_seq = a_phase.copy()
            print(f"[PhaseLocal] improved to {val_phase:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing mass‑swap refinement
    if best_seq is not None and time.time() < deadline - 5.0:
        a_ann, val_ann = _anneal_refine(
            best_seq, n, conv_fft_len, deadline, max_moves=200_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip according to evaluator’s safety checks
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
