def construct_sidon_set(n, m_target, max_trials=1000):
    for trial in range(max_trials):
        candidate_indices = np.random.permutation(n)
        S = []
        sums = set()
        for idx in candidate_indices:
            # compute sums idx + s for all s in S plus idx+idx
            conflict = False
            new_sums = []
            for s in S:
                s_val = idx + s
                if s_val in sums:
                    conflict = True
                    break
                new_sums.append(s_val)
            if not conflict:
                ss = idx + idx
                if ss in sums:
                    continue
                # Add
                S.append(idx)
                sums.add(ss)
                for s_val in new_sums:
                    sums.add(s_val)
                if len(S) >= m_target:
                    return np.array(S, dtype=int)
        # try again
    raise RuntimeError("Failed to construct Sidon set")

def hill_climb(a, n, deadline, max_moves=200_000):
    a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()
    while time.time() < deadline and max_moves > 0:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] < 1e-12: continue
        # move small fraction
        frac = np.random.rand() * 0.2  # up to 20%
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        max_moves -= 1
    return best_a, best_val
```

def _hill_climb_optimize(a0, n, conv_fft_len, deadline, max_moves=20000):
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    moves = 0
    while time.time() < deadline and moves < max_moves:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] < 1e-12:
            continue
        frac = np.random.rand() * 0.2  # up to 20% of mass at i
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        moves += 1
        if best_val <= 1.503:   # early break
            break
    return best_a, best_val

"""
Hybrid search for a low‑max‑convolution step function.

Strategy
--------
1. Generate a diverse pool of candidate probability vectors:
   * warm start (if available)
   * random Dirichlet draws with several concentration values
   * a few analytic shapes (uniform, exponential, power‑law, triangular)
   * each analytic shape is also tested after a random permutation, which
     tends to break the symmetry that causes the central peak of the
     convolution.

2. For each candidate we run a cheap “flattening” stage (L2‑Adam) for a
   few seconds, followed by a smooth‑max Adam stage.  The best vector
   seen so far is kept.

3. When the global time budget permits we switch to a long refinement
   on the current best vector:
   * exponential‑gradient (exact max sub‑gradient)
   * projected gradient descent on the true max‑convolution
   * simulated‑annealing
   * a final hill‑climbing phase that moves a tiny amount of mass
     between two coordinates if it reduces the objective.

4. If at any point the objective falls below the target 1.503 we stop
   early and return the current best sequence.

All sequences are kept on the probability simplex (non‑negative,
normalised to sum 1) because the evaluation function is scale invariant.
The routine is time‑budget aware and guarantees a non‑empty return.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories --------------------------------------------------
def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_permuted(base_func, *args, n):
    """Create a candidate by permuting the output of a base generator."""
    a = base_func(n, *args)
    np.random.shuffle(a)
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the L2 distance to a uniform convolution target."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Pure deterministic hill‑climbing: move a random fraction of mass from
    one coordinate to another whenever the move decreases the objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    moves = 0

    while time.time() < deadline and moves < max_moves:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] < 1e-12:
            continue

        # move at most 30 % of the source mass
        frac = np.random.rand() * 0.30
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2

        moves += 1
        # early exit if we already beat the target
        if best_val <= 1.503:
            break

    return best_a, best_val

# Main search routine --------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine is time‑budget aware and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – larger n gives more flexibility, but n=4096 is fast enough
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best) if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of initial vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())

    # Dirichlet draws with several concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))

    # Analytic families + permuted variants
    pool.append(_make_uniform(n))
    pool.append(_make_exponential(n, gamma=0.7))
    pool.append(_make_powerlaw(n, expo=-0.9))
    pool.append(_make_triangle(n))

    # permuted versions – break symmetry
    pool.append(_make_permuted(_make_exponential, 0.7, n=n))
    pool.append(_make_permuted(_make_powerlaw, -0.9, n=n))
    pool.append(_make_permuted(_make_triangle, n=n))

    # a few extra pure random Dirichlets
    for _ in range(4):
        pool.append(_make_dirichlet(n, np.random.rand() * 2.0))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # uniform convolution target used by the L2‑flattening phase
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # First pass: cheap optimisation of many candidates
    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ---- Phase A: short L2‑Adam (flatten convection) ----
        phase_dead = min(deadline, time.time() + 7.0)   # ~7 s
        a_l2, val_l2 = _adam_optimize_l2(
            a0, n, conv_fft_len, corr_fft_len, target, phase_dead,
            max_iters=250_000, lr=0.05
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[Cand {idx}] L2‑Adam best {val_l2:.6f}")

        # ---- Phase B: quick smooth‑max Adam ----
        if time.time() < deadline - 5:
            phase_dead = min(deadline, time.time() + 5.0)
            a_adam, val_adam = _adam_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, phase_dead,
                max_iters=150_000
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Cand {idx}] Smooth‑max Adam improved to {val_adam:.6f}")

        # early stop if we already beat the target
        if best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # Long refinement on the current best vector
    if best_seq is not None and time.time() < deadline - 30:
        print("[Refine] Long exponentiated‑gradient phase")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=800_000, eta0=0.8
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Refine] ExpGrad improved to {val_sub:.6f}")

    if best_seq is not None and time.time() < deadline - 20:
        print("[Refine] Projected gradient descent on exact max‑conv")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Refine] GD improved to {val_gd:.6f}")

    if best_seq is not None and time.time() < deadline - 10:
        print("[Refine] Simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Refine] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic hill‑climb – cheap but often squeezes the last few
    # percent out of the objective.
    if best_seq is not None and time.time() < deadline - 2:
        print("[Fine] Hill‑climbing refinement")
        a_hc, val_hc = _hill_climb_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"[Fine] Hill‑climb improved to {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = float("inf")

    # Clip to allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation for reporting (uses the provided function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
