def pgd_optimize(a0, n, conv_fft_len, deadline, max_iters=2_000_000, step0=0.1, step_decay=0.9999):
    a = a0.copy()
    best_a = a.copy()
    best_val = objective(a)
    it=0
    while time.time()<deadline and it<max_iters:
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0*n*max_c
        if val<best_val:
            best_val=val; best_a=a.copy()
        # subgradient as before: active indices equal to max_c with tolerance.
        active = np.isclose(conv, max_c, atol=1e-12) # maybe rtol
        if not active.any():
            break
        w=zeros; w[active]=1/active.sum()
        grad = 2.0*cross_correlation(w, a, conv_fft_len) # note uses same fft len maybe fine
        # Gradient descent step opposite direction
        step = step0 / np.sqrt(it+1) * 1.0 # decaying
        a = a - step * grad
        # Ensure non-negative and projection to simplex
        a = np.maximum(a, 0)
        a = project_simplex(a)
        # Optional symmetrize:
        a = (a + a[::-1]) / 2
        it+=1
    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) with a low max‑convolution bound.

Algorithm (≈ 1000 s budget)
1. **Rich initialisation** – generate many diverse candidates:
   * the supplied warm start (if any)
   * uniform, linear‑ramp, triangular, and a dense grid of Beta(α,β) shapes
     (α,β drawn from a wide log‑uniform range, covering very “U‑shaped’’ cases)
   * Dirichlet draws with concentrations from 0.01 to 10.
   The best candidate (lowest exact objective) seeds the optimisation.

2. **Exponentiated‑gradient sub‑gradient phase** – mirror‑descent on the
   simplex using the exact sub‑gradient of `max(conv)`.  A large number of
   cheap iterations quickly drives the objective down.

3. **Adam on a smooth‑max surrogate** – improves the solution by
   optimising a log‑sum‑exp approximation of the maximum with an
   exponentially increasing temperature `lam`.

4. **Projected‑gradient descent** – a few thousand steps of classic
   gradient descent (opposite to the sub‑gradient) with Euclidean
   projection onto the simplex.  The iterate is symmetrised after each
   step (mirroring) because the objective is invariant under reversal.

5. **Local stochastic refinement** – tiny random mass transfers between
   two coordinates, kept if they improve the exact objective.

6. **Cyclic shift test** – try several cyclic shifts of the final vector
   (the linear convolution changes under a shift) and keep the best.

All phases respect the total time budget.  The returned list is a valid
non‑negative sequence; its quality is reported with `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) for a with Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimisation."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # sub‑gradient w.r.t. max(conv)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)        # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.03
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    step0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Projected‑gradient descent on the exact max‑convolution objective."""
    a = np.maximum(a0, 0.0)
    a /= max(a.sum(), 1e-12)

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[PGD it={it:07d}] new best {val:.6f}")

        # sub‑gradient (same as in the sub‑gradient phase)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient‑descent step (move opposite to the sub‑gradient)
        step = step0 / np.sqrt(it + 1.0)
        a = a - step * grad_a

        # keep feasible: non‑negative and on the simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # enforce symmetry (optional but stabilises the search)
        a = (a + a[::-1]) / 2.0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # safety margin to avoid missing the deadline
    deadline = time.time() + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0)  Collect a very diverse pool of initial vectors
    candidates = []

    # warm start (if supplied)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # simple linear ramp (increasing then decreasing – symmetric triangle)
    ramp = np.arange(1, n // 2 + 1)
    ramp = np.concatenate([ramp, ramp[::-1][: n - len(ramp)]])
    a = ramp.astype(np.float64)
    a = a / a.sum()
    candidates.append(a)

    # grid of symmetric Beta–shapes
    x = (np.arange(n) + 0.5) / n
    alphas = np.geomspace(0.05, 2.0, 20)          # explore very small as well
    for alpha in alphas:
        beta = alpha
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # random asymmetric Beta shapes
    for _ in range(30):
        alpha = np.exp(np.random.uniform(np.log(0.05), np.log(5.0)))
        beta = np.exp(np.random.uniform(np.log(0.05), np.log(5.0)))
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # random Dirichlet draws with a wide concentration range
    for _ in range(100):
        conc = np.exp(np.random.uniform(np.log(0.01), np.log(10.0)))
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ------------------------------------------------------------------
    # 1)  Pick the best among the cheap candidates (exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] candidate obj={val:.6f}")

    # ------------------------------------------------------------------
    # 2)  Long exponentiated‑gradient sub‑gradient optimisation
    sub_deadline = deadline - 200.0          # reserve ~200 s for later phases
    if sub_deadline > time.time():
        print("[Phase] Sub‑gradient optimisation")
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=5_000_000, eta0=2.0
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Subgrad] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3)  Adam fine‑tuning on smooth‑max
    adam_deadline = deadline - 100.0
    if adam_deadline > time.time():
        print("[Phase] Adam fine‑tuning")
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=400_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4)  Projected‑gradient descent (few hundred‑thousand cheap steps)
    pgd_deadline = deadline - 30.0
    if pgd_deadline > time.time():
        print("[Phase] Projected‑gradient descent")
        a_pgd, val_pgd = _pgd_optimize(
            best_a, n, conv_fft_len, corr_fft_len, pgd_deadline,
            max_iters=1_200_000, step0=0.3
        )
        if val_pgd < best_val:
            best_val, best_a = val_pgd, a_pgd.copy()
            print(f"[PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # 5)  Tiny stochastic local search (last few seconds)
    local_deadline = deadline - 5.0
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move a random fraction of the smaller of the two masses
        delta = min(best_a[i], best_a[j]) * np.random.rand() * 1e-2
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 6)  Try a few cyclic shifts (linear conv changes under a shift)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 7)  Final sanity checks, clipping and reporting
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)

    # official evaluation (used only for logging)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
