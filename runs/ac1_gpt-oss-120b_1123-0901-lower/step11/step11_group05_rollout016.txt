def _solve_lp_max_sum(a: np.ndarray, max_val: float):
    n = a.shape[0]
    b = cp.Variable(n)
    # Convolution a * b
    conv = cp.conv(a, b)  # will be length 2n-1
    constraints = [conv <= max_val, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if b.value is None:
        return None

```python
"""
Hybrid multi‑restart exponential‑gradient + Frank‑Wolfe LP refinement.

The algorithm:
1. Build a diverse pool of candidate step‑functions (uniform, warm‑start,
   symmetric and asymmetric beta densities, Dirichlet draws, exponential
   decays).  Keep the best few according to the exact max‑convolution
   objective.
2. Starting from the best candidate, run a long exponentiated‑gradient
   sub‑gradient phase (the same update rule as the reference code) to
   quickly drive the max‑convolution down.
3. Fine‑tune the result with a short Adam optimisation on a smooth‑max
   surrogate.
4. Perform a few Frank‑Wolfe style linear‑programming steps: given the
   current vector a, maximise Σ b_j under the constraint that the
   convolution a★b does not exceed the current maximal convolution M.
   The optimiser returns a direction b that is “anti‑correlated’’ with a;
   we mix a towards b (with a small step‑size) and re‑project onto the
   probability simplex.  This step is cheap (only the active bound
   indices are used) and further shrinks the max‑convolution.
5. Tiny stochastic local search and a handful of cyclic shifts are used
   as a final polishing stage.
6. The best sequence found before the deadline is returned as a Python
   list; it is guaranteed to be non‑negative and respects the limits of
   `evaluate_sequence`.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Utility functions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a with Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Optimisers
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimiser (exact max)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------
# Frank‑Wolfe LP refinement (active‑set version)
def _solve_lp_active(a: np.ndarray, max_val: float, conv_a: np.ndarray) -> np.ndarray | None:
    """
    Solve   maximize Σ b_j   s.t.   (a ★ b)_k ≤ max_val  for k in a small
    active set (indices where a ★ a is close to its maximum).
    Returns the optimal b as a dense NumPy array, or None if the solve fails.
    """
    n = a.shape[0]
    # choose active indices (those attaining the max up to a tiny tolerance)
    eps = max_val * 1e-6
    active = np.where(conv_a >= max_val - eps)[0]
    # limit number of active constraints to keep the LP tiny
    if active.size > 20:
        # keep the 20 largest values (they are already close to max)
        order = np.argsort(-conv_a[active])[:20]
        active = active[order]

    b = cp.Variable(n)
    constraints = [b >= 0]

    a_const = cp.Constant(a)

    for k in active:
        # build the linear expression for (a ★ b)_k
        # (a ★ b)_k = Σ_i a_{k-i} b_i over valid i
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # slice of a in reverse order
        a_slice = a_const[k - np.arange(i_start, i_end + 1)]
        b_slice = b[i_start : i_end + 1]
        constraints.append(cp.sum(cp.multiply(a_slice, b_slice)) <= max_val)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except cp.error.SolverError:
        return None
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)


def _frank_wolfe_refine(a: np.ndarray, n: int, conv_fft_len: int,
                       deadline: float, max_steps: int = 6) -> np.ndarray:
    """
    Apply a few Frank‑Wolfe style LP steps to lower the maximal convolution.
    Each step solves a tiny LP (active set only) and mixes the current vector
    towards the LP solution with a modest step‑size.
    """
    step_size = 0.2  # mixing coefficient
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        conv_a = _full_convolution(best_a, conv_fft_len)
        max_c = conv_a.max()
        b_opt = _solve_lp_active(best_a, max_c, conv_a)
        if b_opt is None:
            print("[FW] LP failed, aborting refinement.")
            break
        sum_b = b_opt.sum()
        if sum_b <= 0:
            print("[FW] Obtained zero‑mass b, aborting.")
            break
        g = b_opt / sum_b
        # convex combination
        new_a = (1.0 - step_size) * best_a + step_size * g
        new_a = _project_simplex(new_a)
        new_val = _objective(new_a, n, conv_fft_len)
        print(f"[FW step={step+1}] objective {new_val:.6f}")
        if new_val < best_val:
            best_val, best_a = new_val, new_a.copy()
            # optionally reduce step size for later passes
            step_size = max(0.05, step_size * 0.9)
        else:
            # no improvement – stop early
            break

    return best_a

    # keep only top_k

    print(f"[Init] best objective {best_val:.6f}")

            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=6_000_000, eta0=1.8
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

"""
Hybrid multi‑restart exponential‑gradient + Frank‑Wolfe LP refinement.

The algorithm:
1.  Build a diverse pool of candidate step‑functions (uniform, warm‑start,
    symmetric and asymmetric beta densities, Dirichlet draws, exponential
    decays).  Keep the best few according to the exact max‑convolution
    objective.
2.  Starting from the best candidate, run a long exponentiated‑gradient
    sub‑gradient phase (the same update rule as the reference code) to
    quickly drive the max‑convolution down.
3.  Fine‑tune the result with a short Adam optimisation on a smooth‑max
    surrogate.
4.  Perform a few Frank‑Wolfe style linear‑programming steps: given the
    current vector a, maximise Σ b_j under the constraint that the
    convolution a★b does not exceed the current maximal convolution M.
    Only the active (tight) convolution indices are used, keeping the LP
    tiny.  The optimiser returns a direction b that is “anti‑correlated’’
    with a; we mix a towards b (small step‑size) and re‑project onto the
    probability simplex.  This step further shrinks the max‑convolution.
5.  Tiny stochastic local search and a handful of cyclic shifts are used
    as a final polishing stage.
6.  The best sequence found before the deadline is returned as a Python
    list; it is guaranteed to be non‑negative and respects the limits of
    `evaluate_sequence`.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Utility functions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a with Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Optimisers
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimiser (exact max)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule via soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------
# Frank‑Wolfe LP refinement (active‑set version)
def _solve_lp_active(a: np.ndarray, max_val: float, conv_a: np.ndarray) -> np.ndarray | None:
    """
    Solve   maximize Σ b_j   s.t.   (a ★ b)_k ≤ max_val  for a small
    active set of indices (those where a ★ a is close to its maximum).
    Returns the optimal b as a dense NumPy array, or None on failure.
    """
    n = a.shape[0]
    eps = max_val * 1e-6
    active = np.where(conv_a >= max_val - eps)[0]

    # keep at most 20 active constraints to stay fast
    if active.size > 20:
        order = np.argsort(-conv_a[active])[:20]
        active = active[order]

    b = cp.Variable(n)
    constraints = [b >= 0]
    a_const = cp.Constant(a)

    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        a_slice = a_const[k - idx]               # constant coefficients
        b_slice = b[i_start : i_end + 1]
        constraints.append(cp.sum(cp.multiply(a_slice, b_slice)) <= max_val)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except cp.error.SolverError:
        return None
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)


def _frank_wolfe_refine(a: np.ndarray, n: int, conv_fft_len: int,
                       deadline: float, max_steps: int = 6) -> np.ndarray:
    """Apply a few Frank‑Wolfe style LP steps to lower the maximal convolution."""
    step_size = 0.2
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        conv_a = _full_convolution(best_a, conv_fft_len)
        max_c = conv_a.max()
        b_opt = _solve_lp_active(best_a, max_c, conv_a)
        if b_opt is None:
            print("[FW] LP failed – stopping refinement.")
            break
        sum_b = b_opt.sum()
        if sum_b <= 0:
            print("[FW] Obtained zero‑mass b – stopping.")
            break
        g = b_opt / sum_b
        new_a = (1.0 - step_size) * best_a + step_size * g
        new_a = _project_simplex(new_a)
        new_val = _objective(new_a, n, conv_fft_len)
        print(f"[FW step={step+1}] objective {new_val:.6f}")
        if new_val < best_val:
            best_val, best_a = new_val, new_a.copy()
            step_size = max(0.05, step_size * 0.9)   # slowly shrink step‑size
        else:
            break
    return best_a
