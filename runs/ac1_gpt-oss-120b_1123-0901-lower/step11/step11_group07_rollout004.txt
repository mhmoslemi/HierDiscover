def soft_expgrad_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                          eta0=0.8, T0=0.1, Tf=1e-6):
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0*n*conv.max()
    iter = 0
    while time.time() < deadline:
        # compute temperature as decay linear
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(Tf, T0 * (1.0 - frac))
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # weight
        w = np.exp((conv - max_c) / T)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(iter + 1.0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        # optional clipping small values to zero to improve numeric
        if iter % 1000 == 0:
            conv = _full_convolution(a, conv_fft_len)
            val = 2.0*n*conv.max()
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[softExpGrad it={iter}] new best {val:.6f}")
        iter += 1
    return best_a, best_val
```

"""
Propose a low‑value step‑function using a two‑stage optimizer:

1.  Adam on a smooth‑max surrogate (log‑sum‑exp) – fast global descent.
2.  Temperature‑annealed exponentiated‑gradient (soft‑max of the
    convolution) – precise fine‑tuning of the exact ‑max objective.

Both stages work on the probability simplex (∑a_i = 1) and keep the
vector non‑negative.  The routine starts from the previous best
sequence (global ``height_sequence_1``) if it exists, symmetrises it
to encourage low autocorrelation, and spends the first ~25 % of the
budget on Adam and the rest on the annealed exp‑gradient.  The best
sequence found is clipped to the evaluator’s limits and returned as a
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (FFT‑based convolution / correlation)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns an array of length len(a) aligned with the original indices.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr_full = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr_full[start:start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimiser 1 – Adam on a smooth‑max surrogate

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Adam with a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.5                # start temperature for smooth‑max
    lam_factor = 1.02        # increase slowly for stability

    best_a = a0.copy()
    # exact value (sum=1 ⇒ denominator =1)
    best_val = 2.0 * n * _full_convolution_fft(a0, conv_fft_len).max()

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation_fft(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # anneal temperature
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 200_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Optimiser 2 – Soft, temperature‑annealed exponentiated gradient

def _soft_expgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.8,
    T_start: float = 0.05,
    T_end: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) with a gradually cooling
    soft‑max over the convolution entries.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution_fft(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    start_time = time.time()
    it = 0
    while time.time() < deadline:
        # linear cooling of the temperature
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        T = max(T_end, T_start * (1.0 - frac))

        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()

        # soft weights (more than just the active set while T>0)
        w = np.exp((conv - max_c) / T)        # ≤1, zero for far‑away terms
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _cross_correlation_fft(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)           # multiplicative update
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        if it % 2000 == 0:
            conv = _full_convolution_fft(a, conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[SoftExpGrad it={it}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine spends ~25 % of the time on Adam (smooth‑max) and the
    remainder on a temperature‑annealed exponentiated‑gradient phase.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # a small safety margin – the caller will kill the process at the deadline
    deadline = time.time() + budget_s - 0.5

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run produced a good sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] > n:
            a0 = a0[:n]
        elif a0.shape[0] < n:
            a0 = np.pad(a0, (0, n - a0.shape[0]))
        a0 = np.clip(a0, 0.0, 1000.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()
    else:
        a0 = np.random.dirichlet(np.ones(n))

    # enforce symmetry – most low‑autocorrelation solutions are palindromes
    a0 = (a0 + a0[::-1]) / 2.0
    a0 = a0 / a0.sum()

    # initial objective value
    conv0 = _full_convolution_fft(a0, conv_fft_len)
    best_val = 2.0 * n * conv0.max()
    best_seq = a0.copy()
    print(f"Start value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – Adam on smooth‑max (takes about 25 % of the total budget)
    time_now = time.time()
    remaining = deadline - time_now
    adam_time = min(200.0, remaining * 0.25)
    if adam_time > 5.0:
        adam_deadline = time.time() + adam_time
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=3_000_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – soft, annealed exponentiated‑gradient (uses the rest)
    if time.time() < deadline:
        a_exp, val_exp = _soft_expgrad_optimize(
            best_seq, n,
            conv_fft_len, corr_fft_len,
            deadline,
            eta0=0.9,
            T_start=0.05,
            T_end=1e-7,
        )
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[SoftExpGrad] final best {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Final sanity: clip and return a plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
