"""
Hybrid optimizer for the non‑negative step‑function problem (improved).

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)^2)
with a_i ≥ 0.
The algorithm is a blend of three ideas:

1. **Smooth‑max Adam** – optimisation of a soft‑max surrogate of the
   maximum convolution.  The temperature (λ) grows very slowly,
   keeping the surrogate smooth for a long time.  A convex combination
   of the soft‑max weight and a uniform weight is used, which
   explicitly flattens the whole convolution during the early stage.

2. **Exponentiated‑gradient sub‑gradient** – after the smooth phase
   we switch to an exact sub‑gradient on the true max‑convolution.
   The step‑size schedule has been tuned (η₀=0.5) and the update is
   performed on the simplex via a multiplicative mirror step.

3. **Final simulated‑annealing pass** – a short but aggressive annealing
   that moves a random fraction of mass between two coordinates.
   This is useful for escaping tiny plateaux and polishing the final
   solution.

All phases respect the global deadline (default 1000 s) and the best
vector seen so far is always returned.  The routine works with either
n=4096 (default) or n=8192 when enough time budget is available, the
larger size giving a better asymptotic lower bound (≈1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max + flattening surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                               # modest learning‑rate
    lam = 0.01                               # start temperature
    lam_factor = 1.005                       # very slow increase
    max_lam = 300.0

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- current point on the simplex ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam] it={it:06d} val={val:.6f}")

        # ---- smooth‑max weights (soft‑max over λ·conv) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # flattening component – blend with a uniform distribution
        t = max(0.0, 1.0 - it / max_iters)          # linearly decay
        uniform = np.full_like(w, 1.0 / w.size)
        w = (1.0 - t) * w + t * uniform            # early iterations are very flat

        # ---- gradient of the surrogate w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, max_lam)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Keeps the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad] it={it:07d} val={val:.6f}")

        # active set of the current maximum (could be several indices)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size (mirror‑descent on simplex)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # project back onto the simplex (avoid underflow)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of diverse start vectors."""
    pool: list[np.ndarray] = []

    if warm is not None:
        pool.append(warm.copy())
        # cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both sides)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid (always a good baseline)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    pool.append(s)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Sparse spike (≈√n spikes)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        pool.append(a)

    return pool


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    The routine respects the supplied time budget and always returns
    the best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose the problem size.
    # Larger n gives a better theoretical floor (≈1) but costs more time.
    # Use n=8192 when we have enough budget, otherwise fall back to 4096.
    if budget_s >= 800:
        n = 8192
    else:
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)   # length for linear convolution
    corr_fft_len = _next_pow2(3 * n - 2)   # length needed for cross‑correlation

    # ------------------------------------------------------------------
    # Warm start from previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build an initial pool and keep the best seed (exact evaluation)
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() > deadline:
            break
        if idx % 10 == 0:
            print(f"[Init] candidate {idx:02d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – we repeat a few restarts until time runs out.
    restart = 0
    while time.time() < deadline - 5.0:
        # random initialisation (Dirichlet) or a noisy perturbation of the current best
        if restart % 3 == 0:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            noise = 0.02 * np.random.randn(n)
            a0 = best_seq + noise
            a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

        # ---- Phase I : Adam on smooth‑max + flattening ----
        sub_dead = min(deadline, time.time() + min(250.0, deadline - time.time() - 30.0))
        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len, corr_fft_len, sub_dead)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Restart {restart}] Adam improved to {val_adam:.6f}")

        # ---- Phase II : Exact sub‑gradient (exponentiated gradient) ----
        if time.time() < deadline - 5.0:
            a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                   conv_fft_len, corr_fft_len,
                                                   deadline, eta0=0.5)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Restart {restart}] Sub‑gradient improved to {val_sub:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Final short simulated‑annealing refinement (if time permits)
    if best_seq is not None and time.time() < deadline - 2.0:
        print("[Final] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len,
                                          deadline, max_moves=150_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Final] annealing improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety net – ensure we always return a valid sequence
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
