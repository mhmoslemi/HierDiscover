"""
Hybrid random‑structure + gradient refinement optimizer.

The search proceeds in three stages:

1. **Diverse random families** – generates candidates from many analytic
   families (Dirichlet, exponential, power‑law, beta, log‑normal,
   window‑segments, uniform).  Each candidate is normalised to sum 1 and
   evaluated with a fast FFT‑based objective
   `2·n·max(conv(a,a))`.  The best of these cheap evaluations supplies a
   strong starting point.

2. **Upscaling** – if time permits the best small‑size vector is linearly
   interpolated to a larger size (doubling the length) to obtain a finer
   discretisation of the shape.

3. **Refinement** – the current best vector is improved by three
   optimisation phases:
   * L2‑flattening Adam ( minimise ‖conv(a,a)‑uniform‖² ),
   * smooth‑max Adam (optimises a log‑sum‑exp surrogate of the max),
   * exact‑max exponentiated‑gradient sub‑gradient descent,
   finally a projected‑gradient step on the exact max.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so the scaling invariance of the original objective is respected.
The routine respects the given time budget and returns the best sequence
found (as a plain list of floats) before the deadline expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Candidate factories ----------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * i)
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    # discretise the beta density on (0,1)
    x = (np.arange(1, n + 1) - 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return _make_uniform(n)
    return a / s


def _make_lognormal(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.exp(- (np.log(i) - mu) ** 2 / (2 * sigma ** 2))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return _make_uniform(n)
    return a / s


def _make_window_segment(
    n: int, kind: str, length: int, offset: int
) -> np.ndarray:
    """Place a window of given type in the array (zeros elsewhere)."""
    if kind == "hann":
        w = np.hanning(length)
    elif kind == "hamming":
        w = np.hamming(length)
    elif kind == "blackman":
        w = np.blackman(length)
    elif kind == "tri":
        idx = np.arange(1, length + 1)
        w = idx * (length - idx + 1)
    else:  # fallback to a rectangular block
        w = np.ones(length, dtype=np.float64)

    a = np.zeros(n, dtype=np.float64)
    a[offset : offset + length] = w
    s = a.sum()
    if s <= 0:
        return _make_uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Fast objective (sum = 1) -----------------------------------------------
def _fast_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max -------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 1.0          # start with a moderately sharp surrogate
    lam_factor = 1.5

    best_a = a0.copy()
    best_val = _fast_objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (stable exponentiation)
        diff = conv - max_c
        w = np.exp(np.clip(lam * diff, -50, 50))
        w /= w.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e6)

        if it % 50000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val

# Adam optimiser – L2 flattening ----------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = _fast_objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser – exponentiated‑gradient ------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _fast_objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Projected gradient descent on exact max -------------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = float(2.0 * n * conv.max())
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# Main search routine ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine combines random analytic families, up‑sampling and several
    gradient‑based refinements within the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Small problem size – fast random exploration
    n_small = 4096
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    families = ["uniform", "dirichlet", "exp", "power", "beta", "lognorm", "window"]

    # Random search – stop roughly halfway through the budget to keep time
    # for refinement phases.
    while time.time() < deadline - 30.0:
        fam = np.random.choice(families)

        if fam == "uniform":
            a = _make_uniform(n_small)

        elif fam == "dirichlet":
            # sample a wide range of concentration parameters
            alpha = np.exp(np.random.uniform(-4.0, 0.5))
            a = _make_dirichlet(n_small, alpha)

        elif fam == "exp":
            gamma = np.random.uniform(5e-4, 0.02)
            a = _make_exponential(n_small, gamma)

        elif fam == "power":
            expo = -np.random.uniform(0.0, 2.5)   # negative exponent ⇒ decreasing
            a = _make_powerlaw(n_small, expo)

        elif fam == "beta":
            alpha = np.random.uniform(0.1, 5.0)
            beta = np.random.uniform(0.1, 5.0)
            a = _make_beta_distribution(n_small, alpha, beta)

        elif fam == "lognorm":
            mu = np.random.uniform(0.0, np.log(n_small))
            sigma = np.random.uniform(0.2, 1.2)
            a = _make_lognormal(n_small, mu, sigma)

        elif fam == "window":
            kind = np.random.choice(["hann", "hamming", "blackman", "tri"])
            length = np.random.randint(int(0.2 * n_small), int(0.8 * n_small) + 1)
            offset = np.random.randint(0, n_small - length + 1)
            a = _make_window_segment(n_small, kind, length, offset)

        else:
            continue

        # Normalise and evaluate
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s

        val = _fast_objective(a, n_small, conv_fft_len_small)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random {fam}] improved to {val:.6f}")

        # early exit if we already beat the target
        if best_val < 1.5035:
            break

    if best_seq is None:
        best_seq = _make_uniform(n_small)
        best_val = _fast_objective(best_seq, n_small, conv_fft_len_small)

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger problem (more degrees of freedom)
    a_cur = best_seq
    n_cur = n_small
    conv_fft_len = conv_fft_len_small
    corr_fft_len = corr_fft_len_small

    if time.time() < deadline - 20.0:
        n_big = n_cur * 2
        x_old = np.arange(n_cur)
        x_new = np.linspace(0, n_cur - 1, n_big)
        a_big = np.interp(x_new, x_old, a_cur)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        a_cur = a_big
        n_cur = n_big
        conv_fft_len = _next_pow2(2 * n_cur - 1)
        corr_fft_len = _next_pow2(3 * n_cur - 2)
        print(f"[Upsample] increased length to {n_cur}")

    # ------------------------------------------------------------------
    # Phase 1 – L2 flattening (quick 20‑s window)
    if time.time() < deadline - 10.0:
        target = np.full(2 * n_cur - 1, 1.0 / (2 * n_cur - 1), dtype=np.float64)
        phase_dead = min(deadline, time.time() + 20.0)
        a_l2, val_l2 = _adam_optimize_l2(
            a_cur,
            n_cur,
            conv_fft_len,
            corr_fft_len,
            target,
            phase_dead,
            max_iters=200_000,
            lr=0.05,
        )
        if val_l2 < best_val:
            best_val = val_l2
            a_cur = a_l2.copy()
            print(f"[L2 flatten] improved to {val_l2:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max Adam (fine‑tuning)
    if time.time() < deadline - 8.0:
        phase_dead = min(deadline, time.time() + 20.0)
        a_adam, val_adam = _adam_optimize(
            a_cur,
            n_cur,
            conv_fft_len,
            corr_fft_len,
            phase_dead,
            max_iters=300_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            a_cur = a_adam.copy()
            print(f"[Adam fine] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact‑max exponentiated gradient
    if time.time() < deadline - 6.0:
        a_exp, val_exp = _subgradient_optimize_expgrad(
            a_cur,
            n_cur,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            eta0=0.8,
        )
        if val_exp < best_val:
            best_val = val_exp
            a_cur = a_exp.copy()
            print(f"[ExpGrad] improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected gradient descent (optional)
    if time.time() < deadline - 4.0:
        a_gd, val_gd = _max_conv_descent(
            a_cur,
            n_cur,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            a_cur = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and reporting
    a_cur = np.clip(a_cur, 0.0, 1000.0)

    try:
        final_val = evaluate_sequence(a_cur.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(a_cur)})")
    except Exception as exc:
        print(f"Evaluation error: {exc}")
        print(f"Best internal value {best_val:.6f}")

    return a_cur.tolist()
