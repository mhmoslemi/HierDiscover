"""
Hybrid optimiser for the step‑function attack (improved).

The optimiser works on the probability simplex (∑a_i = 1, a_i ≥ 0),
which is safe because the evaluation function is homogeneous.
The pipeline is a single‑run version that starts from the best known
construction (global variable ``height_sequence_1``) and then applies,
in sequence, the following refinements, each stopping only when the
overall deadline (≈1000 s) is reached:

1.  Spectral flattening – force the FFT magnitudes (except the DC term)
    to be (almost) equal while staying on the simplex.
2.  Adam optimisation on a smooth‑max surrogate of the max‑convolution.
3.  Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
4.  Linear‑programming refinement – maximise Σb_j under the
    constraints (a★b)_k ≤ M (M = maxᵏ(a★a)_k) and mix the resulting
    b with the current a.
5.  Projected gradient descent directly on the exact max‑convolution.
6.  Simulated‑annealing polish.

All stages respect the shared deadline, keep the best candidate seen,
and finally return it as a plain ``list[float]``.  The implementation
uses only NumPy, SciPy and the provided ``evaluate_sequence`` for the
final check.  The algorithm is deliberately longer‑running than the
previous baseline, allowing the optimiser to explore more iterations
and thereby push the bound below the target 1.503.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _spectral_flatten_optimize(a0: np.ndarray,
                               n: int,
                               conv_fft_len: int,
                               deadline: float,
                               max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Iteratively enforce (almost) constant FFT magnitude."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # FFT of current a
        A = np.fft.fft(a)
        energy = np.dot(a, a)                     # Σ a_i²

        # keep DC (k=0) unchanged, equalise all other magnitudes
        if n > 1:
            # total energy = (|A0|² + Σ_{k>0}|A_k|²) / n  =>  Σ_{k>0}|A_k|² = n*energy - |A0|²
            rest_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rest_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # preserve DC to keep the sum unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))
        # inverse transform
        a_new = np.real(np.fft.ifft(A_new))
        # enforce non‑negativity + simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            # fall back to a random Dirichlet if numerical trouble
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s
        it += 1
    return best_a, best_val


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)      # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over convolution values)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient (exponentiated‑gradient) phase."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _lp_improve(a0: np.ndarray,
                n: int,
                conv_fft_len: int,
                deadline: float,
                max_iter: int = 20,
                tau: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current sequence f we maximise Σb subject to (f★b)_k ≤ M,
    where M = max (f★f)_k.  The solution b is normalised and mixed with f.
    """
    a = a0.copy()
    best_a = a.copy()
    conv_a = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv_a.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active indices (allow a tiny tolerance)
        eps = 1e-12
        active_idx = np.where(conv_a >= M - eps)[0]

        # augment with a few well‑spread indices to keep LP well‑posed
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active_idx = np.unique(np.concatenate((active_idx, extra)))

        # build sparse constraint matrix A (|active| × n)
        rows, cols, data = [], [], []
        for row, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))

        # Linear programme: maximise Σ b ⇔ minimise -Σ b
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active_idx))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.time())},
        )

        if not lp_res.success:
            print(f"[LP it={it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # convex combination
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        val_new = 2.0 * n * np.max(_full_convolution(a_new, conv_fft_len))
        print(f"[LP it={it}] value {val_new:.6f} (M={M:.6f})")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            # shrink mixing factor if no progress
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Return a non‑negative step function (list of floats) with low bound."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # spare margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from the best known construction, if available.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    if warm is None:
        # fallback: random Dirichlet
        warm = np.random.dirichlet(np.ones(n))

    best_seq = warm.copy()
    conv = _full_convolution(best_seq, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    print(f"[Init] starting value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – aggressive spectral flattening
    a_spec, val_spec = _spectral_flatten_optimize(
        best_seq, n, conv_fft_len, deadline, max_iters=10_000_000
    )
    if val_spec < best_val:
        best_val, best_seq = val_spec, a_spec.copy()
        print(f"[Phase I] spectral flatten improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase II – Adam on smooth‑max
    a_adam, val_adam = _adam_optimize(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=10_000_000
    )
    if val_adam < best_val:
        best_val, best_seq = val_adam, a_adam.copy()
        print(f"[Phase II] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase III – exact sub‑gradient (exponentiated‑gradient)
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_seq, n, conv_fft_len, corr_fft_len, deadline,
        max_iters=10_000_000, eta0=0.8
    )
    if val_sub < best_val:
        best_val, best_seq = val_sub, a_sub.copy()
        print(f"[Phase III] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – linear‑programming refinement (more iterations)
    a_lp, val_lp = _lp_improve(
        best_seq, n, conv_fft_len, deadline, max_iter=20, tau=0.5
    )
    if val_lp < best_val:
        best_val, best_seq = val_lp, a_lp.copy()
        print(f"[Phase IV] LP improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase V – projected gradient descent on exact max‑conv
    a_gd, val_gd = _max_conv_descent(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=10_000_000
    )
    if val_gd < best_val:
        best_val, best_seq = val_gd, a_gd.copy()
        print(f"[Phase V] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase VI – final simulated‑annealing polishing
    a_ann, val_ann = _anneal_optimize(
        best_seq, n, conv_fft_len, deadline, max_moves=10_000_000
    )
    if val_ann < best_val:
        best_val, best_seq = val_ann, a_ann.copy()
        print(f"[Phase VI] Annealing improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Guarantee non‑empty result and enforce evaluator bounds.
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
