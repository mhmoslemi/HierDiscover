            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # Build a large pool of analytic candidates with varying parameters
    pool = []

        pool.append(warm.copy())

    for alpha in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):

    for gamma in (0.2, 0.5, 1.0, 2.0):

"""
Improved optimiser for the step‑function evaluation problem.

The programme builds a diverse pool of analytic candidates (uniform,
exponential, power‑law, beta‑U‑shapes, arcsine‑like and square‑root‑tails),
quickly discards the worse ones and then subjects the current best to a
multi‑phase continuous optimisation:

1️⃣  L2‑flattening (Adam) – drives the whole convolution towards the uniform
    lower‑bound 1/(2 n‑1) using a smooth‑L2 loss.
2️⃣  Smooth‑max (Adam) – minimises the true max‑convolution via a
    log‑sum‑exp surrogate with a temperature that is increased
    gradually.
3️⃣  Exact‑max sub‑gradient (exponentiated‑gradient) – performs many
    multiplicative updates on the simplex using the true sub‑gradient.
4️⃣  Projected gradient descent on the exact max‑convolution.
5️⃣  Simulated‑annealing fine‑tuning.

Each stage runs for a time slice that respects the global deadline
(≈ 1000 s).  The implementation works entirely with NumPy, projects on the
probability simplex, and uses FFT‑based convolutions for speed.
The final sequence is clipped to the allowed range and returned as a plain
Python list for the external `evaluate_sequence` checker.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric beta‑U‑shape (α,β) on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(n - 1, 1)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1) * np.power(1 - x, beta - 1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a /= a.sum()
    return a


def _make_arcsine(n, eps):
    """Discrete analogue of the arcsine law: a_i ∝ 1/√(x(1‑x))."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(n - 1, 1)
    a = 1.0 / np.sqrt(np.maximum(x * (1 - x), eps))
    a /= a.sum()
    return a


def _make_sqrt_shape(n):
    """Symmetric 1/√i‑tails: a_i ∝ 1/√i + 1/√(n‑i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    rev = i[::-1]
    a = 1.0 / np.sqrt(i) + 1.0 / np.sqrt(rev)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max (log‑sum‑exp surrogate) ------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser – L2 flattening ---------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 400_000,
    lr: float = 0.07,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        # gradient of ||conv-target||_2^2 w.r.t. a
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser – exponentiated gradient -----------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on exact max -------------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing ---------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Helper to evaluate objective without the security wrapper (faster)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Assumes sum(a) = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ budget_s seconds. Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Problem size (feel free to increase)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start if a previous best exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Build a rich pool of analytic initial vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())

    # Dirichlet families with many concentration parameters
    for alpha in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))

    # Simple analytic families
    pool.append(_make_uniform(n))
    for gamma in (0.2, 0.5, 1.0, 2.0):
        pool.append(_make_exponential(n, gamma))
    for expo in (-0.5, -1.0, -1.2, -1.5, -2.0):
        pool.append(_make_powerlaw(n, expo))
    pool.append(_make_triangle(n))

    # Symmetric beta‑U‑shapes (α=β<1)
    for alpha in (0.2, 0.3, 0.4, 0.5):
        pool.append(_make_beta(n, alpha, alpha))

    # Arcsine‑like shapes
    for eps in (1e-6, 1e-5, 1e-4):
        pool.append(_make_arcsine(n, eps))

    # Square‑root tail shape
    pool.append(_make_sqrt_shape(n))

    # --------------------------------------------------------------
    # Quick evaluation of raw candidates, retain the best few
    best_seq = None
    best_val = float("inf")
    for idx, a0 in enumerate(pool):
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        val = _objective(a0, n, conv_fft_len)
        print(f"[Init {idx:02d}] obj={val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a0.copy()

    print(f"[Init] Best initial objective {best_val:.6f}")

    # --------------------------------------------------------------
    # Uniform target for the L2‑flattening phase
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # --------------------------------------------------------------
    # Phase 1 – L2 flattening (Adam) – longer than before
    if time.time() < deadline - 30:
        phase_dead = min(deadline, time.time() + 30.0)
        best_seq, val_l2 = _adam_optimize_l2(
            best_seq, n, conv_fft_len, corr_fft_len,
            target, phase_dead, max_iters=400_000, lr=0.07
        )
        if val_l2 < best_val:
            best_val, best_seq = val_l2, best_seq.copy()
            print(f"[Refine] L2‑Adam improved to {val_l2:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Smooth‑max Adam (log‑sum‑exp surrogate)
    if time.time() < deadline - 20:
        phase_dead = min(deadline, time.time() + 20.0)
        best_seq, val_smooth = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            phase_dead, max_iters=500_000
        )
        if val_smooth < best_val:
            best_val, best_seq = val_smooth, best_seq.copy()
            print(f"[Refine] Smooth‑max Adam improved to {val_smooth:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Exact‑max sub‑gradient (exponentiated‑gradient)
    if time.time() < deadline - 15:
        best_seq, val_exp = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=600_000, eta0=0.8
        )
        if val_exp < best_val:
            best_val, best_seq = val_exp, best_seq.copy()
            print(f"[Refine] ExpGrad improved to {val_exp:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Projected gradient descent on the exact max‑convolution
    if time.time() < deadline - 10:
        best_seq, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=300_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, best_seq.copy()
            print(f"[Refine] GD improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Simulated‑annealing fine‑tuning
    if time.time() < deadline - 5:
        best_seq, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, best_seq.copy()
            print(f"[Refine] Anneal improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Safety net – never return an empty sequence
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the allowed numeric range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation through the official checker (for reporting)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Internal best value {best_val:.6f}")

    return best_seq.tolist()
