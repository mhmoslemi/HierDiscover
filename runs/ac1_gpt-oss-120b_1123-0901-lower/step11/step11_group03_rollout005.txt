def _subgradient_optimize_improved(a0, n, conv_fft_len, corr_fft_len, deadline, eta0=1.5):
    a = np.maximum(a0, 0.0)
    a = a / max(a.sum(), 1e-12)
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it=0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it%5000==0:
                print(f"[Subgrad it={it}] new best {val:.6f}")
        # compute subgradient
        active = np.isclose(conv, max_c, atol=1e-15)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # to avoid underflow, renorm
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        it += 1
        if it % 1000 == 0 and time.time() >= deadline:
            break
    return best_a, best_val

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

"""
Hybrid multi‑stage optimizer for the non‑negative step‑function problem.

Goal
----
Minimise   2·n·max(conv(a,a)) / (sum(a)^2)   with a_i ≥ 0.
Because the objective is homogeneous of degree 0 we keep the vector on the
probability simplex Σ a_i = 1 throughout the whole run.

Strategy
--------
1. **Diverse seed generation** – a small pool of candidates (uniform, Dirichlet,
   exponential tails, double‑Gaussian spikes, a sinusoid, a Beta(½,½) “arcsine”
   shape and a few sparse spike vectors).  The best seed (by the exact objective)
   becomes the starting point.

2. **Adam on a smooth‑max surrogate** – we optimise a soft‑max version of the
   max‑convolution.  The temperature λ starts very low (≈0.02) and is raised
   extremely slowly (factor 1.002 per iteration) so that the surrogate stays
   smooth for a long time.  A standard Adam update on the logits (soft‑max
   parametrisation) is used; the best exact objective encountered is kept.

3. **Exponentiated‑gradient sub‑gradient phase** – after the surrogate has
   converged we switch to the true non‑smooth objective.
   The active set (indices where the convolution attains its maximum) is used
   to form a sub‑gradient and an exponentiated‑gradient (mirror‑descent) update
   on the simplex with a step‑size η_t = η₀ / √t (η₀≈1.5).  The best vector is
   retained.

4. **Focused local‑search (pairwise mass moves)** – a cheap stochastic
   hill‑climber that repeatedly moves a random fraction of mass from one
   coordinate to another, accepting only improving moves (a tiny simulated‑annealing
   component can be added if time permits).  This often shaves the last few
   thousandths off the score.

All phases respect the global deadline (default 1000 s) and always return the
best sequence discovered.  The implementation works for any length `n` (here
`n = 4096` gives a good trade‑off between quality and speed) and uses only
NumPy and the provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Returns the best vector (on the simplex) and the exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.045                         # modest base learning rate
    lam = 0.02                         # start temperature
    lam_factor = 1.002                 # very slow increase

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution (used for early stopping & logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – approach the true max very slowly
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 120_000 == 0 and it > 0:
            lr *= 0.92

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent on the exact max‑convolution.
    Runs until the deadline and keeps the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set for the current maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max‑convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size (mirror‑descent on simplex)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * grad_a)

        # renormalise to the simplex, protecting against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        # cheap checkpoint – break if we are out of time
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_search_optimize(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           deadline: float,
                           max_moves: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing local search on the simplex.
    Moves a random fraction of mass between two coordinates and keeps
    the improvement.  Very cheap but can trim the final objective a bit.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = rng.integers(0, n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(0, n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 40 %) from i to j
        frac = rng.random() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
            if it % 5_000 == 0:
                print(f"[Local it={it:06d}] new best {val2:.6f}")

        it += 1

    return best_a, best_val

def _arcsine_distribution(n: int) -> np.ndarray:
    """
    Beta(½, ½) distribution on {0,…,n‑1} (the discrete arcsine law).
    a_i ∝ 1 / sqrt(i+1)·1 / sqrt(n‑i) .
    """
    i = np.arange(1, n + 1, dtype=np.float64)           # 1 … n
    shape = 1.0 / np.sqrt(i * (n + 1 - i))
    shape /= shape.sum()
    return shape


def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    pool = []

    # warm start (if supplied)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform
    pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws with several concentration levels
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both sides)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law (triangular product) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian spikes
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Arcsine (Beta(½,½)) distribution
    pool.append(_arcsine_distribution(n))

    # A few sparse spike candidates (≈√n spikes)
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    Runs for at most `budget_s` seconds and always returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more flexibility but costs more time.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)        # 8192
    corr_fft_len = _next_pow2(3 * n - 2)        # ≥ 12288

    # ------------------------------------------------------------------
    # Warm start from the previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build the pool of initial candidates and pick the best one by exact eval
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        # fast exact objective (sum=1 → 2·n·max(conv))
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() > deadline:
            break
        if idx % 10 == 0:
            print(f"[Init] candidate {idx} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – Adam optimisation (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase I] Adam optimisation started")
        sub_dead = deadline - 5.0
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=500_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – exponentiated‑gradient sub‑gradient optimisation
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase II] Sub‑gradient optimisation started")
        sub_dead = deadline - 5.0
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            eta0=1.5
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase III – focused local search
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase III] Local search refinement")
        a_loc, val_loc = _local_search_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=30_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase III] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the validator’s allowed range and final normalisation
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"Final best objective {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
