def subgradient_steps(a0, steps, conv_fft_len, corr_fft_len):
    a = a0.copy()
    a = np.maximum(a, 0)
    total = a.sum()
    if total < 1e-12:
        a = np.random.dirichlet(np.ones(len(a)))
        total = a.sum()
    a = a / total
    for it in range(steps):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set indices
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = 0.2 / np.sqrt(it + 1.0)   # step size
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum < 1e-12 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(len(a)))
            a_sum = a.sum()
        a = a / a_sum
    return a
```

def hill_climb_optimize(a0, n, conv_fft_len, deadline, max_moves=200000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val
    it = 0
    while it < max_moves and time.time() < deadline:
        # temperature schedule
        T = max(0.001, (cur_val - best_val) * 0.1) # maybe linear
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move fraction
        frac = np.random.rand() * 0.5
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < cur_val or (np.random.rand() < np.exp(-(val2 - cur_val) / T)):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
        it += 1
    return best_a, best_val
```

"""
Hybrid optimizer for the non‑negative step‑function problem.

Goal: minimise   2·n·max(conv(a,a)) / (sum(a)²)   with a_i ≥ 0.
The value is scale‑invariant, therefore the search is performed on the
probability simplex (∑a_i = 1).  The optimiser consists of the following
stages, each respecting the global time budget (default 1000 s).

1. **Initial pool** – a few diverse seeds (warm start, Dirichlet draws,
   sparse spike vectors).  The best seed according to the exact objective
   is kept.

2. **Adam on a smooth‑max surrogate** – a long Adam run where the
   non‑differentiable max is approximated by a soft‑max.  The temperature
   λ is increased slowly, allowing the optimiser to approach the true
   objective.

3. **Exponentiated‑gradient (mirror‑descent) sub‑gradient optimisation**.
   This directly minimises the exact max‑convolution and runs for the
   remaining time.

4. **Hill‑climbing refinement** – a simple simulated‑annealing style
   local search that repeatedly transfers a random fraction of mass
   between two coordinates.  This phase can often shave off the last
   few ×10⁻⁴ of the score.

The routine always returns the best sequence discovered before the time
deadline, guaranteeing a valid (non‑empty) answer.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of max(conv).
    Returns the best vector (on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)             # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.05
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over λ·(conv−max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – slowly move towards the true max
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Runs until the deadline and keeps the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set for the current max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size (mirror‑descent on simplex)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise to the simplex, protecting against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style hill‑climbing.
    Randomly transfers a fraction of mass between two coordinates.
    Accepts improving moves and, with a temperature schedule, occasional
    worsening moves.  Returns the best vector seen.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    it = 0
    while it < max_moves and time.time() < deadline:
        # linear temperature decay – a few ·10⁻⁴ is enough for fine‑tuning
        T = max(1e-6, (cur_val - best_val) * 0.5 + 1e-4)

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of the mass at i towards j
        frac = np.random.rand() * 0.4          # up to 40 % of a[i]
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        # Metropolis acceptance
        if val2 < cur_val or np.random.rand() < np.exp(-(val2 - cur_val) / T):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
        it += 1

        if it % 20_000 == 0:
            print(f"[HillClimb it={it:06d}] best {best_val:.6f}")

    return best_a, best_val
