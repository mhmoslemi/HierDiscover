def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]
```

"""
Propose a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm (no external state, works under the 1000 s budget):
1. Build a few diverse starting points (warm start, uniform, Dirichlet draws,
   simple parametric families).  Each start is normalized to sum = 1.
2. For every start run a *smooth‑max projected gradient descent* with momentum.
   The smooth‐max surrogate Sλ(a)=log∑exp(λ·(a∗a))/λ is optimized; the
   gradient is 2·cross_corr(w,a) where w are soft‑max weights.
   After each step a is projected onto the probability simplex.
3. Keep the best vector obtained by the previous phase and improve it with a
   *penalty gradient descent*: we push the maximum convolution value below a
   decreasing target by minimising   L(a)=∑max(0,(a∗a−M))² .
   Its gradient is 4·cross_corr(overshoot,a).  Again a is projected onto the
   simplex.
4. Finally polish the best vector with a lightweight *simulated‑annealing*
   local search that repeatedly transfers a random fraction of mass between two
   positions.
All phases respect the global deadline and always return a valid (≥0, sum>0)
sequence.  The implementation avoids closures and uses only NumPy and the
provided convolution helpers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple parametric families (used for initialisation)

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_laplacian(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-gamma * np.abs(i - center))
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_half_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation primitives

def _eval_seq(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a∗a) for a already normalised (sum=1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


def _smooth_max_grad_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max surrogate.
    Returns the best vector found and its evaluation.
    """
    a = a0.copy()
    lr = 0.5               # initial step size
    lam = 0.5              # start temperature for smooth‑max
    lam_factor = 1.02
    beta = 0.9             # momentum coefficient
    v = np.zeros_like(a)   # momentum buffer
    best = a.copy()
    best_val = _eval_seq(a, n, conv_fft_len)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[SmoothGD it={it}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (2·cross‑correlation)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # momentum update and step
        v = beta * v + grad
        a_new = a - lr * v

        # Euclidean projection onto the simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-12:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new = a_new / s

        # simple back‑tracking: accept if the objective improves,
        # otherwise shrink the step size.
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new > val:
            lr *= 0.5            # make the next step smaller
        else:
            a = a_new
            lr = min(lr * 1.01, 1.0)   # gently increase again
            lam = min(lam * lam_factor, 1e7)

        it += 1
    return best, best_val


def _penalty_grad_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent that directly minimises the penalty
        L(a)=∑ max(0, (a∗a)−M)^2
    for a slowly decreasing target M.
    """
    a = a0.copy()
    step = 0.05           # base step size
    best = a.copy()
    best_val = _eval_seq(a, n, conv_fft_len)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[Penalty it={it}] new best {val:.6f}")

        # target is slightly below the current maximum; this creates
        # a positive overshoot only where conv exceeds the target.
        target = max_c * (1.0 - 5e-4)          # 0.05 % reduction per iteration
        overshoot = np.maximum(conv - target, 0.0)

        if overshoot.max() == 0.0:        # already below target, shrink further
            target = max_c * (1.0 - 1e-4)
            overshoot = np.maximum(conv - target, 0.0)
            if overshoot.max() == 0.0:
                # nothing to push – reduce step and continue
                step *= 0.99
                it += 1
                continue

        # gradient of the quadratic penalty
        grad = 4.0 * _cross_correlation(overshoot, a, corr_fft_len)

        # gradient step + simplex projection
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # slowly decay the raw step size
        step = max(step * 0.9999, 1e-6)
        it += 1
    return best, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing polish on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction from i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget, combines three optimisation phases
    (smooth‑max gradient descent, penalty descent and simulated annealing) and
    always returns a valid sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 works well for speed / quality trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run produced a good sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a modest pool of diverse initial vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())
    # uniform baseline
    pool.append(_make_uniform(n))
    # few Dirichlet draws with different concentrations
    for alpha in (0.005, 0.05, 0.2, 1.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))
    # a couple of Gaussian bumps
    for sigma in (0.02, 0.07, 0.15):
        pool.append(_make_gaussian(n, sigma))
    # exponential / Laplacian families
    for gamma in (0.1, 0.4, 0.8):
        pool.append(_make_exponential(n, gamma))
        pool.append(_make_laplacian(n, gamma))
    # beta shapes
    for a_par, b_par in ((0.5, 0.5), (2.0, 2.0), (0.2, 2.0), (2.0, 0.2)):
        pool.append(_make_beta(n, a_par, b_par))
    # half‑sine / cosine / triangle
    pool.append(_make_half_sine(n))
    pool.append(_make_cosine(n))
    pool.append(_make_triangle(n))

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    total_candidates = len(pool)

    # allocate a slice of time for each initial candidate (reserve ~300 s for fine‑tuning)
    remaining = deadline - time.time()
    reserve = 300.0
    time_per_cand = max(5.0, (remaining - reserve) / max(1, total_candidates))

    for idx, a0 in enumerate(pool):
        if time.time() >= deadline - reserve:
            break
        a0 = np.clip(a0, 0.0, 1000.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        a_local, val_local = _smooth_max_grad_descent(
            a0, n, conv_fft_len, corr_fft_len, time_per_cand
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[Init cand {idx}] improved to {val_local:.6f}")

    # ------------------------------------------------------------------
    # Phase II – penalty‑gradient descent on the current best
    if best_seq is not None and time.time() < deadline - 120.0:
        time_left = deadline - time.time() - 120.0
        a_local, val_local = _penalty_grad_descent(
            best_seq, n, conv_fft_len, corr_fft_len, time_left
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[Penalty] improved to {val_local:.6f}")

    # ------------------------------------------------------------------
    # Phase III – simulated‑annealing polish
    if best_seq is not None and time.time() < deadline - 30.0:
        print("[Anneal] polishing")
        a_local, val_local = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=600_000
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[Anneal] improved to {val_local:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
