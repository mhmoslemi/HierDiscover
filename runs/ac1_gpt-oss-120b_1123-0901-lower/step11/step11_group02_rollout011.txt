        # adapt length to n
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0,n-warm.shape[0]), mode='constant')
        warm = np.maximum(warm, 0.)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None

    # Build small set of candidate sequences (maybe 5) for initial quick evaluation
    candidates = []

    if warm is not None:
        candidates.append(warm.copy())

    candidates.append(_make_uniform(n))
    candidates.append(_make_triangle(n))
    candidates.append(_make_exponential(n, 0.5))
    candidates.append(_make_beta(n, 0.5, 0.5))
    # random Dirichlet draws
    for _ in range(4):
        alpha = 10 ** np.random.uniform(-2, 0)  # between .01 and 1
        candidates.append(_make_dirichlet(n, alpha))

    # Evaluate each using internal metric (without calling evaluate_sequence) to get quick (2*n*max_conv) value.
    def fast_score(a):
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()
    best_seq = None
    best_score = float('inf')
    for idx, a in enumerate(candidates):
        a = np.maximum(a,0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        score = fast_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[Init] candidate {idx} score {score:.6f}")

    # Phase A: Adam on best_seq
    if time.time() < deadline - 20:
        sub_dead = min(deadline, time.time() + min(180.0, deadline - time.time() - 10.0))
        a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=500_000)
        if val_adam < best_score:
            best_score = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # Phase B: Exponentiated gradient
    if time.time() < deadline - 30:
        a_exp, val_exp = _subgradient_optimize_expgrad(best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=800_000, eta0=0.8)
        if val_exp < best_score:

            best_seq = a_exp.copy()
            print(f"[ExpGrad] improved to {val_exp:.6f}")

    # Phase C: Mass transfer local search
    if time.time() < deadline - 5:
        a_mt, val_mt = _mass_transfer_opt2(best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=500_000)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[MassTransfer] improved to {val_mt:.6f}")

    # Phase D: Simulated annealing for final polishing
    if time.time() < deadline - 2:
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len, deadline, max_moves=200_000)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # Phase E: Upsample to 8192 if time remains > 30
    if time.time() + 30 < deadline:
        try:

            conv_fft_len_big = _next_pow2(2*n_big-1)

            x_old = np.linspace(0, 1, n, endpoint=False)

            a_big = np.interp(x_big, x_old, best_seq)
            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            if val_big < best_score:
                best_score = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] improved to {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # final evaluation (optional)
    try:

        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()

def _mass_transfer_opt2(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    epsilons = [0.5, 0.33, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active indices (max conv)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        else:
            # fallback: uniform w
            w = np.ones_like(conv) / conv.size
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # find indices with high and low gradient
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        # ensure positive source mass
        if a[i_high] < 1e-15:
            # random fallback
            i_high = np.argmax(a)  # largest mass
        # test transfers
        improved = False
        src_mass = a[i_high]
        for eps_frac in epsilons:
            eps = src_mass * eps_frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-12:
                continue
            # compute objective quickly by conv
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer2 it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # random small transfer to escape
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.03
                a[i] -= eps
                a[j] += eps
        it += 1
    return best_a, best_val

"""
Propose a low‑score step‑function.

Algorithm
---------
1. Load the optional warm‑start (global ``height_sequence_1``) and
   resample it to the working size n=4096 (or initialise a uniform grid).

2. Build a rich pool of candidate probability vectors (Dirichlet draws,
   exponential “U‑shapes”, beta‑U‑shapes, power‑laws, triangular,
   cosine, sparse spikes, …).  All candidates are normalised to sum‑=‑1.

3. Quick screen: evaluate each candidate with the exact objective
   ``2·n·max(conv)`` (no scaling needed) and keep the best.

4. Deep local optimisation on the selected best vector.
   * Adam optimiser on a smooth‑max surrogate (≈30 s).
   * Exponentiated‑gradient sub‑gradient phase (≈30 s).
   * Refined mass‑transfer local search – moves a small amount of mass
     from the coordinate with the largest gradient to the one with the
     smallest, trying a fine grid of step‑sizes (≈30 s).
   * Short simulated‑annealing polish (≈15 s).

5. If time permits up‑sample the current best to a finer grid
   (n=8192), re‑normalise and repeat the whole optimisation pipeline on
   the larger problem (the larger dimensionality gives the optimiser
   more freedom and often yields a slightly smaller constant).

6. Clip the final vector to the evaluator’s limits, print the exact
   score (using the provided ``evaluate_sequence``), and return it as a
   plain Python list.

The routine respects the 1000 s time budget, never returns an empty
sequence, and keeps the best vector found so far at every stage.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Utility helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Beta pdf sampled on the mid‑points of a uniform grid."""
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 800_000,
                                 eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimiser."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _mass_transfer_opt2(a0: np.ndarray, n: int,
                       conv_fft_len: int, corr_fft_len: int,
                       deadline: float,
                       max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Very fine‑grained mass‑transfer local search.
    Moves a fraction of the source mass from the index with the largest
    gradient to the index with the smallest gradient, trying a dense list
    of step‑sizes.  If no improvement is found a tiny random perturbation
    is applied to escape plateaus.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    epsilons = (0.5, 0.33, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # ensure we have some mass to move
        if a[i_high] <= 1e-15:
            # fallback: use the currently largest coordinate as source
            i_high = int(np.argmax(a))

        improved = False
        src_mass = a[i_high]

        for eps_frac in epsilons:
            eps = src_mass * eps_frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer2 it={it:06d}] new best {val_new:.6f}")
                break

        if not improved:
            # tiny random shake to try to leave a flat region
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.03
                a[i] -= eps
                a[j] += eps

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int,
                    deadline: float,
                    max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """Short simulated‑annealing pass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _optimize_on_n(init_seq: np.ndarray, n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the full pipeline (Adam → ExpGrad → Mass‑Transfer → Anneal)
    on a problem of size ``n`` starting from ``init_seq``.
    Returns the best vector found and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ensure proper normalisation
    a = np.maximum(init_seq, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")

    # -------- Adam --------
    if time.time() < deadline - 60:
        sub_dead = min(deadline, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(a, n, conv_fft_len, corr_fft_len,
                                          sub_dead, max_iters=400_000)
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Phase] Adam improved to {val_adam:.6f}")

    # -------- Exponentiated‑gradient --------
    if time.time() < deadline - 40:
        a_exp, val_exp = _subgradient_optimize_expgrad(best_a, n,
                                                      conv_fft_len,
                                                      corr_fft_len,
                                                      deadline,
                                                      max_iters=600_000,
                                                      eta0=0.85)
        if val_exp < best_val:
            best_val, best_a = val_exp, a_exp.copy()
            print(f"[Phase] ExpGrad improved to {val_exp:.6f}")

    # -------- Mass‑Transfer fine search --------
    if time.time() < deadline - 20:
        a_mt, val_mt = _mass_transfer_opt2(best_a, n,
                                           conv_fft_len, corr_fft_len,
                                           deadline, max_iters=700_000)
        if val_mt < best_val:
            best_val, best_a = val_mt, a_mt.copy()
            print(f"[Phase] Mass‑Transfer improved to {val_mt:.6f}")

    # -------- Simulated annealing --------
    if time.time() < deadline - 5:
        a_ann, val_ann = _anneal_optimize(best_a, n,
                                          conv_fft_len,
                                          deadline,
                                          max_moves=250_000)
        if val_ann < best_val:
            best_val, best_a = val_ann, a_ann.copy()
            print(f"[Phase] Anneal improved to {val_ann:.6f}")

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ 1000 s.  Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ----------------------------------------------------------
    # 1. Warm‑start handling (previous best)
    # ----------------------------------------------------------
    warm = globals().get("height_sequence_1")
    n_small = 4096
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_small:
            warm = warm[:n_small]
        elif warm.shape[0] < n_small:
            warm = np.pad(warm, (0, n_small - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ----------------------------------------------------------
    # 2. Build a diverse pool of starting vectors (small n)
    # ----------------------------------------------------------
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n_small // 4, n_small // 2, 3 * n_small // 4):
            pool.append(np.roll(warm, shift))

    # deterministic shapes
    pool.append(_make_uniform(n_small))
    pool.append(_make_triangle(n_small))
    pool.append(_make_cosine(n_small))

    # exponential (U‑shape) family
    for gamma in (0.2, 0.4, 0.6, 0.9, 1.5, 2.5, 4.0):
        pool.append(_make_exponential(n_small, gamma))

    # beta‑U‑shape family – include very sharp ends
    for alpha in (0.2, 0.3, 0.4, 0.5, 0.8, 1.0):
        pool.append(_make_beta(n_small, alpha, alpha))

    # Dirichlet draws with a range of concentration parameters
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_make_dirichlet(n_small, alpha))

    # power‑law tails
    for expo in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw(n_small, expo))

    # sparse spike patterns
    for k in (32, 64, 128, 256):
        pool.append(_make_spikes(n_small, k))

    # a few completely random Dirichlet draws
    for _ in range(6):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        pool.append(_make_dirichlet(n_small, alpha))

    # ----------------------------------------------------------
    # 3. Quick screening to pick the most promising start
    # ----------------------------------------------------------
    def _fast_score(a):
        """Exact objective without the denominator (sum=1)."""
        conv = _full_convolution(a, _next_pow2(2 * n_small - 1))
        return 2.0 * n_small * conv.max()

    best_seq = None
    best_val = float("inf")
    for idx, a0 in enumerate(pool):
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n_small))
        a0 = a0 / a0.sum()
        score = _fast_score(a0)
        if score < best_val:
            best_val = score
            best_seq = a0.copy()
            print(f"[Init] candidate {idx:02d} score {score:.6f}")

    # ----------------------------------------------------------
    # 4. Deep optimisation on the chosen start (n = 4096)
    # ----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        best_seq, best_val = _optimize_on_n(best_seq, n_small, deadline)

    # ----------------------------------------------------------
    # 5. Upsample to a finer grid (n = 8192) if time permits
    # ----------------------------------------------------------
    if time.time() + 30 < deadline:
        try:
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # linear interpolation of the current best onto the finer grid
            x_old = np.linspace(0, 1, n_small, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 25.0)
            a_big, val_big = _optimize_on_n(a_big, n_big, sub_dead_big)

            # keep the finer result if it beats the current best
            if val_big < best_val:
                best_val, best_seq = val_big, a_big.copy()
                print(f"[Upsample] refined best to {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # ----------------------------------------------------------
    # 6. Final preparation and reporting
    # ----------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
