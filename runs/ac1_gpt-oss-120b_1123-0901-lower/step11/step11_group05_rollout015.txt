"""
Propose a low‑value step‑function.

The optimizer works in three phases inside the 1000 s budget:

1. **Diverse random‑start exploration** –  many families of
   probability vectors (beta‑shapes, power‑law, exponential,
   sparse “spike” vectors and Dirichlet draws with a wide range of
   concentration) are generated, normalised to sum = 1 and evaluated
   with an exact FFT‑based convolution.  The best candidate is kept.

2. **Two‑stage exact optimisation** – starting from the best
   candidate we run
   * an exponentiated‑gradient sub‑gradient descent on the exact
     max‑convolution (the same routine that was used in the reference
     code, but with a more aggressive stepsize schedule),
   * a short Adam optimisation on a smooth‑max surrogate of the max
     (temperature λ grows up to 1e16).  Both phases respect the time
     limit and keep the best intermediate vector.

3. **Fine‑grained local search** – tiny stochastic moves that transfer
   a random amount of mass between two coordinates, followed by a
   scan over a handful of cyclic shifts.  The current best vector is
   returned as a plain Python list.

All helper functions are top‑level, there are no closures or lambdas,
and the routine never exceeds the supplied time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv=1, v≥0."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (the factor 2·n is the only part that matters,
# because scaling the vector does not change the score)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa=1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisers (same building blocks as the reference implementation)
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set = indices attaining the maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e16           # push close to the true max

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a valid probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (Bregman‑gradient)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        # modest learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric beta‑shaped density on {0,…,n‑1}."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Power‑law (inverse) shape a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _exponential_shape(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _spike_shape(n: int, k: int) -> np.ndarray:
    """k random spikes (Dirichlet on a random subset)."""
    pos = np.random.choice(n, size=k, replace=False)
    weights = np.random.dirichlet(np.ones(k))
    a = np.zeros(n, dtype=np.float64)
    a[pos] = weights
    return a

    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few beta‑shaped families (including asymmetric ones)

    # a handful of random Dirichlet draws (wide concentration range)

    # ------------------------------------------------------------------
    # 2) Very fast random‑search phase (≈ 30 % of the total budget)

        else:                               # Dirichlet with random concentration

        if s < 1e-9:

        a /= s
        candidates.append(a)

    # ------------------------------------------------------------------

    best_val = np.inf
    for a in candidates:
