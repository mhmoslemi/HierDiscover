"""
Hybrid optimizer for the “min‑max convolution” problem.

The script searches for a non‑negative sequence a of length n that
minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the linear convolution (np.convolve).  
The algorithm consists of three phases

1. **Fast Adam on a smooth‑max surrogate** – the max is approximated by a
   log‑sum‑exp with a temperature λ that is increased slowly.
   Variables are parametrised by logits (soft‑max) so the simplex
   constraint Σa = 1 and a ≥ 0 are satisfied automatically.

2. **Exact‑max sub‑gradient (exponentiated‑gradient)** – once the
   surrogate has roughly converged we switch to the true non‑smooth
   objective and perform a classic exponentiated‑gradient update on the
   simplex.

3. **Local random refinement** – a cheap simulated‑annealing style
   move that transfers a small amount of mass between two positions.
   It is run while time permits and often gives a tiny final improvement.

The routine is time‑budget aware (default 1000 s).  It starts from a
handful of diverse initial vectors (including the previous best if
available) and keeps the best sequence ever seen.  All heavy linear
algebra (convolution and cross‑correlation) is performed with FFTs, so
each iteration costs only O(n log n).

The final sequence is returned as a plain Python list; it will be
validated by the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection -----------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories ----------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_sine(n: int, power: float = 1.0) -> np.ndarray:
    """Symmetric sine‑shaped vector a_i ∝ sin(π·(i+0.5)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser ----------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lr: float = 0.07,
    lam_init: float = 0.01,
    lam_factor: float = 1.01,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of max(conv).
    Returns the best (exact) objective value seen and the associated vector.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise soft‑max logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_init

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- soft‑max to obtain a on the simplex -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- compute convolution and exact objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                # Σa = 1  → denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # slowly raise the temperature → tighter approximation of max
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay (helps final polishing)
        if (it + 1) % 200_000 == 0:
            lr *= 0.5

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient on the exact max --------------------
def _expgrad_subgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimiser (exponentiated‑gradient).
    Works directly on the simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # indices where the maximum is attained (the active set)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)      # multiplicative EG update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple random local search (simulated annealing) -----------------------
def _local_anneal(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing refinement that moves a small fraction of mass
    from one index to another.  Very cheap (uses the FFT‑based convolution
    each move) and often squeezes a few 1e‑4 improvements out of a good
    candidate.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.05
    T_end = 1e-5

    i = 0
    while time.time() < deadline and i < max_moves:
        T = max(T_end, T_start * (1.0 - i / float(max_moves)))
        # pick two distinct indices
        src = np.random.randint(n)
        if a[src] < 1e-14:
            i += 1
            continue
        dst = np.random.randint(n)
        if src == dst:
            i += 1
            continue

        # move a random fraction of src's mass to dst
        frac = np.random.rand() * 0.25   # up to 25 % of src
        delta = a[src] * frac
        a2 = a.copy()
        a2[src] -= delta
        a2[dst] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
            # always accept improvement
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2

        i += 1

    return best_a, best_val

        a_refined, val_refined = _local_anneal(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_refined < best_val:
            best_val = val_refined
            best_seq = a_refined.copy()
            print(f"[Phase 4] Anneal improved to {val_refined:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a return value
    if best_seq is None:
        best_seq = _make_uniform(n)
        best_val = float("inf")

    # Clip to the limits imposed by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final reporting (uses the provided evaluate_sequence)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:  # pragma: no cover
        print(f"Evaluation error: {exc}")
        final_val = best_val
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
