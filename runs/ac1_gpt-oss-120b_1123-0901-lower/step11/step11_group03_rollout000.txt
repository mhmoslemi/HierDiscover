"""
propose_candidate – Gradient‑plus‑annealing optimiser for the non‑negative
step‑function problem.

Goal
----
Given a non‑negative vector a (length n) we minimise

    2·n·max(conv(a,a)) / (∑a)²

The objective is scale‑invariant; we therefore keep the vector on the
probability simplex (∑a = 1).  The optimisation proceeds in four stages:

1. **Diverse initial pool** – a collection of analytic shapes (uniform,
   sine, arcsine, exponential, power‑law, sparse spikes, …) together with
   a warm‑start from the previously found sequence (if any).  The best
   candidate by the exact objective is retained.

2. **Adam on a smooth‑max surrogate** – we replace `max(b)` by a
   log‑sum‑exp smooth approximation with a temperature λ that is slowly
   increased.  Adam updates are performed on the log‑softmax parameters
   (`logits`) of the probability vector, always projecting back onto the
   simplex.

3. **Exponentiated‑gradient sub‑gradient descent on the true max** – a
   mirror‑descent step with an adaptive step size driven by the current
   sub‑gradient (active set of the maximal convolution entries).

4. **Fine‑grained simulated‑annealing** – random mass transfers between
   two coordinates are accepted if they improve the objective or with a
   Metropolis probability that decays with the temperature schedule.

The whole routine respects the global deadline (default 1000 s) and
always returns the best vector found.  The length `n` is set to 8192
(2× larger than the previous run) which gives many more degrees of
freedom and a lower theoretical lower bound
    2·n/(2·n−1) ≈ 1.00006.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT (returns length 2·len(a)−1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a).
    Returns a vector of length len(a) equal to the gradient of
    Σ_k w_k·(a★a)_k with respect to a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors of length n."""
    pool: list[np.ndarray] = []

    # warm‑start (if supplied)
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w.copy())
            # a few circular shifts of the warm start
            for shift in (n // 4, n // 2, 3 * n // 4):
                pool.append(np.roll(w, shift))

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    pool.append(s)

    # Exponential tails (both sides)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law (U‑shaped) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Arcsine (Beta(0.5,0.5)) – strong ends, low centre
    x = (np.arange(n) + 0.5) / n
    arcs = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    arcs /= arcs.sum()
    pool.append(arcs)

    # Double‑Gaussian spikes (edges)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.04, 0.08, 0.12):
        g = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
            + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        g /= g.sum()
        pool.append(g)

    # Sparse spikes (≈√n spikes)
    for k in (64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        pool.append(a)

    # Dirichlet draws with varying concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    return pool

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   corr_len: int,
                   deadline: float,
                   max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp smooth approximation of max(conv).
    Returns the best vector (on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.08                     # start temperature
    lam_factor = 1.012             # increase slowly

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # Soft‑max to retrieve a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # log‑sum‑exp (smooth max)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # Chain rule through soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e9)

        # occasional learning‑rate decay
        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        # keep exact best
        exact_val = 2.0 * n * max_c
        if exact_val < best_val:
            best_val = exact_val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam it={it:06d}] new best {exact_val:.6f}")

        it += 1

    # final exact evaluation
    conv = _full_convolution(best_a, fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          fft_len: int,
                          corr_len: int,
                          deadline: float,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns best vector and best exact objective seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    lower_bound = 2.0 * n / (2 * n - 1)          # theoretical minimum
    best_a = a.copy()
    conv = _full_convolution(a, fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 40_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices achieving the current max)
        active = np.isclose(conv, max_c, atol=1e-15, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # Polyak‑type adaptive step size (clipped)
        # η = (f(x) - f*) / ||g||²  (with safety factor)
        grad_norm2 = np.dot(grad_a, grad_a) + 1e-12
        eta = eta0 * (val - lower_bound) / grad_norm2
        # keep η positive and not too large
        eta = max(min(eta, 0.5), 1e-5)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float,
                     max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly moves a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

        if val < best_val:

        if time.time() > deadline:

    # ------------------------------------------------------------------
    # Phase I – Adam optimisation on a smooth‑max surrogate

        sub_dead = deadline - 5.0
        a_adam, val_adam = _adam_optimize(

    if best_seq is not None and time.time() < deadline - 8:

    # ------------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
