def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n-1)

    # get warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.shape[0] != n:
            if a.shape[0] > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.shape[0]))
    else:
        a = np.full(n, 1.0/n)

    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        a = np.full(n, 1.0/n)
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"Initial value {best_val}")

    # iterative LP refinement
    it = 0
    while time.time() < deadline - 5 and it < 200:
        # ensure positivity for LP
        a_pos = np.maximum(a, 1e-12)
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        # If many active, limit to e.g., 30 randomly chosen
        if len(active) > 30:
            active = active[:30]  # or random subset
        # Build LP matrices
        k = len(active)
        A = np.zeros((k, n), dtype=np.float64)
        for row_idx, t in enumerate(active):
            j_start = max(0, t - (n - 1))
            j_end = min(t, n - 1)
            j_range = np.arange(j_start, j_end + 1)
            i_vals = t - j_range
            A[row_idx, j_range] = a_pos[i_vals]
        b_ub = np.full(k, M)
        c = -np.ones(n)  # maximize sum(b)

        # Solve LP
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                          options={'presolve': True, 'time_limit': max(0, int(deadline - time.time()))})
        except Exception as e:
            print(f"LP solve failed: {e}")
            break

        if not res.success:
            print(f"LP not successful, status {res.status}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-8:
            print("LP cannot improve sum, stopping")
            break

        a_new = b / sum_b
        # Compute new objective
        val = evaluate_sequence(a_new.tolist())
        print(f"[LP it={it}] new val {val:.6f} sum_b={sum_b:.6f}")
        if val < best_val:
            best_val = val
            best_a = a_new.copy()

        a = a_new
        it += 1

    # after LP, perform fine-grained mass transfer
    if time.time() < deadline - 2:
        # use mass transfer on best_a
        a_mt, val_mt = _mass_transfer_optimize(best_a, n, conv_fft_len,
                                               _next_pow2(3*n-2), deadline)
        if val_mt < best_val:
            best_val = val_mt
            best_a = a_mt

    # final clip and evaluate exact
    best_a = np.clip(best_a, 0.0, 1000.0)
    # ensure sum >0.01
    if best_a.sum() < 0.01:
        best_a = np.full(n, 1.0/n)
    else:
        best_a = best_a / best_a.sum()
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Final score {final_score}")
    return best_a.tolist()

"""
Hybrid optimizer with an active‑set linear‑programming refinement.

Algorithm
---------
1. Warm‑start from the previously discovered sequence (global variable
   ``height_sequence_1``) or fall back to a uniform distribution.
2. Re‑scale to a probability simplex (sum = 1) – the objective is then
   ``2·n·max(conv)``.
3. **LP refinement** – repeatedly solve a tiny linear program that
   maximises the total ℓ₁‑mass of a new vector `b` while keeping the
   current worst convolution value `M` feasible on the *active* set
   (the indices where the convolution attains its maximum).  The update
   `a ← b / sum(b)` strictly lowers the maximum if `sum(b) > 1`.
4. **Fine‑grained mass‑transfer** – a fast local‑search that moves a
   small amount of probability mass from the coordinate with the largest
   sub‑gradient of the max‑convolution to the one with the smallest.
5. Return the best vector found (clipped to the evaluator’s bounds).

The LP phase is very cheap because only the (usually one) active
constraint(s) are used, yet it often yields a noticeable drop in the
evaluation score.  The final mass‑transfer stage polishes the result.
The whole routine respects a 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Active‑set LP refinement

def _lp_refine_optimize(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        deadline: float,
                        max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements of a vector `a0` (sum = 1).
    Returns the best vector encountered and its exact objective
    (2·n·max(conv)).
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("[LP] SciPy not available – skipping LP refinement.")
        return a0, 2.0 * n * _full_convolution(a0, conv_fft_len).max()

    a = a0.copy()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()

        # active indices – usually just the peak (t where conv == M)
        # use an absolute tolerance far below numerical noise
        active = np.where(conv >= M - 1e-12)[0]
        if active.size == 0:
            break

        # limit the number of constraints (the active set is tiny anyway)
        if active.size > 20:
            active = active[:20]

        k = active.size
        A = np.zeros((k, n), dtype=np.float64)

        # each row corresponds to a shifted copy of `a`
        for r, t in enumerate(active):
            j_start = max(0, t - (n - 1))
            j_end = min(t, n - 1)
            j_range = np.arange(j_start, j_end + 1, dtype=np.int64)
            i_vals = t - j_range
            A[r, j_range] = a[i_vals]               # a[t‑j] for valid j

        b_ub = np.full(k, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)            # maximise Σb  → minimise –Σb

        # Solve the LP (max Σb subject to A·b ≤ M, b ≥ 0)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'presolve': True})

        if not res.success:
            # either infeasible or unbounded – stop the refinement
            print(f"[LP it={it}] LP failed (status {res.status})")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # no improvement possible
            break

        # Normalise to the simplex
        a_new = b / sum_b
        val_new = 2.0 * n * _full_convolution(a_new, conv_fft_len).max()

        if val_new + 1e-12 < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new                                   # continue from the improvement
            print(f"[LP it={it}] improved objective {val_new:.6f} (sum_b={sum_b:.6f})")
        else:
            # No improvement – stop early
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine‑grained mass‑transfer optimiser (as in the previous state‑of‑the‑art)

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _mass_transfer_optimize(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the coordinate with the
    largest sub‑gradient to the one with the smallest, accepting only
    strict improvements of the exact objective.
    """
    a = a0.copy()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        improved = False
        src_mass = a[i_high]
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it}] new best {val_new:.6f}")
                break
        if not improved:
            # tiny random perturbation to escape possible plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) with a
    low evaluation score.  Uses a warm start, an active‑set LP refinement,
    and a mass‑transfer local search.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # problem dimensions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Initialise the candidate vector
    if warm is not None:
        a = warm.copy()
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform fallback

    # ------------------------------------------------------------------
    # Phase I – active‑set LP refinement
    if time.time() < deadline - 10:
        print("[LP] Starting active‑set linear‑program refinement")
        a_lp, val_lp = _lp_refine_optimize(a, n, conv_fft_len, deadline,
                                           max_iters=40)
        a = a_lp
        print(f"[LP] best objective after refinement {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase II – fine‑grained mass‑transfer local search
    if time.time() < deadline - 2:
        print("[MassTransfer] Starting fine‑grained refinement")
        a_mt, val_mt = _mass_transfer_optimize(
            a, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000
        )
        a = a_mt
        print(f"[MassTransfer] final objective {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Exact evaluation (for the competition harness)
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(a)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return a.tolist()
