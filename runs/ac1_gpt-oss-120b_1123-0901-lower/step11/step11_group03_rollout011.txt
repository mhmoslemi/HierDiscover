"""
Hybrid search for a low–autocorrelation step function.

We try many parametric families (Beta‑type, sinusoid powers, classic
window functions, Kaiser, exponentials, double‑Gaussians, random Dirichlet
samples), keep the best of those, and finally polish the winner with a
projected exponentiated‑gradient (mirror‑descent) on the exact
max‑convolution.  All work is done on the simplex (sum = 1) so the
objective reduces to 2·n·max(conv).  The routine respects the 1000 s
budget and always returns the best sequence found.
"""
import time
import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)          # 8192
    corr_fft_len = _next_pow2(3 * n - 2)          # enough for cross‑corr

    # ------------------------------------------------------------------
    # helper to compute the exact objective (sum is 1 → 2·n·max(conv))
    def obj(a: np.ndarray) -> float:
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # keep the best vector seen so far
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1. warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s < 0.01:
            w = np.random.dirichlet(np.ones(n))
        else:
            w = w / s
        val = obj(w)
        best_seq, best_val = w.copy(), val
        print(f"[Init] warm start value {val:.6f}")

    # ------------------------------------------------------------------
    # 2. systematic exploration of parametric families
    def consider(cand: np.ndarray, label: str):
        """Normalise, evaluate and keep the candidate if it improves."""
        nonlocal best_seq, best_val
        if time.time() >= deadline:
            return
        if cand.sum() < 0.01:
            return
        a = cand.astype(np.float64)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        val = obj(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{label}] new best {val:.6f}")

    # uniform
    consider(np.full(n, 1.0 / n), "Uniform")

    # --------------------------------------------------------------
    # symmetric Beta(α,α) family  (α>0 gives bell‑shaped, α<1 gives U‑shaped)
    idx = np.arange(n, dtype=np.float64) + 0.5
    for alpha in np.linspace(0.2, 3.0, 24):
        # a_i ∝ x^{α-1}(1-x)^{α-1}
        a = (idx ** (alpha - 1)) * ((n + 0.5 - idx) ** (alpha - 1))
        consider(a, f"Betaα={alpha:.2f}")

    # power‑law product with *negative* exponent (already explored in the
    # previous run but we add a denser grid)
    prod = (np.arange(1, n + 1, dtype=np.float64) *
            (n - np.arange(1, n + 1, dtype=np.float64) + 1.0))
    for alpha in np.linspace(-2.0, -0.5, 16):
        a = np.where(prod > 0, prod ** alpha, 0.0)
        consider(a, f"Prodα={alpha:.2f}")

    # --------------------------------------------------------------
    # sinusoid power family: a_i ∝ sin^p (π(i+0.5)/n)
    for p in np.linspace(0.2, 4.0, 20):
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        consider(a, f"SinPow={p:.2f}")

    # classic window functions
    i = np.arange(n, dtype=np.float64)
    consider(0.5 * (1 - np.cos(2 * np.pi * i / (n - 1))), "Hann")
    consider(0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1)), "Hamming")
    consider(0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) +
            0.08 * np.cos(4 * np.pi * i / (n - 1)), "Blackman")

    # Kaiser windows (β from 0 to 10)
    for beta in np.linspace(0.0, 10.0, 11):
        arg = beta * np.sqrt(1 - ((i - (n - 1) / 2) /
                                 ((n - 1) / 2)) ** 2)
        a = i0(arg) / i0(beta)
        consider(a, f"Kaiserβ={beta:.1f}")

    # --------------------------------------------------------------
    # exponential tails (symmetric)
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        consider(shape, f"Expγ={gamma}")

    # double‑Gaussian centred at the middle
    for sigma in (0.03, 0.05, 0.1, 0.2, 0.4):
        a = np.exp(-0.5 * ((i - (n - 1) / 2) /
                           (sigma * n)) ** 2)
        consider(a, f"Gaussσ={sigma}")

    # --------------------------------------------------------------
    # random Dirichlet draws (various concentrations)
    for conc in (0.1, 0.3, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            a = np.random.dirichlet(conc * np.ones(n))
            consider(a, f"Dirα={conc}")

    # sparse spike candidates (≈√n spikes)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        consider(a, f"Sparse{k}")

    # ------------------------------------------------------------------
    # 3. local refinement with exponentiated‑gradient sub‑gradient
    if best_seq is not None and time.time() < deadline - 5:
        print("[Refine] start mirror‑descent")
        a = best_seq.copy()
        it = 0
        while time.time() < deadline - 1.0:
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Refine] improved to {val:.6f} (it={it})")
            # active set (indices achieving the current max)
            active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
            # exact sub‑gradient of max‑convolution
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
            # step size schedule (exponentiated‑gradient / mirror descent)
            eta = 0.6 / np.sqrt(it + 1.0)
            a = a * np.exp(-eta * grad)
            # simplex projection (here scaling suffices because all entries stay non‑negative)
            s = a.sum()
            if s <= 1e-14 or np.isnan(s):
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a = a / s
            it += 1
            if it % 30_000 == 0:
                print(f"[Refine] it={it} current val={val:.6f}")

    # ------------------------------------------------------------------
    # safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
