Optimisation of a non‑negative step‑function (height sequence) for the
objective

The algorithm works inside a fixed time budget (≈1000 s) and always returns
a feasible sequence.

Main ideas
----------
1.  Produce a diverse set of warm‑starts (uniform, beta‑shaped, Dirichlet,
    optional previous solution).
2.  Refine the best start with a mirror‑descent (exponentiated‑gradient)

    autoconvolution.
3.  Switch to a *smooth‑max* surrogate of the max‑convolution and minimise

5.  Aggressive stochastic local moves (transfer a small fraction of mass

6.  A symmetry‑averaging post‑process (a←(a+a_rev)/2) can sometimes lower the
    peak further.

    Both arguments are real arrays; w has length 2·len(a)−1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a that obeys Σa = 1."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------------------- #
#  Optimisers used in the pipeline
# --------------------------------------------------------------------------- #
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation via exponentiated‑gradient updates.
    a0 must already be normalised (Σa = 1, a ≥ 0).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set (indices where convex combination of linear pieces)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)          # mirror‑descent step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 50_000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# --------------------------------------------------------------------------- #
#  Smooth‑max + L‑BFGS (continuation on λ)
# --------------------------------------------------------------------------- #
def _smooth_max_and_grad(logits: np.ndarray, n: int, conv_fft_len: int,
                         corr_fft_len: int, lam: float):
    """
    Returns (objective, gradient) of the smooth‑max surrogate
        2·n·S_λ(a)   with a = softmax(logits).
    """
    # soft‑max parametrisation
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # smooth‑max via log‑sum‑exp (stable)
    max_c = conv.max()
    exp_term = np.exp(lam * (conv - max_c))
    w_sum = exp_term.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w_norm = np.full_like(conv, 1.0 / conv.size)
        smooth_max = max_c
    else:
        w_norm = exp_term / w_sum
        smooth_max = max_c + np.log(w_sum) / lam

    obj = 2.0 * n * smooth_max

    # gradient w.r.t a (include the factor 2·n from the outer objective)
    grad_a = 4.0 * n * _cross_correlation(w_norm, a, corr_fft_len)

    # back‑prop to logits (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_logits = a * (grad_a - dot)

    return obj, grad_logits


# --------------------------------------------------------------------------- #
#  Main entry point
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine respects the supplied time budget (≈ 1000 s) and always
    returns a valid sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3    # small safety margin

    # ------------------------------------------------------------------- #
    #  Problem size – 4096 works well for the time budget.
    # ------------------------------------------------------------------- #
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------- #
    #  0) initialise a diverse set of candidates
    # ------------------------------------------------------------------- #
    candidates = []

    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-8:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        candidates.append(a)

    # uniform start
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric beta‑shaped densities (U‑shaped for α<1)
    alphas = [0.55, 0.60, 0.65, 0.70, 0.75, 0.80, 0.85, 0.90, 0.95]
    x = (np.arange(n) + 0.5) / n
    for alpha in alphas:
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            a = a / a.sum()
            candidates.append(a)

    # random Dirichlet draws with varied concentration
    for _ in range(30):
        conc = np.random.uniform(0.3, 3.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # ------------------------------------------------------------------- #
    #  1) pick the best cheap candidate (exact objective)
    # ------------------------------------------------------------------- #
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] objective {val:.6f}")

    # ------------------------------------------------------------------- #
    #  2) sub‑gradient optimisation (mirror descent / exponentiated‑gradient)
    # ------------------------------------------------------------------- #
    sub_deadline = deadline - 12.0
    if time.time() < sub_deadline:
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=5_000_000, eta0=1.5
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Subgrad] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------- #
    #  3) L‑BFGS on a smooth‑max surrogate (continuation on λ)
    # ------------------------------------------------------------------- #
    # try to import scipy – if unavailable we simply skip this phase
    try:
        from scipy.optimize import minimize
        have_scipy = True
    except Exception:
        have_scipy = False
        print("[Info] SciPy not available – skipping L‑BFGS phase.")

    if have_scipy and time.time() < deadline - 6.0:
        # start from the current best (logits of soft‑max)
        logits = np.log(best_a + 1e-15)

        # continuation schedule for λ (temperature)
        lam_schedule = [10.0, 50.0, 200.0, 800.0, 3200.0, 1e4, 5e4, 2e5]
        for lam in lam_schedule:
            if time.time() >= deadline - 4.0:
                break
            maxiter = 800            # generous but still cheap

            def fun(x):
                # SciPy expects (f, grad) when jac=True
                f, g = _smooth_max_and_grad(x, n, conv_fft_len, corr_fft_len, lam)
                return f, g

            res = minimize(fun, logits, method='L-BFGS-B', jac=True,
                           options={'maxiter': maxiter, 'disp': False})
            logits = res.x

            # evaluate the exact objective for the new point
            a = np.exp(logits - np.max(logits))
            a /= a.sum()
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[L‑BFGS λ={lam}] new best {val:.6f}")

    # ------------------------------------------------------------------- #
    #  4) short Adam fine‑tuning (same surrogate as before)
    # ------------------------------------------------------------------- #
    adam_deadline = deadline - 2.0
    if time.time() < adam_deadline:
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=200_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------- #
    #  5) aggressive stochastic local search (mass transfers)
    # ------------------------------------------------------------------- #
    local_deadline = deadline - 0.6
    while time.time() < local_deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # move a small random fraction of the source mass
        if best_a[i] <= 0:
            continue
        frac = np.random.uniform(0.001, 0.02)          # between 0.1 % and 2 %
        delta = min(best_a[i], best_a[i] * frac)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project onto the simplex
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # ------------------------------------------------------------------- #
    #  6) optional symmetry averaging (a←(a+a_rev)/2)
    # ------------------------------------------------------------------- #
    sym_a = (best_a + best_a[::-1]) / 2.0
    sym_a = _project_simplex(sym_a)
    sym_val = _objective(sym_a, n, conv_fft_len)
    if sym_val < best_val:
        best_val, best_a = sym_val, sym_a.copy()
        print(f"[Symmetrisation] improved to {sym_val:.6f}")

    # ------------------------------------------------------------------- #
    #  7) final sanity checks and return
    # ------------------------------------------------------------------- #
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a)   # ensure exact normalisation

    # official evaluation (mainly for logging)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
