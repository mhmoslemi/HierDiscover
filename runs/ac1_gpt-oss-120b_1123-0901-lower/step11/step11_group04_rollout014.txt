"""
Hybrid search for low‑max‑convolution step functions.

The routine combines a cheap parametric sweep (exponential / power‑law mixes
with the uniform distribution), a few random Dirichlet seeds and a
multi‑stage gradient refinement:
    * L2‑flattening Adam (drive the convolution towards the uniform target)
    * smooth‑max Adam (directly minimises the max‑convolution via a
      log‑sum‑exp surrogate)
    * exponentiated‑gradient sub‑gradient descent on the exact max
    * a final projected gradient descent and a short simulated‑annealing
      polishing step.

Time is split adaptively so that the expensive gradient phases get the
majority of the 1000 s budget while still exploring a diverse set of
initial candidates.  The algorithm works entirely on the probability
simplex (∑a = 1) – the evaluation function is scale‑invariant – and
returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories ----------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n: int, expo: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _mix(seq1: np.ndarray, seq2: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination alpha*seq1 + (1‑alpha)*seq2, already sums to 1."""
    mixed = alpha * seq1 + (1.0 - alpha) * seq2
    # numerical drift may happen – re‑normalise safely
    s = mixed.sum()
    if s < 1e-12:
        return _make_uniform(len(seq1))
    return mixed / s


# ----------------------------------------------------------------------
# Fast evaluation (same formula as evaluate_sequence) --------------------
def _fast_evaluate(seq: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)^2)."""
    conv = _full_convolution(seq, conv_fft_len)
    max_c = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max (log‑sum‑exp surrogate) ------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser – L2 flattening ----------------------------------------
def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser – exponentiated‑gradient ------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on exact max ---------------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# Simple simulated‑annealing refinement -----------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main search routine ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)^2).
    The routine performs:
      * a parametric sweep over exponential / power‑law families and
        convex mixtures with the uniform distribution,
      * a few random Dirichlet seeds,
      * a long L2‑flattening Adam phase,
      * a smooth‑max Adam phase,
      * exponentiated‑gradient sub‑gradient descent,
      * a projected‑gradient polishing step,
      * an optional short annealing refinement.
    It respects the allotted time budget (default 1000 s) and returns
    the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – can be increased if time permits
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously stored solution (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Initialise best candidate
    best_seq = _make_uniform(n)
    best_val = _fast_evaluate(best_seq, n, conv_fft_len)
    print(f"[Init] uniform value {best_val:.6f}")

    if warm is not None:
        val_warm = _fast_evaluate(warm, n, conv_fft_len)
        if val_warm < best_val:
            best_val = val_warm
            best_seq = warm.copy()
            print(f"[Warm] improved to {val_warm:.6f}")

    # ------------------------------------------------------------------
    # 1) Parametric sweep – exponential mixtures
    alphas = np.linspace(0.0, 1.0, 11)               # mixture weight for uniform
    gammas = np.linspace(0.01, 3.0, 61)              # exponential decay
    uni = _make_uniform(n)

    for a in alphas:
        if time.time() > deadline:
            break
        for g in gammas:
            exp_seq = _make_exponential(n, g)
            cand = _mix(uni, exp_seq, a)
            val = _fast_evaluate(cand, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Grid Exp] α={a:.2f} γ={g:.2f} → {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Parametric sweep – power‑law mixtures
    exps = np.linspace(-2.5, -0.1, 25)
    for a in alphas:
        if time.time() > deadline:
            break
        for e in exps:
            pow_seq = _make_powerlaw(n, e)
            cand = _mix(uni, pow_seq, a)
            val = _fast_evaluate(cand, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Grid Pow] α={a:.2f} ρ={e:.2f} → {val:.6f}")

    # ------------------------------------------------------------------
    # 3) Random Dirichlet seeds (different concentration parameters)
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0):
        for _ in range(20):
            if time.time() > deadline:
                break
            cand = _make_dirichlet(n, conc)
            val = _fast_evaluate(cand, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Dirichlet] α={conc:.2f} → {val:.6f}")

    # ------------------------------------------------------------------
    # 4) L2‑flattening Adam phase (takes the lion’s share of the budget)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # allocate up to ~70 % of the remaining budget to this phase
    while time.time() < deadline - 30:
        phase_dead = min(deadline, time.time() + 30.0)   # 30 s chunk
        a_l2, val_l2 = _adam_optimize_l2(
            best_seq, n, conv_fft_len, corr_fft_len,
            target, phase_dead,
            max_iters=600_000, lr=0.07
        )
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = a_l2.copy()
            print(f"[L2Adam] improved to {val_l2:.6f}")
        # one refinement cycle is enough; break to move on to next phase
        break

    # ------------------------------------------------------------------
    # 5) Smooth‑max Adam fine‑tuning
    if time.time() < deadline - 15:
        phase_dead = min(deadline, time.time() + 15.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            phase_dead, max_iters=500_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Smooth‑max Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 6) Exact‑max sub‑gradient (exponentiated‑gradient)
    if time.time() < deadline - 10:
        a_exp, val_exp = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len,

            print(f"[ExpGrad] improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # 7) Projected gradient descent (optional, time permitting)
    if best_seq is not None and time.time() < deadline - 10:
        print("[GD] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=300_000
        )

    # ------------------------------------------------------------------
    # 8) Final simulated‑annealing polishing (if any time left)
    if best_seq is not None and time.time() < deadline - 10:
        print("[Anneal] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=120_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure non‑negativity and legal range, then evaluate for reporting
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Evaluation error: {e}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
