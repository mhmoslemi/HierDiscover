"""
Search for a low‑peak convolution step‑function.

The algorithm works on a moderately large length (4096 – 16384) and
consists of three stages:

1. **Diverse random starts** – Dirichlet draws, very sparse “spike”
   vectors and vectors obtained by inverse FFT of random‑phase spectra.
   All starts are projected onto the probability simplex
   (non‑negative, sum = 1).

2. **Alternating projection** – each start is iteratively projected onto
   the set of vectors whose discrete Fourier transform has (almost) constant
   magnitude (except for the DC component) and back onto the simplex.
   This drives the autocorrelation of the vector towards a flat shape,
   which empirically reduces the max‑convolution value.

3. **Fine‑grained sub‑gradient refinement** – the best candidate from the
   previous stage is further optimized with an exponentiated‑gradient
   method that directly minimizes the exact max‑convolution
   (the same objective used by ``evaluate_sequence``).

The whole routine respects the given time budget (≈ 1000 s) and always
returns the best sequence found before the deadline.
"""


import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the full sequence of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Both arguments are real; the result has the same length as ``a``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _constant_magnitude_projection(a: np.ndarray) -> np.ndarray:
    """
    Replace the magnitude of the DFT of *a* (except the DC term)
    by a constant while keeping the phase unchanged.
    """
    n = a.shape[0]
    A = np.fft.fft(a)
    energy = np.dot(a, a)                # ‖a‖₂²
    dc = A[0]                             # sum(a)  – after projection = 1
    # remaining energy that must be shared equally among the non‑DC bins
    if n > 1:
        rem = max(0.0, n * energy - (np.abs(dc) ** 2))
        const_mag = np.sqrt(rem / (n - 1))
    else:
        const_mag = np.sqrt(energy)

    A_new = np.empty_like(A)
    A_new[0] = dc
    A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))
    a_new = np.real(np.fft.ifft(A_new))
    return a_new


def _evaluate_fast(a: np.ndarray, conv_fft_len: int, n: int) -> float:
    """
    Compute the exact objective 2·n·max(a*a) (sum(a) is 1),
    using an FFT‑based convolution for speed.
    """
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)
    max_c = conv[: 2 * n - 1].max()
    return 2.0 * n * max_c


def _gen_random_phase(n: int) -> np.ndarray:
    """
    Produce a vector whose spectrum has unit magnitude and random phase,
    then project onto the simplex.
    """
    half = n // 2
    phi = np.random.random(half - 1) * 2.0 * np.pi
    A = np.empty(n, dtype=np.complex128)
    A[0] = 1.0                         # DC = 1 → sum = 1
    A[1:half] = np.exp(1j * phi)       # positive frequencies
    A[-(half - 1):] = np.conj(A[1:half][::-1])   # negative frequencies
    if n % 2 == 0:
        # Nyquist frequency (real); pick a random sign
        A[half] = 1.0 if np.random.rand() < 0.5 else -1.0
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s
    return a


def _gen_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a vector from Dirichlet(alpha·1)."""
    a = np.random.dirichlet(alpha * np.ones(n))
    return a


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient minimisation of the exact
    max‑convolution objective.
    Operates directly on the simplex; the loop stops early when the
    deadline is reached.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    (≈ 1000 s) and always returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # Choose a problem size that balances quality and speed.
    # Larger sizes give a lower theoretical limit (≈ 1.5) but need more time.
    if budget_s >= 800:
        n = 16384
    elif budget_s >= 400:
        n = 8192
    else:
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    cand_id = 0

    # ------------------------------------------------------------------
    # Phase I – generate diverse candidates and apply a cheap alternating
    # projection (2000 iterations max).  The loop stops when time runs out.
    max_candidates = 30
    while time.time() < deadline and cand_id < max_candidates:
        # rotate through three simple generators
        if cand_id % 3 == 0:
            a0 = _gen_random_phase(n)
        elif cand_id % 3 == 1:
            a0 = _gen_dirichlet(n, alpha=0.2)
        else:
            a0 = _gen_dirichlet(n, alpha=2.0)

        # alternating projection: constant‑magnitude spectra + simplex
        a = a0.copy()
        proj_steps = 2000
        for it in range(proj_steps):
            if time.time() > deadline - 0.5:
                break
            a = _constant_magnitude_projection(a)
            a = _project_simplex(a)

        val = _evaluate_fast(a, conv_fft_len, n)
        if val < best_val:

    if best_seq is not None and time.time() < deadline - 5.0:

        a_opt, val_opt = _subgradient_optimize_expgrad(

        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Refine] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee we return a non‑empty sequence.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
