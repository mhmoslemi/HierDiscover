def _lp_iterative_refine(a0, n, conv_fft_len, deadline, max_iter=30, tau_start=0.5, tau_decay=0.9):
    a = a0.copy()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    tau = tau_start
    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()
        # active constraints
        eps = 1e-12
        active = np.where(conv_a >= M - eps)[0]
        extra = np.array([0, len(conv_a)//2, len(conv_a)-1])
        active = np.unique(np.concatenate((active, extra)))
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))
        lp_res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                        method='highs',
                                        options={'presolve': True,
                                                 'time_limit': max(0.0, deadline - time.time())})
        if not lp_res.success:
            print(f"[IterLP {it}] linprog failed: {lp_res.message}")
            break
        b = lp_res.x
        if b.sum() < 1e-12:
            break
        b_norm = b / b.sum()
        a_new = (1 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        val_new = 2.0 * n * _full_convolution(a_new, conv_fft_len).max()
        if val_new < best_val:
            best_val, best_a = val_new, a_new.copy()
            print(f"[IterLP {it}] improved to {val_new:.6f}")
        else:
            # reduce tau if not improved
            tau *= 0.5
            if tau < 1e-4:
                break
        a = a_new
    return best_a, best_val

def _random_local_search(a0, n, conv_fft_len, deadline, max_moves=200_000):
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    T_start = 0.05
    T_end = 1e-5
    for it in range(max_moves):
        if time.time() > deadline:
            break
        # temperature schedule
        T = T_end + (T_start - T_end) * (1 - it / max_moves)
        # pick i with significant mass
        i = np.random.randint(n)
        if a[i] <= 1e-15:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        frac = np.random.rand() * 0.3
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # optional: re-normalize? Since total mass unchanged, sum unchanged.

            a = a2
        else:
            if np.random.rand() < np.exp(-(val2 - best_val) / max(T, 1e-12)):

"""
Hybrid optimiser for the step‑function bound.

The pipeline is:
1. Build a modestly diverse pool of start vectors (warm‑start, Dirichlet,
   exponential, Gaussian, spikes, etc.).
2. For each start (a few best only):
   – Adam optimisation on a smooth‑max surrogate (time‑bounded).
   – Exact sub‑gradient phase (exponentiated‑gradient) to push the max‑conv
     down further.
3. Take the overall best candidate and run a *dedicated LP refinement loop*:
   repeatedly solve the linear programme
        maximise Σ b   s.t.   (f ★ b)_k ≤ M,
   where M is the current peak of f ★ f.
   The solution b is normalised and mixed into f with a decreasing mixing
   factor τ.  Doing many (≈30) iterations typically lowers the peak a few
   more percents.
4. Finish with a cheap random‑local‑search (pairwise mass transfers) that
   accepts improving moves (and occasional uphill moves) – a lightweight
   simulated annealing.
5. Return the best non‑negative sequence found.  All operations keep the
   vector on the probability simplex (Σa_i = 1), which does not affect the
   evaluation because the target function is homogeneous.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) used for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _lp_iterative_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 30,
    tau_start: float = 0.5,
    tau_decay: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Repeated LP improvement: maximise Σb subject to (f★b) ≤ M,
    then mix b (normalised) into f with a decreasing τ.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    tau = tau_start
    eps = 1e-12

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv_a = _full_convolution(a, conv_fft_len)
        M = conv_a.max()

        # active set – indices where the convex peak is attained (plus a few extras)
        active = np.where(conv_a >= M - eps)[0]
        extra = np.array([0, len(conv_a) // 2, len(conv_a) - 1])
        active = np.unique(np.concatenate((active, extra)))

        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            rows.extend([row_idx] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        c = -np.ones(n)                     # maximise Σb  ⇔ minimise −Σb
        bounds = [(0, None)] * n
        b_ub = M * np.ones(len(active))

        lp_res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={
                "presolve": True,
                "time_limit": max(0.0, deadline - time.time()),
            },
        )

        if not lp_res.success:
            print(f"[LPIter {it}] linprog failed: {lp_res.message}")
            break

        b = lp_res.x
        if b.sum() <= 1e-12:
            break
        b_norm = b / b.sum()

        # mixing step
        a_new = (1.0 - tau) * a + tau * b_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        val_new = 2.0 * n * _full_convolution(a_new, conv_fft_len).max()
        if val_new < best_val - 1e-12:
            best_val, best_a = val_new, a_new.copy()
            print(f"[LPIter {it}] improved to {val_new:.6f}")
        else:
            # shrink mixing factor if no progress
            tau *= 0.5
            if tau < 1e-4:
                break

        a = a_new

    return best_a, best_val


def _random_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing style local search.
    Randomly moves a fraction of mass from one index to another.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    T_start = 0.08
    T_end = 1e-5

    for it in range(max_moves):
        if time.time() > deadline:
            break
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        frac = np.random.rand() * 0.4   # move up to 40 % of a[i]
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        # total mass unchanged, no need to renormalise
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val - 1e-12:
            best_val, best_a = val2, a2.copy()
            a = a2
        else:
            if np.random.rand() < np.exp(-(val2 - best_val) / max(T, 1e-12)):
                a = a2

    return best_a, best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Return a non‑negative step‑function sequence (list of floats)."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096                     # length of the sequence
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from a previously discovered construction, if available.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Build a rich pool of start vectors.
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # Add a few extra random permutations / block‑wise patterns.
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 12   # only the most promising starts will be fully refined

    # --------------------------------------------------------------
    # Phase I – deep optimisation of a few promising starts.
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Allocate a small time slice for Adam (≈8 s) and sub‑gradient (≈8 s).
        sub_dead = min(deadline, time.time() + 8.0)

        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # Sub‑gradient phase (if still time)
        if time.time() < deadline - 5:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=300_000, eta0=0.8,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase II – dedicated multi‑iteration LP refinement on the best candidate.
    if best_seq is not None and time.time() < deadline - 10:
        print("[Final LP] Starting iterative LP refinement")
        a_lp, val_lp = _lp_iterative_refine(
            best_seq, n, conv_fft_len, deadline, max_iter=30, tau_start=0.5, tau_decay=0.9
        )
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[Final LP] improved to {val_lp:.6f}")

    # --------------------------------------------------------------
    # Phase III – random local search (light simulated annealing).
    if best_seq is not None and time.time() < deadline - 5:
        print("[Local Search] Running random local search")
        a_rand, val_rand = _random_local_search(
            best_seq, n, conv_fft_len, deadline, max_moves=200_000
        )
        if val_rand < best_val:
            best_val, best_seq = val_rand, a_rand.copy()
            print(f"[Local Search] improved to {val_rand:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (0 ≤ x ≤ 1000).
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```

This implementation follows the hybrid strategy described in the prompt, adds a deeper **iterative LP refinement** (up to 30 iterations with a decreasing mixing factor) and a **random local search** phase.  The extra stages give the optimiser more opportunities to shave off the final few hundredths of the bound while staying comfortably within the 1000 s time budget.
