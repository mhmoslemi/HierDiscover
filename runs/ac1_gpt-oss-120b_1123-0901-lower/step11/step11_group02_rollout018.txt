"""
Hybrid optimiser with a Fourier‑based “optimal” start.

The algorithm
----------------
1. Pick a moderately large dimension (default 8192).  Larger n lowers the
   theoretical lower bound 2 n · max(conv) → 1, so we have more freedom.
2. Build a pool of diverse initial guesses:
   • warm‑start (if supplied)
   • a discretisation of the analytically optimal density whose
     self‑convolution is (almost) constant (obtained by an inverse FFT of
     √(sin ω/ω)·e^{-iω/2})
   • several Beta‑shapes (including the arcsine β(½,½) and more extreme
     exponents)
   • a few random Dirichlet draws.
3. For each candidate (a few only – the larger n makes each optimisation
   expensive) run a fast Adam optimiser on a smooth‑max surrogate of the
   peak of the self‑convolution.
4. Refine the best vector with a long, deterministic mass‑transfer local
   search that repeatedly moves a small amount of probability mass from the
   coordinate with the largest sub‑gradient to the one with the smallest.
5. If time permits up‑sample the best vector to 16384 points and repeat
   steps 3–4.
6. Clip to the evaluator’s allowed range and return the best sequence
   found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions ----------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories -------------------------------------------------

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_beta(n, alpha, beta):
    """Symmetric (or asymmetric) Beta pdf on a uniform grid."""
    i = (np.arange(n) + 0.5) / n          # mid‑points, avoids 0/1
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_optimal_continuous(n):
    """
    Discretisation of the (theoretically) optimal density whose
    self‑convolution is constant on [0,2].  The continuous Fourier
    transform is φ(ω)=sqrt(sin ω/ω)·e^{-iω/2}; we obtain samples by an IFFT.
    """
    # oversample the frequency domain to reduce aliasing
    L = _next_pow2(8 * n)               # generous zero‑padding
    i = np.arange(L)
    # frequencies centred at zero, step = 2π/L
    omega = 2.0 * np.pi * (i - L // 2) / L

    # safe compute sin(ω)/ω, with limit = 1 at ω=0
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = np.sin(omega) / omega
    ratio = np.where(omega == 0.0, 1.0, ratio)

    # take the principal complex square‑root (allows negative values)
    phi = np.sqrt(ratio.astype(np.complex128)) * np.exp(-1j * omega / 2.0)

    # inverse FFT → time domain (samples on [0,1))
    f_complex = np.fft.ifft(np.fft.ifftshift(phi))
    f_real = np.real(f_complex)

    # undo the 0.5 shift that was introduced in φ
    f_real = np.roll(f_real, -L // 2)

    a = f_real[:n]
    a = np.maximum(a, 0.0)               # clip any tiny negative ringing
    if a.sum() < 1e-12:
        a = _make_uniform(n)
    else:
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.2                # temperature for smooth‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 5e6)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Fine‑grained heuristic that repeatedly moves a fraction of mass
    from the coordinate with the largest sub‑gradient to the one with
    the smallest.  Moves are accepted only if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        if grad[i_high] <= grad[i_low] + 1e-15:
            break   # stationary

        src_mass = a[i_high]
        improved = False
        for frac in (0.5, 0.35, 0.2, 0.1, 0.05, 0.02, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            if a_new[i_high] < -1e-14:
                continue
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # small random shake to escape plateaus
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.04
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 18.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=200_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] refined best {val_big:.6f}")

            # a short final mass‑transfer on the big grid
            if time.time() < deadline - 5:
                a_big, val_big = _mass_transfer_optimize(
                    best_seq, n_big, conv_fft_len_big, corr_fft_len_big,
                    deadline, max_iters=200_000,
                )
                if val_big < best_val:
                    best_val = val_big
                    best_seq = a_big.copy()
                    print(f"[Upsample‑MT] improved best {val_big:.6f}")
        except Exception as e:
            print("[Upsample] exception:", e)

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final safety clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Exact evaluation (for logging only)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print("Final evaluation failed:", e)

    return best_seq.tolist()
