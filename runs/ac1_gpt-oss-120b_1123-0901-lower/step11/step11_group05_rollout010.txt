def propose_candidate(seed=0, budget_s=1000.0):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose length based on budget (increase if enough time)
    n = 8192

    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    # Build candidates (...)
    candidates = []
    # Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))
    # arcsine
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1 - x) + 1e-20) # avoid zero
    a = a / a.sum()
    candidates.append(a)
    # beta shapes
    alphas = np.linspace(0.2, 0.9, 15)
    for alpha in alphas:
        a = np.power(x, alpha - 1) * np.power(1 - x, alpha - 1)
        a = a / a.sum()
        candidates.append(a)

    # Dirichlet draws
    for _ in range(30):
        conc = np.random.uniform(0.1, 4.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)
    
    # Warm start if global variable present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.size != n:
            # Resample or pad
            if warm_arr.size > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.size), mode='constant')
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() < 1e-6:
            warm_arr = np.random.dirichlet(np.ones(n))
        warm_arr = warm_arr / warm_arr.sum()
        candidates.append(warm_arr)
    
    # Evaluate candidates
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Init] val={val:.6f}")
    # subgradient optimization
    sub_deadline = deadline - 15.0
    if time.time() < sub_deadline:
        a_opt, val_opt = _expgrad_optimize(best_a, n, conv_fft_len, corr_fft_len, sub_deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[ExpGrad] improved to {val_opt:.6f}")

    # Adam fine-tune
    adam_deadline = deadline - 5.0
    if time.time() < adam_deadline:
        a_opt2, val_opt2 = _adam_optimize(best_a, n, conv_fft_len, corr_fft_len, adam_deadline)
        if val_opt2 < best_val:
            best_val, best_a = val_opt2, a_opt2.copy()
            print(f"[Adam] improved to {val_opt2:.6f}")

    # local stochastic search
    local_deadline = deadline - 0.5
    while time.time() < local_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move small random mass
        delta = np.random.rand() * 0.001 * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local] improved to {val:.6f}")

    # final: clip to max allowed 1000 (should be fine)
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _project_simplex(best_a) # re-normalise

    # evaluate official
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f}, length {len(best_a)}")
    return best_a.tolist()

def _expgrad_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=10_000_000, eta0=1.0):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it=0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100000 == 0:
                print(f"[ExpGrad it={it}] new best {val:.6f}")
        # subgradient: determine active indices
        active = np.isclose(conv, max_c, atol=1e-14)
        # there might be multiple maxima; allocate weight equally
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # learning rate
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        it += 1
    return best_a, best_val

local search is performed.  The whole routine respects the supplied time
budget (≈ 1000 s) and always returns a valid list of floats.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx = 1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

def _expgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      max_iters: int = 2_000_000,
                      eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimisation on the exact max‑conv."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the maximum up to floating tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)            # mirror‑descent update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature
    lam_factor = 1.03
    lam_max = 1e12

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # obtain a from logits (softmax)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        # optional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
