"""
Hybrid optimizer for the non‑negative step‑function problem.

Goal: minimise  2·n·max(conv(a,a)) / (sum(a)^2)  with a_i ≥ 0.
The algorithm proceeds as follows:

1.  Build a diverse pool of probability vectors (uniform, Dirichlet,
    exponential tails, sparse spikes, etc.) and keep the best seed.
2.  Run a long Adam optimisation on a smooth‑max surrogate of the
    max‑convolution.  The temperature λ is increased very slowly so that
    the surrogate stays close to the true max for most of the run.
3.  Switch to a *projected gradient descent* on the exact max‑convolution:
    we compute the sub‑gradient 2·(w★a) (w is the normalised indicator of
    the active set where the convolution attains its maximum) and take
    a step on the simplex via Euclidean projection.  The step‑size
    decays as η₀/√t.
4.  Finish with a short simulated‑annealing pass to escape tiny local
    minima.
All phases respect the global 1000 s budget and the best sequence found
is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _make_initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    pool = []

    # warm start (if supplied)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (both sides)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law family (triangular product)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑Gaussian spikes
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # A few sparse spike candidates
    for k in (32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    return pool


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # learning rate
    lam = 0.05                        # initial temperature
    lam_factor = 1.008                # slow increase → smoother surrogate

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        # --- retrieve current probability vector from logits ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & objective (surrogate uses only max_c)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (softmax over λ·(conv−max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    # final exact evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    return best_a, best_val


# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  corr_fft_len: int,
                  deadline: float,
                  eta0: float = 0.5,
                  max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    Uses the sub‑gradient 2·(w★a) where w is the uniform weight on the
    active set of maximal convolution entries.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        # compute current convolution and active set
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        # form sub‑gradient weight vector
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient of max_c
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # gradient step and simplex projection
        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        # evaluate
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()

        # keep best
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            if it % 10_000 == 0:
                print(f"[PGD it={it:07d}] new best {val_new:.6f}")

        # always move to the new point (even if not improving) to keep exploration
        a = a_new

        it += 1
        if it % 1_000 == 0 and time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 120_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096                                   # problem dimension
    conv_fft_len = _next_pow2(2 * n - 1)       # 8192
    corr_fft_len = _next_pow2(3 * n - 2)       # 16384

    # --------------------------------------------------------------
    # Warm start (if available from previous run)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build initial pool and keep the best seed (exact evaluation)
    pool = _make_initial_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() > deadline:
            break
        if idx % 8 == 0:
            print(f"[Init] candidate {idx} val={val:.6f}")

    # --------------------------------------------------------------
    # Phase I – Adam optimisation (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase I] Adam optimisation started")
        sub_dead = deadline - 5.0
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=600_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase II – Projected Gradient Descent on exact max‑convolution
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase II] Projected gradient descent started")
        sub_dead = deadline - 5.0
        a_pgd, val_pgd = _pgd_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            eta0=0.5, max_iters=2_000_000
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[Phase II] PGD improved to {val_pgd:.6f}")

    # --------------------------------------------------------------
    # Phase III – Simulated annealing refinement
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase III] Simulated annealing final pass")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=120_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase III] improved best to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Safety net – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the validator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
