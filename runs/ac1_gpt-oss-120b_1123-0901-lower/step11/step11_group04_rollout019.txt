def _lp_refine(a, n, M, time_limit):
    import cvxpy as cp
    b = cp.Variable(n)
    constraints = [b >= 0]
    # Build constraints for each k
    for k in range(2*n - 1):
        # Determine range of i
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # Build expression sum_{i=i_start}^{i_end} a[i] * b[k-i]
        # Use list comprehension
        expr = cp.sum([a[i] * b[k - i] for i in range(i_start, i_end + 1)])
        constraints.append(expr <= M)
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    # Set solver options for time limit: e.g., CP solver param warm_start, etc.
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, warm_start=True, verbose=False)
    except Exception as e:
        # fallback
        return a
    if b.value is None:
        return a
    b_opt = np.maximum(b.value, 0.0)
    sum_b = b_opt.sum()
    if sum_b < 1e-12:
        return a

def propose_candidate(seed=0, budget_s=1000):

                warm = np.pad(warm, (0, n-warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # initial candidates
    candidates = []
    if warm is not None:
        candidates.append(warm)

    # arcsine

    # uniform, triangle
    candidates.append(_make_uniform(n))
    candidates.append(_make_triangle(n))

    best_seq = None

    # main loop
    for idx, a0 in enumerate(candidates):
        if time.time() > deadline:
            break
        # Normalise
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Phase allocation based on remaining time
        rem = deadline - time.time()
        # allocate fractions
        phases = [
            ('l2', 0.25),
            ('adam', 0.25),
            ('expgrad', 0.25),
            ('gd', 0.15),
            ('anneal', 0.10),
        ]
        cur_seq = a0.copy()
        cur_val = 2.0 * n * _full_convolution(cur_seq, conv_fft_len).max()
        # ... run each phase if time left
        for phase_name, frac in phases:
            if time.time() > deadline:
                break
            time_alloc = min(rem * frac, deadline - time.time())
            sub_deadline = time.time() + time_alloc
            if phase_name == 'l2':
                cur_seq, cur_val = _adam_optimize_l2(cur_seq, n, conv_fft_len, corr_fft_len,
                                                     target, sub_deadline,

                cur_seq, cur_val = _adam_optimize(cur_seq, n, conv_fft_len, corr_fft_len,

            elif phase_name == 'anneal':

            # update best

                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Candidate {idx} {phase_name}] new best {cur_val:.6f}")
            # recompute remaining time

"""
Hybrid optimisation for the max‑convolution objective.

* Build a small diverse pool of seed sequences (including a
  “U‑shaped” power‑law family a_i ∝ ((i+1)*(n‑i))^{‑β} with several β).
* Pick the seed with the lowest raw objective (fast FFT evaluation).
* Spend the remaining budget on a multi‑stage refinement:
    - L2‑flattening (Adam on the squared‑error to the uniform target).
    - Smooth‑max Adam (log‑sum‑exp surrogate of the max‑convolution).
    - Exact‑max exponentiated‑gradient sub‑gradient.
    - Projected gradient descent on the true max‑convolution.
    - A short simulated‑annealing fine‑tune.
* All steps respect the global time budget; the best sequence found is
  cached in the global variable ``height_sequence_1`` for warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_beta(n, beta):
    """U‑shaped power‑law a_i ∝ ((i+1)*(n‑i))^{‑beta}."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    j = np.arange(n, 0, -1, dtype=np.float64)        # n … 1
    a = (i * j) ** (-beta)
    a = np.where(np.isfinite(a), a, 0.0)
    a /= a.sum()
    return a

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional log
            if it % 100_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _adam_optimize_l2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    target: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the L2 distance to the uniform convolution target."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        residual = conv - target
        grad_a = 4.0 * _cross_correlation(residual, a, corr_fft_len)

        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[L2Adam it={it:06d}] new best {val:.6f}")

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function with a low evaluation score.
    The routine builds a few informative seeds, picks the most promising one
    and then spends the remaining time on a multi‑stage refinement (L2‑Adam,
    smooth‑max Adam, exponentiated‑gradient, projected GD, annealing).
    The best sequence found is cached globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 4096                                            # problem size
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from previous run, if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of initial vectors
    pool = []
    if warm is not None:
        pool.append(warm.copy())

    # power‑law “U‑shaped” family with several β values

        pool.append(_make_beta(n, beta))

    # a few classical families
    pool.append(_make_uniform(n))
    pool.append(_make_triangle(n))
    pool.append(_make_exponential(n, gamma=0.5))
    pool.append(_make_powerlaw(n, expo=-1.2))

    # random Dirichlets with different concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0):
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    # Quick scan: pick the seed with the lowest raw objective
    best_seq = None
    best_val = float("inf")
    for idx, a in enumerate(pool):
        a = np.maximum(a, 0.0)

        a = a / a.sum()
        val = _objective_value(a, n, conv_fft_len)
        print(f"[Seed {idx}] raw value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"[Info] best seed value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Allocate the remaining budget among refinement stages
    remaining = deadline - time.time()
    if remaining <= 0:
        # fallback – return the best seed we already have
        final_seq = best_seq.clip(0.0, 1000.0) / best_seq.sum()
        globals()["height_sequence_1"] = final_seq.tolist()
        return final_seq.tolist()

    # time fractions (tuned empirically)
    t_l2 = remaining * 0.30
    t_adam = remaining * 0.20
    t_exp = remaining * 0.25
    t_gd = remaining * 0.15
    t_ann = remaining * 0.10

    # Uniform target for the L2 flattening phase
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 1 – L2 flattening (smooths the whole convolution)
    sub_dead = min(deadline, time.time() + t_l2)
    best_seq, best_val = _adam_optimize_l2(
        best_seq, n, conv_fft_len, corr_fft_len, target, sub_dead,
        max_iters=500_000, lr=0.07
    )
    print(f"[Phase L2] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max Adam (fine‑tuning the true max)
    sub_dead = min(deadline, time.time() + t_adam)
    best_seq, best_val = _adam_optimize(
        best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
        max_iters=400_000
    )
    print(f"[Phase Adam] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact‑max exponentiated‑gradient
    sub_dead = min(deadline, time.time() + t_exp)
    best_seq, best_val = _subgradient_optimize_expgrad(
        best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
        max_iters=800_000, eta0=1.2
    )
    print(f"[Phase ExpGrad] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected gradient descent on the true max‑conv
    sub_dead = min(deadline, time.time() + t_gd)
    best_seq, best_val = _max_conv_descent(
        best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
        max_iters=300_000
    )
    print(f"[Phase GD] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – short annealing refinement
    sub_dead = min(deadline, time.time() + t_ann)
    best_seq, best_val = _anneal_optimize(
        best_seq, n, conv_fft_len, sub_dead, max_moves=150_000
    )
    print(f"[Phase Anneal] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety – ensure a valid return
    if best_seq is None:
        best_seq = _make_uniform(n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    globals()["height_sequence_1"] = best_seq.tolist()

    # ------------------------------------------------------------------
    # Final evaluation using the provided checker (for reporting)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Evaluation error: {exc}")
        print(f"Best internal value {best_val:.6f}")

    return best_seq.tolist()
