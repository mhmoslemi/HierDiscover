def simplex_projection(v, z=1.0):
    """Project v onto simplex {x >= 0, sum x = z}."""
    n = len(v)
    if v.sum() == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def fast_score(seq):
    a = np.clip(seq, 0, 1000)  # but seq already nonnegative small
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    max_b = conv.max()
    return 2 * n * max_b / (sum_a**2)
```

def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    # compute convolution via FFT
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)
```

def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # choose length
    n = 2000
    # initial random distribution
    cur = np.random.rand(n)
    cur /= cur.sum()
    best = cur.copy()
    best_score = fast_score(best)
    cur_score = best_score
    temp = 1.0
    decay = 0.9997
    step_scale = 0.1
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # choose two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # compute delta as fraction of source
        if cur[i] == 0:
            continue
        delta = cur[i] * step_scale * np.random.rand()
        # ensure delta <= cur[i]
        # propose new sequence
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # no need to renormalize (sum unchanged)
        # compute score
        new_score = fast_score(new)
        # acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        # update temperature
        temp *= decay
        # adapt step scale occasionally
        if iter_count % 2000 == 0:
            # if no improvement over last 2000 steps reduce step scale
            step_scale = max(step_scale * 0.9, 0.001)
            # also optionally increase if improvements plateau
            # Print progress
            print(f"Iter {iter_count}, best_score {best_score:.6f}, cur_score {cur_score:.6f}, temp {temp:.6f}, step {step_scale:.5f}")
    print(f"Finished after {iter_count} iterations, best_score {best_score:.6f}")
    return best.tolist()

"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(f, f)) / (sum(f) ** 2)

where `conv` is the ordinary (linear) convolution.
The algorithm starts from a random distribution with unit ℓ₁‑norm and
performs a simple simulated‑annealing / coordinate‑descent search.
At each iteration a small amount of mass is moved from one index to another,
the new sequence is accepted according to a decreasing temperature schedule
(if it improves the objective or with a Metropolis probability otherwise).
All updates keep the sum constant and respect non‑negativity, and the
convolution is evaluated using an FFT‑based routine for speed.
The search runs until the allocated time budget (≈1000 s) is exhausted and
returns the best sequence found.
"""

import numpy as np
import time
import math

# --------------------------------------------------------------
# Fast evaluation that mimics `evaluate_sequence` but uses FFT.
# --------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the same value as `evaluate_sequence` for a 1‑D array."""
    a = np.clip(seq, 0.0, 1000.0)                 # security checks
    s = a.sum()
    if s < 0.01:                                 # avoid div‑by‑zero
        return float('inf')
    n = a.shape[0]
    # length for a power‑of‑two FFT (at least 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# --------------------------------------------------------------
# Simple simulated‑annealing optimizer on the simplex.
# --------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds early).

    Returns
    -------
    list[float]
        The best height sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave a small safety margin

    # ----------------------------------------------------------
    # Problem size.  A few thousand entries give enough freedom
    # while keeping each iteration cheap.
    # ----------------------------------------------------------
    n = 2000                     # you may increase this up to ~7000
    # start from a random point on the simplex
    cur = np.random.rand(n)
    cur /= cur.sum()
    best = cur.copy()
    best_score = _fast_score(best)
    cur_score = best_score

    # annealing parameters
    temperature = 1.0            # initial temperature
    decay = 0.9997               # exponential cooling
    step_fraction = 0.15         # typical proportion of source mass to move
    iter_cnt = 0
    last_improve_iter = 0

    # ----------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        # amount to transfer (never exceed the source entry)
        delta = cur[i] * step_fraction * np.random.rand()
        delta = min(delta, cur[i])

        # propose a new point on the simplex (exactly preserves the sum)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # evaluate
        new_score = _fast_score(new)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability ~ exp(-(Δ)/T)
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                last_improve_iter = iter_cnt

        # temperature schedule
        temperature *= decay

        # adapt step size every few thousand iterations
        if iter_cnt % 2000 == 0:
            # if we have not improved for a while, shrink the step
            if iter_cnt - last_improve_iter > 20000:
                step_fraction = max(step_fraction * 0.7, 0.005)
            else:
                # otherwise allow a slightly larger step to explore
                step_fraction = min(step_fraction * 1.05, 0.3)

            print(
                f"Iter {iter_cnt:6d} | best {best_score:.6f} | cur {cur_score:.6f} "
                f"| temp {temperature:.5e} | step {step_fraction:.4f}"
            )

    print(f"Search finished after {iter_cnt} iterations – best score: {best_score:.6f}")
    return best.tolist()
