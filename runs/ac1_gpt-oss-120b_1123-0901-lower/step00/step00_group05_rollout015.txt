    u = sorted(v, reverse=True)
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - s) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    best_seq = None
    best_score = float('inf')
    # define candidate lengths
    lengths = [500, 1000, 2000, 4000, 6000]
    # generate geometric sequences
    for n in lengths:
        seq = generate_geometric_sequence(n)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
    # optionally refine best_seq with gradient descent
    if best_seq is not None:
        # convert to numpy array
        a = np.array(best_seq, dtype=np.float64)
        # compute constants
        n = len(a)
        # gradient descent loop
        iteration = 0
        while time.time() < deadline:
            # Compute convolution a*a
            b = fft_convolve(a, a)   # length 2n-1
            # compute error from uniform constant
            const = 1.0/(2*n - 1)
            error = b - const
            # compute gradient
            rev_a = a[::-1]
            conv = fft_convolve(error, rev_a)
            grad = 4.0 * conv[n-1:n-1+n]  # length n
            # step size schedule
            iteration += 1
            eta = 0.3 / np.sqrt(iteration)
            a = a - eta * grad
            # projection onto simplex (sum=1)
            a = project_onto_simplex(a, 1.0)
            # occasionally evaluate
            if iteration % 10 == 0:
                seq = a.tolist()
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
        # final ensure best_seq assigned
    if best_seq is None:
        # fallback random
        best_seq = [float(np.random.rand()) for _ in range(1000)]
    print('Best score:', best_score, ' length:', len(best_seq))
    return best_seq

Propose a candidate step‐function sequence.

The algorithm generates several random initial distributions, then performs
projected gradient descent on a smooth approximation of the L∞ convolution
norm.  The loss is   L = log‑sum‑exp(γ·(a*a)) / γ,
whose gradient can be computed efficiently with a few FFT convolutions.
After each gradient step we project onto the probability simplex (non‑negative,
sum=1).  The best sequence seen across all lengths and restarts is returned.

import numpy as np
import time

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    n = v.shape[0]
    # shift to guarantee at least one solution
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_convolve(x, y):
    """Convolution via real FFT (returns length len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    # next power of two for speed
    L = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = X * Y
    conv = np.fft.irfft(Z, n=L)
    return conv[:n]

def _smooth_max_step(a, gamma, lr):
    """
    One gradient‑descent step on
        L(a)= (1/γ) log Σ exp(γ·(a*a))
    The gradient is   ∇L = 2 * (w * a_rev)   where w is the soft‑max of a*a.
    After the step we project the vector back onto the simplex.
    Returns the projected vector and the current raw maximum entry of a*a.
    """
    n = a.shape[0]
    rev_a = a[::-1]

    # convolution a*a
    b = _fft_convolve(a, a)  # length 2n‑1
    # smooth maximum via log‑sum‑exp
    max_b = b.max()
    b_shift = b - max_b          # improve numerical stability
    w = np.exp(gamma * b_shift)

    a_proj = _project_onto_simplex(a_new, s=1.0)
