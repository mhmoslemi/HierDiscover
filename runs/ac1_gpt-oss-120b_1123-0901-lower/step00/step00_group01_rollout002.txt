def fft_convolve(a,b):
    size = len(a) + len(b) - 1
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    conv = np.fft.irfft(fa * fb, m)
    return conv[:size]   # only needed part

"""
Mirror-descent optimiser for the self‑convolution max problem.
The algorithm keeps the distribution on the probability simplex,
minimises a smoothed max of the self‑convolution (via soft‑max)
using exponentiated‑gradient updates, and restarts with random
initialisations until the time budget expires. The best sequence
found (as a list of floats) is returned.
"""

import time
import numpy as np

def _fft_convolve(a, b):
    """Convolution of two real vectors using an FFT of power‑of‑two length."""
    size = len(a) + len(b) - 1
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    conv = np.fft.irfft(fa * fb, m)
    return conv[:size]

def _project_to_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v≥0). """
    # Implementation based on the algorithm from
    # Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    evaluate_sequence().  The method runs a mirror‑descent
    optimiser on a smoothed max of the self‑convolution.
    Several random restarts are performed until the time
    budget expires; the best sequence found is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a small safety margin

    # Length of the step function – use the previously known length if possible
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2000  # a safe default

    best_seq = None
    best_score = np.inf
    run_id = 0

    # Hyper‑parameters (tuned empirically)
    max_outer_iters = 1000            # how many restarts we will try at most
    max_inner_iters = 5000           # iterations per restart
    eta0 = 0.5                       # base step size for exponentiated gradient
    alpha_start = 2.0                # soft‑max temperature at the beginning
    alpha_end = 80.0                 # temperature at the end of a run

    while time.time() < deadline and run_id < max_outer_iters:
        run_id += 1

        # -----------------------------------------------------------------
        # 1) Initialise a random probability vector on the simplex.
        # -----------------------------------------------------------------
        p = np.random.rand(n).astype(np.float64)
        p = p / p.sum()

        # -----------------------------------------------------------------
        # 2) Main optimisation loop.
        # -----------------------------------------------------------------
        for it in range(max_inner_iters):
            # Time‑out check
            if time.time() >= deadline:
                break

            # --- self‑convolution (FFT) ------------------------------------
            conv_self = _fft_convolve(p, p)          # length 2n‑1
            max_conv = conv_self.max()
            # The true evaluate_sequence value for a normalized vector (sum=1)
            # is simply 2 * n * max_conv.
            cur_score = 2.0 * n * max_conv

            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(f"[{time.time():.1f}] Run {run_id:03d} iter {it:04d}  best = {best_score:.6f}")

            # --- soft‑max weights over convolution entries ------------------
            # α grows linearly from alpha_start to alpha_end.
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_inner_iters)
            # Stabilise exponentials
            c_shift = conv_self - conv_self.max()
            w = np.exp(alpha * c_shift)

            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum

            # q_j = Σ_i p_i * w_{i+j}

            conv_pw = _fft_convolve(p_rev, w)       # length 3n‑2
            q = conv_pw[n - 1 : n - 1 + n]          # slice of length n
            grad = q + p

            # Step size decays like 1/√t.
            eta = eta0 / np.sqrt(it + 1)
            # Prevent overflow/underflow in the exponent.

    return best_seq.tolist()

"""
Mirror‑descent optimiser for the self‑convolution max problem.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is invariant to scaling.  By normalising the vector to the
probability simplex (sum = 1) the expression becomes simply
`2 * n * max(conv(a, a))`.  The algorithm therefore seeks a
probability distribution whose self‑convolution is as flat as possible.

A smooth approximation of the max (soft‑max with a temperature
parameter α) is minimised by exponentiated‑gradient (mirror descent)
updates on the simplex.  Several random restarts are performed until
the allotted time budget expires; the best sequence encountered is
returned.

The implementation uses only NumPy and the standard library, runs
well inside the 1000 s budget and does not rely on any external files.
"""

import time
import numpy as np


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Convolution of two real 1‑D arrays via FFT.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    size = len(a) + len(b) - 1
    # next power of two – faster FFTs
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    conv = np.fft.irfft(fa * fb, m)
    return conv[:size]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  A mirror‑descent optimiser on a
    smoothed max of the self‑convolution is run with many random
    restarts; the best sequence found before the time budget runs out
    is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # -----------------------------------------------------------------
    # Determine the length of the candidate.  Use the length of the
    # previously‑known construction if it exists; otherwise fall back to
    # a default of 2000.
    # -----------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the evaluation harness
    except Exception:
        n = 2000

    # -----------------------------------------------------------------
    # Hyper‑parameters (tuned empirically for the 6000‑element regime)
    # -----------------------------------------------------------------
    max_outer_iters = 200          # how many random restarts at most
    max_inner_iters = 20000        # iterations per restart (stops early on timeout)
    eta0 = 0.5                     # base step size for the exponentiated gradient
    alpha_start = 2.0              # soft‑max temperature at iteration 0
    alpha_end = 80.0               # temperature at the final iteration

    best_seq = None
    best_score = np.inf
    run_id = 0

    # -----------------------------------------------------------------
    # Main optimisation loop – keep restarting while time permits.
    # -----------------------------------------------------------------
    while time.time() < deadline and run_id < max_outer_iters:
        run_id += 1

        # ----- 1) Random start on the simplex --------------------------------
        p = np.random.rand(n).astype(np.float64)
        p /= p.sum()                       # now p is a probability vector

        # ----- 2) Iterative mirror‑descent ------------------------------------
        for it in range(max_inner_iters):
            # Stop if we have exhausted the budget.
            if time.time() >= deadline:
                break

            # ---- self‑convolution (FFT) -------------------------------------
            conv_self = _fft_convolve(p, p)          # length 2n‑1
            max_conv = conv_self.max()
            cur_score = 2.0 * n * max_conv           # exact value returned by evaluate_sequence

            # Keep the best sequence seen so far.
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(
                    f"[{time.time():.1f}] run {run_id:03d}  it {it:05d}  "
                    f"best={best_score:.6f}"
                )

            # ---- soft‑max weights over convolution entries --------------------
            # α grows linearly from alpha_start to alpha_end.
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_inner_iters)

            # Stabilise the exponentials by subtracting the max entry.
            shifted = conv_self - conv_self.max()
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                # Extremely unlikely, but keep the algorithm alive.
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ---- gradient of the smoothed max (q + p) -----------------------
            # q_j = Σ_i p_i * w_{i+j}
            p_rev = p[::-1]                                 # reversed copy of p
            conv_pw = _fft_convolve(p_rev, w)              # length 3n‑2
            q = conv_pw[n - 1 : n - 1 + n]                 # slice of length n
            grad = q + p                                    # gradient of the smoothed objective

            # ---- exponentiated‑gradient (mirror descent) --------------------
            eta = eta0 / np.sqrt(it + 1)                   # 1/√t decay
            step = -eta * grad
            # Clip to avoid overflow/underflow in the exponential.
            step = np.clip(step, -50.0, 50.0)
            p = p * np.exp(step)

            # ---- re‑normalise onto the simplex (positivity is guaranteed) -
            total = p.sum()
            if total == 0 or np.isnan(total):
                # Degeneration – restart from a fresh random point.
                p = np.random.rand(n)
                p /= p.sum()
            else:
                p /= total

        # -----------------------------------------------------------------
        # End of a single restart – a new random start will be tried if
        # time permits.
        # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # Fallback: if, for any reason, no sequence improved (should never
    # happen), return the uniform distribution.
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert the NumPy array to a plain Python list as required by the
    # external evaluator.
    return best_seq.tolist()
