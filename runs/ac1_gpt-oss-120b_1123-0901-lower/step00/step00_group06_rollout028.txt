def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Initialize best_seq using inverse sqrt shape
    n = 8000
    i = np.arange(n, dtype=float)
    seq = 1.0/np.sqrt(i+1)   # shape p=0.5
    best_seq = seq.tolist()
    best_score = evaluate_sequence(best_seq)

    # main loop
    t = 0.05  # blending factor
    sqrt2n = np.sqrt(2*n)
    # Precompute maybe
    while time.time() < deadline:
        # normalize current seq
        sum_seq = np.sum(seq)
        if sum_seq < 1e-12:
            break
        f_norm = seq * (sqrt2n / sum_seq)
        # compute bound M
        conv_f = np.convolve(f_norm, f_norm)
        M = np.max(conv_f)
        # solve LP for direction g
        g = solve_convolution_lp(f_norm, M)
        if g is None:
            # random tweak
            idx = np.random.randint(n)
            seq[idx] = max(0, seq[idx] + np.random.randn()*0.01)
            continue
        # normalize g
        sum_g = np.sum(g)
        if sum_g < 1e-12:
            continue
        g_norm = g * (sqrt2n / sum_g)
        # update seq: blend
        seq = (1 - t) * seq + t * g_norm
        # evaluate
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"New best score {best_score:.6f} at time {time.time():.2f}")
            # optionally reduce t for fine-tuning
            if t > 0.001:
                t *= 0.9
        # optional adjust t
    return best_seq

    i = np.arange(n, dtype=np.float64) + 1.0
    return (i ** (-p))


def _grid_seq_sqrt(n, beta):
    """Shifted inverse‑sqrt a[i] = 1 / sqrt(i + β)."""
    i = np.arange(n, dtype=np.float64)
    return 1.0 / np.sqrt(i + beta)


def _initial_candidates():

    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 15000]
    # power‑law around the optimal p = 0.5

def _perturb(seq, rng, step_scale=0.05):
    """Create a neighbour by moving a small amount of mass."""
    n = len(seq)
    new_seq = seq.copy()
    i = rng.integers(0, n)
    # relative change, keep positivity
    delta = rng.normal() * step_scale * new_seq[i]
    new_val = max(0.0, new_seq[i] + delta)
    new_seq[i] = new_val
    return new_seq
