def project_onto_simplex(v, s=1.0):
    """Projection of vector v onto the simplex {x >=0, sum x = s}."""
    # Implementation by sorting.
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
A heuristic optimizer that tries to flatten the auto‑convolution of a
non‑negative sequence.  The objective of evaluate_sequence() is
    2 * n * max(conv) / (sum(seq)^2)
which, after fixing sum(seq)=1, is proportional to max(conv).  The
algorithm therefore minimises max(conv) under the simplex constraint.
It combines a quadratic (L2) loss that pushes the whole convolution
towards its average value with a sub‑gradient of the current peak.
Updates are performed with multiplicative (mirror‑descent) steps and a
projection onto the probability simplex.  Random perturbations and
occasional restarts help escape local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x >= 0, sum(x)=s}
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex of size s."""
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        # all entries zero – return uniform on the simplex
        return np.full_like(v, s / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Search for a step‑function (list of non‑negative heights) with a
    low value of evaluate_sequence().  The routine runs for at most
    *budget_s* seconds and returns the best sequence found.

    The method:
      * start from a random point on the simplex (sum = 1)
      * repeatedly:
          - compute the convolution of the sequence with itself
          - build a gradient that tries to flatten the convolution:
              + an L2‑loss term (conv – average) and
              + a sub‑gradient of the max‑entry of the convolution
          - apply a multiplicative mirror‑descent step and renormalise
          - occasionally add a small random perturbation or restart
          - keep the best sequence seen so far
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # spare a little margin

    # ------------------------------------------------------------------
    # Parameters (feel free to tune)
    n = 3000                     # length of the sequence (≈ current 6k)
    eta0 = 1.0                   # initial step size for mirror descent
    alpha = 0.5                  # blend factor between the two gradients
    perturb_interval = 200       # how often (iterations) to add noise
    restart_interval = 3000      # restart if stuck for many iterations
    max_no_improve = 5000        # after this many iters without improvement -> restart

    # Initial point – random point on the simplex (sum = 1)
    a = np.random.rand(n)
    a = a / a.sum()

    # Keep the best discovered sequence
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    it = 0
    no_improve_counter = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Convolution and basic statistics
        conv = np.convolve(a, a)                     # length 2n-1
        max_c = conv.max()
        k_max = conv.argmax()                        # index of the peak
        # target average of the convolution (sum(conv) = 1)
        avg = 1.0 / (2 * n - 1)
        err = conv - avg

        # --------------------------------------------------------------
        # 2) Gradient of the L2 loss  L = Σ (conv - avg)^2
        #    ∇_a L = 4 * Σ_j a_j * (conv_{i+j} - avg)
        #    which can be obtained by a single convolution.
        conv_err = np.convolve(err, a[::-1])        # length 3n-2
        grad_l2 = 4.0 * conv_err[n - 1 : n - 1 + n]   # size n

        # --------------------------------------------------------------
        # 3) Sub‑gradient of the max‑entry (peak) of the convolution
        grad_max = np.zeros_like(a)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        # For entries participating in the maximal sum, the derivative of
        # conv[k_max] w.r.t. a_i is a_{k_max-i}.
        grad_max[mask] = a[partner[mask]]

        # --------------------------------------------------------------
        # 4) Combine the two directions
        grad = alpha * grad_max + (1.0 - alpha) * grad_l2

        # --------------------------------------------------------------
        # 5) Mirror‑descent (multiplicative) update and re‑normalisation
        #    Clip the gradient to avoid overflow in the exponent.
        clip_val = 20.0
        grad_clip = np.clip(grad, -clip_val, clip_val)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_clip)

        # Normalise back to the simplex (sum = 1, all entries >=0)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            # Degenerate case – re‑initialise
            a = np.random.rand(n)
        a = a / a.sum()

        # --------------------------------------------------------------
        # 6) Occasional random perturbation (helps escaping plateaus)
        if it % perturb_interval == 0 and it > 0:
            noise = np.random.randn(n) * 0.01
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()

        # --------------------------------------------------------------
        # 7) Evaluate the current candidate
        try:
            cur_score = evaluate_sequence(a.tolist())
        except Exception as exc:
            print(f"[error] evaluation failed at iter {it}: {exc}")
            cur_score = float("inf")

        # --------------------------------------------------------------
        # 8) Keep the best solution observed so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve_counter = 0
            print(f"[iter {it}] new best score = {best_score:.6f}")
        else:
            no_improve_counter += 1

        # --------------------------------------------------------------
        # 9) Simple termination condition (target already reached)
        if best_score < 1.503:
            print(f"[done] target reached at iteration {it} (score={best_score:.6f})")
            break

        # --------------------------------------------------------------
        # 10) Restart if we've made no progress for a while
        if it % restart_interval == 0 and it > 0:
            if no_improve_counter > max_no_improve:
                print(f"[restart] no improvement for {no_improve_counter} iters; re‑initialising.")
                a = np.random.rand(n)
                a = a / a.sum()
                no_improve_counter = 0

        it += 1

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_seq.tolist()
