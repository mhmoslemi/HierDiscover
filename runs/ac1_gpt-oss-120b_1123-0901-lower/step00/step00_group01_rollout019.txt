"""
Search for a step‑function (non‑negative sequence) that minimizes the
evaluation metric

    2 * n * max(conv(f,f)) / (sum(f)^2)

The algorithm consists of three stages:

1. **Parametric warm‑start** – generate candidate sequences of the form
   a_i = C / (i + offset)^p (with p≈0.5), normalise their L1‑norm to
   sqrt(2n) and keep the best one.  This family is known to give a
   nearly constant self‑convolution (the continuous analogue of
   1/√x * 1/√x ≡ constant), yielding a value around 1.57, already
   improving the naïve uniform sequence (value 2.0).

2. **LP‑guided refinement** – repeatedly solve a linear programme that
   finds a direction g with a larger L1‑norm while keeping the
   convolution bound unchanged.  Mixing the current sequence f with g
   (f←(1−t)f + t·g) strictly reduces the objective.  The LP is solved
   with SciPy’s HiGHS solver; a moderate step size t=0.02 is used.

3. **Stochastic hill‑climbing** – occasional random pairwise perturbations
   (adding δ to one entry and subtracting δ from another) are performed
   to escape possible local minima.  The better of the perturbed and the
   LP‑updated sequence is kept.

Fast evaluation of the objective uses FFT‑based convolution
(O(n log n)) instead of the O(n²) NumPy `convolve`, allowing thousands of
iterations within the 1000 s budget.

The function `propose_candidate` returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Fast evaluation utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_max_convolution(seq: np.ndarray) -> float:
    """Return max value of the full convolution seq * seq using FFT."""
    n = seq.shape[0]
    # length for linear convolution = 2*n-1, round up to next power of two
    m = 1 << ((2 * n - 1).bit_length())
    # real FFT is enough because the sequence is real
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    # only first 2*n-1 entries belong to the linear convolution
    return float(np.max(conv[: 2 * n - 1]))


def fast_score(seq: np.ndarray) -> float:
    """Fast surrogate of evaluate_sequence (identical formula)."""
    s = float(np.sum(seq))
    if s < 1e-8:
        return float("inf")
    n = seq.shape[0]
    maxb = _fft_max_convolution(seq)
    return 2.0 * n * maxb / (s * s)


# ----------------------------------------------------------------------
# Parametric warm‑start (power‑law family)
# ----------------------------------------------------------------------
def make_powerlaw_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    Create a sequence a_i = C / (i + 1 + offset)^p, scaled so that
    sum(a) = sqrt(2*n).  All entries are non‑negative.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    raw = idx ** (-p)
    total_raw = raw.sum()
    scale = math.sqrt(2.0 * n) / total_raw
    return raw * scale


# ----------------------------------------------------------------------
# Linear‑programming direction finder (same as the reference code)
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve:
        maximise   sum(b)
        subject to  conv(f_seq, b) <= rhs   (for every convolution index)
                    b >= 0
    Returns the optimal b as a numpy array, or None if LP fails.
    """
    n = f_seq.shape[0]
    # objective = -1  (maximise sum => minimise -sum)
    c = -np.ones(n)

    # build constraints for each possible sum index k = 0..2*n-2
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # for each i where j = k-i lies in [0,n)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # fill the row: coefficient of b_j is f_i where i+j=k
        # equivalently, row[j] = f_i
        row[i_low : i_high + 1] = f_seq[k - np.arange(i_low, i_high + 1)]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints b >= 0  ->  -b <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # use HiGHS solver; give a tiny time limit per call (0.5 s)
    opts = {"time_limit": 0.5, "disp": False}
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method="highs", options=opts)
    if res.success:
        return np.maximum(res.x, 0.0)   # safety clamp
    return None


def get_good_direction_to_move_into(seq: list[float], step: float = 0.02) -> list[float] | None:
    """
    LP based improvement direction.  The input sequence is first normalised
    to have L1‑norm sqrt(2*n).  A feasible g with larger L1‑norm under the
    same convolution bound is obtained, normalised again and mixed with the
    original sequence using the given step size.
    """
    n = len(seq)
    seq_arr = np.asarray(seq, dtype=np.float64)
    sum_seq = np.sum(seq_arr)
    if sum_seq < 1e-8:
        return None
    # normalise to the canonical L1 norm
    norm_seq = seq_arr * (math.sqrt(2.0 * n) / sum_seq)

    rhs = np.max(np.convolve(norm_seq, norm_seq))
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None

    # re‑normalise g to the same L1 norm
    g = g * (math.sqrt(2.0 * n) / np.sum(g))

    # convex combination
    new_seq = (1.0 - step) * seq_arr + step * g
    # enforce the problem’s safety limits (non‑negative, ≤1000)
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq.tolist()


# ----------------------------------------------------------------------
# Stochastic perturbation (keeps total sum unchanged)
# ----------------------------------------------------------------------
def random_perturb(seq: list[float], magnitude: float = 0.02) -> list[float]:
    """
    Pick two distinct indices i, j and transfer a small amount delta from
    j to i (or vice‑versa).  The total L1‑norm stays unchanged.
    """
    n = len(seq)
    a = np.array(seq, dtype=np.float64)
    i, j = np.random.choice(n, size=2, replace=False)
    # typical scale = magnitude * mean value
    scale = magnitude * np.mean(a)
    delta = (np.random.rand() - 0.5) * 2.0 * scale
    # ensure we stay non‑negative
    if delta > a[j]:
        delta = a[j]
    if -delta > a[i]:
        delta = -a[i]
    a[i] += delta
    a[j] -= delta
    return a.tolist()

    # ------------------------------------------------------------------
    # Determine length.  Use the length of the provided global reference
    # sequence if it exists, otherwise fall back to 5000.
    # ------------------------------------------------------------------
    global height_sequence_1
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 5000

    # ------------------------------------------------------------------
    # 1) Parametric warm‑start: explore exponents around 0.5 and a few
    #    offsets.  Keep the best raw candidate (according to the fast
    #    surrogate objective).
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    candidates_tested = 0

    exponents = [0.45, 0.48, 0.49, 0.50, 0.51, 0.52, 0.55]
    offsets = [0.0, 0.5, 1.0, 2.0, 4.0]

    for p in exponents:
        for off in offsets:
            seq_arr = make_powerlaw_sequence(n, p=p, offset=off)
            val = fast_score(seq_arr)
            candidates_tested += 1
            if val < best_val:
                best_val = val
                best_seq = seq_arr.tolist()
            # early break if we are already close to target
            if best_val < 1.52:
                break
        if best_val < 1.52:
            break

    # If for some reason no candidate was produced, fall back to uniform.
    if best_seq is None:
        uniform = [math.sqrt(2.0 * n) / n] * n
        best_seq = uniform
        best_val = fast_score(np.asarray(best_seq))

    print(f"[INIT] Tested {candidates_tested} parametric candidates → best score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑guided iterative refinement
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    current_val = best_val
    it = 0
    while time.time() < deadline:
        it += 1
        # try LP direction
        new_seq = get_good_direction_to_move_into(current_seq, step=0.02)
        if new_seq is None:
            # fall back to a random perturbation
            new_seq = random_perturb(current_seq, magnitude=0.02)

        # compute the true (library) score for safety (avoid drift)
        try:
            new_val = evaluate_sequence(new_seq)
        except Exception:
            # any unexpected failure – skip this candidate
            continue

        # accept if better (strict improvement)
        if new_val < current_val:
            current_seq = new_seq
            current_val = new_val

        # keep the best ever seen
        if new_val < best_val:
            best_seq = new_seq
            best_val = new_val
            print(f"[{it:06d}] New best score: {best_val:.6f}")

        # every 200 iterations inject a small random shake to escape plateaus
        if it % 200 == 0:
            shaken = random_perturb(current_seq, magnitude=0.01)
            try:
                shake_val = evaluate_sequence(shaken)
            except Exception:
                shake_val = float("inf")
            if shake_val < current_val:
                current_seq = shaken
                current_val = shake_val
                if shake_val < best_val:
                    best_seq = shaken
                    best_val = shake_val
                    print(f"[{it:06d}] Best after shake: {best_val:.6f}")

        # safety check for time
        if time.time() >= deadline:
            break

    # final verification using the provided evaluate_sequence (just in case)
    final_score = evaluate_sequence(best_seq)
    print(f"[DONE] Best final score: {final_score:.6f} (iterations: {it})")
    return best_seq
