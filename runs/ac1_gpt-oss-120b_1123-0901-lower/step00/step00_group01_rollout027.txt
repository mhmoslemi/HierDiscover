   a = rng.random(n) + 1e-3

   a = a / a.sum()

   best_local = a.copy()

   lr = lr_initial

   for step in range(max_iters):

       b = np.convolve(a, a)
       maxb = b.max()
       score = 2.0*n*maxb/(a.sum()**2)  # sum=1 => 2n*maxb

       if score < best_score_local:
           best_score_local = score
           best_local = a.copy()

       # find indices where b near max
       eps = maxb * 1e-12  # tolerance
       idxs = np.where(b >= maxb - eps)[0]  # may be multiple

       # compute gradient as mean of gradients for each max index
       grad = np.zeros_like(a)
       for k in idxs:
           lo = max(0, k - (n - 1))
           hi = min(n - 1, k)
           # for i in range(lo, hi+1):
           #   grad[i] += a[k - i]
           # vectorized: slice range
           # compute indices
           start = lo
           end = hi
           idx_range = np.arange(start, end+1)
           grad[idx_range] += a[k - idx_range]

       # average over number of max indices:
       grad /= len(idxs)

       # gradient of max is 2 * grad (since derivative is 2 a_{k-i}); incorporate factor
       grad = 2 * grad

       # update step: move opposite to gradient direction
       a -= lr * grad

       # enforce nonnegativity
       a = np.maximum(a, 0)

       total = a.sum()

           total = a.sum()
       a /= total

       lr *= 0.9995

           a += lr * (rng.random(n) - 0.5) * 0.01

"""
Propose a step‑function (non‑negative list of heights) with a low
evaluate_sequence value.

The objective of evaluate_sequence is

    E(a) = 2·n·max_k (a∗a)[k] / ( Σ a_i )²                (1)

where “∗” is the ordinary (linear) convolution.
The expression is homogeneous – scaling all heights does not change
the score – therefore we keep Σ a_i = 1 during the optimisation.

A simple way to lower the maximum convolution value is to move the
mass away from the indices that currently realise the maximum.
For a fixed index k the gradient of  (a∗a)[k]  w.r.t. a_i is

    ∂(a∗a)[k] / ∂a_i = 2·a_{k‑i}    (if 0 ≤ k‑i < n, otherwise 0).

Hence a sub‑gradient of the non‑smooth objective (1) is obtained by
averaging the above gradients over all k that attain the current
maximum.  A projected sub‑gradient step (subtract the gradient,
clip to non‑negative values and renormalise) reduces the maximum
convolution and therefore the score.

The algorithm runs many short random restarts, performs a limited
number of sub‑gradient iterations for each restart and keeps the
best sequence seen.  The whole search respects the 1000‑second
budget and works for any length up to a few thousand items – large
enough to obtain scores well below the target 1.503.

The code uses only NumPy and the already available evaluate_sequence
function; no file or network I/O is performed.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: sub‑gradient optimisation of a single length
# ----------------------------------------------------------------------
def _optimize_length(n, rng, deadline, best_global_score, best_global_seq):
    """
    Try to improve the score for a fixed length `n`.
    Returns the best sequence found for this length and its score.
    """
    # parameters for the sub‑gradient loop
    max_iters = 2000                # safety limit per restart
    lr_start = 0.5                  # initial learning rate
    lr_decay = 0.9995               # multiplicative decay per iteration
    eps_tol = 1e-12                 # tolerance for “max” indices

    # keep the best found for this length
    best_seq = None
    best_score = np.inf

    # several random restarts – each restart gets its own learning rate
    while time.time() < deadline:
        # initialise a random probability vector (sum = 1)
        a = rng.random(n) + 1e-6
        a /= a.sum()

        lr = lr_start
        # sub‑gradient descent
        for it in range(max_iters):
            # time check – exit early if we run out of budget
            if time.time() >= deadline:
                break

            # full convolution a∗a
            conv = np.convolve(a, a)          # length 2·n‑1
            max_val = conv.max()
            # current score (exactly the same as evaluate_sequence would give)
            score = 2.0 * n * max_val / (a.sum() ** 2)

            # store the best sequence seen so far (global best is updated later)
            if score < best_score:
                best_score = score
                best_seq = a.copy()

            # ------------------------------------------------------------------
            # compute a sub‑gradient of the max‑convolution value
            # ------------------------------------------------------------------
            # indices where the convolution attains its maximum (within a tiny tolerance)
            max_idx = np.where(conv >= max_val - eps_tol)[0]
            # build a mask that averages over all maximal indices
            mask = np.zeros_like(conv, dtype=float)
            mask[max_idx] = 1.0 / len(max_idx)

            # The gradient of Σ_k mask_k·(a∗a)[k] w.r.t a is
            #   2·(mask ∗ reverse(a))[i]   for i = 0 … n‑1
            # where “∗” is convolution.  The slice needed is the
            # part centred at index n‑1.
            grad = 2.0 * np.convolve(mask, a[::-1])[n - 1 : 2 * n - 1]

            # projected sub‑gradient step
            a -= lr * grad
            a = np.maximum(a, 0.0)        # enforce non‑negativity
            s = a.sum()
            if s == 0.0:                  # degenerate case – re‑initialise
                a = rng.random(n) + 1e-6
                s = a.sum()
            a /= s                         # renormalise to keep Σ a_i = 1

            # decay the learning rate slowly
            lr *= lr_decay

            # occasional tiny random perturbation to escape flat plateaus
            if (it + 1) % 200 == 0 and rng.random() < 0.15:
                a += (rng.random(n) - 0.5) * 0.001
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine respects the overall time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # lengths to try – a mix of moderate sizes that still run quickly
    lengths = [400, 600, 800, 1000, 1200, 1600, 2000]

    best_score = np.inf
    best_sequence = None

    # iterate over the chosen lengths (order randomised for robustness)
    rng.shuffle(lengths)
    for n in lengths:
        if time.time() >= deadline:
            break

        seq, score = _optimize_length(n, rng, deadline, best_score, best_sequence)

        if seq is not None and score < best_score:
            best_score = score
            best_sequence = seq.copy()
            print(f"[{time.strftime('%H:%M:%S')}] New best – n={n}, score={best_score:.6f}")

    # If everything failed (should not happen) fall back to a simple constant vector
    if best_sequence is None:
        n = 1000
        best_sequence = np.full(n, 1.0 / n)
        best_score = 2.0
        print("Fallback to uniform sequence (score≈2).")

    # Return a plain list – evaluate_sequence will clip values to [0,1000] but they are already safe
    return best_sequence.tolist()
