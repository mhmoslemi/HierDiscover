    np.random.seed(seed)
    random.seed(seed)
    end_time = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')

        # pick new length
        n = np.random.randint(200, 1500)   # limit size for speed
        # init seq: random Dirichlet
        seq = np.random.rand(n)
        seq = seq / np.sum(seq) * np.sqrt(2*n)  # optional scaling
        # maybe random heavypoints
        # Simulated annealing loop for this seq for limited time
        start = time.time()
        # allocate time: e.g., 5 seconds per restart
        while time.time() - start < 5 and time.time() < end_time:
            # compute conv and max index
            conv = np.convolve(seq, seq)
            max_idx = int(np.argmax(conv))
            max_val = conv[max_idx]
            # find contributors
            # indices i where j = max_idx - i in [0,n)
            # compute contributions
            i_vals = []
            contributions = []
            for i in range(n):
                j = max_idx - i
                if 0 <= j < n:
                    contributions.append(seq[i] * seq[j])
                    i_vals.append(i)
            # If empty: fallback random
            if not contributions:
                # random move
                i, j = np.random.choice(n, 2, replace=False)
                #...
                continue
            # choose i with probability proportional to contributions
            probs = np.array(contributions) / np.sum(contributions)
            i = np.random.choice(i_vals, p=probs)
            j = max_idx - i
            if i == j:
                # choose another pair
                continue
            # step fraction
            step = 0.02
            d = step * min(seq[i], seq[j])
            if d <= 0:
                continue
            # move mass to random other index
            l = np.random.choice(n)
            # apply temporary changes
            old_i, old_j, old_l = seq[i], seq[j], seq[l]
            seq[i] = max(0.0, seq[i] - d)
            seq[j] = max(0.0, seq[j] - d)
            seq[l] = seq[l] + 2*d
            # evaluate
            val = evaluate_sequence(seq.tolist())
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"New best {best_val:.6f} len {n}")
            if val <= evaluate_sequence([seq[i], seq[j], seq[l]]):
                # accept
                pass
            else:
                # revert
                seq[i] = old_i
                seq[j] = old_j
                seq[l] = old_l
        # optionally allow random perturbations
    # final return best_seq as list
    return best_seq.tolist()

def propose_candidate(seed=0, budget_s=1000):
    """Return a sequence of non‑negative heights that yields a low value of
    `evaluate_sequence`.  The routine uses a heuristic based on the
    continuous extremal function a_i ≈ 1/√(i+1) and a simple simulated
    annealing/greedy refinement to lower the maximal self‑convolution
    while keeping the ℓ₁‑norm unchanged.
    """
    import numpy as np, time, random, math
    # set seeds
    np.random.seed(seed)
    random.seed(seed)
    time_limit = time.time() + budget_s - 0.5
    # fallback constant sequence
    best_seq = [1.0] * 10
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_score = float('inf')
    print(f"Initial best score {best_score:.6f}")
    # helper to evaluate a numpy array (list conversion)
    def eval_seq(arr):
        return evaluate_sequence(arr.tolist())
    # generate an initial shape using power law exponent near 0.5
    def generate_initial(n, exponent):
        idx = np.arange(n, dtype=np.float64) + 1.0
        seq = (idx) ** (-exponent)   # power law
        # add small jitter
        seq += 0.01 * np.random.rand(n)
        return seq
    # main loop: many restarts
    while time.time() < time_limit:
        # choose length
        n = np.random.randint(300, 1200)   # moderate size
        # exponent around 0.5 with slight random perturbation
        exponent = 0.5 + (np.random.rand() - 0.5) * 0.1   # between 0.45 and 0.55
        seq = generate_initial(n, exponent)
        # normalise to keep values small (optional)
        #clip to [0,1000] automatically handled
        # compute current score
        cur_score = eval_seq(seq)
        # keep track of best within this restart
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"New global best (init) {best_score:.6f} len {n}")
        # annealing parameters
        start = time.time()
        # allocate about 2 seconds for this restart (or remaining time)
        max_iter = 5000
        temperature = cur_score * 0.2 + 1e-6
        for it in range(max_iter):
            if time.time() >= time_limit:
                break
            # recompute convolution and max index
            conv = np.convolve(seq, seq)
            max_idx = int(np.argmax(conv))
            # list of contributing indices
            i_vals = np.arange(n)
            j_vals = max_idx - i_vals
            mask = (j_vals >= 0) & (j_vals < n)
            i_candidates = i_vals[mask]
            j_candidates = j_vals[mask]

                    if new_score < best_score:

                        print(f"New global best {best_score:.6f} len {n} it {it}")

                continue

def generate_power_law(n: int, exponent: float) -> np.ndarray:
