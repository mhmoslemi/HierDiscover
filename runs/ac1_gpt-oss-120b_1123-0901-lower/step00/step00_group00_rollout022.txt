"""
Simple stochastic optimizer for the step‑function evaluation problem.

The algorithm works in two phases:

1. **Exponential seed search** – we try many geometric sequences
   a_i = r^i (0 < r < 1).  For a given length n the evaluation depends only
   on the shape of the sequence, not on its scale, so r is the only free
   parameter.  By sampling r on a log‑scale we quickly locate a good
   starting point (the best exponential usually yields a score ≈ 1.8,
   already better than the uniform bound 2.0).

2. **Local refinement** – starting from the best exponential we perform
   random “mass‑move” updates.  Each step transfers a fraction of the
   value at one index to another index (or perturbs a single entry).  The
   new candidate is kept if it improves the objective, otherwise it may
   be accepted with a tiny probability that decays with time (simulated
   annealing).  The step size σ gradually shrinks, allowing coarse
   exploration early on and fine tuning later.

The routine respects the 1000‑second time budget, prints occasional
progress messages, and finally returns the best sequence found as a
plain Python list of floats.
"""

import time
import random
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    evaluate_sequence().  The function runs for at most *budget_s* seconds.
    """
    # ------------------------------------------------------------------
    # basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # length of the sequence – use the length of the known construction if
    # it exists, otherwise fall back to a reasonable size.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)          # expected 2304
    else:
        n = 1500

    deadline = time.time() + budget_s - 5   # a few seconds margin for cleanup

    # ------------------------------------------------------------------
    # 1) exponential seed search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # we sample r = exp(-α) with α drawn uniformly from a small interval.
    # The interval (5e‑4…1e‑2) corresponds to r ∈ (≈0.99 … ≈0.9995) – the region
    # where the exponential family gives the best theoretical bound.
    n_r_samples = 300
    for _ in range(n_r_samples):
        log_r = -np.random.uniform(5e-4, 1e-2)        # log(r) is negative
        r = np.exp(log_r)
        seq = np.power(r, np.arange(n, dtype=np.float64))
        # evaluate_sequence expects a plain list, but a NumPy array works as well.
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[init] best exponential score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) local stochastic refinement
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    sigma = 0.12                 # typical relative step size
    it = 0
    # temperature for occasional uphill moves (simulated annealing)
    def temperature():
        # linear cooling to a small floor
        remaining = max(0.0, deadline - time.time())
        return max(1e-3, remaining / budget_s)

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # propose a neighbour
        # --------------------------------------------------------------
        if np.random.rand() < 0.7:                 # move a fraction of mass
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            # transfer a random proportion of the mass at i to j
            amount = cur_seq[i] * np.random.uniform(0.0, sigma)
            new_i = max(0.0, cur_seq[i] - amount)
            new_j = cur_seq[j] + amount
            new_seq = cur_seq.copy()
            new_seq[i] = new_i
            new_seq[j] = new_j
        else:                                       # perturb a single entry
            i = np.random.randint(n)
            delta = cur_seq[i] * np.random.normal(scale=sigma)
            new_val = max(0.0, cur_seq[i] + delta)
            new_seq = cur_seq.copy()
            new_seq[i] = new_val

        # note: values stay far below the 1000‑cap, no explicit clipping needed

        # --------------------------------------------------------------
        # evaluate the proposal
        # --------------------------------------------------------------
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        # --------------------------------------------------------------
        # acceptance logic
        # --------------------------------------------------------------
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            cur_seq = new_seq.copy()
            print(f"[iter {it}] NEW BEST = {best_score:.6f}")
            # gradually shrink the step size to focus the search
            sigma = max(1e-4, sigma * 0.99)
        else:
            # occasional uphill move to escape local minima
            T = temperature()
            if np.random.rand() < np.exp((best_score - new_score) / T):
                cur_seq = new_seq
                # allow a tiny increase of sigma on such moves
                sigma = min(0.5, sigma * 1.01)

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    # cast to a plain Python list of floats as required by the evaluator
    return best_seq.tolist()
