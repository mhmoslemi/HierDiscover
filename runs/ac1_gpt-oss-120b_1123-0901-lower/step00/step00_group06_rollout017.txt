def simplex_projection(v):
    # Returns projection of v onto simplex sum=1, v>=0
    # Implementation from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def max_convolution(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[:2*n - 1]

Now compute gradient:

def compute_gradient(a, k):
    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    # i from lo to hi inclusive:
    i = np.arange(lo, hi+1)
    j = k - i
    g = np.zeros_like(a)
    g[i] = 2 * a[j]
    return g

Now projection:

def project_simplex(v):
    # Duchi et al. (2008)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def improve_sequence(a_init, time_allow):
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    # Ensure a normalized sum=1
    a = a / np.sum(a)
    start = time.time()
    eta = 0.5  # initial step size maybe
    iteration = 0
    while time.time() - start < time_allow:
        # compute conv & max
        conv = max_convolution(a)
        k = int(np.argmax(conv))
        maxb = conv[k]
        # evaluate score
        score = 2 * len(a) * maxb / (np.sum(a) ** 2)  # sum =1 mostly
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # gradient
        g = compute_gradient(a, k)
        # random perturb to avoid zero gradient
        if iteration % 100 == 0:
            # add small random noise
            noise = 0.001 * np.random.randn(len(a))
            g = g + noise
        # update
        a_new = a - eta * g
        # project onto simplex
        a_new = project_simplex(a_new)
        # compute new score for acceptance
        conv_new = max_convolution(a_new)
        k_new = int(np.argmax(conv_new))
        maxb_new = conv_new[k_new]
        new_score = 2 * len(a_new) * maxb_new / (np.sum(a_new) ** 2)
        # accept or not
        if new_score < score:
            a = a_new
            # increase step size
            eta *= 1.02
        else:
            # reject or keep with probability?
            # simple: keep a (old) but reduce step size
            eta *= 0.95
            # optionally still adopt new randomly with small prob
            if np.random.rand() < 0.01:
                a = a_new
        iteration += 1
    return best_a, best_score

"""
gradient_descent_convolution.py

Search for a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the discrete convolution (numpy.convolve).  
The algorithm:

* Several simple initialisations are built:
  – uniform (all entries equal)
  – optimal two‑level block distribution (ratio ≈ 1.6875)
  – linearly decreasing sequence (≈ 1.8)
  – the previous best `height_sequence_1` if it exists.
* Each start point is refined by a projected sub‑gradient descent on the
  current maximal convolution entry.
  The sub‑gradient of `max(conv(a,a))` at the active shift `k` is
  `g_i = 2 * a_{k-i}` for the overlapping indices.
* After taking a step we project the vector onto the probability simplex
  (non‑negative, sum = 1) – this guarantees scale invariance and respects
  the non‑negativity required by `evaluate_sequence`.
* An adaptive step‑size is used; occasional tiny random perturbations
  break symmetry and help escape plateaus.
* The best sequence found within the allotted wall‑clock time is returned.

The implementation relies only on NumPy, `time` and the already imported
`evaluate_sequence`.  No file‑IO or external solvers are used.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> np.ndarray:
    """
    Fast convolution a * a using a real FFT.  Returns the full
    convolution of length 2*n-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


def _gradient_of_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry at shift k.
    For indices i with 0 ≤ k‑i < n the gradient is 2*a_{k-i},
    otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.  Implements the algorithm of
    Duchi et al. (2008).  Returns a new array (the input is not modified).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """
    Compute the evaluation metric 2*n*max(conv(a,a)) / (sum(a)**2)
    without the safety‑checks of evaluate_sequence (the vector is already
    non‑negative and reasonably small, so the result coincides).
    """
    s = a.sum()
    if s == 0.0:
        return np.inf
    conv = _max_convolution(a)
    maxb = conv.max()
    return 2.0 * a.shape[0] * maxb / (s * s)


# ----------------------------------------------------------------------
# Core optimiser
# ----------------------------------------------------------------------


def _improve_sequence(a0: np.ndarray, time_budget: float) -> (np.ndarray, float):
    """
    Refine a starting sequence `a0` using projected sub‑gradient descent.
    The function runs for at most `time_budget` seconds and returns the
    best sequence observed together with its score.
    """
    a = a0.copy()
    a = _project_simplex(a)               # ensure a ∈ simplex
    best_a = a.copy()
    best_score = _score(a)

    start = time.time()
    eta = 0.5                               # initial step size
    it = 0
    n = a.shape[0]

    while time.time() - start < time_budget:
        # current convolution and its maximal shift
        conv = _max_convolution(a)
        k = int(conv.argmax())
        cur_score = 2.0 * n * conv[k] / (a.sum() ** 2)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # sub‑gradient at the active shift
        g = _gradient_of_max(a, k)

        # occasional tiny random jitter to break symmetry
        if it % 100 == 0:
            g += 1e-4 * np.random.randn(n) * a.mean()

        # gradient step and projection onto simplex
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        # evaluate the candidate
        conv_new = _max_convolution(a_new)
        new_score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        # accept if we improved; otherwise keep the old point
        if new_score < cur_score:
            a = a_new
            eta *= 1.02                     # slightly enlarge step on success
        else:
            eta *= 0.95                     # shrink step on failure
            # with a very small probability we still move to the new point
            if np.random.rand() < 0.01:
                a = a_new
        it += 1

    return best_a, best_score
