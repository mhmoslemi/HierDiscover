import numpy as np
import time

def compute_score(seq):
    s = np.asarray(seq, dtype=np.float64)
    n = s.size
    if n == 0:
        return np.inf
    ss = s.sum()
    if ss == 0:
        return np.inf
    b = np.convolve(s, s)
    max_b = np.max(b)
    return float(2.0 * n * max_b / (ss*ss))

def shape_power(n, a):
    idx = np.arange(n) + 0.5
    # product of distances to ends:
    prod = idx * (n + 0.5 - idx)
    # Avoid zero for a negative exponent
    w = prod ** a
    # If a is negative, large values at edges
    # Normalize to sum 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0 / n)
    return (w / w_sum).tolist()

"""
Propose a candidate sequence of non‑negative heights that (hopefully) beats the
baseline value 2.0.

The algorithm works in three stages:

1. **Parametric initialisation** – generate several families of deterministic
   sequences whose shape is controlled by a small number of parameters
   (power‑law and exponential “edge‑heavy” profiles).  Each candidate is
   evaluated with a fast local copy of ``evaluate_sequence`` and the best
   one is kept.

2. **Optional use of the supplied construction** – if the global variable
   ``height_sequence_1`` is present it is examined as a possible starting
   point (the sequence is normalised to sum‑1 before scoring).

3. **Simple simulated‑annealing refinement** – starting from the best
   deterministic shape we perform many tiny mass‑transfer moves
   (pick two indices i≠j and move a random fraction of the mass from i to j).
   Moves are accepted if they improve the score or with a probability that
   depends on a slowly decreasing temperature.  The best sequence ever seen
   is returned.

The evaluation function is scale‑invariant, so the sequences are normalised
to sum‑1 for convenience.  All numbers stay non‑negative; the routine finishes
well before the 1000 s budget.
"""

import time
import numpy as np

def _score(seq):
    """Fast version of evaluate_sequence – assumes a list/array of non‑negative numbers."""
    s = np.asarray(seq, dtype=np.float64)
    n = s.size
    if n == 0:
        return np.inf
    total = s.sum()
    if total == 0.0:
        return np.inf
    conv = np.convolve(s, s)
    return float(2.0 * n * conv.max() / (total * total))


def _shape_power(n, a):
    """
    Power‑law shape: w_i ∝ (i+½)*(n+½‑i)  raised to exponent *a*.
    For a < 0 the distribution concentrates mass near the ends,
    for a > 0 it concentrates near the centre.
    """
    idx = np.arange(n, dtype=np.float64) + 0.5          # distance from the left edge
    prod = idx * (n + 0.5 - idx)                       # product of distances to both edges
    # prod is always > 0, so raising to a negative power is safe
    w = prod ** a
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w_sum
    return w


def _shape_exponential(n, a):
    """
    Exponential edge‑heavy shape: w_i ∝ exp(-a*i) + exp(-a*(n‑1‑i)).
    Larger *a* makes the ends much heavier than the centre.
    """
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-a * idx) + np.exp(-a * (n - 1 - idx))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w_sum
    return w

                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    print(f"[perturb {iteration}] score {best_score:.6f}")

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Convert to plain Python list as required by the harness
    return best_seq.tolist()
