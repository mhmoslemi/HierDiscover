def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b)
    # constraints: for each k 0..2n-2: sum_i f_i * b_{k-i} <= rhs
    a_ub = []
    b_ub = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    a_ub = np.array(a_ub)
    b_ub = np.array(b_ub)
    # nonnegativity via bounds
    bounds = [(0, None) for _ in range(n)]
    # Solve using high-speed solver
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

def get_good_direction_to_move_into(seq):
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return None
    # normalize to sum sqrt(2n)
    normalized = seq * np.sqrt(2 * n) / sum_seq
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    normalized_g = g * np.sqrt(2 * n) / sum_g
    t = 0.01
    new_seq = (1 - t) * seq + t * normalized_g
    return new_seq

Proposes a candidate non‑negative step‑function sequence that minimizes the
evaluation function.  The algorithm starts from the known sequence (or a random
initialisation), repeatedly performs a greedy mass‑transfer step that reduces the
contribution of the indices that dominate the maximal convolution value,
occasionally refines the sequence with a linear‑programming direction (the
previous SOTA move) and applies small random perturbations.  All steps keep the
sum of the coefficients normalised to sqrt(2 · n) so that the objective equals
the maximal self‑convolution value.  The best sequence found before the time
budget expires is returned.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog

# Helper functions
# -------------------------------------------------

def normalize_sequence(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative sequence so that its L1‑norm equals sqrt(2 · len)."""
    n = len(seq)
    target = np.sqrt(2.0 * n)
    total = seq.sum()
    if total <= 0:
        seq = np.full(n, target / n)
    else:
        seq = seq * (target / total)
    # clip to the limits enforced in evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def evaluate(seq: np.ndarray) -> float:
    """Wrapper around the supplied evaluate_sequence (expects a list)."""
    return float(evaluate_sequence(seq.tolist()))


def convolution(seq: np.ndarray) -> np.ndarray:
    """Return the linear convolution seq * seq using FFT (real valued)."""
    n = len(seq)
    m = 2 * n - 1
    # real FFT, be careful with length
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)
    return conv


def max_convolution_info(seq: np.ndarray):
    """Return (max_value, index_of_max) of the self‑convolution."""
    conv = convolution(seq)
    max_val = conv.max()
    max_idx = int(conv.argmax())
    return max_val, max_idx, conv


def contribution_vector(seq: np.ndarray, k: int) -> np.ndarray:
    """
    For a given index k (where the convolution attains its maximum) compute
    the vector c where c[i] = sum_{j: i+j=k} seq[j]  (i.e. the contribution of
    entry i to the value at k).  The product seq[i] * c[i] is the total amount
    that entry i adds to the maximal convolution value.
    """
    n = len(seq)
    contrib = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_range = np.arange(i_start, i_end + 1)
    j_range = k - i_range
    contrib[i_range] = seq[j_range]
    return contrib


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """Linear program:
       maximise sum b_j subject to (f * b)[k] ≤ rhs  for all k,
       b_j ≥ 0.
       Returns the optimal b as a numpy array or None if LP fails.
    """
    n = len(f_seq)
    # objective: maximise sum b --> minimise -sum b
    c = -np.ones(n)

    # Build A_ub, b_ub for the convolution constraints.
    # For each k we have sum_i f_i * b_{k-i} ≤ rhs.
    a_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_rows.append(row)
        b_rows.append(rhs)
    A_ub = np.vstack(a_rows)
    b_ub = np.array(b_rows)

    # Bounds enforce b_j ≥ 0.
    bounds = [(0.0, None) for _ in range(n)]

    # Use the high‑performance HiGHS solver (available in recent SciPy).
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None

    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq: np.ndarray):
    """Return a new sequence obtained by moving a small step towards the LP
    direction described in the reference solution.  If the LP fails, return
    None.
    """
    n = len(seq)
    sum_seq = seq.sum()
    if sum_seq == 0:
        return None
    # Normalise the current sequence to the standard L1‑norm.
    norm_seq = seq * np.sqrt(2.0 * n) / sum_seq
    # Upper bound for the convolution (the current maximum).
    rhs = np.max(np.convolve(norm_seq, norm_seq))
    # Solve the LP.
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    sum_g = g.sum()
    if sum_g == 0:
        return None
    # Normalise the LP solution to the same L1‑norm.
    norm_g = g * np.sqrt(2.0 * n) / sum_g
    # Take a small convex combination.
    t = 0.01
    new_seq = (1.0 - t) * seq + t * norm_g
    return new_seq


def smooth_sequence(seq: np.ndarray, radius: int = 2) -> np.ndarray:
    """Simple moving‑average smoothing (radius cells on each side)."""
    kernel = np.ones(2 * radius + 1)
    # Pad with zeros to keep length unchanged.
    padded = np.pad(seq, (radius, radius), mode='constant')
    smoothed = np.convolve(padded, kernel, mode='valid') / kernel.sum()
    return smoothed
