def project_simplex(v):
    # Return projection onto simplex sum=1, v_i >= 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent_step(a, step):
    # compute convolution
    c = np.convolve(a, a)
    max_idx = np.argmax(c)
    # compute gradient
    n = len(a)
    g = np.zeros(n)
    # contributions where i in [0,n-1] and j = max_idx - i in [0,n-1]

```python
import numpy as np
import time

def project_onto_simplex(v):
    # algorithm from https://gist.github.com/...
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_score(a):
    # assume sum(a)=1, but compute general version for safety
    a = np.array(a, dtype=float)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    # scale to sum 1
    a = a / sum_a
    c = np.convolve(a, a)
    max_c = c.max()
    n = len(a)
    return float(2 * n * max_c)
```

a(x) ∝ 1/√x on a finite interval.  In the discrete case we use

    a_i ∝ 1/√(i+1) ,   i = 0 … n‑1

and normalise the vector to a probability distribution.  For this shape
the value of `evaluate_sequence` is close to the theoretical lower bound
π/2 ≈ 1.57, which is a large improvement over the uniform sequence (value 2.0).

To possibly push the bound a little further a fast simulated‑annealing
refinement is applied: two random positions exchange a small amount of mass
while keeping the total sum unchanged and never producing negative entries.
The move is accepted according to a decreasing temperature schedule,
and the best sequence seen so far is returned.

The routine works under the time limit given by `budget_s`.  No external
files or network access are needed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex.
    Result satisfies: w_i >= 0 and sum w_i = 1.
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho where the condition holds
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Compute the value that `evaluate_sequence` would return, assuming
    the sum of `a` is not zero.  The function normalises to sum 1,
    because the evaluation is homogeneous.
    """
    total = a.sum()
    if total < 1e-12:
        return np.inf
    a = a / total
    c = np.convolve(a, a)
    return float(2.0 * a.shape[0] * c.max())


def _initial_sqrt_shape(n: int) -> np.ndarray:
    """
    Initialise a vector of length n with entries proportional to 1/sqrt(i+1)
    and normalised to sum to one.
    """
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a


def _random_move(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Perform a small mass‑exchange move that preserves the total sum.
    Two distinct indices i, j are selected; an amount `delta` is transferred
    from one index to the other (while keeping all entries non‑negative).
    """
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # magnitude of the transfer (both signs possible)
    delta = (rng.random() - 0.5) * sigma
    if delta > 0:
        # move mass from i to j
        delta = min(delta, a[i])           # cannot take more than a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
    else:
        # move mass from j to i
        delta = min(-delta, a[j])
        a_new = a.copy()
        a_new[i] += delta
        a_new[j] -= delta
    # small numerical cleaning
    a_new = np.maximum(a_new, 0.0)
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The algorithm:

    1. Build a deterministic 1/√(i+1) shape.
    2. Optionally improve it by a fast simulated‑annealing loop that
       swaps tiny amounts of weight between random positions.
    3. Keep the best sequence encountered within the time budget and
       return it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Choose a base length.  If a previous construction is available we
    # reuse its length (the variable may be defined in the global scope);
    # otherwise we fall back to a modest size that still gives a good bound.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1108                    # default length used by the benchmark

    # ------------------------------------------------------------------
    # 1. Deterministic starting point: the 1/√ shape.
    # ------------------------------------------------------------------
    current = _initial_sqrt_shape(n)
    best = current.copy()
    best_score = _objective(best)

    print(f"Initial 1/√ shape – length {n}, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing refinement.  The loop runs until the time
    #    budget expires.  At each iteration a tiny mass exchange is tried;
    #    the move is accepted if it improves the objective or with a
    #    probability that decreases with temperature.
    # ------------------------------------------------------------------
    # annealing schedule
    max_iter = 200000       # large upper bound – the loop will stop earlier
    start_temp = 0.05
    end_temp = 1e-6

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        # linear cooling
        t = start_temp + (end_temp - start_temp) * (iteration / max_iter)
        sigma = 0.02 * t + 1e-4   # step magnitude shrinks with temperature

        candidate = _random_move(current, sigma, rng)
        cand_score = _objective(candidate)

        # acceptance rule (Metropolis)
        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            print(f"[{iteration:6d}] New best score: {best_score:.6f}")

        # decide whether to move the current state
        if cand_score < _objective(current):
            accept = True
        else:
            # probability = exp( -(new - old) / temperature )
            accept = rng.random() < np.exp(-(cand_score - _objective(current)) / max(t, 1e-12))

        if accept:
            current = candidate

        iteration += 1

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # ensure the returned sequence satisfies the requirements (non‑negative)
    best = np.maximum(best, 0.0)
    # final safety: re‑normalise (the objective is scale‑invariant)
    best /= best.sum() if best.sum() > 0 else 1.0
    return best.tolist()
