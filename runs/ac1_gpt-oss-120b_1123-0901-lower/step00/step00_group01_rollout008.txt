```
def simplex_projection(v):
    # v: array
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho = max i such that u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all entries zero
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Propose a step‑function (sequence of non‑negative heights) with a low
evaluation score.

Algorithm
---------
1. Generate a diverse set of seed sequences:
   - Uniform, random, linear ramps, cosine‑hump, sine‑squared,
     and a V‑shaped distribution.
   - The existing `height_sequence_1` (if available) is also tried.
2. Keep the best seed according to the evaluation function.
3. Refine the best seed with a simple projected‑gradient/entropy‑descent
   on the objective  max(conv(a,a))  while keeping a on the probability
   simplex (sum = 1, a_i ≥ 0).  The gradient of the *active* convolution
   index k is  g_i = 2·a_{k-i} (zero outside the overlap).
   The update is multiplicative:
        a ← a * exp(‑η·g)   then   a ← a / sum(a)
   This guarantees non‑negativity and keeps the sum constant.

5. The loop runs until the time budget is exhausted (≈ budget_s seconds)

The convolution is computed with an FFT (O(n log n)) for speed.

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho (last index where the condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear (full) convolution a * a using FFT (real input)."""
    n = a.shape[0]
    # Length needed for a * a (full) = 2*n-1
    conv_len = 2 * n - 1
    # Next power of two for efficient FFT
    size = 1 << ((conv_len - 1).bit_length())
    # Real FFT of a padded to 'size'
    fft_a = np.fft.rfft(a, n=size)
    # Point‑wise product (a * a in frequency domain)
    fft_c = fft_a * fft_a
    # Inverse real FFT
    conv_full = np.fft.irfft(fft_c, n=size)
    return conv_full[:conv_len]


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a * a."""
    # Using the FFT based convolution for speed
    return full_convolution(a).max()

    arr = np.maximum(np.arange(n) + 1, n - np.arange(n))

    arr = 0.5 * (1 - np.cos(np.pi * np.arange(n) / (n - 1)))

    arr = np.arange(1, n + 1, dtype=float)

def gen_ramp_down(n: int) -> np.ndarray:
    """Linear decrease from left to right."""

        gen_ramp_down,
        gen_random,
        gen_existing,
    ]

    # Different lengths give the optimiser more flexibility.
    lengths = [200, 500, 1000, 2000, 4000, 6000]

    best_seq = None
    best_score = np.inf

    for n in lengths:
        for gen in generators:
            a = gen(n).astype(float)
            # evaluate using the provided function (scale‑invariant)
            try:
                score = evaluate_sequence(a.tolist())
            except Exception:
                # if the external routine misbehaves, fall back to our own metric
                score = 2.0 * n * conv_max(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[init] {gen.__name__:14s} n={n:<5d}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine the best seed with a simple entropy‑gradient descent.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    a = a / a.sum()                     # ensure exact normalisation
    it = 0

    # pre‑compute a few constants for step‑size adaptation
    eta0 = 5.0                         # initial learning rate (entropy style)
    while time.time() < deadline:
        it += 1

        # ----- compute current max convolution and its index ----------
        conv = full_convolution(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # ----- gradient of the active convolution entry ---------------
        # g_i = 2 * a_{k-i}  for indices where the term exists
        g = np.zeros_like(a)
        if k < n:
            # overlapping region is a[0..k] with reversed a[k..0]
            g[: k + 1] = 2.0 * a[k :: -1]
        else:
            start = k - (n - 1)
            # overlapping region is a[start..] with reversed tail of a
            g[start:] = 2.0 * a[::-1][: n - start]

        # ----- entropy (multiplicative) update -----------------------
        # learning rate decays with sqrt(iteration) to ensure stability
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        # re‑normalise to the simplex
        a = a / a.sum()

        # ----- occasional random jitter to escape plateaus -----------
        if it % 101 == 0:
            jitter = np.random.randn(n) * 0.001
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = simplex_projection(a)

        # ----- keep the best solution seen so far --------------------
        # Our own cheap metric is accurate because the evaluation is
        # scale‑invariant (sum == 1).
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter] it={it:5d}  max_conv={max_val:.6e}  score={cur_score:.6f}")

        # Limit the number of iterations to a reasonable amount
        if it >= 5000:
            break

    # ------------------------------------------------------------------
    # 3.  Return the best sequence encountered.
    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}  length = {len(best_seq)}")
    # The evaluation function internally normalises and clips, so returning
    # a list that already sums to 1 is safe.
    return best_seq.tolist()
