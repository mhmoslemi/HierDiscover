"""
Propose a step‑function that has a very low evaluation score.

The idea is to use a geometric (exponentially decaying) sequence.
For a sequence of length n and ratio r = 1 - α/n  (α > 0) the
convolution peak is approximately max{ n·r^{n‑1}, 2·r } while the
ℓ₁‑norm is   S ≈ n·(1−e^{‑α})/α.  The evaluation value

    2·n·max_b / S²  ≈  2·α²·e^{‑α} / (1−e^{‑α})²

is independent of n as long as the centre term n·e^{‑α} dominates.
The minimum of this expression for realistic n is attained for
α ≈ ln(n/2).  Choosing α ≈ log(n) gives scores of order 10⁻²,
far below the target 1.503.

The function enumerates a few candidate lengths and α values,
keeps the best sequence, and then performs a short random‑local‑search
to squeeze a little more improvement before the time budget runs out.
All numbers stay non‑negative, and the algorithm never violates the
security checks in ``evaluate_sequence``.
"""

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    that minimizes ``evaluate_sequence``.

    Parameters

        before the limit to be safe.
    kwargs : dict
        Ignored – present only to match the expected signature.
    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1.  Explore geometric sequences of several lengths.
    # ------------------------------------------------------------------

    # exists, otherwise fall back to a few hand‑picked lengths.
    try:

    # ------------------------------------------------------------------
    # small random moves that preserve the total sum (approximately)
    # are tried; they often reduce the maximal convolution a bit.
    if best_seq is None:          # fall back to a trivial uniform sequence
        best_seq = [1.0] * max(2, base_len)

    while time.time() < deadline:
        cur = best_seq.copy()
        n = len(cur)

        # pick two distinct indices and move a tiny amount of mass between them
        i, j = np.random.randint(n, size=2)
        # magnitude of the perturbation (relative to the current average)
        avg = sum(cur) / n
        delta = (np.random.rand() - 0.5) * 0.02 * avg   # about ±1 % of avg
        cur[i] = max(0.0, cur[i] + delta)
        cur[j] = max(0.0, cur[j] - delta)

        # evaluate the perturbed candidate
        cur_score = evaluate_sequence(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur
            print(f"Improved by local move: score={best_score:.6f}")

    # Final report (useful when the script is run interactively)
    print(f"Final best score: {best_score:.6f}  (length={len(best_seq)})")
    return best_seq
