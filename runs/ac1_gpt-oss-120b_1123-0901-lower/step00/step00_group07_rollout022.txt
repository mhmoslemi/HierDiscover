Search for a short non‑negative step sequence that gives a low value of the

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search is based on the following observation.

    a_i = (i + c) ** (-p) ,   i = 0 … n‑1      (p > 0, c ≥ 0)

then, after normalisation, the maximum of the convolution is attained either
at the very first index (a_0²) or near the centre (where many terms overlap).
For a given length `n` the two contributions are

    endpoint   ≈ (1-p)² * n^{2p-1}
    centre     ≈ (1-p)² * B(1-p, 1-p)   (independent of n)

where B is the beta‑function.  For 0.5 < p < 1 the centre term is the main
driver of the score and the score is essentially

    score ≈ 2 * (1-p)² * B(1-p, 1-p)

The optimum occurs for p ≈ 0.7; with a modest length (≈150–200) the endpoint
term is already smaller, giving scores well below the target 1.503.

The function below performs a tiny grid‑search over
* length n  (a few values, all ≤ 300 – cheap to evaluate),
* exponent p (0.55 … 0.75),
* a small offset c (0, 0.5, 1).

It keeps the best sequence found before the time budget expires and returns
it as a plain Python list of floats.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    # ------------------------------------------------------------------
    # configuration
    # ------------------------------------------------------------------
    # lengths we will try – short sequences give much smaller scores
    lengths = [80, 100, 120, 150, 180, 200, 250, 300]
    # power‑law exponents – we know the sweet spot is around 0.6‑0.75
    p_grid = np.arange(0.55, 0.76, 0.01)
    # small offsets to reduce the a_0² contribution
    offsets = [0.0, 0.5, 1.0]

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    best_seq = None
    best_score = float('inf')
    best_descr = ""

    # ------------------------------------------------------------------
    # grid search
    # ------------------------------------------------------------------
    for n in lengths:
        # early exit if we run out of time
        if time.time() > deadline:
            break

        idx = np.arange(n, dtype=np.float64)       # 0 … n‑1

        for c in offsets:
            # (idx + c + 1) ensures the first element is never larger than 1
            base = (idx + c + 1.0)

            for p in p_grid:
                if time.time() > deadline:
                    break

                # power‑law shape, all entries are non‑negative
                seq = base ** (-p)

                # evaluate – the function is scale invariant, therefore we do
                # not normalise the sequence; plain values work fine.
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:                # just in case
                    continue

                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    best_descr = f"n={n}, p={p:.3f}, offset={c}"
                    print(f"*** new best: {best_score:.6f}  ({best_descr})")

        # --------------------------------------------------------------
        # after the coarse scan we try a finer step around the best p for
        # the current length (if we found any candidate for it)
        # --------------------------------------------------------------
        if best_seq is not None and best_descr.startswith(f"n={n}"):
            # extract the p that gave the current best for this n
            p0 = float(best_descr.split(",")[1].split("=")[1])
            fine_grid = np.arange(p0 - 0.009, p0 + 0.010, 0.001)
            for p in fine_grid:
                if p <= 0 or p >= 1:
                    continue
                if time.time() > deadline:
                    break
                seq = (np.arange(n, dtype=np.float64) + best_descr.split(",")[2].split("=")[1].strip() + 1.0) ** (-p)
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    best_descr = f"n={n}, p={p:.4f}, offset={c}"
                    print(f"*** refined best: {best_score:.6f}  ({best_descr})")

    # ------------------------------------------------------------------
    # safety: if the search somehow failed we fall back to a short uniform
    # sequence which is guaranteed to give a score of exactly 2.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 100
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print(f"fallback uniform sequence (score={best_score})")

    print(f"FINAL best score: {best_score:.6f}   ({best_descr})")
    return best_seq
