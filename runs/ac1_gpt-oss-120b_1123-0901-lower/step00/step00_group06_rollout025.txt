def project_simplex(v):
    # v is array
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u - (sv - 1) / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = (sv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

propose_candidate
~~~~~~~~~~~~~~~~~

Generate a step‑function (non‑negative list of heights) whose evaluation
value 2·n·max(conv(a,a))/sum(a)² is as small as possible.

The core idea is to start from the analytical construction

distribution.
All updates keep the vector on the probability simplex (non‑negative,
sum = 1), therefore the evaluation is scale‑invariant and the clipping
performed inside ``evaluate_sequence`` never interferes.

The algorithm runs entirely inside the allotted time budget (default
1000 s) and returns the best sequence found.

    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_gradient(a, beta):
    """
    Compute the gradient of the smooth max
        f(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
    w.r.t. a.  The gradient is
        ∇f = 2·conv(p, a_rev)
    where p_k = exp(β·b_k)/Σ_j exp(β·b_j) and a_rev is a reversed.
    """
    n = a.shape[0]
    # linear convolution a*a
    b = np.convolve(a, a)                 # length 2n‑1
    max_b = b.max()
    # soft‑max weights (shifted for numerical stability)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        # degenerate case – fall back to uniform weights
        p = np.full_like(b, 1.0 / b.size)
    else:
        p = w / w_sum
    # gradient = 2·conv(p, a_rev)  (extract the n relevant entries)
    rev_a = a[::-1]
    full = np.convolve(p, rev_a)          # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]    # centre slice gives length n
    return grad, max_b


def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights (floats).  The algorithm:

    1. Initialise with the analytically optimal power‑law a_i ∝ 1/√(i+1).
    2. Perform a short sweep over mixtures of the power‑law and the uniform
       distribution to obtain a better starting point.
    3. Run a projected gradient descent on a smooth approximation of the
       maximum of the self‑convolution, with an adaptive step size and
       increasing smoothing parameter β.
    4. Keep the best sequence ever seen (according to evaluate_sequence).

    The routine respects the time limit (budget_s seconds) and prints
    progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a small safety margin

    # ---- 1. Power‑law initialisation ------------------------------------
    n = 3000                                   # length of the sequence
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(idx)                     # a_i ∝ 1/√(i)
    a = a / a.sum()                            # ℓ₁‑normalise (sum = 1)

    # ---- 2. Mixture sweep (uniform vs. power‑law) -----------------------
    uniform = np.full(n, 1.0 / n)
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] score {best_score:.6f}")

    for lam in np.linspace(0.0, 1.0, 21):      # 21 points, inclusive
        if time.time() > deadline:
            break
        mix = (1.0 - lam) * uniform + lam * a   # already sums to 1
        score = evaluate_sequence(mix.tolist())
        if score < best_score:
            best_score = score
            best_a = mix.copy()
            print(f"[mix λ={lam: .2f}] improved to {best_score:.6f}")

    # ---- 3. Projected gradient descent on smooth max --------------------
    a = best_a.copy()
    curr_score = best_score

    # hyper‑parameters (tuned heuristically)
    beta = 10.0          # initial smoothing; will be increased
    lr = 0.5             # learning rate
    min_lr = 1e-6
    decay_lr = 0.9995    # slow decay
    increase_beta = 1.005

    while time.time() < deadline:
        # gradient of smooth max
        grad, max_b = _smooth_max_gradient(a, beta)

        # take a (projected) gradient step
        a_candidate = a - lr * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        # evaluate the candidate
        cand_score = evaluate_sequence(a_candidate.tolist())

        # simple acceptance rule: keep the move if it improves the score
        if cand_score < curr_score:
            a = a_candidate
            curr_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[{time.time():.1f}] new best {best_score:.6f}")

            # if the step was good we can be a bit more aggressive
            lr = min(lr / decay_lr, 1.0)   # undo a tiny part of decay
        else:
            # reject – shrink the step size a little
            lr = max(lr * decay_lr, min_lr)

        # increase β gradually so the smooth approximation becomes tighter
        beta *= increase_beta

    # ---- 4. Return the best sequence found -----------------------------
    print(f"[final] best score {best_score:.6f} with length {len(best_a)}")
    return best_a.tolist()
