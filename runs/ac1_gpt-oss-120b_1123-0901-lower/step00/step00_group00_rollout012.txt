import numpy as np
import time
from scipy.optimize import linprog

def arcsine_sequence(n):
    """Return a non‑negative sequence of length n whose entries approximate
    the discrete arcsine law.  After scaling the sum is sqrt(2n) which is the
    natural normalisation used by evaluate_sequence."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    target = np.sqrt(2.0 * n)
    vals = vals * (target / np.sum(vals))
    return vals.tolist()

def solve_convolution_lp(f_seq, rhs):
    """Linear programme:
    maximise   ∑ b_j
    subject to   (f_seq * b)[k] ≤ rhs  for every k,
                b_j ≥ 0.
    Returns the optimal b as a numpy array or None if the solver fails."""
    n = len(f_seq)
    # objective – maximise sum(b)  ⇔ minimise -1·b
    c = -np.ones(n, dtype=float)

    # Build the matrix for the convolution constraints.
    # Row k contains f_i for the pair (i, j) with i+j = k.
    a_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        # i runs over indices of f_seq, j = k - i must be a valid index of b.
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # the coefficients are the f_seq values that multiply the corresponding b_j.
        # For each i in [i_min, i_max] we have j = k - i.
        idxs = np.arange(i_min, i_max + 1)
        row[k - idxs] = f_seq[idxs]
        a_rows.append(row)
        b_rows.append(rhs)

    # non‑negativity constraints  -b ≤ 0
    a_rows.append(-np.eye(n))
    b_rows.append(np.zeros(n))

    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_rows)

    # Use the HiGHS interior‑point solver – it is fast and robust.
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    except Exception:
        return None

    if res.success:
        # Small negative values can appear due to numerical noise.
        return np.maximum(0.0, res.x)
    return None

def propose_candidate(seed=42, budget_s=1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence().  The routine starts from an arcsine‑shaped distribution
    and repeatedly solves a linear programme that finds a new feasible direction
    with the same convolution bound.  The new direction is normalised to the
    canonical L1‑norm sqrt(2 n) and, when it strictly reduces the maximal
    additive convolution, it replaces the current candidate.  Random perturbations
    and occasional restarts keep the search from getting stuck.
    The function respects the 1000 s time budget and returns the best sequence
    found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a working length.  1000–2000 gives a good trade‑off
    #     between quality of the bound and the size of the LP.
    # ------------------------------------------------------------------
    n = 1200

    # ------------------------------------------------------------------
    # 2.  Initialise the sequence – arcsine shape plus a tiny random jitter.
    # ------------------------------------------------------------------
    seq = np.array(arcsine_sequence(n), dtype=float)
    seq += np.random.rand(n) * 1e-4
    seq = np.clip(seq, 0.0, 1000.0)

    # Normalise – evaluate_sequence expects the raw sum, not a fixed value.
    target_sum = np.sqrt(2.0 * n)
    seq *= target_sum / np.sum(seq)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score: {best_score:.6f}')

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop.
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    while time.time() < deadline:
        it += 1

        # current bound (also the objective value after normalisation)
        cur_max = np.max(np.convolve(seq, seq))

        # --------------------------------------------------------------
        # LP step – find a feasible direction g with the same bound.
        # --------------------------------------------------------------
        g = solve_convolution_lp(seq.tolist(), cur_max)
        if g is None:
            # fall back to a tiny random move
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            continue

        # Normalise g to the standard L1‑norm √(2 n)
        g_sum = np.sum(g)
        if g_sum == 0:
            continue
        g = g * (target_sum / g_sum)

        # --------------------------------------------------------------
        # Candidate: replace the whole sequence by g.
        # --------------------------------------------------------------
        new_max = np.max(np.convolve(g, g))
        new_score = new_max                     # because sum is √(2n)

        # Accept if the bound strictly improves.
        if new_score < best_score:
            best_score = new_score
            best_seq = g.copy()
            print(f'Iter {it:4d}: new best {best_score:.6f}')
            no_improve = 0
        else:
            no_improve += 1

        # If g actually reduces the bound we keep it; otherwise we fall back to a
        # convex combination with a modest step size.
        if new_score < cur_max:
            seq = g
        else:
            # convex blend – the factor is chosen dynamically.
            t = 0.2
            seq = (1.0 - t) * seq + t * g
            # re‑normalise after blending (tiny drift may occur)
            seq = seq * (target_sum / np.sum(seq))

        # ------------------------------------------------------------------
        # 4.  Periodic random perturbations – help escape plateaus.
        # ------------------------------------------------------------------
        if it % 20 == 0:
            # make a shallow noisy copy of the current best
            pert = best_seq.copy()
            for _ in range(5):
                idx = np.random.randint(n)
                pert[idx] = max(0.0, pert[idx] + np.random.randn() * 5e-3)
            pert = pert * (target_sum / np.sum(pert))
            pert_score = np.max(np.convolve(pert, pert))
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert.copy()
                seq = pert
                print(f'Iter {it:4d}: perturb improved to {best_score:.6f}')
                no_improve = 0

        # ------------------------------------------------------------------
        # 5.  Early exit if we already reached the target bound.
        # ------------------------------------------------------------------
        if best_score < 1.503:
            print(f'Target reached at iteration {it}.')
            break

        # ------------------------------------------------------------------
        # 6.  Restart strategy – after many unsuccessful iterations we
        #     re‑initialise from a fresh random vector.

"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().  

The algorithm starts from the discrete arcsine law which is known to give a
nearly flat additive convolution.  The sequence is normalised so that
∑ a_i = √(2 n) – the normalisation used in the scoring function, therefore
the score equals the maximal value of the self‑convolution.

We repeatedly improve the sequence by solving a linear programme:

    maximise   ∑ b_j
    subject to (a * b)[k] ≤ M   for every k,
                b_j ≥ 0,

where M is the current maximal convolution value of the candidate a.
The LP is linear because a is fixed, while the objective pushes the mass
as far away from the current “hot spots’’ as possible.  The feasible vector
b is renormalised to the same L¹‑norm and replaces the current candidate if
it strictly reduces the maximal convolution.  Small convex blends are used
as a safety net when a full replacement does not improve the bound.
Random perturbations and occasional restarts avoid stagnation.

The routine respects a 1000 s time budget, prints progress, and returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: arcsine‐shaped initial sequence.
# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """Return a length‑n non‑negative vector approximating the discrete
    arcsine law and scaled to have ℓ₁‑norm √(2 n)."""
    i = np.arange(n, dtype=float)
    # 1 / √((i+½)·(n‑½‑i)) is the discretised arcsine density.
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    target = np.sqrt(2.0 * n)
    vals = vals * (target / np.sum(vals))
    # Guard against the (very unlikely) overflow of the clipping in
    # evaluate_sequence().
    vals = np.clip(vals, 0.0, 1000.0)
    return vals.tolist()


# ----------------------------------------------------------------------
# Linear programme that, for a fixed f, finds the feasible vector g with
# maximal total mass under the convolution bound M = max(f*f).
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: list[float], rhs: float) -> np.ndarray | None:
    """Solve:
       maximise   Σ b_j
       subject to Σ_i f_i·b_{k‑i} ≤ rhs  for all k = 0 … 2n‑2,
                   b_j ≥ 0.
    Returns the optimal b (as a NumPy array) or None on failure."""
    n = len(f_seq)
    # Objective: minimise –Σ b_j  (i.e. maximise Σ b_j)
    c = -np.ones(n, dtype=float)

    # Build the convolution constraints.
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        # i runs over indices of f_seq that actually contribute.
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx_i = np.arange(i_min, i_max + 1, dtype=int)
        # Corresponding indices in b are j = k - i.
        idx_j = k - idx_i
        row[idx_j] = np.array(f_seq)[idx_i]
        rows.append(row)
        rhs_vec.append(rhs)

    # Non‑negativity constraints: –b ≤ 0.
    rows.append(-np.eye(n))
    rhs_vec.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(rhs_vec)

    try:
        # HiGHS is the default high‑performance solver in recent SciPy.
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    except Exception:          # pragma: no cover
        return None

    if res.success:
        # Numerical noise can produce tiny negative entries.
        return np.maximum(0.0, res.x)
    return None


# ----------------------------------------------------------------------
# Main entry point required by the challenge.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises evaluate_sequence().
    The function runs for at most `budget_s` seconds, prints progress, and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # safety margin for clean exit

    # ------------------------------------------------------------------
    # 1. Choose a working length.
    #    1000–1500 gives a good balance between LP size and achievable bound.
    # ------------------------------------------------------------------
    n = 1200

    # ------------------------------------------------------------------
    # 2. Initialise with an arcsine shape plus a tiny jitter.
    # ------------------------------------------------------------------
    seq = np.array(arcsine_sequence(n), dtype=float)
    seq += np.random.rand(n) * 1e-4
    seq = np.clip(seq, 0.0, 1000.0)

    # Normalise to the required ℓ₁‑norm √(2 n).
    target_sum = np.sqrt(2.0 * n)
    seq *= target_sum / np.sum(seq)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    it = 0
    stagnation = 0   # counts iterations without improvement

    while time.time() < deadline:
        it += 1

        # Current maximal self‑convolution (the actual objective value).
        cur_max = np.max(np.convolve(seq, seq))

        # --------------------------------------------------------------
        # LP step: find a feasible direction g with the same bound M.
        # --------------------------------------------------------------
        g = solve_convolution_lp(seq.tolist(), cur_max)
        if g is None:
            # Very occasional fallback – a tiny random move.
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            continue

        # Renormalise g to the canonical ℓ₁‑norm √(2 n).
        g_sum = np.sum(g)
        if g_sum == 0.0:
            continue
        g = g * (target_sum / g_sum)

        # ----------------------------------------------------------------
        # Candidate: full replacement by g.
        # ----------------------------------------------------------------
        new_max = np.max(np.convolve(g, g))
        new_score = new_max          # because ∑g = √(2 n)

        # Track the overall best.
        if new_score < best_score:
            best_score = new_score
            best_seq = g.copy()
            stagnation = 0
            print(f"Iter {it:4d}: improved score {best_score:.6f}")

        # Accept the replacement if it truly decreases the bound;
        # otherwise perform a safe convex blend.
        if new_score < cur_max:
            seq = g
        else:
            t = 0.2
            seq = (1.0 - t) * seq + t * g
            # Re‑normalise after blending (tiny drift may appear).
            seq *= target_sum / np.sum(seq)

        # ----------------------------------------------------------------
        # Periodic random perturbations of the current best – help escape
        # plateaus.
        # ----------------------------------------------------------------
        if it % 20 == 0:
            pert = best_seq.copy()
            for _ in range(5):
                idx = np.random.randint(n)
                pert[idx] = max(0.0, pert[idx] + np.random.randn() * 5e-3)
            pert *= target_sum / np.sum(pert)
            pert_score = np.max(np.convolve(pert, pert))
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert.copy()
                seq = pert
                stagnation = 0
                print(f"Iter {it:4d}: perturb improved to {best_score:.6f}")

        # ----------------------------------------------------------------
        # Simple restart logic if we have been stuck for a while.
        # ----------------------------------------------------------------
        if new_score >= best_score:
            stagnation += 1
        if stagnation >= 80:
            seq = np.random.rand(n) * 0.01
            seq *= target_sum / np.sum(seq)
            stagnation = 0
            print(f"Iter {it:4d}: restart due to stagnation")

        # ----------------------------------------------------------------
        # Early exit if we already beat the target bound.
        # ----------------------------------------------------------------
        if best_score < 1.503:
            print(f"Target reached after {it} iterations.")
            break

    print(f"Finished after {it} iterations. Best score {best_score:.6f}")
    return best_seq.tolist()
