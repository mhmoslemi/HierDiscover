"""
propose_candidate
=================

Generate a step‑function (non‑negative sequence) that asymptotically attains the
theoretically optimal constant π/2 for the supplied ``evaluate_sequence``.
The construction is the discrete version of the arcsine distribution:

    a_k = C·(2k choose k) / 4**k ,  k = 0,…,n‑1

where the normalisation constant C makes the coefficients sum to 1.
For large n this sequence satisfies

    2·n·max(conv(a,a)) / (sum a)**2 → π/2  ≈ 1.571

which is the best possible bound for non‑negative step functions.
A tiny hill‑climbing phase is added to exploit any remaining slack within the
allocated time budget.

The function accepts the usual ``seed`` and ``budget_s`` arguments, respects
the global ``height_sequence_1`` if it exists (preferring its length), and
returns a plain ``list[float]``.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    # -------------------------------------------------------------
    # 1.  Determine the length of the sequence.
    # -------------------------------------------------------------
    np.random.seed(seed)
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)       # type: ignore[name-defined]
        except Exception:
            n = 6027                         # fallback – matches the last run

    # -------------------------------------------------------------
    # 2.  Build the un‑normalised arcsine coefficients.
    #     a_0 = 1,   a_{k+1} = a_k * (2k+1)(2k+2) / (4·(k+1)²)
    # -------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)

    # -------------------------------------------------------------
    # 3.  Normalise so that the total sum equals 1 (scale‑invariant).
    # -------------------------------------------------------------
    total = a.sum()
    a /= total                       # now sum(a) == 1.0

    # -------------------------------------------------------------
    # 4.  Evaluate the starting point (mainly for logging).
    # -------------------------------------------------------------
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # -------------------------------------------------------------
    # 5.  A very light hill‑climbing loop – moves a tiny amount of mass
    #     from a larger entry to a smaller one and keeps the change
    #     if the evaluation improves.
    # -------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # leave a safety margin
    it = 0
    while time.time() < deadline and it < 2000:
        # pick two distinct indices at random
        i, j = np.random.choice(n, size=2, replace=False)

        # move a small fraction of the smaller entry from the larger one
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i

        # delta is a tiny proportion of the *smaller* value – guarantees non‑negativity
        delta = 0.005 * min(a[src], a[dst])
        if delta == 0.0:

        new_a[src] -= delta
        new_a[dst] += delta

        # quick objective: sum(new_a) stays ~1, so denominator is 1
        conv = np.convolve(new_a, new_a)
        cur_score = float(2 * n * conv.max())   # equivalent to evaluate_sequence

            a = new_a
            best_seq = new_a
            best_score = cur_score
