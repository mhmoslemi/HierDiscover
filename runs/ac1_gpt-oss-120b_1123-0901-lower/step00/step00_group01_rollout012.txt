def project_onto_simplex(v):
    """Projects a vector onto the probability simplex (sum=1, v_i>=0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time, math

def project_simplex(v):
    # v: 1D array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all entries <=0, return uniform?
        return np.maximum(v, 0)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def full_convolution(a):
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    return conv

def gradient_maxconv(a, conv, offsets):
    n = len(a)
    grad = np.zeros_like(a)
    for k in offsets:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        # indices for a[i] and a_j where j = k-i
        # compute j = k - np.arange(i_start,i_end)
        js = k - np.arange(i_start, i_end)
        grad[i_start:i_end] += 2.0 * a[js]
    return grad

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Choose length: use height_sequence_1 if available else 2000
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    # Initialize with binomial central coefficients
    a = np.zeros(n)
    # compute central binomial / 4**i iteratively
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)  # ratio (2i-1)/(2i)
    # Normalize to sum 1
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2 * n * full_convolution(a).max()  # eval with sum=1
    # parameters
    eta0 = 0.1
    iteration = 0
    while time.time() < deadline:
        # compute convolution
        conv = full_convolution(a)
        max_val = conv.max()
        # compute max indices
        tol = max_val * (1 - 1e-12)  # compares
        # Use tolerance maybe 1e-12 relative
        mask = np.isclose(conv, max_val, rtol=0, atol=1e-12 * max_val)
        # find offsets
        offsets = np.where(mask)[0]
        # gradient
        grad = gradient_maxconv(a, conv, offsets)
        # step size
        eta = eta0 / math.sqrt(iteration + 1)
        # update
        a_new = a - eta * grad
        # project onto simplex
        a_new = project_simplex(a_new)
        a = a_new
        # evaluate
        score = 2 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iteration}: new best score {best_score:.6f}")
        # occasional random perturb to escape
        if iteration % 500 == 0 and iteration > 0:
            # add small random noise and renormalize
            noise = np.random.randn(n) * 0.01
            a = np.maximum(a + noise, 0)
            a = project_simplex(a)
        iteration += 1
    # return best as list
    return best_a.tolist()

"""
Projected sub‑gradient descent on the peak of the self‑convolution.

The evaluation metric is  2·n·max(conv(a,a)) / (sum(a)^2).
Because the metric is homogeneous of degree 0 we may fix sum(a)=1.
Then the problem reduces to: minimise  M(a)=max(conv(a,a))  with a_i≥0, Σa_i=1.

The algorithm:
    • Initialise a with the first n coefficients of the series
          a_i = C(2i,i) / 4**i    (the Taylor coefficients of (1‑z)^‑½).
      This choice is known to give a fairly flat convolution (≈π/2 bound).
    • Repeatedly:
        – Compute the full convolution a*a (FFT for speed).
        – Locate the index k where the convolution attains its maximum.
        – The sub‑gradient of max(conv) at a is
              g_i = 2·a_{k‑i}   for indices where k‑i is in range, else 0.
        – Take a step a←a−η·g, clip negatives to zero and renormalise to
          Σa_i=1 (projection onto the probability simplex by simple scaling).
        – If the step does not improve the peak, shrink η (back‑tracking).
    • Occasionally add a tiny random perturbation to escape plateaus.
    • Keep the best vector seen during the run and return it.

The method is fast (FFT of length ≈ 2·n) and works directly with the
non‑negative constraint, enabling us to improve the known bound (≈ 2.0)
towards the target ≈ 1.5 within the allocated time.
"""

import numpy as np
import time
import math

def _project_onto_simplex(v):
    """Project a vector onto the simplex {x≥0, sum(x)=1}."""
    # Simple scaling after clipping works because we never need the exact
    # Euclidean projection – it keeps the iterate feasible and is cheap.
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        # fall back to the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _full_convolution(a):
    """Return the linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a, conv, k_max):
    """Sub‑gradient of max(conv) w.r.t. a for the given maximising index."""
    n = len(a)
    grad = np.zeros_like(a)
    # contributions only where 0 ≤ k_max-i < n
    i_start = max(0, k_max - (n - 1))
    i_end = min(n, k_max + 1)      # exclusive
    # j = k_max - i
    js = k_max - np.arange(i_start, i_end)
    grad[i_start:i_end] += 2.0 * a[js]
    return grad

    current bound.  The algorithm uses projected sub‑gradient descent
    starting from the “square‑root of the all‑ones” series.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # use the existing length if available
    except NameError:

    # ------------------------------------------------------------------
    # 2. Initialise a with the central‑binomial coefficients.
    #    a_i = C(2i,i) / 4**i   (i = 0 … n‑1)
    # ------------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_i = a_{i‑1} * (2·i‑1)/(2·i)
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a = _project_onto_simplex(a)         # normalise Σa_i = 1

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    # ------------------------------------------------------------------
    best_a = a.copy()
    conv = _full_convolution(a)
    best_score = 2.0 * n * conv.max()    # same as evaluate_sequence for Σ=1
    print(f"Initial score: {best_score:.6f}")

    # parameters for the step‑size schedule
    eta0 = 0.1
    iteration = 0

    while time.time() < deadline:
        # compute current convolution and its peak
        conv = _full_convolution(a)
        max_val = conv.max()
        k_max = int(conv.argmax())     # a single maximising index is enough

        # sub‑gradient of the peak
        grad = _subgradient_maxconv(a, conv, k_max)

        # back‑tracking line‑search
        eta = eta0 / math.sqrt(iteration + 1.0)
        improved = False
        for _ in range(5):
            a_new = a - eta * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new)
            new_max = conv_new.max()
            if new_max <= max_val:      # accept any non‑worsening step
                a = a_new
                max_val = new_max
                improved = True
                break
            eta *= 0.5                  # shrink step‑size and retry

        if not improved:
            # if none of the trial steps improved, fall back to a tiny random kick
            noise = np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a + noise)

        # keep the best solution seen so far
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: new best score {best_score:.6f}")

        # occasional random perturbation to escape shallow plateaus
        if iteration % 1000 == 0 and iteration > 0:
            perturb = np.random.randn(n) * 0.005
            a = _project_onto_simplex(a + perturb)

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain list of python floats)
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations. Best score = {best_score:.6f}")
    return best_a.tolist()
