def propose_move(p):
    n = len(p)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:

    # propose moving some mass from i to j
    amount = np.random.exponential(scale=0.01) * p[i]  # fraction
    amount = min(amount, p[i])
    new_p = p.copy()
    new_p[i] -= amount

"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm is a simple stochastic optimisation that works directly on the
probability vector `p = a / sum(a)`.  The evaluation metric
   f(p) = 2·n·max(p * p)      (where “*” denotes discrete convolution)
is independent of overall scaling, so we keep the vector normalised to sum 1.

* A fast convolution via FFT (O(n log n)) is used to obtain the maximum
  value and its index.
* For the current maximal index `k_max` the sub‑gradient of the objective is
      g_i = p_{k_max-i}   (if the term exists)  +  p_i  (if 2*i == k_max)
  (this is the exact derivative of the active max‑convolution term).
* One iteration moves a small amount of probability mass from the index with
  the largest gradient component to the index with the smallest component.
  This reduces the contribution to the current maximal convolution value.
* Occasionally a completely random pairwise move is tried to escape plateaus.
* A simple simulated‑annealing acceptance rule with a linearly decreasing
  temperature allows occasional uphill moves.
* The whole process is repeated with many random restarts (different lengths,
  different initialisations) until the time budget is exhausted.  The best
  sequence found (with the lowest `evaluate_sequence` value) is returned.

All numbers stay non‑negative and are far below the built‑in limits
(0 ≤ x ≤ 1000).  The final sequence is returned as a plain Python list of
floats.
"""

import time
import math
import random
import numpy as np

# --------------------------------------------------------------
# Helper functions – no closures, top‑level only
# --------------------------------------------------------------

def _conv_fft(p: np.ndarray) -> np.ndarray:
    """Return the full linear convolution p * p using a power‑of‑2 FFT."""
    n = p.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fp * fp, size)[: 2 * n - 1]
    return conv


def _max_conv(p: np.ndarray):
    """Maximum value of p * p and the index where it occurs."""
    conv = _conv_fft(p)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _gradient_at_max(p: np.ndarray, kmax: int):
    """Sub‑gradient of max(p*p) at the active index kmax."""
    n = p.shape[0]
    i = np.arange(n)
    j = kmax - i
    g = np.zeros_like(p)
    mask = (j >= 0) & (j < n)
    g[mask] = p[j[mask]]
    diag = (2 * i == kmax)
    g[diag] += p[diag]                # diagonal terms have factor 2
    return g


def _random_dirichlet(n: int, alpha: float = 0.3) -> np.ndarray:
    """Draw a random probability vector of length n (Dirichlet with given alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.random.rand(n)
    return x / x.sum()
