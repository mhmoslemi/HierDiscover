def generate_sequence(M, max_len=10000, eps=1e-12):
    a = []
    a0 = math.sqrt(M)
    a.append(a0)
    for k in range(1, max_len):
        # compute inner sum
        inner = 0.0
        # i from 1 to k-1
        for i in range(1, k):
            inner += a[i] * a[k - i]
        ak = (M - inner) / (2 * a0)
        if ak <= 0 or math.isnan(ak):
            break
        a.append(ak)
    return a

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)
    # a_ub rows for each shift k
    rows = []
    for k in range(2*n -1):
        row = np.zeros(n)
        # i from max(0, k-(n-1)) to min(n-1,k)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # loop i and set row[j] += f_seq[i] where j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f_seq[i]  # note for each i there is coefficient f_i for variable b_j
        rows.append(row)
    a_ub = np.vstack(rows + [-np.eye(n)])
    b_ub = np.hstack([np.full(2*n-1, rhs), np.zeros(n)])
    result = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'time_limit': 5.0})
    if result.success:
        return result.x
    return None
```

def get_improved_direction(seq, t=0.1):
    n = len(seq)
    total = np.sum(seq)
    if total == 0:
        return None
    normalized = seq * np.sqrt(2 * n) / total
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    normalized_g = g * np.sqrt(2 * n) / sum_g
    new_seq = [(1 - t) * x + t * y for x, y in zip(seq, normalized_g)]
    return new_seq
```

"""
Propose a high‑quality step‑function (non‑negative height sequence).

The core idea is a known analytic construction:
    a_k = C(2k,k) / 4**k   (central binomial coefficients)
which satisfies  (a * a)_k = 1  for every k.
For a finite prefix this gives a score ≈ π/2 ≈ 1.571, already better than the
uniform bound 2.0.

To push the score further we iteratively improve the sequence with a linear
program that, for the current sequence `f`, finds a direction `g` that
maximises the total sum while keeping the convolution `f * g` bounded by the
current maximum `M = max(f*f)`.  A small convex combination
    f ← (1‑t)·f + t·g
strictly increases the total sum without raising the peak of the autocorrelation,
thereby lowering the evaluation value
    2·n·max(f*f) / (sum(f))**2.

The algorithm works under the 1000 s time budget, keeps everything in memory,
and never uses filesystem or network I/O.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
#  Analytic base sequence (central binomial coefficients)
# ----------------------------------------------------------------------
def analytic_sequence(max_len: int) -> list[float]:
    """Return the first `max_len` terms of (2k choose k)/4**k."""
    seq = [1.0]          # a_0 = 1
    for k in range(1, max_len):
        # recurrence: a_k = a_{k-1} * (2k-1) / (2k)
        seq.append(seq[-1] * (2 * k - 1) / (2.0 * k))
    return seq


# ----------------------------------------------------------------------
#  Linear program: maximise sum(b) subject to (f * b)_k <= M   and b >= 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: list[float], bound: float) -> list[float] | None:
    """
    Solve  max  Σ b_j
          s.t. Σ_i f_i * b_{k-i} ≤ bound   for each k (0 … 2n‑2)
               b_j ≥ 0
    Returns the optimal b (as a dense np.ndarray) or None on failure.
    """
    n = len(f_seq)
    # objective (negative because linprog does minimisation)
    c = -np.ones(n)

    # Build A_ub row‑by‑row.  For each shift k we need:
    #   Σ_i f_i * b_{k-i} ≤ bound
    a_ub_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill the row: b_{j} gets coefficient f_i where j = k-i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        a_ub_rows.append(row)

    # non‑negativity constraints  -b_j ≤ 0
    a_ub_rows.append(-np.eye(n))
    a_ub = np.vstack(a_ub_rows)

    # RHS vector
    b_ub = np.hstack([np.full(2 * n - 1, bound), np.zeros(n)])

    # use HiGHS solver; give a modest per‑solve time limit
    res = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
#  One improvement step: obtain a better direction and take a convex step
# ----------------------------------------------------------------------
def improve_step(sequence: list[float], step: float = 0.05) -> list[float] | None:
    """
    Given a sequence `sequence`, compute a better neighbour using the LP described
    above and return the mixed sequence.  Returns None if the LP fails.
    """
    n = len(sequence)
    total = float(np.sum(sequence))
    if total < 1e-12:
        return None

    # normalise to the conventional scale Σf = √(2n)
    norm_factor = math.sqrt(2.0 * n) / total
    f_norm = [x * norm_factor for x in sequence]

    # current maximal convolution value
    conv = np.convolve(f_norm, f_norm)
    bound = float(np.max(conv))

    # solve the LP for the direction g
    g_raw = solve_convolution_lp(f_norm, bound)
    if g_raw is None:
        return None

    # re‑normalise g to the same Σ = √(2n)
    sum_g = float(np.sum(g_raw))
    if sum_g < 1e-12:
        return None
    g_norm = [x * math.sqrt(2.0 * n) / sum_g for x in g_raw]

    # convex combination
    new_seq = [(1.0 - step) * a + step * b for a, b in zip(sequence, g_norm)]
    return new_seq


# ----------------------------------------------------------------------
#  Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a sequence of non‑negative heights that (hopefully) beats the
    uniform upper‑bound of 2.0.  The routine:
        1. builds the analytic base sequence (≈π/2 score),
        2. repeatedly applies the LP improvement step while time permits,
        3. keeps and returns the best sequence seen.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a safety margin

    # ---- 1. Analytic start -------------------------------------------------
    max_start_len = 1500                      # modest size – fits into LP memory
    base_seq = analytic_sequence(max_start_len)
    # Trim negligible tail – very small entries add length without profit
    base_seq = [x for x in base_seq if x > 1e-12]
    curr_seq = base_seq.copy()
    best_seq = curr_seq.copy()
    best_score = evaluate_sequence(best_seq)
    print(f"Initial length {len(curr_seq)}  score {best_score:.6f}")

    # ---- 2. Iterative LP improvements -------------------------------------
    step_size = 0.07   # moderately aggressive at the start
    while time.time() < deadline:
        new_seq = improve_step(curr_seq, step=step_size)
        if new_seq is None:
            # LP failed – fall back to a tiny random jitter to escape stagnation
            idx = np.random.randint(len(curr_seq))
            jitter = np.random.randn() * 0.005
            curr_seq[idx] = max(0.0, curr_seq[idx] + jitter)
            continue

        # evaluate the candidate
        try:
            cur_score = evaluate_sequence(new_seq)
        except Exception:          # safety net – shouldn't happen
            continue

        # keep the best one
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            print(f"New best {best_score:.6f}  length {len(best_seq)}  step {step_size:.4f}")

        # accept the move (always move – the LP is constructed to not increase M)
        curr_seq = new_seq

        # slowly decay the step size to fine‑tune later
        step_size = max(0.001, step_size * 0.9995)

    # ---- 3. Final tidy‑up --------------------------------------------------
    # discard trailing entries that are effectively zero (they only increase n)
    final_seq = [x for x in best_seq if x > 1e-12]
    final_score = evaluate_sequence(final_seq)
    print(f"Finished – best score {final_score:.6f}  length {len(final_seq)}")
    return final_seq
