"""
Simple projected sub‑gradient optimizer for the max‑convolution objective.

The evaluation function is scale invariant:
    score = 2·n·max(conv(a,a)) / (sum(a)²)

Hence we fix the ℓ₁‑norm to sqrt(2·n) and minimise the maximal
convolution value.  The algorithm works as follows

1. Initialise a sequence from the analytically known “constant‑convolution”
   coefficients (c₀,c₁,…) where  c_k = binom(2k,k)/4^k .  After scaling its
   ℓ₁‑norm to sqrt(2·n) this provides a good starting point (score ≈1.6).

2. Repeatedly perform a projected sub‑gradient step.
   * The current maximal convolution entry is located (index k*).
   * The sub‑gradient of that entry w.r.t. the vector a is
         g_i = 2·a_{k*‑i}   (zero when the index is out of range).
   * A step a ← a – η·g is taken and the result is projected back onto the
     simplex {a_i ≥ 0, Σ a_i = sqrt(2·n)}.

   The step size η is adapted (increase on improvement, decrease otherwise).

3. Occasionally a small random perturbation is added to escape shallow
   plateaux; after a number of iterations a fresh restart with a different
   length is performed.

The whole search runs for the allotted time (default 1000 s) and the best
sequence found is returned.  All numbers are ordinary Python ``float`` objects,
as required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)          # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]        # last index where condition holds
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient step on the maximal convolution entry
def _subgradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step.
    a    – current non‑negative vector (ℓ₁‑norm already fixed)
    eta  – step size (positive float)

    Returns the new vector (still on the simplex).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))               # index of the maximal entry
    # build the sub‑gradient of conv_k w.r.t. a
    grad = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    inds = np.arange(i_min, i_max + 1, dtype=int)
    # grad[i] = 2 * a[k-i] for the admissible indices
    grad[inds] = a[k - inds]
    grad *= 2.0

    # take a step and re‑project onto the simplex
    a_new = a - eta * grad
    a_new = _project_onto_simplex(a_new, np.sum(a))
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided `evaluate_sequence` metric.  The routine runs for at most
    `budget_s` seconds and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored (kept for compatibility with the evaluation harness).

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5      # a small safety margin

    best_seq = None
    best_score = np.inf

    # a few different problem sizes are explored – they are cheap enough
    # for the allocated budget and give the optimiser several “land‑scapes”.
    length_candidates = [200, 400, 800, 1200, 1600]

    # ------------------------------------------------------------------
    # Main optimisation loop: each outer iteration picks a length,
    # builds a fresh start (biased towards the analytically known solution),
    # then runs a limited number of sub‑gradient steps.
    while time.time() < deadline:
        n = int(rng.choice(length_candidates))
        target_sum = np.sqrt(2.0 * n)

        # ----- start from the constant‑convolution coefficients (c_k)
        c = np.empty(n, dtype=float)
        c[0] = 1.0
        for i in range(1, n):
            # recurrence: c_{i} = c_{i-1} * (2(i-1)+1) / (2(i-1)+2)
            c[i] = c[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * (i - 1) + 2.0)
        a = c * (target_sum / c.sum())

        # add a tiny random perturbation so that the algorithm does not
        # sit exactly on a stationary point of the analytical construction
        a += rng.normal(scale=target_sum * 0.001, size=n)
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, target_sum)

        # evaluate the starting point
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()

        # ----- sub‑gradient optimisation for this length
        # initial step size – a fraction of the typical entry magnitude
        eta = target_sum / n * 0.5
        it = 0
        while time.time() < deadline and it < 2500:
            a_new = _subgradient_step(a, eta)
            new_score = evaluate_sequence(a_new.tolist())

            # accept if we are strictly better; otherwise shrink step size
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                # on improvement we try to enlarge the step a little
                eta *= 1.02
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.tolist()
            else:
                eta *= 0.94   # back‑off
                # occasional random “kick” to avoid stagnation
                if rng.random() < 0.03:
                    kick = rng.normal(scale=eta, size=n)
                    a_kick = a + kick
                    a_kick = np.maximum(a_kick, 0.0)
                    a_kick = _project_onto_simplex(a_kick, target_sum)
                    a = a_kick
                    cur_score = evaluate_sequence(a.tolist())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.tolist()
            it += 1

        # small chance to reshuffle the length pool (helps diversification)
        if rng.random() < 0.1:
            rng.shuffle(length_candidates)

    # ------------------------------------------------------------------
    # If nothing was found (should never happen) fall back to the uniform vector
    if best_seq is None:
        n = 1000
        uniform_value = np.sqrt(2.0 * n) / n
        best_seq = [float(uniform_value)] * n

    return best_seq
