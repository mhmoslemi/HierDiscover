The projection onto simplex: given vector v, we return w = max(v - τ, 0) where τ chosen such that sum w = 1.

Implementation via sorting descending.

Pseudo:

def project_simplex(v):
    # ensure sum to 1, v >=0 after projection
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Note this works for any vector v (not necessarily nonnegative). We'll use numpy.

"""
Propose a candidate step‑function height sequence.

The algorithm works in two phases:

1. **Coarse grid search** – try short exponential sequences
   a_i = (1‑r) r**i for several values of length ``n`` and ratio
   ``r``.  This quickly finds a good starting point (geometric
   sequences give ≈ 1.84, already better than the uniform 2.0 bound).

2. **Projected sub‑gradient descent** – minimise the maximal
   convolution value ``max(conv(a,a))`` while keeping the vector
   non‑negative and normalised (∑a_i = 1).  The sub‑gradient of the
   active convolution entry ``k_max`` is ``g_j = 2*a_{k_max‑j}``
   (zero where the index is out of range).  After a gradient step the
   vector is projected back onto the probability simplex (non‑negative,
   sum‑to‑one) so the evaluation function remains invariant to scaling.
   The step size decays like ``η = η0 / sqrt(t+1)``.

The whole search runs for the allotted time (≈ 1000 s) and returns the
best sequence found.  All numbers are ≤ 1000 after the security
checks in ``evaluate_sequence``.
"""

import time
import numpy as np

# ---------- utilities -------------------------------------------------

def project_simplex(v):
    """
    Euclidean projection of a vector onto the simplex {x >= 0, sum(x)=1}.
    Implements the O(n log n) algorithm from:
    Duchi et al., "Efficient Projections onto the L1-Ball for
    Learning in High Dimensions", ICML 2008.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1  # cumulative sum minus 1
    ind = np.arange(n) + 1
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def geometric_sequence(n, r):
    """Return a length‑n geometric sequence a_i = (1‑r) r**i."""
    # (1‑r) makes the L1‑norm close to 1 (exactly 1‑r**n)
    idx = np.arange(n, dtype=np.float64)
    seq = (1.0 - r) * np.power(r, idx)
    return seq


def evaluate(seq):
    """Helper that forwards to the already imported evaluate_sequence."""
    return evaluate_sequence(seq.tolist())


# ---------- main optimisation routine --------------------------------

def propose_candidate(budget_s=1000, seed=123):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a few seconds spare

    best_seq = None
    best_score = float('inf')

    # ----- Phase 1: coarse grid over geometric families -----------------
    print("Phase 1: grid search over geometric sequences")
    candidate_lengths = [500, 800, 1200, 2000]   # moderate sizes, fast eval
    r_grid = np.linspace(0.80, 0.99, 191)       # step ≈ 0.001

    for n in candidate_lengths:
        for r in r_grid:
            seq = geometric_sequence(n, r)
            score = evaluate(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
        print(f"  n={n:4d}  best score so far = {best_score:.6f}")

    # if something went wrong keep a uniform fallback
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate(best_seq)
        print("Fallback to uniform sequence, score:", best_score)

    # ----- Phase 2: projected sub‑gradient descent --------------------
    print("\nPhase 2: sub‑gradient descent to reduce max(conv)")
    a = best_seq.copy()
    n = a.shape[0]

    # Normalise to sum = 1 (scaling does not affect the objective)
    a = a / np.sum(a)
    current_score = evaluate(a)

    itr = 0
    eta0 = 0.1   # initial step size
    while time.time() < deadline:
        # compute convolution and identify the active index
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # compute sub‑gradient of the active convolution term
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi + 1, dtype=int)
            grad[indices] = 2.0 * a[k_max - indices]

        # step size schedule
        eta = eta0 / np.sqrt(itr + 1)

        # gradient step and projection onto simplex
        a_candidate = project_simplex(a - eta * grad)

        # evaluate candidate
        cand_score = evaluate(a_candidate)

        # accept if we made progress
        if cand_score < current_score:
            a = a_candidate
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
            print(f"Iter {itr:5d}: improved score = {cand_score:.6f}")
        # occasional random perturbation to escape flat regions
        elif itr % 50 == 0:
            # small random jitter, keep feasibility via projection
            jitter = np.random.normal(scale=0.01, size=n)
            a_candidate = project_simplex(a + jitter)
            cand_score = evaluate(a_candidate)
            if cand_score < current_score:
                a = a_candidate
                current_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                print(f"Iter {itr:5d} (jitter): improved score = {cand_score:.6f}")

        itr += 1
        # safety break for very long runs
        if itr > 5000:
            break

    print("\nFinished optimisation")
    print(f"Best score found: {best_score:.6f} (length = {len(best_seq)})")
    # Return as a plain Python list of floats
    return best_seq.tolist()
