def generate_sequence_exponential(n, eps):
    r = 1.0 - eps
    # compute denominator (1 - r**n) robustly
    log_r = np.log(r)
    rn = np.exp(log_r * n)
    denom = 1.0 - rn
    S = np.sqrt(2.0 * n)
    A = S * eps / denom
    # generate a_i
    i = np.arange(n, dtype=np.float64)
    seq = A * np.exp(log_r * i)   # A * r**i
    # Ensure nonnegative and clamp to 0..1000 just in case
    seq = np.clip(seq, 0, 1000)
    return seq.tolist()

def generate_exponential_sequence(n, eps):
    # compute r = 1 - eps; handle eps may be zero.
    if eps <= 0.0:
        r = 1.0
    else:
        r = 1.0 - eps
    # compute denominator (1 - r**n) robustly
    if r == 0:
        denom = 1.0
    else:
        # Use log to compute r**n
        log_r = math.log(r)
        rn = math.exp(log_r * n)
        denom = 1.0 - rn
    S = math.sqrt(2.0 * n)
    A = S * eps / denom if denom != 0 else 0.0
    # generate using exponentials
    idx = np.arange(n, dtype=np.float64)
    seq = A * np.exp(log_r * idx) if r != 1.0 else np.full(n, S / n)
    # Ensure after rounding float not negative
    seq = np.clip(seq, 0.0, 1000.0)
    return seq.tolist()

            best_score = score
            best_seq = seq

    # linear decreasing shape: a_i = (n - i)
    # But scaling ensures sum.
    # Might try multiple offsets
    slopes = np.linspace(0.0, 2.0, 20)
    for a in slopes:
        if time.time()>deadline: break

        # Actually we want decreasing: start high to low:
        raw = np.linspace(a, 1.0, n)  # descending
        sum_raw = np.sum(raw)
        seq = (raw / sum_raw) * S
        seq = np.clip(seq, 0.0, 1000.0).tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"Linear a={a:.3f} score={score:.6f}")

    # random perturbation from best
    if best_seq is not None:
        seq = best_seq.copy()
        # hill climbing
        while time.time() < deadline:
            # propose small local move: pick two indices, transfer some mass
            i, j = np.random.randint(0, n, size=2)
            delta = (np.random.rand() - 0.5) * 0.01 * seq[i]
            new_seq = seq.copy()
            new_seq[i] = max(0.0, new_seq[i] + delta)
            new_seq[j] = max(0.0, new_seq[j] - delta)
            # Ensure sum not too far off; can re-normalize to sum S
            # Compute diff
            diff = sum(new_seq) - S
            if abs(diff) > 1e-9:
                # adjust uniformly

            score = evaluate_sequence(new_seq)
            if score < best_score:
                best_score = score
                best_seq = new_seq
                seq = new_seq
                print(f"Hill climb improved score {best_score:.6f}")
            else:

"""
Simple search for a low‑value step‑function.

The core idea is to start from a few analytically‑motivated families
(exponential decay, piecewise linear, random Dirichlet) that already give
a bound below 2.0 and then improve them with a small number of linear‑
programming “direction” steps (the same idea as the published SOTA
iteration).  Each LP finds a non‑negative sequence g with a larger
ℓ₁‑norm while respecting the current convolution bound M = max(f∗f).
After rescaling g to the same ℓ₁‑norm as f we mix the two sequences.
A short hill‑climbing phase finishes the search.

The algorithm respects the 1000 s time budget, prints progress and
returns the best sequence found.
"""
import math
import time

import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: generate an exponential‑decay sequence, exactly normalized.
def gen_exponential(n: int, eps: float) -> list[float]:
    """
    a_i = A * (1 - eps)^i ,  i = 0 .. n-1,
    scaled so that sum_i a_i = sqrt(2*n).
    """
    if eps <= 0.0:
        # degenerate case – fall back to uniform
        S = math.sqrt(2.0 * n)
        return [S / n] * n

    r = 1.0 - eps
    # use log to avoid under‑flow for r**n
    log_r = math.log(r)
    rn = math.exp(log_r * n)                # r**n
    denom = 1.0 - rn                         # 1 - r**n
    S = math.sqrt(2.0 * n)                   # target ℓ₁‑norm
    A = S * eps / denom                       # scaling factor
    i = np.arange(n, dtype=np.float64)
    seq = A * np.exp(log_r * i)               # A * r**i
    # clip only to obey the evaluator’s limits
    seq = np.clip(seq, 0.0, 1000.0)
    return seq.tolist()


# ----------------------------------------------------------------------
# Helper: piecewise linear (descending) sequence, normalized.
def gen_linear_desc(n: int, start_factor: float) -> list[float]:
    """
    descending arithmetic progression: a_0 = start, a_{n-1}=0,
    scaled to the required ℓ₁‑norm.
    start_factor controls the slope (larger => higher start value).
    """
    raw = np.linspace(start_factor, 0.0, n, dtype=np.float64)
    S = math.sqrt(2.0 * n)
    raw = raw / np.sum(raw) * S
    raw = np.clip(raw, 0.0, 1000.0)
    return raw.tolist()


# ----------------------------------------------------------------------
# Helper: random Dirichlet‑distributed sequence, normalized.
def gen_dirichlet(n: int, alpha: float) -> list[float]:
    """Draw a Dirichlet vector with concentration alpha and scale to ℓ₁‑norm."""
    draws = np.random.gamma(alpha, 1.0, size=n)
    S = math.sqrt(2.0 * n)
    seq = draws / draws.sum() * S
    seq = np.clip(seq, 0.0, 1000.0)
    return seq.tolist()


# ----------------------------------------------------------------------
# Linear programming step – identical to the SOTA routine but with a
# configurable time limit.
def solve_convolution_lp(f_seq: list[float], rhs: float, lp_timelimit: float = 5.0):
    """
    Solve:
        max  sum(b)
        s.t. conv(f_seq, b) ≤ rhs   (for every shift)
             b ≥ 0
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                      # maximize sum(b)  →  minimize -sum(b)

    # build the convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints   –b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # SciPy’s HiGHS backend is very fast for this size.
    try:
        res = optimize.linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": lp_timelimit, "presolve": True},
        )
        if res.success:
            return np.maximum(0.0, res.x)   # numerical safety
    except Exception:
        pass
    return None


# ----------------------------------------------------------------------
# Small hill‑climbing / simulated‑annealing style refinement.
def local_refine(seq: list[float], budget: float) -> list[float]:
    """
    Randomly transfer a tiny amount of mass between two positions,
    accepting any move that improves the evaluation score.
    Runs for ``budget`` seconds.
    """
    n = len(seq)
    best = seq[:]
    best_score = evaluate_sequence(best)
    deadline = time.time() + budget

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a small fraction of a_i to a_j
        delta = (np.random.rand() - 0.5) * 0.02 * best[i]  # up to ±1 % of a_i
        if best[i] + delta < 0.0:
            continue
        cand = best[:]
        cand[i] = max(0.0, cand[i] + delta)
        cand[j] = max(0.0, cand[j] - delta)

        # re‑normalise to keep the ℓ₁‑norm exact (tiny drift may appear)
        sum_cand = sum(cand)
        target = math.sqrt(2.0 * n)
        if sum_cand != 0.0:
            scale = target / sum_cand
            cand = [x * scale for x in cand]

        sc = evaluate_sequence(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            # occasional log
            if int(time.time()) % 30 == 0:
                print(f"  local refine improved → {best_score:.6f}")
    return best

    lp_deadline = time.time() + lp_budget
    t_mix = 0.05                     # start mixing weight

    while time.time() < lp_deadline:
        # Normalise (it should already be normalised, but guard against drift)
        cur_sum = sum(best_seq)
        if cur_sum == 0:
            break
        factor = target_norm / cur_sum
        f_norm = [x * factor for x in best_seq]

        # current bound M = max(f∗f); for a normalised sequence it equals the
        # evaluation score.
        M = evaluate_sequence(f_norm)

        g = solve_convolution_lp(f_norm, M, lp_timelimit=4.0)
        if g is None:
            break

        # Scale g back to the target ℓ₁‑norm.
        sum_g = float(g.sum())
        if sum_g == 0:
            break
        g_scaled = (g * (target_norm / sum_g)).tolist()

        # Convex combination.
        new_seq = [(1.0 - t_mix) * a + t_mix * b for a, b in zip(best_seq, g_scaled)]

        new_score = evaluate_sequence(new_seq)
        if new_score < best_score:
            best_seq, best_score = new_seq, new_score
            print(f"LP step improvement → score={best_score:.6f}")
            # be slightly more aggressive after a win
            t_mix = min(t_mix * 1.2, 0.3)
        else:
            # back‑off if we did not improve
            t_mix *= 0.5
            if t_mix < 1e-3:
                break

    # ------------------------------------------------------------------
    # 3)  Light local optimisation ----------------------------------------
    # Use the remaining time for a simple hill‑climb.
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"starting local refinement, {remaining:.1f}s left")
        best_seq = local_refine(best_seq, budget=remaining * 0.9)

    print(f"final best score = {best_score:.6f}")
    return best_seq
