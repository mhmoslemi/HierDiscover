def project_to_simplex(v):
    # returns projection onto {x >=0, sum x = 1}
    v = np.array(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def optimize_n(n, iterations, step_scale):
    a = np.random.exponential(size=n)
    a = a / a.sum()
    best_a = a.copy()
    best_val = evaluate(a)
    cur = a.copy()
    cur_val = best_val
    T0 = 1.0
    for it in range(iterations):
        # propose move
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * step_scale
        if cur[i] + delta < 0:
            delta = -cur[i]
        cur_new = cur.copy()
        cur_new[i] += delta
        cur_new[j] -= delta
        # ensure nonneg
        if cur_new[j] < 0:
            continue
        cur_new = cur_new / cur_new.sum()
        val = evaluate(cur_new)
        if val < best_val:
            best_val = val
            best_a = cur_new.copy()
        # acceptance
        if val < cur_val or np.random.rand() < math.exp(-(val - cur_val) / (T0 * (1 - it/iterations))):
            cur = cur_new
            cur_val = val
```

def propose_candidate(seed=0, budget_s=990, **kwargs):
    import numpy as np, time, math, random
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5
    best_seq = None
    best_val = float('inf')
    # try multiple lengths
    lengths = list(range(2, 20)) + [30,40,50,70,100]
    while time.time() < deadline:
        n = random.choice(lengths)
        # init random Dirichlet
        a = np.random.dirichlet(alpha=np.ones(n))
        cur_val = evaluate_a(a)
        # keep local best for this n
        # SA parameters
        T0 = 1.0
        for it in range(2000):
            # propose move
            i, j = np.random.randint(n), np.random.randint(n)
            if i == j:
                continue
            delta = (np.random.rand() - 0.5) * 0.1 * a[i]  # relative move
            if a[i] - delta < 0:
                delta = a[i]
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            # re-normalize maybe not needed because sum unchanged
            # compute value
            val = evaluate_a(new_a)
            # acceptance
            if val < cur_val or np.random.rand() < math.exp(-(val - cur_val) / (T0 * (1 - it/2000))):
                a = new_a
                cur_val = val
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
        # occasional gradient step:
        # we may compute conv and gradient to reduce max
        # Not needed for simple version
    # fallback if no seq
    if best_seq is None:
        # simple 2-element optimum
        best_seq = np.array([1/3, 2/3])
        best_val = evaluate_a(best_seq)
    # Convert to list of floats
    result = [float(x) for x in best_seq]

def propose_candidate(seed=123, budget_s=900):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2
    best_seq = None
    best_val = float('inf')
    lengths = [2,3,4,5,6,7,8,9,10,12,15,20,30,40,50]
    for n in lengths:
        # restart several times
        for restart in range(5):
            a = np.random.dirichlet(alpha=np.ones(n))
            cur_val = evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
            temp = 1.0
            for it in range(5000):
                if time.time() >= deadline:
                    break
                # propose move
                i = np.random.randint(n)
                j = np.random.randint(n-1)
                if j >= i:
                    j += 1
                eps = (np.random.rand() - 0.5) * 0.1  # step size fraction
                if a[i] + eps < 0:
                    eps = -a[i]
                a_new = a.copy()
                a_new[i] += eps
                a_new[j] -= eps
                if a_new[j] < 0:
                    # adjust to be nonnegative
                    continue
                # renormalize to sum 1
                a_new = a_new / a_new.sum()
                new_val = evaluate(a_new)
                # accept if better or with probability
                if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
                    a = a_new
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best_seq = a.copy()
                # cooldown temp
                temp *= 0.9999
    # ensure non-negative and maybe small epsilon
    result = [float(x) for x in best_seq]
    print("Best value:", best_val)
    return result

"""
Simple stochastic optimizer for the self‑convolution bound.

The objective (ignoring the overall scale) is:
    f(a) = 2 * n * max(conv(a,a))
with n = len(a) and sum(a)=1 (the sum is normalized because the
evaluation function is scale‑invariant).  The optimizer searches over
different lengths, uses a projected‑gradient step that reduces the
currently maximal convolution entry and a random “mass transfer”
move to escape plateaus.  The best vector seen within the time budget
is returned.
"""

import time
import math
import random
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – the projection is the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Computes 2 * n * max(conv(a,a)).  The vector `a` is assumed to sum to 1.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv)


def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient step that tries to decrease the presently maximal
    convolution entry.  The step size is a positive scalar.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(conv.argmax())                     # index with the current maximum
    # gradient of conv_k w.r.t. a_i is 2 * a_{k-i} (zero if out of bounds)
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    # take a step opposite to the gradient, keep non‑negative and project
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new)
    return a_new


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small amount of mass from one coordinate to another.
    The amount is drawn from a symmetric uniform distribution scaled by `scale`.
    """
    n = a.shape[0]
    i, j = random.sample(range(n), 2)
    delta = (random.random() - 0.5) * scale
    # make sure we stay non‑negative
    if a[i] + delta < 0.0:
        delta = -a[i]
    if a[j] - delta < 0.0:
        delta = a[j]          # move the opposite direction instead
        if a[i] + delta < 0.0:
            return a          # give up this move
    a_new = a.copy()
    a_new[i] += delta
    a_new[j] -= delta
    # the sum is unchanged, only a projection is required if rounding errors appear
    a_new = _project_to_simplex(a_new)
    return a_new


def propose_candidate(seed: int = 0, budget_s: int = 980, **kwargs):
    """
    Search for a sequence of non‑negative heights that gives a low value
    of `evaluate_sequence`.  The routine runs for up to `budget_s` seconds
    (a few seconds are kept for the final return) and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5.0  # a safety margin for the return

    # lengths that we will try – a mix of very short and moderately long vectors
    lengths = [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20, 30, 40, 50, 80, 120]

    best_vec = None
    best_val = float('inf')

    # main outer loop – pick a length, initialise a vector and optimise it
    while time.time() < deadline:
        n = random.choice(lengths)

        # initialise with a Dirichlet draw (all components positive, sum = 1)
        a = np.random.dirichlet(alpha=np.ones(n))
        cur_val = _objective(a)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_vec = a.copy()
            print(f"New global best (len={n}): {best_val:.6f}")

        # SA/gradient parameters
        step_size = 0.5          # initial gradient step
        temp = 1.0               # temperature for accepting uphill moves
        max_inner = 3000         # iteration budget per restart

        for it in range(max_inner):
            if time.time() >= deadline:
                break

            # decide whether to do a gradient move or a random transfer
            if random.random() < 0.6:
                # sub‑gradient descent on the current maximum entry
                a_new = _gradient_step(a, step_size)
            else:
                # random mass transfer (helps to escape plateaus)
                a_new = _random_transfer(a, scale=0.1)

            new_val = _objective(a_new)

            # accept if better, otherwise accept with a Metropolis probability
            if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
                a = a_new
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_vec = a.copy()
                    print(f"New global best (len={n}): {best_val:.6f}")

            # geometric cooling of step size and temperature
            step_size *= 0.9995
            temp *= 0.9997

    # -----------------------------------------------------------------
    #  Post‑processing – ensure a valid output format
    # -----------------------------------------------------------------
    if best_vec is None:
        # fallback: a simple 2‑element distribution (still better than uniform)
        best_vec = np.array([1.0 / 3.0, 2.0 / 3.0])

    # make sure we respect the `evaluate_sequence` sanitisation:
    # - all entries are non‑negative
    # - sum is well above 0.01 (our vectors sum to 1)
    # - values are far below the 1000‑clip ceiling
    result = [float(x) for x in best_vec.tolist()]
    print(f"Final best value: {best_val:.6f}  (length {len(result)})")
    return result
