def project_simplex(v):
    """Projects vector v onto the probability simplex (nonnegative entries summing to 1)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def compute_score(a):
    n = len(a)
    # use FFT
    nfft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

def gradient_descent(seq, max_iter=200):
    a = seq.copy()
    n = len(a)
    eta = 0.1
    for it in range(max_iter):
        conv = np.convolve(a, a)
        maxb = conv.max()
        ks = np.where(conv == maxb)[0]
        k = ks[0]  # take first
        # gradient
        grad = np.zeros_like(a)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        idx = np.arange(i_min, i_max+1)
        counterpart = k - idx
        grad[idx] = 2.0 * a[counterpart]
        # step
        a = a - eta * grad
        # projection onto simplex
        a = project_simplex(a)
        # reduce step size
        eta *= 0.99
    return a

"""
Propose a sequence of non‑negative heights that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm performs many independent restarts.  For each restart we

* sample a random Dirichlet vector (the concentration can be tuned)
* improve it with a projected sub‑gradient descent that reduces the
  currently maximal convolution entry.
* keep the best sequence seen so far.

Both the random sampling and the sub‑gradient steps are cheap – the
convolution is computed with numpy’s FFT (O(n log n)) and the projection
onto the simplex guarantees that the sum stays constant (≈1) and that all
entries stay non‑negative.

The whole procedure runs until the allotted time (default 1000 s) is
nearly exhausted and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Small helper – return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_convolution(a: np.ndarray) -> float:
    """
    Return the maximal value of a * a (discrete linear convolution).
    Uses an FFT of length next power of two for speed.
    """
    n = a.shape[0]
    L = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Compute the evaluation function (without the security checks).
    The sequence is assumed to be non‑negative and its sum > 0.01.
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { w ≥ 0, sum(w) = 1 }.
    Standard O(n log n) algorithm (sorting).
    """
    # make a copy to avoid mutating the input
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return the uniform simplex point
        return np.full_like(v, 1.0 / n)

    rho = ind[cond].max()
    theta = cssv[cond][ -1] / rho   # cssv[rho-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step that tries to reduce the
    currently maximal convolution entry.
    - a : current probability vector (sum = 1).
    - eta : step size.
    Returns the updated vector (projected onto the simplex).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()

    # indices where the maximum is attained (there may be several)
    ks = np.where(conv == max_val)[0]
    k = ks[np.random.randint(ks.size)]

    # gradient of the active term g_k(a) = sum_i a_i a_{k-i}
    # (ordered pairs, hence factor 2)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]

    # take a step and project back onto the simplex
    a_new = a - eta * grad
    a_new = _project_simplex(a_new)
    return a_new


def _local_optimize(x0: np.ndarray,
                    max_iter: int = 300,
                    eta_start: float = 0.5,
                    eta_decay: float = 0.995) -> np.ndarray:
    """
    Refine a candidate sequence using projected sub‑gradient descent.
    Returns the improved vector.
    """
    a = x0.copy()
    eta = eta_start
    for _ in range(max_iter):
        a = _subgradient_step(a, eta)
        eta *= eta_decay
    return a
