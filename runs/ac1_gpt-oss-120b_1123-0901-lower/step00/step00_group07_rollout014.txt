def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2  # small margin
    best_seq = None
    best_score = float('inf')
    # parameter sets list
    ns = [256, 512, 1024, 2048, 3072]
    # maybe also random lengths.
    while time.time() < deadline:
        n = np.random.choice(ns)
        # run gradient descent with random init
        a = np.random.rand(n)
        target_sum = np.sqrt(2 * n)
        a = a / a.sum() * target_sum
        # adam states
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        # inner loop steps limited by time and max_iter
        max_iter = 2000
        for iter_idx in range(1, max_iter+1):
            if time.time() >= deadline:
                break
            # evaluate eval
            conv = np.convolve(a, a)
            max_b = conv.max()
            if max_b < best_score:
                best_score = max_b
                best_seq = a.copy()
                print(f"New best score {best_score:.6f} at n={n}, iter={iter_idx}")
            # compute smooth weights
            beta = 0.02 + (iter_idx / max_iter) * 4.0  # range 0.02 to 4.02
            # compute stable exponentials
            # subtract max for numeric stability
            shift = max_b
            exp_vals = np.exp(beta * (conv - shift))
            w = exp_vals / exp_vals.sum()
            # gradient via convolution
            grad_full = np.convolve(a, w)  # length n+2n-2 = 3n-2
            grad = 2 * grad_full[:n]
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** iter_idx)
            v_hat = v / (1 - beta2 ** iter_idx)
            lr = 0.05 / (np.sqrt(v_hat) + epsilon)  # base lr
            a = a - lr * m_hat
            # clip negatives
            a = np.maximum(a, 0.0)
            # re-normalize sum
            s = a.sum()
            if s == 0:
                # reinitialize
                a = np.random.rand(n)
                a = a / a.sum() * target_sum
                m = np.zeros_like(a)
                v = np.zeros_like(a)
                continue
            a = a * (target_sum / s)
            # clip high values for safety
            if a.max() > 1000:
                a = np.minimum(a, 1000.0)
                # re-normalize again
                a = a * (target_sum / a.sum())
        # maybe after finishing inner loop, optionally do random perturbation
        # optional: small random jitter to escape local minima
    # final best_seq must be list, ensure non-negative truncated values.
    if best_seq is None:
        # fallback: constant sequence
        n = 100  # small default
        avg = np.sqrt(2 * n) / n
        best_seq = [avg] * n
    # convert to python list, ensure floats
    return [float(x) for x in best_seq]

"""
Propose a step‑function (non‑negative height sequence) that minimizes the
evaluation function defined in the challenge.

The evaluation is scale‑invariant, because multiplying all heights by a constant
c multiplies both the numerator (max convolution) and the denominator
(sum²) by c².  Therefore we can fix the L¹‑norm of the sequence to
`sum = sqrt(2·n)`.  With that normalisation the score returned by
`evaluate_sequence` is exactly the maximal entry of the self‑convolution
`a*a`.

The search combines two complementary ideas:

1. **Smooth‑max gradient descent** – we minimise a differentiable
   approximation of the maximal convolution entry:
   ```
       f(a) = Σ_k w_k · (a*a)_k   with   w_k = softmax(β·(a*a)_k)
   ```
   The gradient of `f` w.r.t. `a` can be written in closed form as
   `∇f = 2·(a * w)` where `*` is a linear (non‑reversed) convolution.
   A simple step of gradient descent (with a small learning rate) followed by
   renormalisation of the L¹‑norm pushes mass away from positions that cause
   high convolution values, flattening the autocorrelation.

2. **Occasional LP move** – the provided linear‑programming routine
   (`get_good_direction_to_move_into`) finds a direction that can increase the
   total sum while keeping the maximal convolution bounded.  Mixing a tiny
   fraction of that direction into the current sequence sometimes yields
   a larger improvement than a pure gradient step.

The algorithm runs until the allotted time budget (default 1000 s) expires,
keeping the best sequence seen so far.  Random restarts are performed if
no improvement has been observed for a while, helping to escape possible
local minima.

All heights are kept non‑negative, clipped to the allowed range
[0, 1000] and finally returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax_weights(conv_vals: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights w_k = exp(β·(c_k‑max))/Σ exp(...)."""
    max_val = conv_vals.max()
    # subtraction for numerical stability
    exp_vals = np.exp(beta * (conv_vals - max_val))
    w = exp_vals / exp_vals.sum()
    return w


def solve_convolution_lp(f_seq: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise Σ b_j
        s.t.   (f * b)_k ≤ rhs  (for all k)
               b_j ≥ 0
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)            # maximise Σ b_j  → minimise −Σ b_j
    a_ub_rows = []
    b_ub_vals = []

    # convolution constraints: for each possible sum index k (0 … 2n‑2)
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub_rows.append(row)
        b_ub_vals.append(rhs)

    # non‑negativity constraints  –b ≤ 0  (i.e. b ≥ 0)
    a_ub_rows.append(-np.eye(n))
    b_ub_vals.append(np.zeros(n))

    a_ub = np.vstack(a_ub_rows)
    b_ub = np.array(b_ub_vals)

    res = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        bounds=(0, None),          # explicit non‑negativity as a safety net
        method="highs",            # robust default
        options={"time_limit": 5.0, "disp": False},
    )
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq: list[float]) -> list[float] | None:
    """
    LP‑based improvement direction.
    The current sequence `seq` is first normalised to have sum sqrt(2·n);
    then a new sequence `g` with a larger sum but respecting the same
    maximal convolution bound is computed.  A tiny convex combination
    (t ≈ 0.01) of the old and the new sequence yields a feasible improvement.
    """
    n = len(seq)
    seq = np.array(seq, dtype=float)
    sum_seq = seq.sum()
    if sum_seq == 0:
        return None
    # Normalise to the standard L¹‑norm used by the evaluation function
    seq_norm = seq * (np.sqrt(2 * n) / sum_seq)

    rhs = np.max(np.convolve(seq_norm, seq_norm))
    g = solve_convolution_lp(seq_norm, rhs)
    if g is None:
        return None

    sum_g = g.sum()
    if sum_g == 0:
        return None
    # Renormalise g to the same L¹‑norm
    g_norm = g * (np.sqrt(2 * n) / sum_g)

    # Small convex step towards g (t ≈ 0.01)
    t = 0.01
    new_seq = (1 - t) * seq_norm + t * g_norm
    # Clip to the allowed range before returning
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    return new_seq.tolist()


def max_self_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)_k for a non‑negative sequence."""
    return np.max(np.convolve(seq, seq))

    except Exception:
        init_seq = None

    if init_seq is not None and init_seq.size > 0:
        n = len(init_seq)
        target_sum = np.sqrt(2.0 * n)
        s = init_seq.sum()
        if s > 0:
            init_seq = init_seq * (target_sum / s)
        else:
            init_seq = np.random.rand(n)
            init_seq = init_seq / init_seq.sum() * target_sum
    else:
        # fall back to a random start; length is chosen from a modest range
        n = np.random.randint(800, 1500)
        init_seq = np.random.rand(n)
        init_seq = init_seq / init_seq.sum() * np.sqrt(2.0 * n)

    # best solution seen so far
    best_seq = init_seq.copy()
    best_score = max_self_convolution(best_seq)

    # ------------------------------------------------------------------
    # Gradient‑descent parameters
    # ------------------------------------------------------------------
    step_size = 0.02          # base learning rate
    beta_start = 0.3          # start of the soft‑max temperature
    beta_end = 6.0            # end of the soft‑max temperature
    max_iterations = 200_000  # upper bound; actual loop stops by time
    last_improve_iter = 0

    a = init_seq.copy()
    target_sum = np.sqrt(2.0 * n)

    iteration = 0
    while time.time() < deadline and iteration < max_iterations:
        iteration += 1

        # ------------------------------------------------------------------
        # Occasionally try a large LP‑based move (probability ≈2 %)
        # ------------------------------------------------------------------
        if np.random.rand() < 0.02:
            candidate = get_good_direction_to_move_into(a.tolist())
            if candidate is not None:
                a = np.array(candidate, dtype=float)
                # Re‑normalise after the LP step (tiny numerical drift may appear)
                a = a * (target_sum / a.sum())

        else:
            # --------------------------------------------------------------
            # Gradient‑descent step on the smooth maximal convolution
            # --------------------------------------------------------------
            beta = beta_start + (beta_end - beta_start) * (iteration / max_iterations)

            conv = np.convolve(a, a)
            w = softmax_weights(conv, beta)                # shape (2n‑1,)

            # gradient = 2 * (a * w)   (linear convolution, not reversed)
            grad_full = np.convolve(a, w)
            grad = 2.0 * grad_full[:n]

            # simple gradient step with a fixed learning rate
            a = a - step_size * grad

            # enforce non‑negativity
            a = np.maximum(a, 0.0)

            # re‑normalise to keep the sum constant (scale‑invariance of the score)
            s = a.sum()
            if s == 0.0:
                # rare degenerate case – restart from scratch
                a = np.random.rand(n)
                a = a / a.sum() * target_sum
                continue
            a = a * (target_sum / s)

            # safety clipping (the evaluation routine also clips)
            if a.max() > 1000.0:
                a = np.minimum(a, 1000.0)
                a = a * (target_sum / a.sum())

        # ------------------------------------------------------------------
        # Evaluate the current candidate (fast local evaluation)
        # ------------------------------------------------------------------
        cur_score = max_self_convolution(a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = iteration
            print(
                f"[{time.time():.2f}] iter {iteration:6d}  new best max={best_score:.6f} "
                f"(n={n})"
            )

        # ------------------------------------------------------------------
        # Random restart if stagnation lasts too long
        # ------------------------------------------------------------------
        if iteration - last_improve_iter > 5000:
            a = np.random.rand(n)
            a = a / a.sum() * target_sum
            last_improve_iter = iteration

    # Final sanity check – convert to plain Python floats and enforce limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
