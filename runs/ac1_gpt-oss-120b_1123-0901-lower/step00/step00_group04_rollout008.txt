"""
Propose a candidate step‑function sequence.

The algorithm performs many short runs of a simple projected
exponentiated‑gradient optimisation on the simplex.
For each run we:

1. Pick a random length n (200‑800) and initialise a random
   probability vector a (∑a_i = 1).
2. Iteratively minimise a smooth surrogate of   max_k (a*a)_k .
   The surrogate is the soft‑max of the convolution values,
   controlled by a temperature λ that slowly rises.
   The gradient of the surrogate is obtained via one extra
   convolution (O(n²) with NumPy’s fast implementation).
3. Apply an exponentiated‑gradient update a_i ← a_i·exp(‑η·∂/∂a_i)
   and renormalise to the simplex, keeping all entries ≥0.
4. Every few iterations we evaluate the true objective
   `evaluate_sequence`.  Whenever a better score is seen we
   keep the corresponding vector.
5. After the gradient phase we try a few cheap random swaps
   (move a small ε from one position to another) as a tiny
   hill‑climbing step.

The whole procedure runs for the allowed time (default 1000 s)
and returns the best sequence found.  The method improves on the
simple uniform construction (score 2.0) by learning an
asymmetric distribution that reduces the maximal convolution
value, aiming for scores below the target 1.503.
"""

import time
import numpy as np


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  Returns the best sequence found before the
    time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).
    kwargs : dict
        Unused additional arguments (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    # keep a small safety margin for the outer environment
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Helper: numerically stable softmax
    # ------------------------------------------------------------------
    def _softmax(x: np.ndarray) -> np.ndarray:
        m = np.max(x)
        e = np.exp(x - m)
        return e / e.sum()

    best_seq = None          # best array (numpy)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑up: evaluate a few uniform candidates of moderate size.
    # ------------------------------------------------------------------
    for n0 in (200, 300, 400):
        a0 = np.full(n0, 1.0 / n0)
        try:
            score0 = evaluate_sequence(a0.tolist())
        except Exception:
            continue
        if score0 < best_score:
            best_score = score0
            best_seq = a0.copy()
            print(f"[init] uniform n={n0}  score={score0:.6f}")

    # ------------------------------------------------------------------
    # Main search loop – many independent restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Choose a random length and start from a random simplex point.
        # ------------------------------------------------------------------
        n = np.random.randint(200, 800)          # size of the sequence
        a = np.random.rand(n)
        a = a / a.sum()                           # project onto the simplex

        # ------------------------------------------------------------------
        # Gradient‑descent phase (soft‑max surrogate of max convolution).
        # ------------------------------------------------------------------
        it = 0
        max_iters = 150                           # per restart
        while time.time() < deadline and it < max_iters:
            # convolution a*a (linear, not circular)
            conv = np.convolve(a, a)               # length 2n‑1

            # slowly increase λ so that the surrogate focuses more on the true max
            lam = 5.0 + 0.05 * it

            # soft‑max weights (sum 1)
            w = _softmax(lam * conv)

            # gradient of the surrogate:  ∂/∂a_i  Σ_k w_k (a*a)_k
            # = 2 * Σ_j a_j w_{j+i}  (indices out of range are ignored)
            w_rev = w[::-1]                       # reversed weights
            rev = np.convolve(a, w_rev)           # length 3n‑2
            idx = 2 * n - 2                       # index to pick from rev
            grad = 2.0 * rev[idx - np.arange(n)]  # vectorised gradient

            # exponentiated‑gradient update (always stays non‑negative)
            lr = 0.2 / (1.0 + 0.01 * it)          # decreasing learning rate
            a = a * np.exp(-lr * grad)

            # re‑normalise onto the probability simplex
            s = a.sum()
            if s <= 0.0 or np.isnan(s):
                # degenerated – re‑initialise randomly
                a = np.random.rand(n)
                a = a / a.sum()
            else:
                a = a / s

            # Periodically evaluate the true objective.
            if it % 5 == 0:
                try:
                    cur_score = evaluate_sequence(a.tolist())
                except Exception:
                    cur_score = float("inf")
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[improve] n={n} it={it}  score={cur_score:.6f}")
                    # early exit if we already beat the target
                    if best_score <= 1.503:
                        print("[target] reached required bound.")
                        return best_seq.tolist()
            it += 1

        # ------------------------------------------------------------------
        # Tiny hill‑climbing using random ε‑swaps (helps escape tiny plateaus)
        # ------------------------------------------------------------------
        for _ in range(30):
            if time.time() >= deadline:
                break
            i, j = np.random.choice(n, 2, replace=False)
            eps = np.random.rand() * 0.02          # move up to 2 % of total mass
            if a[i] <= eps:
                continue
            a_candidate = a.copy()
            a_candidate[i] -= eps
            a_candidate[j] += eps
            a_candidate = a_candidate / a_candidate.sum()
            try:
                cand_score = evaluate_sequence(a_candidate.tolist())
            except Exception:
                continue
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                a = a_candidate                     # keep the improved point
                print(f"[swap]   n={n}  score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing was ever found (should not happen) fall back to a simple uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    return best_seq.tolist()
