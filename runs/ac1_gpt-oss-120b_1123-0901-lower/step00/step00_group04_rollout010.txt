def compute_max_conv(a):
    # compute full convolution using FFT
    size = len(a)
    n = size
    # pad to next power-of-two maybe for speed
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    fb = fa * fa
    conv = np.fft.irfft(fb, n=m)[:2*n-1]
    return conv.max()

def improve_sequence(seq, rng, deadline):
    n = len(seq)
    # Precompute constant factor: 2*n
    constant = 2 * n
    # compute conv via FFT for speed
    # define function to compute max conv
    def max_conv(a):
        m = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(a, n=m)
        fb = fa * fa
        conv = np.fft.irfft(fb, n=m)[:2*n-1]
        return conv.max()
    # initial best
    max_b = max_conv(seq)
    best_score = constant * max_b
    best_seq = seq.copy()
    # parameters
    lr = 0.5
    decay = 0.99995
    # iterations
    while time.time() < deadline:
        # compute full conv
        m = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(seq, n=m)
        fb = fa * fa
        conv = np.fft.irfft(fb, n=m)[:2*n-1]
        max_val = conv.max()
        max_idx = np.argmax(conv)
        # compute gradient as reversed a (masked)
        grad = np.zeros_like(seq)
        # for i in-range, if j=k-i within 0..n-1: grad[i] = 2*seq[j]
        # this is just 2*seq_rev shifted by k
        # Equivalent: grad = 2 * np.pad(seq, (0, n-1), 'constant')
        # Actually we can implement loop.
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                grad[i] = 2 * seq[j]
        # exponentiated gradient step
        seq = seq * np.exp(-lr * grad)
        # renormalize
        s = seq.sum()
        if s == 0:
            # reinitialize
            seq = rng.random(n)
            s = seq.sum()
        seq = seq / s
        # evaluate new max_b
        max_b = max_conv(seq)
        score = constant * max_b
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # optionally accept occasionally if not improved? Keep track
        # decay lr
        lr *= decay
        # break if lr too small
        if lr < 1e-8:
            break
    return best_seq, best_score

"""
Propose a candidate sequence that reduces the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

which is scale‑invariant.  The algorithm works on a probability vector
(a_i ≥ 0, sum = 1) and directly minimises the L∞‑norm of the
self‑convolution.

Main ideas
-----------
* Treat a as a distribution of a die.  The convolution b = a * a is the
  distribution of the sum of two independent draws.
* Use a smooth approximation of the max – a soft‑max (log‑sum‑exp) – so
  that we get a differentiable objective.
* Compute the gradient efficiently with FFTs:
      ∇_a  Σ_k w_k  (a * a)_k = 2 * (reverse(a) ⊛ w)
  where w are the soft‑max weights.
* Perform projected gradient descent on the simplex (multiplicative
  update → always stays non‑negative) and re‑normalise after each step.
* Several random restarts are performed until the time budget expires.
  The best sequence found over all restarts is returned.

The method is completely self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for the final check, and runs well within
the 1000 s limit.
"""

import numpy as np
import time

# --------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of a with itself using FFT.
    Returns a real array of length 2*len(a)-1.
    """
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=m)
    fb = fa * fa
    conv = np.fft.irfft(fb, n=m)[: 2 * n - 1]
    return conv


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute softmax weights w_k = exp(beta * b_k) / Σ exp(beta * b_k)
    in a numerically safe way.
    """
    # shift by the maximum to avoid overflow
    shift = beta * (b - b.max())
    # clip extreme negative values (exp(-800) is already ~0)
    shift = np.clip(shift, -800, None)
    exp_shift = np.exp(shift)
    w = exp_shift / exp_shift.sum()
    return w


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective w.r.t. a.
    Implements  grad = 2 * (reverse(a) ⊛ w)[n-1 : n-1+n]
    using FFT for speed.
    """
    n = a.shape[0]
    a_rev = a[::-1]  # reversed a
    # convolution length = n + (2n-1) - 1 = 3n - 2
    size = _next_pow2(3 * n - 2)
    fa_rev = np.fft.rfft(a_rev, n=size)
    fw = np.fft.rfft(w, n=size)
    conv_full = np.fft.irfft(fa_rev * fw, n=size)
    # slice the needed part (indices n-1 … n-1+n‑1)
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad


def _improve_distribution(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Run a projected (multiplicative) gradient descent on a single
    restart.  The sequence is kept normalised to sum = 1, which makes
    the objective simply ``2 * n * max(conv)``.
    Returns the best sequence seen in this run and its (raw) score.
    """
    n = a.shape[0]
    const = 2.0 * n

    # hyper‑parameters – they are deliberately conservative
    lr = 0.6          # learning rate for the multiplicative update
    beta = 1.0        # initial soft‑max temperature
    beta_mul = 1.001  # increase beta slowly (approach true max)
    lr_mul = 0.9995   # slowly decay the step size

    best_a = a.copy()
    b = _conv_full(a)
    best_score = const * b.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        # 1️⃣ compute convolution and soft‑max weights
        b = _conv_full(a)
        w = _softmax_weights(b, beta)

        # 2️⃣ gradient of the smoothed max
        grad = _gradient_softmax(a, w)

        # 3️⃣ multiplicative (exponentiated) update, then re‑normalise
        #    a_new = a * exp(-lr * grad)
        a = a * np.exp(-lr * grad)
        s = a.sum()
        if s == 0.0:         # occasional numerical collapse – re‑seed
            a = np.maximum(np.random.random(n), 1e-12)
            s = a.sum()
        a = a / s

        # 4️⃣ evaluate the true objective (using the exact max)
        b = _conv_full(a)
        score = const * b.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # small progress report
            print(
                f"[{time.time():.2f}] iter={it} n={n} score={best_score:.6f}"
            )

        # 5️⃣ schedule parameters
        lr *= lr_mul
        beta *= beta_mul
        it += 1

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point required by the framework
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` function.
    The search consists of several random restarts, each performing a
    fast projected gradient descent on the simplex (see the module
    docstring for details).  The best sequence found before the time
    budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. optionally start from the known SOTA sequence (if it exists)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be a very long list – truncate for speed
        base_seq = np.array(height_sequence_1, dtype=float)
        if base_seq.ndim == 0 or base_seq.size == 0:
            raise NameError
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # 2. Main loop: keep launching restarts until time runs out
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # decide length; keep it modest for speed (800–1500 works well)
        n = rng.integers(800, 1500)

        if restart == 0 and base_seq is not None and rng.random() < 0.5:
            # use a clipped slice of the SOTA vector (preserve its scale)
            if base_seq.size >= n:
                a0 = base_seq[:n].copy()
            else:
                # pad with small random numbers if needed
                pad = rng.random(n - base_seq.size)
                a0 = np.concatenate([base_seq, pad])
            a0 = np.maximum(a0, 0.0)
        else:
            # random Dirichlet‑like start (positive, sum = 1)
            a0 = rng.random(n)

        # normalise to a probability vector
        a0 = a0 / a0.sum()

        # run the local optimiser on this restart
        seq, score = _improve_distribution(a0, deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(
                f"*** New global best (restart {restart}) "
                f"score={best_score:.6f}, n={n}"
            )

        restart += 1

    # ------------------------------------------------------------------
    # 3. Fallback: if nothing succeeded, return a simple uniform seq
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)

    # Convert to plain Python list of floats as required
    return best_seq.tolist()
