def generate_sequence(n, p):
    seq = [(i+1)**p for i in range(n)]
    # scale to max <=1000
    max_val = max(seq)
    if max_val > 1000:
        factor = 1000.0 / max_val
        seq = [x * factor for x in seq]
    return seq

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Candidate lengths: maybe 500, 800, 1000, 1108? But we need final one.
    lengths = [500, 800, 1000, 1108]   # ensure within max 1000 values.
    p_values = np.linspace(0.2, 2.5, 24)  # e.g., 24 points.

            seq = generate_sequence(n, p)
            score = evaluate_sequence(seq)

def _make_sequence(n: int, p: float) -> list[float]:
    """Return a power‑law sequence a[i] = (i+1)^p of length *n*.

    The sequence is scaled so that its maximum does not exceed 1000.
    This scaling does **not** change the evaluation result because the
