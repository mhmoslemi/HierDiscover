def proj_onto_simplex(v, S):
    # Implementation based on sorting
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

    # Setup
    import...
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1

    # Determine length n
    try:
        init_seq = list(height_sequence_1)
    except NameError:
        n = kwargs.get('length', 6850)
        init_seq = [np.random.rand() for _ in range(n)]

    a = np.array(init_seq, dtype=np.float64)
    a = a * target_sum / a.sum()

    a = np.clip(a, 0, None)

    step_fraction = 0.1  # fraction of average weight to move
    temperature = best_score  # initial temperature for annealing
    decay = 0.9995

    iteration = 0
    while time.time() < deadline:

        b = np.convolve(a, a)
        t = b.max()

        # Compute gradient for k_max
        grad = np.zeros_like(a)
        i_min = max(0, k_max - (n - 1))

        idx = np.arange(i_min, i_max_idx + 1)

        grad[idx] = a[j] + a[idx]

        i_high = np.argmax(grad)  # location where moving mass decreases max most (since derivative large)

        a_candidate[i_high] -= delta
        a_candidate[i_low] += delta

"""
Propose a candidate sequence that minimises the evaluation function

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The function is homogeneous: scaling the whole vector does not change the
score.  Hence we normalise the vector to have
    sum(a) = sqrt(2*n)
so that the score equals simply  max(conv(a, a)).

The algorithm is a fast projected‑gradient / simulated‑annealing search.

* Initialise a with a (slightly noisy) uniform distribution of length *n*.
* Compute the full self‑convolution by FFT once; keep the convolution array
  `b` up‑to‑date after each accepted move using a cheap O(n) rank‑one update.
* At each iteration:
    – find the index `k_max` where `b` is maximal,
    – compute the sub‑gradient g_i = a_i + a_{k_max-i},
    – pick a location `i_high` with the largest gradient (where moving mass
      away reduces the peak the most) and `i_low` with the smallest gradient,
    – move a small amount `delta` from `i_high` to `i_low`,
    – update the convolution `b` analytically (no FFT),
    – accept the move if it improves the peak, otherwise accept it
      probabilistically with an annealing temperature that decays over time.
* Occasionally inject a tiny random perturbation to escape stagnation.
* Keep the best sequence seen so far and return it when the time budget
  expires.

All operations are O(n) per iteration (n ≈ 7 000), allowing many thousands
of iterations within the 1000 s budget.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------------------- #

def _fft_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector using an FFT of length power‑of‑2."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two ≥ m
    L = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[:m]


def _gradient_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution term with respect to a.
    For the active shift k the derivative of b_k = Σ_i a_i a_{k-i} is
        g_i = a_i + a_{k-i}  (a_{k-i}=0 when out of range)
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - idx
    g[idx] = a[idx] + a[j]
    return g


def _shift_array(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Return an array s of length 2*n-1 such that
        s[k] = a_{k-idx}  (zero if k-idx outside [0,n-1]).
    This is a “shifted view” of a needed for the rank‑one update of the
    convolution after moving a mass from one coordinate to another.
    """
    n = a.shape[0]
    m = 2 * n - 1
    s = np.zeros(m, dtype=np.float64)
    s[idx:idx + n] = a
    return s

    # initial convolution and its maximum

    best_score = cur_max                     # for the normalised vector score == max conv

    print(f"[init] n={n}, initial_score={best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Hyper‑parameters (tuned empirically)
    # ------------------------------------------------------------------- #
    avg_weight = target_sum / n
    max_delta = 0.5 * avg_weight             # never move more than half an average slot
    temperature = 0.2 * cur_max              # initial annealing temperature
    temp_decay = 0.9999
    iter_no_improve = 0
    no_improve_limit = 4000                  # after this many iterations we add noise

    iteration = 0
    while time.time() < deadline:
        # ---- current active shift -------------------------------------------------
        k_max = int(np.argmax(b))
        # ---- gradient of the active shift -----------------------------------------
        grad = _gradient_for_k(a, k_max)

        # pick source (high gradient) and destination (low gradient)
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # ensure we can move something
        if a[i_high] <= 1e-12 or i_high == i_low:
            # fallback to a random tiny move
            i_high = np.random.randint(n)
            i_low = np.random.randint(n)
            while i_low == i_high:
                i_low = np.random.randint(n)
            delta = min(a[i_high], max_delta) * np.random.rand()
        else:
            # move a fraction of the source weight (capped by max_delta)
            delta = min(a[i_high] * 0.5, max_delta)

        # ---- rank‑one update of the convolution -----------------------------------
        # shift arrays for the two indices
        shift_add = _shift_array(a, i_low)
        shift_rem = _shift_array(a, i_high)

        const_term = delta * (a[i_low] - a[i_high])          # same for every k
        # new convolution without recomputing FFT
        b_candidate = b + delta * (shift_add - shift_rem) + const_term

        # add the tiny δ² corrections on the diagonal terms
        two_low = 2 * i_low
        two_high = 2 * i_high
        if two_low < b_candidate.shape[0]:
            b_candidate[two_low] += delta * delta
        if two_high < b_candidate.shape[0]:
            b_candidate[two_high] += delta * delta

        new_max = b_candidate.max()

        # ---- acceptance -----------------------------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            # probabilistic uphill move (simulated annealing)
            if temperature > 1e-12:
                prob = math.exp(-(new_max - cur_max) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # apply the move permanently
            a[i_high] -= delta
            a[i_low] += delta
            # numerical safety: clip negativity caused by round‑off
            a = np.clip(a, 0.0, None)
            # renormalise to keep the exact sum (tiny drift may appear)
            a_sum = a.sum()
            if abs(a_sum - target_sum) > 1e-12:
                a *= target_sum / a_sum

            b = b_candidate
            cur_max = new_max
            iter_no_improve = 0

            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{iteration:06d}] new best score = {best_score:.6f}")

        else:
            iter_no_improve += 1

        # ---- temperature schedule -------------------------------------------------
        temperature *= temp_decay

        # ---- occasional random perturbation to escape stagnation ------------------
        if iter_no_improve >= no_improve_limit:
            # add a tiny Gaussian jitter, then re‑normalise
            jitter = np.random.randn(n) * 0.001 * avg_weight
            a += jitter
            a = np.clip(a, 0.0, None)
            a *= target_sum / a.sum()
            b = _fft_convolution(a)           # recompute convolution after a shake
            cur_max = b.max()
            iter_no_improve = 0
            temperature = 0.2 * cur_max        # reset temperature a bit
            print(f"[{iteration:06d}] jitter applied, score={cur_max:.6f}")

        iteration += 1

    # ------------------------------------------------------------------- #
    # final reporting
    # ------------------------------------------------------------------- #
    print(f"Finished after {iteration} iterations. Best score found: {best_score:.6f}")
    # convert to plain Python list for the competition harness
    return best_a.tolist()
