    """
    Generate a deterministic non‑negative step‑function sequence with a
    polynomially increasing profile (a_i = (i+1)^p).  After scaling the
    maximum entry to 1000 (the clipping limit used by `evaluate_sequence`)
    the sequence achieves an evaluation score of 4/3 * (p+2)/(p+1) ?
    ...
    """
    import math
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000  # fallback
    p = 3  # exponent
    # Build raw sequence
    raw = [(float(i+1) ** p) for i in range(n)]
    # Scale to keep values <= 1000
    max_val = max(raw)
    scale = 1000.0 / max_val if max_val > 0 else 0.0
    seq = [x * scale for x in raw]
    # Optional sanity check: evaluate score
    try:
        score = evaluate_sequence(seq)
        print(f"Proposed sequence length {n}, exponent {p}, score {score:.6f}")
    except Exception as e:
        print("Evaluation failed:", e)
    return seq
