def gradient_step(seq, eta, target_sum):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    mask = (conv == max_val).astype(float)
    # gradient = 2 * conv(mask, reversed(seq))
    rev_seq = seq[::-1]
    grad_full = np.convolve(mask, rev_seq)
    start = n-1
    grad = 2.0 * grad_full[start:start+n]
    new_seq = seq - eta * grad
    new_seq = np.maximum(new_seq, 0)
    if new_seq.sum() == 0:
        new_seq = np.full_like(seq, target_sum/len(seq))
    else:
        new_seq *= target_sum / new_seq.sum()
    return new_seq, max_val
```

    x = np.maximum(x, 0)
    total = x.sum()
    if total == 0:
        return np.full_like(x, target_sum / len(x))
    return x * (target_sum / total)

def solve_convolution_lp(f, rhs):
    n = len(f)
    c = -np.ones(n)
    # Build A_ub and b_ub
    rows = []
    b = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # compute contributions
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # for i in range(i_min, i_max+1):
        #   j = k - i
        #   row[j] = f[i]
        # Optimize loops
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        b.append(rhs)
    # nonnegativity constraints: -I x <= 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([b, np.zeros(n)])
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
    if res.success:
        # Clip small negatives due to tolerance
        x = np.maximum(res.x, 0)
        return x
    return None

    n = len(seq)

    max_val = conv.max()
    mask = (conv == max_val).astype(float)

"""
Search for a step‑function (non‑negative height list) that minimises
evaluate_sequence.

The algorithm mixes three complementary ideas:

1. **LP direction** – given the current sequence f,
   solve a linear program that finds a non‑negative sequence g with maximal
   ℓ₁‑mass while respecting the convolution bound
   (f * g)(k) ≤ M where M = max(f * f).  Mixing f and g reduces the
   peak of the self‑convolution.  A short line‑search over the mixing
   coefficient t selects the best convex combination.

2. **Sub‑gradient step** – the objective  max(f * f) is convex.
   At the positions where the convolution attains its maximum we build a
   (sub‑)gradient and take a step opposite to it, followed by a projection
   onto the non‑negative simplex (fixed total mass).  This helps to escape
   plateaus where the LP offers no improvement.

3. **Random perturbations / restarts** – if the search stagnates, a small
   Gaussian tweak or a complete restart from a fresh random point is applied.

All candidates are kept normalised to the same total sum
   S = sqrt(2·n)       (n = length of the sequence).
Because the evaluation function is homogeneous,
   2 n max(f * f) / S² = max(f * f),
so minimising the maximum convolution entry is equivalent to minimising
the original score.

The routine runs for at most `budget_s` seconds and returns the best
sequence found.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog


def _project_and_normalise(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative, then rescale so that the entries sum to target_sum."""
    x = np.maximum(x, 0.0)
    total = x.sum()
    if total == 0.0:
        # fallback: uniform distribution
        return np.full_like(x, target_sum / x.size)
    return x * (target_sum / total)


def _solve_convolution_lp(f: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve:
        maximise   Σ b_j
        subject to  (f * b)_k ≤ rhs   for all k = 0,…,2n‑2
                    b_j ≥ 0
    Returns the optimal b as a NumPy array, or None if the LP fails.
    """
    n = f.size
    # objective (maximise Σ b ⇒ minimise –Σ b)
    c = -np.ones(n)

    # build the convolution constraints: for each output index k we need
    # sum_{j} f_{k-j} * b_j ≤ rhs  (where the index k‑j is inside [0,n-1])
    rows = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs over indices of f that contribute to output k
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  –b ≤ 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])

    # solve with HiGHS (fast interior‑point); give a modest time limit
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": 5.0},
        )
    except Exception:  # pragma: no cover – safety net
        return None

    if not res.success:
        return None

    # tiny negative values may appear due to tolerance – clip them
    return np.maximum(res.x, 0.0)


def _gradient_step(seq: np.ndarray,
                   eta: float,
                   target_sum: float) -> tuple[np.ndarray, float]:
    """
    Perform one sub‑gradient descent step on max(conv(seq,seq)).
    Returns the new sequence and the current maximum convolution value.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    mask = (conv == max_val).astype(float)

    # sub‑gradient: 2 * (mask *rev* seq)  (cross‑correlation)
    rev_seq = seq[::-1]
    grad_full = np.convolve(mask, rev_seq)
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]

    new_seq = seq - eta * grad
    new_seq = _project_and_normalise(new_seq, target_sum)
    return new_seq, max_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function respects the global time budget
    (`budget_s`) and returns the best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=float)
        n = init.size
    else:
        n = np.random.randint(1500, 2500)
        init = np.random.rand(n).astype(float)

    target_sum = np.sqrt(2.0 * n)            # we keep the total mass fixed
    seq = _project_and_normalise(init, target_sum)

    best_seq = seq.copy()
    best_metric = np.max(np.convolve(best_seq, best_seq))
    print(f"[init] length={n}, metric={best_metric:.6f}")

    # ------------------------------------------------------------------
    # parameters for the iterative search
    # ------------------------------------------------------------------
    t_grid = np.linspace(0.02, 0.5, 25)      # mixing coefficients explored after LP
    no_improve = 0
    it = 0

    while time.time() < deadline:

                    best_local_seq = cand

            if best_local_seq is not None:

                    best_metric = best_local_metric
                    best_seq = seq.copy()
