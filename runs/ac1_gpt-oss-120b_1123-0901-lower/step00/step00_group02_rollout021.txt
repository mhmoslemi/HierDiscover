    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2100
    # Build linear increasing shape a_i ∝ i+1
    indices = np.arange(1, n+1)  # 1..n
    seq = indices.astype(float)
    seq = seq / seq.sum()
    best_seq = seq.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"Initial linear sequence length {n}, score {best_score}")
    curr_seq = np.copy(seq)
    step = 0.01  # initial step size relative to values
    while time.time() < deadline:
        # generate perturbation
        perturb = np.random.randn(n) * step
        # make zero-sum perturb (to keep sum roughly unchanged)
        perturb -= np.mean(perturb)
        cand = curr_seq + perturb
        # enforce nonnegative
        cand = np.maximum(cand, 0.0)
        # renormalize to sum=1 (scale invariance)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s
        score = evaluate_sequence(cand.tolist())
        # accept if improvement (lower)
        if score < best_score:
            best_score = score
            best_seq = cand.tolist()
            curr_seq = cand
            print(f"New best {best_score:.6f} at time {time.time():.2f}")
        # simple simulated annealing acceptance condition
        else:
            # sometimes accept worse to escape local minima
            # probability decays with time
            t = time.time()
            # simple temperature schedule
            temp = max(0.001, (deadline - t) / (deadline - time.time() + 1e-9))
            # compute acceptance probability
            if np.random.rand() < np.exp((best_score - score) / (temp * 0.1)):
                curr_seq = cand
        # gradually reduce step
        if np.random.rand() < 0.01:
            step *= 0.9
    print(f"Best score after search: {best_score}")

"""
propose_candidate
-----------------
Generate a step‑function (a list of non‑negative heights) that beats the
current benchmark (~2.0) and aims for the target of ~1.5.
The method is based on the following insight:

* For a sequence a with Σa = 1 the evaluation reduces to
      2·n·max_k (a * a)[k] .
* A linear increasing shape  a_i ∝ (i+1)  (after normalising the sum)
  yields a convolution whose maximum is
      max_k (a*a)[k] = (2/3)·(n+2) / (n·(n+1))   ≈  (2/3)/n .
  Consequently the score is
      2·n·max ≈ 4/3 ≈ 1.33,
  which already improves the target 1.503.

The function builds this linear shape, then performs a cheap
stochastic hill‑climbing/annealing phase that transfers a small amount
of mass between random indices while keeping the total mass = 1.
Only modifications that lower the score are kept (occasionally a
worse move is accepted to escape local minima).  The search stops
shortly before the time budget expires and returns the best sequence
found.

The algorithm uses only NumPy, time and the already‑imported
evaluate_sequence function.
"""

import time
import numpy as np

def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) gives a low
    value of evaluate_sequence.  The function respects the time budget
    (leaves a few seconds margin for the caller).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # spare a few seconds

    # -------------------------------------------------------------
    # 1) Choose the length.  Use the length of the provided reference
    #    sequence if it exists, otherwise fall back to 2100 (the
    #    length mentioned in the statement).
    # -------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore  # noqa: F821
    except Exception:
        n = 2100

    # -------------------------------------------------------------
    # 2) Build the simple linear increasing distribution:
    #        a_i = (i+1) / Σ_{j=1}^{n} j
    #    This already gives a score ≈ 4/3.
    # -------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=float)          # 1 … n
    seq = idx / idx.sum()                           # normalised to sum = 1
    best_seq = seq.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={n}, linear start score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3) Stochastic improvement loop.
    #    - perturb a few entries (zero‑mean Gaussian noise)
    #    - enforce non‑negativity and re‑normalise to sum = 1
    #    - accept if the score improves; occasionally accept a worse
    #      move with a simple temperature schedule.
    # -------------------------------------------------------------
    cur_seq = seq.copy()
    step = 0.02                     # typical relative magnitude of a perturbation
    it = 0
    while time.time() < deadline:
        it += 1

        # ---- random zero‑sum perturbation ---------------------------------
        perturb = np.random.randn(n) * step
        perturb -= perturb.mean()                # keep total mass roughly unchanged
        cand = cur_seq + perturb
        cand = np.maximum(cand, 0.0)             # non‑negative constraint

        total = cand.sum()
        if total < 1e-12:                        # degenerate, retry
            continue
        cand /= total                             # normalise (scale‑invariant)

        # ---- evaluate ------------------------------------------------------
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # ---- acceptance ----------------------------------------------------
        if score < best_score:
            best_score = score
            best_seq = cand.tolist()
            cur_seq = cand
            # slowly shrink the step size when we make progress
            step *= 0.99
            print(f"[{it:07d}] NEW BEST {best_score:.6f}  step={step:.5f}")
        else:
            # simple simulated‑annealing acceptance to escape plateaus
            # temperature decays linearly to a small floor
            remaining = deadline - time.time()
            frac = max(0.0, remaining / (budget_s - 5.0))
            temperature = 0.1 * (0.5 + 0.5 * frac)   # in (0.05,0.1)
            prob = np.exp((best_score - score) / temperature)
            if np.random.rand() < prob:
                cur_seq = cand

        # ---- occasional step‑size refresh ---------------------------------
        if it % 5000 == 0:
            step = max(step, 0.001)  # keep a minimum exploration radius

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_seq
