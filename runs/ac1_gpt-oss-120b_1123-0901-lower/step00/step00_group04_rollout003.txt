    # v is numpy array, may have negative values. Returns projection onto {x>=0, sum x=1}
    n = len(v)

"""
A fast projected‑gradient descent that tries to flatten the self‑convolution
of a non‑negative step function.

The idea is to minimise the deviation of the convolution
C = a * a from its average value.
For a vector a with sum S we have
    sum_k C_k = S²  and  avg(C)=S²/(2·n‑1).
If C is (almost) constant then the evaluation function
    2·n·max(C)/S² ≈ 2·n/(2·n‑1) → 1
and we obtain a bound < 2 (the uniform distribution gives 2).

The algorithm:
1. Pick a random length n (200‑4000) and a random positive vector a,
   normalise it to sum = 1.
2. Repeatedly:
      * compute C = a ⊛ a   (FFT convolution)
      * set target T = mean(C)
      * compute the gradient of ½·∑(C‑T)² w.r.t. a,
        which is 4·[(C‑T) ⊛ a_rev]  (a_rev is a reversed)
      * take a step a←a‑η·grad and project back onto the probability
        simplex (non‑negative, sum = 1).
   The step size η starts at 0.5 and is geometrically decayed.
3. Keep the best sequence (according to evaluate_sequence) seen in any
   restart.  The whole search stops a few seconds before the given budget.

The projection onto the simplex is exact and inexpensive.
FFT based convolutions make each iteration O(n·log n), allowing thousands
of iterations within the 1000 s time limit.

The returned sequence has only non‑negative numbers and satisfies the
required format for evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero, return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution using real FFT (fast for the sizes we use)
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the (linear) convolution of x and y using real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    # next power of two for efficient FFT
    n_fft = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n_fft)
    fy = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(fx * fy, n_fft)[:n]
    return conv


# ----------------------------------------------------------------------
