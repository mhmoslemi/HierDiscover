"""
propose_candidate
-----------------

A fast heuristic for the self‑convolution problem.

The evaluation function is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is invariant under scaling, so we may (and do) normalise the
sequence to have unit sum.  The quantity to minimise is therefore

    2 * n * max(conv(a, a))

with a_i ≥ 0 and Σ a_i = 1.

**Main ideas**

1.  Analytic family – a power‑law sequence  
    a_i ∝ (i + offset) ** (‑p)  
    (i = 0 … n‑1).  For p = 0.5 the continuous analogue gives a
    convolution that is almost flat; the corresponding asymptotic
    constant is π/2 ≈ 1.57, already a solid improvement over the
    trivial uniform construction (score = 2).

2.  Parameter sweep – we try several values of the length `n`,
    the exponent `p` (around the critical 0.5) and an additive offset.
    Every candidate is evaluated, and the best one is kept.

3.  Simple projected gradient descent – starting from the best
    candidate we repeatedly locate the index `k` where the convolution
    attains its maximum and move the sequence opposite to the gradient
    of that entry (`∂ conv_k / ∂ a_i = 2 * a_{k‑i}` for valid i).  After
    each step the vector is projected back onto the probability simplex
    (non‑negative, unit sum).  The step size `eta` is adapted
    automatically.  This cheap iteration often reduces the peak a bit
    further.

The whole routine respects the 1000‑second budget, uses only NumPy
(and the already available `evaluate_sequence`) and returns a plain
list of floats ready for the final scoring.  All values are clipped
to the interval [0, 1000] before returning so that the security checks
in `evaluate_sequence` never modify the result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas as required)
# ----------------------------------------------------------------------


def _score(seq):
    """Fast wrapper around the provided evaluate_sequence."""
    # The function evaluate_sequence is assumed to be already imported.
    # We call it directly – it performs all security checks.
    return evaluate_sequence(seq)


def _make_power_law(n, p, offset):
    """
    Return a normalised power‑law sequence:
        a_i ∝ (i + offset) ** (‑p)   for i = 0 … n‑1
    The result sums to 1.
    """
    idx = np.arange(n, dtype=float)
    seq = (idx + offset) ** (-p)
    total = seq.sum()
    if total == 0:
        seq = np.ones(n, dtype=float) / n
    else:
        seq = seq / total
    return seq


def _project_to_simplex(v):
    """Project a non‑negative vector onto the simplex Σ v_i = 1."""
    # Simple renormalisation works because we only ever set negatives to 0.
    s = v.sum()
    if s == 0:
        # fallback to the uniform distribution
        return np.ones_like(v) / len(v)
    return v / s


def _gradient_step(a, eta):
    """
    One gradient descent step that reduces the current maximal entry of
    the auto‑convolution.  The gradient for the index k_max is
        g_i = 2 * a_{k_max - i}   (if the index is inside the vector)
    The step is followed by a projection onto the probability simplex.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(conv.argmax())                # index of the current peak
    # build the gradient (zero where the reversed index would be out of range)
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    # i runs from i_start … i_end  and  j = k - i  lies in 0 … n‑1
    for i in range(i_start, i_end + 1):
        j = k - i
        grad[i] = 2.0 * a[j]               # ∂conv_k/∂a_i
    # take a step and re‑project
    a_new = a - eta * grad
    a_new[a_new < 0] = 0.0
    a_new = _project_to_simplex(a_new)
    return a_new, conv.max()


def _refine_by_descent(start_seq, deadline):
    """
    Small projected‑gradient descent loop.  The routine stops when time
    runs out or when the step size becomes tiny.
    """
    a = start_seq.copy()
    eta = 0.1       # initial step size
    best_local = _score(a.tolist())
    while time.time() < deadline:
        a_next, peak = _gradient_step(a, eta)
        new_score = _score(a_next.tolist())
        # accept if we improve; otherwise shrink the step size
        if new_score < best_local:
            a = a_next
            best_local = new_score
            # slowly increase the step size when we are making progress
            eta = min(eta * 1.02, 0.5)
        else:
            eta = max(eta * 0.9, 1e-5)
            # occasional random jitter to escape tiny local plateaus
            if np.random.rand() < 0.02:
                idx = np.random.randint(len(a))
                a[idx] = max(0.0, a[idx] + np.random.randn() * 0.01)
                a = _project_to_simplex(a)
                best_local = _score(a.tolist())
    return a, best_local

# The required entry point
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 5

    # 1) start from the best known construction if it exists
    # ------------------------------------------------------------------

    best_val = np.inf

        best_seq = seq0.tolist()

    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) explore a small family of power‑law sequences
    # ------------------------------------------------------------------
    n_options = [256, 512, 1024, 2048, 4096]
    p_options = [0.45, 0.48, 0.5, 0.52, 0.55]
    offset_options = [0.5, 1.0, 2.0]

    for n in n_options:
        if time.time() > deadline:
            break
        for offset in offset_options:
            if time.time() > deadline:
                break
            for p in p_options:
                if time.time() > deadline:
                    break
                seq = _make_power_law(n, p, offset)
                val = _score(seq.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = seq.tolist()
                    print(
                        f"[grid] n={n:4d} p={p:4.2f} off={offset:3.1f} → score {val:.6f}"
                    )

    # ------------------------------------------------------------------
    # 3) a quick refined gradient descent from the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=float)
        a = _project_to_simplex(a)          # ensure exact sum = 1
        a_refined, refined_val = _refine_by_descent(a, deadline)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = a_refined.tolist()
            print(f"[descent] improved to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # 4) safety net – if everything failed fall back to a simple uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = (np.ones(n, dtype=float) / n).tolist()
        best_val = _score(best_seq)
        print(f"[fallback] uniform sequence score = {best_val:.6f}")

    # 5) final sanitisation (clip to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    # final report
    final_score = _score(best_seq)
    print(f"[result] length={len(best_seq)} final score={final_score:.6f}")

    return best_seq
