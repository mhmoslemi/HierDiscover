def greedy_build(L, n, max_trials):
    sum_counts = np.zeros(2*n-1, dtype=np.int16)
    occupied = np.full(n, False, dtype=bool)
    S = []
    # candidate pool: list of indices not in S
    candidates = list(range(n))
    np.random.shuffle(candidates)
    for x in candidates:
        # compute sums of x with existing S
        if not S:
            # only self sum count = 1
            if sum_counts[2*x] + 1 <= L:
                sum_counts[2*x] += 1
                occupied[x] = True
                S.append(x)
            continue
        sums = np.array(S) + x  # shape (len(S),)
        # collisions check:
        # need existing counts at sums + 2 <= L
        if np.any(sum_counts[sums] + 2 > L):
            continue
        if sum_counts[2*x] + 1 > L:
            continue
        # accept
        sum_counts[sums] += 2
        sum_counts[2*x] += 1
        occupied[x] = True
        S.append(x)
    return np.array(S), sum_counts
```

import numpy as np
import time
import random

def greedy_build(L, n, rng):
    sum_counts = [0] * (2 * n - 1)
    S = []
    occupied = [False] * n
    indices = list(range(n))
    rng.shuffle(indices)
    for x in indices:
        # Check self contribution
        if sum_counts[2 * x] + 1 > L:
            continue
        # Check contributions with existing elements
        ok = True
        for y in S:
            if sum_counts[x + y] + 2 > L:
                ok = False
                break
        if not ok:
            continue
        # Add contributions
        for y in S:
            sum_counts[x + y] += 2
        sum_counts[2 * x] += 1
        occupied[x] = True
        S.append(x)
    max_b = max(sum_counts)
    return S, sum_counts

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a binary step function (0/1 heights) whose
    convolution maximum is small compared to the squared sum.
    The algorithm builds sets of positions with a bound on the
    number of ordered pairs that can share the same sum (greedy
    construction of low‑additive‑energy sets).  Several random
    restarts for increasing limits are performed within the time
    budget; the best sequence found is returned.
    """
    import numpy as np
    import time
    import random
    # Determine length of the step function
    try:
        n = len(height_sequence_1)  # use the provided reference length
    except Exception:
        n = 6850  # fallback length used in the statement
    deadline = time.time() + budget_s - 2.0
    rng = random.Random(seed)
    best_seq = None
    best_score = float('inf')
    attempt = 0
    # Choose a collection of target maximal pair counts (ordered)
    L_candidates = [5, 7, 9, 11, 13, 15]  # larger values allow larger supports
    while time.time() < deadline:
        L = rng.choice(L_candidates)
        S, sum_counts = greedy_build(L, n, rng)
        m = len(S)
        if m < 2:
            continue
        max_b = max(sum_counts)   # this is the actual max of the convolution
        # Approximate objective (the exact objective is the same for binary values)
        approx_score = 2.0 * n * max_b / (m * m)
        # Keep the best (lowest) score discovered so far
        if approx_score < best_score:
            best_score = approx_score
            # Build the actual sequence list of floats
            seq = [0.0] * n
            for idx in S:
                seq[idx] = 1.0
            best_seq = seq
            print(f'New best @ attempt {attempt}: L={L}, m={m}, max_b={max_b}, approx={approx_score:.6f}')
        attempt += 1
    # As a safety net, if nothing was found we fall back to the uniform baseline
    if best_seq is None:
        # uniform sequence of length n with value 1
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print('Fallback to uniform sequence, score', best_score)
    else:
        # Verify with the official evaluator (could be slightly different due to clipping)
        true_score = evaluate_sequence(best_seq)
        print(f'Final sequence: length={len(best_seq)}, true_score={true_score:.6f}, best_approx={best_score:.6f}')
    return best_seq

"""
Search for a low‑value step‑function by constructing a sparse
set of positions whose pairwise sums collide only a few times.
The algorithm builds many random “low‑additive‑energy’’ sets
using a greedy insertion that never lets a sum appear more than
L times (ordered pairs).  For each constructed set we compute the
exact objective 2·n·max(conv)/sum² (which, for binary heights,
is simply 2·n·max_count/|S|²).  The best sequence found within the
time budget is returned.
"""

import time
import numpy as np


def greedy_build(L: int, n: int, rng: np.random.Generator):
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that the ordered
    convolution count of its indicator never exceeds L.

    Returns
    -------
    S : list[int]
        List of chosen positions.
    sum_counts : list[int]
        Length 2·n‑1 list where sum_counts[k] is the number of ordered
        pairs (i,j)∈S×S with i+j = k.
    """
    sum_counts = [0] * (2 * n - 1)        # ordered pair counts for each sum
    S = []                               # chosen positions
    indices = list(range(n))
    rng.shuffle(indices)                 # random order of candidates

    for x in indices:
        # self‑pair (x,x) would increase sum 2*x by 1
        if sum_counts[2 * x] + 1 > L:
            continue

        # check all mixed pairs (x,y) with y already in S
        ok = True
        for y in S:
            if sum_counts[x + y] + 2 > L:   # (x,y) and (y,x) add 2
                ok = False
                break
        if not ok:
            continue

        # all constraints satisfied → insert x
        for y in S:
            sum_counts[x + y] += 2
        sum_counts[2 * x] += 1
        S.append(x)

    max_b = max(sum_counts)
    return S, sum_counts
