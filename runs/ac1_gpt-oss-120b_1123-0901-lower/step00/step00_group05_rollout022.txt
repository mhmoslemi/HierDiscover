"""
Propose a candidate step‑function sequence.

The construction starts from the (discrete) inverse‑square‑root law
    a_i ∝ 1/√(i+1) .
For a sequence of length n with ℓ₁‑norm √(2n) this shape gives a
self‑convolution that is almost constant, attaining an evaluation
≈π/2≈1.57 (much better than the naïve uniform sequence which scores 2).

From this analytically motivated start we run a few iterations of a
smooth gradient‑descent that tries to flatten the convolution further.
The smooth objective is a soft‑max (log‑sum‑exp) of the convolution,
whose gradient can be written in closed form and computed efficiently
with FFTs.  After each step we clip to non‑negativity, re‑normalise the
sum to √(2n) and keep the best sequence seen.

The whole routine respects the 1000 s time budget and returns a list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a sequence that (empirically) improves the evaluation score.

    The algorithm:
        1. Initialise a_i = 1/√(i+1) (inverse‑square‑root law) and scale so
           that Σ a_i = √(2n).
        2. Optionally try a few small shifts of the denominator to get a
           slightly better start.
        3. Run a simple gradient‑descent on a smooth version of the
           max‑convolution (log‑sum‑exp) using FFTs for speed.
        4. Keep the best sequence encountered and return it as a plain list.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – use the length of the previously found construction
    #    if it exists, otherwise fall back to the reported length (6269).
    # ------------------------------------------------------------------
    n = None
    if "height_sequence_1" in globals():
        try:
            n = int(len(height_sequence_1))
        except Exception:
            pass
    if n is None:
        n = 6269
    target_sum = np.sqrt(2.0 * n)            # Σ a_i must equal √(2n)

    # ------------------------------------------------------------------
    # 2. Simple analytical start: a_i ∝ 1/√(i+1) (inverse‑square‑root law).
    #    Try a few tiny shifts of the denominator (β) and keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    betas = [0.0, 0.5, 1.0, 2.0, 5.0]   # cheap one‑dimensional sweep
    idx = np.arange(n, dtype=float) + 1.0          # i+1 to avoid division by zero
    for beta in betas:
        a = 1.0 / np.sqrt(idx + beta)              # shape a_i = 1/√(i+1+β)
        a = a * (target_sum / a.sum())             # scale to the required ℓ₁‑norm
        try:
            sc = evaluate_sequence(a.tolist())
        except Exception:
            continue
        print(f"beta={beta:.2f} → initial score {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # If for any reason the analytical start failed, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # ------------------------------------------------------------------
    # 3. Refine the sequence with a few gradient‑descent steps.
    #    We minimise a smooth approximation of max(b) where
    #        b = a * a  (full discrete convolution).
    #    The smooth objective is
    #        F(a) = (1/α)·log( Σ_k exp(α·b_k) )
    #    whose gradient is
    #        ∇F = 2·(a * p)_valid,
    #    where p_k = softmax(α·b_k).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    alpha = 10.0                     # temperature for the soft‑max (larger = sharper)
    step = target_sum * 0.001        # a modest learning‑rate in the original scale
    max_iters = 2000                # safety cap
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- full convolution a*a (via FFT) ---------------------------------
        size = 1 << ((2 * n - 1).bit_length())            # next power of two
        A_hat = np.fft.rfft(a, size)
        b_full = np.fft.irfft(A_hat * A_hat, size)[: 2 * n - 1]

        # ---- evaluation (max‑convolution) -----------------------------------
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter {it:04d}] new best score {best_score:.6f}")

        # ---- soft‑max weights -------------------------------------------------
        max_b = b_full.max()
        # shift by max_b to avoid overflow
        w = np.exp(alpha * (b_full - max_b))
        p = w / w.sum()                                   # prob. vector, Σp = 1

        # ---- gradient  g = 2·(a * p)_valid ----------------------------------
        # convolution of a (len n) with p (len 2n‑1) – we need the 'valid' part
        size2 = 1 << ((n + len(p) - 1).bit_length())
        A_hat = np.fft.rfft(a, size2)
        P_hat = np.fft.rfft(p, size2)
        conv_full = np.fft.irfft(A_hat * P_hat, size2)[: n + len(p) - 1]
        # valid slice: indices n‑1 … 2n‑2  (length n)
        grad = 2.0 * conv_full[n - 1 : 2 * n - 1]

        # ---- simple projected gradient step ----------------------------------
        a = a - step * grad
        a = np.maximum(a, 0.0)                     # enforce non‑negativity
        s = a.sum()
        if s == 0.0:                               # re‑initialise if we collapsed to zero
            a = np.random.random(n)
            s = a.sum()
        a = a * (target_sum / s)                   # re‑scale to the required ℓ₁‑norm

        # ---- modest parameter decay -----------------------------------------
        step *= 0.9995
        alpha *= 0.9999
        it += 1

    print(f"Optimization finished after {it} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
