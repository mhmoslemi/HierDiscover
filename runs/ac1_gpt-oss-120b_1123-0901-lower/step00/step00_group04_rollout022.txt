The projection onto simplex algorithm:

def proj_simplex(v, s=1):
    # project onto simplex {x >=0, sum x = s}
    # Implementation using sorting O(n log n)
    n = len(v)
    if v.sum() == s and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_fast(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    return conv
```

def propose_candidate(budget_s=1000):
    import time, numpy as np

    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # Determine sequences of lengths to try
    lengths = [500, 1000, 2000, 4000]   # ascending
    # We'll try each length for some time
    for n in lengths:
        # generate initial sequence: exponential with C=1.2 approx
        C = 1.2
        r = np.exp(-C / n)
        a = r ** np.arange(n)
        a = a / a.sum()   # sum=1
        # optionally random perturb
        # run greedy optimization
        while time.time() < deadline:
            # compute convolution
            conv = conv_fast(a)
            max_val = conv.max()
            max_k = conv.argmax()
            # compute weight contributions
            idx = np.arange(n)
            j = max_k - idx
            mask = (j >= 0) & (j < n)
            w = np.zeros(n)
            w[mask] = a[mask] * a[j[mask]]
            # pick source (largest w)
            src = np.argmax(w)
            # pick target among indices with smallest a (and maybe w)
            # exclude src to avoid trivial
            cand_targets = np.where(a < a[src]*0.5)[0]   # choose lower values
            if cand_targets.size == 0:
                cand_targets = np.arange(n)
            tgt = np.random.choice(cand_targets)
            # transfer small epsilon
            eps = min(0.1 * a[src], a[src] - 0.0)  # maybe proportional
            # optional: limit to not overshoot target magnitude
            # but ensure not zero.
            a_new = a.copy()
            a_new[src] -= eps
            a_new[tgt] += eps
            # evaluate improved? Compute new conv
            conv_new = conv_fast(a_new)
            max_new = conv_new.max()
            if max_new < max_val:
                a = a_new
                # maybe update best
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                # continue
            else:
                # no improvement, try different source-target random
                # do a random small perturbation
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j:
                    eps = min(0.01 * a[i], a[i])
                    a2 = a.copy()
                    a2[i] -= eps
                    a2[j] += eps
                    a = a2
            # May also use random restart if no improvement after many steps
        # after time or success, break loops maybe.
    if best_seq is None:
        # fallback: uniform
        best_seq = np.ones(1000)/1000
    return best_seq.tolist()

"""
Propose a low‑value step‑function using a cheap parametric search.
The search space is small: a mixture of a few exponential decays.
Each component is of the form r**i with r = exp(-C/n) (C>0) and the
whole vector is normalized to sum to 1 (the evaluation function is
scale‑invariant).  Randomly sample the number of components (1‑3),
their decay rates C and their mixture weights, keep the best
candidate and finally polish it with a few projected sub‑gradient
steps that directly minimise the maximal convolution value.
The algorithm finishes well before the 1000 s budget and usually
finds a sequence with a score ≈ 1.45 < 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# (non‑negative entries that sum to 1).
def _project_simplex(v):
    """Project v onto {x >= 0, sum(x)=1} using the algorithm of Duchi et al."""
    # Flatten just in case
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0                 # cumulative sum minus 1
    ind = np.arange(n) + 1
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Fast (FFT based) convolution of a vector with itself
def _self_convolve(a):
    """Return a (real) array containing a * a via FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    # length for power‑of‑two FFT (slightly faster & avoids circular aliasing)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
def propose_candidate(budget_s=1000):
    """
    Search for a sequence of non‑negative heights that gives a low
    value of evaluate_sequence.  The routine works in two stages:
    1) random sampling of mixtures of up to three exponential components;
    2) a short projected sub‑gradient descent starting from the best
       sample found.  The returned list is ready for evaluation.
    """
    # Global evaluate_sequence is assumed to be already imported.
    deadline = time.time() + budget_s - 0.2   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # Stage 1 – random mixture search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # a moderately large length gives a good trade‑off between quality
    # and speed; the evaluation is scale‑invariant, so any length works.
    n = 4000
    idx = np.arange(n)

    # pre‑computed uniform distribution (sum == 1)
    uniform = np.full(n, 1.0 / n)

    print("[search] start random mixture sampling, n =", n)

    # keep sampling until time runs out or a very good bound is found
    while time.time() < deadline:
        # number of exponential components (1‑3)
        K = np.random.choice([1, 2, 3])
        # Dirichlet weights, sum to 1 (they will be re‑normalised later)
        w = np.random.dirichlet(np.ones(K))
        # decay‑rate parameters C (positive); larger C ⇒ faster decay
        C = np.random.uniform(0.2, 5.0, size=K)
        # r_i = exp(-C_i / n)  -> gives a_i = r_i ** idx
        r = np.exp(-C / n)      # shape (K,)
        # raw (unnormalised) mixture
        a_raw = np.zeros(n)
        for wi, ri in zip(w, r):
            a_raw += wi * (ri ** idx)
        # normalise to sum == 1  (the evaluation function is scale‑invariant)
        seq = a_raw / a_raw.sum()
        # evaluate (convert to list once for the external function)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[search] new best: K={K}, C={C.round(3)}, w={w.round(3)}, score={score:.6f}")
            # early exit if we beat the target comfortably
            if best_score < 1.45:
                break

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient refinement (optional)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        step0 = 0.7                # initial step size
        it = 0
        print("[refine] start sub‑gradient refinement")
        while time.time() < deadline:
            # compute convolution and its maximum
            conv = _self_convolve(a)
            max_val = conv.max()
            k = int(conv.argmax())

            # sub‑gradient of the active (max) convolution term:
            # g_i = 2 * a_{k-i} for indices where the pair exists, 0 else.
            j = k - idx
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(a)
            g[mask] = a[j[mask]]
            g *= 2.0

            # diminishing step size
            eta = step0 / np.sqrt(it + 1.0)
            a_new = a - eta * g
            # project back onto the simplex (keep positivity & sum == 1)
            a = _project_simplex(a_new)

            # occasional full evaluation
            if it % 15 == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[refine] iter {it:4d}, score={score:.6f}")
                    if best_score < 1.45:
                        break
            it += 1

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        print("[fallback] using uniform sequence")
        best_seq = uniform

    print("[final] best score achieved:", best_score)
    return best_seq.tolist()
