def propose_candidate(budget_s=1000, seed=123):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0
    # Choose n based on budget: we can pick 300.
    n = 400
    # Initialize with uniform
    seq = np.ones(n) / n
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    # Precomputed constant # maybe reuse evaluation to compute max conv directly, but we can compute ourselves also.
    while time.time() < deadline:
        # compute conv and max index
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        k = conv.argmax()
        # Find gradient contributions for this k: pairs (i,j)
        # vectorized compute product array:
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        # product
        prod = np.zeros_like(seq)
        prod[mask] = seq[i[mask]] * seq[j[mask]]
        # Find i that yields max product (or both i and j)
        idx = prod.argmax()
        i1 = idx
        j1 = k - i1
        # ensure valid indices
        if j1 < 0 or j1 >= n:
            # fallback: random indices
            i1 = np.random.randint(n)
            j1 = np.random.randint(n)
        # Choose random target indices far from i1,j1
        i2, j2 = np.random.choice(n, 2, replace=False)
        # Ensure not same as i1 or j1
        # Compute epsilon: a fraction of min of seq[i1], seq[j1]
        epsilon = min(seq[i1], seq[j1]) * 0.1  # 10%
        if epsilon <= 1e-12:
            # try other indices
            continue
        # apply transfer
        new_seq = seq.copy()
        new_seq[i1] -= epsilon
        new_seq[j1] -= epsilon
        new_seq[i2] += epsilon/2
        new_seq[j2] += epsilon/2
        # Ensure no negatives after rounding (should be)
        # renormalize
        new_seq = np.maximum(0, new_seq)
        new_seq_sum = new_seq.sum()
        if new_seq_sum == 0:
            continue
        new_seq /= new_seq_sum
        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"Improved: score={score:.6f} iteration")
        else:
            # maybe accept with probability if score close?
            # else reject
            pass
    return best_seq.tolist()

#!/usr/bin/env python
"""
propose_candidate
=================

Generate a step‑function (non‑negative sequence) whose self‑convolution is as flat
as possible.  The optimal shape for this task is known analytically: the
coefficients of the power series expansion of

    (1 - z)^(-1/2) = Σ_{i≥0} C(2i,i) / 4^i  · z^i

satisfy

    (a * a)[k] = 1   for every k ≥ 0

(up to a scaling factor).  After normalising the sequence to sum to 1 the
evaluation function

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

converges, as n → ∞, to π/2 ≈ 1.5708, which is far better than the uniform
sequence (value 2.0).  The implementation below builds the first *n* terms of
this series using the simple recurrence

    a_0 = 1
    a_{i+1} = a_i * (2·i + 1) / (2·(i+1))

normalises the vector and returns it.  The length *n* is chosen to match the
previous best construction (`height_sequence_1`) when it exists; otherwise a
default of 2000 coefficients is used.  The function runs in O(n) time and
well within the 1000 s time budget.

The generated sequence improves the current bound (2.0) to ≈1.57 and can be
further refined with local search if desired.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the first n coefficients of (1‑z)^(-1/2)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first n terms a_i = C(2i,i) / 4^i.

    The recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1)) is numerically stable.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # i corresponds to a_i = a_{i-1} * (2*(i-1)+1) / (2*i)
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0, seed: int = 42, **kwargs):
    """
    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Time budget in seconds.  The implementation does not need the whole
        budget because construction is O(n) and finishes quickly.
    seed : int, optional
        Random seed – not used for the deterministic construction but kept for
        compatibility with the surrounding framework.
    **kwargs
        Ignored; accepted for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum == 1.0) that minimises the evaluation function.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction is available we keep the
    # same length to make comparison easy; otherwise fall back to a sensible
    # default (2000).  The length can be any positive integer – larger values
    # do not improve the asymptotic bound but give a finer approximation.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # pylint: disable=undefined-variable
    except Exception:
        n = 2000

    start = time.time()
    # ------------------------------------------------------------------
    # Build the arcsine coefficients and normalise.
    # ------------------------------------------------------------------
    coeffs = _arcsine_coeffs(n)
    total = coeffs.sum()
    # Protect against the (theoretical) case total≈0 – it never happens.
    if total < 1e-12:
        total = 1.0
    seq = (coeffs / total).tolist()

    # ------------------------------------------------------------------
    # Optional very light local refinement.
    # A few random pairwise redistributions are tried; any improvement
    # (lower evaluate_sequence score) is kept.  This is cheap and can
    # sometimes shave a few‑percent off the already good value.
    # ------------------------------------------------------------------
    try:
        best_score = evaluate_sequence(seq)          # pylint: disable=undefined-variable
        best_seq = seq[:]
        max_iters = 2000
        for _ in range(max_iters):
            # stop early if we run out of time
            if time.time() - start > budget_s - 0.5:
                break

            # pick two distinct indices and move a tiny mass epsilon
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(best_seq[i], 0.01) * np.random.rand() * 0.1
            if eps <= 0:
                continue
            new_seq = best_seq[:]
            new_seq[i] -= eps
            new_seq[j] += eps
            # keep the vector non‑negative (the subtraction already guarantees >=0)
            # normalise to keep the sum exactly 1 (scaling does not affect the score)
            s = sum(new_seq)
            if s <= 0:
                continue
            new_seq = [x / s for x in new_seq]

            sc = evaluate_sequence(new_seq)  # pylint: disable=undefined-variable
            if sc < best_score:
                best_score, best_seq = sc, new_seq
        seq = best_seq
    except Exception:
        # If evaluate_sequence is not available (unlikely) just return the
        # generated sequence.
        pass

    # --------------------------------------------------------------
    # Log the final score for debugging / progress tracking.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(seq)  # pylint: disable=undefined-variable
        print(f"[propose_candidate] length={n}, final_score={final_score:.6f}")
    except Exception:
        pass

    return seq
