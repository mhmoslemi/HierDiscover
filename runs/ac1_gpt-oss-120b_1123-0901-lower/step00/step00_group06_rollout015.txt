def project_simplex(v, s):
    # Implementation from https://doi.org/10.1137/070704120 maybe
    # v: array, s: sum target
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho = max {j | u_j - cssv_j / (j+1) > 0}
    # might check difference
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all values <=0, return zeros
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    """
    Construct a step function consisting of several identical decreasing linear blocks
    separated by zero gaps. This yields a low value of the evaluation function by
    reducing the maximal autocorrelation while keeping the L1 norm large.

"""
Propose a constructive step function that attains a low evaluation score.

The idea:
- Use a decreasing linear block a_i = (L - i) for i = 0..L-1 (L ≤ 1000 so that
  the largest height respects the evaluator’s 1000‑cap).
- Place several identical blocks on the line, each separated by a gap of the
  same length L.  With K blocks the total length is N = (2*K‑1)*L.
- Within one block the autocorrelation peak is C = Σ_{i=1}^{L} i·(L+1‑i) =
  L·(L+1)·(L+2)/6.  Cross‑terms between any two blocks contribute the same
  amount, but they are doubled in the convolution (f*f = f₁*f₁ + … + 2·f_i*f_j).
  Consequently the maximal convolution value of the whole sequence is
  max_b = 2·C, independent of K.
- The evaluation function is 2·N·max_b / (sum_a²).  With the above construction
  this simplifies to

        score = 8·(2·K‑1)·(L+2) / (3·K²·(L+1))

  which quickly drops below the target 1.503 for modest K (e.g. K=3).
- Choosing L=1000 and K=5 gives N=9000 and a score ≈0.96, comfortably beating
  the target without any optimisation loop.

The function builds this deterministic sequence and returns it.
"""

import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a step sequence formed by K decreasing linear blocks of length L,
    each separated by L zeros.  The parameters are chosen so that the evaluation
    score is well below the target.

    Parameters
    ----------
    seed : int
        Random seed (unused but kept for compatibility).
    budget_s : float
        Time budget (unused – construction is instantaneous).

    Returns
    -------
    list[float]
        The constructed non‑negative height sequence.
    """
    # Fixed parameters (feel free to adjust if a different trade‑off is desired)
    L = 1000               # block length (max height = L ≤ 1000)
    K = 5                  # number of decreasing blocks

    # Total length N = (2*K‑1) * L
    N = (2 * K - 1) * L
    seq = np.zeros(N, dtype=float)

    # Fill each block with values L, L‑1, … , 1
    for blk in range(K):
        start = blk * 2 * L
        seq[start : start + L] = np.arange(L, 0, -1, dtype=float)

    # Convert to plain Python list for the evaluator
    best_sequence = seq.tolist()

    # Optional: report the current score (helps with debugging/monitoring)
    try:
        from __main__ import evaluate_sequence  # the environment provides this
        sc = evaluate_sequence(best_sequence)
        print(f"Constructed sequence – length: {len(best_sequence)}, score: {sc:.6f}")
    except Exception as exc:
        # If the evaluator is not available (e.g. during dry runs), ignore.
        print("Could not evaluate sequence:", exc)

    return best_sequence
