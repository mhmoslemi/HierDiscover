```
def project_simplex(v):
    # returns projection onto simplex {x>=0, sum=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
def project_to_simplex(v):
    # ensure sum=1 and v>=0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
def compute_subgradient(a, b, max_idxs):
    n = len(a)
    g = np.zeros(n)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        g[i] += 2 * a[k - i]
    g /= len(max_idxs)
    return g
```

```
def anneal(a, n, rng, time_limit):
    # parameters
    T_start = 1.0
    T_end = 1e-4
    max_iter = 1000000 # but break early based on time
    start = time.time()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    cur_a = a.copy()
    cur_score = best_score
    it = 0
    while time.time() < time_limit and it < max_iter:
        # temperature schedule geometric
        T = T_start * (T_end / T_start) ** (it / max_iter)
        i, j = rng.integers(0,n, size=2)
        if i == j:
            continue
        # choose delta as small fraction of a[i]
        delta = rng.random() * 0.1 * cur_a[i]  # up to 10% of mass at i
        if delta <= 0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # compute new score

        new_score = evaluate_sequence(new_a.tolist())
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
        it += 1
    return best_a, best_score
```

```
def compute_score(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))
```

```
def local_refine(a, time_budget, rng):
    # run subgradient descent for a limited number of steps
    start = time.time()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    # ensure sum=1
    a = a / a.sum()
    step0 = 1.0
    decay = 1e-4
    it = 0
    while time.time() - start < time_budget and it < 5000:
        # compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        max_idxs = np.where(b >= max_b - 1e-12)[0]
        # compute subgradient
        g = np.zeros_like(a)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1)
            g[i] += 2 * a[k - i]
        g = g / len(max_idxs)
        step = step0 / (1 + it * decay)
        a_new = a - step * g
        # project onto simplex
        a = project_simplex(a_new)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # random small perturbation occasionally
        if it % 500 == 0:
            perturb = rng.normal(scale=0.001, size=n)
            a = project_simplex(a + perturb)
        it += 1
    return best_a
```

assistantfinal**Solution Overview**

The goal is to find a non‑negative sequence `a = (a₀,…,a_{n‑1})` that minimises  

```
score(a) = 2·n·max_k (a∗a)_k / ( Σ_i a_i )²
```

(`∗` is the linear convolution).  
The score is *scale‑invariant*: multiplying the whole vector by a constant does not change the value, therefore we can enforce `Σ_i a_i = 1` and simply minimise  

```
F(a) = 2·n·max_k (a∗a)_k .
```

Uniform weights (`a_i = 1/n`) give `F = 2`.  Small perturbations of the uniform vector can lower the maximum entry of the convolution, therefore we perform a stochastic search:

1. **Random initialisation** – draw many candidates from Dirichlet‑type distributions with different concentration parameters.  This provides both nearly uniform vectors and very sparse vectors.
2. **Projection onto the simplex** – every candidate is normalised to have non‑negative entries that sum to 1.
3. **Fast evaluation** – a custom `conv_max` routine computes `max_k (a∗a)_k` by NumPy’s FFT based convolution ( `O(n log n)` ).  
4. **Local improvement** – for the best candidate of a batch we run a few hundred *sub‑gradient steps* that move mass away from indices that actively contribute to the current maximum convolution entry.  After each step we project back onto the simplex.
5. **Iterative refinement** – the whole process is repeated while time permits.  The best sequence found is returned.

The algorithm uses only NumPy, no external files, and respects the 1 000 s time budget.

```python
"""Propose a low‑score step‑function sequence.

The search consists of three phases:
1. Random Dirichlet draws with various concentration parameters.
2. Projection onto the probability simplex (sum = 1, non‑negative).
3. A cheap sub‑gradient descent that reduces the peak of the
   self‑convolution while keeping the vector on the simplex.

The routine respects the given time budget and returns the best
sequence discovered.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast convolution max using FFT (real valued)
def _conv_max(a: np.ndarray) -> float:
    """Return max_k (a * a)_k for a real non‑negative vector a."""
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()     # next power of two
    A = np.fft.rfft(a, size)
    B = A * A
    conv = np.fft.irfft(B, size)[: 2 * n - 1]
    return float(conv.max())


# --------------------------------------------------------------
# Sub‑gradient descent to locally reduce max(convolution)
def _refine_sequence(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
) -> np.ndarray:
    """Improve a by moving mass away from indices that attain the max."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2 * n * _conv_max(a)
    step0 = 0.5          # initial step size (relative to a's magnitude)
    decay = 1e-4         # simple decay schedule

    it = 0
    while time.time() < deadline and it < max_iters:
        # current convolution and indices attaining the maximum
        b = np.convolve(a, a)
        max_val = b.max()
        max_idx = np.where(b >= max_val - 1e-14)[0]

        # sub‑gradient of the max‑norm part (average over all active k)
        grad = np.zeros_like(a)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1)
            grad[i] += 2.0 * a[k - i]
        grad /= len(max_idx)

        # step size decreases slowly
        step = step0 / (1.0 + it * decay)
        a_new = a - step * grad
        a = _project_simplex(a_new)

        cur_val = 2 * n * _conv_max(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny random jitter to escape plateaux
        if (it & 63) == 0:
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            a = _project_simplex(a + jitter)

        it += 1

    return best_a

    """
```
