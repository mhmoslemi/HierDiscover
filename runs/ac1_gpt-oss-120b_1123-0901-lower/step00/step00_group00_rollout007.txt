def simplex_projection(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

That's O(n log n) per iteration.

"""
Simple projected gradient descent on a smoothed ℓ∞‑objective.

We minimise   max_k (a * a)[k]   under the constraints
    a_i ≥ 0 ,  Σ a_i = 1 .
The scaling of a does not affect the evaluation, therefore we keep the
ℓ₁‑norm fixed to 1 (simplex projection each step).

The non‑smooth max is replaced by a log‑sum‑exp with a temperature β.
For a given β the gradient is a weighted sum of the convolution
coefficients; the weights are a probability distribution
p_k ∝ exp(β·conv_k).  The gradient can be written as a cross‑correlation
of a and p and is computed with a single convolution.

The algorithm runs for the allocated time budget, keeps the best
sequence (lowest value of ``evaluate_sequence``) and finally returns it.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (ℓ₁ = 1, non‑negative)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projects vector v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Gradient‑descent search for a step‑function that minimises the
    evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs :
        Optional keyword ``length`` to set the size of the sequence.
        If omitted the length of the global ``height_sequence_1`` is used
        (or a default of 2000).

    Returns
    -------
    list[float]
        The best non‑negative sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine the length of the sequence
    default_len = 2000
    if 'length' in kwargs:
        n = int(kwargs['length'])
    else:
        try:
            n = len(height_sequence_1)          # type: ignore[name-defined]
            if n < 2:
                n = default_len
        except Exception:
            n = default_len

    # ------------------------------------------------------------------
    # Initialise a feasible point (simplex, sum = 1)
    # Use a random Dirichlet start to avoid symmetry.
    a = np.random.rand(n).astype(np.float64)
    a = a / a.sum()

    # Keep the best sequence seen so far
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())   # type: ignore[name-defined]
    except Exception:
        best_score = np.inf

    # ------------------------------------------------------------------
    # Hyper‑parameters
    beta = 10.0               # smoothing for the max (log‑sum‑exp temperature)
    eta = 0.08                # learning rate
    min_eta = 1e-5
    stagnation = 0            # iterations without improvement
    max_stagnation = 300      # after this we restart from a new random point

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- convolution of the current vector with itself
        conv = np.convolve(a, a)               # length 2n‑1
        max_conv = conv.max()

        # ----- smoothed max: weights p_k ∝ exp(beta*(conv_k - max_conv))
        # shift by max_conv to avoid overflow
        exp_arg = beta * (conv - max_conv)
        # for numerical stability clamp very small values
        np.maximum(exp_arg, -700, out=exp_arg)       # avoid underflow in exp
        p = np.exp(exp_arg)
        p_sum = p.sum()
        if p_sum == 0 or not np.isfinite(p_sum):
            # fallback to uniform weights (rare)
            p = np.ones_like(p) / p.size
        else:
            p = p / p_sum

        # ----- gradient of the smoothed objective
        # g_j = 2 * Σ_i a_i * p_{i + j}
        # this is a cross‑correlation a ⋆ p, which can be obtained via
        # convolution of a with the reversed p.
        corr = np.convolve(a, p[::-1])          # length n + (2n‑1) - 1 = 3n‑2
        start = (len(p) - 1)                    # = 2n‑2
        g = 2.0 * corr[start:start + n]

        # ----- gradient step and simplex projection
        a_new = a - eta * g
        a_new = _project_onto_simplex(a_new)

        # occasional random jitter to escape flat regions
        if np.random.rand() < 0.02:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new)

        a = a_new

        # ----- evaluate
        try:
            cur_score = evaluate_sequence(a.tolist())   # type: ignore[name-defined]
        except Exception:
            cur_score = np.inf

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
            # shrink temperature a little when we improve (focus more on true max)
            beta = min(beta * 1.05, 100.0)
            # optionally increase learning rate a bit
            eta = min(eta * 1.05, 0.2)
        else:
            stagnation += 1
            # slowly decay learning rate when no progress
            eta = max(eta * 0.99, min_eta)

        # restart if we are stuck for a while
        if stagnation >= max_stagnation:
            a = np.random.rand(n)
            a = a / a.sum()
            stagnation = 0
            beta = 10.0
            eta = 0.08

        # ----- occasional logging
        if iteration % 100 == 0:
            print(f"Iter {iteration:6d} | best = {best_score:.6f} | cur = {cur_score:.6f} | η={eta:.4g} | β={beta:.2f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    return best_seq.tolist()
