```python
import numpy as np
import time

def next_pow2(x):
    """Return the next power of two >= x."""
    return 1 << (x - 1).bit_length()

def compute_conv(f, N):
    """Return full linear convolution of f with itself using FFT size N."""
    F = np.fft.fft(f, N)
    conv = np.real(np.fft.ifft(F * F))[:2*len(f)-1]
    return conv, F

def compute_score_from_conv(conv, total):
    """Compute the objective value given convolution and total sum."""
    n = (len(conv) + 1) // 2
    max_b = np.max(conv)
    return 2 * n * max_b / (total * total)

def softmax_weights(conv, eps):
    """Return a probability vector proportional to exp((conv - max)/eps)."""
    c = conv
    max_c = np.max(c)
    # subtract max for numerical stability
    exps = np.exp((c - max_c) / eps)
    s = exps.sum()
    if s == 0:
        # fallback: put all weight on the max index
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return exps / s

def gradient_via_softmax(f, F_fft, w):
    """Compute gradient of the smoothed max with respect to f."""
    N = len(F_fft)
    W_fft = np.fft.fft(w, N)
    # correlation of w and f (linear, non‑circular) = ifft(conj(F) * W)
    corr = np.real(np.fft.ifft(np.conj(F_fft) * W_fft))[:len(f)]
    return 2.0 * corr

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Gradient‑descent based search for a step‑function that minimises
    evaluate_sequence.  The method works on the simplex (sum = 1) and
    repeatedly smooths the max‑convolution with a soft‑max, follows the
    resulting gradient and injects small random perturbations to escape
    the uniform stationary point.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)   # a small safety margin

    # --------------------------------------------------------------
    # Choose a starting length and initialise the vector.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=float)
        n = len(init)
        # normalise to sum = 1 (the scale does not affect the objective)
        f = init.copy()
        s = f.sum()
        if s <= 0:
            f = np.full(n, 1.0 / n)
        else:
            f = f / s
    else:
        # pick a length that is large enough to allow a sub‑uniform peak
        n = 2500
        f = np.full(n, 1.0 / n)

    # Ensure strict positivity to avoid zero‑gradient plateaus.
    f = np.maximum(f, 1e-12)
    f = f / f.sum()

    # FFT size (next power of two >= 3*n)
    N = next_pow2(3 * n)

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    max_eps = 0.5          # initial soft‑max temperature
    min_eps = 1e-5
    eps = max_eps
    eps_decay = 0.998      # multiplicative decay per iteration
    step_factor = 0.2      # step size = step_factor * eps
    min_step = 1e-6

    best_f = f.copy()
    # Compute initial score directly (fast, no extra call)
    conv, F_fft = compute_conv(f, N)
    best_score = compute_score_from_conv(conv, f.sum())
    print(f"Initial length={n}, score={best_score:.6f}")

    iteration = 0
    while time.time() < deadline:
        # ----------------------------------------------------------
        # Soft‑max on the convolution to obtain a smooth approximation
        # of the max and its gradient.
        # ----------------------------------------------------------
        conv, F_fft = compute_conv(f, N)
        max_b = np.max(conv)

        # (objective = 2*n*max_b because sum=1)
        cur_score = 2.0 * n * max_b

        w = softmax_weights(conv, eps)

        g = gradient_via_softmax(f, F_fft, w)

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_self_convolution(f: np.ndarray, N: int):
    """
    Compute the full linear convolution f * f using FFT size N.
    Returns the convolution (length 2*len(f)-1) and the FFT of f.
    """
    F = np.fft.fft(f, N)
    conv = np.real(np.fft.ifft(F * F))[: 2 * len(f) - 1]
    return conv, F


def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """
    Soft‑max approximation of the max of ``conv``.
    The temperature ``eps`` is gradually reduced during the run.
    """
    max_c = np.max(conv)
    # stable exponentiation
    exp_vals = np.exp((conv - max_c) / eps)
    s = exp_vals.sum()
    if s == 0.0:                              # under‑flow – put all mass on the max index
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_vals / s


def _gradient_softmax(f: np.ndarray, F_fft: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smoothed max with respect to the vector ``f``.
    Derived from g = 2 * (w ⋆ f) where “⋆” denotes linear correlation.
    Implements the correlation via FFTs.
    """
    W_fft = np.fft.fft(w, N)
    # linear correlation (w ⋆ f) – the first len(f) entries correspond to lags 0…len(f)-1
    corr = np.real(np.fft.ifft(np.conj(F_fft) * W_fft))[: len(f)]
    return 2.0 * corr


def _objective_from_convolution(conv: np.ndarray, total: float) -> float:
    """
    The exact objective used by ``evaluate_sequence`` (up to the internal clamping,
    which is irrelevant for our tiny positive numbers):
        2 * n * max(conv) / total**2
    """
    n = (len(conv) + 1) // 2          # because conv length = 2n‑1
    return 2.0 * n * np.max(conv) / (total * total)
