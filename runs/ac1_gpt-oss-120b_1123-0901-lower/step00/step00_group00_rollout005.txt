"""
Propose a candidate step‑function (list of non‑negative heights) that improves
the evaluation score.

The algorithm is a simple simulated‑annealing / targeted‑mass‑move search:

* The sequence length is taken from the previously known construction
  ``height_sequence_1`` (default 2000 if unavailable).
* The sum of the coefficients is fixed to  √(2·n)  – this normalization makes
  the evaluation equal to the maximal self‑convolution value.
* At each iteration we compute the full self‑convolution using an FFT
  (O(n log n)) and locate the index ``k_max`` where it attains its maximum.
* The contribution of each coefficient ``a[i]`` to that maximal term is
  ``a[i]·a[k_max−i]`` (when the second index is in range).  We choose a source
  index with probability proportional to that contribution and move a small
  amount of mass to a uniformly chosen target index.  The total sum is kept
  constant, so the move preserves the normalisation.
* Moves are accepted greedily if they improve the score or probabilistically
  according to a temperature schedule (simulated annealing) to escape local
  minima.
* Occasionally a random Gaussian perturbation of a few entries is applied
  (followed by renormalisation) to increase diversity.
* The best sequence seen so far is remembered and returned.

The method runs for up to ``budget_s`` seconds (default 1000) and prints
progress messages with the current best score.
"""

import time
import math
import numpy as np


def _fft_convolve_self(x: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full convolution x * x using real FFT of length ``fft_len``."""
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)
    # only the first 2*len(x)-1 entries are the linear convolution
    return conv[: 2 * len(x) - 1]

    Search for a low‑value sequence using simulated annealing.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds before
        this limit to guarantee a return value).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """

    # ----------------------------------------------------------------------

        # ``height_sequence_1`` is provided by the surrounding environment.
        n = len(height_sequence_1)  # type: ignore[name-defined]
        start_arr = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
    except Exception:
        # Fall back to a moderate size if the global is missing.
        n = 2000
        start_arr = None

    target_sum = math.sqrt(2.0 * n)

        cur *= (target_sum / cur.sum())

    # 2. Pre‑compute FFT length (next power of two) for fast convolutions.
    # ----------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Helper that returns (score, max_convolution, full_convolution)
    def _score(arr: np.ndarray):
        s = arr.sum()
        conv = _fft_convolve_self(arr, fft_len)
        maxb = conv.max()
        # The official objective (identical to evaluate_sequence for normalized sum)
        return 2.0 * n * maxb / (s * s), maxb, conv

    cur_score, cur_maxb, cur_conv = _score(cur)
    best_arr = cur.copy()
    best_score = cur_score

    print(f"Initial score: {best_score:.6f} (n={n})")

    # ----------------------------------------------------------------------
    # 3. Simulated annealing parameters.
    # ----------------------------------------------------------------------
    temperature = 0.5 * best_score            # initial temperature
    decay = 0.9995                            # exponential cooling per iteration
    step_factor = 0.2                         # typical fraction of a source value to move
    perturb_prob = 0.05                       # chance to perform a random perturbation
    deadline = time.time() + budget_s - 5.0   # stop a few seconds early
    it = 0
    last_improve = time.time()

    # ----------------------------------------------------------------------
    # 4. Main optimisation loop.
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # Decide on move type.
        if np.random.rand() < perturb_prob:
            # ---- Random perturbation of a few entries ----
            new_arr = cur.copy()
            # pick a small set of indices and add Gaussian noise
            idxs = np.random.choice(n, size=min(5, n), replace=False)
            noise = np.random.randn(len(idxs)) * step_factor * target_sum / n
            new_arr[idxs] = np.maximum(new_arr[idxs] + noise, 0.0)
            # Renormalise to keep the prescribed sum.
            new_arr *= (target_sum / new_arr.sum())
        else:
            # ---- Targeted mass transfer -------------------------------------------------
            # Identify the index where the convolution is maximal.
            k_max = int(np.argmax(cur_conv))

            # Determine the range of i for which a[i] participates in that term.
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)

            # Contributions a[i] * a[k_max-i] (zero where counterpart outside range).
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contrib = cur[i_range] * cur[j_range]

            total_contrib = contrib.sum()
            if total_contrib > 0.0 and np.random.rand() < 0.9:
                # Choose a source index biased by its contribution.
                probs = contrib / total_contrib
                src = np.random.choice(i_range, p=probs)
            else:
                # Fallback: pick any index with non‑zero mass.
                nonzero = np.flatnonzero(cur > 0)
                if nonzero.size == 0:
                    src = np.random.randint(n)
                else:
                    src = np.random.choice(nonzero)

            # Choose a distinct target index uniformly.
            tgt = np.random.randint(n)
            while tgt == src:
                tgt = np.random.randint(n)

            # Amount to move: a fraction of the source's current value.
            max_move = cur[src]
            delta = max_move * step_factor * np.random.rand()

            new_arr = cur.copy()
            new_arr[src] -= delta
            new_arr[tgt] += delta
            # No renormalisation needed – the total sum stays unchanged.

        # ---- Evaluate the candidate -------------------------------------------------
        new_score, new_maxb, new_conv = _score(new_arr)

        # Accept if better, otherwise accept with Metropolis probability.
        if new_score < cur_score or np.exp(-(new_score - cur_score) / max(temperature, 1e-12)) > np.random.rand():
            cur = new_arr
            cur_score = new_score
            cur_maxb = new_maxb
            cur_conv = new_conv

            if new_score < best_score:
                best_score = new_score
                best_arr = new_arr.copy()
                last_improve = time.time()
                print(f"Iter {it:07d} – new best: {best_score:.6f}")

        # Cool down.
        temperature *= decay

        # Occasionally shake the system if we have been stuck for a while.
        if it % 5000 == 0 and (time.time() - last_improve) > 30.0:
            # Small random reshuffle around the current best.
            perm = np.random.permutation(n)
            cur = best_arr[perm].copy()
            # Add tiny noise and re‑normalise.
            cur += np.random.randn(n) * step_factor * target_sum / n
            cur = np.maximum(cur, 0.0)
            cur *= (target_sum / cur.sum())
            cur_score, cur_maxb, cur_conv = _score(cur)
            print(f"Iter {it:07d} – stall reset, best still {best_score:.6f}")

    print(f"Finished after {it} iterations. Best score: {best_score:.6f}")
    return best_arr.tolist()
