def propose_candidate(seed=0, budget_s=1000):
    """Generate a sequence of non‑negative heights that tries to minimise
    evaluate_sequence.  The algorithm searches over many random starts of a
    probability vector f (∑ f = 1) and performs a projected exponentiated
    gradient descent on the smooth approximation of the maximum convolution
    value.  The best vector found is returned as a plain Python list.
    """
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a little margin

    best_score = math.inf
    best_vec = None

    # parameters that can be tuned on the fly
    # we explore a range of lengths to give the optimiser more flexibility
    lengths = list(range(120, 401, 20))   # 120,140,...,400
    # pre‑shuffled order to avoid deterministic pattern
    np.random.shuffle(lengths)

    # hyper‑parameters
    inner_iters = 600      # number of gradient steps for each start
    eta = 5.0              # step size for exponentiated update
    alpha_start = 0.5
    alpha_end = 20.0

    # a tiny function that projects onto the probability simplex
    def proj_simplex(v):
        """Euclidean projection of v onto {x >=0, sum x = 1}."""
        # Duchi et al. algorithm
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # fallback – return uniform
            return np.full_like(v, 1.0 / len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w

    # main loop – keep trying new random starts until time runs out
    while time.time() < deadline:
        # pick a length (cycle through the list)
        n = lengths[np.random.randint(len(lengths))]
        # initialise a random probability vector
        f = np.random.rand(n)
        f /= f.sum()

        # gradually increase the smooth‑max temperature
        alpha = alpha_start
        alpha_increment = (alpha_end - alpha_start) / max(inner_iters, 1)

        for it in range(inner_iters):
            # time guard
            if time.time() >= deadline:
                break

            # convolution and current score (sum(f)=1 ⇒ score = 2 n * max)
            conv = np.convolve(f, f)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_vec = f.copy()
                print(f"New best: {best_score:.6f} (n={n}, it={it})")

            # soft‑max weights – a smooth sub‑gradient of the max operator
            # stability trick: subtract the max before exponentiation
            exp_vals = np.exp(alpha * (conv - max_conv))
            w = exp_vals / exp_vals.sum()

            # gradient of the smooth max w.r.t. f.
            # Using: grad_i = 2·∑_j w_{i+j}·f_j
            rev_f = f[::-1]
            # full convolution of w with rev_f gives the summed terms
            tmp = np.convolve(w, rev_f)          # length = (2n‑1)+n‑1
            grad = 2.0 * tmp[n-1 : n-1 + n]      # extract the n relevant entries

            # exponentiated (mirror) update – guarantees non‑negativity
            f = f * np.exp(-eta * grad)

            # project back onto the simplex (simple normalisation is enough,
            # but we use a robust Euclidean projection to avoid numerical drift)
            f = proj_simplex(f)

            # increase the temperature so that the approximation becomes tighter
            alpha = min(alpha + alpha_increment, alpha_end)

            # occasional tiny random jitter to escape shallow plateaus
            if it % 50 == 0 and np.random.rand() < 0.3:
                jitter = np.random.randn(n) * 1e-4
                f = proj_simplex(f + jitter)

        # end of inner optimisation – try a fresh random start
        # optionally we could also try a short “refine” run on the current best,
        # but the outer loop already revisits many starts.

    # ----------------------------------------------------------------------
    # If nothing was found (should never happen) fall back to a uniform vector
    if best_vec is None:
        best_vec = np.ones(1)

    # Convert to a plain Python list of floats – evaluate_sequence will clip
    # values that are out of the allowed range, so we keep them as they are.
    return [float(v) for v in best_vec]

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative numbers that (hopefully) gives a low
    ``evaluate_sequence`` score.

    Approach
    --------
    * Work with a probability vector ``f`` (sum = 1).  The score of a vector
      with sum 1 is ``2 * n * max(conv(f, f))`` where ``n = len(f)``.
    * The problem is to minimise the peak of the self‑convolution.
      This is a non‑convex optimisation but the gradient of a smooth
      approximation of the ``max`` operator is easy to compute.
    * For a given ``f`` we form ``conv = np.convolve(f, f)`` and replace
      ``max(conv)`` by the soft‑max
          ``w_k = exp(alpha * (conv_k - max(conv))) / sum(exp(...))``.
      The gradient of ``sum_k w_k * conv_k`` w.r.t. ``f`` is
          ``grad_i = 2 * sum_j w_{i+j} * f_j``.
      Using a reversed copy of ``f`` this can be obtained with a single
      convolution.
    * An *exponentiated* (mirror) gradient step
          ``f = f * exp(-eta * grad)`` guarantees non‑negativity.
      Afterwards the vector is projected back onto the probability simplex.
    * Many short runs with random initial vectors and different lengths are
      performed until the time budget expires.  The best vector seen
      (lowest true score) is returned.

    The algorithm runs entirely in NumPy, is deterministic given the seed,
    and respects the 1000‑second limit.
    """
    import time
    import math
    import numpy as np

    # ------------------------------------------------------------------
    # helpers
    def proj_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
        # Duchi et al. (2008) – O(n log n) sort based projection
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
        if rho.size == 0:
            # fallback – uniform vector (should not happen)
            return np.full_like(v, 1.0 / len(v))
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # ------------------------------------------------------------------
    # main search parameters
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    best_score = math.inf
    best_vec = None

    # a set of lengths we will explore (shuffled to avoid deterministic loops)
    lengths = list(range(120, 401, 20))               # 120, 140, …, 400
    np.random.shuffle(lengths)

    # optimisation hyper‑parameters
    inner_iters = 600            # gradient steps per random start
    eta = 5.0                    # step size for the exponentiated update
    alpha_start = 0.5            # temperature for the smooth‑max
    alpha_end = 20.0

    # ------------------------------------------------------------------
    # outer loop – keep trying fresh random starts until time runs out
    while time.time() < deadline:
        # pick a length (sample with replacement)
        n = lengths[np.random.randint(len(lengths))]

        # random initial probability vector
        f = np.random.rand(n)
        f = f / f.sum()

        # smooth‑max temperature schedule (linear increase)
        alpha = alpha_start
        alpha_inc = (alpha_end - alpha_start) / max(inner_iters, 1)

        for it in range(inner_iters):
            # quick time check
            if time.time() >= deadline:
                break

            # -----------------------------------------------------------
            # evaluate current objective (exact score, not the smooth one)
            conv = np.convolve(f, f)          # length 2n‑1
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv     # because sum(f)=1
            if cur_score < best_score:
                best_score = cur_score
                best_vec = f.copy()
                print(f"New best: {best_score:.6f} (n={n}, it={it})")

            # -----------------------------------------------------------
            # smooth‑max weights – a differentiable surrogate for the max
            exp_vals = np.exp(alpha * (conv - max_conv))   # stable: max(conv) subtracted
            w = exp_vals / exp_vals.sum()                 # shape (2n‑1,)

            # -----------------------------------------------------------
            # gradient of the smooth‑max w.r.t. f:
            #   grad_i = 2 * sum_j w_{i+j} * f_j
            # Compute via a single convolution with a reversed copy of f
            rev_f = f[::-1]
            tmp = np.convolve(w, rev_f)     # length = (2n‑1) + n‑1 = 3n‑2
            grad = 2.0 * tmp[n - 1 : n - 1 + n]   # extract n entries

            # -----------------------------------------------------------
            # mirror (exponentiated) gradient step – guarantees non‑negativity
            f = f * np.exp(-eta * grad)

            # projection back onto the probability simplex
            f = proj_simplex(f)

            # gradually tighten the smooth‑max
            alpha = min(alpha + alpha_inc, alpha_end)

            # occasional tiny random jitter to avoid stagnation
            if it %  fifty == 0 and np.random.rand() < 0.3:
                jitter = np.random.randn(n) * 1e-4
                f = proj_simplex(f + jitter)

        # end of inner optimisation loop – a new random start will be tried

    # ------------------------------------------------------------------
    # fallback if nothing was discovered (should never happen)
    if best_vec is None:
        best_vec = np.ones(1)

    # evaluate_sequence works with a plain Python list; keep the values as
    # plain floats (they already satisfy the non‑negative and sum>0 checks)
    return [float(v) for v in best_vec]
```
