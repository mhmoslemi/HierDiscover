"""
Minimal‑max‑convolution search.

The evaluation metric is   2·n·max(convolution) / (sum(sequence)^2).
Because the objective is homogeneous we may fix   sum(sequence)=1   and
only minimise the maximal aperiodic autocorrelation (the convolution).
For a fixed length n this is the optimisation problem

    minimise   t
    s.t.       sum_i a_i = 1 ,      a_i ≥ 0
              ∀k∈{0,…,2n‑2} :   Σ_i a_i·a_{k‑i} ≤ t .

The constraints are quadratic, but SciPy’s SLSQP can handle them as
non‑linear inequality constraints.  The routine below tries a few
different sequence lengths (small n gives a smaller prefactor 2·n) and
several random starts.  The best feasible sequence found within the
time budget is returned.

If the optimiser fails to produce a feasible point we fall back to a
simple uniform sequence.
"""

import time
import numpy as np
from scipy import optimize

def propose_candidate(seed: int = 1234, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) with a low
    evaluation score.  The function is called by the evaluation harness,
    receives a time budget and must return a list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3   # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # sequence lengths that are cheap enough for the SLSQP solver.
    # Small n makes the factor 2·n small, while the theoretical lower
    # bound 2·n/(2·n‑1) quickly approaches 1.
    lengths = [10, 12, 15, 20, 25, 30, 40, 50, 70, 100, 150, 200, 300]

    # ------------------------------------------------------------------
    # Helper: solve the min‑max‑convolution problem for a given length.
    # ------------------------------------------------------------------
    def solve_for_n(n, start_a):
        """Return a feasible sequence of length n (or None)."""
        a0 = start_a.copy()
        a0 /= a0.sum()
        conv0 = np.convolve(a0, a0)
        t0 = conv0.max() + 1e-12                     # feasible initial t
        x0 = np.concatenate([a0, [t0]])             # optimisation vector

        # equality: sum_i a_i = 1
        def eq_sum(x):
            return np.sum(x[:n]) - 1.0

        # inequality: t - (a * a)[k] ≥ 0   for every shift k
        def ineq_conv(x):
            a = x[:n]
            t = x[n]
            conv = np.convolve(a, a)
            return t - conv                         # vector valued

        cons = [
            {"type": "eq",  "fun": eq_sum},
            {"type": "ineq","fun": ineq_conv},
        ]
        bounds = [(0.0, None)] * n + [(0.0, None)]

        # SLSQP is a gradient‑free (finite‑difference) optimizer that
        # handles non‑linear constraints.
        try:
            res = optimize.minimize(
                lambda x: x[n],                # minimise t
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-12, "disp": False},
            )
        except Exception:                      # any numerical trouble
            return None

        if not res.success:
            return None

        a_opt = res.x[:n]
        a_opt = np.maximum(a_opt, 0.0)          # numerical safety
        s = a_opt.sum()
        if s < 1e-12:
            return None
        a_opt /= s                               # enforce sum=1 exactly
        return a_opt

    # --------------------------------------------------------------
    # Main search loop over lengths and random starts.
    # --------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        # a few independent starts per length (more attempts for small n)
        attempts = 8 if n <= 30 else 4
        for _ in range(attempts):
            if time.time() > deadline:
                break

            # Random Dirichlet start (positive, sums to 1)
            start = rng.random(n)
            start /= start.sum()

            seq = solve_for_n(n, start)
            if seq is None:
                continue

            # Evaluate – evaluate_sequence does its own clipping and
            # normalisation, but we already have sum(seq)=1.
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue

            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time():.1f}] new best: n={n} score={score:.6f}")

    # --------------------------------------------------------------
    # If nothing succeeded (unlikely) fall back to a uniform vector.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 200
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print("fallback uniform sequence, score =", best_score)

    print("final best score =", best_score, "length =", len(best_seq))
    return best_seq
