import time
import numpy as np

def generate_B2_sequence(max_n, g, rng):
    """
    Greedy construction of a set S ⊂ {0,…,max_n-1} whose unordered sums appear at most `g` times.
    Returns a list of floats (0.0/1.0) of length max(S)+1.
    """
    sum_counts = np.zeros(2*max_n, dtype=np.int16)
    selected = []
    perm = rng.permutation(max_n)
    for i in perm:
        # quick test for self sum
        if sum_counts[i+i] >= g:
            continue
        feasible = True
        for j in selected:
            s = i + j
            if sum_counts[s] >= g:
                feasible = False
                break
        if not feasible:
            continue
        # add i
        for j in selected:
            sum_counts[i + j] += 1
        sum_counts[i + i] += 1
        selected.append(i)
    if not selected:
        return [0.0]
    n = max(selected) + 1
    seq = np.zeros(n, dtype=np.float64)
    seq[selected] = 1.0
    return seq.tolist()

"""
Propose a step‑function with a very small evaluation score.

The core idea is to build a *B₂[g]* set (also called a generalized Sidon set):
a subset S ⊂ {0,…,N‑1} such that every unordered sum i+j (i≤j) occurs at most
`g` times.  For a binary sequence a_i = 1_{i∈S} the discrete convolution
a*a at index k is

    (a*a)[k] = 2·(# unordered pairs i<j with i+j=k) + (# i with 2i=k)

Hence if `g = 2` the convolution never exceeds 5, regardless of how many
elements are chosen.  The score returned by `evaluate_sequence` is

    score = 2·n·max(a*a) / (∑a)² .

With `max(a*a) ≤ 5` we obtain, e.g. for n≈5000 and |S|≈200,
score ≈ 1.25 < 1.503.  The algorithm repeatedly draws a random ordering of
the indices, greedily inserts feasible indices while respecting the `g`
constraint, and finally augments the set with a deterministic pass that tries
to insert any still‑feasible index.  Many independent trials are performed
until the allotted time expires; the best sequence found is returned.

No external files or network access are used; only numpy and the supplied
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: greedy construction of a B₂[g] set (unordered sums ≤ g)
# ----------------------------------------------------------------------
def _make_B2_sequence(max_n: int, g: int, rng: np.random.Generator) -> list[float]:
    """
    Create a binary sequence (list of 0.0/1.0) of length n = max(S)+1 where
    S ⊂ {0,…,max_n‑1} satisfies: for every integer s,
    the number of unordered pairs {i,j}⊂S with i≤j and i+j=s is at most g.

    Parameters
    ----------
    max_n : int
        Upper bound for allowed indices (the range is 0 … max_n‑1).
    g : int
        Maximal allowed multiplicity of any unordered sum.
    rng : np.random.Generator
        Random number generator.

    Returns
    -------
    seq : list[float]
        Binary sequence of length n (n = max(S)+1) with 1.0 at the selected
        positions.
    """
    # count of unordered pairs that give each possible sum
    sum_counts = np.zeros(2 * max_n, dtype=np.int16)

    selected = []            # list of chosen indices (order does not matter)
    selected_set = set()

    # ------------------------------------------------------------------
    # Random insertion phase
    # ------------------------------------------------------------------
    perm = rng.permutation(max_n)
    for i in perm:
        if sum_counts[i + i] >= g:
            continue  # self‑sum would exceed the limit

        feasible = True
        for j in selected:
            if sum_counts[i + j] >= g:
                feasible = False
                break
        if not feasible:
            continue

        # add i to the set
        for j in selected:
            sum_counts[i + j] += 1
        sum_counts[i + i] += 1
        selected.append(i)
        selected_set.add(i)

    # ------------------------------------------------------------------
    # Deterministic augmentation phase – try to insert any remaining index
    # ------------------------------------------------------------------
    improved = True
    while improved:
        improved = False
        for i in range(max_n):
            if i in selected_set:
                continue
            if sum_counts[i + i] >= g:
                continue

            feasible = True
            for j in selected:
                if sum_counts[i + j] >= g:
                    feasible = False
                    break
            if not feasible:
                continue

            # insert i
            for j in selected:
                sum_counts[i + j] += 1
            sum_counts[i + i] += 1
            selected.append(i)
            selected_set.add(i)
            improved = True
        # if we added at least one element we repeat the scan,
        # because the new element may enable further insertions.

    if not selected:
        # fallback – a single positive entry (the evaluate function would reject
        # a sum that is too close to zero, so we keep at least something).
        return [1.0]

    n = max(selected) + 1
    seq = np.zeros(n, dtype=np.float64)
    seq[selected] = 1.0
    return seq.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Randomised search for a sparse binary sequence with a low
    `evaluate_sequence` score.

    The search repeatedly:
        * picks a random target length N (2000–8000),
        * builds a B₂[g] set with g∈{2,3},
        * evaluates the produced sequence,
        * keeps the best one found before the time budget runs out.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds (the function must return before the
        budget expires).

    Returns
    -------
    best_seq : list[float]
        The best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a small safety margin

    best_seq = None
    best_score = float('inf')
    attempts = 0

    # We will try a few different limits on the sum multiplicity.
    g_choices = [2, 3]

    while time.time() < deadline:
        # Randomly choose the maximal index range for this trial.
        max_n = int(rng.integers(2000, 8000))
        g = int(rng.choice(g_choices))

        seq = _make_B2_sequence(max_n, g, rng)

        # evaluate – the provided function takes a plain Python list.
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # Any unexpected problem (should not happen) – skip this trial.
            continue

        attempts += 1
        if score < best_score:
            best_score = score
            best_seq = seq
            print(
                f"[{attempts}] new best score = {best_score:.6f} "
                f"| length = {len(seq)} | g = {g}"
            )

    # In the extremely unlikely case that no candidate was accepted,
    # fall back to a trivial uniform sequence (score = 2.0).
    if best_seq is None:
        best_seq = [1.0] * 10
        best_score = evaluate_sequence(best_seq)
        print(f"fallback uniform sequence, score = {best_score:.6f}")

    return best_seq
