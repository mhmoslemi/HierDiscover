def compute_score(seq):
    s = np.sum(seq)
    if s < 1e-8: return large.
    normalized = seq * np.sqrt(2*len(seq)) / s
    # compute convolution via FFT, using real FFT for speed
    fft_len = 1 << (2*len(seq)-1).bit_length()
    fft_seq = np.fft.rfft(normalized, fft_len)
    conv = np.fft.irfft(fft_seq * fft_seq, fft_len)
    # conv length > needed, take first 2*len(seq)-1
    conv = conv[:2*len(seq)-1]
    max_b = np.max(conv)

    # Since we normalized, denominator is 2n, thus score = max_b (should be approx).
    return max_b

import scipy.sparse as sp

def solve_convolution_lp(f, rhs):
    n = len(f)
    # each constraint: sum_i f[i] * b[k - i] <= rhs for k in 0..2n-2, with indices within bounds.
    rows = []
    cols = []
    data = []
    b = []

    for k in range(2*n-1):
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # For each i, j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            # coefficient f[i] multiplies variable b[j]
            rows.append(k)
            cols.append(j)
            data.append(f[i])
        b.append(rhs)

    # nonnegativity constraints: -b_i <= 0
    for i in range(n):
        rows.append(2*n-1 + i)
        cols.append(i)
        data.append(-1.0)
        b.append(0.0)

    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1 + n, n))
    c = -np.ones(n)  # maximize sum(b) -> minimize -sum(b)
    result = linprog(c, A_ub=A, b_ub=b, method='highs', bounds=(0, None), options={'time_limit': 10.0})

    if result.success:
        return result.x
    else:
        return None

def compute_score(seq):
    n = len(seq)
    # No scaling needed; evaluate_sequence will scale but we can compute same.
    a = np.array(seq, dtype=np.float64)
    # Use real FFT with length = next power of two >= 2*n
    L = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[:2*n-1]
    max_b = np.max(conv)
    # denominator: sum_a^2
    sum_a = np.sum(a)
    if sum_a < 1e-8:
        return np.inf
    score = float(2 * n * max_b / (sum_a**2))
    return score

    ``evaluate_sequence``.  The search combines three ideas:
    – an alternating‑direction scheme based on a linear program (LP)
      that maximises the total mass while keeping the convolution bound,
    – an aggressive line‑search over the mixing factor,
    – occasional random perturbations (simulated‑annealing style).

    import numpy as np, time, random
    from scipy.optimize import linprog
    import scipy.sparse as sp

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    def fast_score(a):
        """Return the same quantity as ``evaluate_sequence`` but using FFT."""
        a = np.asarray(a, dtype=np.float64)
        n = a.size
        if a.sum() < 1e-8:
            return np.inf
        # length for FFT – next power of two for speed
        L = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
        maxb = np.max(conv)
        return float(2.0 * n * maxb / (a.sum() ** 2))

    def lp_g(f, rhs):
        """Solve  max 1·b  s.t.  conv(f,b) ≤ rhs,  b ≥ 0.
        Returns the optimal b (as a dense numpy array) or None on failure.
        """
        n = f.size
        # build sparse matrix of shape (2n-1 + n, n)
        rows = []
        cols = []
        data = []

        # convolution constraints
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # iterate over i where j = k-i is inside the interval
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(k)
                cols.append(j)
                data.append(f[i])

        # non‑negativity constraints  –b ≤ 0
        offset = 2 * n - 1
        for i in range(n):
            rows.append(offset + i)
            cols.append(i)
            data.append(-1.0)

        A = sp.csr_matrix((data, (rows, cols)), shape=(offset + n, n))
        b = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

        # we maximise sum(b)  → minimise –sum(b)
        c = -np.ones(n)

        # use HiGHS –good for large sparse problems
        res = linprog(c, A_ub=A, b_ub=b, method='highs', bounds=(0, None),
                      options={'presolve': True, 'time_limit': 8.0})
        if res.success:
            return res.x
        return None

    def mix(f, g, t):
        """Return (1‑t)·f + t·g after re‑normalising g to the same ℓ₁‑norm as f."""
        # re‑scale g so that its ℓ₁ equals that of f
        sf = f.sum()
        sg = g.sum()
        if sg == 0:
            return f.copy()
        g = g * (sf / sg)
        return (1.0 - t) * f + t * g

    # Search parameters
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [512, 768, 1024, 1536]   # all comfortably fit into memory for the LP
    # we will cycle over them until the budget runs out
    length_idx = 0

        n = lengths[length_idx % len(lengths)]
        length_idx += 1

        # initialise a random vector (keep the mass around 1)
        f = random_state.rand(n)
        f = f / f.sum()               # ℓ₁‑normalise; absolute scale does not matter
        # occasionally start from the best solution discovered so far
        if best_seq is not None and best_seq.size == n and random_state.rand() < 0.3:
            f = best_seq.copy()
            f = f / f.sum()

        # local best for this length
        local_best_f = f.copy()
        local_best_val = fast_score(f)

        # annealing temperature for accepting deteriorating moves
        temp = 0.5

        # inner iteration – LP + line‑search + occasional random jitter
        while time.time() < deadline:
            # current convolution bound
            conv = np.convolve(f, f)
            M = conv.max()

            # solve the LP that looks for a direction with larger total mass
            g = lp_g(f, M)
            candidates = []

            if g is not None:
                # try several mixing factors; 1.0 means replace f by the re‑scaled g
                for t in (0.1, 0.25, 0.5, 0.75, 1.0):
                    cand = mix(f, g, t)
                    candidates.append(cand)
            else:
                # LP failed – fallback to a small random perturbation
                cand = f.copy()
                idx = random_state.randint(n)
                cand[idx] = max(0.0, cand[idx] + random_state.randn() * 0.05)
                candidates.append(cand)

            # occasionally add a pure random kick (helps escape plateaus)
            if random_state.rand() < 0.05:
                jitter = random_state.rand(n) * 0.02
                cand = np.maximum(0.0, f + jitter)
                candidates.append(cand / cand.sum())

            # evaluate all candidates and pick the best
            improved = False
            for cand in candidates:
                val = fast_score(cand)
                # accept if strictly better or with a probability that decays with temperature
                if val < local_best_val or random_state.rand() < np.exp((local_best_val - val) / max(temp, 1e-8)):
                    f = cand
                    local_best_val = val
                    local_best_f = cand.copy()
                    improved = True
                    # update global best as well
                    if val < best_val:
                        best_val = val
                        best_seq = cand.copy()
                        print(f"[{time.time() - now:.1f}s] New global best: {best_val:.6f} (len={n})")

            # decrease temperature slowly
            temp *= 0.9995

            # if no improvement for a while, restart with a fresh random vector
            if not improved and random_state.rand() < 0.001:
                f = random_state.rand(n)
                f = f / f.sum()
                temp = 0.5

        # end of inner loop for this length
        # keep the best vector of this length for possible future restarts
        # (already stored in best_seq)

    # Return the best sequence found (convert to plain Python list)
    # ------------------------------------------------------------------

        # safety fallback – a constant vector of modest size
        best_seq = np.ones(256, dtype=float)

        n = a.size

        if s < 1e-12:

        # FFT length – next power of two for speed
        L = 1 << ((2 * n - 1).bit_length())

        maxb = np.max(conv)

    # ------------------------------------------------------------------
