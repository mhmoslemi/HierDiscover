"""
Propose a candidate sequence that beats the uniform bound (score 2.0).

The sequence a_i = C_i / 4**i = binom(2*i, i) / 4**i satisfies
a_{i+1} = a_i * (2*i+1) / (2*i+2) and decays like 1/√i.
Its discrete convolution is almost flat:
for every k ≤ n‑1 we have (a * a)[k] ≈ 1,
so the evaluation 2*n*max(conv)/sum(a)^2 approaches π/2 ≈ 1.57,
well below the trivial uniform score 2.0.

The code builds this “inverse‑square‑root’’ sequence for the same length as the
previous best construction (or a sensible default), evaluates it once, and then
spends the remaining time doing a very cheap random local search (multiplicative
perturbations of single entries) that keeps only improving moves.
Because the evaluation is scale‑invariant, no additional normalisation is
required.

The function `propose_candidate` is the entry point that will be called by the
runner.
"""

import time
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Return a list a_0,…,a_{n‑1} with a_i = binom(2*i,i)/4**i.
    The recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) avoids large intermediate numbers."""
    seq = [1.0]                     # a_0 = 1
    for i in range(1, n):
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq

    """
    Build the inverse‑square‑root sequence (theoretical near‑optimal shape)
    and perform a lightweight stochastic refinement while time permits.
    Returns a list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Determine the length of the construction.
    #    Use the length of the previously found sequence if available,
    #    otherwise fall back to a modest default.
    # ----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000

    # ----------------------------------------------------------------------
    base_seq = generate_inverse_sqrt_sequence(n)

    # Evaluate once – this also applies the security checks inside evaluate_sequence.
    best_seq = base_seq
    try:
        best_score = evaluate_sequence(best_seq)   # type: ignore[name-defined]
    except Exception:
        # Should never happen – fallback to a uniform sequence.
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
