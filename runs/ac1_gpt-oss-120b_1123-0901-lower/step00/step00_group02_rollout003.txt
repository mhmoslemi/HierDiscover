def simplex_projection(v):
    """Project onto the probability simplex."""
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑based simplex projection optimizer.

The evaluation function is homogeneous, therefore we may restrict the
candidate to a probability simplex (non‑negative entries with sum = 1).
For a fixed length `n` the objective reduces to `2*n*max(conv)`.
We minimise `max(conv)` by smoothing it with a soft‑max
(`log‑sum‑exp`) and performing projected gradient descent.
The gradient of the smoothed objective can be computed efficiently with
a single convolution:
    ∂/∂a_i  Σ_k w_k (a*a)_k = 2· Σ_k w_k a_{k‑i}
which is exactly a convolution of the weight vector `w` with the
reversed sequence `a`.  After each gradient step the vector is projected
onto the simplex using the O(n) algorithm of Duchi et al.

Multiple random restarts and several lengths are tried while respecting
the time budget.  The best sequence (lowest evaluation score) is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # due to numerical issues the sum may be slightly off; renormalise
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smoothed max (soft‑max) of the convolution
# ----------------------------------------------------------------------
def grad_softmax_convolution(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a probability vector `p` (length n) and soft‑max weights `w`
    (length 2n‑1), compute ∇_p Σ_k w_k (p*p)_k.
    The formula reduces to 2 * (w ⋆ p)[i] where ⋆ is the discrete
    cross‑correlation.  It can be obtained via a single convolution.
    """
    n = p.shape[0]
    # cross‑correlation via convolution with the reversed vector
    rev = p[::-1]
    conv_full = np.convolve(w, rev)          # length = 3n‑2
    start = n - 1                            # index that corresponds to shift 0
    grad = 2.0 * conv_full[start:start + n]  # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    `evaluate_sequence`.  The algorithm performs several
    projected‑gradient runs on different lengths, each starting from a
    random Dirichlet distribution.  A soft‑max approximation of the
    infinity‑norm of the auto‑convolution is minimised; the final best
    sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # lengths to try – larger lengths give a tighter theoretical bound
    lengths = [50, 100, 150, 200, 250, 300]

    best_seq = None
    best_score = float("inf")
    total_iters = 0

    # ---------- main outer loop over lengths ----------
    for n in lengths:
        if time.time() > deadline:
            break

        # allocate a modest amount of work per length
        restarts = 5
        max_iter_per_restart = 800   # can be cut short by the time budget

        for restart in range(restarts):
            if time.time() > deadline:
                break

            # start from a random point on the probability simplex
            p = np.random.dirichlet(np.ones(n))

            # parameters for the smooth max
            alpha = 5.0          # start with a gentle smoothing
            lr = 0.12            # base learning rate

            # ---------- inner gradient descent ----------
            for it in range(max_iter_per_restart):
                if time.time() > deadline:
                    break

                # convolution a*a
                conv = np.convolve(p, p)                 # length 2n‑1
                # soft‑max weights (numerically stable)
                scaled = alpha * conv
                shift = np.max(scaled)
                exp_terms = np.exp(scaled - shift)
                w = exp_terms / exp_terms.sum()           # shape (2n‑1,)

                # gradient of the smoothed objective
                grad = grad_softmax_convolution(p, w)

                # diminishing step size
                eta = lr / np.sqrt(it + 1.0)
                p = p - eta * grad

                # project back onto the simplex
                p = simplex_projection(p)

                # slowly increase the sharpening parameter
                if (it + 1) % 120 == 0:
                    alpha = min(40.0, alpha * 1.4)

                # occasional evaluation (every 40 steps)
                if (it + 1) % 40 == 0:
                    seq = p.tolist()
                    try:
                        score = evaluate_sequence(seq)
                    except Exception:
                        score = float("inf")
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        print(
                            f"[{time.time():.1f}] n={n} restart={restart} "
                            f"it={it+1}  score={score:.6f}"
                        )
                    # early stop if we already beat the target
                    if best_score < 1.503:
                        print("Target achieved – terminating early.")
                        return best_seq

            # final evaluation for this restart
            seq = p.tolist()
            try:
                score = evaluate_sequence(seq)
            except Exception:
                score = float("inf")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(
                    f"[{time.time():.1f}] n={n} restart={restart} final "
                    f"score={score:.6f}"
                )
            if best_score < 1.503:
                print("Target achieved – terminating early.")
                return best_seq

    # Fallback: if nothing was found (highly unlikely) return a uniform vector
    if best_seq is None:
        n = 100
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence, score={best_score:.6f}")

    print(f"Best score after search: {best_score:.6f}")
    return best_seq
