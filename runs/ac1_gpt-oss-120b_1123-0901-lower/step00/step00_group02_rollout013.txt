"""
Propose a better height sequence using a simple gradient–descent optimizer.

The evaluation function is
    E(a) = 2 * n * max_k (a * a)[k] / (sum_i a_i)^2,
where * is the discrete convolution.
E is homogeneous, so we keep the total sum fixed (to sqrt(2n)) to avoid
numerical overflow and to make the step size independent of the scale.

For a given sequence a we locate the index k_max at which the convolution
attains its maximum.  Around this index the derivative of max_k (a*a)[k]
with respect to a_p is 2 * a_{k_max-p} (zero when the index is out of range).
Using the chain rule we obtain the exact gradient of E:

    ∂E/∂a_p =
        4 n / S^2 * a_{k_max-p}  -  4 n * M / S^3,

with S = sum_i a_i and M = max_k (a*a)[k].

A projected gradient step is performed:
    a ← a - η * grad,
negative entries are clipped to zero,
and the vector is renormalised to the fixed sum.
The step size η decays slowly with the iteration counter.
Every few hundred iterations a tiny random perturbation is added
to escape shallow local minima.

The algorithm starts from the best known construction *height_sequence_1*
(if available) and runs until the allotted time expires.  The best
sequence seen during the run is returned.
"""

import time
import math
import numpy as np
from scipy import optimize  # only used for a fallback LP (optional)

# ----------------------------------------------------------------------
# Helper: compute convolution max and its index
def _conv_max(a: np.ndarray):
    """Return (M, k_max) where M = max convolution of a with itself."""
    b = np.convolve(a, a)
    M = b.max()
    k_max = int(b.argmax())
    return M, k_max, b

# ----------------------------------------------------------------------
# Helper: single gradient descent step
def _gradient_step(a: np.ndarray, target_sum: float, lr: float):
    """
    Perform one projected gradient descent step on the evaluation cost.
    Returns the updated array.
    """
    n = a.shape[0]
    S = a.sum()
    if S == 0.0:
        return a.copy()

    M, k_max, _ = _conv_max(a)

    # Build reversed part a_{k_max-p}
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]

    # Gradient of the objective (derived in the docstring)
    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

    # Gradient step
    a_new = a - lr * grad

    # Projection onto the non‑negative orthant
    a_new = np.maximum(a_new, 0.0)

    # Re‑scale to keep the total sum fixed (scale invariance of the objective)
    sum_new = a_new.sum()
    if sum_new > 0.0:
        a_new = a_new * (target_sum / sum_new)
    else:
        # If everything collapsed to zero, fall back to the previous vector
        a_new = a.copy()

    # Clamp to the limits imposed by evaluate_sequence (0 ≤ x ≤ 1000)
    np.clip(a_new, 0.0, 1000.0, out=a_new)

    return a_new

# Main entry point required by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    supplied `evaluate_sequence` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total wall‑clock time (seconds) available for the search.
    kwargs : dict
        Unused, kept for compatibility.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Initialise a starting sequence
    try:
        # Use the global sequence if it exists (the SOTA construction)
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        # Fallback: a uniform vector of length 2100
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)                # the “normalisation” used by the paper
    a = base.copy()
    # Normalise to the fixed sum
    a = a * (target_sum / a.sum())

    # Keep the best solution seen so far
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Gradient descent parameters
    iteration = 0
    lr0 = 0.5                                      # initial learning rate
    decay = 1e-4                                   # controls how fast the step shrinks

    # Main optimisation loop
    while time.time() < deadline:
        # Adaptive learning rate
        lr = lr0 / (1.0 + decay * iteration)

        # Gradient step
        a_new = _gradient_step(a, target_sum, lr)

        # Occasionally add a tiny random jitter to avoid stagnation
        if iteration % 1000 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = a_new * (target_sum / a_new.sum())

        # Evaluate the candidate
        try:
            cur_score = evaluate_sequence(a_new.tolist())
        except Exception:  # safety – should never happen
            cur_score = float("inf")

        # Keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[iter {iteration:06d}] new best score = {best_score:.6f}")

        # Accept the step if it improved the objective, otherwise keep the old vector
        # (a simple hill‑climbing scheme; the occasional jitter already adds exploration)
        if cur_score <= best_score:
            a = a_new
        else:
            # Occasionally accept a worse move to escape tiny plateaus
            if np.random.rand() < 0.005:
                a = a_new

        iteration += 1

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
