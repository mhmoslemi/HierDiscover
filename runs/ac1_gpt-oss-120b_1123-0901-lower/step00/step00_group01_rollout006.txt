def simplex_projection(v):
    # standard algorithm: sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

"""
Propose a step‑function (non‑negative list) that minimizes

    2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)

using a smooth‑max surrogate and projected gradient descent.
The algorithm works on the normalized simplex (sum = 1) – the
objective is scale invariant – and repeatedly pushes mass away from
indices that contribute most to the current maximal convolution entry.
A simple simplex projection keeps the iterates feasible.
The code is self‑contained, prints progress and stops as soon as the
target value (≈1.5) is reached or the time budget expires.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the closest point to v in the simplex {x >= 0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    ind = np.arange(len(v)) + 1.0
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative – return uniform simplex point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise (tiny errors may appear)
    w /= w.sum()
    return w
