def proj_simplex(v):
    """Project v onto probability simplex: nonnegative and sum =1."""
    # Standard algorithm: sort descending, find rho, compute theta.
    # Use numpy.
    n=len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def proj_simplex(v):
    """Projection of a vector v onto the probability simplex (sum=1, v_i>=0)."""
    # reference: https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def conv_fft(p):
    n = p.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    P = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(P * P, n=L)
    return conv[:2*n - 1]

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Gradient‑descent on the max‑convolution objective.
    
    The algorithm works on a probability vector p (sum=1).  The evaluation
    value is 2·n·max(conv(p,p)).  Starting from random points we move
    opposite the gradient of the maximum convolution entry and project
    back onto the simplex.  Multiple short restarts are performed until the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # leave a small safety margin

    # length of the sequence – use the length of the known construction if available
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 6000  # fallback
    print(f'Using sequence length n = {n}')

    # keep the best sequence seen so far (list of floats)
    best_seq = None
    best_score = np.inf

    # a quick reference: the uniform distribution gives score exactly 2
    uniform = np.full(n, 1.0 / n)
    uniform_score = 2.0
    best_seq = uniform.tolist()
    best_score = uniform_score
    print(f'Baseline uniform score = {best_score:.6f}')

    # small helper to compute score from probability vector
    def score_from_p(p):
        conv = conv_fft(p)
        maxc = conv.max()
        return 2.0 * n * maxc

    # main optimisation loop – many short restarts
    restart = 0
    while time.time() < deadline:
        restart += 1
        # initialise a random point on the simplex
        p = np.random.rand(n)
        p = p / p.sum()
        # optional tiny perturbation of the uniform start for the first restart
        if restart == 1 and best_seq is not None:
            p = np.array(best_seq)
            # add a little random noise and re‑project
            p = proj_simplex(p + 0.01 * np.random.randn(n))
        # store best score in this run
        local_best_score = np.inf
        local_best_p = None

        # gradient descent parameters
        max_iter = 2000                     # hard cap per restart
        eta0 = 0.5                          # initial step size
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            # compute convolution and the current score
            conv = conv_fft(p)
            maxc = conv.max()
            cur_score = 2.0 * n * maxc
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.copy()
                print(f'[{time.time()-start_time:6.1f}s] New global best {best_score:.6f}')
            # keep the best of this run (might be slightly worse than global best)
            if cur_score < local_best_score:
                local_best_score = cur_score
                local_best_p = p.copy()

            # stop early if we have already reached the target gap
            if best_score <= 1.503 + 1e-3:
                break

            # identify the index k where the maximum occurs
            k_star = int(conv.argmax())

            # gradient of maxC wrt p (positive direction).  We move opposite.
            idx = np.arange(n)
            j = k_star - idx
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(p)
            g[mask] = 2.0 * p[j[mask]]

            # simple back‑tracking line search
            eta = eta0 / np.sqrt(it + 1.0)
            accepted = False
            for _ in range(5):
                p_candidate = proj_simplex(p - eta * g)
                cand_score = score_from_p(p_candidate)
                if cand_score < cur_score:
                    p = p_candidate
                    accepted = True
                    break
                eta *= 0.5
            if not accepted:
                # if no improvement, perform a tiny random jitter to escape
                jitter = 0.001 * np.random.randn(n)
                p = proj_simplex(p + jitter)

        # after the inner loop, optionally retain the best of this restart
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_p.copy()
            print(f'[{time.time()-start_time:6.1f}s] Restart {restart} gave new best {best_score:.6f}')

    # ensure the result respects the positivity constraints (already true)
    # scale to a reasonable magnitude (optional – keep sum=1)
    result = [float(x) for x in best_seq]
    print(f'Finished. Best score = {best_score:.6f}')
    return result

import numpy as np
import time

def proj_simplex(v):
    """Project vector v onto the probability simplex."""
    # based on the algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative => return uniform zero?
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1   # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

"""
Gradient‑descent optimiser for the step‑function bound.

The evaluation of a sequence a is
    2·n·max(conv(a,a)) / (sum(a))² .
Because the expression is scale‑invariant we can enforce sum(a)=1.
The problem then reduces to minimising 2·n·max(conv(p,p)) over
probability vectors p (p_i≥0, Σp_i=1).

The algorithm works on a probability vector `p`:

* start from a random point on the simplex,
* repeatedly locate the index k* where the convolution is maximal,
* compute the gradient of max(conv) w.r.t. p (∂=2·p_{k*-i}),
* move a step opposite to the gradient,
* project back onto the simplex (so the sum stays 1 and all entries stay ≥0).

A simple back‑tracking line‑search guarantees descent.
If no progress is made for many steps the current iterate is re‑initialised
randomly (a cheap way to escape potential plateaus).  The best sequence
found during the allotted budget is returned.

The convolution is evaluated with an FFT (O(n log n)) to keep the inner
loop fast even for several thousand coefficients.  The final sequence
conforms to all security checks of `evaluate_sequence`.
"""

import numpy as np
import time


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v_i≥0)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return the uniform zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1          # zero‑based index of the last positive entry
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(p: np.ndarray) -> np.ndarray:
    """Convolution of a real vector with itself using an FFT."""
    n = p.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    P = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(P * P, n=L)
    return conv[: 2 * n - 1]


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()

    # Choose a length.  Use the known construction if it exists, otherwise a
    # default that is comfortably big for the search.
    # ------------------------------------------------------------------ #
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 8000                               # default length
    print(f"Sequence length n = {n}")

    # baseline (uniform) – score is exactly 2.0
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.tolist()
    best_score = 2.0
    print(f"Baseline uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------ #
    # Helper that returns the objective value for a probability vector.
    # ------------------------------------------------------------------ #
    def _objective(p_vec: np.ndarray) -> float:
        conv = _conv_fft(p_vec)
        return 2.0 * n * conv.max()   # because sum(p)=1

    # ------------------------------------------------------------------ #
    # Optimisation parameters.
    # ------------------------------------------------------------------ #
    stagnation_limit = 3000          # restart after this many non‑improving steps
    iter_no = 0
    stagnation = 0

    # start from a random point on the simplex
    p = np.random.rand(n)
    p = p / p.sum()

    while time.time() < deadline:
        cur_score = _objective(p)

        # update global best if improved
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = p.copy()
            print(f"[{time.time() - start_time:6.1f}s] New best score = {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # early exit once we have comfortably passed the target gap
        if best_score <= 1.503 + 1e-4:
            break

        # ------------------------------------------------------------------
        # Gradient step (move opposite to the gradient of the maximal conv entry)
        # ------------------------------------------------------------------
        conv = _conv_fft(p)
        k_star = int(conv.argmax())

        idx = np.arange(n)
        j = k_star - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(p)
        g[mask] = 2.0 * p[j[mask]]                     # ∂maxC/∂p_i

        # back‑tracking line search
        eta = 0.5 / np.sqrt(iter_no + 1.0)
        improved = False
        for _ in range(6):
            p_candidate = _proj_simplex(p - eta * g)
