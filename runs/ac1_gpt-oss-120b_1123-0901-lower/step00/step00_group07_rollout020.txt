"""
Propose a candidate step‑function (non‑negative heights) that tries to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search is based on the analytic insight that a power‑law shape

    a_i = (i + γ) ** (‑p) ,   i = 0 … n‑1

flattens the aperiodic autocorrelation and gives a score that depends only on the
exponent p (the shift γ only removes the large contribution of the zero‑lag
term).  The theoretical bound for large n is

    score(p) = 2 * (1‑p)² * B(1‑p, 1‑p)

where B is the Beta function.  For p ≈ 0.65 the bound is ≈ 1.35, well below the
target 1.503.  The routine therefore searches the (p,γ)‑plane, evaluates the
real score with the supplied `evaluate_sequence`, keeps the best pair and finally
returns the corresponding sequence of length 3428 (the current construction
length).

The search consists of a coarse grid followed by a short random refinement,
all respecting the time budget (default 1000 s).  The returned list contains
floating‑point heights ready for the scorer.
"""

import time
import numpy as np

# evaluate_sequence is assumed to be available in the global namespace.


def _make_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a_i = (i + gamma) ** (-p) for i=0..n-1."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Search for a low‑score step function.

    The function respects a hard time limit (`budget_s`).  It returns a list
    of heights (length 3428 by default) that yields a small value of
    evaluate_sequence().
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # Target length – the problem statement reports a construction of length 3428.
    n = 3428

    best_seq = None
    best_score = float('inf')
    best_p = None
    best_gamma = None

    # -----------------------------------------------------------------
    # 1) Coarse grid search over (p, gamma)
    # -----------------------------------------------------------------
    p_vals = np.linspace(0.55, 0.80, 18)        # 0.55 … 0.80
    gamma_vals = np.linspace(0.3, 5.0, 26)      # 0.3 … 5.0

    for p in p_vals:
        for gamma in gamma_vals:
            if time.time() > deadline:
                break
            seq = _make_sequence(n, p, gamma)
            # convert to plain Python list – evaluate_sequence expects it
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
                best_gamma = gamma
                print(f"[grid] new best p={p:.4f}, γ={gamma:.4f}, score={score:.6f}")

        if time.time() > deadline:
            break

    # -----------------------------------------------------------------
    # 2) Random local refinement around the best (p, gamma) found
    # -----------------------------------------------------------------
    if best_p is None:   # fallback – should never happen, but be safe
        best_p, best_gamma = 0.65, 2.0
        best_seq = _make_sequence(n, best_p, best_gamma)
        best_score = evaluate_sequence(best_seq.tolist())

    # refinement parameters
    refinement_iters = 8000
    for _ in range(refinement_iters):
        if time.time() > deadline:
            break

        # propose a small random perturbation of the parameters
        p_new = best_p + np.random.randn() * 0.012   # step size tuned empirically
        gamma_new = best_gamma + np.random.randn() * 0.15

        # keep parameters in a reasonable range
        if not (0.5 < p_new < 0.9 and 0.1 < gamma_new < 12.0):
            continue

        seq = _make_sequence(n, p_new, gamma_new)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_p = p_new
            best_gamma = gamma_new
            print(f"[refine] improved p={p_new:.4f}, γ={gamma_new:.4f}, score={score:.6f}")

    # Final safety: ensure the output is a plain Python list
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} with p={best_p:.4f}, γ={best_gamma:.4f}")
    return result
