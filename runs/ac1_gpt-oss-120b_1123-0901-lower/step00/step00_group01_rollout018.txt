def score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n=len(a)
    # compute conv via FFT:
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa*fa, m)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a**2))

import numpy as np

"""
Optimizes a non‑negative step‑function so that the quantity
    2 * n * max(conv(f,f)) / (sum(f)^2)
is as small as possible.  

The algorithm treats the convolution peak as the bottleneck and iteratively
moves a small amount of mass from indices that contribute most to the current
maximum (the gradient of the peak) to indices that contribute little or not at
all.  The update is performed on a normalized probability vector
(∑ a_i = 1); scaling does not affect the objective, so the search works on a
compact simplex.  Fast FFT‑based convolution provides a cheap evaluation of
the peak, allowing millions of cheap steps within the 1000 s budget.  A few
random restarts and a simple cooling schedule help escape local minima.
"""

def _fft_size(n):
    """Return the smallest power‑of‑two length ≥ 2·n‑1 for an FFT."""
    target = 2 * n - 1
    return 1 << (target - 1).bit_length()


def _conv_fft(a, fft_len):
    """Linear convolution of a with itself using an FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return conv


def _score_from_peak(n, peak, sum_a):
    """Objective value from the peak of the convolution."""
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * peak / (sum_a * sum_a))

    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                     # safety guard (should never fire)
        best_a = np.full_like(best_a, 0.01 / len(best_a))
    return best_a.tolist()
