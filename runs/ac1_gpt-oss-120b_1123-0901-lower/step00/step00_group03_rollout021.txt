def simplex_projection(v):
    # v is ndarray
    # Implementation from https://github.com/jaulinus/simplexproj
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.max(np.where(cond)[0])  # last index where condition true
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1000
    # start from random probabilities
    best_seq = None
    best_score = float('inf')

    def eval_seq(seq):
        # compute evaluation directly using evaluate_sequence for final check,
        # but also compute fast approx: 2 * n * max(conv(seq, seq)) / (sum(seq)**2)
        arr = np.array(seq, dtype=float)
        max_b = np.max(np.convolve(arr, arr))
        sum_a = np.sum(arr)
        if sum_a < 0.01:
            return np.inf
        return float(2 * len(arr) * max_b / (sum_a**2))

    def project_to_simplex(v):
        # Simplex projection: sum(v)=1, v>=0
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
        if len(rho) == 0:
            theta = 0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # local search routine
    def improve(p):
        # p is numpy array, sum to 1
        nonlocal best_seq, best_score
        it = 0
        while time.time() < deadline:
            q = np.convolve(p, p)
            kmax = int(np.argmax(q))
            # gradient (2*rev)
            rev = np.zeros_like(p)
            low = max(0, kmax - (len(p)-1))
            high = min(len(p)-1, kmax)
            idx = np.arange(low, high+1)
            rev[idx] = p[kmax - idx]
            grad = 2 * rev
            # step size schedule

            p_new = project_to_simplex(p_new)

                best_score = cur_score
                best_seq = p_new.copy()

"""
propose_candidate – binary‐set hill‑climbing search

The evaluation of a sequence a[0..n‑1] is

    score = 2·n·max_k (a * a)[k] / (sum a)^2

where “*” is the linear convolution.  Because the score is homogeneous,
the absolute scale of a does not matter – only the pattern of
non‑zero positions matters.  For a binary sequence (entries 0 or 1)
the sum equals the number of ones m and the convolution counts, for
each possible sum k, the number of ordered pairs (i,j) in the set with
i+j = k.  Hence

    score = 2·n·max_count / m²

with *max_count* the largest number of representations of a sum.
The goal is to pick a subset S⊂{0,…,n‑1} of size m that makes the
pair‑sum representation numbers as uniform as possible, i.e. minimise
max_count while keeping m reasonably large.

The algorithm explores several set sizes m.  For each m it performs a
local search:

* initialise S with m random indices,
* build the ordered‑pair count array `cnt` (length 2·n‑1),
* repeatedly try a random swap “remove i_in ∈ S, insert i_out ∉ S”.
  The swap is evaluated by updating `cnt` incrementally (O(m) work);
  the swap is kept only if it strictly reduces the current `max_count`
  (occasionally a worsening move is accepted to escape local minima).
* keep the best set seen for this m.

The best binary sequence across all tried m is returned.  All numbers
are 0.0 or 1.0, satisfying the non‑negativity requirement of
`evaluate_sequence`.  The search runs within the supplied time budget
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: simplex projection (not used for binary search but kept for
# possible future extensions)
def _project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Core search for a good binary set of given size m
def _search_fixed_m(n, m, deadline, rng):
    """
    Return a set S of size m that (approximately) minimises the maximum
    ordered‑pair sum count.  Also returns that maximum count.
    """
    best_S = None
    best_max = None

    # Theoretical lower bound for max count (average representation count)
    avg_cnt = m * m / (2 * n)

    # ------------------------------------------------------------------
    # Helper to initialise count array for a given set
    def _init_counts(S):
        cnt = np.zeros(2 * n - 1, dtype=np.int32)
        for i in S:
            for j in S:
                cnt[i + j] += 1
        return cnt

    # ------------------------------------------------------------------
    # Main loop – several random restarts until time runs out
    while time.time() < deadline:
        # Random start
        S = set(rng.choice(n, size=m, replace=False))
        cnt = _init_counts(S)
        cur_max = int(cnt.max())

        # Keep the best of this restart
        if best_max is None or cur_max < best_max:
            best_max = cur_max
            best_S = set(S)

        # Simple hill‑climbing with occasional restarts
        no_improve = 0
        # number of attempts before a forced restart
        max_no_improve = max(2000, 5 * m)

        while time.time() < deadline and no_improve < max_no_improve:
            # choose elements to swap
            i_in = rng.choice(list(S))
            i_out = rng.integers(0, n)
            if i_out in S:
                continue

            # --------------------------------------------------------------
            # Apply removal of i_in (update cnt in‑place)
            # we iterate over a snapshot because S will be mutated
            for j in list(S):
                s = i_in + j
                if j == i_in:
                    cnt[s] -= 1            # self‑pair contributes 1
                else:
                    cnt[s] -= 2            # ordered pair (i_in,j) and (j,i_in)

            S.remove(i_in)

            # --------------------------------------------------------------
            # Apply insertion of i_out
            for j in S:                     # S currently does NOT contain i_out
                cnt[i_out + j] += 2
            cnt[2 * i_out] += 1              # self‑pair of the new element
            S.add(i_out)

            new_max = int(cnt.max())

            # --------------------------------------------------------------
            # Acceptance test
            if new_max < cur_max:
                cur_max = new_max
                no_improve = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_S = set(S)
                # keep the move
            else:
                # revert the move (restore previous state)
                # remove i_out
                cnt[2 * i_out] -= 1
                for j in S:
                    if j == i_out:
                        continue
                    cnt[i_out + j] -= 2
                S.remove(i_out)

                # re‑insert i_in
                for j in S:
                    cnt[i_in + j] += 2
                cnt[2 * i_in] += 1
                S.add(i_in)

                no_improve += 1
                # occasional uphill acceptance to escape plateaus
                if rng.random() < 0.001:
                    cur_max = new_max
                    no_improve = 0

            # early exit if we already reached the theoretical target
            # (max count not much larger than the average)
            if cur_max <= avg_cnt * 1.5:
                return best_S, best_max

        # restart automatically after many non‑improving attempts
        # (the outer while loop will start a new random set)

    return best_S, best_max

    """

    a low value of `evaluate_sequence`.  The method builds binary sequences
    of various densities and improves them by a customized hill‑climbing
    that directly minimises the maximal ordered‑pair sum count.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2   # leave a tiny safety margin

    # Length of the problem – use the length of the previously found

    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, list):

        n = 1108

    target_score = 1.503

    m_candidates = [150, 180, 200, 220, 250, 300, 350, 400]
