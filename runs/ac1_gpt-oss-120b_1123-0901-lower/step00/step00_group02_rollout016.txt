"""
propose_candidate
=================

A stand‑alone optimisation routine that searches for a step‑function (a
non‑negative list of heights) minimising the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the ordinary (linear) convolution.  The algorithm
does **not** use the reference linear‑programming approach – instead it
optimises a surrogate objective (the variance of the convolution) by a
projected gradient descent, complemented by occasional random
perturbations and several random restarts.

The main ideas
--------------

* **Scale‑invariance** – the objective does not change by multiplying all
  entries by a constant, therefore the search works with a normalised
  sequence (`sum == 1`).  After every update the sequence is re‑normalised
  to keep the scale fixed.

* **Flatten the convolution** – a uniform convolution (the triangular
  shape obtained for a constant sequence) yields the worst possible
  bound (`2`).  Minimising the variance of the convolution pushes the
  values towards a flatter profile, which directly reduces the maximum
  entry and consequently the final score.

* **Projected gradient** – for a sequence `a` let

      conv = a * a               # np.convolve
      avg  = conv.mean()
      err  = conv - avg           # deviation from the mean

  The gradient of `sum(err**2)` w.r.t. `a` is

      ∇ = 4 * ( err  ⊛  a[::-1] )[n-1 : n-1+n]

  (where `⊛` is convolution).  Updating `a ← a - η·∇` and then
  projecting back onto the simplex (`a ≥ 0`, `sum(a)=1`) yields a
  monotone descent on the variance and usually also lowers the original
  objective.

* **Exploration** – the optimisation can get stuck at saddle points
  (the uniform vector is a stationary point of the original max‑convolution
  objective).  Random restarts, different initial shapes (random,
  exponential, linear) and occasional stochastic perturbations ensure
  sufficient exploration.

The function respects the 1000 s time limit, logs progress and finally
returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _compute_score(seq: np.ndarray) -> float:
    """Exact evaluation of the target objective for a given sequence."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _variance_gradient(a: np.ndarray) -> np.ndarray:
    """
    Gradient of the variance of the convolution w.r.t. the sequence ``a``.
    The factor 4 is absorbed into the step size later on.
    """
    conv = np.convolve(a, a)
    avg = conv.mean()
    err = conv - avg                     # deviation from the mean
    # 4 * (err ⊛ reverse(a)) restricted to the original indices
    grad_full = np.convolve(err, a[::-1])
    n = a.shape[0]
    grad = grad_full[n - 1 : n - 1 + n]   # length ``n``
    return 4.0 * grad


def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a non‑negative start vector of length ``n``."""
    if kind == "random":
        a = np.random.rand(n)
    elif kind == "exponential":
        # decay chosen close to 1 – the exact value is not critical
        r = 0.9995
        a = np.array([r ** i for i in range(n)], dtype=float)
    elif kind == "linear":
        # decreasing linear ramp
        a = np.arange(n, 0, -1, dtype=float)
    elif kind == "previous" and "height_sequence_1" in globals():
        # use the supplied best construction (trim / pad as needed)
        src = np.asarray(height_sequence_1, dtype=float)
        if src.size >= n:
            a = src[:n].copy()
        else:
            a = np.concatenate([src, np.zeros(n - src.size, dtype=float)])
    else:
        # fallback – uniform vector
        a = np.ones(n, dtype=float)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.random.rand(n)
        s = a.sum()
    return a / s  # normalise to sum == 1


# ----------------------------------------------------------------------
# Core optimiser (projected gradient on variance)
# ----------------------------------------------------------------------
def _optimize_start(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Perform projected gradient descent starting from ``a0`` until ``deadline``.
    Returns the best sequence found from this run and its score.
    """
    a = a0.copy()
    best_local = a.copy()
    best_score = _compute_score(best_local)

    n = a.shape[0]
    iteration = 0
    stagnation = 0
    # initial learning rate – will be tapered
    base_lr = 0.2

    while time.time() < deadline and iteration < 5000:
        # current convolution statistics
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        M = conv[max_idx]
        sum_a = a.sum()
        curr_score = 2.0 * n * M / (sum_a * sum_a)

        # gradient of the variance (a smoother surrogate)
        grad = _variance_gradient(a)

        # simple diminishing step size
        lr = base_lr / (1.0 + iteration * 0.01)

        # projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new == 0.0:
            # restart from a small random perturbation
            a_new = np.random.rand(n)
            a_new /= a_new.sum()
        else:
            a_new /= s_new  # re‑normalise

        new_score = _compute_score(a_new)

        # accept only improving moves (strict descent)
        if new_score < curr_score:
            a = a_new
            stagnation = 0
            if new_score < best_score:
                best_score, best_local = new_score, a_new.copy()
        else:
            stagnation += 1
            # every few hundred non‑improving steps inject a tiny random kick
            if stagnation % 250 == 0:
                kick = a.copy()
                idx = np.random.randint(n)
                kick[idx] = max(0.0, kick[idx] + np.random.randn() * 0.02)
                kick /= kick.sum()
                kick_score = _compute_score(kick)
                if kick_score < curr_score:
                    a = kick
                    if kick_score < best_score:
                        best_score, best_local = kick_score, kick.copy()
        iteration += 1

    return best_local, best_score

    np.random.seed(seed)
    start_time = time.time()

    # Length to work with – use the length of the supplied construction if

    try:

    lengths = [base_len, max(500, base_len // 2), min(5000, base_len * 2)]
    lengths = sorted(set(lengths))
