"""
Propose a low‑value step‑function using an LP‑based iterative refinement.

Algorithm
---------
1. Choose a moderate length `n` (200‑800) – large enough to allow improvement
   but small enough that a linear program can be solved many times within the
   budget.
2. Initialise a few diverse candidate sequences:
      * uniform (the trivial solution, value = 2)
      * linearly decreasing
      * exponential decay
      * random positive vector
   Every candidate is normalised so that   Σ a_i = sqrt(2·n)   (the
   evaluation of `evaluate_sequence` then equals the maximal convolution entry).
3. For each candidate run an inner loop:
      * Compute the current maximal convolution value `M = max(conv(a,a))`.
      * Solve the LP

            maximise   Σ b_j
            subject to  conv(a, b) ≤ M          (2·n‑1 linear constraints)
                        b_j ≥ 0

        The LP asks for a non‑negative vector `b` that keeps the same convolution
        bound but has the largest possible ℓ₁‑norm.
      * If the LP succeeds, scale the solution to the required ℓ₁‑norm
        `sqrt(2·n)`:
                b̂ = b · sqrt(2·n) / Σb .
        Because the scaling factor is < 1, the new maximal convolution
        `M̂ = max(conv(b̂, b̂))` is strictly smaller than `M`.
        Accept `b̂` as the new current sequence.
      * If the LP fails (e.g. the problem is unbounded) perform a tiny random
        perturbation of the current sequence to escape a degenerate corner.
      * Stop the inner loop when no improvement is observed or the time budget
        runs out.

4. Keep the best sequence seen across all random starts and lengths.
5. Return the best sequence as a plain Python list of floats.

The method builds on the published “LP‑direction” idea but removes the small
step‑size (`t = 0.01`) used in the reference implementation – after each LP
solve we fully rescale the obtained vector, which yields much larger and
typically monotone improvements.  Random restarts and several simple families
of initial sequences help avoid local minima.

The whole routine respects the 1000 s wall‑time limit, prints progress, and
does not depend on any external files or network access.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _convolution_max(a: np.ndarray) -> float:
    """
    Return max_{k} (a * a)[k] using FFT‑based convolution (O(n log n)).
    The sequence length is typically a few hundred, so this is very fast.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _normalise_to_target_sum(a: np.ndarray) -> np.ndarray:
    """Rescale a non‑negative vector so that its ℓ₁‑norm equals sqrt(2·n)."""
    n = a.size
    target = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        # fall back to uniform
        return np.full(n, target / n, dtype=float)
    return a * (target / s)


def _build_lp_constraints(f: np.ndarray, rhs: float):
    """
    Build `A_ub` and `b_ub` for the LP:
        conv(f, b) <= rhs,  b >= 0
    Returns (A_ub, b_ub) as dense `np.ndarray`s.
    """
    n = f.size
    n2 = 2 * n - 1                     # length of a * a
    # Allocate matrix for the convolution constraints
    A_conv = np.zeros((n2, n), dtype=float)
    for k in range(n2):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # rows correspond to Σ_i f_i * b_{k-i}
        # set A_conv[k, j] = f_i where j = k - i
        for i in range(i_low, i_high + 1):
            j = k - i
            A_conv[k, j] = f[i]
    # Non‑negativity constraints  -I·b ≤ 0
    A_nonneg = -np.eye(n, dtype=float)
    b_nonneg = np.zeros(n, dtype=float)

    A_ub = np.vstack([A_conv, A_nonneg])
    b_ub = np.concatenate([np.full(n2, rhs, dtype=float), b_nonneg])
    return A_ub, b_ub


def _solve_lp_for_direction(f: np.ndarray, max_conv: float):
    """
    Solve the LP that maximises Σ b_j under the convolution bound.
    Returns the optimal vector `b` (numpy array) or None on failure.
    """
    n = f.size
    A_ub, b_ub = _build_lp_constraints(f, max_conv)
    c = -np.ones(n, dtype=float)          # maximise Σ b ⇒ minimise -Σ b
    # Use the modern HiGHS solver; give a modest time limit per solve.
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None


def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random positive vector and normalise it."""
    vec = rng.random(n)
    return _normalise_to_target_sum(vec)


def _linear_decreasing_initial(n: int) -> np.ndarray:
    """Linearly decreasing from 1 to a small positive value, then normalised."""
    raw = np.linspace(1.0, 0.0, n, dtype=float)
    return _normalise_to_target_sum(raw)


def _exponential_initial(n: int, alpha: float = 0.01) -> np.ndarray:
    """Exponential decay a_i = exp(-alpha * i), normalised."""
    raw = np.exp(-alpha * np.arange(n, dtype=float))
    return _normalise_to_target_sum(raw)


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Wall‑clock seconds available for the whole search.
    **kwargs : dict
        Ignored (present for compatibility).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # keep a safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Try several lengths; smaller n means cheaper LP solves.
    # The objective after normalisation equals the maximal convolution entry,
    # so length does not influence the bound directly.
    # ------------------------------------------------------------------
    lengths = [200, 400, 600]        # tuneable: still fast enough
    for n in lengths:
        # ----------------------------------------------------------------
        # Generate a few diverse starting vectors for this length.
        # ----------------------------------------------------------------
        starters = []

        # 1) uniform (the known 2‑value bound)
        uniform = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)
        starters.append(uniform)

        # 2) linearly decreasing
        starters.append(_linear_decreasing_initial(n))

        # 3) exponential (a couple of decay rates)
        starters.append(_exponential_initial(n, alpha=0.005))
        starters.append(_exponential_initial(n, alpha=0.02))

        # 4) random
        for _ in range(2):
            starters.append(_random_initial(n, rng))

        # --------------------------------------------------------------
        # Run a short optimisation loop for each starter.
        # --------------------------------------------------------------
        for start_idx, a in enumerate(starters):
            # Ensure the vector is non‑negative and has the proper ℓ₁‑norm.
            a = np.maximum(a, 0.0)
            a = _normalise_to_target_sum(a)

            current = a.copy()
            cur_val = _convolution_max(current)

            # Keep the best seen for the current length.
            if cur_val < best_val:
                best_val = cur_val
                best_seq = current.copy()
                print(
                    f"[Init] n={n}, starter={start_idx}, val={best_val:.6f}"
                )

            # Inner optimisation loop: LP → rescale → repeat.
            inner_iter = 0
            while time.time() < deadline and inner_iter < 50:
                M = _convolution_max(current)

                # Solve LP to increase the ℓ₁‑norm under the same convolution bound.
                b = _solve_lp_for_direction(current, M)
                if b is None:
                    # LP failed (e.g. unbounded). Apply a tiny random jitter.
                    idx = rng.integers(0, n)
                    perturb = (rng.random() - 0.5) * 0.02 * current[idx]
                    current[idx] = max(0.0, current[idx] + perturb)
                    current = _normalise_to_target_sum(current)
                    inner_iter += 1
                    continue

                # Scale the LP solution to the required sum.
                b_scaled = _normalise_to_target_sum(b)

                # Clip tiny negatives that may appear due to numerical noise.
                b_scaled = np.maximum(b_scaled, 0.0)

                # New maximal convolution after scaling.
                new_M = _convolution_max(b_scaled)

                # Accept improvement if we have strictly decreased the peak.
                if new_M + 1e-9 < M:
                    current = b_scaled
                    if new_M < best_val:
                        best_val = new_M
                        best_seq = current.copy()
                        print(
                            f"[Improved] n={n}, starter={start_idx}, "
                            f"iter={inner_iter}, val={best_val:.6f}"
                        )
                else:
                    # No improvement – break out of the inner loop.
                    break

                inner_iter += 1

            # End of inner loop for this starter.
            if time.time() >= deadline:
                break

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # If we could not improve beyond the uniform bound, fall back to the
    # provided height_sequence_1 (if it exists) – it is guaranteed to be
    # a valid sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        try:
            # height_sequence_1 is supplied in the execution environment.
            best_seq = np.array(height_sequence_1, dtype=float)
        except Exception:
            # As a last resort, return a uniform sequence of modest size.
            n = 200
            best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    # Final sanity checks – enforce constraints that evaluate_sequence also enforces.
    best_seq = np.maximum(best_seq, 0.0)                 # non‑negative
    best_seq = np.minimum(best_seq, 1000.0)              # upper clip (problem spec)
    best_seq = _normalise_to_target_sum(best_seq)       # exact ℓ₁‑norm

    final_val = evaluate_sequence(best_seq.tolist())
    print(f"[Result] final evaluation = {final_val:.6f} (n={best_seq.size})")
    return best_seq.tolist()
