def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = list(np.random.dirichlet(np.ones(200))) # sum=1
    best_score = evaluate_sequence(best_seq)

    seq = best_seq.copy()
    n = len(seq)
    S = sum(seq)

    while time.time() < deadline:
        # Compute convolution and max
        conv = np.convolve(seq, seq)
        M = conv.max()
        k = conv.argmax()
        # Compute gradient gsub for M
        grad = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = seq[i] + seq[j]  # includes double for i=j
        # Compute direction that reduces M: move opposite to grad, but also keep sum constant
        # Choose step size relative to M
        step = 0.01  # tune
        # Proposed new seq: subtract step*grad from all, then renorm
        new_seq = np.array(seq) - step * grad
        # Ensure nonnegative
        new_seq = np.maximum(new_seq, 0)
        # Renorm to sum S
        total = new_seq.sum()
        if total > 0:
            new_seq = new_seq * (S / total)
        else:
            new_seq = np.array(seq)
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_seq = new_seq.tolist()
            best_score = new_score
        if new_score < evaluate_sequence(seq):
            seq = new_seq.tolist()
        # occasional random shuffle to escape saddle
        if np.random.rand() < 0.01:
            # random small perturb
            idx = np.random.randint(n)
            seq[idx] = max(0, seq[idx] + np.random.randn() * 0.01)
            total = sum(seq)
            if total > 0:
                seq = [x * (S / total) for x in seq]

    return best_seq

"""
propose_candidate
-----------------

A fast heuristic for minimizing the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm follows the “direction‑search + mixing’’ idea that was
used in the reference implementation, but with a few practical
enhancements:

* the sequence length is deliberately kept modest (≈200–300) – a dense
  LP with O(n²) constraints is cheap and still gives a strong improvement
  over the naïve uniform distribution;
* the linear program is solved with SciPy’s high‑performance HiGHS
  solver; only the convolution constraints are required – non‑negativity
  is enforced via variable bounds;
* after the LP step the new direction is mixed with the current
  sequence using a *large* step size (0.5 by default).  The step size is
  reduced if the mixed sequence does not improve the score;
* simple random perturbations are used as a fallback when the LP
  fails or gets stuck, and a full random restart is performed every
  few seconds to avoid local minima;
* the best sequence seen so far is kept and returned at the end.

The method runs for the whole allotted time budget (default 1000 s) and
produces a sequence with an evaluation value usually well below the
uniform bound of 2.0 – often approaching the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: build the convolution constraints matrix for a given "f".
# The constraint  f * g <= M  (entry‑wise) is linear in g and can be
# written as   A @ g <= M·1 .
def _build_conv_constraints(f):
    """Return matrix A (shape (2*len(f)-1, len(f))) such that
       A @ g = np.convolve(f, g) ."""
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # each row k contains the coefficients f[i] where i + j = k and
    # j is the column index.
    for i in range(n):
        # contribution of f[i] appears in rows k = i .. i + n - 1,
        # at column j = k - i.
        A[i:i + n, i] = f[: rows - i]  # slice fits automatically
    return A


# ----------------------------------------------------------------------
def _solve_lp_direction(f, M, time_limit=5.0):
    """
    Solve the LP:
        maximise   sum(g)
        subject to conv(f, g) <= M   (element‑wise)
                  g >= 0
    Returns the optimal g (as a NumPy array) or None if the solver fails.
    """
    n = len(f)
    A = _build_conv_constraints(f)
    b = np.full(A.shape[0], M, dtype=np.float64)

    # objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # variable bounds enforce g >= 0
    bounds = [(0.0, None)] * n

    # HiGHS solver (default for scipy.linprog) respects 'time_limit' in the
    # options dictionary.
    opts = {"disp": False, "time_limit": time_limit}
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options=opts)

    if res.success:
        return res.x
    return None

            cur = np.random.dirichlet(np.ones(n))
            cur = cur / cur.sum()
            step_size = 0.5
            continue

    # ----------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    print(f"Finished – best score {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
