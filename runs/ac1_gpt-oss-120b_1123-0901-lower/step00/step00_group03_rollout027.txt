def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5 # small buffer
    best_seq = None
    best_score = float('inf')
    # optionally start from given height_sequence_1 if exists (global).
    try:
        init_seq = list(height_sequence_1)
        score = evaluate_sequence(init_seq)
        best_seq = init_seq
        best_score = score
        print("Initial known sequence score", score)
    except NameError:
        pass

    # Define helper to generate shape
    def generate_sequence(n):
        typ = np.random.choice(['gaussian', 'poly', 'mix'])
        center = (n-1)/2.0
        if typ == 'gaussian':
            sigma = np.random.uniform(0.3, 0.9) * n
            arr = np.exp(-0.5 * ((np.arange(n)-center)/sigma)**2)
        elif typ == 'poly':
            p = np.random.uniform(0.5, 5.0)
            offset = np.random.uniform(0.0, 0.2)
            arr = (np.abs(np.arange(n)-center)+offset) ** p
        else: # mix
            # start with gaussian and add small random spikes
            sigma = np.random.uniform(0.4, 0.8) * n
            arr = np.exp(-0.5 * ((np.arange(n)-center)/sigma)**2)
            # add random spikes
            for _ in range(np.random.randint(1,5)):
                pos = np.random.randint(n)
                val = np.random.rand() * arr.max()
                arr[pos] += val
        # Normalize sum to sqrt(2n) (optional)
        sum_a = arr.sum()
        if sum_a < 1e-6:
            arr += 1.0
            sum_a = arr.sum()
        scaling = np.sqrt(2*n) / sum_a
        arr = arr * scaling
        return arr.tolist()
    
    # Local refinement function
    def hill_climb(seq, steps=200):
        seq = np.array(seq)
        n = len(seq)
        cur_score = evaluate_sequence(seq.tolist())
        best_local_seq = seq.copy()
        best_local_score = cur_score
        for s in range(steps):
            i, j = np.random.randint(n), np.random.randint(n)
            if i==j: continue
            # transfer epsilon from i to j
            epsilon = np.random.uniform(-0.02, 0.02) * seq[i]  # relative change
            # ensure new values not negative
            delta = max(-seq[i], min(seq[j], epsilon))
            if abs(delta) < 1e-12:
                continue
            seq[i] -= delta
            seq[j] += delta
            new_score = evaluate_sequence(seq.tolist())
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_seq = seq.copy()
            if new_score < cur_score:
                cur_score = new_score
            else:
                # revert with probability
                if np.random.rand() > np.exp(-(new_score - cur_score)/0.01):
                    seq[i] += delta
                    seq[j] -= delta
                else:
                    cur_score = new_score
        return best_local_seq.tolist(), best_local_score

    # Main search loop
    while time.time() < deadline:
        n = np.random.randint(300, 2000)  # length random
        seq = generate_sequence(n)
        # quick evaluation: maybe compute shallow version
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best from random shape: score {best_score:.6f}, len {len(seq)}")
        # local refinement
        if score < best_score * 1.2:  # focus refinements on promising candidates
            refined_seq, refined_score = hill_climb(seq, steps=500)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"Improved by hill climb: score {best_score:.6f}, len {len(best_seq)}")
        # occasional LP direction improvement from current best
        if best_seq is not None and np.random.rand() < 0.1:
            # use existing get_good_direction_to_move_into if defined globally

                        best_score = new_score
                        best_seq = new_seq.copy()
                        print(f"LP step improvement: score {best_score:.6f}")
            except Exception:

    # Final check: ensure non-negative and at least one positive

        best_score = evaluate_sequence(best_seq)
    print("Best final score:", best_score)

def reduce_peak(seq, steps=200):
    seq = np.array(seq)
    n = len(seq)
    for _ in range(steps):
        b = np.convolve(seq, seq)
        p = int(np.argmax(b))
        # contributions to p: For each i s.t. 0<=i<n and 0<=p-i<n,
        # contribution = seq[i] * seq[p-i] (counted double except when i == p-i)

            if 0 <= j < n:
                prod = seq[i] * seq[j]
                if prod > 0:
                    candidates.append((prod, i))
        if not candidates:
            break
        # Get i with largest product
        i = max(candidates)[1]
        # Choose j' random index far from i and not overlapping p maybe: 
        # We want new index k where k does not pair with any large index to produce large product at p.
        # Choose random k where p - k outside [0,n-1] or seq[p-k] small.

    Search for a step-function sequence with low evaluation score.
    The algorithm randomly samples sequences of the form 
    a_i = (|i - (n-1)/2| + eps) ** p,
    normalizes them to have L1 = sqrt(2n), and keeps the best.
    Occasionally a short hill‑climbing phase (mass transfer between
    two positions) is applied to fine‑tune promising candidates.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    best_seq = None
    best_score = float('inf')
    # optionally start from known sequence
    try:
        best_seq = list(height_sequence_1)
        best_score = evaluate_sequence(best_seq)
        print(f"Start from known sequence, score={best_score:.6f}")
    except Exception:
        pass

    def make_power_shape(n, p):
        idx = np.arange(n)
        center = (n - 1) / 2.0
        dist = np.abs(idx - center) + 1e-9   # epsilon to avoid zero
        a = dist ** p
        # normalize L1 to sqrt(2*n)
        s = a.sum()
        if s == 0:
            a = np.ones_like(a)
            s = a.sum()
        scale = np.sqrt(2.0 * n) / s
        a = a * scale
        return a.tolist()

    def hill_climb(seq, iters=200):
        seq = np.array(seq)
        n = len(seq)
        cur_score = evaluate_sequence(seq.tolist())
        best_local = seq.copy()
        best_local_score = cur_score
        for _ in range(iters):
            i, j = np.random.randint(n), np.random.randint(n)
            if i == j:
                continue
            # take small fraction from i to j
            # magnitude proportional to current value at i
            delta = np.random.uniform(-0.05, 0.05) * seq[i]
            # ensure non‑negativity
            delta = max(-seq[i], delta)
            seq[i] -= delta
            seq[j] += delta
            # evaluate
            new_score = evaluate_sequence(seq.tolist())
            # accept if improvement, otherwise revert with probability
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = seq.copy()
                cur_score = new_score
            else:
                # metropolis acceptance
                if np.random.rand() < np.exp(-(new_score - cur_score) / 0.01):
                    cur_score = new_score
                else:
                    # revert
                    seq[i] += delta
                    seq[j] -= delta
        return best_local.tolist(), best_local_score

    while time.time() < deadline:
        # sample length and exponent
        n = np.random.randint(500, 2000)
        # bias exponent towards 1..3, but allow broader
        p = np.random.uniform(0.5, 4.0)
        seq = make_power_shape(n, p)
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"New best raw shape: n={n}, p={p:.3f}, score={score:.6f}")
        # occasionally refine
        if score < best_score * 1.5 and np.random.rand() < 0.3:
            refined_seq, refined_score = hill_climb(seq, iters=300)
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq.copy()
                print(f"Refined best: n={len(best_seq)}, p~{p:.3f}, score={best_score:.6f}")

    # If still nothing found, fall back to uniform
    if best_seq is None:
        n = 800
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)

    print("Finished search, best score =", best_score)
    return best_seq

def improve_by_gradient(seq, steps):
    seq = np.array(seq)
    n=len(seq)
    for t in range(steps):
        b = np.convolve(seq, seq)
        p = int(np.argmax(b))
        # compute counterpart values
        counterpart = np.zeros(n)
        # valid indices where p-i in [0,n-1]
        start = max(0, p-(n-1))
        end = min(n-1, p)
        # for i from start to end, counterpart[i] = seq[p-i]
        counterpart[start:end+1] = seq[p - np.arange(start,end+1)]
        w = counterpart # we could multiply by 2 ignoring constant
        # find i with max w (i_max) and min w (i_min)
        i_max = int(np.argmax(w))
        i_min = int(np.argmin(w))
        # ensure i_max != i_min and seq[i_max] > 0
        if i_max == i_min or seq[i_max] < 1e-12:
            break
        epsilon = min(seq[i_max]*0.05, 0.01) # small step
        seq[i_max] -= epsilon
        seq[i_min] += epsilon
        # optionally renormalize sum to sqrt(2n)

        # Actually sum stays same because we transfer epsilon, so sum unchanged.
    return seq.tolist()

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that yields a low
value of ``evaluate_sequence``.  The evaluation is

    2 * n * max(convolution(a, a)) / (sum(a) ** 2)

and, after normalising the L1‑norm to ``sqrt(2*n)`` (the normalisation used in
the reference code), the score coincides with the maximal convolution value.
Hence the problem is to make the self‑convolution as flat as possible.

The algorithm works in two phases:

1. **Random initialisation** – a length ``n`` (chosen uniformly in a
   reasonable interval) is drawn, a random positive vector is generated and
   normalised to ``sqrt(2*n)``.  Various sensible shapes (Gaussian, Hann,
   Hamming, Blackman, power‑cosine) are also tried – they are cheap to build
   and often already give scores below the previous bound of 2.

2. **Gradient‑descent style flattening** – for a given sequence the index
   ``p`` where the convolution attains its maximum is located.  The
   contribution of each coefficient ``a[i]`` to this peak is proportional to
   ``a[p‑i]`` (if the index is in range).  Mass is moved from an index with a
   large contribution to an index with a small contribution, keeping the
   total L1‑norm unchanged.  Repeating this simple “move‑mass” step lowers the
   peak of the convolution and therefore the score.

The search runs until the allotted time (``budget_s``) is exhausted, keeping
the best sequence found.  Simple hill‑climbing refinements are applied
occasionally, and the code prints progress messages for debugging.

The final returned list satisfies all security checks of
``evaluate_sequence`` (non‑negative, values ≪ 1000, and a non‑zero sum).
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions ------------------------------------------------------
# ----------------------------------------------------------------------


def _normalise_to_sqrt2n(seq):
    """Scale the sequence so that sum(seq) == sqrt(2*len(seq))."""
    seq = np.asarray(seq, dtype=float)
    n = len(seq)
    total = seq.sum()
    if total <= 0:
        return np.full_like(seq, np.sqrt(2.0 * n) / n)
    scale = np.sqrt(2.0 * n) / total
    seq *= scale
    return seq


def random_sequence(n):
    """Uniform random positive vector, normalised to sqrt(2n)."""
    a = np.random.rand(n) + 1e-6          # avoid exact zeros
    return _normalise_to_sqrt2n(a)


def gaussian_sequence(n, sigma_frac):
    """Discrete Gaussian, sigma = sigma_frac * n, normalised."""
    x = np.arange(n)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return _normalise_to_sqrt2n(a)


def hann_sequence(n):
    """Classic Hann window, normalised."""
    x = np.arange(n)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * x / (n - 1)))
    return _normalise_to_sqrt2n(a)


def hamming_sequence(n):
    """Classic Hamming window, normalised."""
    x = np.arange(n)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    return _normalise_to_sqrt2n(a)


def blackman_sequence(n):
    """Classic Blackman window, normalised."""
    x = np.arange(n)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
    return _normalise_to_sqrt2n(a)


def power_cos_sequence(n, power):
    """w[i] = sin(pi * i/(n-1)) ** power, normalised."""
    x = np.arange(n)
    a = np.sin(np.pi * x / (n - 1)) ** power
    return _normalise_to_sqrt2n(a)


def _convolve(a):
    """Fast linear convolution (real) using FFT."""
    # length needed for linear convolution
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return conv


def flatten_peak(seq, steps=200):
    """
    Simple greedy mass‑transfer that reduces the maximal convolution entry.

    The sequence is kept at the same L1 norm; only two entries are altered at
    each step.
    """
    a = np.asarray(seq, dtype=float)
    n = len(a)

    for _ in range(steps):
        # compute convolution and the index of the current peak
        conv = _convolve(a)
        p = int(np.argmax(conv))

        # build the “partner” vector: partner[i] = a[p-i] when the index is
        # legal, otherwise 0.  This encodes the contribution of a[i] to the
        # peak at position p.
        low = max(0, p - (n - 1))
        high = min(n - 1, p)
        idx = np.arange(low, high + 1)                # indices that have a partner
        partner = np.zeros(n)
        partner[low : high + 1] = a[p - idx]

        # choose a source (large contribution) and a sink (small contribution)
        i_src = int(np.argmax(partner))
        i_dst = int(np.argmin(partner))

        if i_src == i_dst or a[i_src] <= 1e-12:
            break

        # move a modest fraction of the source mass to the destination
        eps = min(a[i_src] * 0.04, 0.02)   # 4 % of source, capped at 0.02
        a[i_src] -= eps
        a[i_dst] += eps
        # the total sum is unchanged, no renormalisation needed

    return a.tolist()


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that gives a low value of
    ``evaluate_sequence``.  The routine spends at most ``budget_s`` seconds
    searching and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5          # a small safety margin

    # best candidate found so far
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try the previously known construction if it exists
    # ------------------------------------------------------------------
    try:
        cand = list(height_sequence_1)            # pylint: disable=E0602
        sc = evaluate_sequence(cand)
        best_seq, best_score = cand, sc
        print(f"[init] known sequence score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main random‑search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a length – larger lengths give more freedom but cost more
        n = np.random.randint(800, 2000)

        # pick a construction method
        method = np.random.choice(
            ["random", "gauss", "hann", "hamming", "blackman", "power_cos"]
        )

        if method == "random":
            seq = random_sequence(n)
        elif method == "gauss":
            sigma_frac = np.random.uniform(0.25, 0.45)   # 0.25‑0.45 × n
            seq = gaussian_sequence(n, sigma_frac)
        elif method == "hann":
            seq = hann_sequence(n)
        elif method == "hamming":
            seq = hamming_sequence(n)
        elif method == "blackman":
            seq = blackman_sequence(n)
        else:  # power_cos
            power = np.random.uniform(0.5, 4.0)
            seq = power_cos_sequence(n, power)

        # ----------------------------------------------------------------
        # Optional local refinement – gradient based flattening
        # ----------------------------------------------------------------
        try:
            # a few aggressive flattening steps
            seq = flatten_peak(seq, steps=300)

            score = evaluate_sequence(seq)
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[new] n={n:4d}, method={method:10s}, score={score:.6f}")

        # ----------------------------------------------------------------
        # Very occasional deeper refinement (more steps)
        # ----------------------------------------------------------------
        if np.random.rand() < 0.15:
            try:
                seq2 = flatten_peak(seq, steps=800)
                score2 = evaluate_sequence(seq2)
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.copy()
                    print(f"[ref] n={len(seq2):4d}, method={method:10s}, score={score2:.6f}")
            except Exception:
                pass

    # ----------------------------------------------------------------------
    # Fallback – uniform sequence (should never be reached, but safe)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = [np.sqrt(2.0 * n) / n] * n
        best_score = evaluate_sequence(best_seq)
        print("[fallback] uniform sequence score =", best_score)

    print("Search finished.  Best score =", best_score)
    return best_seq
