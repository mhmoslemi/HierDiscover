"""
Propose a candidate sequence for the step‑function problem.

The algorithm performs a stochastic projected‑gradient descent on the
objective

    E(a) = 2·n·max_k (a * a)[k] / (sum a)^2

which is equivalent (up to a scale) to minimizing the maximum entry of the
self‑convolution of a non‑negative vector a.
The gradient of the max‑convolution is obtained analytically; after each
gradient step the vector is projected back onto the probability simplex
(non‑negative with unit sum).  Random perturbations and occasional restarts
are added to escape possible plateaus.  The best sequence found before the
time budget expires is returned.
"""

import time
import numpy as np


def _score(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` (without the safety checks)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * maxb / (s * s)


def _subgradient_maxconv(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  w.r.t. a.
    Only the indices that achieve the current maximum are considered.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    for k in max_idx:
        # indices i that contribute to conv[k] (0 <= i < n, 0 <= k-i < n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)                # inclusive
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # derivative of a_i * a_j appears twice (i,j) and (j,i) except when i=j
        g[i] += 2.0 * a[j]

        # correct the double‑count on the diagonal term (i == j)
        if k % 2 == 0:
            ic = k // 2
            if i_start <= ic <= i_end:
                # we added 2*a[ic] above, but the true derivative is a[ic]
                g[ic] -= a[ic]
    return g


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
        { x | x_i >= 0, sum_i x_i = 1 }.
    Uses the classic O(n log n) algorithm based on sorting.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays.")
    n = v.shape[0]

    # shift by the mean to avoid numerical issues when the vector is already
    # nearly a probability distribution
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero (should not happen); return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] n={n:4d} it={it:6d} new best = {best_val:.6f}")

    # Main search loop over the chosen lengths -------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # --- initialise a random point on the simplex ----------------------------------------
        a = np.random.rand(n)
        a = a / a.sum()

        cur_val = _score(a)
        _update_best(a, cur_val, 0, n)

        it = 0
        no_improve = 0          # counter for stagnation detection

        while time.time() < deadline:
            # compute convolution and identify the (possibly multiple) max indices
            conv = np.convolve(a, a)
            maxb = conv.max()
            max_idx = np.where(conv >= maxb - 1e-12)[0]

            # sub‑gradient of the max‑convolution
            g = _subgradient_maxconv(a, max_idx)

            # step‑size schedule (decreases slowly with iterations)
            eta = 0.05 / (1.0 + it * 0.001)

            # gradient step, projection onto the simplex
            a_next = a - eta * g
            a_next = np.maximum(a_next, 0.0)
            a_next = _project_to_simplex(a_next)

            # evaluate
            next_val = _score(a_next)

            # accept if improvement, otherwise keep the old vector
            if next_val < cur_val:
                a = a_next
                cur_val = next_val
                no_improve = 0
                _update_best(a, cur_val, it, n)
            else:
                # small probability to accept a worse move (helps escaping)
                if np.random.rand() < 0.001:
                    a = a_next
                    cur_val = next_val
                no_improve += 1

            # occasional random perturbation to re‑inject diversity
            if np.random.rand() < 0.015:
                i, j = np.random.choice(n, 2, replace=False)
                # move a tiny amount from i to j (keeping positivity)
                delta = min(a[i], np.random.rand() * 0.02)
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a)   # guarantee sum = 1
                cur_val = _score(a)
                _update_best(a, cur_val, it, n)

            # if we have been stuck for a while, restart from a fresh random point
            if no_improve > 5000:
                a = np.random.rand(n)
                a = a / a.sum()
                cur_val = _score(a)
                no_improve = 0
                _update_best(a, cur_val, it, n)

            # early exit if we have already hit the target bound
            if best_val <= 1.503:
                print("Target reached – stopping search.")
                return best_seq.tolist()

            it += 1

    # -------------------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # -------------------------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple uniform vector (guaranteed to be feasible)
        best_seq = np.full(100, 1.0 / 100.0)
    return best_seq.tolist()
