def project_onto_simplex(v):
    # ensure nonnegative and sum=1
    # uses algorithm: sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    """Project vector v onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim !=1:
        raise ValueError
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑height sequence that (almost) minimizes the given
`evaluate_sequence` function.

The evaluation depends only on three quantities of the sequence `a`:
    * n = len(a)
    * A = sum(a)
    * M = max_k (a * a)[k]   (self‑convolution, linear convolution)

Since the score 2·n·M/A² is invariant under positive scaling,
we may fix A = 1 and minimise M.  M is the ℓ∞‑norm of the
self‑convolution, i.e. the worst anti‑concentration of the sum of two
independent draws from the distribution defined by `a`.

The algorithm proceeds in two stages:

1. **Warm‑start** – a one‑parameter exponential family
   `a_i ∝ exp(-c·i/n)` is scanned for the best constant `c`
   (≈1.2 gives a score ≈1.8, already better than the uniform
   construction which scores exactly 2.0).

2. **Projected sub‑gradient descent** on the quantity `M`.
   For the current `a` we compute the convolution,
   locate the index `k*` where it attains its maximum,
   and use the known sub‑gradient
       ∂M/∂a_i = 2·a_{k*‑i}   (if the index is in range, else 0).
   A step opposite to this gradient is taken and the resulting
   vector is projected back onto the probability simplex
   (non‑negative entries that sum to 1).  The simplex projection
   preserves the scaling invariance of the original objective.
   The step size decays like 1/√t; occasional random swaps are added
   to escape shallow plateaus.

The whole loop runs for the allotted time (default 1000 s) and the
best sequence found is returned as a plain list of `float`s.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex {x ≥ 0, sum x = 1}
def _project_simplex(v):
    """Project the 1‑D array v onto the simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("Only 1‑D arrays are supported")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) gives a low
    score for `evaluate_sequence`.  The implementation follows the
    description in the module docstring.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length ≤ the known construction (if any) – long enough
    #    to give the algorithm room, but not so large that each iteration
    #    becomes expensive.
    try:
        n = len(height_sequence_1)               # pre‑existing construction
    except Exception:
        n = 3000                                 # a reasonable default

    # ------------------------------------------------------------------
    # 2. Warm‑start: scan a small grid of exponent‑decay parameters.
    #    a_i ∝ exp(-c·i/n)  (c > 0).  The best among the grid seeds the
    #    subsequent optimisation.
    best_seq = None
    best_score = float('inf')

    # grid of c values (the useful region is roughly 0.5 … 3.0)
    for c in np.linspace(0.5, 3.0, 30):
        # raw exponential shape, un‑normalised
        raw = np.exp(-c * np.arange(n) / n)
        seq = raw / raw.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # early exit if time is running out
        if time.time() >= deadline:
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent on the ℓ∞‑norm of the self‑convolution.
    seq = best_seq.copy()
    it = 0
    step0 = 0.5    # initial step magnitude (will be scaled each iteration)

    while time.time() < deadline:
        it += 1

        # --- convolution and index of the current maximum ----------------
        conv = np.convolve(seq, seq)               # length 2n‑1
        k_max = int(np.argmax(conv))

        # --- sub‑gradient of the max entry --------------------------------
        # grad_i = 2 * seq_{k_max - i}  when the index is in range
        grad = np.zeros_like(seq)
        i_start = max(0, k_max - (n - 1))
        i_end   = min(k_max, n - 1)
        if i_start <= i_end:
            # slice of seq that participates in the max convolution term
            # reversed order yields the required a_{k_max‑i}
            grad[i_start:i_end + 1] = 2.0 * seq[i_start:i_end + 1][::-1]

        # --- step size – decays like 1/√t and normalised by the gradient norm
        #    (the extra normalisation prevents overly aggressive moves)
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = step0 / (np.sqrt(it) * grad_norm)

        # --- take a step opposite to the gradient and re‑project onto the simplex
        cand = seq - step * grad
        cand = _project_simplex(cand)

        # --- evaluate the candidate ----------------------------------------
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand                # keep the improvement
            # occasionally increase the step a little after a good move
            step0 *= 1.02
        else:
            # Simulated‑annealing style acceptance to escape tiny local minima
            # Temperature slowly cools down.
            temperature = 0.1 / np.sqrt(it + 1)
            if np.random.rand() < np.exp((best_score - cand_score) / max(temperature, 1e-12)):
                seq = cand

        # --- occasional random perturbation to inject diversity -------------
        if it % 250 == 0:
            # pick two random positions and transfer a small random mass
            i, j = np.random.randint(n, size=2)
            delta = min(seq[i], 0.01 * np.random.rand())
            if delta > 0:
                seq[i] -= delta
                seq[j] += delta
                seq = _project_simplex(seq)

        # --- gentle step‑size decay -----------------------------------------
        if it % 500 == 0:
            step0 *= 0.95

    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
