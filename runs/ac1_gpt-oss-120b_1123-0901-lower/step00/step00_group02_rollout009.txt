def solve_lp_for_direction(f_norm, M):
    n = len(f_norm)
    # build matrix A_ub for convolution constraints
    # using scipy.sparse
    rows = []
    cols = []
    data = []
    # for each i (col), assign entries to rows i..i+n-1
    for i in range(n):
        # rows indices for col i
        start = i
        end = i + n  # exclusive
        # length = n
        rows.extend(range(start, end))
        cols.extend([i] * n)
        data.extend(f_norm.tolist())
    import scipy.sparse as sp
    A_conv = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # nonnegativity constraints: -I * b <= 0
    A_neg = sp.identity(n) * -1
    A_ub = sp.vstack([A_conv, A_neg], format='csr')
    b_ub = np.hstack([np.full(2*n - 1, M), np.zeros(n)])
    # objective: maximize sum b => minimize -sum b
    c = -np.ones(n)
    # call linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
        return result.x
    else:
        return None
```

```
def solve_convolution_lp(f_seq, M):
    n = len(f_seq)
    import scipy.sparse as sp
    # build A as Toeplitz using sparse matrix
    # rows: for each i (col), we add f_seq[i] to rows i to i+n-1.
    data = np.repeat(f_seq, n)  # shape (n*n)
    rows = np.concatenate([np.arange(i, i+n) for i in range(n)])
    cols = np.concatenate([np.full(n, i) for i in range(n)])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # combine with -I for nonnegativity
    A_ub = sp.vstack([A, -sp.identity(n)], format='csr')
    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

def proj_simplex(v, s):
    # project onto simplex {x >= 0, sum x = s}
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function sequence that minimises the
`evaluate_sequence` objective.

The algorithm is an accelerated version of the published SOTA:
* keep the sum of the coefficients fixed to √(2·n) (scale‑invariant)
* at each iteration solve a linear program that maximises the total sum
  of a new sequence `g` under the constraint  conv(f,g) ≤ M where
  `M = max(conv(f,f))`.
* after normalising `g` to the same total sum as `f`, take the largest
  possible convex combination `f←(1‑t)f + t·g` that does **not** exceed the
  current bound `M`.  The maximal feasible `t` is found by a short
  binary search using the quadratic formula for the convolution of the
  mixture.
* repeat while time permits, keeping the best sequence seen.

A small random perturbation is applied when progress stalls.
The linear programme is built once per iteration using a sparse
Toeplitz matrix, which makes each solve fast enough for the 1000 s
budget.
"""

import time
import numpy as np
from scipy import optimize, sparse

linprog = optimize.linprog


def _normalize_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale `seq` so that its L1‑norm equals `target` (preserving sign)."""
    total = np.sum(seq)
    if total == 0:
        return seq
    return seq * (target / total)


def _solve_direction_lp(f_seq: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(b)
        subject to conv(f_seq, b) ≤ M   (2·n‑1 linear constraints)
                   b ≥ 0
    Returns the optimal `b` or None if the solver fails.
    """
    n = f_seq.shape[0]

    # Build the sparse Toeplitz matrix representing the convolution constraints.
    # Each column i contributes f_seq[i] to rows i … i+n‑1.
    data = np.repeat(f_seq, n)                                      # length n*n
    rows = np.concatenate([np.arange(i, i + n) for i in range(n)])   # length n*n
    cols = np.concatenate([np.full(n, i, dtype=int) for i in range(n)])

    A_conv = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Non‑negativity: -I * b ≤ 0
    A_ub = sparse.vstack([A_conv, -sparse.identity(n)], format="csr")
    b_ub = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])

    # maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n)

    # Use the HiGHS solver (fast, handles sparse matrices)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if result.success:
        return result.x
    return None


def _max_feasible_t(conv_ff: np.ndarray,
                    conv_fg: np.ndarray,
                    conv_gg: np.ndarray,
                    M: float,
                    eps: float = 1e-5) -> float:
    """
    Binary‑search the largest t ∈ [0,1] such that
        max( (1‑t)^2·ff + 2t(1‑t)·fg + t^2·gg ) ≤ M .
    Returns that t (may be 0).
    """
    lo, hi = 0.0, 1.0

    # If the full step is already feasible we can take t = 1.
    mix_hi = conv_gg  # t = 1 ⇒ mixture = g
    if np.max(mix_hi) <= M:
        return 1.0

    for _ in range(25):          # enough for ~1e‑7 precision
        mid = (lo + hi) * 0.5
        mix = ((1.0 - mid) ** 2) * conv_ff \
              + 2.0 * mid * (1.0 - mid) * conv_fg \
              + (mid ** 2) * conv_gg
        if np.max(mix) <= M:
            lo = mid
        else:
            hi = mid
    return lo


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value sequence within the given time budget.
    Returns the best sequence (a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Initialise ---------------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # Use the provided SOTA construction if it exists.
        init_seq = np.array([float(x) for x in height_sequence_1], dtype=float)
        n = init_seq.shape[0]
    except Exception:
        # Fallback: random start, length roughly 2000 (as in the problem statement)
        n = 2100
        init_seq = np.random.rand(n).astype(float)

    target_sum = np.sqrt(2.0 * n)
    f = _normalize_sum(init_seq, target_sum)

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    it = 0
    stagnant = 0          # counts iterations without improvement

    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # Normalise for the current iteration (keeps the sum constant)
        # --------------------------------------------------------------
        f = _normalize_sum(f, target_sum)
        conv_ff = np.convolve(f, f)
        M = np.max(conv_ff)

        # --------------------------------------------------------------
        # Find a better direction by solving the LP
        # --------------------------------------------------------------
        g_raw = _solve_direction_lp(f, M)
        if g_raw is None:
            # LP failed – perform a tiny random walk and continue
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01 * f[idx])
            continue

        # Normalise g to have the same L1‑norm as f
        g = _normalize_sum(g_raw, target_sum)

        # --------------------------------------------------------------
        # Prepare convolution terms needed for the binary search
        # --------------------------------------------------------------
        conv_fg = np.convolve(f, g)
        conv_gg = np.convolve(g, g)

        # --------------------------------------------------------------
        # Largest admissible mixing coefficient t
        # --------------------------------------------------------------
        t = _max_feasible_t(conv_ff, conv_fg, conv_gg, M)

        # If progress is negligible, inject a random perturbation
        if t < 1e-6:
            stagnant += 1
            if stagnant >= 20:
                # random shake‑up of the whole vector
                noise = np.random.randn(n) * (0.001 * target_sum / n)
                f = np.maximum(f + noise, 0.0)
                stagnant = 0
            continue
        stagnant = 0

        # --------------------------------------------------------------
        # Update the current sequence
        # --------------------------------------------------------------
        f = (1.0 - t) * f + t * g

        # --------------------------------------------------------------
        # Evaluate and keep the best seen sequence
        # --------------------------------------------------------------
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[iter {it:4d}] score={best_score:.6f}, t={t:.4f}, M={M:.6f}")

        # Simple early‑stop: if we already beat the target, stop.
        if best_score <= 1.503:
            print("[target reached]")
            break

    print(f"[finished] iterations={it}, best_score={best_score:.6f}")
    return best_seq.tolist()
