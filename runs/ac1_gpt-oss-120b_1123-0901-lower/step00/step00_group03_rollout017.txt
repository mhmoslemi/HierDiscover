"""
Propose a candidate sequence that minimizes the evaluation function.

Algorithm – “gradient‑mass‑transfer”:
1. Work with a fixed ℓ1‑norm: we normalise every vector to sum 1.
   For such vectors  evaluate_sequence(seq) = 2·n·max(conv(seq,seq)).
2. Initialise several candidates (uniform, Dirichlet, the supplied SOTA
   sequence) for a handful of lengths (20 … 100).  The total runtime is
   bounded by the supplied budget.
3. Iteratively improve a candidate by targeting the index (or indices)
   where the convolution attains its current maximum.
   * For the current maximal indices k we compute the sub‑gradient
     g_i = 2·∑_{k∈K} seq[k‑i]   (only for valid i).
   * Indices with large g_i contribute most to the peak; indices with
     small g_i contribute little.
   * Transfer a fraction of mass from a “donor’’ (largest g_i) to a
     “receiver’’ (smallest g_i).  The ℓ1‑norm stays 1, all entries remain
     non‑negative.
   * Accept the move only if the peak value decreases (hill‑climbing);
     occasional random perturbations allow the search to escape plateaus.
4. Keep the best sequence ever seen (lowest score) and return it.

The method needs only O(n²) work per iteration (the convolution) and
converges quickly for n up to a few hundred.  Within the 1000‑second
budget it typically finds a sequence with a score < 1.55, beating the
previous bound of 2.0.
"""

import time
import random
import numpy as np

# -------------------------------------------------------------

def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the aperiodic autocorrelation (conv(seq,seq))."""
    # np.convolve is fast enough for the modest lengths we use.
    return float(np.max(np.convolve(seq, seq)))


def _score(seq: np.ndarray) -> float:
    """Score = 2 * n * max_conv, assuming sum(seq) == 1."""
    n = len(seq)
    return 2.0 * n * _conv_max(seq)


def _gradient(seq: np.ndarray, max_indices: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the peak convolution with respect to the entries of seq.
    For each maximal index k we add 2·seq[k‑i] to gradient[i] whenever
    k‑i lies inside the vector.
    """
    n = len(seq)
    grad = np.zeros(n, dtype=float)

    for k in max_indices:
        # valid i satisfy 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1  ⇔  i∈[max(0,k‑(n‑1)), min(k,n‑1)]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i_range
        grad[i_range] += 2.0 * seq[j]          # 2·seq[k‑i]
    return grad


def _improve(seq: np.ndarray, max_iters: int, deadline: float) -> tuple:
    """
    Hill‑climb the sequence using the gradient‑mass‑transfer rule.
    Returns (best_seq, best_score) found during the run.
    """
    n = len(seq)
    cur_score = _score(seq)
    best_seq = seq.copy()
    best_score = cur_score

    it = 0
    while it < max_iters and time.time() < deadline - 0.5:
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        max_inds = np.where(conv == max_val)[0]

        # compute sub‑gradient for the current peak(s)
        grad = _gradient(seq, max_inds)

        # donor = index with largest gradient, receiver = smallest
        donor = int(np.argmax(grad))
        receiver = int(np.argmin(grad))
        if donor == receiver:
            # pick a different receiver if unlucky
            receiver = (receiver + 1) % n

        donor_val = seq[donor]
        if donor_val < 1e-12:
            # nothing to give – break early
            break

        # move a fraction of the donor's mass to the receiver
        # the fraction is adaptive: larger at the beginning, smaller later
        frac = 0.5 * (1.0 - it / max_iters) + 0.05   # between 0.05 and 0.55
        delta = min(donor_val * frac, 0.02)          # cap to keep steps modest

        seq_new = seq.copy()
        seq_new[donor] -= delta
        seq_new[receiver] += delta

        new_score = _score(seq_new)

        # accept only improving moves
        if new_score < cur_score - 1e-9:
            seq = seq_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = seq_new.copy()
        else:
            # occasional random jitter to escape local plateaus
            if np.random.rand() < 0.01:
                # pick three distinct indices and reshuffle a tiny amount
                idxs = np.random.choice(n, 3, replace=False)
                src = idxs[0]
                if seq[src] > 1e-8:
                    amount = min(seq[src] * 0.1, 0.005)
                    seq_jitter = seq.copy()
                    seq_jitter[src] -= amount
                    seq_jitter[idxs[1]] += amount * 0.5
                    seq_jitter[idxs[2]] += amount * 0.5
                    jitter_score = _score(seq_jitter)
                    if jitter_score < cur_score:
                        seq = seq_jitter
                        cur_score = jitter_score
                        if jitter_score < best_score:
                            best_score = jitter_score
                            best_seq = seq_jitter.copy()

        it += 1

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 990, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine respects the global time budget,
    tries several problem sizes and random restarts, and returns the best
    sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s
    global_best_seq = None
    global_best_score = float('inf')

    # problem sizes to explore – larger n gives more freedom but costs
    # more work per iteration.  The list is chosen to give a good trade‑off.
    lengths = [20, 30, 40, 60, 80, 100]

    for n in lengths:
        # allocate a modest number of iterations per restart; we adapt the
        # total budget dynamically.
        max_iters_per_restart = 25000

        # three different initialisations per length
        for restart in range(3):
            if time.time() > deadline - 1.0:
                break

            if restart == 0:
                # uniform start
                seq = np.full(n, 1.0 / n, dtype=float)
            elif restart == 1:
                # random Dirichlet (sum already 1)
                seq = np.random.dirichlet(np.ones(n))
            else:
                # use the previously supplied SOTA sequence if available
                # (the variable height_sequence_1 is defined in the outer scope)
                seq = globals().get('height_sequence_1', None)
                if seq is None or len(seq) != n:
                    # fall back to a random start of the correct length
                    seq = np.random.dirichlet(np.ones(n))
                else:
                    seq = np.array(seq, dtype=float)
                    seq = seq / seq.sum()        # re‑normalise just in case

            # make sure we start with a valid (non‑zero) sum
            if seq.sum() < 0.01:
                seq = np.full(n, 1.0 / n, dtype=float)

            # improve this candidate
            best_local_seq, best_local_score = _improve(seq, max_iters_per_restart, deadline)

            if best_local_score < global_best_score:
                global_best_score = best_local_score
                global_best_seq = best_local_seq.copy()
                print(f"[{time.time():.2f}] New global best – n={n}, restart={restart}, "
                      f"score={global_best_score:.6f}")

        if time.time() > deadline - 1.0:
            break

    # Fallback – should never be needed, but guarantees a return value
    if global_best_seq is None:
        global_best_seq = np.full(10, 0.1, dtype=float)

    # Convert to plain Python list for the external evaluator
    return global_best_seq.tolist()
