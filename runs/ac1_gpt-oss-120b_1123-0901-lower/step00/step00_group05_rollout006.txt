def conv_fft(x, y):
    N = len(x) + len(y) - 1
    size = 1 << (N-1).bit_length()   # next power of two

"""
Gradient‑descent optimizer for the auto‑convolution bound.

The algorithm works with a probability vector `a` (non‑negative, sum = 1).
Its objective is to make the auto‑convolution `b = a*a` as flat as possible,
i.e. to minimise the variance of `b`.  This directly reduces the
`max(b)` term in the evaluation function:
    score = 2·n·max(b) / (sum(a)^2)   (sum(a)=1 → score = 2·n·max(b))

At each iteration we:
  * compute `b` via an FFT‑based convolution,
  * form the residual `diff = b – avg`, where `avg` is the theoretical average
    of the convolution (S²/(2n‑1) with S=1),
  * compute the gradient of the variance w.r.t. `a`:
        grad_i = 4·∑_j a_j·diff_{i+j}
    which is a correlation that can also be obtained with FFTs,
  * take a momentum step, then project back onto the probability simplex
    (non‑negative, sum = 1).

The best sequence seen before the time limit is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution using a power‑of‑two FFT (real input)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of two real vectors using FFT."""
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    z = np.fft.irfft(X * Y, n=size)
    return z[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Fast score computation (identical to evaluate_sequence, but without I/O)
def _score_from_vec(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) / (sum(a)^2)."""
    n = a.shape[0]
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = _conv_fft(a, a)
    return float(2.0 * n * b.max() / (S * S))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Gradient‑descent search for a low‑score step‑function.
    Returns the best vector (list of floats) found before the time limit.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function stops a few tenths of a
        second before the limit to avoid being cut off).

    Returns
    -------
    list[float]
        The sequence of heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose the length.  Prefer the known construction if present.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 6000                             # fallback length

    # Initialise a near‑uniform probability vector.
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += np.random.randn(n) * 1e-4           # tiny perturbation to break symmetry
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    best_a = a.copy()
    best_score = _score_from_vec(best_a)
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optimiser hyper‑parameters
    eta0 = 0.5           # initial learning rate
    decay = 1e-4         # decay per second
    momentum = 0.9
    vel = np.zeros_like(a)

    # Pre‑compute FFT sizes (powers of two) to avoid re‑allocation each step.
    size_conv = 1 << ((2 * n - 1).bit_length())        # for a*a
    size_grad = 1 << ((3 * n - 2).bit_length())        # for a_rev * diff

    iteration = 0
    while time.time() < deadline:
        # ---------- auto‑convolution ----------
        Fa = np.fft.rfft(a, n=size_conv)
        b = np.fft.irfft(Fa * Fa, n=size_conv)[: 2 * n - 1]

        # target average of the convolution (S² / (2n‑1)); S = 1 here.
        avg = 1.0 / (2 * n - 1)
        diff = b - avg

        # ---------- gradient of the variance ----------
        # grad_i = 4 * Σ_j a_j * diff_{i+j}
        Fr = np.fft.rfft(a[::-1], n=size_grad)
        Fd = np.fft.rfft(diff, n=size_grad)
        grad_full = np.fft.irfft(Fr * Fd, n=size_grad)[: 3 * n - 2]
        grad = 4.0 * grad_full[n - 1 : 2 * n - 1]       # length n

        # ---------- momentum step ----------
        elapsed = time.time() - start_time
        eta = eta0 / (1.0 + decay * elapsed)
        vel = momentum * vel - eta * grad
        a = a + vel

        # ---------- projection onto the probability simplex ----------
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s                                    # enforce sum = 1

        # ---------- occasional evaluation ----------
        if iteration % 30 == 0:
            # we already have the convolution `b`
            cur_score = 2.0 * n * b.max()            # sum(a)=1 → simplified formula
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[iter {iteration:5d}] improved score = {best_score:.6f}")

        iteration += 1

    # Final check with the official evaluator (optional, just for logging)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluated score = {final_score:.6f}")
    except Exception:
        # If evaluate_sequence is unavailable for some reason, fall back to our fast score.
        final_score = _score_from_vec(best_a)
        print(f"[final] fast‑score = {final_score:.6f}")

    return best_a.tolist()
