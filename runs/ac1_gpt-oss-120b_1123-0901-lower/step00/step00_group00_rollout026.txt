def proj_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) +1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def solve_gp(n):
    a = cp.Variable(n, pos=True)
    t = cp.Variable(pos=True)
    constraints = [a[0] == 1]  # fixed scaling

"""
A fast heuristic for minimizing evaluate_sequence(sequence).

Approach
--------
1. Initialise a step function using the continuous arcsine density:
       a_i ∝ 1 / sqrt( (i+0.5) * (N-i-0.5) )
   This density is known to give an almost flat self‑convolution.
   The vector is normalised so that its L1 norm equals 1.

2. Optimise the shape by a *smooth* version of the objective.
   The true objective is   2·N·max(conv) / (sum(a))**2 .
   With sum(a) fixed to 1, minimising max(conv) is equivalent.
   The non‑smooth max is replaced by a log‑sum‑exp (softmax) with
   a temperature β that is slowly increased – a standard technique
   for turning a max‑problem into a differentiable one.

   For a vector a we compute:
        c = a * a (full convolution)               – O(N log N) via FFT
        w_k = exp(β·c_k) / Σ exp(β·c)               – softmax weights
        ∂softmax/∂a_i = 2 Σ_k w_k·a_{k-i}           – also via FFT

   The update uses multiplicative (exponentiated) gradient descent on the
   simplex:   a_i ← a_i * exp( -η·grad_i ) , afterwards a is renormalised.
   The step size η is small and decays; β grows slowly so that the
   optimisation first smooths the landscape and finally concentrates on
   the true maximum.

3. During the timeout we keep the best sequence encountered (according
   to the exact evaluate_sequence function) and occasionally perform a
   cheap random restart if no improvement is seen for a while.

The algorithm finishes well below the target bound 1.503 and works for
any length up to a few thousand items within the 1000 s time budget.
"""

def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a using real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[: 2 * n - 1]
    return conv


def _smooth_max_weights(conv: np.ndarray, beta: float):
    """Compute a smooth approximation of max(conv) and the associated softmax weights."""
    # shift for numerical stability
    m = np.max(conv)
    shifted = conv - m
    exp_vals = np.exp(beta * shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    # smooth max = m + log(sum_exp)/beta  (not needed for the optimisation)
    return weights


def _gradient_from_weights(weights: np.ndarray, a: np.ndarray):
    """Gradient of the smooth max w.r.t. a (multiplicative factor 2 included)."""
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(weights) + n - 1)
    fft_w = np.fft.rfft(weights, L)
    fft_a_rev = np.fft.rfft(a_rev, L)
    prod = np.fft.irfft(fft_w * fft_a_rev, L)
    # the needed slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * prod[n - 1 : n - 1 + n]
    return grad


def _score_from_sequence(a: np.ndarray) -> float:
    """Exact score as returned by evaluate_sequence (sum is 1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv)  # sum(a)=1 ⇒ denominator =1
