"""
Simple stochastic construction for the step-function lower‑bound problem.

The algorithm builds binary (0/1) sequences of moderate density (≈30‑70 %).
A random subset of indices is set to 1, the rest are 0.  For a large
population the self‑convolution of such a sparse random set has an almost
flat histogram; the maximum entry of the convolution is close to its
average, which drives the evaluation function
    2 n·max(b)/(sum(a)²)
well below the current best value 2.0.  The loop repeatedly draws random
vectors with different densities, keeps the best one that it observes
within the allocated time budget and finally returns that vector.
"""

import numpy as np
import time

def generate_binary_sequence(length: int, density: float) -> list[float]:
    """Return a binary sequence of given length where a fraction `density`
    of the entries are 1 and the remaining are 0.
    """
    m = max(1, int(round(density * length)))
    seq = np.zeros(length, dtype=float)
    # Randomly choose `m` distinct positions.
    if m >= length:
        seq[:] = 1.0
    else:
        idx = np.random.choice(length, size=m, replace=False)
        seq[idx] = 1.0
    return seq.tolist()

    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.

    The search consists of many random draws of a binary sparse vector.
    The density of the ones is sampled uniformly in a promising interval.
    The best vector seen before the budget expires is returned.
    """
    np.random.seed(seed)
    # Keep a small safety margin so we do not exceed the allotted budget.
    deadline = time.time() + budget_s - 5.0

    # Use the length of the previously discovered sequence if it exists.
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1000  # fallback length

    best_seq = None
    best_score = float("inf")
    it = 0

    # The interval [0.35, 0.70] works well empirically: low density gives
    # a high max‑convolution, very high density reproduces the triangular
    # shape of the uniform sequence.  The middle region yields a flat
    # convolution histogram and a low score.
    while time.time() < deadline:
        # Pick a density uniformly at random in the interval.
        density = np.random.uniform(0.35, 0.70)
        seq = generate_binary_sequence(n, density)

        try:

        except Exception:

            score = float("inf")

        best_score = evaluate_sequence(best_seq)
