"""
Optimise a non‑negative step‑function for the objective

    F(a) = 2·n·max(a∗a) / (sum a)²

The code builds many diverse start‑ups, refines each one with three
phases (smooth‑max Adam → exact sub‑gradient → cheap hill‑climb) and
keeps the best result.  Improvements over the reference implementation:

*   Up to 30 initial candidates per length (instead of 12) covering a
    richer family of edge‑heavy, centre‑light and multi‑peak shapes.
*   A more aggressive temperature schedule for the smooth‑max (λ grows
    faster) and a learning‑rate decay that helps the optimiser settle.
*   Adaptive iteration limits: the number of Adam/sub‑gradient/hill‑climb
    steps is chosen from the remaining wall‑time, guaranteeing we use
    the full 1000 s budget without overrunning.
*   After the main search a final long Adam run (with a very low
    learning‑rate) is performed on the current global best.
*   Simple simulated‑annealing style perturbations are applied when a
    candidate stagnates for a few seconds.

All helper functions are top‑level, no closures or lambdas are used,
and the returned value is a plain Python list as required by the
external evaluator.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective 2·n·max(conv) / (sum a)².
    a is assumed non‑negative.
    """
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_c: int,
                   fft_len_x: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate.
    The distribution is kept on the simplex via a soft‑max (logits).
    """
    # Normalise to sum = 1 and work with logits for positivity.
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.08            # initial learning rate
    lam = 0.5             # temperature (inverse smoothness)
    lam_factor = 1.06     # grow λ faster than the reference code
    lr_decay = 0.9995     # mild decay each iteration

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_c)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---------- simplex via soft‑max ----------
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---------- smooth max ----------
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        # soft‑max weights for the surrogate gradient
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # back‑prop through soft‑max (log‑its)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 * (lr_decay ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- schedule ----------
        lam = min(lam * lam_factor, 1e8)

        cur_val = _objective(a, n, fft_len_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:05d}] new best {cur_val:.6f}")

        # occasional shake‑up if stuck > 3 s
        if time.time() - last_improve > 3.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_c: int,
                      fft_len_x: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max (sub‑gradient).
    """
    a = a0 / a0.sum()
    if init_val is None:
        best_val = _objective(a, n, fft_len_c)
    else:
        best_val = init_val
    best_a = a.copy()
    last_improve = time.time()
    eta = 0.7   # step‑size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # exponentiated‑gradient step (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} SubGrad it={it:05d}] new best {cur_val:.6f}")

        # stop if stuck for a few seconds
        if time.time() - last_improve > 2.5:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_c: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: transfer a small fraction of mass between two
    random coordinates; accept if improvement.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move a random proportion (up to 10 %) of mass from i to j
        eps = np.random.rand() * 0.10 * a[i]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        if new_a[i] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 100 == 0:
                print(f"[n={n} HillClimb it={it:05d}] new best {new_val:.6f}")

        # Occasionally accept a worsening move (simulated‑annealing style)
        elif np.random.rand() < 0.001:
            a = new_a

        # If no improvement for a while, inject a tiny random jitter
        if time.time() - last_improve > 4.0:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.clip(a + jitter, 0, None)
            a /= a.sum()
            last_improve = time.time()

    return best_a, best_val

def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    total = new.sum()
    if total < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / total
    return new


def _generate_initial_candidates(n: int,
                                 global_best: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a varied pool of start‑up sequences (all sum‑to‑1).
    """
    cand = []

    # 1. Warm‑start from the current global best (exact length)
    if global_best is not None and global_best.shape[0] == n:
        cand.append(global_best.copy())
    # 2. Interpolated version of the global best
    if global_best is not None and global_best.shape[0] != n:
        cand.append(_interpolate(global_best, n))

    # 3. Pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 4‑9. Parameterised families ------------------------------------------------
    idx = np.arange(n, dtype=np.float64) + 1.0
    rev = n - idx + 1.0
    prod = idx * rev               # i·(n‑i+1)

    # Edge‑heavy power‑law (α negative, more extreme than before)
    for alpha in np.linspace(-3.5, -0.2, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # Inverse‑root edges with a broader exponent range
    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5):
        shape = np.power(idx, -alpha) + np.power(rev, -alpha)
        cand.append(shape / shape.sum())

    # Two‑Gaussian edges with a finer set of γ
    for gamma in (0.1, 0.3, 0.6, 1.2, 2.5):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # Multi‑peak Gaussian mixtures (random centres & widths)
    for _ in range(5):
        peaks = np.zeros(n, dtype=np.float64)
        for centre in (0.2, 0.5, 0.8):
            sigma = np.random.uniform(0.04, 0.15)
            x = np.arange(n, dtype=np.float64) / (n - 1)
            peaks += np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        cand.append(peaks / peaks.sum())

    # Random piecewise‑linear (few breakpoints)
    for _ in range(3):
        bpts = sorted(np.random.choice(n, size=4, replace=False))
        values = np.random.rand(len(bpts))
        seq = np.interp(np.arange(n), bpts, values)
        seq = np.maximum(seq, 0)
        cand.append(seq / seq.sum())

    random.shuffle(cand)
    # Keep a reasonable number (≈30) to stay within the time budget
    return cand[:30]

            best_seq,
            n,
            fft_c,
            fft_x,
            deadline,

            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Optional: report the final score using the provided evaluator (cheap call)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal evaluation score reported by evaluator: {final_score:.6f}")
    except Exception:
        # evaluate_sequence may not be available in the offline test environment
        pass

    return best_seq.tolist()
```

This implementation respects the 1000 s time limit, exploits a richer set of initial shapes, uses more aggressive annealing for the smooth‑max, adapts iteration counts to the remaining budget, and finishes with a high‑precision Adam refinement. The returned list can be fed directly to the `evaluate_sequence` function.
