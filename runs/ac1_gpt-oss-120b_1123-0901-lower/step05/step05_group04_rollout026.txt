    n = a.shape[0]
    b = cp.Variable(n)

def simplex_projection(v, s=1.0):
    n = len(v)
    if v.sum() == s and np.all(v>=0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def solve_lp_max_sum(a, M, solver='ECOS'):
    import cvxpy as cp
    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(a, b) <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, warm_start=True, max_iters=5000, feastol=1e-9, abstol=1e-9, reltol=1e-8)
    except cp.error.SolverError:
        return None
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)

"""
Search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

* A cheap FFT based evaluator is used for the inner loop.
* Many random “shapes’’ (power law, exponential, mixed, uniform) are
  sampled.  Each shape is normalised so that ∑a = √(2 n).
* If the optional ``cvxpy`` package is available we run a small
  “active‑set’’ optimisation inspired by the literature:

        a₀ ← initial shape
        repeat
            M  ← max(conv(a₀, a₀))
            b  ← solution of   maximise ∑b   s.t.  b≥0,
                               conv(a₀, b) ≤ M               (LP)
            g  ← (√(2 n) / ∑b) · b          (scale to the canonical sum)
            a₁ ← (1‑t)·a₀ + t·g  for a few trial step‑sizes  t∈{0.5,0.3,0.1,0.05}
        until no improvement or time runs out

  The LP is expressed as a linear convolution constraint using
  ``cvxpy.conv`` and solved with the high‑speed ECOS solver.
* When ``cvxpy`` is missing we fall back to the Adam optimiser on a
  soft‑max parametrisation (the code from the previous state‑of‑the‑art).
* The best sequence found over all restarts is returned.  If time permits a
  final refinement (again the active‑set method) is performed on the incumbent.

The routine respects the 1000 s budget, prints progress information and
finally returns a plain Python list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – fast full convolution via FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Fast evaluator (identical to evaluate_sequence but without the safety layers)
# ----------------------------------------------------------------------
def _fast_evaluation(a: np.ndarray) -> float:
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max() / (s ** 2))


# ----------------------------------------------------------------------
# Random initial shapes (all will be re‑scaled later)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.5)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:                                   # uniform
        shape = np.ones_like(idx)

    shape += rng.random(n) * 1e-6          # tiny jitter
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables (fallback when cvxpy is missing)
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """Adam on a smoothed max‑convolution surrogate."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # 1) map logits to the simplex (a = target_sum·softmax(x))
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # 2) current max‑convolution (exact)
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  local best = {max_conv:.8f}")

        # 3) smooth‑max weights (log‑sum‑exp surrogate)
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4) gradient w.r.t. a (derivative of Σ w_k·conv_k)
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # 5) gradient w.r.t. logits x (chain rule through the soft‑max)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # 6) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        alpha *= 0.999

        if iteration - last_improve > 2000:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear programme (maximise ∑b  s.t.  b≥0 , conv(a,b) ≤ M)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float):
    """Return the optimal b for the LP described in the doc‑string.
    Returns None on failure."""
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    # conv(a,b) ≤ M  (vector inequality, one constraint per shift)
    constraints = [cp.conv(a, b) <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:
        return None
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)


# ----------------------------------------------------------------------
# Active‑set refinement (LP + mixing)
# ----------------------------------------------------------------------
def _active_set_optimize(n: int,
                         a0: np.ndarray,
                         target_sum: float,
                         max_iters: int,
                         deadline: float):
    """Iteratively improve a sequence using the LP‑based active‑set method."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _fast_evaluation(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current bound
        conv = _full_convolution_fft(a, a)
        M = conv.max()

        # linear programme
        b = _solve_lp_max_sum(a, M)
        if b is None or b.sum() <= 0.0:
            break

        # scale to the canonical sum
        sum_b = b.sum()
        g = b * (target_sum / sum_b)

        improved = False
        for t in (0.5, 0.3, 0.1, 0.05):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, 1000.0)
            score_new = _fast_evaluation(a_new)
            if score_new < best_score - 1e-8:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                print(f"[{time.time():.2f}] n={n}  active‑set iter {it}  t={t}  score={score_new:.8f}")
                break
        if not improved:
            break
        it += 1
    return best_a, best_score
