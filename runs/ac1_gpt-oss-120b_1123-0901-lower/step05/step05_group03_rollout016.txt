import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.shape[0]-1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a, n, conv_fft_len):
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   start_lr=0.03, max_iters=500000, lam_start=0.3, lam_factor=1.05):
    """Adam optimizer on the log‑sum‑exp smooth‑max surrogate.
    Returns (best_seq, best_val, last_a) where last_a is the final iterate."""
    # Initialise soft‑max logits
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = start_lr

    best_seq = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weighting
        w = np.exp(lam_start * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam_start = min(lam_start * lam_factor, 1e6)

        # occasional shake if stuck
        if time.time() - last_improve > 3.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    # return best and final a (decoded)
    final_exp = np.exp(logits - logits.max())
    a_last = final_exp / final_exp.sum()
    return best_seq, best_val, a_last

def _subgradient_refine(a0, n, conv_fft_len, corr_fft_len, deadline,
                        start_eta=0.6, max_iters=20000):
    """Exponentiated‑gradient sub‑gradient phase on the true max."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # multiplicative update
        a = a * np.exp(-start_eta * grad)

        s = a.sum()
        if s < 1e-12:
            # re‑initialize to avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if not improving for a while
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val

def _initial_candidates(n, warm):
    """Return a list of diverse starting vectors of length n."""
    cand = []

    # warm start and cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 6, n // 3, n // 2):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet (dense)
    cand.append(np.random.dirichlet(np.ones(n)))

    # sparse random: choose k positions
    for k in (int(np.sqrt(n) // 2), int(np.sqrt(n)), int(np.sqrt(n) * 2)):
        mask = np.zeros(n, dtype=bool)
        idx = np.random.choice(n, size=min(k, n), replace=False)
        mask[idx] = True
        weights = np.random.rand(mask.sum())
        vec = np.zeros(n)
        vec[mask] = weights
        if vec.sum() > 0:
            vec /= vec.sum()
            cand.append(vec)

    # power‑law edge shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # alternating high‑low pattern (comb)
    comb = np.zeros(n)
    comb[::2] = 1.0
    comb /= comb.sum()
    cand.append(comb)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand

      * a diverse pool of initial candidates (dense, sparse, edge‑biased…)
      * for each candidate an Adam optimisation on a smooth‑max surrogate
      * a dedicated exponentiated‑gradient sub‑gradient refinement
      * a final long‑run refinement on the currently best sequence
    The total wall‑clock time respects the supplied budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # primary length (the known good bound was obtained for 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:

        else:
            warm_start = None
    else:
        warm_start = None

    np.random.shuffle(candidates)

    # Phase 1: quick Adam runs on each candidate (limited iterations)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # short Adam run (max 30 k iterations or until time runs out)
        short_deadline = min(deadline, time.time() + 20.0)  # max 20 s per candidate
        seq, val, _ = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, short_deadline,
            start_lr=0.03, max_iters=150_000, lam_start=0.4, lam_factor=1.07
        )
        # sub‑gradient refinement (fast)
        seq, val = _subgradient_refine(seq, n, conv_fft_len, corr_fft_len, short_deadline)

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Candidate {cid}] improved global best to {val:.6f}")

    if best_seq is not None and time.time() < deadline - 30.0:
        remaining = deadline - time.time()

        long_deadline = time.time() + remaining - 5.0

        seq, val, final_a = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, long_deadline,
            start_lr=0.01, max_iters=500_000, lam_start=0.2, lam_factor=1.03

        seq, val = _subgradient_refine(final_a, n, conv_fft_len, corr_fft_len, long_deadline)

            print(f"[Upscaled] improved to {best_val:.6f}")

    # Clip to evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list
    return best_seq.tolist()

"""
Propose a low‑objective step‑function.

The search proceeds in three stages:

1.  Build a diverse pool of start vectors (dense Dirichlet,
    sparse random masks, edge‑biased power‑law, Gaussian edges,
    sinusoid, comb, uniform and optional warm‑start).  For each start
    run a short Adam optimisation on a log‑sum‑exp surrogate of the
    maximum convolution, followed by a quick exponentiated‑gradient
    sub‑gradient phase.  The best sequence found over all starts is
    kept.

2.  Perform a long‑running refinement of that best sequence using
    a slower Adam schedule (smaller learning rate, slower temperature
    increase) and a longer exponentiated‑gradient phase.

3.  If time permits, interpolate the best sequence to a larger
    length (8192) and run another short refinement; any improvement
    is down‑sampled back to the primary length.

All work respects the supplied time budget (default 1000 s) and
returns a plain Python list of non‑negative numbers ready for
``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Result length = len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (sum(a) assumed ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    start_lr: float = 0.03,
    max_iters: int = 500_000,
    lam_start: float = 0.3,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  Returns the best
    sequence seen, its value, and the final iterate decoded from the
    logits.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = start_lr

    best_seq = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    last_improve = time.time()
    it = 0

    lam = lam_start
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & smooth‑max weighting
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature (smooth‑max → true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional shake when stuck
        if time.time() - last_improve > 3.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    # decode final logits
    final_exp = np.exp(logits - logits.max())
    a_last = final_exp / final_exp.sum()
    return best_seq, best_val, a_last


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    start_eta: float = 0.6,
    max_iters: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient phase on the exact max (no surrogate).
    Returns the best sequence found and its objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:05d}] new best {val:.6f}")

        # active set of maximal entries
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps non‑negativity)
        a = a * np.exp(-start_eta * grad_a)

        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop early if stagnant
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Produce a modest pool of diverse start vectors."""
    cand = []

    # warm start and a few shifted copies
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 6, n // 3, n // 2):
            cand.append(np.roll(warm, shift))

    # dense random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # sparse random masks of a few non‑zero positions
    for k in (int(np.sqrt(n) // 2), int(np.sqrt(n)), int(np.sqrt(n) * 2)):
        mask = np.zeros(n, dtype=bool)
        idx = np.random.choice(n, size=min(k, n), replace=False)
        mask[idx] = True
        weights = np.random.rand(mask.sum())
        vec = np.zeros(n)
        vec[mask] = weights
        if vec.sum() > 0:
            vec /= vec.sum()
            cand.append(vec)

    # power‑law edge‑biased shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave (positive)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # alternating comb pattern
    comb = np.zeros(n)
    comb[::2] = 1.0
    comb /= comb.sum()
    cand.append(comb)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand
