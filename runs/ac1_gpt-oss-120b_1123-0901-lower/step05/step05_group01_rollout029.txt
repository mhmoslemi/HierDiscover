"""
Optimisation of non‑negative step‑function heights to minimise
    F(a) = 2·n·max(a∗a) / (sum a)²

The algorithm is a multi‑stage search:

* generate a diverse pool of initial vectors (Dirichlet, power‑law,
  edge‑exponential, mixture‑Gaussian, interpolation of the current best);
* for each start run a large number of exponentiated‑gradient (sub‑gradient)
  iterations on the exact max‑convolution; the step‑size decays as
  η₀/√t and the active set of constraints is updated every iteration;
* finish the candidate with a cheap stochastic hill‑climb that swaps a
  small amount of mass between two random coordinates; occasional uphill
  moves allow escaping shallow plateaus;
* keep the globally best sequence across a ladder of lengths
  (2048 → 4096 → 8192 → 16384 → 32768) and finally polish the best vector
  with a longer Adam run on a smooth‑max surrogate.

The whole routine respects a user‑supplied time budget (default 1000 s)
and returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int, best_seq: np.ndarray | None
) -> list[np.ndarray]:
    """
    Produce a moderate sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # warm‑start from the current best (exact length) or interpolated
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes: ((i+1)*(n‑i))**α
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 6):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑sided exponential (edge‑Gaussian) shapes
    for gamma in np.linspace(0.05, 2.0, 6):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in np.linspace(0.4, 1.5, 5):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # two‑peak Gaussian mixture
    for sigma in (0.05, 0.10, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = (
            np.exp(-0.5 * ((x - 0.33) / sigma) ** 2)
            + np.exp(-0.5 * ((x - 0.66) / sigma) ** 2)
        )
        cand.append(peaks / peaks.sum())

    # three‑peak Gaussian mixture
    for sigma in (0.07, 0.15):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(
            np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75)
        )
        cand.append(peaks / peaks.sum())

    # random mixtures of a few Gaussians (stochastic)
    for _ in range(3):
        centers = np.random.rand(3)
        sigmas = np.random.uniform(0.02, 0.15, size=3)
        amps = np.random.rand(3)
        x = np.arange(n, dtype=np.float64) / (n - 1)
        mix = sum(
            amp * np.exp(-0.5 * ((x - cen) / sigma) ** 2)
            for amp, cen, sigma in zip(amps, centers, sigmas)
        )
        cand.append(mix / mix.sum())

    np.random.shuffle(cand)
    # Keep a reasonably small pool – the optimisation of each candidate is expensive.
    return cand[:20]


# ----------------------------------------------------------------------
def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (sub‑gradient) optimisation of max(a*a)
    with a decreasing step‑size η₀/√t.  The vector is kept on the simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()
    eta0 = 0.8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution and active set
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        act_cnt = active.sum()
        if act_cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / act_cnt

        # gradient of the active max‑convolution
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[subgrad n={n} it={it:05d}] new best {cur_val:.6f}")

        # early stop if stagnating
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a random fraction
    of the mass from one coordinate to another and keep the move
    if it improves the objective.  Occasionally an uphill move is
    accepted to escape tiny plateaus.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv)
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # transfer up to 5 % of the mass from i to j
        eps = np.random.rand() * 0.05 * a[i]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # re‑normalise (the sum stays 1)
        # (numerical drift may appear, so we renorm explicitly)
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 200 == 0:
                print(f"[hill n={n} it={it}] new best {new_val:.6f}")
        else:
            # very rare uphill acceptance
            if np.random.rand() < 0.001:
                a = new_a

        # stop early after a while without progress
        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate.
    The surrogate uses a soft‑max over the convolution entries;
    the temperature (lam) grows exponentially so that the surrogate
    becomes tighter as optimisation proceeds.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.3               # initial temperature parameter
    lam_factor = 1.025

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex projection via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # avoid overflow/underflow by shifting by max_c
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (same as for exact max but weighted)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={t}] new best {cur_val:.6f}")

        # occasional random shake when stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val

"""
Optimisation of non‑negative step‑function heights to minimise
    F(a) = 2·n·max(a∗a) / (sum a)²

The algorithm is a multi‑stage search:

* generate a diverse pool of initial vectors (Dirichlet, power‑law,
  edge‑exponential, mixture‑Gaussian, interpolation of the current best);
* for each start run a large number of exponentiated‑gradient (sub‑gradient)
  iterations on the exact max‑convolution; the step‑size decays as
  η₀/√t and the active set of constraints is updated every iteration;
* finish the candidate with a cheap stochastic hill‑climb that swaps a
  small amount of mass between two random coordinates; occasional uphill
  moves allow escaping shallow plateaus;
* keep the globally best sequence across a ladder of lengths
  (2048 → 4096 → 8192 → 16384 → 32768) and finally polish the best vector
  with a longer Adam run on a smooth‑max surrogate.

The whole routine respects a user‑supplied time budget (default 1000 s)
and returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int, best_seq: np.ndarray | None
) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # warm‑start from the current best (exact length) or interpolated
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law: ((i+1)*(n‑i))**α
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 6):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑sided exponential (edge‑Gaussian) shapes
    for gamma in np.linspace(0.05, 2.0, 6):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in np.linspace(0.4, 1.5, 5):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # two‑peak Gaussian mixture
    for sigma in (0.05, 0.10, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = (
            np.exp(-0.5 * ((x - 0.33) / sigma) ** 2)
            + np.exp(-0.5 * ((x - 0.66) / sigma) ** 2)
        )
        cand.append(peaks / peaks.sum())

    # three‑peak Gaussian mixture
    for sigma in (0.07, 0.15):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # random mixtures of a few Gaussians (stochastic)
    for _ in range(3):
        centers = np.random.rand(3)
        sigmas = np.random.uniform(0.02, 0.15, size=3)
        amps = np.random.rand(3)
        x = np.arange(n, dtype=np.float64) / (n - 1)
        mix = sum(
            amp * np.exp(-0.5 * ((x - cen) / sigma) ** 2)
            for amp, cen, sigma in zip(amps, centers, sigmas)
        )
        cand.append(mix / mix.sum())

    np.random.shuffle(cand)
    return cand[:20]   # keep the pool moderate

def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation of max(a*a) with a
    decreasing step‑size η₀/√t.  The vector stays on the simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()
    eta0 = 0.8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        act_cnt = active.sum()
        w = np.zeros_like(conv)
        if act_cnt > 0:
            w[active] = 1.0 / act_cnt

        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[subgrad n={n} it={it:05d}] new best {cur_val:.6f}")

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing: move a random fraction of mass
    from one index to another and keep the move if it improves the objective.
    Very occasional uphill moves help escape tiny plateaus.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv)
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        eps = np.random.rand() * 0.05 * a[i]   # up to 5 % of mass at i
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 200 == 0:
                print(f"[hill n={n} it={it}] new best {new_val:.6f}")
        else:
            if np.random.rand() < 0.001:   # rare uphill acceptance
                a = new_a

        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val
