def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = np.inf

    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]   # include smaller for warm-ups
    # may also attempt 32768 if time.

    for n in lengths:
        if time.time() > deadline: break
        # compute FFT lengths
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(3*n-2)))
        # build candidates
        candidates = []
        # uniform
        candidates.append(np.full(n, 1.0/n))
        # arcsine shape
        idx = np.arange(1, n+1)
        shape = 1.0/np.sqrt(idx * (n - idx + 1))
        shape = shape / shape.sum()
        candidates.append(shape)
        # if previous best of same size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq))
        # upsample
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq)
            x_old = np.linspace(0, 1, old.shape[0])
            x_new = np.linspace(0, 1, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum()
            candidates.append(up)
        # random Dirichlet a few times
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))
        # power law shapes
        prod = idx * (n - idx + 1).astype(float)
        for alpha in np.linspace(-2.5, -0.2, 5):
            shape = np.where(prod > 0, prod**alpha, 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # Two Gaussian edges
        i = np.arange(n)
        for gamma in [0.1, 0.3, 0.6, 1.5]:
            shape = np.exp(-gamma*i**2) + np.exp(-gamma*(n-1-i)**2)
            shape = shape/shape.sum()
            candidates.append(shape)

        # Process each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break
            # Ensure positivity
            eps = 1e-16
            a = np.maximum(a0, eps)
            a = a / a.sum()
            # parameterize by logits
            logits = np.log(a)

            # adam optimization
            best_val_local = best_val
            # hyperparameters: learning rate schedule decays maybe
            lr = 0.08
            beta1 = 0.9
            beta2 = 0.999
            eps_adam = 1e-8

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            lam = 0.5
            lam_factor = 1.02
            # iteration budget: keep enough for convergence
            max_iters = int(30000 * 1024 / n) + 3000
            last_improve = time.time()
            for it in range(1, max_iters+1):
                # compute a from logits via softmax
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()
                # convolution
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                # compute smooth-max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum
                # gradient
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1-beta1) * grad_log
                v = beta2 * v + (1-beta2) * (grad_log**2)
                m_hat = m / (1 - beta1**it)
                v_hat = v / (1 - beta2**it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal lam
                lam = min(lam * lam_factor, 1e7)

                # evaluate
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[len={n} cand={cand_id} Adam it={it}] new best {val:.6f}")
                # stagnation
                if time.time() - last_improve > 4.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()
                # time check
                if time.time() > deadline:
                    break

            # subgradient refinement
            # Starting from final a
            a = a / a.sum()
            eta0 = 1.0
            eta = eta0
            sub_last = time.time()
            max_sub_iters = int(50000 * 1024 / n) + 5000
            for sub_it in range(1, max_sub_iters+1):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                w = np.zeros_like(conv)
                if active.any():
                    w[active] = 1.0 / active.sum()
                # gradient
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                # step size decreasing
                eta = eta0 / np.sqrt(sub_it)
                # mirror descent (exponential update)
                a = a * np.exp(-eta * grad_a)
                # renormalize
                s = a.sum()
                if s <= 0 or np.isnan(s) or np.isinf(s):
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluate
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[len={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 6.0:
                    break
                if time.time() > deadline:
                    break

    # after optimizing all lengths, optionally refine global best using local random perturbations
    if time.time() < deadline and best_seq is not None:
        a = np.array(best_seq, copy=True)
        n = len(a)
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(3*n-2)))
        last_improve = time.time()
        # local random perturbation iterations
        # We'll keep total iterations limited by time
        it = 0
        while time.time() < deadline - 0.5:
            it += 1
            # randomly select two indices
            i, j = np.random.randint(0, n, size=2)
            # propose moving a small amount delta from i to j
            # choose delta proportion of a[i]
            delta = a[i] * np.random.uniform(0.001, 0.01)
            if delta > a[i]:
                delta = a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # ensure nonnegatives
            # evaluate objective
            conv_new = _full_convolution(a_new, conv_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new
            # accept if improvement
            if val_new < best_val:
                best_val = val_new
                best_seq = a_new.copy()
                a = a_new
                last_improve = time.time()
                print(f"[refine] it={it} new best {val_new:.6f}")
                continue
            # else accept with probability based on temperature that decays
            # Simulated annealing
            # Temperature schedule T ~ initial_T / sqrt(it)
            T = 0.01 / np.sqrt(it)  # small
            if np.random.rand() < np.exp(-(val_new - best_val)/T):
                a = a_new
            # else reject

            if time.time() - last_improve > 10.0:
                # if no improvement for 10s, break
                break

    # final fallback uniform for safety
    if best_seq is None:
        best_seq = np.full(2048, 1.0/2048)

    return best_seq.tolist()

"""
Step‑function optimiser (alternative to the reference implementation).

The objective to minimise is
        F(a) = 2·n·max(a∗a) / (∑a)² .
Because the problem is homogeneous we can normalise ∑a = 1,
so that F(a) = 2·n·max(a∗a).

Algorithm
---------
1. Work on a cascade of lengths (256 … 16384).  For each length we build a
   diverse pool of initial vectors:
   – uniform,
   – an “arcsine” shape   a_i ∝ 1/√(i·(n‑i+1)),
   – a warm‑start from the best solution of the same size,
   – an up‑sampled version of the best shorter solution,
   – a few random Dirichlet draws,
   – edge‑concentrated power‑law families,
   – double‑exponential edge shapes.
2. Each start is optimised by Adam on a smooth‑max surrogate
   (log‑sum‑exp) of the convolution maximum.  The surrogate temperature
   λ is annealed (λ←λ·1.01) up to 1e7, which makes the surrogate converge to
   the true max.
3. After the Adam phase we run a long mirror‑descent / sub‑gradient
   refinement on the exact max.  The step size decays as 1/√t.
4. When the overall time budget is almost exhausted we take the current
   global best and perform a cheap stochastic local‑search (move a tiny
   amount of mass between two coordinates, simulated‑annealing acceptance)
   to escape possible plateaus.
5. The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns the convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr).real
    start = n - 1
    return corr[start:start + n]  # length n


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   best_val: float,
                   best_seq: np.ndarray) -> tuple:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max(a*a).
    Returns (new_best_val, new_best_seq, last_a, last_logits).
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    # Adam hyper‑parameters
    lr = 0.08
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    lam = 0.5            # smooth‑max temperature
    lam_factor = 1.02
    last_improve = time.time()

    # iteration budget (more iterations for smaller n)
    max_iters = int(30000 * 1024 / n) + 3000

    for it in range(1, max_iters + 1):
        # --- soft‑max to obtain a on the simplex ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- convolution and smooth‑max weights ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smooth‑max w.r.t. a ---
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑propagation through the soft‑max (mirror‑gradient)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update on logits ---
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        lam = min(lam * lam_factor, 1e7)

        # evaluate exact objective (∑a=1 ⇒ denominator = 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[len={n} Adam it={it}]  new best {val:.6f}")

        # stagnation handling – tiny random perturbation of logits
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        if time.time() > deadline:
            break

    # final a, logits (to be used for sub‑gradient phase)
    mx = logits.max()
    a = np.exp(logits - mx)
    a = a / a.sum()
    return best_val, best_seq, a, logits


def _subgradient_refine(a: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        corr_fft_len: int,
                        deadline: float,
                        best_val: float,
                        best_seq: np.ndarray) -> tuple:
    """
    Mirror‑descent / sub‑gradient refinement on the true max.
    Returns (new_best_val, new_best_seq, refined_a).
    """
    # Step‑size schedule η_t = η0 / sqrt(t)
    eta0 = 1.0
    sub_last = time.time()

    # iteration budget – generous for large n
    max_iters = int(50000 * 1024 / n) + 5000

    for t in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad_a)

        # renormalise, protect against numerical collapse
        s = a.sum()
        if s <= 0 or np.isnan(s) or np.isinf(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluate
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last = time.time()
            print(f"[len={n} Subgrad it={t}] new best {val:.6f}")

        # stagnation break
        if time.time() - sub_last > 6.0:
            break
        if time.time() > deadline:
            break

    return best_val, best_seq, a


def _local_refine(a: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  deadline: float,
                  best_val: float,
                  best_seq: np.ndarray) -> tuple:
    """
    Cheap stochastic local search (move a tiny amount of mass between
    two coordinates, Simulated‑Annealing acceptance) for the final few
    seconds of the budget.
    """
    last_improve = time.time()
    it = 0
    while time.time() < deadline - 0.5:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of mass from i to j
        delta = a[i] * np.random.uniform(0.001, 0.01)
        if delta > a[i]:
            delta = a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = _full_convolution(a_new, conv_fft_len)
        max_c = conv.max()
        val_new = 2.0 * n * max_c

        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
            print(f"[refine] it={it}  new best {val_new:.6f}")
            continue

        # simulated‑annealing acceptance (very low temperature)
        T = 0.01 / np.sqrt(it)
        if np.random.rand() < np.exp(-(val_new - best_val) / T):
            a = a_new

        if time.time() - last_improve > 10.0:
            # give up if no improvement for a while
            break

    return best_val, best_seq, a


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    best_val = float("inf")

        cand_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        idx = np.arange(1, n + 1, dtype=np.float64)
        shape = 1.0 / np.sqrt(idx * (n - idx + 1.0))
        shape = shape / shape.sum()
        cand_pool.append(shape)

        # warm‑start from exact same length (if we already have one)

        # up‑sample a shorter best solution (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum()
            cand_pool.append(up)

        # a few random Dirichlet draws
        for _ in range(3):
            cand_pool.append(np.random.dirichlet(np.ones(n)))

        # power‑law edge‑concentrated families

        for alpha in np.linspace(-2.5, -0.2, 5):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                cand_pool.append(shape)

        # double‑exponential edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.1, 0.3, 0.6, 1.5]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            cand_pool.append(shape)

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cand_id, a0 in enumerate(cand_pool):
            if time.time() > deadline:
                break

            # --- Adam on smooth‑max surrogate ---
            best_val, best_seq, a, logits = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )

            # --- Sub‑gradient refinement on the exact max ---
            best_val, best_seq, a = _subgradient_refine(
                a, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )

    # ----------------------------------------------------------------------
    # Final cheap stochastic refinement (uses the current global best)
    if best_seq is not None and time.time() < deadline:
        a = np.asarray(best_seq, dtype=np.float64)
        n = a.shape[0]
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        best_val, best_seq, _ = _local_refine(
            a, n, conv_fft_len, deadline, best_val, best_seq
        )

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # expose the solution for potential warm‑starts in a later run
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
