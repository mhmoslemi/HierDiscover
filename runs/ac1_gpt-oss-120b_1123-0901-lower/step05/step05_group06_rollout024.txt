def _project_onto_simplex(v, target_sum):
    # from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _subgradient_refine(a, n, deadline, best_score, rng):
    # a is numpy array with sum target_sum
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    it=0
    while time.time() < deadline:
        # compute conv via FFT
        Nfft = 1 << ((2*n-1).bit_length())
        A_fft = np.fft.rfft(cur_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        max_val = conv.max()
        # note: need to compute objective: 2*n*max_val/(target_sum**2) => but since target sum constant: compute D = 2*n*max_val/(target_sum**2)
        # we can compute later but max_val needed for gradient

        # find indices achieving max (within tolerance)
        max_inds = np.where(conv >= max_val - 1e-12)[0]
        # pick one maybe randomly if many
        k = rng.choice(max_inds)

        # compute rev: gradient = 2 * rev
        rev = np.zeros_like(cur_seq)
        if k < n:
            rev[:k+1] = cur_seq[k::-1]
        else:
            start = k - (n-1)
            rev[start:] = cur_seq[:2*n-1 - k][::-1]
        grad = 2.0 * rev

        # step size schedule
        it += 1
        eta = (target_sum * 0.2) / np.sqrt(it)   # adjust factor
        # gradient descent step (negative gradient)
        new_seq = cur_seq - eta * grad

        # project onto simplex of target sum
        new_seq = _project_onto_simplex(new_seq, target_sum)

        # compute new max
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv_new = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        new_max = conv_new.max()
        new_score = float(2*n*new_max/(target_sum**2))

        # acceptance: if better accept; else accept with SA
        if new_score < cur_score - 1e-12:
            cur_seq, cur_score = new_seq, new_score
            # maybe print improvement
            print(f"[subgrad] improvement: {cur_score:.6f}")
        else:
            # acceptance probability decaying
            T = max(0.01 * cur_score, 1e-5) * (1.0 - it / 500000)  # bound of steps
            prob = math.exp(-(new_score - cur_score) / max(T,1e-12))
            if rng.random() < prob:
                cur_seq, cur_score = new_seq, new_score
        # maybe jitter occasionally
        if it % 10000 == 0:
            # small random jitter on sequence (like add Gaussian noise then renorm)
            cur_seq += rng.normal(scale=0.01*target_sum/n, size=n)
            cur_seq = _project_onto_simplex(cur_seq, target_sum)
    return cur_seq, cur_score

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Diverse starts** – for each chosen length we generate a handful of
   normalised vectors (uniform, Dirichlet, exponential, windowed, interpolated
   from a previous best …).

2. **Smooth max‑Adam optimisation** – we optimise the *log‑softmax* of the
   coefficients with Adam while approximating the max‑convolution by a
   temperature‑controlled soft‑max (log‑sum‑exp).  This gives a fast descent
   to a good basin.

3. **Projected sub‑gradient refinement** – the exact max‑convolution is used.
   The sub‑gradient (2 · reverse‑convolution at the maximal shift) is taken,
   a step is performed and the vector is projected back onto the probability
   simplex (sum = √(2 n)).  A simple simulated‑annealing acceptance rule
   helps escape shallow plateaux.  This stage can dramatically improve the
   final bound.

If time permits the best vector is *up‑sampled* to a larger length (up to 8000)
and refined again, because a longer sequence provides more degrees of freedom
and often yields a smaller max‑convolution.

The whole search respects the 1000 s budget; the final vector is returned as
a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers -------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = target_sum}.  Implementation follows Duchi et al.
    """
    if target_sum <= 0:
        return np.zeros_like(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = cssv[-1] / len(v)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max over the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution  rev_i = Σ_k w_k·a_{k‑i}."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]          # 3 n‑2 length
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """A small pool of diverse, normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    arr = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * arr / (n - 1)) + a2 * np.cos(4.0 * np.pi * arr / (n - 1))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(0.0, 0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _interp_to_length(old: np.ndarray, new_n: int, target_sum: float) -> np.ndarray:
    """Linear interpolation of a vector to a new length, preserving non‑negativity."""
    old_n = old.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(new_x, old_x, old)
    interp = np.maximum(interp, 0.0)
    interp = interp / interp.sum() * target_sum
    return interp

def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) with a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen in this
    run together with its exact objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    # temperature schedule
    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    # FFT size for linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current vector (softmax of logits)
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # exact self‑convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # soft‑max weights for the max
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # weighted reverse convolution (gradient of the smoothed max)
        rev = _weighted_rev(a, w, Nfft)

        # gradient wrt a (2·rev)
        g_a = 2.0 * rev

        # chain rule to logits (mirror‑descent on the simplex)
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % jitter_interval == 0:
            y += rng.normal(scale=0.01, size=n)

        # keep the best exact score seen so far
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _subgradient_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    The step direction is 2·rev where rev corresponds to the current maximal
    shift(s).  After each step we project onto the simplex Σa = √(2 n).
    A simple simulated‑annealing acceptance rule helps to escape flat regions.
    """
    target_sum = math.sqrt(2.0 * n)
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for convolution (length 2 n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    step = 0
    max_steps = 500_000  # a soft cap; the loop also stops at the deadline
    while time.time() < deadline and step < max_steps:
        # exact convolution
        A_fft = np.fft.rfft(cur_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        max_val = conv.max()
        # exact objective (same as max_val because sum = √(2 n))
        cur_obj = max_val

        # locate all maximal indices (allow tiny tolerance)
        mask = conv >= max_val - 1e-12
        max_inds = np.nonzero(mask)[0]
        # pick a random one among them
        k = rng.choice(max_inds)

        # compute rev_i = a_{k-i} (zero outside range) – O(n) slice
        rev = np.zeros_like(cur_seq)
        if k < n:
            rev[: k + 1] = cur_seq[k :: -1]
        else:
            start = k - (n - 1)
            rev[start:] = cur_seq[: 2 * n - 1 - k][::-1]

        grad = 2.0 * rev

        # step size – decreasing like 1/√t
        step += 1
        eta = (target_sum * 0.25) / math.sqrt(step)

        new_seq = cur_seq - eta * grad
        new_seq = _project_onto_simplex(new_seq, target_sum)

        # evaluate new objective
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv_new = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_obj = conv_new.max()

        # deterministic improvement
        if new_obj < cur_obj - 1e-12:
            cur_seq, cur_obj = new_seq, new_obj
            cur_score = float(2 * n * new_obj / (target_sum ** 2))
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_obj - cur_obj) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_obj = new_obj
                cur_score = float(2 * n * new_obj / (target_sum ** 2))

        # occasional random jitter to shake the system
        if step % 10000 == 0:
            cur_seq += rng.normal(scale=0.005 * target_sum / n, size=n)
            cur_seq = _project_onto_simplex(cur_seq, target_sum)

    return cur_seq, cur_score
