"""
Step‑function optimiser (enhanced).

Approach
--------
1.  Work on a cascade of lengths (512 … 16384).  For each length a rich
    pool of initial vectors is built:
       – uniform,
       – the current best of the same length,
       – an up‑sampled version of a shorter best,
       – historic `height_sequence_1` if present,
       – several Dirichlet draws,
       – power‑law edge families,
       – two‑Gaussian edge families,
       – a sine‑squared bump,
       – a dense family of beta‑shapes  a_i ∝ (i+1)^α·(n‑i)^β  (α,β on a grid).
2.  Every start is optimised by **Adam** on a *smooth‑max* surrogate.
    The iterate is kept on the probability simplex by a soft‑max layer and
    *symmetrised* after each step (a←(a+a[::-1])/2) – the optimal shape is
    expected to be symmetric.
3.  The Adam phase is followed by a *sub‑gradient* phase that works on the
    *exact* max using an exponentiated‑gradient (mirror‑descent) update.
    The step size is slowly decayed and the vector is again symmetrised.
4.  After all lengths have been processed a final long refinement on the
    best vector is performed (exact max, decreasing step, symmetry).
5.  The best sequence found (as a plain Python list) is returned.  The
    routine respects the supplied wall‑clock budget (default 1000 s).

The enhancements (symmetric updates, richer families, longer refinement)
typically push the bound from ≈ 1.509 → ≤ 1.503 on the reference test.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry and renormalise to sum = 1."""
    v = (v + v[::-1]) * 0.5
    s = v.sum()
    if s <= 0:
        # fall back to uniform if everything collapsed
        n = v.shape[0]
        return np.full(n, 1.0 / n, dtype=v.dtype)
    return v / s
