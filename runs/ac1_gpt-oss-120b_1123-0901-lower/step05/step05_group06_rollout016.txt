def _generate_parametric_vectors(n, target_sum, rng):
    vectors = []
    x = np.arange(n)
    center = (n-1)/2.0
    # Gaussian sigmas
    for sigma_factor in [0.15, 0.2, 0.25, 0.33, 0.5]:
        sigma = sigma_factor * n
        g = np.exp(-0.5 * ((x - center)/sigma)**2)
        if g.sum() == 0: continue
        g = g / g.sum() * target_sum
        vectors.append(g)
    # Cosine taper window (raised cosine)
    for alpha in [0.2, 0.5, 0.8]:
        w = np.ones(n)
        l = int(alpha * n)
        # left taper
        if l > 0:
            t = np.arange(l) / l
            w[:l] = 0.5 * (1 - np.cos(np.pi * t))
            w[-l:] = 0.5 * (1 - np.cos(np.pi * (1 - t[::-1])))
        w = np.maximum(w, 0)
        w = w / w.sum() * target_sum
        vectors.append(w)
    # Exponential tails
    for rate in [0.001, 0.002, 0.005]:
        exp = np.exp(-rate * np.arange(n))
        exp = exp / exp.sum() * target_sum
        vectors.append(exp)
        vectors.append(exp[::-1])
    # random sparsity pattern: random set of positions
    for _ in range(3):
        mask = rng.random(n) < 0.2
        vals = rng.random(mask.sum())
        if vals.size == 0:
            continue
        seq = np.zeros(n)
        seq[mask] = vals
        seq = seq / seq.sum() * target_sum
        vectors.append(seq)
    return vectors

def _max_convolution_value(a):
    n = a.shape[0]

    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2 * n - 1]
    return float(conv.max())

"""
Propose a step‑function (non‑negative vector) that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Choose several lengths (up to 8000).  For each length generate a *diverse*
    pool of initial vectors:
        – classical starters (uniform, Dirichlet, windows …)
        – parametric families (Gaussian, cosine‑taper, exponential, sparse)
        – linear interpolation of the previously discovered best sequence
        – a few random sparse vectors.
2.  Quickly evaluate every starter by the exact max‑convolution (FFT) and keep
    the best few per length.
3.  Refine each kept starter with a soft‑max Adam optimiser on the log‑its
    (mirror‑descent on the simplex).  The temperature τ controlling the smooth
    max is slowly annealed; occasional jitter on the logits helps escaping
    plateaus.  The optimiser works directly with the exact convolution and
    its weighted reverse‑convolution gradient.
4.  After the global time budget for the Adam phase is exhausted, perform a
    cheap stochastic hill‑climbing on the current global best:
    * repeatedly move a small fraction of mass from a source index to a
      destination index,
    * accept every improvement and otherwise accept with a decaying
      simulated‑annealing probability.
5.  Return the best vector discovered (plain Python list).  All numbers are
    non‑negative; the optimiser enforces the normalisation
    ``sum(a) = sqrt(2·n)`` which makes the evaluation equal to the maximal
    convolution value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Exact maximum of the linear convolution a * a using an FFT.
    """
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Weighted reverse convolution needed for the gradient of the
    (soft‑max) smoothed max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]          # length = 3n‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


# ----------------------------------------------------------------------
# Initial vector factories ----------------------------------------------

def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """
    Classic diverse starts (uniform, Dirichlet, windows, etc.).
    """
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n,
                                                  size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _generate_parametric_vectors(n: int,
                                 target_sum: float,
                                 rng: np.random.Generator):
    """
    Additional families of smooth shapes (Gaussian, cosine‑taper,
    exponential tails, sparse random).
    """
    vectors = []
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0

    # Gaussian windows with several widths
    for sigma_factor in (0.12, 0.18, 0.25, 0.35, 0.45):
        sigma = sigma_factor * n
        g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        if g.sum() > 0:
            vectors.append(g / g.sum() * target_sum)

    # Cosine‑taper (raised‑cosine) windows
    for alpha in (0.1, 0.3, 0.5, 0.7):
        w = np.ones(n)
        l = int(alpha * n)
        if l > 0:
            t = np.arange(l) / l
            w[:l] = 0.5 * (1 - np.cos(np.pi * t))
            w[-l:] = 0.5 * (1 - np.cos(np.pi * (1 - t[::-1])))
        vectors.append(w / w.sum() * target_sum)

    # Exponential tails (both directions)
    for rate in (0.0008, 0.0015, 0.003):
        e = np.exp(-rate * x)
        vectors.append(e / e.sum() * target_sum)
        vectors.append(e[::-1] / e[::-1].sum() * target_sum)

    # Random sparse vectors (≈20 % non‑zero)
    for _ in range(4):
        mask = rng.random(n) < 0.2
        if not mask.any():
            continue
        vals = rng.random(mask.sum())
        v = np.zeros(n)
        v[mask] = vals
        vectors.append(v / v.sum() * target_sum)

    return vectors


# ----------------------------------------------------------------------
# Optimiser (soft‑max Adam on the log‑its) ------------------------------

def _exp_gradient_optimize(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int,
                           best_seq: np.ndarray,
                           best_score: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the log‑its with a temperature‑controlled
    soft‑max approximation of ``max(a*a)``.
    """
    target_sum = math.sqrt(2.0 * n)

    # Initialise logits from the start vector
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.09                     # base learning rate (log‑its)

    # temperature schedule
    tau_factor = 0.28
    decay_rate = max_iters / 7.0   # exponential decay

    # FFT size for linear convolution (length 3n‑2)
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute current distribution (softmax on logits) -----
        logits = y - np.max(y)                # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()     # on the simplex
        a = target_sum * p                    # guarantees the normalisation

        # ----- exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max weights for the (smooth) max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- gradient wrt a (2·rev) -----
        rev = _weighted_rev(a, w, Nfft)
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on simplex) -----
        avg = np.dot(p, g_a)                # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)  # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape flat regions
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            y += jitter

        # keep the best vector seen so far
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local stochastic refinement -------------------------------------------

def _local_refine(a: np.ndarray,
                  n: int,
                  deadline: float,
                  best_score: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing:
    move a small random fraction of mass from a source index to a destination
    index, always accepting improvements and otherwise using a decaying
    simulated‑annealing probability.
    """
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size (only needed for the convolution)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # Simulated‑annealing schedule
    max_steps = 250_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 % – 6 %)
        frac = rng.uniform(0.01, 0.06)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            if interp.sum() > 0:

        scores = []

    if best_seq is not None and time.time() < deadline:
