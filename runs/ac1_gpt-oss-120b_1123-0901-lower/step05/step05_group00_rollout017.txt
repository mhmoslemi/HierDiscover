def obj_and_grad(logits):
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    # compute smooth max with lam: use logsumexp
    # We can compute logsumexp with stable method:
    lse = max_c + np.log(np.exp(lam*(conv - max_c)).sum())
    f = lse / lam   # smoothed max
    # gradient in a space: grad_a = 2 * cross_correlation(w, a) where w_i ∝ exp(lam*(conv_i - max_c))
    w = np.exp(lam*(conv - max_c))
    w = w / w.sum()
    grad_a = 2 * _cross_correlation(w, a, fft_len=corr_fft_len)
    # transform to logits gradient
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return f, grad_log

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_val = float('inf')
    best_seq = None

    lengths = [1024, 2048, 4096, 8192, 16384, 32768] # could include 65536 but may be heavy.

    for n in lengths:
        if time.time() > deadline: break
        # compute FFT lengths
        conv_fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3*n-2)))
        # generate candidates list
        candidates = []
        # uniform
        candidates.append(np.full(n, 1.0/n))
        # warm start from best same n
        if best_seq is not None and len(best_seq)==n:
            candidates.append(np.array(best_seq))
        # upsample shorter best
        if best_seq is not None and len(best_seq) < n:
            up = np.interp(np.linspace(0,1,n), np.linspace(0,1,len(best_seq)), best_seq)
            candidates.append(up/up.sum())
        # add height_sequence_1 if exists
        hseq = globals().get('height_sequence_1')
        if hseq is not None:
            h = np.array(hseq, dtype=float)
            # crop/pad
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]), mode='constant')
            h = np.maximum(h, 0.0)
            if h.sum()>0:
                h = h/h.sum()
            else:
                h = np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # random Dirichlet draws for quick screening
        random_candidates = []
        for _ in range(30):
            cand = np.random.dirichlet(np.ones(n))
            random_candidates.append(cand)
        # evaluate quick objective and keep best 5
        vals = []
        for cand in random_candidates:
            conv = _full_convolution(cand, fft_len=conv_fft_len)
            val = 2.0 * n * conv.max()
            vals.append(val)
        sorted_idx = np.argsort(vals)[:5]
        for idx in sorted_idx:
            candidates.append(random_candidates[idx])

        # Add parametric families: power-law edges, power-law center, Gaussian, sinusoidal, Beta shapes.
        idx = np.arange(1, n+1, dtype=float)
        prod = idx * (n - idx + 1)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod>0, np.power(prod, alpha), 0.0)
            if shape.sum()>0:
                shape = shape/shape.sum()
                candidates.append(shape)
        for alpha in np.linspace(0.2, 2.0, 4):
            shape_center = np.where(prod>0, np.power(prod, alpha), 0.0)
            if shape_center.sum()>0:
                shape_center = shape_center/shape_center.sum()
                candidates.append(shape_center)
        # Two-Gaussian
        i = np.arange(n, dtype=float)
        for gamma in [0.2,0.5,1.0,2.0]:
            shape = np.exp(-gamma*i*i) + np.exp(-gamma*(n-1-i)*(n-1-i))
            shape = shape/shape.sum()
            candidates.append(shape)
        # Sinusoidal powers
        base = np.sin(np.pi * (np.arange(n)+0.5)/n)
        for p in [1,2,4]:
            shape = base**p
            shape = shape/shape.sum()
            candidates.append(shape)
        # Beta distributions
        x = (np.arange(n)+0.5)/n
        for a_param in [0.5, 0.8, 1.2, 1.5, 2.0]:
            b_param = a_param
            shape = np.power(x, a_param-1)*np.power(1-x, b_param-1)
            if shape.sum()>0:
                shape = shape/shape.sum()
                candidates.append(shape)

        # Remove duplicates approximate
        # Not necessary.

        # Optimize each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break
            # ensure nonneg and normalized
            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            # param via logits
            logits = np.log(a)

            # Adam phase
            m = np.zeros_like(logits); v = np.zeros_like(logits)
            beta1=0.9; beta2=0.999; eps_adam=1e-8
            lr=0.05
            lam=0.5
            lam_factor=1.01
            last_improve=time.time()
            max_iters=int(25000*1024/n)+2000
            for it in range(1, max_iters+1):
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam*(conv-max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0/w.size)
                else:
                    w = w / w_sum
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                m = beta1*m + (1-beta1)*grad_log
                v = beta2*v + (1-beta2)*(grad_log**2)
                m_hat = m/(1-beta1**it)
                v_hat = v/(1-beta2**it)
                logits -= lr * m_hat/(np.sqrt(v_hat)+eps_adam)
                lam = min(lam*lam_factor, 1e7)
                val = 2.0*n*max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it}] new best {val:.6f}")
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n)-0.5)*1e-3
                    last_improve = time.time()
                if time.time() > deadline:
                    break

            # L-BFGS refinement with larger lam to approximate true max
            # Use final logits as start
            try:
                import scipy.optimize as opt
                # Use lam large, e.g., lam=10.
                lam_ref = 10.0
                # Define closure that accesses a, conv_fft_len, corr_fft_len
                def f_and_g(logits_vec):
                    mx = logits_vec.max()
                    exp_log = np.exp(logits_vec - mx)
                    a_vec = exp_log / exp_log.sum()
                    conv = _full_convolution(a_vec, fft_len=conv_fft_len)
                    max_c = conv.max()
                    # compute smooth max
                    lse = max_c + np.log(np.exp(lam_ref*(conv - max_c)).sum())
                    f_val = lse/lam_ref
                    w = np.exp(lam_ref*(conv-max_c))
                    w = w / w.sum()
                    grad_a = 2.0 * _cross_correlation(w, a_vec, fft_len=corr_fft_len)
                    avg = np.dot(a_vec, grad_a)
                    grad_log = a_vec * (grad_a - avg)
                    return f_val, grad_log
                res = opt.minimize(f_and_g, logits, method='L-BFGS-B', jac=True,
                                   options={'maxiter': 500, 'disp': False})
                logits = res.x
                # compute final a
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()
            except Exception as e:
                # if SciPy not available, skip
                pass

            # Subgradient refinement
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(2000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0/active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0*n*max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")
                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

    # fallback
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0/fallback_n)

    return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)**2)

where * denotes linear convolution.  The algorithm is a multi‑scale
stochastic optimiser:

  * Process a cascade of lengths (small → large).  Larger lengths give
    a finer discretisation of the continuous problem and typically lower
    the bound.
  * For each length generate a diverse pool of start points:
        – uniform,
        – warm‑starts (previous best, up‑sampled shorter best,
          historic `height_sequence_1`),
        – random Dirichlet draws,
        – analytic families (edge‑power‑law, centre‑power‑law,
          Gaussian bumps, sinusoidal bumps, symmetric Beta shapes).
    The best few random draws (by the exact objective) are kept.
  * Optimise each start with three phases:
        1. Adam on a smooth‑max surrogate (log‑sum‑exp) of the
           convolution maximum.  The surrogate temperature λ is slowly
           increased, turning the problem into the true max.
        2. A short L‑BFGS‑B refinement on the same surrogate with a
           larger λ (closer to the true max).
        3. An exact sub‑gradient descent on the true max using an
           exponentiated‑gradient (mirror‑descent) step.
  * After the cascade, perform a cheap local random‑move search
    (mass transfer between two coordinates) for a few seconds to
    escape any remaining plateau.
  * All steps respect a wall‑clock budget (default 1000 s) and the best
    vector found is returned as a plain Python list.

The implementation uses only NumPy, SciPy (for L‑BFGS‑B) and standard
Python libraries and works on any machine where the evaluator
`evaluate_sequence` is available.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution / cross‑correlation (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine respects the given wall‑clock budget and returns the best
    sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Length schedule (small → large).  Larger lengths give a finer grid
    # and therefore allow a smaller bound, at the cost of slower per‑step.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))   # (2n‑1)+(n‑1)

        # --------------------------------------------------------------
        # Build a rich pool of start points
        # --------------------------------------------------------------
        candidates = []

        # 1. Uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2. Warm‑start of exactly this size (if we already have one)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 3. Upsampled shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            if up.sum() > 0:
                up /= up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 4. Historic global variable (if present)
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]), mode='constant')
            h = np.maximum(h, 0.0)
            if h.sum() > 0:
                h /= h.sum()
            else:
                h = np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # 5. Random Dirichlet draws (screened cheaply)
        random_pool = []
        random_vals = []
        for _ in range(30):
            cand = np.random.dirichlet(np.ones(n))
            random_pool.append(cand)
            conv = _full_convolution(cand, fft_len=conv_fft_len)
            val = 2.0 * n * conv.max()
            random_vals.append(val)
        # keep the 5 best random draws
        best_random_idx = np.argsort(random_vals)[:5]
        for idx in best_random_idx:
            candidates.append(random_pool[idx])

        # 6. Edge‑concentrated power‑law families (negative exponent)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # 7. Centre‑concentrated power‑law families (positive exponent)
        for alpha in np.linspace(0.2, 2.0, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # 8. Two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            candidates.append(shape)

        # 9. Smooth sine‑squared / sine‑quartic bumps
        base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        for p in [1, 2, 4]:
            shape = base ** p
            shape /= shape.sum()
            candidates.append(shape)

        # 10. Symmetric Beta distributions (a = b)
        x = (np.arange(n) + 0.5) / n
        for a_beta in [0.5, 0.8, 1.2, 1.5, 2.0]:
            shape = np.power(x, a_beta - 1) * np.power(1 - x, a_beta - 1)
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # --------------------------------------------------------------
        # Optimise each candidate
        # --------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure positivity, normalise to the simplex
            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            logits = np.log(a)

            # ---------------------- Adam on smooth‑max -----------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            lam = 0.5                # start temperature
            lam_factor = 1.01
            last_improve = time.time()
            max_iters = int(25000 * 1024 / n) + 2000   # fewer iters for large n

            for it in range(1, max_iters + 1):
                # soft‑max ⇒ simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # back‑propagation through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (sum is 1 ⇒ denominator = 1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it}] new best {val:.6f}")

                # stagnation handling – tiny random shake
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---------------------- L‑BFGS‑B refinement -----------------
            # Use a larger temperature (closer to the true max) and let
            # SciPy's quasi‑Newton method polish the result.
            try:
                import scipy.optimize as opt

                lam_ref = 10.0

                def f_and_g(logits_vec):
                    mx = logits_vec.max()
                    exp_log = np.exp(logits_vec - mx)
                    a_vec = exp_log / exp_log.sum()
                    conv = _full_convolution(a_vec, fft_len=conv_fft_len)
                    max_c = conv.max()
                    # smooth‑max = log‑sum‑exp / λ
                    lse = max_c + np.log(np.exp(lam_ref * (conv - max_c)).sum())
                    f_val = lse / lam_ref

                    w = np.exp(lam_ref * (conv - max_c))
                    w /= w.sum()
                    grad_a = 2.0 * _cross_correlation(w, a_vec, fft_len=corr_fft_len)
                    avg = np.dot(a_vec, grad_a)
                    grad_log = a_vec * (grad_a - avg)
                    return f_val, grad_log

                res = opt.minimize(
                    f_and_g,
                    logits,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": 500, "disp": False},
                )
                logits = res.x
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()
            except Exception:
                # SciPy not available – silently skip this phase.
                pass

            # ---------------------- Exact sub‑gradient -----------------
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(2000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                # active set of maximal entries
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

        # --------------------------------------------------------------
        # Optional cheap local random‑move refinement for the current best
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_local = np.asarray(best_seq, dtype=np.float64)
            while time.time() < deadline - 0.5:
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                # transfer a small amount of mass from i to j
                eps = 0.02 * a_local[i] * np.random.rand()
                if eps <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps
                a_new[j] += eps
                if a_new[i] < 0.0:
                    continue
                # re‑normalise (tiny drift)
                a_new /= a_new.sum()
                conv = _full_convolution(a_new, fft_len=conv_fft_len)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a_new.copy()
                    a_local = a_new
                    print(f"[n={n} local move] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – return a uniform vector if nothing was discovered.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list (required by the external evaluator)
    return best_seq.tolist()
