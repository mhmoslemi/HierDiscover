import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

def lp_step(f, M, target_sum):
    n = len(f)
    # Build convolution matrix rows: each row k (0..2n-2) has elements f_i for i in [max(0,k-n+1), min(n-1,k)]
    # Actually conv[k] = sum_i f_i * g_{k-i} where g index must be within [0,n-1].
    # So for given f, row k: column j = f_{k-j} if 0 <= k-j < n.
    # So we can build matrix rows as shifted copies of f.
    # Use sparse diag for each shift.
    data = []
    rows = []
    cols = []
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(n-1, k)  # inclusive
        # j = k - i
        # For each i in [i_start,i_end], column j = i, coefficient f[i_start? Wait we need coefficient of g_j, with j = k - i
        # Actually row k * g = sum_i f_i * g_{k-i} = sum_{j} f_{k-j} * g_j, where j in allowed range.
        # So column j corresponds to g_j, coefficient = f_{k-j} if 0 <= k-j < n.
        # So for j range: max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)
        # j_start <= j <= j_end
        # For each j, coefficient = f_{k-j}
        # So we can iterate j.
        for j in range(j_start, j_end+1):
            coeff = f[k - j]
            rows.append(k)
            cols.append(j)
            data.append(coeff)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ g <= M, g >= 0
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # bounds: (0, None) each variable
    bounds = [(0, None) for _ in range(n)]
    # Use linprog with high accuracy but early tolerance
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve':True, 'time_limit': ??? })
    if not res.success:
        return None
    g = res.x
    # Scale to target sum
    sum_g = g.sum()
    if sum_g == 0:
        return None
    g = g * (target_sum / sum_g)
    return g
```

def _deterministic_refine(a, deadline, rng):
    n = a.shape[0]
    target_sum = a.sum()
    cur = a.copy()
    cur_score = _max_convolution(cur, 1 << ((2*n-1).bit_length())) # compute exact max conv
    # Use top-lag approach
    Nfft = 1 << ((2*n-1).bit_length())
    step = 0
    while time.time() < deadline and step < 500_000:
        conv_all = np.fft.irfft(np.fft.rfft(cur, n=Nfft) * np.fft.rfft(cur, n=Nfft), n=Nfft)[:2*n-1]
        max_val = conv_all.max()
        # find k's near max within tolerance
        tol = 1e-9
        active = np.where(conv_all >= max_val - tol)[0]
        # or pick top few lags
        if len(active) > 5:
            active = active[:5]
        # aggregated influence
        weight = np.zeros_like(cur)
        for k in active:
            # contribution vector for lag k: reversed shift of cur
            # The influence of each index i for this lag is cur[k-i] (if within bounds)
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            # i ranges from lo to hi, weight[i] += cur[k-i]
            # vectorized:
            idx = np.arange(lo, hi+1)
            weight[idx] += cur[k - idx]
        # Choose source i_max where weight large and cur large also, and target i_min where weight small
        # compute combined score: cur[i] * weight[i]
        src_score = cur * weight
        i_max = np.argmax(src_score)
        # exclude i_max from target selection
        # compute target weight (small weight)
        i_min_candidates = np.where(cur > 0)[0]  # any index could receive mass
        # we want smallest weight
        i_min = i_min_candidates[np.argmin(weight[i_min_candidates])]
        if i_max == i_min:
            # pick second best
            sorted_src = np.argsort(-src_score)
            for cand in sorted_src:
                if cand != i_min:
                    i_max = cand
                    break
        # transfer small fraction
        frac = rng.uniform(0.005, 0.02)
        delta = cur[i_max] * frac
        if delta <= 0:
            step += 1
            continue
        new_seq = cur.copy()
        new_seq[i_max] -= delta
        new_seq[i_min] += delta
        new_score = _max_convolution(new_seq, Nfft)
        if new_score < cur_score - 1e-12:
            cur = new_seq
            cur_score = new_score
            print(f"[deterministic] improved score {cur_score:.6f} step {step}")
        else:
            # accept some deterioration with small prob
            T = cur_score * 0.01 * (1 - step/500_000)
            if rng.random() < math.exp(-(new_score - cur_score)/max(T, 1e-12)):
                cur = new_seq
                cur_score = new_score
        step += 1
    return cur, cur_score

"""
Hybrid gradient‑Adam + deterministic max‑convolution flattening.

The objective of ``evaluate_sequence`` is simply ``max(a∗a)`` when the
sequence satisfies the required normalization Σa_i = √(2 n).  This code
searches over several lengths, starts from a diverse pool of
initializations and runs an Adam optimiser on a smooth approximation of
the max‑convolution (soft‑max).  After the Adam phase a cheap,
deterministic “flatten‑lag’’ refinement repeatedly moves a small amount
of mass from indices that contribute most to the currently largest
convolution lag toward indices that contribute least.  The two phases
are repeated for many random starts and the best vector seen is
returned.

The implementation uses only NumPy (no external solvers) and respects
the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Gradient of the (soft‑max) max‑convolution w.r.t. ``a``.
    Implements the reverse weighted convolution by FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)      # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution (length = len(w)+len(a)-1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # take the appropriate slice to obtain the gradient vector
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact maximum of ``a * a`` (linear convolution)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small, diverse set of start vectors for length ``n``."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    if tri.sum() > 0:
        init.append(tri / tri.sum() * target_sum)

    return init


# ----------------------------------------------------------------------
# Phase‑A: Adam optimisation on a smooth max‑convolution

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   rng: np.random.Generator):
    """Adam optimiser on the log‑its using a soft‑max approximation of max‑conv."""
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from a0 (add epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.10                     # base learning rate on logits

    # temperature schedule parameters
    tau_factor = 0.45               # τ₀ = τ_factor·M
    decay_iters = max_iters / 6.0   # slower decay produces smoother optimisation

    # FFT size – must accommodate linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    best_seq = None
    best_score = float('inf')

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- current vector (softmax of logits) ----
        logits = y - np.max(y)               # shift for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()    # probability simplex
        a = target_sum * p                   # guaranteed Σa = target_sum, a_i≥0

        # ---- exact convolution ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- soft‑max weights (smooth max) ----
        tau = tau_factor * M * math.exp(-it / decay_iters)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---- weighted reverse convolution (gradient of smoothed max) ----
        rev = _weighted_rev(a, w, Nfft)

        # ---- gradient w.r.t. a (2·rev) ----
        g_a = 2.0 * rev

        # ---- chain rule to logits (mirror‑descent on the simplex) ----
        avg = np.dot(p, g_a)                 # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % 2500 == 0:
            y += rng.normal(scale=0.02, size=n)

        # ---- keep best exact score seen so far ----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best max = {best_score:.6f}")

    return best_seq, best_score

# Phase‑B: deterministic flattening of the dominant lag(s)

def _flatten_lag_refine(a: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator):
    """Move a small fraction of mass from indices that contribute most to
    the current maximum convolution lag(s) toward indices that contribute
    least.  Deterministic, fast, and works well as a final polish."""
    n = a.shape[0]
    target_sum = a.sum()                     # already √(2 n)
    cur = a.copy()
    # FFT size for exact max‑convolution queries
    Nfft = 1 << ((2 * n - 1).bit_length())
    cur_score = _max_convolution(cur, Nfft)

    step = 0
    max_steps = 400_000                       # generous but bounded

    while time.time() < deadline and step < max_steps:
        # recompute full convolution (cost O(n log n))
        A_fft = np.fft.rfft(cur, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        max_val = conv.max()
        # indices of the 3–5 largest lags (active set)
        top_lags = np.argpartition(-conv, range(5))[:5]

        # build aggregated influence vector
        weight = np.zeros_like(cur)
        for k in top_lags:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            weight[idx] += cur[k - idx]

        # combine with current magnitude – indices with large a_i·weight_i are prime donors
        donor_score = cur * weight
        i_src = int(np.argmax(donor_score))

        # choose a recipient with the smallest weight (allow any index)
        # avoid the trivial case i_src == i_tgt
        candidates = np.where(cur > 0)[0]
        if len(candidates) == 0:
            break
        i_tgt = int(candidates[np.argmin(weight[candidates])])
        if i_src == i_tgt:
            # fall back to second‑best donor
            sorted_src = np.argsort(-donor_score)
            for idx in sorted_src:
                if idx != i_tgt:
                    i_src = int(idx)
                    break

        # transfer a small fraction of the source mass
        frac = rng.uniform(0.006, 0.022)          # 0.6 % – 2.2 %
        delta = cur[i_src] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur.copy()
        new_seq[i_src] -= delta
        new_seq[i_tgt] += delta

        new_score = _max_convolution(new_seq, Nfft)

        # accept strict improvements; otherwise accept with tiny probability
        if new_score < cur_score - 1e-12:
            cur = new_seq
            cur_score = new_score
            print(f"[flatten] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-6) * (1.0 - step / max_steps)
            if rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                cur = new_seq
                cur_score = new_score
        step += 1

    return cur, cur_score


# ----------------------------------------------------------------------
# Phase‑C: cheap stochastic hill‑climbing (fallback if time remains)

def _stochastic_refine(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_score: float,
                       rng: np.random.Generator):
    """Same style as the original code – random mass transfers with simulated‑annealing."""
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())
    step = 0
    max_steps = 250_000
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_score = _max_convolution(new_seq, Nfft)
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[stoch] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            if rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                cur_seq = new_seq
                cur_score = new_score
        step += 1
    return cur_seq, cur_score

# Main entry point required by the harness

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load any previously discovered construction
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    n0 = base.shape[0] if base is not None else 2100

    # choose a handful of lengths (up to the practical limit of 8000)
    length_factors = [1.0, 1.5, 2.0, 2.5, 3.0]
    lengths = sorted({max(500, min(int(n0 * f), 8000)) for f in length_factors})
    if 8000 not in lengths:
        lengths.append(8000)

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Phase‑A: Adam optimisation on several lengths
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # diverse starts; also interpolate the historic best if shapes differ
        init_vectors = _make_initial_vectors(n, target_sum, rng)
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # allocate a sensible iteration budget (more for small n)
        remaining = deadline - time.time()
        approx_iters = int(min(2_500_000 * 2100 / n, 500_000, remaining * 150))
        max_iters = max(200_000, approx_iters)

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for start_idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {start_idx+1}/{len(init_vectors)} for length {n}")
            seq, score = _adam_optimize(a0.copy(), n, deadline, max_iters, rng)
            if seq is not None and score < best_score:
                best_seq = seq
                best_score = score
                print(f"[global] new best max={best_score:.6f} (len {n})")

    # ------------------------------------------------------------------
    # Phase‑B: deterministic flatten‑lag refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 25.0)
        best_seq, best_score = _flatten_lag_refine(best_seq, refine_deadline, rng)
        print(f"[post‑flatten] best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑C: cheap stochastic refinement (fallback)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        best_seq, best_score = _stochastic_refine(best_seq,
                                                  best_seq.shape[0],
                                                  time.time() + remaining,
                                                  best_score,
                                                  rng)
        print(f"[post‑stochastic] best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform sequence if nothing was found (should never happen)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        print("[fallback] using uniform sequence")

    print(f"[done] best max = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
