```
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

for a non‑negative vector `a`.  
Because the objective is homogeneous, we may fix the ℓ‑1 norm to  

\[
\sum_i a_i=\sqrt{2n},
\]

so that `F(a)` is simply the maximal entry of the (linear) self‑convolution
`c = a * a`.  The problem is non‑convex; we therefore use a **stochastic
mirror‑descent** on the simplex together with a smooth approximation of the
max‑operator and a cheap simulated‑annealing refinement.

Key ideas

| Idea | Why it helps |
|------|--------------|
| **Soft‑max (log‑sum‑exp) smoothing** of the max‑convolution – differentiable, provides a useful gradient. | Allows gradient‑based optimisation while still driving the true max down. |
| **Mirror‑descent on the probability simplex** (logits → soft‑max) – enforces non‑negativity and the ℓ‑1‑norm automatically. | Guarantees feasibility without explicit projections. |
| **Adam optimiser** on the logits – adaptive step‑sizes, fast convergence. | Handles the ill‑scaled landscape of the problem. |
| **Temperature schedule** – start with a large τ (very smooth) and anneal exponentially. | Prevents early plateaus and later focuses on the true max. |
| **Jitter & random restarts** – occasional random perturbations of the logits. | Helps escaping shallow local minima. |
| **Highly diversified initial vectors** – uniform, Dirichlet (dense & sparse), exponential / power‑law decays, triangular, random sparse patterns and interpolations of any known solution. | Gives the optimiser many different basins of attraction. |
| **Local stochastic refinement** – repeatedly transfer a tiny amount of mass between two positions, accepting worsening moves with a cooling schedule. | Removes the last “noise’’ left by the gradient phase. |
| **Dynamic time allocation** – the global time budget is split automatically between many short optimisation runs on several lengths (300 … 8000). | Explores the dependence on the sequence length without exhausting the budget. |

The code below follows the algorithm described above.  It is completely
self‑contained, uses only `numpy`/`math`/`time`, and returns the best
sequence found before the 1000 s deadline.

```python
"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm
---------
* Choose a set of candidate lengths (300 … 8000).  For each length we
  generate a diverse pool of start vectors (uniform, Dirichlet,
  exponential, power‑law, sparse etc.; also interpolate a previously
  known solution if available).

* For every start vector we run a mirror‑descent optimiser on the
  logits (exponentiated‑gradient).  The objective is the smooth
  max‑convolution obtained with a soft‑max temperature τ that is
  exponentially annealed.  Adam updates are applied to the logits.

* Whenever a new lower max‑convolution is observed we keep the
  corresponding vector globally.

* After the time budget for the gradient phase expires we perform a
  cheap simulated‑annealing refinement that repeatedly moves a small
  fraction of mass from one position to another.

The whole procedure respects the 1000 s wall‑clock limit and returns
the best sequence (plain Python list) found.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures

def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns (weights, smoothed max).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)          # shift for stability
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)      # log‑sum‑exp value
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Weighted reverse convolution.
    Returns rev_i = Σ_k w_k * a_{k‑i} (indices outside the array are ignored).
    Implemented via FFT, works for any length.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse w before FFT – required for the “reverse” convolution
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)

    # keep only the linear part (length = len(w) + len(a) – 1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]

    # the required slice (see derivation in the original implementation)
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _make_diverse_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """
    Produce a small, very diverse set of start vectors for length `n`.
    All vectors are already normalised so that Σ a_i = target_sum.
    """
    vecs = []

    # 1) uniform
    vecs.append(np.full(n, target_sum / n))

    # 2) Dirichlet α = 1   (dense)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    vecs.append(g / g.sum() * target_sum)

    # 3) Dirichlet α = 0.3 (sparse)
    g = rng.gamma(shape=0.3, scale=1.0, size=n)
    vecs.append(g / g.sum() * target_sum)

    # 4) exponential decay (random rate)
    rate = rng.uniform(0.001, 0.010)
    x = np.exp(-rate * np.arange(n))
    vecs.append(x / x.sum() * target_sum)

    # 5) reversed exponential
    vecs.append(x[::-1] / x[::-1].sum() * target_sum)

    # 6) power‑law decay (random exponent)
    p = rng.uniform(0.5, 2.5)
    i = np.arange(1, n + 1, dtype=float)
    x = i ** (-p)
    vecs.append(x / x.sum() * target_sum)

    # 7) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    vecs.append(tri / tri.sum() * target_sum)

    # 8) random sparse pattern (≈√n non‑zero entries)
    k = max(2, int(math.sqrt(n)))
    idxs = rng.choice(n, size=k, replace=False)
    mass = rng.uniform(0.5, 1.5, size=k)
    sparse = np.zeros(n)
    sparse[idxs] = mass
    sparse = sparse / sparse.sum() * target_sum
    vecs.append(sparse)

    # 9) uniform + tiny perturbation
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.05 * target_sum / n, size=n)
    pert = np.clip(pert, 0.0, None)
    pert = pert / pert.sum() * target_sum
    vecs.append(pert)

    # 10) interpolation of a previously found solution, if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        if base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
        else:
            interp = base.copy()
        interp = np.clip(interp, 0.0, None)
        interp = interp / interp.sum() * target_sum
        vecs.append(interp)
    except Exception:
        pass

    return vecs


# ----------------------------------------------------------------------
# Optimisation core – gradient descent on the soft‑max of the max‑convolution

def _exp_gradient_optimize(
        a0: np.ndarray,
        n: int,
        deadline: float,
        max_iters: int,
        best_seq: np.ndarray,
        best_score: float,
        rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the logits of a simplex vector.
    The objective is the exact maximal entry of a*a; a smooth max with a
    temperature τ is used for the gradient.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (tiny epsilon avoids log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.12                     # a little larger than the original code

    # temperature schedule parameters
    tau_factor = 0.45              # start slightly “hot”
    decay_rate = max_iters / 6.0   # slower decay ⇒ smoother search

    # FFT size – must accommodate a convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # -------- current vector (soft‑max of logits) --------
        logits = y - np.max(y)                 # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()      # lies on the simplex
        a = target_sum * p                     # satisfies Σa = target_sum, a≥0

        # -------- exact convolution a*a (linear, via FFT) ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # -------- smooth max (soft‑max) weights -------------
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # near‑exact max: uniform weight over all entries that tie
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # -------- gradient of the smooth max -----------------
        rev = _weighted_rev(a, w, Nfft)        # reverse‑convolution
        g_a = 2.0 * rev                         # ∂‖a*a‖_τ / ∂a

        # -------- chain rule to logits (mirror‑descent) -----
        avg = np.dot(p, g_a)                   # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)     # ∂f/∂y_i

        # -------- Adam update --------------------------------
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter – escapes flat regions
        if it % 1500 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            y += jitter

        # -------- keep the best vector seen -----------------
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} it {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement (pairwise mass transfers)

def _local_refine(
        a: np.ndarray,
        n: int,
        deadline: float,
        best_score: float,
        rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style refinement:
    move a small fraction of mass from one index to another.
    Accept improvements always, otherwise accept with a temperature decreasing linearly.
    """
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for the fast convolution used at each trial
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer 1 %–5 % of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate new max‑convolution (exact)
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.02 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (the problem permits up to 8000)
    lengths = [512, 768, 1024, 1536, 2048, 3072, 4096, 6144, 8000]
    lengths = [n for n in lengths if n >= 300]   # sanity check

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # Cache of start vectors for each length (generated once)
    init_cache = {}

    # ------------------------------------------------------------------
    # Main optimisation loop – runs many short Adam searches until time runs out
    while time.time() < deadline:
        # pick a length (biased towards larger sizes, they have more expressive power)
        n = int(rng.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        if n not in init_cache:
            init_cache[n] = _make_diverse_initial_vectors(n, target_sum, rng)

        starts = init_cache[n]
        # random start for this run
        a0 = starts[int(rng.integers(0, len(starts)))]

        # allocate a modest number of iterations – enough to make progress but cheap enough to try many runs
        if n <= 1500:
            max_iters = 300_000
        elif n <= 3000:
            max_iters = 200_000
        else:
            max_iters = 120_000

        # run the Adam optimiser on this start
        best_seq, best_score = _exp_gradient_optimize(
            a0=a0.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq if best_seq is not None else a0,
            best_score=best_score,
            rng=rng,
        )

        # light refinement after each successful run (costs only a few seconds)
        if time.time() < deadline - 2.0:
            refine_deadline = time.time() + min(4.0, (deadline - time.time()) * 0.15)
            best_seq, best_score = _local_refine(
                a=best_seq,
                n=best_seq.shape[0],
                deadline=refine_deadline,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Final, more thorough refinement on the overall best vector
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        final_deadline = time.time() + min(remaining, 30.0)   # ≤ 30 s for the final polish
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=final_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a return value
    if best_seq is None:
        n = 2000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
```
