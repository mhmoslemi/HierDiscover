"""
Propose a non‑negative step function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The implementation combines three ideas that improve on the baseline:

1.  Adam optimisation on a log‑softmax parametrisation, but with a
    more aggressive learning‑rate schedule and a much longer annealing
    of the smooth‑max temperature.
2.  A cheap “spectral‑flattening” stage (Gerchberg‑Saxton style) that
    forces the DFT magnitude of the padded sequence to be (approximately)
    constant – a known way to flatten the auto‑convolution.
3.  A final simulated‑annealing / mass‑shuffling refinement that can
    escape remaining local minima.

The whole search respects a 1000 s wall‑time budget, uses only NumPy
(and optionally a global variable ``height_sequence_1`` for warm‑starts),
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures) as required.
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    The result is a real array of length 2*n-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max.  Both inputs are real.  Returns a vector of length len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑controlled smooth approximation of max(conv) and its
    softmax weights (∂smooth_max/∂conv).
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def objective_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """
    Compute the evaluation function from the convolution.
    The sequence is assumed to be non‑negative.
    """
    return float(2.0 * n * conv.max() / (sum_a * sum_a))


def normalize(a: np.ndarray) -> np.ndarray:
    """Scale to unit ℓ₁‑norm (sum = 1), guarding against division by zero."""
    s = a.sum()
    if s < 1e-12:
        # fallback to uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a low‑value step‑function under the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # best probability vector (np.ndarray)
    best_val = float("inf")  # best objective value seen so far

    # ------------------------------------------------------------------
    # 1) MAIN optimisation loop over several problem sizes.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]   # larger n can theoretically reach ≈1

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused many times.
        fft_len_conv = next_pow2(2 * n - 1)          # for a*a
        fft_len_corr = next_pow2(3 * n)              # for cross‑correlation

        # Choose a modest number of restarts – adapt to remaining time.
        remain = deadline - time.time()
        restarts = 6
        if remain < 120:
            restarts = 2
        if n >= 4096:
            restarts = max(1, restarts - 2)

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Initialise a probability vector.
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a0 = best_seq.copy()
                a0 = normalize(a0)
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                # Use the historic construction if available.
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]), mode='constant')
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    a0 = np.random.dirichlet(np.ones(n))
                else:
                    a0 = normalize(cand)
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random start")

            # --------------------------------------------------------------
            # Adam optimisation on the log‑softmax parameters.
            # --------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.12                # initial learning rate

            # iteration budget – larger n gets fewer iterations, but we keep
            # the total work within the global deadline.
            if n <= 1024:
                max_iters = 180000
            elif n <= 2048:
                max_iters = 140000
            elif n <= 4096:
                max_iters = 100000
            else:
                max_iters = 60000

            # smooth‑max temperature schedule (linear annealing)
            tau_initial = 0.15
            tau_final = 5e-8

            last_improve = time.time()

            for it in range(max_iters):
                if time.time() > deadline:
                    break

                # soft‑max → probability vector (guarantees Σa = 1, a_i ≥ 0)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # convolution a*a
                conv = full_convolution(a, fft_len=fft_len_conv)

                # temperature for this iteration
                tau = max(tau_final,
                          tau_initial * (1.0 - it / max_iters))

                smooth, weights = smooth_max_and_weights(conv, tau)

                cur_val = objective_from_conv(conv, n, a.sum())

                # update global best
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} it={it:06d}] new best {cur_val:.6f}")

                # gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑prop through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update (with bias correction)
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** (it + 1))
                v_hat = v / (1.0 - beta2 ** (it + 1))
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # simple learning‑rate decay
                if (it + 1) % 5000 == 0:
                    lr *= 0.85

                # stagnation handling – small random jitter in log‑space
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={r} it={it}] shake (stagnation)")

            # --------------------------------------------------------------
            # 2) Local greedy mass‑shuffling refinement.
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for _ in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    if a_local[i] <= 1e-14:
                        continue
                    # move a modest fraction of mass from i → j
                    eps_move = min(a_local[i], 0.008) * (np.random.rand() * 0.6 + 0.2)
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = objective_from_conv(conv_new, n, a_new.sum())
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine n={n}] improved to {val_new:.6f}")

            # --------------------------------------------------------------
            # 3) Spectral flattening (Gerchberg‑Saxton style).
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
                a_spec = best_seq.copy()
                L = next_pow2(2 * n - 1)       # zero‑padding length for linear conv
                spec_iters = 2000
                for it in range(spec_iters):
                    if time.time() > deadline:
                        break
                    # FFT of the zero‑padded candidate
                    A = np.fft.rfft(a_spec, n=L)
                    # target magnitude = √(average power) → constant across frequencies
                    mag_target = np.sqrt(np.mean(np.abs(A) ** 2))
                    # keep original phase, enforce the constant magnitude
                    phase = np.angle(A)
                    A = mag_target * np.exp(1j * phase)
                    # inverse FFT, truncate to original length
                    a_new_full = np.fft.irfft(A, n=L)[:n]
                    a_new = np.maximum(a_new_full, 0.0)   # enforce non‑negativity
                    a_new = normalize(a_new)
                    conv = full_convolution(a_new, fft_len=fft_len_conv)
                    val = objective_from_conv(conv, n, a_new.sum())
                    if val < best_val - 1e-12:
                        best_val = val
                        best_seq = a_new.copy()
                        a_spec = a_new.copy()
                        print(f"[spec flatten n={n} it={it}] improved to {val:.6f}")

            # --------------------------------------------------------------
            # 4) Optional simulated‑annealing fine‑tuning (if time permits).
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
                a = best_seq.copy()
                cur_conv = full_convolution(a, fft_len=fft_len_conv)
                cur_val = objective_from_conv(cur_conv, n, a.sum())
                T = 0.2
                T_decay = 0.9997
                sa_iters = 20000
                for it in range(sa_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    if a[i] <= 1e-14:
                        continue
                    delta = min(a[i], 0.01) * (np.random.rand() * 0.5 + 0.1)
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = objective_from_conv(conv_new, n, a_new.sum())
                    diff = val_new - cur_val
                    if diff < 0 or np.exp(-diff / T) > np.random.rand():
                        a = a_new
                        cur_val = val_new
                        cur_conv = conv_new
                        if val_new < best_val - 1e-12:
                            best_val = val_new
                            best_seq = a.copy()
                            print(f"[SA n={n} it={it}] new best {val_new:.6f}")
                    T *= T_decay

    # ----------------------------------------------------------------------
    # 5) Fallback if nothing was discovered (should never happen).
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = float(2.0)  # uniform vector gives 2.0

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
