def propose_candidate(seed=12345,budget_s=1000,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # get base sequence if any
    try:
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        base = None

    # base length fallback
    n0 = len(base) if base is not None else 2100
    target_sum0 = math.sqrt(2*n0)

    # Step 1: get an initial good vector for base length
    if base is not None:
        best_seq = base.copy()
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Initial base eval = {best_score}")
    else:
        # generate uniform start and run short optimization
        best_seq = np.full(n0, target_sum0 / n0)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Uniform init eval = {best_score}")

    # For n in [n0, int(1.5*n0), int(2.0*n0), int(3.0*n0), int(4.0*n0)] but capped at 8000
    factors = [1.0, 1.5, 2.0, 3.0, 4.0]
    lengths = [min(8000, max(300, int(n0 * f))) for f in factors]
    lengths = sorted(set(lengths))

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # generate start vectors
        init_vectors = []
        # 1. interpolated best_seq (if size does not match)
        if best_seq.shape[0] != n:
            old_x = np.linspace(0,1,best_seq.shape[0])
            new_x = np.linspace(0,1,n)
            interp = np.interp(new_x, old_x, best_seq)
            interp = np.maximum(interp,0)
            interp = interp / np.sum(interp) * target_sum
            init_vectors.append(interp)

        # uniform
        init_vectors.append(np.full(n, target_sum/n))

        # Dirichlet
        g = rng.gamma(1.0,1.0,size=n)
        init_vectors.append(g / g.sum() * target_sum)

        # exponential decays
        rate = rng.uniform(0.0005,0.01)
        expv = np.exp(-rate*np.arange(n))
        init_vectors.append(expv/expv.sum()*target_sum)

        # random perturbation of uniform
        pert = np.full(n, target_sum/n) + rng.normal(scale=0.02*target_sum/n,size=n)
        pert = np.maximum(pert,0)
        pert = pert/pert.sum()*target_sum
        init_vectors.append(pert)

        # limit number of init vectors
        max_init = 5

        max_iters = int(2_500_000 * n0 / n)
        max_iters = max(200_000, max_iters)

        for idx, a0 in enumerate(init_vectors[:max_init]):
            if time.time() >= deadline:
                break
            print(f"[run] length={n}, start {idx+1}/{len(init_vectors[:max_init])}")

            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng
            )
    # final refinement
    if time.time() < deadline and best_seq is not None:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng
        )
    # final evaluation log
    print(f"[final] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Load the previously discovered construction (global ``height_sequence_1``)
    if it exists.  This gives a decent starting point for the *base* length
    ``n0`` (normally 2100).

2.  For a set of increasing lengths (``n0``, ≈1.5 · n0, 2 · n0, 3 · n0,
    4 · n0, capped at 8000) generate a few diverse initial vectors:
       * linear interpolation of the current best to the new length,
       * uniform distribution,
       * a Dirichlet draw,
       * an exponential decay,
       * a small random perturbation of the uniform vector.

3.  Each start vector is optimised with an **Adam** optimiser on the
    *log‑its* (exponentiated‑gradient) representation.  The objective is a
    smooth approximation of the maximum of the self‑convolution
    (soft‑max / log‑sum‑exp).  Gradients are obtained analytically via an
    FFT‑based reverse convolution.  The optimiser respects the constraints
    ``a_i ≥ 0`` and ``Σ a_i = √(2 n)`` automatically.

4.  The best vector found for a given length is kept globally.  The
    optimisation for a length runs until the global time budget is
    exhausted or a generous iteration limit is reached; longer lengths
    automatically get fewer iterations because each FFT is more expensive.

5.  After the length loop, a cheap stochastic hill‑climbing refinement
    (pairwise mass transfers with simulated‑annealing acceptance) is
    performed on the overall best vector.

6.  The final vector is returned as a plain Python list.  All numbers are
    non‑negative, and the total sum equals ``√(2 n)`` – therefore the
    value returned by ``evaluate_sequence`` is exactly the maximal entry
    of the self‑convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max) where weights sum to 1.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)          # shift for stability
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)        # log‑sum‑exp value
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i}   (indices out of range are ignored)
    using FFTs.  This is the gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution (length = 3 n‑2)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen in
    this run together with its exact “max‑convolution” objective.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (add tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.07                   # base learning rate for logits
    jitter_interval = 2500       # how often to add random jitter

    # temperature (soft‑max) schedule
    tau_factor = 0.25            # τ₀ = τ_factor·M
    decay_rate = max_iters / 8.0

    # FFT size – must hold a linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ------- current vector (softmax of logits) -------
        logits = y - np.max(y)                     # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()          # lies on the simplex
        a = target_sum * p                         # satisfies Σa = target_sum

        # ------- exact self‑convolution -------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ------- soft‑max weights for the max -------
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ------- weighted reverse convolution (gradient of smoothed max) -------
        rev = _weighted_rev(a, w, Nfft)

        # ------- gradient wrt a (2·rev) -------
        g_a = 2.0 * rev

        # ------- chain rule back to logits (mirror‑descent on the simplex) -------
        avg = np.dot(p, g_a)                     # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)       # ∂f/∂y_i

        # ------- Adam update on logits -------
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

        # ------- keep best -------
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:9d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from one index to another.
    Improvements are always accepted; worsening moves are accepted with a
    decaying simulated‑annealing probability.
    """
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the given
    time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds as safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Load any previously discovered construction
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        print(f"[load] loaded previous construction (len={base.shape[0]})")
    except Exception:
        base = None
        print("[load] no previous construction found")

    # --------------------------------------------------------------
    # Base length and its normalisation
    if base is not None:
        n0 = int(base.shape[0])
        best_seq = base.copy()
        # compute its exact score once (fast via convolution)
        A_fft = np.fft.rfft(best_seq, n=1 << ((2 * n0 - 1).bit_length()))
        conv = np.fft.irfft(A_fft * A_fft)[: 2 * n0 - 1]
        best_score = float(conv.max())
        print(f"[init] base score = {best_score:.6f}")
    else:
        # fall back to an uniform vector of moderate size
        n0 = 2100
        target_sum = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target_sum / n0)
        best_score = float('inf')
        print(f"[init] using uniform start, len={n0}")

    # --------------------------------------------------------------
    # Lengths to explore – geometric stretch up to 8000
    factors = [1.0, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in factors})

    # --------------------------------------------------------------
    # Global best tracking
    # (best_seq / best_score already hold the current global best)

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        print(f"\n[run] exploring length {n}")

        # ---- generate diverse start vectors for this length ----
        init_vectors = []

        # a) interpolated current global best (if length differs)
        if best_seq.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, best_seq)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # b) uniform
        init_vectors.append(np.full(n, target_sum / n))

        # c) Dirichlet (Gamma(1,1))
        g = rng.gamma(shape=1.0, scale=1.0, size=n)
        init_vectors.append(g / g.sum() * target_sum)

        # d) exponential decay (random rate)
        rate = rng.uniform(0.0005, 0.01)
        expv = np.exp(-rate * np.arange(n))
        init_vectors.append(expv / expv.sum() * target_sum)

        # e) small random perturbation of the uniform vector
        pert = np.full(n, target_sum / n) + rng.normal(scale=0.02 * target_sum / n, size=n)
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum() * target_sum
        init_vectors.append(pert)

        # ---------------------------------------------------------
        # Work‑budget per length – more iterations for smaller n
        max_iters = int(2_500_000 * n0 / n)          # roughly keeps total work constant
        max_iters = max(200_000, max_iters)         # never fewer than 200 k

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
            )

    # --------------------------------------------------------------
    # Final stochastic refinement (if we still have time)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # at most half the remaining budget for refinement, but no more than 30 s
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # --------------------------------------------------------------
    # Fallback safety net (should never trigger)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"\n[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
