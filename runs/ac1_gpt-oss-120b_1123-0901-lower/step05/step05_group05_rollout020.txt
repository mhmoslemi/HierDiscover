def propose_candidate(seed=42, budget_s=1000.0):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    best_seq = None
    best_score = np.inf

    # 1. warm-start from known sequence
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        # normalize sum
        n0 = hs_arr.shape[0]
        target = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm-start score={best_score}")
    except Exception:
        pass

    # 2. exponential family search (fast)
    while time.time() < deadline:
        # adjust remaining time
        remaining = deadline - time.time()
        # choose n in 1500-6000
        n = np.random.randint(1500, 5000+1)
        target = np.sqrt(2.0 * n)
        # lambda range: 0.05 to 4.0
        # number of samples adapt: maybe 30 per iteration
        num_lambdas = 30
        lambdas = np.linspace(0.05, 4.0, num_lambdas)
        for lam in lambdas:
            r = np.exp(-lam / n)
            # generate exponential shape
            a = np.empty(n, dtype=float)
            # compute powers iteratively for speed
            a[0] = 1.0
            for i in range(1,n):
                a[i] = a[i-1] * r
            # normalize to target sum
            a = a * (target / a.sum())
            # compute max conv using analytic formula
            # conv_max = c^2 * max(L * r^k) where c = a[0]
            c = a[0]
            L = np.concatenate([np.arange(1,n+1), np.arange(n-1,0,-1)])
            r_powers = np.exp(np.arange(2*n-1) * np.log(r))
            conv_max = (c*c) * np.max(L * r_powers)
            # candidate objective = conv_max (since scaling invariant)
            # use evaluate_sequence to get precise result (optional)
            if conv_max < best_score:
                # compute exact score to verify
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = float(exact)
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] exponential n={n} r={r:.6f} new best={best_score:.6f}")

        # break earlier if time low
        if time.time() > deadline - 5:
            break

    # 3. hillclimb fine tuning
    if best_seq is not None:
        n = best_seq.shape[0]
        target = np.sqrt(2.0 * n)
        hill_end = deadline - 0.5
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target, hill_end)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] final hillclimb best={best_score}")

    # fallback uniform
    if best_seq is None:
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback score={best_score}")

    return best_seq.tolist()

"""
Hybrid random‑search / local‑refinement for the autocorrelation minimisation problem.

The evaluation metric is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

and is scaling‑invariant, i.e. only the shape of the non‑negative vector `a`
matters.  The algorithm proceeds in three stages:

1. **Parametric random search** –  a large number of cheap candidates are
   generated from a few low‑dimensional families (exponential, power‑law,
   linear taper, mixed).  Each candidate is normalised to the required
   ℓ₁‑norm √(2 n) and its exact objective is obtained with a fast FFT‑based
   convolution.

2. **Projected sub‑gradient descent** –  starting from the best candidate,
   the true non‑smooth objective (the maximum convolution entry) is reduced
   by a sub‑gradient step and Euclidean projection onto the simplex
   {x ≥ 0, Σx = √(2 n)}.

3. **Fine‑grained stochastic hill‑climbing** –  a simulated‑annealing
   style local search that repeatedly transfers a tiny amount of mass
   between two positions (or averages a pair) while keeping the simplex
   constraint.  This stage refines the solution found by the previous
   stages.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (full length)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)      # remove tiny negatives from rounding
    return conv


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {w ≥ 0, Σw = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_conv_max(a: np.ndarray) -> float:
    """Exact objective (maximum entry of the auto‑convolution) for a given vector."""
    conv = _full_convolution_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Parametric families (shape generators)
# ----------------------------------------------------------------------
def _gen_exponential(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i / n).  `lam` > 0."""
    r = np.exp(-lam / n)
    a = r ** np.arange(n, dtype=float)
    a *= target_sum / a.sum()
    return a


def _gen_powerlaw(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   with p ≥ 0."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _gen_linear_taper(n: int, frac: float, target_sum: float) -> np.ndarray:
    """
    Decreasing linear taper.
    `frac` ∈ (0,1) controls the length of the non‑zero part (≈frac·n).
    The values drop linearly from a positive height to zero.
    """
    m = max(2, int(frac * n))            # at least two non‑zero entries
    # heights: m, m‑1, … , 1   (scaled later)
    a = np.arange(m, 0, -1, dtype=float)
    a = np.concatenate([a, np.zeros(n - m, dtype=float)])
    a *= target_sum / a.sum()
    return a


def _gen_mixed(n: int, lam: float, p: float, target_sum: float) -> np.ndarray:
    """Mixture of exponential and power‑law: a_i ∝ exp(-lam*i/n)·(i+1)^{-p}."""
    r = np.exp(-lam / n)
    idx = np.arange(1, n + 1, dtype=float)
    a = (r ** np.arange(n, dtype=float)) * (idx ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation stages
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_conv_max(a)
    step = 0.1 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the selected peak (conv_k)
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _objective_conv_max(cand)

        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02
        else:
            step *= 0.70

        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] subgrad it={it} step={step:.3e} best={best_val:.6f}")

        if best_val < 1.40:           # generous early‑stop
            break

    return best_a, best_val


def _hillclimb(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing style stochastic hill‑climb.
    Two move types are used:
      * transfer a small mass from i → j,
      * replace a pair (i, j) by their average (mass conserving).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_conv_max(a)

    T0 = best_val * 0.3
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - frac))

        # pick a move type
        if np.random.rand() < 0.7:
            # transfer a tiny amount
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] <= 0:
                continue
            delta = a[i] * np.random.rand() * 0.08
            new = a.copy()
            new[i] -= delta
            new[j] += delta
        else:
            # average a pair (keeps the sum exactly)
            i, j = np.random.choice(n, 2, replace=False)
            avg = (a[i] + a[j]) * 0.5
            new = a.copy()
            new[i] = avg
            new[j] = avg

        new_val = _objective_conv_max(new)

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = new
            if new_val < best_val:
                best_val = new_val
                best_a = new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 2000 == 0:
            # numerical drift correction (project back onto the simplex)
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.40:      # early stop if we are already very good
            break

    return best_a, best_val
