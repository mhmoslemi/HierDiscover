"""
Propose a non‑negative step‑function that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm works in three phases.

1️⃣  *Broad parametric sweep* – a large set of analytic families
    (edge‑concentrated power‑law, two‑side exponentials, Gaussians,
    Beta‑shaped, sine‑bump, uniform and random Dirichlet draws) are
    generated for the target length n.  Each shape is normalised to a
    probability vector and its objective value is evaluated with a
    fast FFT‑based convolution.  The best few candidates are kept.

2️⃣  *High‑precision refinement* – every retained candidate undergoes a
    smooth‑max Adam optimisation on the logits of the soft‑max
    parametrisation (guaranteeing the simplex constraint).  The smooth‑max
    temperature λ is annealed to the true max; Adam’s learning‑rate,
    momentum and a simple stagnation‑shake are tuned for robustness.

3️⃣  *Exact sub‑gradient + tiny random‑move local search* – after the
    surrogate phase a small number of exponentiated‑gradient steps on the
    true max are performed, followed by a handful of random mass‑transfer
    moves that are accepted only when they improve the exact objective.
    All work respects the wall‑clock budget (default 1000 s).

The routine returns the best sequence found as a plain Python list.
"""

# FFT‑based convolution utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Families of analytic shapes (all return a *non‑normalised* vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law(alpha: float, n: int) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    return shape


def _edge_exp(beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
    return shape


def _two_gaussian(gamma: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    return shape


def _beta_shape(p: float, q: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    eps = 1e-12
    # avoid 0**negative or 1**negative
    left = np.where(x > 0, (x + eps) ** p, 0.0)
    right = np.where(x < 1, ((1 - x) + eps) ** q, 0.0)
    shape = left * right
    return shape


def _sine_bump(n: int) -> np.ndarray:
    return np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2


# ----------------------------------------------------------------------
# Helper to evaluate the exact objective (sum of a is assumed 1)
# ----------------------------------------------------------------------
def _objective(val_vec: np.ndarray, n: int) -> float:
    """
    ``val_vec`` must be a probability vector (non‑negative, sum = 1).
    Returns 2·n·max(conv) where conv = val_vec * val_vec.
    """
    conv = _full_convolution(val_vec, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Heavy optimiser (Adam on smooth‑max + exact sub‑gradient + random moves)
# ----------------------------------------------------------------------
def _refine_candidate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    best_val: float,
    best_seq: np.ndarray,
    deadline: float,
) -> tuple[float, np.ndarray]:
    """
    Perform the aggressive optimisation described in the doc‑string.
    Returns possibly improved ``best_val`` and ``best_seq``.
    """

    # --------------------------------------------------------------
    # Initialisation
    # --------------------------------------------------------------
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)                     # simplex via soft‑max

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.07                               # a bit larger than before

    lam = 0.5                               # smooth‑max temperature
    lam_factor = 1.02
    last_improve = time.time()

    # iteration budget – more iterations for the large problem
    max_iters = 12000                       # ample for n≈16 k

    # --------------------------------------------------------------
    # Adam on smoothed maximum
    # --------------------------------------------------------------
    for it in range(1, max_iters + 1):
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (mirror‑gradient)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] new best {val:.6f} (it {it})")

        # stagnation handling – tiny shake
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # Exact sub‑gradient refinement (exponentiated‑gradient)
    # --------------------------------------------------------------
    a = a / a.sum()
    eta = 0.6
    sub_last = time.time()
    for sub_it in range(3000):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last = time.time()
            print(f"[Subgrad] new best {val:.6f} (it {sub_it})")

        # active set (ties are split uniformly)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        if time.time() - sub_last > 5.0:
            break
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # Light random‑move local search
    # --------------------------------------------------------------
    a = best_seq.copy()
    move_trials = 0
    while time.time() < deadline and move_trials < 5000:
        i, j = np.random.choice(n, 2, replace=False)
        # transfer a tiny amount of mass from i to j
        delta = min(a[i], np.random.rand() * 1e-4)  # ≤ 1e‑4 of total mass
        if delta <= 0:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # normalisation stays 1, positivity holds by construction
        conv = _full_convolution(a2, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a2.copy()
            a = a2
            print(f"[Random] new best {val:.6f} (trial {move_trials})")
        move_trials += 1

    return best_val, best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects a wall‑clock budget (default 1000 s) and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # target length – the larger the better (within the time budget)
    n = 16384

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1️⃣  Broad parametric sweep
    # ------------------------------------------------------------------
    candidate_pool = []

    # uniform
    candidate_pool.append(_uniform(n))

    # historic global variable, if present
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        candidate_pool.append(h)

    # power‑law edge‑concentrated families (dense sweep)
    for alpha in np.linspace(-3.0, -0.1, 50):
        candidate_pool.append(_power_law(alpha, n))

    # two‑side exponentials
    for beta in np.linspace(0.02, 2.5, 40):
        candidate_pool.append(_edge_exp(beta, n))

    # dual Gaussians
    for gamma in np.linspace(0.02, 2.5, 40):
        candidate_pool.append(_two_gaussian(gamma, n))

    # beta‑shaped distributions (p,q grid)
    for p in (-1.5, -0.5, 0.0, 0.5, 1.5):
        for q in (-1.5, -0.5, 0.0, 0.5, 1.5):
            candidate_pool.append(_beta_shape(p, q, n))

    # smooth sine‑squared bump
    candidate_pool.append(_sine_bump(n))

    # a few random Dirichlet draws (cheap but diverse)
    for _ in range(20):
        candidate_pool.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # Evaluate all candidates (fast) and keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for idx, raw in enumerate(candidate_pool):
        if time.time() > deadline:
            break
        shape = np.asarray(raw, dtype=np.float64)
        s = shape.sum()
        if s <= 0:
            continue
        a = shape / s                     # normalise to sum = 1
        val = _objective(a, n)
        evaluated.append((val, a))
        print(f"[Sweep] cand {idx:03d} value {val:.6f}")

    # sort by objective (ascending) and keep the top K for deep optimisation
    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_candidates = [a for (_, a) in evaluated[:top_k]]

    # initialise global best
    best_val = float("inf")
    best_seq = None
    if top_candidates:
        best_val = _objective(top_candidates[0], n)
        best_seq = top_candidates[0].copy()

    # ------------------------------------------------------------------
    # 2️⃣  Heavy optimisation on the most promising candidates
    # ------------------------------------------------------------------
    for cid, cand in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(
            f"\n=== Heavy optimisation start: candidate {cid} (init val {_objective(cand, n):.6f}) ==="
        )
        best_val, best_seq = _refine_candidate(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            best_val,
            best_seq,
            deadline,
        )

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Return as plain Python list
    return best_seq.tolist()
