# ----------------------------------------------------------------------
# FFT‑based convolution utilities

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (used only for occasional resampling)

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            pert = best_seq + 0.02 * np.random.randn(n) * best_seq
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                cand.append(pert / s)

    np.random.shuffle(cand)
    # Keep a reasonable number – the later stages filter by time.
    return cand[:16]


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


# ----------------------------------------------------------------------
# Adam optimisation of a smooth‑max surrogate

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best (a, value) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)          # initialise logits for simplex

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # learning‑rate scales mildly with problem size
    lr = 0.05 * np.sqrt(1024.0 / n)
    lr = max(lr, 0.001)

    lam = 0.1                         # start temperature for smooth‑max
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Project logits to simplex via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # Smooth‑max weights over convolution entries
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # Gradient of the surrogate = 2 * (w ★ a)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # Gradient w.r.t. logits (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # Random shake if stuck for a while
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient optimisation (exponentiated gradient)

def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    Uses exponentiated‑gradient updates with a decaying step size.
    """
    a = a0 / a0.sum()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    best_a = a.copy()
    last_improve = time.time()
    eta0 = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # Identify active indices (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # Decaying step size
        eta = eta0 / np.sqrt(it)

        # Exponentiated‑gradient step – keeps positivity automatically
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        # If no improvement for a few seconds, give up early
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climbing local search (enhanced)

def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that shifts mass between two or three indices.
    Occasionally accepts uphill moves to escape tiny plateaus.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- two‑point move -------------------------------------------------
        i1, i2 = np.random.randint(0, n, size=2)
        if i1 != i2:
            eps = np.random.rand() * 0.05 * a[i1]   # up to 5 % of mass from i1 to i2
            if eps > 0:
                new_a = a.copy()
                new_a[i1] -= eps
                new_a[i2] += eps
                if new_a[i1] >= -1e-15:
                    new_a /= new_a.sum()
                    new_val = _objective(new_a, n, fft_len_conv)
                    if new_val < best_val:
                        best_val = new_val
                        best_a = new_a.copy()
                        a = new_a
                        cur_val = new_val
                        last_improve = time.time()
                        print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
                        continue

        # ----- three‑point cyclic move ---------------------------------------
        i = np.random.choice(n, size=3, replace=False)
        eps = np.random.rand() * 0.03   # move up to 3 % of total mass
        if eps > 0:
            new_a = a.copy()
            # rotate eps among three positions
            new_a[i[0]] = max(new_a[i[0]] - eps, 0.0)
            new_a[i[1]] += eps * 0.5
            new_a[i[2]] += eps * 0.5
            new_a /= new_a.sum()
            new_val = _objective(new_a, n, fft_len_conv)
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                a = new_a
                cur_val = new_val
                last_improve = time.time()
                continue

        # ----- occasional uphill acceptance (simulated annealing) ----------
        if np.random.rand() < 0.001:
            # propose a tiny random perturbation
            idx = np.random.randint(0, n)
            delta = (np.random.rand() - 0.5) * 0.01 * a[idx]
            new_a = a.copy()
            new_a[idx] = max(new_a[idx] + delta, 0.0)
            new_a /= new_a.sum()
            new_val = _objective(new_a, n, fft_len_conv)
            if new_val < cur_val or np.random.rand() < 0.01:
                a = new_a
                cur_val = new_val

        # early exit if stuck
        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence (step‑function) with a low
    evaluation score.  Returns the best sequence discovered as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Load a historic solution if the notebook already defined one.
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # Length schedule – start small, finish with a larger resolution.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    # Pre‑compute FFT lengths needed for each n.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # Create a diverse pool of starting points.
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Adam iteration budget shrinks with n (quadratic cost of FFT).
        max_adam_iters = int(15000 * 1024 / n) + 2000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – Adam on smooth‑max surrogate ------------------
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact sub‑gradient (exponentiated‑gradient) ---
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline, init_val=val_adam
            )

            # ---- Phase 3 – cheap hill‑climbing -------------------------
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=val_sub
            )

            # ---- Optional small Adam refinement to escape local minima ---
            # Only if time permits and improvement is plausible.
            if time.time() < deadline - 5.0 and val_hc > 1.4:
                a_ref, val_ref = _adam_optimize(
                    a_hc, n, fft_len_c, fft_len_x, deadline,
                    max_iters=int(0.5 * max_adam_iters)
                )
                if val_ref < val_hc:
                    a_hc, val_hc = a_ref, val_ref

            # Update the global best.
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run refinement on the best sequence (if any time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n, 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(2.0 * (15000 * 1024 / n) + 4000)
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as a plain Python list (required by the external evaluator).
    return best_seq.tolist()
