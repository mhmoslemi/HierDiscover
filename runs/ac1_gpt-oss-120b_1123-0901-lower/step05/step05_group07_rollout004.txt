"""
Propose a non‑negative step function that (approximately) minimizes

        F(a) = 2·n·max(a*a) / (sum(a)^2)

Our approach differs from the previous gradient‑only method.
We reduce the dimensionality by expressing the logits of the probability
vector in a truncated Fourier basis and optimise those coefficients with
Adam.  The smooth‑maximum (log‑sum‑exp) is used during annealing, and a
final cheap stochastic “mass‑shuffling’’ refinement is applied.
The algorithm works for several lengths (1024, 2048, 4096) while honouring
the 1000 s wall‑time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (all top‑level – no closures or lambdas)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    Implemented by FFT‑based convolution with the reversed vector.
    """
    n = a.shape[0]
    # convolution of w with reversed a
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a[::-1], fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, fft_len)
    start = n - 1
    return corr_full[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑smoothed max of `conv` together with its softmax weights.
    Returns (smooth_max, weights) where
        smooth_max = max(conv) + τ·log Σ exp((conv‑max)/τ)
        weights[i] = ∂smooth_max/∂conv[i] = exp((conv[i]-max)/τ) / Σ exp(...)
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------

    Search for a step‑function (list of non‑negative floats) that minimises

    Fourier‑basis parametrisation and Adam optimisation, and finishes with a
    cheap stochastic local refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None          # best probability vector found so far
    best_val = float("inf")  # best objective value (2·n·max(conv))

    # ------------------------------------------------------------------
    # Helper to obtain an initial probability vector for a given length
    # ------------------------------------------------------------------
    def initialise_a(n):
        """Return a normalised non‑negative array of length `n` to start from."""
        nonlocal best_seq
        # 1. Warm‑start from a previously discovered sequence (if size matches)
        if best_seq is not None:
            if best_seq.shape[0] == n:
                a = best_seq.copy()
            elif best_seq.shape[0] < n:
                # linear interpolation / up‑sampling
                old = best_seq
                x_old = np.linspace(0, 1, old.shape[0])
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, old)
            else:   # best_seq longer than needed
                a = best_seq[:n].copy()
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            return a

                cand = np.random.dirichlet(np.ones(n))
            cand /= cand.sum()
            return cand

        # 3. Pure random Dirichlet start
        a = np.random.dirichlet(np.ones(n))
        return a

    # Optimisation over a few problem sizes

        # ------------------------------------------------------------------
        # Number of sinusoidal frequencies (controls expressive power)
        freq_cnt = max(5, min(80, int(n * 0.04)))   # ≈4 % of n, capped at 80
        m = 1 + 2 * freq_cnt                      # bias + cos/sin pairs
        idx = np.arange(n, dtype=np.float64)
        B = np.empty((n, m), dtype=np.float64)
        B[:, 0] = 1.0
        col = 1
        for k in range(1, freq_cnt + 1):
            omega = 2.0 * np.pi * k / n
            B[:, col] = np.cos(omega * idx)
            B[:, col + 1] = np.sin(omega * idx)
            col += 2

        # ------------------------------------------------------------------
        # Initialise the optimisation variables (theta = basis coefficients)
        # ------------------------------------------------------------------
        a0 = initialise_a(n)
        # log‑softmax logits of the starting vector
        x0 = np.log(a0 + 1e-15)
        # Least‑squares solve B·theta ≈ x0 to obtain a sensible start
        theta, *_ = np.linalg.lstsq(B, x0, rcond=None)

        # Adam state
        m_t = np.zeros_like(theta)
        v_t = np.zeros_like(theta)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.05                     # learning rate for theta

        # Iteration limits (more for smaller n)
        if n <= 1024:
            max_iters = 200000
        elif n <= 2048:
            max_iters = 150000
        else:
            max_iters = 120000

        # Smooth‑max temperature schedule
        tau_initial = 0.18
        tau_final = 5e-7

        t = 0
        last_improve = time.time()

        while time.time() < deadline and t < max_iters:
            # ----- logits → probability vector (softmax) -----
            logits = B @ theta
            max_log = np.max(logits)
            exp_log = np.exp(logits - max_log)
            a = exp_log / exp_log.sum()          # a_i ≥ 0, Σ a_i = 1

            # ----- convolution of a with itself -----
            conv = full_convolution(a, fft_len=fft_len_conv)

            # ----- annealed smooth‑max -----
            tau = tau_initial * (1.0 - t / max_iters)
            if tau < tau_final:
                tau = tau_final
            smooth, weights = smooth_max_and_weights(conv, tau)

            # ----- true objective (for bookkeeping) -----
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[n={n} iter={t:06d}] new best {cur_val:.6f}")

            # ----- gradient of smooth‑max w.r.t. a -----
            # ∂smooth/∂a = 2·(weights ★ a)
            grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

            # ----- back‑prop through the soft‑max -----
            inner = np.dot(a, grad_a)
            grad_x = a * (grad_a - inner)   # dF/d(logits)

            # ----- chain rule to the Fourier coefficients -----
            grad_theta = B.T @ grad_x

            # ----- Adam update for theta -----
            t += 1
            m_t = beta1 * m_t + (1.0 - beta1) * grad_theta
            v_t = beta2 * v_t + (1.0 - beta2) * (grad_theta * grad_theta)
            m_hat = m_t / (1.0 - beta1 ** t)
            v_hat = v_t / (1.0 - beta2 ** t)
            theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- learning‑rate decay (geometric) -----
            if t % 30000 == 0:
                lr *= 0.7
                print(f"[n={n} iter={t}] lr decayed to {lr:.4f}")

            # ----- stagnation handling (random jitter) -----
            if time.time() - last_improve > 30.0:
                theta += (np.random.rand(*theta.shape) - 0.5) * 1e-2
                last_improve = time.time()
                print(f"[n={n} iter={t}] jitter (stagnation)")

        # ------------------------------------------------------------------
        # Cheap stochastic local refinement (mass shuffling)
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            a_local = best_seq.copy()
            local_iters = 4000
            for it in range(local_iters):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                # move a small random fraction from i to j
                move = min(a_local[i], 0.01) * (np.random.rand() * 0.8 + 0.2)
                if move <= 0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= move
                a_new[j] += move
                conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                val_new = 2.0 * n * conv_new.max()
                if val_new < best_val - 1e-12:
                    best_val = val_new
                    best_seq = a_new.copy()
                    a_local = a_new
                    print(f"[local refine n={n} iter={it}] improved to {val_new:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the unlikely event we never found a feasible candidate
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
