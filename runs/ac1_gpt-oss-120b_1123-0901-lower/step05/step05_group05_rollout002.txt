"""
Hybrid search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine consists of three stages:

1. **Randomised initialisation** – draw a shape (uniform, sine, triangular,
   exponential, or Dirichlet) and normalise it to Σa = √(2 n).

2. **Linear‑program refinement** – for a fixed sequence f, solve the LP

       maximise   Σ b_j
       subject to  (f * b)_k ≤ M   (k = 0 … 2n‑2)
                   b_j ≥ 0

   where M = max(conv(f, f)).  The solution b is scaled to the required
   ℓ₁‑norm and mixed with f (convex combination).  Repeating this step
   shrinks the convolution peak.

3. **Lighthill stochastic local search** – a final hill‑climb that moves a
   tiny amount of mass between two indices, accepting moves that improve
   (or occasionally worsen) the peak.

All stages respect a global time budget (default 1000 s) and keep the
best sequence found.  The final list is returned to the evaluator.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σw = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex with radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build the (2n‑1) × n convolution matrix T such that T @ b = f * b.
# T_kn = f_k where 0 ≤ k‑i < n, otherwise 0.
# ----------------------------------------------------------------------
def _build_convolution_matrix(f: np.ndarray) -> csr_matrix:
    n = f.shape[0]
    # non‑zero count = n * n
    nnz = n * n
    data = np.empty(nnz, dtype=np.float64)
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)

    idx = 0
    for i, fi in enumerate(f):
        # rows where column i participates: i … i+n‑1
        rows[idx: idx + n] = np.arange(i, i + n, dtype=np.int32)
        cols[idx: idx + n] = i
        data[idx: idx + n] = fi
        idx += n
    shape = (2 * n - 1, n)
    return csr_matrix((data, (rows, cols)), shape=shape)


# ----------------------------------------------------------------------
# Simple sub‑gradient descent on max(conv(a, a))
# ----------------------------------------------------------------------
def _subgradient_descent_simple(a: np.ndarray,
                               target_sum: float,
                               deadline: float,
                               max_iter: int = 30) -> np.ndarray:
    """A few sub‑gradient steps to quickly lower the peak."""
    n = a.shape[0]
    step = 0.1 * target_sum / n
    a_best = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        a_cand = a - step * grad
        a_cand = _project_onto_simplex(a_cand, target_sum)

        val_cand = _full_convolution_fft(a_cand, a_cand).max()
        if val_cand < best_val - 1e-12:
            a = a_cand
            a_best = a_cand.copy()
            best_val = val_cand
            step *= 1.02
        else:
            step *= 0.70

        # occasional jitter to escape plateaus
        if it % 10 == 0 and random.random() < 0.1:
            a += np.random.rand(n) * 1e-5
            a = _project_onto_simplex(a, target_sum)

    return a_best


# ----------------------------------------------------------------------
# Linear‑program refinement (the “alternating LP” idea)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iter: int = 6) -> np.ndarray:
    """
    Iteratively improve a by solving the LP:
        maximise Σ b_j  subject to (a * b)_k ≤ M   and   b ≥ 0.
    The solution is mixed back into a.
    """
    n = a.shape[0]
    a_cur = a.copy()
    best_val = _full_convolution_fft(a_cur, a_cur).max()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution_fft(a_cur, a_cur)
        M = float(conv.max())

        # Build convolution matrix for the current a
        T = _build_convolution_matrix(a_cur)

        # Linear program data
        c = -np.ones(n, dtype=np.float64)          # maximise Σb  → minimise –Σb
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve with HiGHS (fast interior‑point)
        res = linprog(c=c,
                      A_ub=T,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # If LP fails, abort further refinement
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # Scale to the required ℓ₁‑norm
        scale = target_sum / sum_b
        g = b * scale

        # Convex combination (t = 0.5 works well in practice)
        t = 0.5
        a_new = (1.0 - t) * a_cur + t * g
        a_new = _project_onto_simplex(a_new, target_sum)

        new_val = _full_convolution_fft(a_new, a_new).max()
        print(f"[{time.time():.2f}] LP iter {it}  M={M:.6f} → new max={new_val:.6f}")

        if new_val < best_val - 1e-12:
            a_cur = a_new
            best_val = new_val
        else:
            # No improvement – stop early
            break

    return a_cur


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_simple(a: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      max_iters: int = 3000) -> np.ndarray:
    """Fine‑tune a by moving small mass chunks between random positions."""
    n = a.shape[0]
    best = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule (hot → cold)
    T0 = best_val * 0.3
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move at most 10 % of the mass at i
        delta = a[i] * random.random() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if random.random() < math.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # occasional full re‑normalisation to avoid drift
        if it % 500 == 0:
            a = _project_onto_simplex(a, target_sum)

    return best

    an LP based refinement, and a final hill‑climb.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.05   # safety margin

    best_seq = None
    best_score = np.inf

    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score
        val = evaluate_sequence(seq.tolist())
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL best={best_score:.6f} (len={len(seq)})")

    # ------------------------------------------------------------------
    # Main search loop – keep generating candidates until time runs out.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # Choose a problem size adapted to the remaining time
        if remaining > 200:
            n = np.random.randint(1500, 2500)
        elif remaining > 100:
            n = np.random.randint(1200, 2000)
        else:
            n = np.random.randint(800, 1500)

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Random initial shape (four families)
        # ------------------------------------------------------------------
        shape = np.random.choice(['random', 'sine', 'triangular', 'exponential'])
        if shape == 'random':
            a = np.random.rand(n)
        elif shape == 'sine':
            idx = np.arange(n)
            a = np.sin(np.pi * idx / (n - 1))
        elif shape == 'triangular':
            idx = np.arange(n)
            peak = (n - 1) / 2.0
            a = 1.0 - np.abs(idx - peak) / peak
            a[a < 0] = 0.0
        else:   # exponential decay from the centre
            idx = np.arange(n)
            centre = (n - 1) / 2.0
            a = np.exp(-np.abs(idx - centre) / (0.2 * n))

        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a = a / a.sum() * target_sum

        # ------------------------------------------------------------------
        # Phase 1 – a handful of sub‑gradient steps (quick improvement)
        # ------------------------------------------------------------------
        a = _subgradient_descent_simple(a, target_sum, deadline, max_iter=15)

        # ------------------------------------------------------------------
        # Phase 2 – LP refinement (core of the algorithm)
        # ------------------------------------------------------------------
        a = _lp_refine(a, target_sum, deadline, max_iter=6)

        # ------------------------------------------------------------------
        # Phase 3 – final hill‑climb for fine‑tuning
        # ------------------------------------------------------------------
        a = _hillclimb_simple(a, target_sum, deadline, max_iters=2500)

        # ------------------------------------------------------------------
        # Record the candidate if it beats the global best
        # ------------------------------------------------------------------
        _maybe_update(a)

    # ----------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Convert to plain Python list as required by the evaluator
    return best_seq.tolist()
