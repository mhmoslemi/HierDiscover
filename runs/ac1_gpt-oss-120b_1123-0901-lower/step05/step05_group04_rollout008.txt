```
def _optimise_subgradient_max(n, a0, target_sum, run_deadline, rng, lr0=0.1):
    a = a0.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    t0 = time.time()
    lr = lr0
    iteration=0
    last_improve_iter=0

    while time.time() < run_deadline:
        # compute convolution
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        kmax = conv.argmax()
        # early exit if no improvement for many iter
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_a = a.copy()
            last_improve_iter = iteration
        # compute gradient (subgradient)
        # compute w one-hot
        w = np.zeros(conv.shape[0])
        w[kmax] = 1.0
        # compute grad a using conv2 = convolution a * w_rev
        w_rev = w[::-1]
        grad_full = _full_convolution_fft(a, w_rev)
        # gradient a
        grad_a = 2.0 * grad_full[2*n-2 - np.arange(n)]
        # step
        a = a - lr * grad_a
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)
        # decrease lr
        lr = lr * 0.9995
        iteration += 1
        # occasional random perturbation
        if iteration % 500 == 0:
            # add small random noise
            a += rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a,0.0)
            a = _project_to_simplex(a, target_sum)
        if iteration - last_improve_iter > 5000:
            break
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, final_score
```

```
def _project_to_simplex(v, target_sum):
    # v may have negative entries, we need to project onto {x >=0, sum = target}
    n = v.shape[0]
    # shift to ensure sum = target
    if target_sum <= 0:
        return np.zeros_like(v)
    # compute sorted v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Projected sub‑gradient descent on the max‑convolution objective.

The evaluation function is invariant to scaling, therefore we can fix
the ℓ₁‑norm of the step heights to √(2 n).  For a fixed length n the
objective becomes simply  max(conv(a,a)).  The algorithm works as follows:

*  Randomly pick a length n (1500‑3500) and generate a diverse initial
   shape (power‑law, exponential, binary sparse, sinusoidal, …).
*  Scale the vector to the canonical simplex (non‑negative, sum=√(2 n)).
*  Run a fast projected sub‑gradient optimiser:
      – compute the full convolution with FFT,
      – locate the index k where the convolution attains its maximum,
      – the sub‑gradient of max(conv) w.r.t. a is 2·a reversed about k,
      – take a step opposite to that gradient and project back onto the
        simplex,
      – slowly decay the step size and occasionally inject a tiny random
        jitter to escape plateaux.
*  Keep the best sequence seen over many restarts.  After the main loop,
  if time permits a final longer refinement is performed on the current
  best candidate.

The method is deliberately lightweight (O(n log n) per iteration) and
uses many independent restarts, giving a good chance to push the bound
below the target 1.503 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using a power‑of‑2 FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (sum = target, >=0)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum(x) = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries would become zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation families
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law / exponential / mixed / uniform random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)

    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


def _initial_shape_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse binary‑like shape: a random subset of positions carries mass."""
    density = rng.uniform(0.05, 0.25)      # proportion of non‑zero entries
    k = max(1, int(density * n))
    idx = rng.choice(n, size=k, replace=False)
    shape = np.zeros(n, dtype=float)
    shape[idx] = rng.random(k) + 0.1      # avoid values that are too tiny
    shape += rng.random(n) * 1e-6
    return shape


def _initial_shape_sinusoidal(n: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth sinusoidal profile, optionally reshaped by a random exponent."""
    x = np.arange(n, dtype=float)
    shape = np.sin(np.pi * (x + 0.5) / n)        # positive half‑wave
    shape = shape ** rng.uniform(0.5, 2.0)       # random stretching
    shape += rng.random(n) * 1e-6
    return shape


# ----------------------------------------------------------------------
# Core sub‑gradient optimiser for max(conv(a,a))
# ----------------------------------------------------------------------
def _refine_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  max(conv(a,a)).
    Returns the best sequence found and its official evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_seq = a.copy()
    best_val = float("inf")
    # Initial step size – a small fraction of the typical entry size
    lr = max(target_sum * 0.02, 1e-3)
    min_lr = target_sum * 1e-5
    last_improve = 0
    it = 0

    while time.time() < deadline:
        # 1. convolution and location of the current maximum
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        k = conv.argmax()                     # index where max is attained

        # record best so far
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_seq = a.copy()
            last_improve = it

        # 2. sub‑gradient of max(conv) at the active index k
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]          # 2·a reversed about k

        # 3. gradient step and projection onto the simplex
        a -= lr * grad
        a = _project_to_simplex(a, target_sum)

        # 4. step‑size schedule and occasional jitter
        lr = max(lr * 0.9995, min_lr)
        it += 1
        if it % 500 == 0 and rng.random() < 0.3:
            a += rng.normal(scale=target_sum * 5e-4, size=n)
            a = _project_to_simplex(a, target_sum)

        # early stop if no progress for a while
        if it - last_improve > 5000:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

        shape = shape / shape.sum() * target_sum

        # --------------------------------------------------------------
        # Allocate a time slice for this run and refine
        # --------------------------------------------------------------
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _refine_subgradient(shape, target_sum,
                                                     run_deadline, rng)

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            # early exit if we have already beaten the target
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final refinement on the incumbent if time permits
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        seq_refined, score_refined = _refine_subgradient(
            best_seq, target_sum, deadline, rng
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] REFINEMENT improved best to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] FALLBACK sequence, score = {best_score:.8f}")

    # Return a plain Python list as required by the harness
    return best_seq.tolist()
