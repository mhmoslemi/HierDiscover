assistantfinal**Solution Overview**

The objective
\[
F(a)=\frac{2\,n\;\max(a\ast a)}{(\sum a)^2}
\]
is scale‑invariant, therefore we can always keep the vector normalised to
\(\sum a = 1\).  
The task is to find a *non‑negative* step‑function whose self‑convolution is
as flat as possible – i.e. minimise the peak of the convolution.

The programme below implements a multi‑stage optimiser that

1. **generates a diverse set of warm‑starts** (Dirichlet draws, edge‑biased
   power‑laws, Gaussian mixtures, random Gaussian mixtures, etc.);
2. **optimises a smooth‑max surrogate** of the convolution peak with Adam
   (large number of iterations, slowly increasing the temperature);
3. **refines the solution with an exact sub‑gradient (exponentiated‑gradient)**
   method;
4. **performs an aggressive hill‑climb** that swaps mass between two
   coordinates many times;
5. **spends the remaining budget on a long Adam run** on the current global
   best (low learning‑rate, very slow temperature schedule).

All stages stop immediately when the global time‑budget (`budget_s`) is
exhausted, so the function always returns a valid sequence before the
deadline.

```python
"""Propose a low‑value step‑function sequence.

The algorithm:
  * builds a portfolio of diverse initial vectors (Dirichlet, edge‑concentrated,
    Gaussian mixture, random Gaussian mixtures);
  * for each length runs a three‑phase optimisation:
        – Adam on a smooth‑max surrogate (many iterations, slow temperature rise);
        – exact sub‑gradient (exponentiated‑gradient) on the true max;
        – cheap hill‑climb that moves small mass between random indices;
  * keeps the best sequence across all lengths;
  * finally spends any remaining time on a long Adam refinement of the global
    best.

All calculations use FFT‑based convolutions for speed.  The routine respects
the 1000 s budget and always returns a list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper to obtain the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)² (sum a is assumed > 0)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

# Candidate generation ----------------------------------------------------
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _random_gaussian_mix(n: int, components: int = 5) -> np.ndarray:
    """Random mixture of `components` Gaussians on a grid of size n."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    weights = np.random.dirichlet(np.ones(components))
    shape = np.zeros(n, dtype=np.float64)
    for w in weights:
        mu = np.random.rand()
        sigma = np.random.uniform(0.02, 0.25)
        shape += w * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    shape = np.maximum(shape, 0.0)
    tot = shape.sum()
    if tot < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape /= tot
    return shape


def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # warm start from the global best (exact length) – keeps a very good seed
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())

    # warm start by resizing the global best
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # historic sequence (if any)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 4):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑Gaussian edge shapes
    for gamma in (0.1, 0.3, 0.7, 1.5):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.4, 0.7, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # three‑Gaussian peaks (0.25,0.5,0.75) with various widths
    for sigma in (0.04, 0.09, 0.15):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # several random Gaussian mixtures (adds genuine diversity)
    for _ in range(3):
        cand.append(_random_gaussian_mix(n, components=np.random.randint(3, 7)))

    # shuffle and keep a limited number (will be trimmed later)
    np.random.shuffle(cand)
    return cand

# Optimisation primitives -------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int,
                   lr: float = 0.05,
                   lam_init: float = 0.01,
                   lam_factor: float = 1.01) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_a, best_val) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lam = lam_init

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to enforce simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights for the convolution entries
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the smooth surrogate (2 * correlation)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max (the usual soft‑max Jacobian trick)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature (decrease smoothness) slowly
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # small random jitter if we are stuck for a few seconds
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient) on the true max.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val

    last_improve = time.time()
    eta0 = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            break

        # equal distribution over all maximal positions
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # decaying step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Cheap random‑pair hill‑climbing: move a small fraction of mass from i → j.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    # step size slowly shrinks (makes the search finer later on)
    step_factor = 0.9995
    cur_step = 0.05

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to cur_step of the mass from i1 to i2
        eps = np.random.rand() * cur_step * a[i1]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 100 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")

        # slowly reduce the max allowed move size
        cur_step *= step_factor

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    The routine respects the total time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # global best so far
    best_seq = None
    best_val = float("inf")

    # lengths to explore – start with modest sizes, finish with the largest.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # pre‑compute FFT lengths for every n
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = _next_pow2(conv_len)
        # correlation needs at most conv_len + n - 1 ≤ 3·n
        corr_fft[n] = _next_pow2(conv_len + n - 1)

    # load historic solution if present
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # Optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # build a (moderately sized) diverse start‑point pool
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # keep the pool reasonably small for large n (to respect the budget)
        max_cand = 12 if n <= 8192 else 4
        if len(candidates) > max_cand:
            candidates = candidates[:max_cand]

        # iteration budgets – they shrink with n
        max_adam_iters = int(20000 * 1024 / n) + 3000
        max_subgrad_iters = 8000
        max_hill_iters = 8000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – Adam on smooth‑max surrogate ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x,
                deadline, max_adam_iters,
                lr=0.05, lam_init=0.01, lam_factor=1.01
            )

            # ---- Phase 2 – exact sub‑gradient (exponentiated‑gradient) ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x,
                deadline, init_val=val_adam,
                max_iters=max_subgrad_iters
            )

            # ---- Phase 3 – cheap hill‑climbing local search ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c,
                deadline, init_val=val_sub,
                max_iters=max_hill_iters
            )

            # Update global best if improved
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run refinement on the current global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, _next_pow2(2 * n - 1))
        fft_len_x = corr_fft.get(n, _next_pow2(3 * n))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x,
            deadline,
            max_iters=int(40000 * 1024 / n) + 6000,
            lr=0.02, lam_init=0.005, lam_factor=1.005
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

        # an extra hill‑climb pass (still cheap) to squeeze out a few more digits
        if time.time() < deadline:
            a_hc2, val_hc2 = _hill_climb_optimize(
                best_seq, n, fft_len_c,
                deadline, init_val=best_val,
                max_iters=12000
            )
            if val_hc2 < best_val:
                best_val = val_hc2
                best_seq = a_hc2.copy()
                print(f"*** Final hill‑climb improvement value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was ever found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
```
