Let's consider implementing the linear programming iteration.

Procedure:

- Start with an initial f (non-negative sequence of length n). Normalize with sum sqrt(2n) (or any scalar, but shape matters). Compute M = max(conv(f,f)).

- Then construct an LP to find g (also length n) maximizing sum_j g_j subject to g_j >=0 and conv(f,g) <= M (elementwise). Since conv operation is linear in g (f fixed). This is a linear program.

- Solve for g via cvxpy; get optimum g_0. Then rescale g_0 to have sum sqrt(2n) (i.e., normalize). Update f = (1-t) f + t g, with small t (e.g., 0.02). Iterate.

This is similar to alternate projection algorithm that reduces M.

def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i range:
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i  # in [0, n-1]
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

"""
Hybrid LP‑based optimizer for the min‑max self‑convolution problem.

We repeatedly:
  * start from a random non‑negative simplex vector,
  * compute its maximal convolution value M,
  * solve a linear program maximizing the total mass of a new vector g
    under the linear constraints (f * g)_k ≤ M,
  * rescale g to the required L1‑norm and average it with the current f.
Repeatedly doing this drives the maximal convolution down.
Finally a short stochastic hill‑climb refines the best candidate.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution via real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)
    return conv

def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return the matrix A such that (A @ g)[k] = (a * g)[k] (linear in g)."""
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def _lp_optimize(n: int, target_sum: float, deadline: float,
                 max_iter: int = 200, t: float = 0.1) -> tuple[np.ndarray, float]:
    """Iterative LP‑based improvement for a fixed length n.

    Returns the best sequence found and its maximal convolution value.
    """
    # start from a random strictly positive vector (to keep the LP bounded)
    a = np.random.rand(n) + 1e-3
    a = a / a.sum() * target_sum

    best_a = a.copy()
    best_M = _full_convolution_fft(a, a).max()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # current maximal convolution
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        # update global best
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP n={n} it={it} new best M={M:.6f}")
            if M < 1.503:
                break

        # build linear constraints A @ g ≤ M
        A = _build_conv_matrix(a)
        c = -np.ones(n)                     # maximise sum(g)  → minimise -sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.2f}] LP solve failed (it={it}): {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # normalise g to the same L1‐norm as a (target_sum)
        g = g * (target_sum / sum_g)

        # convex update
        a = (1.0 - t) * a + t * g

    return best_a, best_M

def _hillclimb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb that keeps the L1 norm constant."""
    n = a.shape[0]
    target_sum = a.sum()
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # exponential cooling of the acceptance temperature
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - frac))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.08   # move ≤ 8 % of the source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new
            # occasional logging
            if it % 2000 == 0:
                print(f"[{time.time():.2f}] hillclimb it={it} new best M={best_val:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                a = a_new

        # keep the sum exactly constant (numerical drift)
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())

    return best_a, best_val

"""
Hybrid LP‑based optimiser for the min‑max self‑convolution task.

The algorithm works on a fixed length `n`:

1. Initialise a strictly positive vector `a` on the simplex
   (∑ a = √(2 n)).
2. Compute its maximal convolution value `M = max(conv(a,a))`.
3. Solve the linear programme

       maximise   Σ g_j
       subject to   (a * g)_k ≤ M   for all k = 0 … 2n‑2
                    g_j ≥ 0

   The convolution `(a * g)` is linear in `g`; the constraints are built
   as a sparse matrix `A` so that `A @ g` equals the convolution.
4. Rescale `g` to the required L¹‑norm and move a small step
   `a ← (1‑t)·a + t·g`.  The maximal convolution never increases.
   Repeat steps 2‑4 until the time budget expires or the bound stops
   improving.
5. A very cheap stochastic hill‑climb (mass swaps that keep the L¹‑norm
   constant) refines the best candidate.
6. The routine is called repeatedly for a few different problem sizes –
   the best result over all runs is returned.

All operations respect the 1000 s budget, avoid closures and use only
NumPy, SciPy and the provided `evaluate_sequence` function.
"""

import time

import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------

    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Build the matrix that maps g → a*g (linear in g)
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return sparse matrix A such that (A @ g)[k] = (a * g)[k] (full linear convolution).
    """
    n = a.shape[0]
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# LP‑based optimiser for a fixed length
# ----------------------------------------------------------------------
def _lp_optimize(
    n: int,
    target_sum: float,
    deadline: float,
    *,
    max_iter: int = 200,
    step_size: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve a non‑negative step function of length `n`
    by solving a linear programme and taking a convex combination.
    Returns the best coefficient vector found and its maximal convolution value.
    """
    # strictly positive start (needed to keep the LP bounded)
    a = np.random.rand(n) + 1e-3
    a = a / a.sum() * target_sum

    best_a = a.copy()
    best_M = _full_convolution_fft(a, a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _full_convolution_fft(a, a)
        M = conv.max()

        # record the best vector seen so far
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP n={n} it={it} new best M={M:.6f}")
            if M < 1.503:                     # early exit if target reached
                break

        # linear constraints: A @ g ≤ M (component‑wise)
        A = _build_conv_matrix(a)
        c = -np.ones(n)                      # maximise Σ g → minimise –Σ g
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        # solve the LP (highs is a fast interior‑point/dual‑simplex method)
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[{time.time():.2f}] LP failed (it={it}): {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # rescale g to the required L¹‑norm (the same as a)
        g = g * (target_sum / sum_g)

        # convex update (the original paper uses a small t>0)
        a = (1.0 - step_size) * a + step_size * g

    return best_a, best_M


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (exact objective)
# ----------------------------------------------------------------------
def _hillclimb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random local search that moves a tiny amount of mass between two indices
    while keeping the L¹‑norm constant.  Accept improvements always,
    otherwise accept with a Metropolis probability that cools linearly.
    """
    n = a.shape[0]
    target_sum = a.sum()
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule – start relatively hot, cool to near zero
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move at most 8 % of the source mass
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                if it % 2000 == 0:
                    print(f"[{time.time():.2f}] hillclimb it={it} new best M={best_val:.6f}")

        # re‑normalise occasionally to avoid drift caused by rounding
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the external evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a step‑function that minimises

            2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine tries several problem sizes, runs the LP optimiser for each
    (with a generous time slice), and finishes with a short hill‑climb.
    The best sequence discovered before the deadline is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Try a handful of lengths – short runs give many independent attempts.
    # ------------------------------------------------------------------
    length_candidates = [600, 800, 1000, 1300, 1700]

    for n in length_candidates:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # give roughly a quarter of the remaining time to this length
        slice_deadline = min(deadline,
                             time.time() + (deadline - time.time()) * 0.25)

        # --------------------------------------------------------------
        # Phase 1 – LP based optimisation
        # --------------------------------------------------------------
        a_lp, M_lp = _lp_optimize(
            n,
            target_sum,
            slice_deadline,
            max_iter=250,
            step_size=0.12,
        )
        score_lp = evaluate_sequence(a_lp.tolist())
        if score_lp < best_score:
            best_score = float(score_lp)
            best_seq = a_lp.copy()
            print(f"[{time.time():.2f}] length {n} LP best score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – cheap hill‑climb for fine tuning (uses remaining time)
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_hill, M_hill = _hillclimb(best_seq.copy(), deadline)
            score_hill = evaluate_sequence(a_hill.tolist())
            if score_hill < best_score:
                best_score = float(score_hill)
                best_seq = a_hill.copy()
                print(f"[{time.time():.2f}] after hill‑climb best score={best_score:.6f}")

        # early exit if we have already passed the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform construction (should never be used).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
