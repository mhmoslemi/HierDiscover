def propose_candidate(seed:int=0, budget_s:float=1000):
    import time, numpy as np
    np.random.seed(seed)
    end = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    # try lengths
    lengths = [4096, 8192, 16384]
    for n in lengths:
        if time.time() > end:
            break
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(3*n)))
        # generate seeds:
        seeds = []
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # previous best if same length
        if best_seq is not None and len(best_seq) == n:
            seeds.append(np.asarray(best_seq, dtype=float))
        # random Dirichlet
        seeds.append(np.random.dirichlet(np.ones(n)))
        # spiky Dirichlet
        for conc in [0.2,0.5,0.8]:
            seeds.append(np.random.dirichlet(np.full(n, conc)))
        # sparse spikes: choose k and assign masses
        k = max(2, n//2000)  # e.g., 2 spikes for 4000 maybe 2; adjust
        for _ in range(5):
            positions = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a = np.zeros(n)
            a[positions] = masses
            seeds.append(a)
        # powerlaw edge shape
        idx = np.arange(n)+1
        prod = idx * (n - idx + 1)
        for alpha in np.linspace(-2.0,-0.2,9):
            shape = np.power(prod, alpha, out=np.zeros_like(prod), where=prod>0)
            if shape.sum()>0:
                shape /= shape.sum()
                seeds.append(shape)
        # two-sided exponential
        for gamma in [0.0005,0.001,0.002,0.005,0.01]:
            shape = np.exp(-gamma*np.arange(n)) + np.exp(-gamma*(n-1-np.arange(n)))
            shape /= shape.sum()
            seeds.append(shape)

        # Loop seeds
        for seed_idx, a0 in enumerate(seeds):
            if time.time() > end:
                break
            # Normalize
            a = a0 / (a0.sum() + 1e-15)
            logits = np.log(a + 1e-15)  # parameters for softmax
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1=0.9; beta2=0.999; eps=1e-8
            lr=0.06
            lm=0.2
            lm_factor=1.02
            # iteration budget
            max_iters = int(15000 * 1024 / n) + 2000
            last_improve = time.time()
            for t in range(1, max_iters+1):
                # softmax
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                # convolution via FFT
                conv = full_convolution(a, conv_len)
                max_c = conv.max()
                # smooth max weights
                w = np.exp(lm * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w = w / w_sum
                grad_a = 2.0 * cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                # Adam
                m = beta1 * m + (1-beta1) * grad_log
                v = beta2 * v + (1-beta2) * (grad_log**2)
                m_hat = m / (1-beta1**t)
                v_hat = v / (1-beta2**t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
                lm = min(lm * lm_factor, 1e6)
                # evaluate objective for our metric (scaled to sum=1)
                val = 2.0 * n * max_c
                # update global best
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} seed={seed_idx} iter={t}] new best {val:.6f}")
                # random shake if stagnant
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n)-0.5) * 1e-2
                    last_improve = time.time()
                if time.time() > end:
                    break
            # subgradient exponentiated gradient
            a = a / a.sum()
            eta = 0.6
            sub_iters = 4000
            sub_last = time.time()
            for s in range(sub_iters):
                conv = full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0/active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * cross_correlation(w, a, corr_len)
                a = a * np.exp(-eta * grad_a)
                total = a.sum()
                if total < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    total = a.sum()
                a = a / total
                max_c = full_convolution(a, conv_len).max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} seed={seed_idx} sub={s}] new best {val:.6f}")
                if time.time() - sub_last > 5.0:
                    break
                if time.time() > end:
                    break
        # fine tuning of best for this length
        if best_seq is not None and len(best_seq) == n:
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits); v = np.zeros_like(logits)
            lr=0.04; lm=0.5; lm_factor=1.01
            fine_iters = int(30000 * 1024 / n) + 5000
            last_improve = time.time()
            for t in range(1, fine_iters+1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lm * (conv - max_c))
                w = w / w.sum()
                grad_a = 2.0 * cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                m = beta1 * m + (1-beta1) * grad_log
                v = beta2 * v + (1-beta2) * (grad_log**2)
                m_hat = m / (1-beta1**t)
                v_hat = v / (1-beta2**t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
                lm = min(lm * lm_factor, 1e6)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine={t}] new best {val:.6f}")
                if time.time() - last_improve > 8.0:
                    logits += (np.random.rand(n)-0.5) * 5e-3
                    last_improve = time.time()
                if time.time() > end:
                    break

    # fallback uniform if not found
    if best_seq is None:
        best_seq = np.full(2048, 1.0/2048)
    # return as list
    return best_seq.tolist()

"""Search for a step‑function that minimises 2·n·max(conv)/sum^2.

The optimizer combines three ideas:
* diverse initial seeds (uniform, Dirichlet, sparse spikes, exponential,
  power‑law and sine‑shaped patterns);
* a long Adam optimisation on a smooth‑max surrogate of the max‑convolution;
* a final simulated‑annealing refinement that perturbs the best candidate
  while recomputing the exact convolution each step.

The routine respects a time‑budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""
import time
import numpy as np

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    The result has length len(a) and is
        (w ★ a)[i] = Σ_k w[k]·a[i+k−(n−1)]   (n = len(a)).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    conv = _full_convolution(a, conv_len)
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths we will try – larger means a richer search space but slower FFTs.
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(3 * n)))   # safe bound for cross‑corr

        # -----------------------------------------------------------------
        # Build a diverse list of starting points
        seeds = []

        # Warm‑start from the current global best if it matches the length
        if best_seq is not None and len(best_seq) == n:
            seeds.append(np.asarray(best_seq, dtype=np.float64))

        # Historic solution, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            cand = cand / (cand.sum() + 1e-15)
            seeds.append(cand)

        # Uniform vector – the trivial baseline
        seeds.append(np.full(n, 1.0 / n))

        # Dirichlet (concentration 1) and spiky Dirichlet (c < 1)
        seeds.append(np.random.dirichlet(np.ones(n)))
        for conc in (0.2, 0.5, 0.8):
            seeds.append(np.random.dirichlet(np.full(n, conc)))

        # Sparse random spikes – a small number of equally weighted positions
        for s in (int(np.sqrt(n)), int(2 * np.sqrt(n)), int(4 * np.sqrt(n))):
            for _ in range(3):
                pos = np.random.choice(n, size=s, replace=False)
                mass = np.random.dirichlet(np.ones(s))
                a = np.zeros(n)
                a[pos] = mass
                seeds.append(a)

        # Edge‑concentrated power‑law shapes
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.2, 9):
            shape = np.power(prod, alpha, out=np.zeros_like(prod), where=prod > 0)
            if shape.sum() > 0:
                shape /= shape.sum()
                seeds.append(shape)

        # Two‑sided exponential shape
        for gamma in (0.0005, 0.001, 0.002, 0.005, 0.01):
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape /= shape.sum()
            seeds.append(shape)

        # Smooth sine window – always non‑negative
        sine = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        sine /= sine.sum()
        seeds.append(sine)

        # -----------------------------------------------------------------
        # Optimise each seed with Adam on the smooth‑max surrogate
        for seed_id, a0 in enumerate(seeds):
            if time.time() > deadline:
                break

            a = a0.copy()
            if a.sum() == 0:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            logits = np.log(a + 1e-15)

            # Adam state
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07
            lam = 0.2
            lam_factor = 1.02

            # Number of inner iterations – scales inversely with the length
            max_iters = int(20000 * 1024 / n) + 5000
            last_impr = time.time()

            for t in range(1, max_iters + 1):
                # soft‑max to enforce the simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()

                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the smooth‑max (chain rule through the soft‑max)
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule – slowly turn the surrogate into the true max
                lam = min(lam * lam_factor, 1e6)

                # evaluate the exact objective (sum(a)=1)
                cur_val = 2.0 * n * max_c
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_impr = time.time()
                    print(f"[n={n} seed={seed_id} Adam iter={t:05d}] new best {cur_val:.6f}")

                # occasional random shake to get out of plateaus
                if time.time() - last_impr > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_impr = time.time()

                if time.time() > deadline:
                    break

            # -------------------------------------------------------------
            # Sub‑gradient (exponentiated‑gradient) phase on the exact max
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(4000):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                a = a * np.exp(-eta * grad_a)
                total = a.sum()
                if total < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    total = a.sum()
                a = a / total

                cur_val = 2.0 * n * _full_convolution(a, conv_len).max()
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} seed={seed_id} Subgrad iter={sub_it}] new best {cur_val:.6f}")

                if time.time() - sub_last > 7.0:
                    break
                if time.time() > deadline:
                    break

        # -----------------------------------------------------------------
        # Simulated‑annealing refinement of the *global* best for this length
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] annealing refinement (current best {best_val:.6f})")
            a = best_seq.copy()
            cur_val = best_val
            temperature = 0.1 * cur_val
            for step in range(50000):
                if time.time() > deadline:
                    break
                # propose a small transfer of mass between two random positions
                i, j = np.random.choice(n, 2, replace=False)
                max_delta = min(a[i], 0.2)                      # we never move more than 20 % of a[i]
                delta = max_delta * np.random.rand()
                if delta <= 0:
                    continue
                a_prop = a.copy()
                a_prop[i] -= delta
                a_prop[j] += delta
                # re‑normalise to avoid accumulated rounding error
                a_prop /= a_prop.sum()
                val_prop = _objective(a_prop, n, conv_len)

                # Metropolis acceptance
                if val_prop < cur_val or np.random.rand() < np.exp(-(val_prop - cur_val) / temperature):
                    a = a_prop
                    cur_val = val_prop
                    if cur_val < best_val:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[n={n} anneal step={step}] new best {cur_val:.6f}")

                # exponential cooling
                temperature *= 0.99995
                if temperature < 1e-8:
                    temperature = 1e-8

            # -----------------------------------------------------------------
            # One final short Adam fine‑tune (starting from the annealed vector)
            logits = np.log(best_seq + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            lr = 0.04
            lam = 0.5
            lam_factor = 1.01
            fine_iters = int(30000 * 1024 / n) + 5000
            last_impr = time.time()
            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w /= w.sum()
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                    best_seq = a.copy()

    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    conv = _full_convolution(a, conv_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a low‑score sequence under the given time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # give a tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more expressive vectors but slower FFTs.
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        # FFT lengths (next power of two) – fast for NumPy's FFT implementation.
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(3 * n)))   # safe upper bound for the correlation

        # --------------------------------------------------------------
        # Build a diverse collection of initial candidates.
        seeds = []

        # Warm‑start from a previous global best (if length matches).
        if best_seq is not None and len(best_seq) == n:
            seeds.append(np.asarray(best_seq, dtype=np.float64))

        # Historic solution, if available.
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            cand /= (cand.sum() + 1e-15)
            seeds.append(cand)

        # Uniform vector – the trivial baseline.
        seeds.append(np.full(n, 1.0 / n))

        # Dirichlet draws with various concentrations.
        seeds.append(np.random.dirichlet(np.ones(n)))
        for conc in (0.2, 0.5, 0.8):
            seeds.append(np.random.dirichlet(np.full(n, conc)))

        # Sparse random spikes (different numbers of active sites).
        for s in (int(np.sqrt(n)), int(2 * np.sqrt(n)), int(4 * np.sqrt(n))):
            for _ in range(3):
                pos = np.random.choice(n, size=s, replace=False)
                mass = np.random.dirichlet(np.ones(s))
                a = np.zeros(n)
                a[pos] = mass
                seeds.append(a)

        # Edge‑concentrated power‑law shapes.
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.2, 9):
            shape = np.power(prod, alpha, out=np.zeros_like(prod), where=prod > 0)
            if shape.sum() > 0:
                shape /= shape.sum()
                seeds.append(shape)

        # Two‑sided exponential decays.
        for gamma in (0.0005, 0.001, 0.002, 0.005, 0.01):
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape /= shape.sum()
            seeds.append(shape)

        # Smooth sine window (always non‑negative).
        sine = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        sine /= sine.sum()
        seeds.append(sine)

        # --------------------------------------------------------------
        # Optimise every seed with a long Adam run on the smooth‑max surrogate.
        for seed_id, a0 in enumerate(seeds):
            if time.time() > deadline:
                break

            a = a0.copy()
            if a.sum() == 0:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            logits = np.log(a + 1e-15)

            # Adam state.
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07
            lam = 0.2
            lam_factor = 1.02

            # Number of inner iterations grows for smaller vectors.
            max_iters = int(20000 * 1024 / n) + 5000
            last_impr = time.time()

            for t in range(1, max_iters + 1):
                # Soft‑max → enforce simplex.
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()

                # Smooth‑max weights.
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # Gradient of the surrogate (chain rule through the soft‑max).
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update.
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # Temperature schedule – slowly turn the smooth‑max into the true max.
                lam = min(lam * lam_factor, 1e6)

                cur_val = 2.0 * n * max_c
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_impr = time.time()
                    print(f"[n={n} seed={seed_id} Adam iter={t:05d}] new best {cur_val:.6f}")

                # Small random shake if we have been idle for a while.
                if time.time() - last_impr > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_impr = time.time()

                if time.time() > deadline:
                    break

            # ----------------------------------------------------------
            # Exponentiated‑gradient sub‑gradient phase on the *exact* max.
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(4000):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                a = a * np.exp(-eta * grad_a)
                total = a.sum()
                if total < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    total = a.sum()
                a = a / total

                cur_val = 2.0 * n * _full_convolution(a, conv_len).max()
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} seed={seed_id} Subgrad iter={sub_it}] new best {cur_val:.6f}")

                if time.time() - sub_last > 7.0:
                    break
                if time.time() > deadline:
                    break

        # --------------------------------------------------------------
        # Simulated‑annealing refinement of the *global* best for this length.
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] annealing refinement (current best {best_val:.6f})")
            a = best_seq.copy()
            cur_val = best_val
            temperature = 0.1 * cur_val
            for step in range(50000):
                if time.time() > deadline:
                    break
                # Propose a tiny transfer of mass between two random indices.
                i, j = np.random.choice(n, 2, replace=False)
                max_delta = min(a[i], 0.2)        # never move more than 20 % of a[i]
                delta = max_delta * np.random.rand()
                if delta == 0.0:
                    continue
                a_prop = a.copy()
                a_prop[i] -= delta
                a_prop[j] += delta
                a_prop /= a_prop.sum()
                val_prop = _objective(a_prop, n, conv_len)

                # Metropolis acceptance.
                if (val_prop < cur_val) or (np.random.rand() < np.exp(-(val_prop - cur_val) / temperature)):
                    a = a_prop
                    cur_val = val_prop
                    if cur_val < best_val:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[n={n} anneal step={step}] new best {cur_val:.6f}")

                # Exponential cooling.
                temperature *= 0.99995
                if temperature < 1e-8:
                    temperature = 1e-8

            # ----------------------------------------------------------
            # One short Adam fine‑tune starting from the annealed vector.
            logits = np.log(best_seq + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            lr = 0.04
            lam = 0.5
            lam_factor = 1.01
            fine_iters = int(30000 * 1024 / n) + 5000
            last_impr = time.time()
            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w /= w.sum()
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                cur_val = 2.0 * n * max_c
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_impr = time.time()
                    print(f"[n={n} fine‑tune iter={t}] new best {cur_val:.6f}")

                if time.time() - last_impr > 10.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_impr = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (should never happen).
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
