"""
Propose a non‑negative step‑function (list of floats) that minimises  

    F(a) = 2·n·max(a*a) / (sum(a)²)   

where n = len(a) and a*a denotes the linear convolution of a with itself.
The optimiser works on a hierarchy of increasing lengths (1024 … 16384).  
For each length it builds a diverse set of seeds (uniform, up‑sampled,
Dirichlet draws with several concentrations, edge‑concentrated power‑laws,
two‑Gaussian edge shapes, sine‑squared bump, arcsine shape, alternating
high‑low pattern, etc.).  
Every seed is refined by two phases:

1. **Adam on a smooth‑max surrogate** – the max is approximated by a
   log‑sum‑exp with an annealed temperature λ.  The vector is kept in the
   probability simplex by a soft‑max parametrisation (logits).  Adam’s
   momentum is used together with a simple stagnation shake‑up.

2. **Exact sub‑gradient refinement** – after the surrogate phase the true
   max is used.  The sub‑gradient (uniform over the active maximal lags)
   is applied with an exponentiated‑gradient (mirror‑descent) update that
   automatically respects non‑negativity and normalisation.

Both phases keep track of the best objective value seen over the whole
run.  At the end the best sequence found (as a plain Python list) is
returned.  The implementation is fully vectorised, uses only NumPy’s FFT
routines and respects a wall‑clock budget (default 1000 s).
"""

import os
import time
import numpy as np

# limit BLAS/FFT threads to avoid exceeding the 2‑CPU limit
os.environ["OMP_NUM_THREADS"] = "2"
os.environ["OPENBLAS_NUM_THREADS"] = "2"
os.environ["MKL_NUM_THREADS"] = "2"
os.environ["VECLIB_MAXIMUM_THREADS"] = "2"
os.environ["NUMEXPR_NUM_THREADS"] = "2"


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a).  Returns a vector of length n such that

        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0 … n‑1 .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _smooth_max(conv: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """
    Returns (max_c, weights) where weights are a probability distribution
    over the entries of ``conv`` obtained from a log‑sum‑exp with temperature
    ``lam``.
    """
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    return max_c, w


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function that minimises the evaluation
    function defined in the problem statement.  The routine respects the
    supplied wall‑clock ``budget_s`` (seconds) and returns the best sequence
    as a plain Python list.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper that computes the true objective for a normalised vector.
    # ------------------------------------------------------------------
    def _objective(a: np.ndarray) -> float:
        # a is assumed to be non‑negative and sum(a) > 0
        n = a.shape[0]
        conv = _full_convolution(a)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # Define the sequence lengths that will be processed.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a diverse pool of seeds.
        # --------------------------------------------------------------
        candidates = []

        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) Warm‑start from a previously found solution of exactly this size

            candidates.append(np.asarray(best_seq, dtype=np.float64))

            candidates.append(up)

        candidates.append(np.random.dirichlet(np.ones(n)))

                break

            logits = np.log(a0)

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            # smooth‑max temperature schedule
            lam = 0.2
            lam_factor = 1.02

            max_iters = int(20000 * 1024 / n) + 4000
            # give a little extra budget for the largest length

            max_c, w = _smooth_max(conv, lam)

            avg = np.dot(a, grad_a)
