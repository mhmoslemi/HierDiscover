def generate_poly_seq(p, n, target_sum):
    # p shape (D+1,)
    i = np.arange(n) / (n-1)
    # compute loga = Σ p[d] * i**d
    # Use Horner's method for efficiency:
    loga = np.polyval(p[::-1], i) # Because polyval expects highest degree first.
    # exponentiate stable:
    max_log = np.max(loga)
    a_raw = np.exp(loga - max_log)
    a_raw = a_raw / a_raw.sum() * target_sum
    return a_raw

def generate_beta_seq(ab, n, target_sum):
    α, β = ab
    i = np.arange(1, n+1)  # 1..n
    a_raw = (i ** α) * ((n - i + 1) ** β)  # ensure shape
    # may have large dynamic range -> stable scaling:
    max_val = np.max(a_raw)
    if max_val == 0:
        a_raw = np.full(n, 1.0)
    else:
        a_raw = a_raw / max_val  # bring to [0,1]
    a_raw = a_raw / a_raw.sum() * target_sum
    return a_raw

def generate_gaussian_seq(params, n, target_sum):
    μ, log_sigma = params
    sigma = np.exp(log_sigma)
    i = np.arange(n)
    loga = - ((i - μ) ** 2) / (2 * sigma ** 2)
    # ensure not overflow:
    max_log = np.max(loga)
    a_raw = np.exp(loga - max_log)
    a_raw = a_raw / a_raw.sum() * target_sum
    return a_raw

"""
Candidate generator for the step‑function minimisation problem.

The algorithm proceeds in four phases:

1. **Warm‑start** – if a previously discovered sequence is available under the
   name ``height_sequence_1`` it is normalised to the canonical L1‑norm
   ``sqrt(2 n)`` and used as the initial best solution.

2. **Low‑dimensional parametric search** – a cheap random / simulated‑annealing
   search over three families of shapes (exponential‑polynomial, Beta‑type and
   Gaussian) with only a few parameters.  Each generated shape is normalised
   to the required sum and evaluated with ``evaluate_sequence``.  The best
   sequence found becomes the new global best.

3. **Soft‑max Adam refinement** – the current best sequence is re‑parameterised
   as a soft‑max of unconstrained logits and optimised with Adam on a smooth
   surrogate of the max‑convolution (log‑sum‑exp).  The routine respects the
   remaining time budget and updates the global best whenever an improvement
   is observed.

4. **Exact local polishing** – first a projected sub‑gradient descent on the
   true max‑convolution, then a stochastic hill‑climb (Metropolis criterion)
   are applied to the best candidate, again keeping the best solution.

All phases are time‑controlled; the function returns the best sequence found
within the allotted budget (default 1000 s).  The returned list consists of
non‑negative floats and the objective value is the evaluation score
``2 · n·max(conv(a,a))/sum(a)²`` (which equals the maximal convolution value
when the sum is normalised to ``sqrt(2n)``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # tiny negative values appear because of rounding – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {w ≥ 0, Σw = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.15,
    min_tau_factor: float = 0.0005,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (as a numpy array) and its *exact*
    evaluation via ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- map logits to simplex (soft‑max) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)            # stable
        p = exp_shift / np.sum(exp_shift)        # probabilities, Σp = 1
        a = target_sum * p

        # ---- exact max of the convolution ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new local best max={cur_max:.6f}")

        # ---- smooth‑max surrogate (log‑sum‑exp) ----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient wrt a (2 · a ⊛ w_rev) ----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # positions that correspond to Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- back‑prop to logits (chain rule) ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # tiny jitter every few hundred iterations
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    # final evaluation via the provided evaluator (identical to max)
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(conv(a, a)) with Euclidean simplex projection.
    Returns the best sequence (numpy array) and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                # index of the current max

        # sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step & projection
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02
        else:
            step_size *= 0.70

        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}")

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Random local search that moves a small mass between two positions while
    keeping the simplex constraint.  Returns the best sequence and its
    maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Low‑dimensional parametric families
# ----------------------------------------------------------------------
def _generate_poly_seq(p: np.ndarray, n: int, target_sum: float) -> np.ndarray:
    """Exponential‑polynomial shape: a_i ∝ exp(∑ p_d·(i/(n‑1))^d)."""
    i = np.arange(n, dtype=float) / (n - 1)
    # polyval expects highest degree first; reverse p for that order
    loga = np.polyval(p[::-1], i)
    # stabilise exponentiation
    max_log = np.max(loga)
    a_raw = np.exp(loga - max_log)
    a_raw = a_raw / a_raw.sum() * target_sum
    return a_raw


def _generate_beta_seq(ab: tuple, n: int, target_sum: float) -> np.ndarray:
    """Beta‑type shape: a_i ∝ (i+1)^α·(n‑i)^β."""
    α, β = ab
    i = np.arange(1, n + 1, dtype=float)
    a_raw = (i ** α) * ((n - i + 1) ** β)
    # avoid overflow/underflow
    max_val = np.max(a_raw)
    if max_val == 0:
        a_raw = np.ones_like(a_raw)
    else:
        a_raw = a_raw / max_val
    a_raw = a_raw / a_raw.sum() * target_sum
    return a_raw


def _generate_gauss_seq(params: tuple, n: int, target_sum: float) -> np.ndarray:
    """Gaussian envelope shape: a_i ∝ exp(- (i‑μ)² / (2σ²))."""
    μ, log_sigma = params
    sigma = np.exp(log_sigma)
    i = np.arange(n, dtype=float)
    loga = - ((i - μ) ** 2) / (2.0 * sigma ** 2)
    max_log = np.max(loga)
    a_raw = np.exp(loga - max_log)
    a_raw = a_raw / a_raw.sum() * target_sum
    return a_raw


# ----------------------------------------------------------------------
# Randomised search over the three families
# ----------------------------------------------------------------------
def _param_random_search(
    current_seq: np.ndarray | None,
    current_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap random / simulated‑annealing search in the low‑dimensional shape space.
    Returns the best sequence found and its maximal convolution value.
    """
    best_seq = None if current_seq is None else current_seq.copy()
    best_score = np.inf if current_score is None else current_score
    it = 0

    while time.time() < deadline:
        it += 1
        # pick a family
        family = np.random.choice(['poly', 'beta', 'gauss'])

        # pick a length (bias towards the current length if we have one)
        if best_seq is not None:
            base_n = best_seq.shape[0]
            n = int(base_n + np.random.randint(-200, 201))
            n = max(600, n)
        else:
            n = np.random.randint(800, 3000)

        target_sum = np.sqrt(2.0 * n)

        if family == 'poly':
            D = 5
            p = np.random.randn(D + 1) * 1.0
            seq = _generate_poly_seq(p, n, target_sum)
        elif family == 'beta':
            α = np.random.uniform(0.0, 5.0)
            β = np.random.uniform(0.0, 5.0)
            seq = _generate_beta_seq((α, β), n, target_sum)
        else:  # Gaussian family
            μ = np.random.uniform(0, n - 1)
            log_sigma = np.random.uniform(np.log(0.01 * n), np.log(0.5 * n))
            seq = _generate_gauss_seq((μ, log_sigma), n, target_sum)

        # safeguard against pathological NaNs / infs
        if np.any(np.isnan(seq)) or np.any(np.isinf(seq)):
            continue

        # evaluate using the official evaluator (returns the full score)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = float(score)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] param-search new best={best_score:.6f} n={n} fam={family}")

        # occasional tiny perturbation of the current best to escape flat regions
        if best_seq is not None and it % 250 == 0:
            eps = best_seq.mean() * 0.02
            pert = best_seq + np.random.randn(n) * eps
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum() * target_sum
            score = evaluate_sequence(pert.tolist())
            if score < best_score:
                best_score = float(score)
                best_seq = pert.copy()
                print(f"[{time.time():.2f}] param-perturb improved best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function with minimal evaluation score.
    Orchestrates warm‑start, parametric random search, Adam refinement,
    projected sub‑gradient descent and final hill‑climbing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Phase 0 – warm‑start from a previously stored construction
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start score={best_score:.6f} n={n0}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – cheap parametric random search (≈10 % of budget)
    # --------------------------------------------------------------
    param_deadline = min(deadline, start_time + budget_s * 0.10)
    best_seq, best_score = _param_random_search(best_seq, best_score, param_deadline)

    # --------------------------------------------------------------
    # Phase 2 – soft‑max Adam optimisation (≈45 % of remaining time)
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        slice_time = min(30.0, remaining * 0.45)   # each Adam run gets ≤30 s
        if best_seq is None:
            n = np.random.randint(1500, 2800)
            target_sum = np.sqrt(2.0 * n)
            a0 = np.random.rand(n)
            a0 = a0 / a0.sum() * target_sum
            x0 = np.log(a0 + 1e-12)
        else:
            n = best_seq.shape[0]
            target_sum = np.sqrt(2.0 * n)
            x0 = np.log(best_seq + 1e-12)

        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            time.time() + slice_time,
            init_tau_factor=0.15,
            min_tau_factor=0.0005,
            alpha0=0.025,
        )
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] Adam improved best={best_score:.6f} n={n}")
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (≈30 % of remaining time)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 8.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(30.0, time_left * 0.40)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] Subgradient improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – stochastic hill‑climbing for final polishing
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = time.time() + time_left
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] Hill‑climb final best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
