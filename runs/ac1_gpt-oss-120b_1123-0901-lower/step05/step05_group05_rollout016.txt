def _optimise_adam_pnorm(n, x0, target_sum, deadline, p_start, p_end, alpha0, beta1, beta2, eps):

    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # compute a
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        # update best
        if max_conv < best_val:
            best_val = max_conv
            best_a = a.copy()
            print(...)

        # schedule p
        p_val = p_start + (p_end - p_start) * (it / (it + 1000.0))  # gradual increase

        # compute weights on conv
        weight = np.power(conv, max(p_val - 1.0, 0.0))  # conv ** (p-1)
        # handle conv zero entries: set weight zero
        # normalize
        w_sum = weight.sum()
        if w_sum == 0.0:
            w = np.full_like(weight, 1.0 / weight.size)
        else:
            w = weight / w_sum

        # gradient wrt a: same as weighted sum: grad_a = 2 * convolution(a, w_rev) [same indices]
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back propagate to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # decay alpha
        alpha *= 0.9995

"""
Hybrid non‑convex optimizer for the step‑function problem.

The algorithm consists of four stages:

1. **Diverse initialisations** – random Dirichlet vectors, exponential
   tails and power‑law shapes (parameterised by a few random numbers) are
   generated for a chosen length `n`.  Each is turned into soft‑max
   logits so that the non‑negativity and sum constraint are handled
   automatically.

2. **Annealed Adam optimisation** – two Adam based surrogates are used:
   * a log‑sum‑exp “soft‑max” surrogate (identical to the original code),
   * a *p‑norm* surrogate `f_p(a)= (∑ conv_i(a)^p)^{1/p}` with a slowly
     increasing exponent `p`.  Both work in the log‑it space (soft‑max)
     and respect the simplex via a built‑in projection.  The p‑norm
     surrogate provides a smoother approximation of the true max for
     intermediate values of `p` and a tighter approximation as `p`
     grows.

3. **Projected sub‑gradient descent** – starting from the best Adam
   candidate, the exact sub‑gradient of the max‑convolution is taken
   and a simple adaptive step‑size rule (increase on success,
   decrease on failure) is applied, with Euclidean projection onto the
   simplex.

4. **Targeted stochastic hill‑climbing** – mass is moved from indices
   with large sub‑gradient values to indices with small ones; moves are
   accepted if they improve the exact objective or according to a
   decreasing temperature schedule.

The whole routine respects a 1000 s wall‑time budget and falls back to
a uniform distribution whenever no candidate has been found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # clip negatives from rounding
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex with radius s (s ≥ 0)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam on the soft‑max parametrisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser using a log‑sum‑exp surrogate for max(conv)."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to the simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] softmax Adam n={n}  new best max={cur_max:.6f}")

        # smooth‑max weights (log‑sum‑exp surrogate)
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a  (2 · a ⊛ w_rev)
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑propagation to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9995

    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Adam with a p‑norm surrogate (gradually increasing exponent)
# ----------------------------------------------------------------------
def _optimise_adam_pnorm(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    p_start: float = 2.0,
    p_end: float = 50.0,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser where the surrogate is (∑ conv_i^p)^{1/p}."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to the simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p_vec = exp_shift / np.sum(exp_shift)
        a = target_sum * p_vec

        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] p‑norm Adam n={n}  new best max={cur_max:.6f}")

        # schedule exponent p (slowly increase)
        p_cur = p_start + (p_end - p_start) * (it / (it + 1500.0))

        # compute weights w_i ∝ conv_i^{p‑1}
        max_c = conv.max()
        if max_c <= 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            # normalise to avoid overflow
            w = (conv / max_c) ** (p_cur - 1.0)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

        # gradient of the surrogate w.r.t. a
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑propagation to logits
        dotp = np.dot(p_vec, grad_a)
        grad_x = target_sum * p_vec * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # jitter & learning‑rate decay
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001
        alpha *= 0.9995

    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with adaptive step‑size."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.05 * target_sum / n  # start with ≈ 5 % of the average entry

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())  # index of the current maximum

        # sub‑gradient of conv_k
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step with projection onto the simplex
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02  # modestly increase after success
        else:
            step_size *= 0.70  # shrink after failure

        # occasional jitter to avoid stagnation
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] sub‑grad it={it}  step={step_size:.3e}  best={best_val:.6f}")

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass from high‑gradient to low‑gradient positions."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        # gradient‑guided index selection
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # pick source (large gradient) and destination (small gradient)
        src = np.argmax(grad)
        dst = np.argmin(grad)

        if src == dst or a[src] <= 0.0:
            # fall back to random move
            i, j = np.random.randint(0, n, size=2)
            src, dst = i, j

        # move up to 15 % of the source mass
        delta = a[src] * np.random.rand() * 0.15
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        # re‑normalise occasionally (numerical drift)
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper functions to generate diverse initial vectors
# ----------------------------------------------------------------------
def _init_random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    a = np.random.rand(n)
    a = a / a.sum() * target_sum
    return a


def _init_exponential(n: int, target_sum: float) -> np.ndarray:
    r = np.random.uniform(0.05, 0.90)
    idx = np.arange(n, dtype=float)
    a = (r ** idx) + (r ** (n - 1 - idx))
    a = a / a.sum() * target_sum
    return a


def _init_powerlaw(n: int, target_sum: float) -> np.ndarray:
    p = np.random.uniform(0.1, 2.0)
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p) + (n + 1 - idx) ** (-p)
    a = a / a.sum() * target_sum
    return a

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a globally available construction, if any.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many short Adam runs with diverse lengths / seeds.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # keep at least ~200 s for later phases
        if remaining < 220.0:
            break

        slice_time = min(15.0, remaining * 0.5)
        run_deadline = time.time() + slice_time

        # choose problem size (bias towards larger n while time permits)
        if remaining > 600:
            n = np.random.randint(2500, 6000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 4500 + 1)
        else:
            n = np.random.randint(1200, 2500 + 1)

        target_sum = np.sqrt(2.0 * n)

        # -------------------- generate an initial vector --------------------
        init_choice = np.random.rand()
        if init_choice < 0.35:
            a0 = _init_random_dirichlet(n, target_sum)
        elif init_choice < 0.60:
            a0 = _init_exponential(n, target_sum)
        elif init_choice < 0.85:
            a0 = _init_powerlaw(n, target_sum)
        else:
            # a noisy copy of the current global best (trim / pad as needed)
            if best_seq is not None:
                if best_seq.shape[0] >= n:
                    a0 = best_seq[:n].copy()
                else:
                    pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                    a0 = np.concatenate([best_seq, pad])
                a0 = a0 / a0.sum() * target_sum
            else:
                a0 = _init_random_dirichlet(n, target_sum)

        x0 = np.log(a0 + 1e-12)                     # logits for the soft‑max

        # -------------------- pick optimiser --------------------
        if np.random.rand() < 0.55:
            # soft‑max (log‑sum‑exp) surrogate
            seq_local, score_local = _optimise_adam_softmax(
                n,
                x0,
                target_sum,
                run_deadline,
                init_tau_factor=0.20,
                min_tau_factor=0.001,
                alpha0=0.03,
            )
        else:
            # p‑norm surrogate
            seq_local, score_local = _optimise_adam_pnorm(
                n,
                x0,
                target_sum,
                run_deadline,
                p_start=2.0,
                p_end=50.0,
                alpha0=0.03,
            )

        # -------------------- update global best --------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f} (n={n})")
            if best_score < 1.503:                 # early exit if we hit the target
                break

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 30.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(40.0, time_left * 0.5)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climbing for fine‑tuning.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 10.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = time.time() + time_left
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
