def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')

    def evaluate(a):
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        # compute conv
        conv = _full_convolution(a)
        maxc = conv.max()
        s = a.sum()
        if s < 1e-12:
            return float('inf')
        return 2.0 * a.shape[0] * maxc / (s*s)

    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768] # some lengths
    # allocate time per length proportional to weighting.
    # We'll loop and break if out of time.

    for n in lengths:
        if time.time() > deadline:
            break
        # compute FFT lengths
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # Build initial candidates (list of numpy arrays)
        candidates = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # warm start if best_seq length matches
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # upsample from prior best
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq)
            x_old = np.linspace(0, 1, old.shape[0])
            x_new = np.linspace(0, 1, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum()
            candidates.append(up)

        # random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))

        # smooth random: generate Gaussian kernel smoothed
        raw = np.random.rand(n)
        raw = np.convolve(raw, np.ones(5)/5, mode='same')
        raw = raw / raw.sum()
        candidates.append(raw)

        # edge-weighted families: power law
        idx = np.arange(1, n+1)
        prod = idx * (n - idx + 1)
        for alpha in [-1.5, -1.0, -0.75]:
            shape = np.power(prod, alpha)
            shape = np.maximum(shape, 0)
            shape = shape / shape.sum()
            candidates.append(shape)

        # two Gaussian edges
        i = np.arange(n, dtype=np.float64)
        for sigma in [0.3, 0.7, 1.5]:
            shape = np.exp(-(i**2)/(2*sigma**2)) + np.exp(-((n-1-i)**2)/(2*sigma**2))
            shape /= shape.sum()
            candidates.append(shape)

        # sine bump
        shape = np.sin(np.pi * (np.arange(n)+0.5)/n) ** 2
        shape /= shape.sum()
        candidates.append(shape)

        # optional: random perturbations of best candidate (if exists)
        if best_seq is not None:
            base = np.asarray(best_seq, dtype=np.float64)
            if base.shape[0] == n:
                pert = base + 0.01 * np.random.randn(n)
                pert = np.maximum(pert, 0.0)
                pert = pert / pert.sum()
                candidates.append(pert)

        # Determine iteration count for each candidate based on n
        # Use base_iters = 20000; scale.
        base_iters = 20000
        max_iters = int(base_iters * 1024 / n) + 5000   # ensures decent count

        # Optimize each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # ensure positivity, normalize
            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            logits = np.log(a)

            # Adam parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.02   # lower learning rate for better stability

            lam = 0.2
            lam_factor = 1.005
            last_improve = time.time()

            for it in range(1, max_iters+1):

                mx = logits.max()

                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # exit if no progress for a while
                if time.time() - sub_last > 4.0:
                    break
                if time.time() > deadline:
                    break

    if best_seq is None:

    # Return list
    return best_seq.tolist()

def lp_step(f, M):
    n = f.shape[0]
    # Build CSR matrix A
    rows = []
    cols = []
    data = []
    for j in range(n):
        row_start = j
        row_end = j + n
        # row indices row_start..row_end-1
        # cols 0..n-1, constant coefficient f[j]
        rows.extend(range(row_start, row_end))

Propose best step‑function sequence.

The algorithm works on a cascade of increasing lengths.  For each length it
generates a heterogeneous set of initial vectors (uniform, Dirichlet,
edge‑heavy power‑law, two‑Gaussian edges, sine‑bump, up‑sampled previous
solution, etc.).  Each candidate is optimised by Adam on a smooth‑max
surrogate of the convolution maximum; the surrogate temperature is
annealed very slowly (λ←λ·1.001) and a low learning rate is used.
After the Adam phase a short sub‑gradient descent on the true ℓ∞‑norm
further refines the vector.  For the smallest size (n≤2048) a linear‑
program is solved that maximises the total mass of a new vector g under
the same convolution bound; the current solution is mixed with g and
optimised again.  Finally a cheap random‑pair local search is performed.
The best sequence found over all lengths is returned.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT (power‑of‑two length)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for the gradient of max(a*a).
    Returns a vector of length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective(a: np.ndarray) -> float:
    """Exact objective F(a)=2·n·max(a*a)/(sum(a)²)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a)
    max_c = conv.max()
    return 2.0 * a.shape[0] * max_c / (s * s)

def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a[i]=a[n‑1-i] and renormalise."""
    a_sym = (a + a[::-1]) * 0.5
    s = a_sym.sum()
    if s < 1e-12:
        return a_sym
    return a_sym / s

def _lp_improve(f: np.ndarray, M: float):
    """
    Solve the LP:
        max   sum_j g_j
        s.t.  (f * g)_i ≤ M   for all i
              g_j ≥ 0
    Returns a normalised vector g (sum(g)=1) or None on failure.
    Only intended for moderate n (≤2048) because the constraint matrix
    grows as O(n²).
    """
    n = f.shape[0]
    # Build a sparse matrix A with shape (2n‑1, n) such that A·g = f*g.
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.concatenate([np.full(n, f_j) for f_j in f])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: minimise -1·g  (i.e. maximise sum(g))
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s

    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray) -> float:
    """Exact objective F(a)=2·n·max(a*a)/(sum(a)²)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a)
    max_c = conv.max()
    return 2.0 * a.shape[0] * max_c / (s * s)


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i]=a[n‑1‑i] and renormalise."""
    a_sym = (a + a[::-1]) * 0.5
    s = a_sym.sum()
    if s < 1e-12:
        return a_sym
    return a_sym / s


def _lp_improve(f: np.ndarray, M: float):
    """
    Solve the LP:
        max   sum_j g_j
        s.t.  (f * g)_i ≤ M   for all i
              g_j ≥ 0
    Returns a normalised vector g (sum(g)=1) or None on failure.
    Intended for n ≤ 2048 (the constraint matrix is O(n²)).
    """
    n = f.shape[0]

    # Build sparse convolution matrix A such that A @ g = f * g.
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.concatenate([np.full(n, fj) for fj in f])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=np.float64)

    # minimise -sum(g)  ⇔ maximise sum(g)
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method="highs")
    if not res.success:
        return None
    g = res.x
    sg = g.sum()
    if sg < 1e-12:
        return None
    return g / sg


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function.  The routine respects the wall‑clock budget
    and returns the best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # Length schedule – start small, finish with a fairly large problem.
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a diversified pool of initial candidates.
        candidates = []

        # 1. Uniform vector.
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2. Warm‑start if we already have a best of the same size.
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # 3. Upsample a shorter best (linear interpolation).
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum() if up.sum() > 0 else np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 4. Random Dirichlet draw.
        candidates.append(np.random.dirichlet(np.ones(n)))

        # 5. Smoothed Dirichlet (tiny Gaussian filter).
        raw = np.random.rand(n)
        raw = np.convolve(raw, np.ones(5) / 5, mode="same")
        raw = np.maximum(raw, 0.0)
        raw = raw / raw.sum()
        candidates.append(raw)

        # 6. Edge‑concentrated power‑law families.
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in (-1.5, -1.2, -0.9):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 7. Two‑Gaussian edge shapes.
        i = np.arange(n, dtype=np.float64)
        for sigma in (0.3, 0.6, 1.2):
            shape = np.exp(- (i ** 2) / (2 * sigma ** 2)) + np.exp(- ((n - 1 - i) ** 2) / (2 * sigma ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 8. Smooth sine‑squared bump.
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # 9. Perturbed copy of the current global best (if same size).
        if best_seq is not None and len(best_seq) == n:
            pert = best_seq + 0.01 * np.random.randn(n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0:
                pert = pert / pert.sum()
                candidates.append(pert)

        # --------------------------------------------------------------
        # Optimise every candidate.
        # Hyper‑parameters for the Adam phase.
        base_iters = 20000
        max_iters = int(base_iters * 1024 / n) + 5000
        lr = 0.02
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lam = 0.2
        lam_factor = 1.001

        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            logits = np.log(a)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            last_improve = time.time()

            for it in range(1, max_iters + 1):
                # Soft‑max → simplex.
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # Occasionally enforce mild symmetry – encourages a flatter convolution.
                if it % 500 == 0:
                    a = _symmetrize(a)

                # Convolution and smooth‑max weights.
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c           # denominator = 1 because a is normalised

                # Global best update.
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # Smooth‑max surrogate.
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w = w / w_sum

                # Gradient of surrogate w.r.t. a.
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # Back‑prop through the soft‑max.
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits.
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # Anneal smooth‑max temperature.
                lam = min(lam * lam_factor, 1e7)

                # Stagnation handling – tiny random shake.
                if time.time() - last_improve > 4.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # --------------------------------------------------------------
            # Sub‑gradient refinement on the true ℓ∞‑norm.
            a = a / a.sum()
            eta = 0.5
            sub_last = time.time()
            for sub_it in range(3000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                # Active set of maximal entries.
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # Exponentiated‑gradient update (keeps a ≥ 0).
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                if time.time() - sub_last > 4.0:
                    break
                if time.time() > deadline:
                    break

        # --------------------------------------------------------------
        # Optional LP refinement for modest sizes (n ≤ 2048).
        if n <= 2048 and best_seq is not None and len(best_seq) == n:
            f = best_seq
            M = _full_convolution(f).max()
            g = _lp_improve(f, M)
            if g is not None:
                t = 0.15
                mixed = (1.0 - t) * f + t * g
                mixed = mixed / mixed.sum()

                # Tiny Adam polishing of the mixed vector.
                a = mixed.copy()
                logits = np.log(np.maximum(a, 1e-16))
                m = np.zeros_like(logits)
                v = np.zeros_like(logits)
                lam = 0.1
                for it in range(500):
                    mx = logits.max()
                    exp_log = np.exp(logits - mx)
                    a = exp_log / exp_log.sum()
                    conv = _full_convolution(a, fft_len=conv_fft_len)
                    max_c = conv.max()
                    val = 2.0 * n * max_c
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        print(f"[n={n} LP‑mix Adam it={it}] new best {val:.6f}")
                    w = np.exp(lam * (conv - max_c))
                    w_sum = w.sum()
                    w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)
                    grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                    avg = np.dot(a, grad_a)
                    grad_log = a * (grad_a - avg)
                    m = beta1 * m + (1 - beta1) * grad_log
                    v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                    m_hat = m / (1 - beta1 ** (it + 1))
                    v_hat = v / (1 - beta2 ** (it + 1))
                    logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
                    lam = min(lam * 1.005, 1e7)

        # --------------------------------------------------------------
        # Local random‑pair search around the current best of this size.
        if best_seq is not None and len(best_seq) == n:
            a = best_seq.copy()
            local_iters = 4000
            for it in range(local_iters):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                # Transfer a random fraction of mass from i to j.
                delta = a[i] * np.random.uniform(-0.2, 0.2)
                if a[i] - delta < 0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                val = _objective(a_new)
                if val < best_val:
                    best_val = val
                    best_seq = a_new.copy()
                    a = a_new
                    print(f"[n={n} local it={it}] new best {val:.6f}")

        # End of this length – larger lengths will re‑use the (upsampled)
        # best sequence automatically.

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely).
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Return as plain Python list.
    return best_seq.tolist()
