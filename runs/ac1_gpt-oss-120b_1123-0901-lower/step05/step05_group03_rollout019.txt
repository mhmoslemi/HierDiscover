"""
Propose_candidate – multi‑scale randomized search + Adam + exact max sub‑gradient.

Algorithm
---------
1.  Choose a set of problem sizes (n = 1024, 4096, 8192).  For each size
    allocate a fraction of the total time budget.
2.  Build a modest pool of diverse initial vectors:
        * warm start (previous best, if any)
        * random Dirichlet draws
        * a few parametric families (exponential, power‑law, twin‑Gaussian,
          sinusoidal power)
3.  For every candidate run a two‑phase optimiser:
        * Adam on a smooth‑max surrogate (log‑sum‑exp).  The temperature
          ``lam`` is increased slowly so the surrogate converges to the true
          max.
        * After Adam, switch to an exponentiated‑gradient sub‑gradient method
          that works on the exact max.  This stage keeps the iterates on the
          simplex (non‑negative and normalised).
    Both phases keep track of the globally best sequence/value.
4.  After the last length has been processed, perform a short stochastic
    local‑search (add small Gaussian noise, renormalise) while time permits.
5.  Return the best sequence found (clipped to the evaluator’s allowed range).

The optimiser works directly on the FFT‑based convolution, which makes each
iteration O(n log n) and therefore feasible up to n≈8192 within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)-1 using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)^2)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _adam_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the (possibly) improved best sequence/value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.3                # start temperature
    lam_factor = 1.03        # increase slowly, keeps the surrogate smooth

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ---------- simplex projection (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()            # a ≥ 0, sum = 1

        # ---------- compute convolution and objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() == 1, but keep formula

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:05d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        w = np.exp(lam * (conv - max_c))        # (conv - max_c) ≤ 0
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                 # ∇_log = a·(grad - avg)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e6)

        # occasional random shake if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        # safety bound – prevent endless loops if deadline is far away
        if it > 300_000:
            break

    return best_seq, best_val


def _subgradient_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                      # enforce the simplex

    eta = 0.6                         # step size (can be tuned)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < 30_000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it:05d}] new best {val:.6f}")

        # active set of maximal entries (allow tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:                     # should never happen, but be safe
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:                     # collapse guard
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # simple stagnation check
        if time.time() - last_improve > 5.0:
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the two‑phase optimiser (Adam → exact sub‑gradient) while
    respecting the global deadline.  Returns the globally best sequence/value.
    """
    # Adam phase
    best_seq, best_val = _adam_phase(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    # Sub‑gradient phase on the true max
    best_seq, best_val = _subgradient_phase(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a modest, diverse pool of start vectors for a given length ``n``.
    Families include random Dirichlet, exponential, edge‑concentrated
    power‑laws, twin Gaussians, and sinusoidal powers.
    """
    cand = []

    # warm start (and a few rolled versions) if provided
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # exponential decay families
    for r in np.linspace(0.80, 0.99, 4):
        shape = np.power(r, np.arange(n, dtype=float))
        shape /= shape.sum()
        cand.append(shape)

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=float)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.6, 4):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # twin‑Gaussian edge shapes
    i = np.arange(n, dtype=float)
    for sigma_frac in (0.06, 0.12, 0.24):
        sigma = sigma_frac * n
        shape = np.exp(-(i ** 2) / (2 * sigma ** 2))
        shape += np.exp(-((n - 1 - i) ** 2) / (2 * sigma ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal power‑law shapes
    base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    base = np.maximum(base, 0.0)
    for gamma in (0.5, 1.0, 2.0, 3.0):
        shape = base ** gamma
        shape /= shape.sum()
        cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (sequence) with a low evaluation
    score.  The routine respects a 1000 s wall‑time budget and returns the
    best sequence it discovered.
    """
    np.random.seed(seed)
    # small safety margin to avoid overshooting the evaluator's timeout
    deadline = time.time() + budget_s - 0.2

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage configuration – length, time‑fraction and number of restarts
    stages = [
        (1024, 0.13, 6),   # quick exploration on a smaller problem
        (4096, 0.42, 6),   # main optimisation stage (the one that gave 1.509)
        (8192, 0.42, 4),   # larger length – we have more time now
    ]

    # fetch previous construction if it exists
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)

    for n, fraction, n_restarts in stages:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # allocate a piece of the remaining budget for this stage
        remaining = deadline - time.time()
        stage_deadline = time.time() + max(remaining * fraction, 0.5)

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # construct the pool of initial vectors for this length
        warm_n = None
        if warm_start is not None:
            # resize/trim the warm start to the current length
            if warm_start.shape[0] > n:
                warm_n = warm_start[:n]
            else:
                warm_n = np.pad(warm_start, (0, n - warm_start.shape[0]))
            warm_n = np.maximum(warm_n, 0.0)
            if warm_n.sum() >= 0.01:
                warm_n = warm_n / warm_n.sum()
            else:
                warm_n = None

        candidates = _initial_candidates(n, warm_n)

        # --------------------------------------------------------------
        # Optimise each candidate; the global best is updated across stages
        for cid, cand in enumerate(candidates):
            if time.time() >= stage_deadline:
                break

            a0 = np.maximum(cand, 0.0)
            s = a0.sum()
            if s < 0.01:
                a0 = np.random.dirichlet(np.ones(n))
                s = a0.sum()
            a0 = a0 / s

            # run the deep optimiser within the stage deadline
            best_seq, best_val = _deep_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                stage_deadline,
                best_seq,
                best_val,
            )

            # occasional progress report
            print(
                f"[Stage n={n}] candidate {cid+1}/{len(candidates)} "
                f"best_val={best_val:.6f}"
            )

        # after a stage we keep the current best as warm start for the next size
        warm_start = best_seq

    # ------------------------------------------------------------------
    # Final short stochastic local‑search (perturbation) while time permits
    if best_seq is not None:
        n_final = best_seq.shape[0]
        conv_fft_len = _next_pow2(2 * n_final - 1)
        corr_fft_len = _next_pow2(3 * n_final - 2)

        while time.time() < deadline - 0.05:
            # small Gaussian perturbation (relative magnitude 0.5 %)
            noise = np.random.randn(n_final) * 0.005
            cand = best_seq * (1.0 + noise)
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum < 1e-12:
                continue
            cand /= cand_sum

            val = _objective(cand, n_final, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand
                print(f"[Local search] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded produce a uniform vector.
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final report
    print(f"=== Final best: value={best_val:.6f}, length={len(best_seq)} ===")
    return best_seq.tolist()
