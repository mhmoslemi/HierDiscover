from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_g(f, M):
    n = len(f)
    # Build constraint matrix A such that A @ g <= M
    # A_ub shape (2n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i runs from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # j = k - i.
        # So for i in range(i_start, i_end+1), set coefficient = f[i]
        # column = j = k-i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

def simplex_projection(v, z=1):
    # v: array, need to project onto {x >=0, sum x = z}
    # returns projected vector.
    # algorithm: sort descending, compute cumulative sum.
    u = np.sort(v)[::-1]

    cond = u - cssv / ind > 0

    return w

def simplex_projection(v, z):
    """Project vector v onto simplex {x >=0, sum x = z}."""
    # Note: if v is already nonnegative and sum equals z, return v.
    if np.sum(v) == z and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1] + 1 # index+1
    theta = cssv[rho - 1] / rho if rho > 0 else 0
    w = np.maximum(v - theta, 0)
    return w

def full_convolution_fft(x, y):
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n-1).bit_length())
    fx = rfft(x, n=size)
    fy = rfft(y, n=size)
    conv = irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0, out=conv)
    return conv

def subgradient_descent(a, target_sum, max_iters, deadline, step_factor=1.0):
    n = len(a)
    best = a.copy()
    best_val = max_conv(a)
    # compute step base
    base_step = step_factor * target_sum / n   # approx sqrt(2/n)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = full_convolution_fft(a, a)
        k = np.argmax(conv)
        # gradient
        rev_idx = k - np.arange(n)
        grad = np.zeros_like(a)
        mask = (rev_idx >= 0) & (rev_idx < n)
        grad[mask] = 2.0 * a[rev_idx[mask]]
        # step size schedule
        step = base_step / (1.0 + it * 0.01)
        a_new = a - step * grad
        a_new = simplex_projection(a_new, target_sum)
        val = max_conv(a_new)
        if val < best_val - 1e-8:
            best = a_new.copy()
            best_val = val
            a = a_new
        else:
            # if not improving maybe reduce step size
            step *= 0.5
    return best, best_val

def lp_refine(f, target_sum, deadline, max_iters=10, t=0.5):
    n = len(f)
    best_f = f.copy()
    best_M = max_conv(f)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = best_M
        # Build constraint matrix
        # Use sparse diags
        if SCIPY_AVAILABLE:
            data = []
            offsets = []
            for d in range(n):
                # diagonal with f[d] repeated length n
                diag = np.full(n, f[d])
                data.append(diag)  # list of arrays
                offsets.append(d)
            A = diags(data, offsets, shape=(2*n-1, n), format='csr')
            b = np.full(2*n-1, M)
            c = -np.ones(n)  # maximize sum -> minimize -sum
            bounds = [(0, None)] * n
            try:
                res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
                if not res.success:
                    break
                g = res.x
            except Exception:
                break
        else:
            break
        sum_g = g.sum()
        if sum_g <= target_sum * (1 + 1e-9):
            # cannot improve sum
            break
        # scale g to target_sum
        g_scaled = g * (target_sum / sum_g)
        f_new = (1 - t) * f + t * g_scaled
        M_new = max_conv(f_new)
        if M_new < best_M - 1e-8:
            f = f_new
            best_f = f_new.copy()
            best_M = M_new
            # continue loop
        else:
            # maybe try smaller t?
            # Set t = t * 0.5 and continue
            t *= 0.5
            if t < 1e-3:
                break
    return best_f, best_M

1. **Random start & sub‑gradient descent** –  the sequence lives on the
   simplex (non‑negative, prescribed ℓ¹‑norm).  In each iteration we
   evaluate the full self‑convolution, locate the index of the maximal
   coefficient and take a sub‑gradient step on that *true* max (no smooth
   surrogate).  The step is followed by Euclidean projection onto the
   simplex.

2. **Linear‑programming refinement (optional)** –  given a current
   sequence `f` with peak `M = max(f∗f)`, we solve a LP that maximises
   `∑g_i` under the linear constraints `f∗g ≤ M`.  The solution `g`
   usually has a larger ℓ¹‑norm; after scaling it to the target sum we
   mix it with `f` (`f←(1‑t)f+t g`) which never raises the peak.
   Repeating this step quickly drives the peak down.

3. **Final Adam optimisation** –  the best candidate obtained so far
   is refined with the Adam optimiser on the soft‑max parametrisation
   (`a = S·softmax(x)`).  A log‑sum‑exp surrogate of the max is used
   with an aggressively annealed temperature, so the optimiser ends
   up almost exactly minimising the true objective.

Multiple lengths `n` (up to ≈ 8000) are tried; the best sequence over
all runs is returned.  If a previous construction `height_sequence_1`
is available it is used as a warm‑start.  All computation respects the
overall time budget (default 1000 s).  The returned list can be fed
directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Optional SciPy imports (LP refinement)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix, diags
    SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Full linear convolution via FFT (real vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = rfft(x, n=size)
    fy = rfft(y, n=size)
    conv = irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return _full_convolution_fft(a, a).max()


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = z}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Project vector v onto the simplex with sum = z."""
    if np.all(v >= 0) and abs(v.sum() - z) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1] + 1
        theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *true* max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
    step_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on  max(conv(a, a)),
    keeping a on the simplex of sum `target_sum`.
    Returns the best sequence encountered and its peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _max_convolution(a)

    # a typical magnitude of a_i is target_sum / n = sqrt(2/n)
    base_step = step_factor * target_sum / n

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the peak w.r.t. a (sub‑gradient)
        rev_idx = k - np.arange(n)
        mask = (rev_idx >= 0) & (rev_idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[rev_idx[mask]]

        # simple diminishing step size
        step = base_step / (1.0 + 0.01 * it)
        a_new = a - step * grad
        a_new = _simplex_projection(a_new, target_sum)

        val = _max_convolution(a_new)
        if val < best_val - 1e-9:
            best = a_new.copy()
            best_val = val
            a = a_new
        else:
            # no improvement – keep current a but keep searching
            a = a_new

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (only for moderate n, requires SciPy)
# ----------------------------------------------------------------------
def _lp_refine(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 12,
    t: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Given a sequence f, repeatedly solve a LP that maximises ∑g_i under
    f∗g ≤ M (M = max(f∗f)).  The resulting g is scaled to the same
    ℓ¹‑norm as f and mixed with f.  The peak never increases and
    typically decreases.
    """
    if not SCIPY_AVAILABLE or f.size > 3000:   # avoid huge LPs
        return f, _max_convolution(f)

    n = f.shape[0]
    best_f = f.copy()
    best_M = _max_convolution(f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = best_M

        # build sparse Toeplitz matrix A where (A @ g)[k] = (f∗g)[k]
        data = []
        offsets = []
        for d in range(n):
            data.append(np.full(n, f[d]))
            offsets.append(d)
        A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
        b = np.full(2 * n - 1, M)

        # LP: maximise sum(g)  == minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        except Exception:   # pragma: no cover
            break
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= target_sum * (1.0 + 1e-9):
            # no improvement possible
            break

        # scale g back to the target ℓ¹‑norm
        g_scaled = g * (target_sum / sum_g)

        # convex combination
        f_new = (1.0 - t) * f + t * g_scaled
        M_new = _max_convolution(f_new)

        if M_new < best_M - 1e-9:
            f = f_new
            best_f = f_new.copy()
            best_M = M_new
            # keep the same mixing weight for the next iteration
        else:
            # try a smaller mixing coefficient
            t *= 0.5
            if t < 1e-3:
                break

    return best_f, best_M


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation (as in the reference code)
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ---- map logits to the simplex ---------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stabilise
        p = exp_shift / exp_shift.sum()               # probabilities, sum = 1
        a = target_sum * p

        # ---- evaluate current smoothed max -----------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  local best = {max_conv:.8f}")

        # ---- smooth‑max weights (log‑sum‑exp) -------------------------
        tau = max(
            init_tau_factor * max_conv * (tau_decay ** iteration),
            min_tau_factor * max_conv,
        )
        scaled = (conv - max_conv) / tau               # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a (derivative of Σ w_k·conv_k) ----------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # slice to pick needed entries
        grad_a = 2.0 * conv2[idx]

        # ---- gradient w.r.t. logits x (chain rule through soft‑max) ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping – prevents occasional huge steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # ---- Adam update -----------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to help escape flat plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.999

        # early stop if we have not improved for a long stretch
        if iteration - last_improve > 2000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the time budget and returns the best
    sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1             # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optional warm‑start from a previously known construction.
    # --------------------------------------------------------------
    warm = None
    try:
        warm = np.array(globals()["height_sequence_1"], dtype=float)
        warm = np.clip(warm, 0.0, 1000.0)
    except Exception:
        warm = None

    # Candidate lengths (larger lengths give more freedom, but are costlier)
    candidate_ns = [1800, 2500, 3400, 4400, 5600, 7200, 8000]
    candidate_ns = list(rng.permutation(candidate_ns))

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # give each length a modest time slice
        slice_time = min(120.0, remaining * 0.7)
        slice_deadline = time.time() + slice_time

        target_sum = math.sqrt(2.0 * n)

        # ----------------------------------------------------------
        # Initialise a sequence (warm‑start or random shape)
        # ----------------------------------------------------------
        if warm is not None and warm.size >= n:
            a0 = warm[:n].copy()
        else:
            typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
            idx = np.arange(1, n + 1, dtype=float)

            if typ == 'power':
                p = rng.uniform(0.1, 2.0)
                shape = idx ** (-p)
            elif typ == 'exponential':
                tau = rng.uniform(0.2, 0.8) * n
                shape = np.exp(-idx / tau)
            elif typ == 'mixed':
                split = rng.integers(1, n - 1)
                left = idx[:split] ** (-rng.uniform(0.2, 1.0))
                right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
                shape = np.concatenate([left, right])
            else:   # uniform
                shape = np.ones_like(idx)

            # tiny jitter to avoid perfect symmetry
            shape += rng.random(n) * 1e-6
            a0 = shape

        a0 = np.maximum(a0, 1e-12)
        a0 = a0 / a0.sum() * target_sum

        # ----------------------------------------------------------
        # Phase 1 – sub‑gradient descent on the true max‑conv
        # ----------------------------------------------------------
        sub_deadline = min(slice_deadline, time.time() + slice_time * 0.35)
        a_sub, val_sub = _subgradient_descent(
            a0,
            target_sum,
            max_iters=5000,
            deadline=sub_deadline,
            step_factor=0.9,
        )

        # ----------------------------------------------------------
        # Phase 2 – LP refinement (only for moderate n)
        # ----------------------------------------------------------
        if SCIPY_AVAILABLE and n <= 3000:
            lp_deadline = min(slice_deadline, time.time() + slice_time * 0.30)
            a_lp, val_lp = _lp_refine(
                a_sub,
                target_sum,
                lp_deadline,
                max_iters=12,
                t=0.5,
            )
        else:
            a_lp, val_lp = a_sub, val_sub

        # ----------------------------------------------------------
        # Phase 3 – final Adam refinement (smooth surrogate)
        # ----------------------------------------------------------
        if time.time() < slice_deadline:
            adam_deadline = slice_deadline
            x0 = np.log(a_lp + 1e-12)
            a_final, final_score = _optimise_adam_softmax(
                n,
                x0,
                target_sum,
                adam_deadline,
                rng,
                init_tau_factor=0.2,
                min_tau_factor=1e-4,
                alpha=0.05,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                tau_decay=0.95,
                max_iter=None,
            )
        else:
            a_final = a_lp
            final_score = evaluate_sequence(a_final.tolist())

        # ----------------------------------------------------------
        # Keep the best candidate seen so far
        # ----------------------------------------------------------
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            warm = best_seq.copy()                 # warm‑start future runs
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            # early stop if we are already below the target
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
