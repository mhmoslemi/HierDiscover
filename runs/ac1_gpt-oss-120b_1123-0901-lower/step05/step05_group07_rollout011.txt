def conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

import cvxpy as cp
import scipy.linalg

def lp_refine(a, M, max_iters, deadline):
    n = a.shape[0]
    # Build convolution matrix
    col = np.concatenate([a, np.zeros(n-1)])
    row = np.concatenate([a[:1], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        cur_val = 2.0 * n * conv.max()

            best_local_val = cur_val

        c_max = conv.max()

def lp_refine(a, M, deadline, max_lp_iters=10):
    n = a.shape[0]
    best_a = a.copy()
    best_M = M
    t = 0.2
    while time.time() < deadline and max_lp_iters>0:
        # build matrix A
        col = np.concatenate([a, np.zeros(n-1)])
        row = np.concatenate([a[:1], np.zeros(n-1)])
        A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
        # solve LP
        c = -np.ones(n)
        ub = np.full(2*n-1, best_M)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs', options={'time_limit': min(30.0, deadline - time.time())})
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b  # normalized
        # convex combination
        a_new = (1 - t) * a + t * g
        conv_new = full_convolution_fft(a_new)
        M_new = conv_new.max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new
            # adapt step size: possibly increase or keep.
            # optionally reduce t for fine tuning later
            if t > 0.02:
                t *= 0.9
            # update a for next iteration
            a = a_new
        else:
            # if no improvement, shrink t or break
            t *= 0.7
            if t < 1e-3:
                break
        max_lp_iters -= 1
    return best_a, best_M

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (vector) minimizing 
    F(a)=2·n·max(a*a)/(sum a)^2. The method first runs a
    temperature‑annealed Adam optimiser on a soft‑max
    parametrisation, then refines the result with a linear‑programming
    step that maximises the total mass while keeping the self‑convolution
    peak unchanged. Finally a cheap greedy local shuffling is applied.
    The whole search respects the given time budget.
    """
    import numpy as np
    import time
    from scipy.fft import rfft, irfft
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small margin

    # helper functions
    def conv_fft(a):
        n = a.shape[0]
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fa = rfft(a, size)
        conv = irfft(fa * fa, size)[:2 * n - 1]
        return conv

    def corr_fft(w, a):
        n = a.shape[0]
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fw = rfft(w, size)
        fr = rfft(a[::-1], size)
        corr = irfft(fw * fr, size)
        start = n - 1
        return corr[start:start + n]

    # choose problem size (start with 1024, maybe go to 2048 later)
    lengths = [1024]
    best_seq = None
    best_val = float('inf')

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # initialise a
        if 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            elif cand.shape[0] < n:
                cand = np.pad(cand, (0, n - cand.shape[0]), constant_values=0.0)
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-12:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            a = cand.copy()
            print("[init] Using height_sequence_1 (truncated/padded)")
        else:
            a = np.random.dirichlet(np.ones(n))
            print("[init] Random Dirichlet start")
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # --------------------------------------------------------------
        # Phase 1 – smooth‑max Adam optimisation
        # --------------------------------------------------------------
        # allocate about half of remaining time
        remaining = deadline - time.time()
        phase1_time = min(remaining * 0.55, 600.0)   # at most 600 s
        phase1_deadline = time.time() + phase1_time
        # Adam hyper‑parameters
        x = np.log(a + 1e-15)          # logits
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.12
        decay_every = 5000
        max_iters = 120000
        tau_start = 0.12
        tau_end = 1e-6

        best_local_seq = a.copy()
        best_local_val = 2.0 * n * conv_fft(a).max()

        t_iter = 0
        last_improve = time.time()
        while time.time() < phase1_deadline and t_iter < max_iters:
            # soft‑max to probability vector
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()

            conv = conv_fft(a)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_local_val - 1e-12:
                best_local_val = cur_val
                best_local_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam iter={t_iter:06d}] new best {cur_val:.6f}")

            # smooth max and its gradient weights
            c_max = conv.max()
            tau = max(tau_end, tau_start * (1.0 - t_iter / max_iters))
            shifted = (conv - c_max) / tau
            exp_shifted = np.exp(shifted)
            sum_exp = exp_shifted.sum()
            weights = exp_shifted / sum_exp   # d smooth_max / d conv

            # gradient w.r.t a
            grad_a = 2.0 * corr_fft(weights, a)

            # back‑prop through soft‑max
            inner = np.dot(a, grad_a)
            grad_x = a * (grad_a - inner)

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            if t_iter % decay_every == 0:
                lr *= 0.7

            # stagnation shake
            if time.time() - last_improve > 30.0:
                x += (np.random.rand(n) - 0.5) * 6e-3
                last_improve = time.time()
                print(f"[Adam iter={t_iter}] shake due to stagnation")

        # store best from phase 1
        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_seq.copy()
            print(f"[phase1] best value so far {best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Linear‑programming refinement
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # use the best sequence found for this length as starting point
        a = best_seq.copy()
        conv = conv_fft(a)
        M = conv.max()
        # parameters for the refinement loop
        t_step = 0.3                     # initial mixing weight
        max_lp_iters = 15               # how many LP solves we attempt
        lp_iter = 0
        while time.time() < deadline and lp_iter < max_lp_iters:
            # Build convolution matrix A where (A @ b)[k] = Σ_i a_i * b_{k-i}
            col = np.concatenate([a, np.zeros(n - 1)])
            row = np.concatenate([a[:1], np.zeros(n - 1)])
            A = toeplitz(col, row)        # dense (2n‑1)×n matrix

            # Solve max Σ b_j  s.t. A @ b ≤ M  ,  b ≥ 0
            c = -np.ones(n)               # minimise -sum(b)
            ub_vec = np.full(2 * n - 1, M)
            # time limit for each LP = a few seconds
            lp_time = min(5.0, deadline - time.time())
            res = linprog(c, A_ub=A, b_ub=ub_vec, bounds=(0, None),
                          method='highs', options={'time_limit': lp_time})
            if not res.success:
                print("[LP] solver failed or ran out of time")
                break

            b = res.x
            sum_b = b.sum()
            if sum_b <= 0:
                print("[LP] zero sum solution")
                break

            # Normalise b to a probability vector (same total mass as a)
            g = b / sum_b

            # convex combination
            a_new = (1.0 - t_step) * a + t_step * g
            conv_new = conv_fft(a_new)
            M_new = conv_new.max()
            val_new = 2.0 * n * M_new

            if val_new < best_val - 1e-12:
                best_val = val_new
                best_seq = a_new.copy()
                print(f"[LP iter={lp_iter}] improved to {val_new:.6f}")

            # Accept the update if it reduces the max‑convolution,
            # otherwise shrink the mixing weight.
            if M_new < M - 1e-14:
                a = a_new
                M = M_new
                # slightly decrease mixing weight for finer steps
                t_step = max(t_step * 0.85, 0.02)
            else:
                t_step *= 0.6
                if t_step < 1e-3:
                    break

            lp_iter += 1

        # --------------------------------------------------------------
        # Phase 3 – cheap random mass shuffling (local refinement)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            a_local = best_seq.copy()
            local_iters = 4000
            for it in range(local_iters):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                # random fraction of mass from i to j
                move = min(a_local[i], 0.01) * (np.random.rand() * 0.7 + 0.2)
                if move <= 0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= move
                a_new[j] += move
                conv_new = conv_fft(a_new)
                val_new = 2.0 * n * conv_new.max()
                if val_new < best_val - 1e-12:
                    best_val = val_new
                    best_seq = a_new.copy()
                    a_local = a_new
                    print(f"[local] improved to {val_new:.6f}")

    # safety fallback
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)  # =2

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain python list as required
    return best_seq.tolist()
```

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)^2)

    The method consists of three phases:

    1. **Smooth‑max Adam optimisation** – a temperature‑annealed Adam optimiser
       works on a soft‑max parametrisation of the vector.  The smooth maximum
       (log‑sum‑exp) provides a differentiable surrogate for the true max.

    2. **Linear‑programming refinement** – given the current vector `a` we solve

           maximise   sum(b)
           subject to a * b ≤ M   (component‑wise),   b ≥ 0

       where `M = max(a*a)` and `*` denotes linear convolution.
       The LP yields a vector `b` that carries as much mass as possible while
       keeping the convolution peak unchanged.  A small convex combination
       `a ← (1‑t)·a + t·b̂` (with `b̂` normalised to unit sum) strictly reduces the
       peak, and the step size `t` is gradually decreased.

    3. **Cheap greedy local shuffling** – tiny amounts of probability mass are
       moved between random coordinates; a move is kept only if it lowers the
       objective.

    All phases respect the global time budget (default 1000 s).  The best
    sequence found (as a plain Python list) is returned.
    """
    import time
    import numpy as np
    from scipy.fft import rfft, irfft
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ----------------------------------------------------------------------
    # Helper utilities (FFT‑based linear convolution and cross‑correlation)
    # ----------------------------------------------------------------------
    def conv_fft(a: np.ndarray) -> np.ndarray:
        """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
        n = a.shape[0]
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fa = rfft(a, size)
        conv = irfft(fa * fa, size)[: 2 * n - 1]
        return conv

    def corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
        """
        Linear cross‑correlation (w ★ a) of length len(a).
        Returns (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]  for i=0…n‑1.
        """
        n = a.shape[0]
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fw = rfft(w, size)
        fr = rfft(a[::-1], size)
        corr = irfft(fw * fr, size)
        start = n - 1
        return corr[start : start + n]

    # ----------------------------------------------------------------------
    # Problem size – we work with n = 1024 (the length that gave the best
    # result so far).  The code can be extended to other lengths if desired.
    # ----------------------------------------------------------------------
    lengths = [1024]

    best_seq = None
    best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # Initial vector (warm‑start from a previously found construction if
        # available, otherwise random Dirichlet).
        # ------------------------------------------------------------------
        if "height_sequence_1" in globals():
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            elif cand.shape[0] < n:
                cand = np.pad(cand, (0, n - cand.shape[0]), constant_values=0.0)
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-12:
                cand = np.random.dirichlet(np.ones(n))
                print("[init] height_sequence_1 empty → random Dirichlet")
            else:
                cand = cand / cand.sum()
                print("[init] warm‑start from height_sequence_1")
            a = cand.copy()
        else:
            a = np.random.dirichlet(np.ones(n))
            print("[init] random Dirichlet start")
        a = np.maximum(a, 0.0)
        a = a / a.sum()                     # normalise to a probability vector

        # ------------------------------------------------------------------
        # Phase 1 – smooth‑max Adam optimisation
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        phase1_time = min(remaining * 0.55, 600.0)     # keep a generous margin
        phase1_deadline = time.time() + phase1_time

        # Adam state (logits parametrisation)
        x = np.log(a + 1e-15)                     # logits for the soft‑max
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.12
        decay_every = 5000
        max_iters = 120_000
        tau_start = 0.12
        tau_end = 1e-6

        best_local_seq = a.copy()
        best_local_val = 2.0 * n * conv_fft(a).max()
        t_iter = 0
        last_improve = time.time()

        while time.time() < phase1_deadline and t_iter < max_iters:
            # ---- soft‑max -> probability vector ----
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()

            # ---- objective and best‑track ----
            conv = conv_fft(a)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_local_val - 1e-12:
                best_local_val = cur_val
                best_local_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam iter={t_iter:06d}] new best {cur_val:.6f}")

            # ---- smooth‑max and gradient w.r.t. a ----
            c_max = conv.max()
            tau = max(tau_end, tau_start * (1.0 - t_iter / max_iters))
            shifted = (conv - c_max) / tau
            exp_shifted = np.exp(shifted)               # stable because we subtracted c_max
            sum_exp = exp_shifted.sum()
            weights = exp_shifted / sum_exp              # ∂smooth_max/∂conv

            grad_a = 2.0 * corr_fft(weights, a)         # ∂smooth_max/∂a

            # ---- back‑prop through the soft‑max ----
            inner = np.dot(a, grad_a)
            grad_x = a * (grad_a - inner)

            # ---- Adam update ----
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            if t_iter % decay_every == 0:
                lr *= 0.7

            # ---- stagnation shake ----
            if time.time() - last_improve > 30.0:
                x += (np.random.rand(n) - 0.5) * 6e-3
                last_improve = time.time()
                print(f"[Adam iter={t_iter}] shake (stagnation)")

        # store the best vector of this phase
        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_seq.copy()
            print(f"[phase 1] best value so far {best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Linear‑programming refinement
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break

        a = best_seq.copy()
        conv = conv_fft(a)
        M = conv.max()                     # current peak of a*a
        t_step = 0.3                       # initial mixing weight
        max_lp_iters = 15                  # how many LP solves we attempt
        lp_iter = 0

        while time.time() < deadline and lp_iter < max_lp_iters:
            # Build the (2n‑1)×n convolution matrix A such that A @ b = a * b.
            col = np.concatenate([a, np.zeros(n - 1)])
            row = np.concatenate([a[:1], np.zeros(n - 1)])
            A = toeplitz(col, row)        # dense but modest size (≈16 MiB for n=1024)

            # Linear programme: maximise Σ b_j  ⇔ minimise -Σ b_j
            c = -np.ones(n)               # objective vector for linprog
            ub = np.full(2 * n - 1, M)    # right‑hand side (all entries = current peak)

            # Give the LP a few seconds; HiGHS respects the time_limit option.
            lp_time = min(5.0, deadline - time.time())
            res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None),
                          method='highs', options={'time_limit': lp_time})

            if not res.success:
                print("[LP] solver failed or timed out")
                break

            b = res.x
            sum_b = b.sum()
            if sum_b <= 0:
                print("[LP] zero‑mass solution")
                break

            # Normalise b to have unit sum (same normalisation as a)
            g = b / sum_b

            # Convex combination
            a_new = (1.0 - t_step) * a + t_step * g
            conv_new = conv_fft(a_new)
            M_new = conv_new.max()
            val_new = 2.0 * n * M_new

            if val_new < best_val - 1e-12:
                best_val = val_new
                best_seq = a_new.copy()
                print(f"[LP iter={lp_iter}] improved to {val_new:.6f}")

            # Accept the update only if the peak decreased; otherwise shrink step.
            if M_new < M - 1e-14:
                a = a_new
                M = M_new
                t_step = max(t_step * 0.85, 0.02)   # finer steps as we converge
            else:
                t_step *= 0.6
                if t_step < 1e-3:
                    break

            lp_iter += 1

        # ------------------------------------------------------------------
        # Phase 3 – cheap random local refinement (mass shuffling)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            a_local = best_seq.copy()
            local_iters = 4000
            for it in range(local_iters):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                # Move a modest fraction of probability mass from i to j
                move = min(a_local[i], 0.01) * (np.random.rand() * 0.7 + 0.2)
                if move <= 0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= move
                a_new[j] += move
                conv_new = conv_fft(a_new)
                val_new = 2.0 * n * conv_new.max()
                if val_new < best_val - 1e-12:
                    best_val = val_new
                    best_seq = a_new.copy()
                    a_local = a_new
                    print(f"[local] improved to {val_new:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)   # = 2

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
