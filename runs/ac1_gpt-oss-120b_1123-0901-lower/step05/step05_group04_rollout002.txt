def solve_lp_for_g(f, M):
    n = len(f)
    # Build convolution matrix A using f
    # Use diags
    data = np.vstack([np.full(n, f_i) for f_i in f])   # shape (n,n)
    offsets = np.arange(n)
    A = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
    b = cvxpy.Variable(n)
    constraints = [b >= 0, A @ b <= M]
    obj = cvxpy.Maximize(cvxpy.sum(b))
    prob = cvxpy.Problem(obj, constraints)
    prob.solve(solver='ECOS', warm_start=True) # set solver
    return b.value

def _solve_lp_g(f, M):

    import scipy.sparse as sp
    n = len(f)
    # Build convolution matrix A (2n-1) x n

    b = cp.Variable(n)

    # Use warm start if needed? Not needed

        return None
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)  # ensure nonneg.

"""
Propose a step‑function with minimal evaluation score.

Algorithm
---------
1. Warm‑start from a global candidate (if provided).
2. Perform many random restarts:
   * pick a length n (1500–2500) and a random non‑negative shape.
   * normalise the shape so that sum = sqrt(2 n) (the denominator of the
     evaluation function becomes constant).
   * run a few Frank–Wolfe / conditional‑gradient iterations:
        – given the current sequence f, compute M = max(conv(f,f)).
        – solve a linear programme:
              maximise   Σ b_i
              subject to b_i ≥ 0
                         conv(f, b) ≤ M  (pointwise)
          The LP returns a “direction’’ g (after rescaling to the same sum).
        – mix f and g with a small step size (line‑search over a few τ).
        – repeat while the max‑convolution strictly decreases.
   * keep the best sequence seen.
3. If time permits, polish the best sequence with a short Adam optimiser
   (the same optimiser used in the previous baseline).
4. Return the best sequence as a plain Python list.

The Frank–Wolfe phase lowers the bound quickly with only a handful of LP
solves (each LP has n variables and 2 n‑1 linear constraints).  Random
restarts explore different lengths / shapes, and the final Adam refinement
fine‑tunes the result.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Random initial shapes (positive, normalised later)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a positive profile of length n (power, exponential, mixed, uniform)."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:   # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Helper: compute the max of a full convolution (fast enough for n≈2500)
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a, a)) using NumPy's direct convolution."""
    conv = np.convolve(a, a)          # full linear convolution
    return float(conv.max())


# ----------------------------------------------------------------------
# Solve the LP: maximise Σ b_i   s.t.   b ≥ 0  and  conv(f, b) ≤ M
# ----------------------------------------------------------------------
def _solve_lp_for_direction(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a non‑negative vector f and an upper bound M,
    solve   max Σ b_i   s.t.   b ≥ 0  and conv(f, b) ≤ M .
    Returns the optimal b (numpy array) or None on failure.
    """
    n = f.shape[0]

    # Build the (2n‑1) × n convolution matrix A where (A b)_k = (f * b)_k.
    # Each diagonal offset i (0 ≤ i < n) contains the constant value f[i].
    diag_data = [np.full(n, val, dtype=float) for val in f]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')

    b = cp.Variable(n)

    constraints = [b >= 0,
                   A @ b <= M]          # pointwise bound on the convolution

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver='ECOS', warm_start=True, max_iters=2000,
                   abstol=1e-8, reltol=1e-8, feastol=1e-8)
    except Exception as exc:
        print(f"[LP] solve failed: {exc}")
        return None

    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)     # force numerical non‑negativity


# ----------------------------------------------------------------------
# Frank–Wolfe style improvement for a given initial sequence
# ----------------------------------------------------------------------
def _frank_wolfe_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 12,
) -> tuple[np.ndarray, float]:
    """
    Perform a few Frank–Wolfe iterations starting from a0.
    Returns the best sequence found and its official evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = _max_convolution(a)

    # keep a copy of the best *evaluation* (not just max conv) for early exit
    best_score = float('inf')

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # 1. current bound
        M = _max_convolution(a)

        # 2. direction from LP
        b = _solve_lp_for_direction(a, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b < 1e-12:
            break

        # Rescale to preserve the canonical sum
        g = target_sum * b / sum_b

        # 3. line‑search over a few mixing coefficients
        best_local_M = None
        best_local_a = None
        best_tau = None
        for tau in (0.05, 0.15, 0.30, 0.45, 0.60, 0.80):
            a_candidate = (1.0 - tau) * a + tau * g
            M_candidate = _max_convolution(a_candidate)

            if (best_local_M is None) or (M_candidate < best_local_M):
                best_local_M = M_candidate
                best_local_a = a_candidate
                best_tau = tau

        # 4. accept if we improved the max‑convolution
        if best_local_M is not None and best_local_M < M - 1e-12:
            a = best_local_a
            if best_local_M < best_M:
                best_M = best_local_M
                best_a = a.copy()
        else:
            # No improvement – stop early.
            break

    # final official evaluation
    score = evaluate_sequence(best_a.tolist())
    return best_a, score


# ----------------------------------------------------------------------
# Optional short Adam refinement (same idea as the previous baseline)
# ----------------------------------------------------------------------
def _adam_refine(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    A lightweight Adam optimiser acting on the soft‑max parametrisation.
    Used only as a final polishing step.
    """
    n = seq.shape[0]
    # initialise logits
    x = np.log(np.maximum(seq, 1e-12))

    # Adam hyper‑parameters
    alpha = 0.07
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = seq.copy()
    best_val = evaluate_sequence(best_seq.tolist())

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    last_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # soft‑max mapping
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # current max‑convolution (smooth surrogate not needed for acceptance)
        conv = np.convolve(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = it

        # smooth‑max weights
        tau = max(0.25 * max_conv * (0.96 ** it), 5e-4 * max_conv)
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        w_rev = w[::-1]
        conv2 = np.convolve(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # gradient w.r.t. logits (soft‑max chain rule)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter
        if it % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        alpha *= 0.9997

        if it - last_improve > 2000:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  Random restarts + Frank–Wolfe LP steps + a short
    Adam polishing phase are employed within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.05   # tiny safety margin

    best_seq = None
    best_score = float("inf")
    top_elite = []    # keep a few elite candidates for possible warm‑starts

    # ------------------------------------------------------------------
    # Warm‑start from an existing global candidate if available.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            top_elite.append((sc, hs_arr.copy()))
            print(f"[{time.time():.2f}] Warm‑start score {sc:.8f} (n={len(hs_arr)})")
    except Exception:
        pass  # no warm‑start provided

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 8.0:
            break

        # pick a problem size – larger early, smaller later
        if remaining > 300:
            n = int(rng.integers(1500, 2600 + 1))
        elif remaining > 120:
            n = int(rng.integers(1300, 2200 + 1))
        else:
            n = int(rng.integers(1000, 1800 + 1))

        # initialise shape (occasionally use an elite candidate as seed)
        if top_elite and rng.random() < 0.4:
            src_seq = top_elite[rng.integers(len(top_elite))][1]
            if src_seq.size >= n:
                init = src_seq[:n].copy()
            else:
                pad = rng.random(n - src_seq.size) * 1e-6
                init = np.concatenate([src_seq, pad])
        else:
            init = _initial_shape_random(n, rng)

        # normalise to the canonical sum sqrt(2 n)
        target_sum = math.sqrt(2.0 * n)
        init = init / init.sum() * target_sum

        # Frank–Wolfe optimisation (fast, a few LP solves)
        seq_fw, score_fw = _frank_wolfe_optimize(
            init, target_sum, deadline, rng, max_iters=12
        )

        if score_fw < best_score:
            best_score = score_fw
            best_seq = seq_fw.copy()
            top_elite.append((best_score, best_seq.copy()))
            top_elite.sort(key=lambda x: x[0])
            top_elite = top_elite[:5]      # keep only a handful
            print(f"[{time.time():.2f}] GLOBAL NEW BEST {best_score:.8f} (n={n})")
            if best_score < 1.503:          # target already met
                break

    # ------------------------------------------------------------------
    # Final polishing with Adam (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 7.0:
        n = best_seq.size
        target_sum = math.sqrt(2.0 * n)
        # re‑scale to exact simplex (just in case numerical drift)
        a = best_seq * (target_sum / best_seq.sum())
        seq_ref, score_ref = _adam_refine(
            a, target_sum, deadline, rng, max_iters=3000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = seq_ref.copy()
            print(f"[{time.time():.2f}] ADAM REFINEMENT improved to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Fallback: uniform sequence (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list (the evaluator expects this)
    return best_seq.tolist()
