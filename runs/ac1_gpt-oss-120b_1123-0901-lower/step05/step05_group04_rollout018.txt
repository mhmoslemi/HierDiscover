def project_onto_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimisation for the step‑function problem.

1.  A fast Adam optimiser works on a soft‑max parametrisation
    (a = S·softmax(x), S = sqrt(2·n)).  A log‑sum‑exp smooth max of the
    self‑convolution is annealed aggressively so that the optimiser
    behaves like a sub‑gradient method on the true max.

2.  The best Adam result is refined by a simple projected sub‑gradient
    descent on the exact max‑convolution (the objective under the
    normalisation S = sqrt(2·n) is simply max(conv(a,a))).  The sub‑gradient
    is known analytically, a step is taken and the vector is projected
    back onto the probability simplex (non‑negative, fixed sum).  The step
    size decays slowly and is occasionally increased when an improvement
    is observed.

The routine respects the 1000 s budget, prints intermediate scores and
returns the best sequence found.  If a previous construction is present
under the name ``height_sequence_1`` it is used as a warm‑start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 size."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-5,
    alpha: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.95,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed‑max surrogate.
    Returns the best sequence encountered (as a NumPy array) and its
    official evaluation score (computed by ``evaluate_sequence``).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ---- map logits to the simplex (a = S·softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stabilise
        p = exp_shift / exp_shift.sum()               # probabilities, sum = 1
        a = target_sum * p

        # ---- convolution and current (smooth) max ----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  Adam local best = {max_conv:.8f}")

        # ---- smooth‑max weights (log‑sum‑exp surrogate) ----
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau               # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a (derivative of Σ w_k·conv_k) ----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # pick needed slice
        grad_a = 2.0 * conv2[idx]

        # ---- gradient w.r.t. logits x (chain rule through the soft‑max) ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping – prevents occasional huge steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # ---- Adam update ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to help escape plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9997

        # early stop if we have not improved for a long stretch
        if iteration - last_improve > 2500:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # final evaluation with the official evaluator (preserves rounding behaviour)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {z ≥ 0, sum(z) = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of sum s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient refinement (direct optimisation of max(conv))
# ----------------------------------------------------------------------
def _subgradient_refine(
    a_init: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective max(conv(a,a))
    under the normalisation sum(a) = sqrt(2·n).
    Returns the best sequence found and its evaluation score.
    """
    a = a_init.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initial step size – a fraction of the total mass
    step = S * 0.02
    decay = 0.995
    best_a = a.copy()
    # under the normalisation the objective equals the max‑convolution entry
    best_score = np.max(_full_convolution_fft(a, a))

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution and the index where the maximum occurs
        conv = _full_convolution_fft(a, a)
        p = int(conv.argmax())

        # sub‑gradient of max(conv) at the current a (one active index p)
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = p - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # gradient step and projection back onto the simplex
        a_tmp = a - step * grad
        a = _project_onto_simplex(a_tmp, S)

        # evaluate (exact, no clipping needed because we stay within bounds)
        cur_score = np.max(_full_convolution_fft(a, a))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # a small increase in step size when we succeed
            step = min(step * 1.05, S * 0.05)
        else:
            # otherwise decay the step size
            step = max(step * decay, S * 1e-6)

        # occasional large jitter to escape flat regions
        if iteration % 500 == 0 and rng.random() < 0.10:
            jitter = rng.normal(scale=step * 0.5, size=n)
            a = _project_onto_simplex(a + jitter, S)

    # final official evaluation
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, final_score
