def _subgradient_refine(a, deadline, rng):
    target_sum = a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = _max_convolution(a, Nfft)
    Nfft = 1 << ((2 * n - 1).bit_length())
    step = 0
    while time.time() < deadline and step < max_steps:
        # compute conv
        A_fft = np.fft.rfft(best_a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2*n - 1]
        M = float(conv.max())
        k = int(conv.argmax())
        # gradient direction
        rev = np.zeros_like(best_a)
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        # iterate i from lo to hi:
        rev_idx = np.arange(lo, hi+1)
        rev[rev_idx] = best_a[k - rev_idx]
        g = 2.0 * rev
        # step size schedule
        alpha = (target_sum / n) * 0.1 * (0.99 ** step)  # decaying
        # project onto simplex after step
        a_new = best_a - alpha * g
        # projection onto simplex
        a_new = _project_simplex(a_new, target_sum)
        # compute new max
        A_fft = np.fft.rfft(a_new, n=Nfft)
        conv_new = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2*n - 1]
        M_new = float(conv_new.max())
        if M_new < best_M - 1e-12:
            best_a = a_new
            best_M = M_new
            step = 0  # reset step count maybe
            print('[subgrad] improved M =', best_M)
        else:
            # possibly accept with some probability?
            step += 1
    return best_a, best_M

"""
Propose a non‑negative step function (vector) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Choose a single “large” length (up to 8000) – more degrees of freedom
   usually allow a lower max‑convolution.
2. Generate a diverse pool of start vectors:
   * uniform, Dirichlet (different concentration), exponential,
     Gaussian, sinusoidal and the historic best (if available).
3. For each start run an **exponentiated‑gradient Adam** optimiser that
   works on the soft‑max approximation of the ℓ∞‑norm of the convolution.
   The optimiser uses a slightly larger learning rate, more frequent jitter
   and a slower temperature decay than the reference implementation.
4. After the budget for the Adam phase expires, apply a **projected
   sub‑gradient descent** that directly targets the (exact) maximum entry
   of the convolution.  The sub‑gradient step is
   ``g = 2·rev`` where ``rev[i] = a[max−i]`` for the current peak position.
   After each step the iterate is projected onto the simplex
   {x≥0, Σx = √(2 n)}.
   This cheap O(n) method can run for many thousands of iterations and
   typically squeezes the objective a few per‑cent lower.
5. Finally a short stochastic “pairwise‑mass‑transfer” refinement (the
   same routine as in the reference code) is performed.
6. The best sequence found over all starts is returned as a plain Python
   list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range ignored)
    using FFTs.  This is the gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # linear part, length = 3 n‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet with moderate concentration
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) Dirichlet with low concentration (spiky)
    g = rng.gamma(shape=0.3, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 4) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 5) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 6) Gaussian centred at the middle
    centre = (n - 1) / 2.0
    sigma = rng.uniform(0.1 * n, 0.3 * n)
    gauss = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    # 7) sinusoidal (sin^p) shape
    p = rng.uniform(1.0, 3.0)
    sin_shape = np.sin(np.pi * (np.arange(n) + 1) / (n + 1)) ** p
    init.append(sin_shape / sin_shape.sum() * target_sum)

    # 8) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 9) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 10) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen in
    this run together with its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (add tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.12                      # slightly larger base learning rate
    jitter_interval = 1000         # more frequent jitter

    # temperature (soft‑max) schedule
    tau_factor = 0.60               # start with a bit more smoothing
    decay_rate = max_iters / 6.0    # slower exponential decay

    # FFT size – must be large enough for linear conv of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector (softmax of logits) -----
        logits = y - np.max(y)               # shift for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()    # probability simplex
        a = target_sum * p                   # guaranteed Σa = target_sum, a_i≥0

        # ----- exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- gradient wrt a (2·rev, see derivation) -----
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on the simplex) -----
        avg = np.dot(p, g_a)                 # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.02, size=n)   # additive jitter on logits
            y += jitter

        # ----- keep best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x >= 0, Σ x = s}.  Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly minimises the exact
    ℓ∞‑norm of the convolution (i.e. its maximum entry).  Each iteration
    lowers the current peak by moving mass away from the positions that
    contribute to it and re‑projecting onto the simplex.
    """
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule for step size
    max_steps = 800_000
    step = 0
    while time.time() < deadline and step < max_steps:
        # compute exact convolution and its peak
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())
        k = int(conv.argmax())

        # if we already beat the current global best, stop early
        if M < best_score - 1e-12:
            best_score = M
            best_seq = a.copy()
            print(f"[subgrad] new global best = {best_score:.6f}")

        # --- sub‑gradient for the max entry ---
        rev = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        rev[i] = a[k - i]
        g = 2.0 * rev                     # ∂ max / ∂ a (sub‑gradient)

        # step size – decays exponentially
        alpha = (target_sum / n) * 0.4 * math.exp(-0.8 * step / max_steps)

        # take a tentative step and project back onto the simplex
        a_candidate = _project_onto_simplex(a - alpha * g, target_sum)

        # evaluate candidate
        A_fft = np.fft.rfft(a_candidate, n=Nfft)
        conv_cand = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M_cand = float(conv_cand.max())

        # accept if improvement, otherwise keep current iterate
        if M_cand < M - 1e-12:
            a = a_candidate
            # occasionally report progress
            if step % 10000 == 0:
                print(f"[subgrad] iter {step:6d} improved M -> {M_cand:.6f}")
        else:
            # with a tiny probability we still accept (simulated annealing)
            T = 0.01 * M
            prob = math.exp(-(M_cand - M) / max(T, 1e-12))
            if rng.random() < prob:
                a = a_candidate

        step += 1

    # final evaluation
    final_score = _max_convolution(a, Nfft)
    return a, final_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            if step % 5000 == 0:
                print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list)
    found within the given time budget.
    """
    # deterministic RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for finalisation

    # ------------------------------------------------------------------
    # Load any pre‑discovered construction (if present)
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a single, relatively large length.  Larger n gives more freedom.
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100  # historic fallback
    # Aim for the maximal allowed length unless n0 is already huge
    n = min(8000, max(2000, int(1.5 * n0)))

    target_sum = math.sqrt(2.0 * n)
    print(f"[setup] using length {n}, target_sum = {target_sum:.3f}")

    # ------------------------------------------------------------------
    # Build a diverse pool of starts
    init_vectors = _make_initial_vectors(n, target_sum, rng)

    # also interpolate the historic best if it has a different size
    if base is not None and base.shape[0] != n:
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        interp = np.interp(new_x, old_x, base)
        interp = np.maximum(interp, 0.0)
        interp = interp / interp.sum() * target_sum
        init_vectors.append(interp)

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Determine an iteration budget that respects the overall time limit.
    # Rough estimate: each Adam iteration costs ~2 FFTs → a few milliseconds.
    # We allocate roughly 70 % of the time to the Adam phase.
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.70 * remaining

    # Roughly 2 000 000 iterations for n≈2000, scale inversely with n.
    max_iters = int(3_000_000 * 2100 / n)
    max_iters = max(200_000, max_iters)
    print(f"[run] length={n}, max_iters≈{max_iters}")

    for idx, a0 in enumerate(init_vectors):
        if time.time() >= adam_deadline:
            break
        print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

        best_seq, best_score = _exp_gradient_optimize(
            a0=a0.copy(),
            n=n,
            deadline=adam_deadline,
            max_iters=max_iters,
            best_seq=best_seq if best_seq is not None else a0,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Sub‑gradient refinement (cheaper, can run many more iterations)
    if best_seq is not None and time.time() < deadline:
        subgrad_deadline = time.time() + 0.20 * (deadline - time.time())
        print("[phase] starting projected sub‑gradient refinement")
        best_seq, best_score = _subgradient_refine(
            a0=best_seq,
            n=n,
            deadline=subgrad_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Local stochastic refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        refine_deadline = time.time() + min(0.10 * (deadline - time.time()), 20.0)
        print("[phase] final stochastic refinement")
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=n,
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
