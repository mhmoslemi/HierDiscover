    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # safety

    best_small_seq = None
    best_val = float('inf')

    n_small = 2048

    # generate candidates
    candidates = []
    # uniform
    candidates.append(np.full(n_small, 1.0/n_small))
    # random dirichlet
    candidates.append(np.random.dirichlet(np.ones(n_small)))
    # edge power-law: for alpha in [-2, -1.5, -1, -0.5] maybe
    idx = np.arange(n_small, dtype=np.float64) + 1.0
    prod = idx * (n_small - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 4):
        shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
        if shape.sum()>0:
            candidates.append(shape/shape.sum())
    # inverse power edges
    for p in (0.5,1.0,2.0,3.0):
        i = np.arange(n_small, dtype=np.float64)+1.0
        shape = np.power(i,-p)+np.power(n_small-i+1.0,-p)
        candidates.append(shape/shape.sum())
    # mixture random
    for _ in range(3):
        mix = np.random.rand()
        base = np.random.dirichlet(np.ones(n_small))
        edge = np.power(np.arange(1,n_small+1, dtype=np.float64), -1.0)
        shape = mix*base + (1-mix)*edge
        candidates.append(shape/shape.sum())
    
    # ensure unique maybe.

    # iterate over candidates
    for cand in candidates:
        if time.time()>deadline:
            break
        a = cand.copy()
        # LP refinement loop
        max_lp_iters = 20
        t_mix = 0.5
        t_decay = 0.9
        for it in range(max_lp_iters):
            if time.time()>deadline:
                break
            conv = np.convolve(a, a)
            M = conv.max()
            # build A matrix
            n=n_small
            A = np.zeros((2*n-1, n), dtype=np.float64)
            for i in range(n):

                break
            g = res.x
            sum_g = g.sum()

                break

            g = g / sum_g

            t_mix *= t_decay

            val = evaluate_sequence(a.tolist())

        a_adam, val_adam = _adam_optimize(a, n_small, fft_len, corr_fft_len, deadline, max_iters=2000)

    # After all candidates for small n, upsample to larger n

    # Choose final length large (maybe 16384 if time, else 8192)

        best_seq = final_seq
    else:

    return best_seq.tolist()

"""
Propose a step‑function sequence that minimises

    F(a) = 2·n·max(a∗a) / (∑a)²

The method is deliberately different from the reference implementation:

1.  work on a moderate length (n≈2048) where a dense linear programme
    can be solved cheaply;
2.  for each start‑point run a few LP‑steps that maximise the total
    mass of a partner sequence g under the current convolution bound.
    This “active‑set” move moves mass towards the edges and reduces the
    peak of the self‑convolution.
3.  optionally polish the result with the smooth‑max Adam optimiser
    (the same routine used before, but with a tighter schedule).
4.  finally up‑sample the best shape to the target length
    (≈16384) and perform a long Adam run.

The whole search respects a global time budget (default 1000 s) and
returns a plain Python list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility helpers -------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum a)² (no clipping)."""
    conv = np.convolve(a, a)          # exact linear convolution
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * a.shape[0] * max_c / (s * s)


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate) ---------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) encountered before deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                     # initial learning rate
    lam = 0.5                     # smooth‑max temperature
    lam_factor = 1.03

    best_a = a.copy()
    best_val = _objective(a)

    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- simplex projection (soft‑max) -------------------------
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- smooth‑max weights ------------------------------------
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # ----- gradient of surrogate --------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={t:06d}] new best {cur_val:.6f}")

        # occasional random jitter if we stall
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programme refinement (edge‑pushing) ----------------------------
def _lp_refine(a: np.ndarray,
               max_iters: int,
               deadline: float,
               mix_start: float = 0.5,
               mix_decay: float = 0.90) -> np.ndarray:
    """
    Repeatedly solve a linear programme that maximises total mass
    under the current convolution bound.
    """
    n = a.shape[0]
    t_mix = mix_start

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current peak of the self‑convolution
        conv = np.convolve(a, a)
        M = conv.max()

        # build the convolution matrix A (size (2n‑1) × n)
        m_rows = 2 * n - 1
        A = np.zeros((m_rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = a

        # linear programme: maximise 1ᵀg  ⇔ minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)     # minimise –sum(g)
        bounds = [(0, None)] * n
        b_ub = np.full(m_rows, M, dtype=np.float64)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # If the LP fails, abort further LP steps.
            break

        g = res.x
        s = g.sum()
        if s < 1e-12:
            break
        g = g / s                     # normalise to sum = 1

        # convex mixing step
        a = (1.0 - t_mix) * a + t_mix * g
        t_mix *= mix_decay

        # early stop if the bound does not improve appreciably
        new_val = _objective(a)
        if new_val >= _objective(a) * (1.0 + 1e-8):
            # no substantial decrease; continue a few more iterations
            pass

    return a


# ----------------------------------------------------------------------
# Main search ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1.   Work on a modest size where a dense LP is affordable.
    n_small = 2048
    fft_len_c_small = 1 << int(np.ceil(np.log2(2 * n_small - 1)))
    fft_len_corr_small = 1 << int(np.ceil(np.log2(3 * n_small)))

    # 2.   Diverse warm‑starts (all normalised to sum = 1)
    candidates = []

    # uniform
    candidates.append(np.full(n_small, 1.0 / n_small))

    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n_small)))

    # two‑spike edge shape (maximal M, good LP slack)
    e = np.zeros(n_small)
    e[0] = e[-1] = 0.5
    candidates.append(e)

    # edge‑concentrated power‑law ((i+1)*(n‑i))**α, α<0
    idx = np.arange(n_small, dtype=np.float64) + 1.0
    prod = idx * (n_small - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 4):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    # inverse‑root edges ((i+1)**‑p + (n‑i)**‑p)
    for p in (0.5, 1.0, 2.0, 3.0):
        i = np.arange(n_small, dtype=np.float64) + 1.0
        shape = np.power(i, -p) + np.power(n_small - i + 1.0, -p)
        candidates.append(shape / shape.sum())

    # mixture of a random Dirichlet and edge‑spike (adds variability)
    for _ in range(3):
        mix = np.random.rand()
        base = np.random.dirichlet(np.ones(n_small))
        edge = np.zeros(n_small)
        edge[0] = edge[-1] = 0.5
        shape = mix * base + (1 - mix) * edge
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # shuffle to avoid systematic bias
    np.random.shuffle(candidates)

    # ------------------------------------------------------------------
    # 3.   Refine every candidate with LP + short Adam.
    for cid, cand in enumerate(candidates):
        if time.time() > deadline:
            break

        print(f"\n--- Candidate {cid} / {len(candidates)} (n={n_small}) ---")

        a = cand.copy()

        # --- LP refinement ------------------------------------------------
        lp_iters = 25               # cheap, yet enough for a clear move
        a = _lp_refine(a, max_iters=lp_iters, deadline=deadline,
                       mix_start=0.5, mix_decay=0.90)

        # --- short Adam fine‑tuning --------------------------------------
        # give each candidate a modest budget; the global best will later
        # receive a long run.
        adam_iters = int(2000 * 1024 / n_small) + 500   # ≈ 1500 for n=2048
        a, val = _adam_optimize(a, n_small,
                                fft_len_c_small,
                                fft_len_corr_small,
                                deadline,
                                max_iters=adam_iters)

        # keep the best across all candidates
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"*** Global new best after cand {cid}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4.   Upsample the current best to a large length and polish it.
    if best_seq is None:
        # fallback (should never happen)
        best_seq = np.full(n_small, 1.0 / n_small)

    # Decide final length depending on remaining time.
    final_n = 16384
    if time.time() + 10 > deadline:          # not enough time → stay small
        final_n = n_small

    if final_n != n_small:
        print(f"\n--- Upsampling to final length {final_n} ---")
        best_seq = _interpolate_seq(best_seq, final_n)

        fft_len_c_big = 1 << int(np.ceil(np.log2(2 * final_n - 1)))
        fft_len_corr_big = 1 << int(np.ceil(np.log2(3 * final_n)))

        # long Adam run (more iterations than the short runs above)
        remaining = deadline - time.time()
        # allocate most of the remaining budget (still keep a safety margin)
        max_iters_big = int(15000 * 1024 / final_n) + 5000
        # safety: never exceed remaining time – the optimiser checks itself.
        best_seq, val_big = _adam_optimize(best_seq, final_n,
                                           fft_len_c_big,
                                           fft_len_corr_big,
                                           deadline,
                                           max_iters=max_iters_big)
        if val_big < best_val:
            best_val = val_big
            print(f"*** Global best after final refinement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list as required.
    return best_seq.tolist()
