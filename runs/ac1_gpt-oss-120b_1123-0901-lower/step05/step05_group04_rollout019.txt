def lp_max_sum_under_conv(f, M):
    n = len(f)
    # Build A
    A = np.zeros((2*n-1, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = f
    # Solve LP: maximize sum(b) subject to A @ b <= M
    c = -np.ones(n)  # negatives for maximizing
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a step‐function whose evaluation

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

is as small as possible.

Algorithm
---------

1. **Random start** – pick a length `n` (1500‑3000) and a random
   non‑negative shape (power‑law, exponential, mixed, uniform).  The
   vector is scaled to the canonical sum `√(2 n)`.

2. **Linear‑programming improvement** – given a current sequence `f`
   with `M = max(conv(f, f))` we solve

        maximise   Σ b_i
        subject to  (f * b)_k ≤ M      for all k = 0 … 2n‑2
                    b_i ≥ 0

   (`f * b` is a linear convolution, therefore the constraints are linear).
   The LP returns an extreme point `b`.  After normalising `b` to the same
   sum as `f` we form a convex combination

        f ← (1‑t)·f + t·b   (t = 0.2, reduced if necessary)

   This step never increases `M`.  Repeating it a few times drives the
   maximum of the auto‑convolution down.

3. **Multiple restarts** – the above LP loop is executed for many
   independently generated starts while the global time budget permits.
   The best sequence (lowest true evaluation) is stored.

4. **Fine‑tuning** – when time remains a short final phase runs a
   standard Adam optimiser on the soft‑max parametrisation of the best
   sequence.  This refines the solution on the true non‑smooth objective.

The whole routine respects the 1000 s budget; if a sequence with a score
below the target 1.503 is found it is returned immediately.

Dependencies:  numpy, scipy (for `linprog`), and the already‑imported
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a power‑of‑2 FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def conv_max_fast(a: np.ndarray) -> float:
    """Return max entry of the auto‑convolution of ``a``."""
    return _full_convolution_fft(a, a).max()


# ----------------------------------------------------------------------
# Random initial shapes
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive vector of length n with a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)

    shape += rng.random(n) * 1e-6          # tiny jitter
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Build a dense convolution‐constraint matrix A for a given f.
#    (f * b)_k = Σ_i f_i * b_{k-i}  ⟹  row k of A contains the shifted copy of f
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A[:] = 0.0
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# One LP improvement step – returns the (possibly) improved sequence.
# ----------------------------------------------------------------------
def _lp_improve_step(f: np.ndarray, M: float, target_sum: float, rng: np.random.Generator,
                     max_inner_iter: int = 25) -> tuple[np.ndarray, float]:
    """
    Perform a short LP‑based refinement of ``f``.

    Returns the improved sequence and its new max‑convolution value.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_M = M

    for it in range(max_inner_iter):
        # 1) build constraints and solve LP
        A = _build_constraint_matrix(f)
        c = -np.ones(n)                         # maximise Σ b_i  ⟹ minimise -Σ b_i
        b_ub = np.full(2 * n - 1, M)           # (f * b)_k ≤ M
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break

        b_raw = res.x
        s = b_raw.sum()
        if s <= 0.0:
            break

        # 2) re‑scale to the canonical sum √(2 n)
        b = b_raw * (target_sum / s)

        # 3) try a convex combination  f ← (1‑t) f + t b
        improved = False
        for t in (0.2, 0.1, 0.05, 0.02, 0.01):
            f_new = (1.0 - t) * f + t * b
            # re‑normalise (tiny numerical drift)
            f_new *= (target_sum / f_new.sum())
            M_new = conv_max_fast(f_new)
            if M_new < best_M - 1e-9:
                f = f_new
                best_f = f_new.copy()
                best_M = M_new
                improved = True
                break
        if not improved:
            # no improvement with any step size → stop
            break

    return best_f, best_M


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation (used for final polishing)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:  # noqa: E302
    """Full linear convolution using FFT (identical to the definition above)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed‑max surrogate.
    Returns the best sequence encountered (as a NumPy array) and its
    exact evaluation score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # map logits to the simplex (a = S·softmax(x))
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # convolution and smooth‑max weight
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] Adam n={n}  best={best_val:.8f}")

        # smooth‑max (log‑sum‑exp) weights
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # gradient wrt logits x (chain rule through soft‑max)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        alpha *= 0.9997

        if iteration - last_improve > 2000:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score
