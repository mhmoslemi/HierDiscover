    # Use Adam
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    # hyperparameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = 0.1 # start temperature low? Actually we want smoothness, so small lam -> less smoothing? lam is factor on exponent? In previous code lam started at 0.5 and increased.
    lam_factor = 1.02

    best_seq_local = best_seq
    best_val_local = best_val
    last_improve = time.time()
    iteration = 0
    max_iters = 500_000   # high
    while time.time() < deadline and iteration < max_iters:
        # compute a via softmax
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # compute convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)  # note a sum = 1
        if val < best_val_local:
            best_val_local = val
            best_seq_local = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={iteration}] best {val:.6f}")
        # compute smooth max weights
        # Using lam as scaling factor, if lam is large becomes sharper. Starting lam small yields more smoothing?

        w_sum = exp_w.sum()

        else:
            w = exp_w / w_sum
        # gradient wrt a: 2 * cross_corr(w,a)

    # early shape: use warm or random.

    # Use similar to previous _initial_candidates but also include some random mixture.
    candidates = []

"""
Propose a low‑score step‑function (non‑negative vector) for the
`evaluate_sequence` benchmark.

Algorithm
---------
1. Build a modest pool of diverse starting vectors (warm‑start,
   random Dirichlet, edge‑heavy power‑law, mirrored Gaussians, sinusoid,
   uniform).
2. For each start run a two‑phase optimiser on a given length *n*:
   * Phase‑A – Adam on a smooth‑max surrogate (log‑sum‑exp) of the
     convolution peak.  The parameters are the logits of a soft‑max
     representation, guaranteeing the simplex constraints.
   * Phase‑B – Exponentiated‑gradient sub‑gradient refinement on the
     exact max‑convolution objective.
   The optimiser keeps the globally best sequence found so far.
3. After the 4096‑point optimisation, if time permits the best shape is
   linearly interpolated to longer grids (8192 → 16384) and the optimiser
   is run again on each size.  Larger *n* gives the algorithm more freedom
   to flatten the autocorrelation and therefore to reduce the evaluation
   score.
4. The routine respects the total time budget, prints intermediate
   improvements and finally returns the best sequence (clipped to the
   evaluator’s allowed range).

The optimiser is fully vectorised (FFT‑based convolutions) and fits well
within the 1000 s budget on a single CPU core.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed ≈ 1)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by an exponentiated‑gradient
    sub‑gradient phase.  Updates and returns the best sequence/value seen
    globally (across all calls).
    """
    # Initialise logits for soft‑max (ensures positivity and simplex)
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    # smooth‑max temperature (lam) – start low, increase gradually
    lam = 0.10
    lam_factor = 1.02

    last_improve = time.time()
    it = 0
    max_adam_iters = 500_000

    while time.time() < deadline and it < max_adam_iters:
        # ----- simplex point from logits -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()          # a lies on the simplex, sum(a)=1

        # ----- objective (exact) for monitoring -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum()≈1
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        # w = softmax(lam * conv)
        exp_w = np.exp(lam * (conv - max_c))
        w_sum = exp_w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_w / w_sum

        # ----- gradient w.r.t. a (through convolution) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through the soft‑max parametrisation -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 5e5)

        # ----- occasional jitter if stagnating -----
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    # ------------------------------------------------------------------
    # Sub‑gradient refinement on the true max‑convolution.
    # Use exponentiated‑gradient updates which stay inside the simplex.
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= max(a.sum(), 1e-12)

    eta = 0.55
    sub_last_improve = time.time()
    max_sub_iters = 20_000
    sub_it = 0
    while time.time() < deadline and sub_it < max_sub_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active set – indices where the convolution attains the max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stagnation handling
        if time.time() - sub_last_improve > 12.0:
            # small random shake to escape flat regions
            a += (np.random.rand(n) - 0.5) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= max(a.sum(), 1e-12)
            sub_last_improve = time.time()

        sub_it += 1

    return global_best_seq, global_best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Construct a small diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # a few rotations of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (uniform on the simplex)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the evaluation score.
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – obtain an optional warm start from the global variable
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # ensure non‑negativity and normalisation later
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – optimise on n = 4096 (fast, decent baseline)
    n0 = 4096
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    candidates = _initial_candidates(n0, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n0))
        else:
            a0 /= s

        best_seq, best_val = _optimize(
            a0, n0, conv_fft_len0, corr_fft_len0, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to 8192 if time permits
    if time.time() + 8 < deadline:
        n1 = 8192
        conv_fft_len1 = _next_pow2(2 * n1 - 1)
        corr_fft_len1 = _next_pow2(3 * n1 - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n1, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n1))

        best_seq, best_val = _optimize(
            a_big,
            n1,
            conv_fft_len1,
            corr_fft_len1,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to 16384 if still time left
    if time.time() + 8 < deadline:
        n2 = 16384
        conv_fft_len2 = _next_pow2(2 * n2 - 1)
        corr_fft_len2 = _next_pow2(3 * n2 - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n2, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n2))

        best_seq, best_val = _optimize(
            a_big,
            n2,
            conv_fft_len2,
            corr_fft_len2,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never happen, but safety)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed domain and return as plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value (estimated): {best_val:.6f}")
    return best_seq.tolist()
