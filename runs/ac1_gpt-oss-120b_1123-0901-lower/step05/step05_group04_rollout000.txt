        # decide weights

        v = beta2*v + (1-beta2)*(grad_x**2)
        m_hat = m/(1 - beta1**iteration)

        alpha *= 0.9998

            break
        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())

"""
Hybrid optimiser for the max‑autocorrelation step‑function problem.

The objective is 2*n*max(conv(a,a))/sum(a)^2.  When the vector a is
scaled to satisfy Σa = √(2n) this reduces to minimising the maximum entry of
the (full) self‑convolution, i.e. the peak of the autocorrelation.

Our search consists of three stages:

1. **Many short Adam restarts** – the coefficients are parametrised by
   logits x so that a = S·softmax(x) with S = √(2n).  The Adam optimiser
   works on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature is aggressively annealed (τ → 0) so that the surrogate
   quickly approaches the true max.

2. **Hard‑max refinement** – after the budget has been partially spent we
   switch to a sub‑gradient optimiser that uses the exact max‑convolution
   (weights are an indicator of the active lag(s)).  The same Adam update
   on the logits is employed.

3. **Stochastic hill‑climbing** – the best candidate from the previous
   phases is further perturbed by moving tiny amounts of mass between two
   random positions while keeping Σa fixed.  Improvements are accepted
   immediately, which helps to escape shallow local minima.

The routine respects the global time budget (default 1000 s) and always
returns a feasible list of non‑negative heights.  If a previously found
construction is present in the global variable `height_sequence_1` it is
used as a warm‑start.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helper – FFT based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities
# ----------------------------------------------------------------------
def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector that sums to 1."""
    max_logit = np.max(logits)
    shifted = np.exp(logits - max_logit)
    return shifted / shifted.sum()


def _logits_to_seq(logits: np.ndarray, target_sum: float) -> np.ndarray:
    """Map logits → non‑negative sequence with prescribed ℓ1‑norm."""
    p = _softmax(logits)
    return target_sum * p


def _grad_logits_from_grad_a(p: np.ndarray, grad_a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Convert a gradient w.r.t. a (∂L/∂a) into a gradient w.r.t. logits x
    using the soft‑max chain rule.
    """
    dotp = np.dot(p, grad_a)
    return target_sum * p * (grad_a - dotp)

#   mode='smooth'  → log‑sum‑exp surrogate with decreasing temperature τ
#   mode='hard'    → exact max‑convolution (indicator weights)
# ----------------------------------------------------------------------

    deadline: float,
    rng: np.random.Generator,
    *,

    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on logits x.  Returns the best sequence (as np.ndarray)
    encountered and its official evaluation score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")          # this stores max(conv) – lower is better
    last_improve = 0
    iteration = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) map logits → sequence a
        # ------------------------------------------------------------------
        p = _softmax(x)
        a = target_sum * p

        # ------------------------------------------------------------------
        # 2) convolution, retrieve max value
        # ------------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        # ------------------------------------------------------------------
        # 3) keep the best seen so far
        # ------------------------------------------------------------------
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n:4d}  new best max_conv = {best_val:.9f}")

        # ------------------------------------------------------------------
        # 4) construct the weight vector w for the gradient
        # ------------------------------------------------------------------
        if mode == "hard":
            # indicator of the (almost) active lag(s)
            mask = np.isclose(conv, max_conv, atol=1e-12)
            w = mask.astype(float)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum
        else:   # smooth surrogate
            tau = max(
                init_tau_factor * max_conv * (tau_decay ** iteration),
                min_tau_factor * max_conv,
            )
            scaled = (conv - max_conv) / tau          # ≤ 0
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

        # ------------------------------------------------------------------
        # 5) gradient w.r.t. a:  d/da Σ_k w_k·conv_k = 2·a * w_rev   (convolution)
        # ------------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                 # pick slice of length n
        grad_a = 2.0 * conv2[idx]

        # ------------------------------------------------------------------
        # 6) gradient w.r.t. logits (chain rule through soft‑max)
        # ------------------------------------------------------------------
        grad_x = _grad_logits_from_grad_a(p, grad_a, target_sum)

        # optional clipping of the raw gradient to avoid explosions
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # ------------------------------------------------------------------
        # 7) Adam update
        # ------------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9997

        # early stop if stagnation
        if iteration - last_improve > 3000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # ------------------------------------------------------------------
    # Final official evaluation (uses the same rounding / clipping as the
    # competition harness).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing – works on a fixed‑norm sequence.
# ----------------------------------------------------------------------
def _hill_climb(
    seq: np.ndarray,
    target_sum: float,
    budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform cheap local moves that preserve Σa = target_sum and a_i ≥ 0.
    Returns the best sequence found and its max‑convolution value.
    """
    a = seq.copy()
    n = a.size
    # initial value
    conv = _full_convolution_fft(a, a)
    best_val = conv.max()
    best_seq = a.copy()
    deadline = time.time() + budget

    while time.time() < deadline:
        # choose two distinct positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # propose to move a small amount of mass from i to j
        # amount is chosen proportionally to the current average height
        eps = rng.uniform(0.0, 0.03) * target_sum / n
        if a[i] < eps:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        conv_new = _full_convolution_fft(a_new, a_new)
        val_new = conv_new.max()
        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
            # optional early break if we have found something significantly better
            # (but continue to search for even better)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence discovered as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.15  # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction if available.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    top_candidates = []  # keep a few elite sequences (score, seq)

    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            top_candidates.append((sc, hs_arr.copy()))
            print(f"[{time.time():.2f}] Warm‑start from height_sequence_1: score = {sc:.9f}")
    except Exception:
        pass  # No warm‑start available

    # ------------------------------------------------------------------
    # Helper to create a random initial shape (before soft‑max scaling)
    # ------------------------------------------------------------------
    def _random_shape(n):
        typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
        idx = np.arange(1, n + 1, dtype=float)

        if typ == 'power':
            p = rng.uniform(0.1, 2.0)
            shape = idx ** (-p)
        elif typ == 'exponential':
            tau = rng.uniform(0.2, 0.9) * n
            shape = np.exp(-idx / tau)
        elif typ == 'mixed':
            split = rng.integers(1, n - 1)
            left = idx[:split] ** (-rng.uniform(0.2, 1.0))
            right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
            shape = np.concatenate([left, right])
        else:  # uniform
            shape = np.ones_like(idx)

        shape += rng.random(n) * 1e-6           # tiny jitter
        shape = np.maximum(shape, 1e-12)
        return shape

    # ------------------------------------------------------------------
    # Main restart loop – many short Adam runs with the smooth surrogate.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose problem size – start larger, shrink later for speed
        if remaining > 300:
            n = rng.integers(1500, 3500 + 1)
        elif remaining > 120:
            n = rng.integers(1000, 2500 + 1)
        else:
            n = rng.integers(600, 1800 + 1)

        # Initial logits (log‑it representation)
        if top_candidates and rng.random() < 0.45:
            # Warm‑start from a good previous candidate (possibly truncate / pad)
            src_seq = top_candidates[rng.integers(len(top_candidates))][1]
            if src_seq.size >= n:
                init = src_seq[:n].copy()
            else:
                pad = rng.random(n - src_seq.size) * 1e-6
                init = np.concatenate([src_seq, pad])
        else:
            init = _random_shape(n)

        target_sum = math.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)

        # Allocate a time slice for this run (max 12 s or a fraction of remaining)
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam(
            n=n,
            x0=x0,
            target_sum=target_sum,
            deadline=run_deadline,
            rng=rng,
            mode="smooth",
            init_tau_factor=0.25,
            min_tau_factor=5e-4,
            tau_decay=0.96,
            alpha=0.07,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            max_iter=None,
        )

        # Keep global best
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            top_candidates.append((best_score, best_seq.copy()))
            top_candidates.sort(key=lambda z: z[0])
            top_candidates = top_candidates[:5]   # retain only a few elites
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.9f} (n={n})")
            if best_score < 1.503:               # early exit if target reached
                break

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max refinement on the current best (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = math.sqrt(2.0 * n)
        a = best_seq * (target_sum / best_seq.sum())
        x0 = np.log(a + 1e-12)
        run_deadline = deadline
        seq_refined, score_refined = _optimise_adam(
            n=n,
            x0=x0,
            target_sum=target_sum,
            deadline=run_deadline,
            rng=rng,
            mode="hard",                     # use exact max‑convolution gradient
            init_tau_factor=0.0,            # ignored in hard mode
            min_tau_factor=0.0,
            tau_decay=0.0,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            max_iter=None,
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] HARD‑MAX REFINEMENT improved best to {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing around the best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 4.0:
        n = best_seq.size
        target_sum = math.sqrt(2.0 * n)
        # Use a modest budget (few seconds) for hill climbing
        hill_budget = min(4.0, deadline - time.time())
        seq_hc, hc_val = _hill_climb(best_seq, target_sum, hill_budget, rng)
        # Evaluate with the official function (to ensure identical rounding)
        hc_score = evaluate_sequence(seq_hc.tolist())
        if hc_score < best_score:
            best_score = hc_score
            best_seq = seq_hc.copy()
            print(f"[{time.time():.2f}] HILL‑CLIMB improved best to {best_score:.9f}")

    # ------------------------------------------------------------------
    # Fallback – ensure we always return something.
    # ------------------------------------------------------------------
    if best_seq is None:
        # simple uniform construction as a safe default
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to plain Python list as required by the external harness
    return best_seq.tolist()
