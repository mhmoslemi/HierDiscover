```
def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    timer = time.time()
    deadline = timer + budget_s - 0.1

    # Determine n:
    global height_sequence_1
    if 'height_sequence_1' in globals():
        best_initial = np.asarray(height_sequence_1, dtype=np.float64)
        n = len(best_initial)
    else:
        n = 16384
        best_initial = np.full(n, 1.0/n)

    # ensure sum=1
    if best_initial.sum() == 0:
        best_initial = np.random.dirichlet(np.ones(n))
    else:
        best_initial = best_initial / best_initial.sum()

    # candidate pool
    candidates = []
    candidates.append(best_initial) # warm start
    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))
    # upsample if we have shorter candidate: let's also draw from 8192 uniform upsample
    # here's a function to upsample d to length n with linear interpolation
    # also we can shape using power law edges
    idx = np.arange(1, n+1)
    prod = idx * (n - idx + 1)
    for alpha in [-1.5, -1.0, -0.75]:
        shape = np.where(prod>0, prod**alpha, 0.0)
        shape = shape / shape.sum()
        candidates.append(shape)

    # also sine-squared bump shape
    shape = np.sin(np.pi * (np.arange(n)+0.5)/n)**2
    shape = shape / shape.sum()
    candidates.append(shape)

    best_seq = None
    best_val = float('inf')
    # constants for optimization
    for cand_id, a0 in enumerate(candidates):
        if time.time() > deadline:
            break
        # Avoid zero entries
        a = np.maximum(a0, 1e-16)
        a /= a.sum()
        # init logits
        logits = np.log(a + 1e-30)  # safe

        # Adam parameters
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        lr = 0.05

        # lambda schedule
        lam = 2.0
        lam_factor = 1.02
        max_iters = 20000  # maybe adapt based on time
        last_improve = time.time()
        for it in range(1, max_iters+1):
            # end if time
            if time.time() > deadline:
                break

            # softmax
            mx = logits.max()
            exp_log = np.exp(logits - mx)
            a = exp_log / exp_log.sum()

            # conv
            conv = _full_convolution(a, fft_len=conv_fft_len)
            max_c = conv.max()
            # produce smoothmax weights
            # compute exponent terms numerically stable
            # w_i = exp(lam*(conv_i - max_c))
            exp_terms = np.exp(lam * (conv - max_c))
            w = exp_terms / exp_terms.sum()
            # gradient
            grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

            # backprop through softmax
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # lambda anneal
            lam = min(lam * lam_factor, 1e7)

            # evaluate exact objective
            val = 2.0 * n * max_c
            if val < best_val - 1e-9:
                best_val = val
                best_seq = a.copy()
```

The optimizer below searches for a non‑negative step‑function `a` that minimises  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{(\sum_i a_i)^2},
\]

where `*` denotes linear convolution.  
Key ideas

* **Symmetry** – the optimal shape is (empirically) symmetric.  
  By enforcing `a[i]=a[n‑1‑i]` we only have to optimise half of the
  coefficients (`m=n/2`).  This cuts the optimisation dimension in half
  and lets us perform many more gradient steps within the 1000 s budget.

* **Re‑parameterisation** – the half‑vector `v` lives on the probability
  simplex (`v_i≥0, Σv_i=1`).  We store `logits` and obtain `v` by a
  soft‑max, guaranteeing feasibility at every step.

* **Smooth‑max surrogate** – the true objective contains a `max`.  
  We replace it by a log‑sum‑exp with a temperature `λ`.  `λ` is
  annealed (starting small, multiplied by `1.02` each iteration) so
  that early iterations see a smooth landscape and later iterations
  approach the exact `max`.

* **Adam on the surrogate** – Adam’s adaptive moments are used for the
  logits of the half‑vector.  Because the full‑vector `a` is obtained
  from `v` by mirroring and a fixed scaling (`a = ½·sym(v)`), gradients
  can be back‑propagated analytically:
  ```
  grad_logits = v * (grad_v - (v·grad_v))
  grad_v = ½·grad_a[first half] + ½·grad_a[second half reversed]
  ```

* **Exact sub‑gradient refinement** – after the smooth‑max phase a few
  exponentiated‑gradient sub‑gradient steps on the *true* `max`
  further tighten the bound.

* **Diverse initialisations** – uniform, warm‑start from the global
  best (`height_sequence_1`), random Dirichlet, edge‑concentrated
  power‑law shapes and a sine‑squared bump are transformed to the
  half‑vector representation.

* **Time‑aware loop** – the routine monitors the wall‑clock and stops
  cleanly before the 1000 s limit.

The returned sequence is a plain Python list of non‑negative numbers,
normalised (its sum is 1) and ready for the external `evaluate_sequence`
function.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Symmetric step‑function optimiser.

The algorithm works on an even length n (default 16384 or the length of the
previous best `height_sequence_1`).  Only the first half of the coefficients
is optimised; the full sequence is obtained by mirroring and a fixed ½‑scale,
which guarantees a sum of 1 and non‑negativity.
A smooth‑max surrogate (log‑sum‑exp) is minimised with Adam, followed by a
few exact sub‑gradient refinements.  Several analytically‑motivated initial
vectors are tried, and the best result found before the time budget expires
is returned (as a plain list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_power_of_two(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    a is a 1‑D real array.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_power_of_two(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _symmetrize_half(v: np.ndarray) -> np.ndarray:
    """
    Given half‑vector v (sum(v)=1), build the full symmetric vector a
    of length n = 2*len(v) with a = ½·sym(v).
    """
    return np.concatenate((v, v[::-1]), axis=0) * 0.5


def _half_from_full(a: np.ndarray) -> np.ndarray:
    """
    Recover the half‑vector v from a symmetric full vector a.
    Works for even length n.
    """
    n = a.shape[0]
    m = n // 2
    # v_i = a_i + a_{n-1-i}
    return a[:m] + a[-1:-m - 1:-1]


def _smooth_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute the smooth‑max weight vector:
        w_i ∝ exp(lam * (conv_i - max(conv))).
    Guarantees sum(w)=1 and numerical stability.
    """
    max_c = conv.max()
    # subtraction avoids overflow
    exp_term = np.exp(lam * (conv - max_c))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        # fall back to uniform (should not happen for lam>0)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / w_sum


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock budget in seconds (default 1000 s).

    Returns
    -------
    list[float]
        The best coefficient sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (even length) and initialise best record
    # ------------------------------------------------------------------
    hseq_global = globals().get("height_sequence_1")
    if isinstance(hseq_global, (list, np.ndarray)) and len(hseq_global) >= 2:
        # force even length
        n_full = len(hseq_global)
        if n_full % 2 == 1:
            n_full -= 1
        n = max(1024, n_full)          # at least 1024 to have enough resolution
    else:
        n = 16384                       # default length
    if n % 2 == 1:
        n += 1                          # enforce even

    m = n // 2                         # optimisation dimension

    # pre‑compute FFT lengths once
    conv_fft_len = _next_power_of_two(2 * n - 1)
    corr_fft_len = _next_power_of_two(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Build a diverse pool of half‑vectors (each sums to 1)
    # ------------------------------------------------------------------
    candidates = []

    # uniform half‑vector
    candidates.append(np.full(m, 1.0 / m, dtype=np.float64))

    # warm‑start from previously discovered global best (if available)
    if hseq_global is not None and len(hseq_global) == n:
        a_full = np.asarray(hseq_global, dtype=np.float64)
        a_full = np.maximum(a_full, 0.0)
        a_full = a_full / a_full.sum()
        v_warm = _half_from_full(a_full)
        if v_warm.sum() > 0:
            v_warm = v_warm / v_warm.sum()
            candidates.append(v_warm)

    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(m)))

    # edge‑concentrated power‑law shapes, several exponents
    idx = np.arange(1, m + 1, dtype=np.float64)
    prod = idx * (m - idx + 1.0)          # triangular factor
    for alpha in np.linspace(-2.0, -0.5, 4):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # smooth sine‑squared bump
    bump = np.sin(np.pi * (np.arange(m) + 0.5) / m) ** 2
    bump = bump / bump.sum()
    candidates.append(bump)

    # ------------------------------------------------------------------
    # Optimise each candidate
    # ------------------------------------------------------------------
    for cand_id, v0 in enumerate(candidates):
        # ------------------------------------------------------------------
        # Time‑out guard before starting a new candidate
        # ------------------------------------------------------------------
        if time.time() > deadline:
            break

        # initialise logits for the half‑vector
        v = np.maximum(v0, 1e-16)
        v = v / v.sum()
        logits = np.log(v + 1e-30)                # safe log

        # Adam state
        m_adam = np.zeros_like(logits)
        v_adam = np.zeros_like(logits)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr = 0.06

        # smooth‑max temperature
        lam = 2.0
        lam_factor = 1.02

        last_improve = time.time()
        # iteration budget – scale with problem size but leave headroom
        max_iters = int(30000 * 1024 / n) + 3000

        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break

            # ---- 1. Soft‑max → valid half‑vector v --------------------
            mx = logits.max()
            exp_log = np.exp(logits - mx)
            v = exp_log / exp_log.sum()

            # ---- 2. Build full symmetric a -------------------------
            a = _symmetrize_half(v)   # a already sums to 1

            # ---- 3. Convolution and smooth‑max weights ---------------
            conv = _full_convolution(a, fft_len=conv_fft_len)
            max_c = conv.max()
            # smooth‑max weights
            w = _smooth_max_weights(conv, lam)

            # ---- 4. Gradient of the surrogate -----------------------
            grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

            # Scaling: a = ½·sym(v)  ⇒  grad_a0 = ½·grad_a
            grad_a0 = 0.5 * grad_a

            # Map gradient to the half‑vector v
            #   grad_v_i = grad_a0[i] + grad_a0[n-1-i]
            grad_v = grad_a0[:m] + grad_a0[-1:-m - 1:-1]

            # Back‑prop through soft‑max
            avg = np.dot(v, grad_v)
            grad_log = v * (grad_v - avg)

            # ---- 5. Adam update on logits ---------------------------
            m_adam = beta1 * m_adam + (1.0 - beta1) * grad_log
            v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m_adam / (1.0 - beta1 ** it)
            v_hat = v_adam / (1.0 - beta2 ** it)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- 6. Temperature annealing ---------------------------
            lam = min(lam * lam_factor, 1e7)

            # ---- 7. Exact objective (sum(a)=1) ----------------------
            val = 2.0 * n * max_c
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[cand {cand_id:02d} Adam it {it:05d}] new best {val:.6f}")

            # ---- 8. Stagnation handling -----------------------------
            if time.time() - last_improve > 5.0:
                logits += (np.random.rand(m) - 0.5) * 1e-3
                last_improve = time.time()

        # ------------------------------------------------------------------
        # Exact sub‑gradient refinement (few hundred iterations)
        # ------------------------------------------------------------------
        if best_seq is None:
            continue

        a = best_seq.copy()                         # already normalised
        # Re‑extract half‑vector for consistent updates
        v = _half_from_full(a)

        eta = 0.6
        sub_last = time.time()
        for sub_it in range(2000):
            if time.time() > deadline:
                break

            a = _symmetrize_half(v)                 # rebuild full vector
            conv = _full_convolution(a, fft_len=conv_fft_len)
            max_c = conv.max()
            # active set (indices attaining the true maximum)
            active = np.isclose(conv, max_c, atol=1e-12)
            if not np.any(active):
                break
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

            grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
            grad_a0 = 0.5 * grad_a
            grad_v = grad_a0[:m] + grad_a0[-1:-m - 1:-1]

            # Exponentiated‑gradient (mirrors the soft‑max re‑parametrisation)
            v = v * np.exp(-eta * grad_v)
            v_sum = v.sum()
            if v_sum < 1e-12:                     # avoid collapse
                v = np.random.dirichlet(np.ones(m))
            else:
                v = v / v_sum

            # evaluate exact objective
            a = _symmetrize_half(v)
            conv = _full_convolution(a, fft_len=conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a.copy()
                sub_last = time.time()
                print(f"[cand {cand_id:02d} Subgrad it {sub_it:04d}] new best {val:.6f}")

            if time.time() - sub_last > 5.0:
                break

    # ------------------------------------------------------------------
    # (Optional) quick random local search around the best solution
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_best = best_seq.copy()
        n = a_best.shape[0]
        for trial in range(200):
            if time.time() > deadline:
                break
            # propose a tiny random perturbation
            a_new = a_best.copy()
            idxs = np.random.choice(n, size=10, replace=False)
            a_new[idxs] += np.random.randn(10) * 0.005
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-12:
                continue
            a_new = a_new / s
            conv = _full_convolution(a_new, fft_len=conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a_new.copy()
                a_best = a_new
                print(f"[local search] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list (required by the external evaluator)
    return best_seq.tolist()
```
