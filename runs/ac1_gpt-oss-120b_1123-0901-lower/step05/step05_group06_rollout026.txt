def optimize_sequence(n, y0, target_sum, deadline):
    a_current_y = y0.copy()
    # Determine initial max conv for y0
    a0 = softmax(y0) * target_sum
    conv0 = conv_fft(a0, Nfft)
    max0 = conv0.max()
    tau = max0 * 0.5  # initial tau.
    # L-BFGS loop
    while time.time()<deadline and tau>1e-5:
        # define closure for objective & grad using current tau
        def fun(y):
            # compute a from y via softmax and scaling
            logits = y - np.max(y)
            exp_logits = np.exp(logits)
            p = exp_logits/exp_logits.sum()
            a = target_sum * p
            # FFT conv
            A_fft = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2*n - 1]
            # compute weight and Ltau
            maxc = conv.max()
            if tau < 1e-9:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
                Ltau = maxc
            else:
                ex = np.exp((conv - maxc)/tau)
                sum_ex = ex.sum()
                w = ex / sum_ex
                Ltau = maxc + tau * np.log(sum_ex)
            # weighted rev
            rev = _weighted_rev(a, w, Nfft)
            g_a = 2.0 * rev
            avg = np.dot(p, g_a)
            g_y = target_sum * p * (g_a - avg)
            return Ltau, g_y

        # Use scipy.optimize.minimize with L-BFGS-B
        res = minimize(fun, a_current_y, method='L-BFGS-B', jac=True,
                       options={'maxiter':200, 'ftol':1e-12, 'gtol':1e-12})
        a_current_y = res.x
        # compute exact max conv after this stage, update best if improved.
        a = softmax(a_current_y)*target_sum
        conv = conv_fft(a,Nfft)
        max_conv = conv.max()
        if max_conv < best_score: update global best.
        # reduce tau
        tau *= 0.5

        if time.time() >= deadline: break

"""
Improved search for the step‑function that minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Diverse initialisations** –  uniform, Dirichlet, exponential,
   Hann/Blackman windows and an interpolation of the previous best
   (if available).  Every start vector is normalised to Σa = √(2 n).

2. **L‑BFGS optimisation on the soft‑max smoothed max‑convolution.**
   The variables are logits y such that a = S·softmax(y) (S = √(2 n)).
   A smoothed max (log‑sum‑exp) with temperature τ provides a
   differentiable surrogate for max(a∗a).  The gradient is obtained
   by a weighted reverse convolution using FFTs.  τ is halved after
   each L‑BFGS run (continuation) which drives the solution towards the
   true max‑convolution.

3. **Targeted stochastic refinement** – a hill‑climbing move that
   transfers a small fraction of mass from a position with large
   gradient (i.e. contributes heavily to the current maximal convolution)
   to a position with small/negative gradient.  A simple simulated‑annealing
   acceptance rule helps to escape shallow plateaux.

All helper functions are top‑level (no closures) and the whole search
must finish within the supplied time budget (default 1000 s).  The best
sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy import optimize  # requires SciPy ≥1.0

# ----------------------------------------------------------------------
# FFT‑based convolution helpers ------------------------------------------------

def _conv_full(a: np.ndarray, Nfft: int) -> np.ndarray:
    """Linear convolution a*a (full) using an FFT of length Nfft."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, tau: float):
    """Soft‑max weights for a (smooth) max over ``conv``."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range indices ignored)
    via FFT – the gradient of the smoothed max‑convolution.
    """
    A = np.fft.rfft(a, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)   # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev * A, n=Nfft)
    # keep the linear part: length = (len(w)+len(a)-1) = 3 n‑2
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _softmax_to_a(logits: np.ndarray, target_sum: float) -> np.ndarray:
    """Map logits → positive vector on the simplex (Σ = target_sum)."""
    maxlog = np.max(logits)
    exp_logits = np.exp(logits - maxlog)
    p = exp_logits / exp_logits.sum()
    return target_sum * p


# ----------------------------------------------------------------------
# Objective class for L‑BFGS (no closures) --------------------------------

class LBFGSObjective:
    """
    Callable used by ``scipy.optimize.minimize``.
    Holds problem data (n, target sum, FFT size) and the current temperature τ.
    Returns the smoothed max and its gradient w.r.t. the logits.
    """

    def __init__(self, n: int, target_sum: float):
        self.n = n
        self.target_sum = target_sum
        # FFT length enough for linear convolution of length 3 n‑2
        self.Nfft = 1 << ((3 * n - 2).bit_length())
        self.tau = 1.0

    def set_tau(self, tau: float) -> None:
        self.tau = tau

    def __call__(self, y: np.ndarray):
        # a = S·softmax(y)
        a = _softmax_to_a(y, self.target_sum)

        # exact convolution a*a
        conv = _conv_full(a, self.Nfft)

        # smooth max (log‑sum‑exp) and its weights
        w, Ltau = _softmax_weights(conv, self.tau)

        # gradient w.r.t. a (2·rev)
        rev = _weighted_rev(a, w, self.Nfft)
        g_a = 2.0 * rev

        # chain rule to logits (mirror‑descent on the simplex)
        p = a / self.target_sum      # because a = S·p
        avg = np.dot(p, g_a)
        g_y = self.target_sum * p * (g_a - avg)

        return float(Ltau), g_y

# Initial‑vector generation --------------------------------------------------

def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Return a short list of diverse start vectors (already normalised)."""
    out = []

    # uniform
    out.append(np.full(n, target_sum / n))

    # Dirichlet (γ(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    out.append(g / g.sum() * target_sum)

    # random exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    out.append(x / x.sum() * target_sum)

    # reversed exponential
    out.append(x[::-1] / x[::-1].sum() * target_sum)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    out.append(hann / hann.sum() * target_sum)

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    out.append(black / black.sum() * target_sum)

    # noisy uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    out.append(pert / pert.sum() * target_sum)

    # triangular (peak in centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    out.append(tri / tri.sum() * target_sum)

    return out

# L‑BFGS optimisation ------------------------------------------------------


def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Optimise a start vector with L‑BFGS on the smooth max‑convolution.
    Returns the best vector (exact max‑convolution) discovered in this run.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector
    eps = 1e-12
    y = np.log(a0 + eps)

    # create objective object
    obj = LBFGSObjective(n, target_sum)

    # initialise temperature from the start vector
    a_start = _softmax_to_a(y, target_sum)
    conv0 = _conv_full(a_start, obj.Nfft)
    tau = max(0.2 * conv0.max(), 1e-6)      # a fairly generous start temperature
    obj.set_tau(tau)

    # L‑BFGS stage loop – continuation with halving τ
    while time.time() < deadline:
        # each L‑BFGS call performs at most ``maxiter`` iterations
        res = optimize.minimize(
            obj,
            y,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 400, "ftol": 1e-12, "gtol": 1e-12},
        )
        y = res.x

        # exact evaluation of the current vector
        a = _softmax_to_a(y, target_sum)
        conv = _conv_full(a, obj.Nfft)
        cur_max = float(conv.max())

        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[L‑BFGS n={n:5d}] new best max = {best_score:.6f}")

        # halve temperature – drives the smooth surrogate toward the true max
        tau *= 0.5
        if tau < 1e-12:
            break
        obj.set_tau(tau)

        if time.time() >= deadline:
            break

    return best_seq, best_score

# Targeted local refinement -------------------------------------------------

def _targeted_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    cur_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that always moves mass away from indices that contribute
    most to the current maximal convolution (gradient‑guided).
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_best = cur_score

    # FFT size for cheap convolutions
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule
    max_steps = 150_000
    step = 0
    while time.time() < deadline and step < max_steps:
        # compute gradient of the exact max (no smoothing)
        # – obtain the shift(s) attaining the max
        A_fft = np.fft.rfft(cur_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        max_val = conv.max()
        mask = conv >= max_val - 1e-12
        w = mask.astype(float)
        if w.sum() == 0:
            w[0] = 1.0
        else:
            w /= w.sum()

        # weighted reverse convolution gives the gradient w.r.t. a
        rev = _weighted_rev(cur_seq, w, Nfft)
        grad = 2.0 * rev                      # ∇_a max

        # indices with largest positive gradient (mass should leave) and
        # smallest/negative gradient (mass can be received)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))

        if i == j:
            # fallback to random pair if gradient has no direction
            i, j = rng.integers(0, n, size=2, endpoint=False)

        if cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a tiny fraction (1‑3 %) of the source mass
        frac = rng.uniform(0.01, 0.03)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate new max convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_max = float(conv.max())

        # deterministic improvement
        if new_max < cur_best - 1e-12:
            cur_seq = new_seq
            cur_best = new_max
            print(f"[refine] improved max → {cur_best:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_best, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_max - cur_best) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_best = new_max
        step += 1

    return cur_seq, cur_best

        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vecs.append(interp)

        # give each start a modest slice of the remaining budget
        remaining = deadline - time.time()
        per_start = remaining / (len(init_vecs) * 1.5)      # leave room for refinement
        per_start = max(5.0, per_start)                    # at least a few seconds

        for idx, a0 in enumerate(init_vecs):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_start
            print(f"[run] length={n}, start {idx + 1}/{len(init_vecs)}")
            best_seq, best_score = _lbfgs_optimize(
                a0=a0,
                n=n,
                deadline=sub_dead,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Final targeted stochastic refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _targeted_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            cur_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        print("[fallback] using a uniform sequence")

    # convert to plain list for the harness
    final_list = best_seq.tolist()
    # sanity check (optional, not required for the final answer)
    try:
        final_score = evaluate_sequence(final_list)  # pylint: disable=E0602
    except Exception:
        final_score = float("nan")
    print(f"[done] best evaluate_sequence = {final_score:.6f}, length = {len(final_list)}")
    return final_list
