def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Candidate generators
def _candidate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _candidate_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Dirichlet with equal concentration α (α>0)."""
    return np.random.dirichlet(np.full(n, alpha))


def _candidate_powerlaw(n: int, alpha: float) -> np.ndarray:
    """Shape ∝ ((i+1)*(n-i))**α  (α may be negative)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
    if shape.sum() <= 0:
        return _candidate_uniform(n)
    return shape / shape.sum()


def _candidate_exponential(n: int, rate: float) -> np.ndarray:
    """Shape ∝ exp(-rate * i) + exp(-rate * (n-1-i)). Symmetric decay."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    if shape.sum() == 0:
        return _candidate_uniform(n)
    return shape / shape.sum()


def _candidate_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑U shape: a,b∈(0,1) give a heavy U‑distribution."""
    x = np.linspace(0.0, 1.0, n)
    # avoid the singularities at the borders by adding a tiny epsilon
    eps = 1e-12
    shape = np.power(x + eps, a - 1.0) * np.power(1.0 - x + eps, b - 1.0)
    if shape.sum() == 0 or np.isnan(shape).any():
        return _candidate_uniform(n)
    return shape / shape.sum()


def _candidate_gaussian_mix(n: int, sigma: float, centers: tuple) -> np.ndarray:
    """Mixture of a few Gaussians anchored at given normalized centres."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    shape = np.zeros_like(x)
    for c in centers:
        shape += np.exp(-0.5 * ((x - c) / sigma) ** 2)
    if shape.sum() == 0:
        return _candidate_uniform(n)
    return shape / shape.sum()


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        return _candidate_uniform(new_n)
    return new / tot


def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of starting points (all sum to 1)."""
    cand: list[np.ndarray] = []

    # 1) Warm‑starts
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    if hist_seq is not None:
        tmp = hist_seq[:n] if hist_seq.shape[0] >= n else np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot > 1e-12:
            cand.append(tmp / tot)

    # 2) Simple uniform
    cand.append(_candidate_uniform(n))

    # 3) Dirichlet with various concentrations
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(_candidate_dirichlet(n, alpha))

    # 4) Power‑law edge shapes
    for alp in (-2.0, -1.0, -0.5, -0.2):
        cand.append(_candidate_powerlaw(n, alp))

    # 5) Exponential decays
    for rate in (0.001, 0.005, 0.02, 0.1):
        cand.append(_candidate_exponential(n, rate))

    # 6) Beta‑U shapes (U‑distribution)
    for a_ in (0.2, 0.4, 0.6):
        for b_ in (0.2, 0.4, 0.6):
            cand.append(_candidate_beta(n, a_, b_))

    # 7) Multi‑peak Gaussians (edges + centre)
    for sigma in (0.02, 0.05, 0.10):
        cand.append(_candidate_gaussian_mix(n, sigma, (0.0, 0.5, 1.0)))

    # 8) Random mixture of two U‑shapes
    for _ in range(2):
        a_ = np.random.rand() * 0.8 + 0.1
        b_ = np.random.rand() * 0.8 + 0.1
        cand.append(_candidate_beta(n, a_, b_))

    # Shuffle and truncate to a manageable size (≈15)
    np.random.shuffle(cand)
    return cand[:15]

# Optimisers
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # start with a moderate learning‑rate; decay slowly
    lr0 = 0.07
    lr = lr0

    # temperature for the smooth‑max (λ). Start low and grow rapidly.
    lam = 0.5
    lam_factor = 1.07

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # schedule
        lam = min(lam * lam_factor, 1e7)
        # gentle learning‑rate decay
        lr = lr0 * (1.0 / (1.0 + 1e-4 * t))

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # random perturbation if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient (exponentiated‑gradient) optimisation."""
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val

    last_improve = time.time()
    eta0 = 0.7

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # active set (allow a small tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # decreasing step‑size
        eta = eta0 / sqrt(it)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = _candidate_dirichlet(n, 1.0)
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb with occasional uphill moves."""
    a = a0.copy()
    cur_val = (
        _objective(a, n, fft_len_conv) if init_val is None else init_val
    )
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move a modest fraction of mass from i1 to i2
        eps = np.random.rand() * 0.04 * a[i1]  # ≤4 % of the source mass
        if eps <= 0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-12:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 100 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # simulated‑annealing style uphill acceptance
            prob = np.exp(-(new_val - cur_val) * 10.0)  # temperature ≈0.1
            if np.random.rand() < prob:
                a = new_a
                cur_val = new_val

        if time.time() - last_improve > 6.0:
            # give up early if no progress
            break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Retrieve a historical solution if available.
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Lengths – start with moderate sizes where optimisation is cheap, end
    # with the largest allowed size for a final refinement.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths (next power‑of‑two) for each n.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # correlation needs conv_len + n - 1 ≤ 3·n, safe to use 2*conv_len
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # ------------------------------------------------------------------
        # Generate a diverse pool of warm‑starts.
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Adam iterations are scaled roughly inversely with n.
        max_adam_iters = int(20000 * 256 / n) + 1200

        for idx, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact sub‑gradient ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline, init_val=val_adam
            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=val_sub
            )

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={idx}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run refinement on the best sequence (if any time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n, 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            fft_len_c,
            fft_len_x,
            deadline,
            max_iters=int(1.5 * (20000 * 256 / n) + 3000),
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required.
    return best_seq.tolist()
