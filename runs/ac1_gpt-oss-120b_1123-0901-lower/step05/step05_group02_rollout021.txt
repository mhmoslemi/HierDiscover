def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    lengths = [8192, 16384, 4096, 2048]  # prioritize larger

    for n in lengths:
        if time.time() > deadline: break
        print(f"Optimising length n={n}")
        conv_pow2 = 1 << ((2*n - 2).bit_length())
        corr_pow2 = 1 << ((3*n - 3).bit_length())
        # candidates
        candidates = []
        # upsample
        if best_seq is not None:
            up = upsample(np.asarray(best_seq), n)
            candidates.append(up)
        # uniform
        candidates.append(np.full(n, 1.0/n))
        # random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))
        for conc in [0.2, 0.5]:
            candidates.append(np.random.dirichlet(np.full(n, conc)))
        # exponential
        for gamma in [0.001, 0.002, 0.005, 0.01]:
            i = np.arange(n)
            shape = np.exp(-gamma*i) + np.exp(-gamma*(n-1-i))
            shape /= shape.sum()
            candidates.append(shape)
        # power law
        idx = np.arange(n)+1
        prod = idx*(n - idx + 1)
        for alpha in np.linspace(-2, -0.2, 8):
            shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)
        # tent
        left = np.linspace(0, 1, n//2, endpoint=False)
        tent = np.concatenate([left, left[::-1]])
        tent = np.maximum(tent,0)
        if len(tent) < n:
            tent = np.append(tent, 0)
        tent /= tent.sum()
        candidates.append(tent)

        # iterate over candidates
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline: break
            # Ensure positivity and non-zero sum
            if np.allclose(a0, 0):
                continue
            a0 = np.maximum(a0, 0)
            sum0 = a0.sum()
            if sum0 < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
                sum0 = a0.sum()
            a0 = a0 / sum0

            # Adam phase
            # We'll allocate some ratio of remaining time
            # compute local deadline per candidate: maybe time left / (remaining candidates)
            local_deadline = deadline - 0.01
            a_opt = adam_optimize(a0, n, conv_pow2, corr_pow2, local_deadline)
            # compute objective
            val = compute_obj(a_opt, n)
            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"New best from Adam cand {cid} n={n}: {val:.6f}")

            # EG phase
            a_opt2 = eg_optimize(a_opt, n, conv_pow2, corr_pow2, local_deadline)
            val2 = compute_obj(a_opt2, n)
            if val2 < best_val:
                best_val = val2
                best_seq = a_opt2.copy()
                print(f"New best from EG cand {cid} n={n}: {val2:.6f}")

    # final fallback
    if best_seq is None:
        best_seq = np.full(2048, 1.0/2048)
    print(f"Final best value: {best_val}")
    return best_seq.tolist()

def adam_optimize(a0, n, conv_len_pow2, corr_len_pow2, deadline):
    a = a0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr=0.05; lam=0.3; lam_factor=1.02
    last_improve = time.time()
    max_iters = 200000  # but break on time
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv
        conv = _full_convolution(a, conv_len_pow2)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_len_pow2)
        # backprop softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1-beta1)*grad_log
        v = beta2 * v + (1-beta2)*(grad_log**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat)+eps)
        lam = min(lam * lam_factor, 1e6)

        # improvement check
        if t % 500 == 0:
            val = 2.0 * n * max_c
            if val < compute_obj(a, n):
                last_improve = time.time()
    return a

def eg_optimize(a0, n, conv_len_pow2, corr_len_pow2, deadline):
    a = a0.copy()
    eta=0.5
    last_improve = time.time()
    max_iters = 5000
    for it in range(max_iters):
        if time.time() >= deadline: break
        conv = _full_convolution(a, conv_len_pow2)
        max_c = conv.max()
        # Determine active indices near max
        # We consider those within tolerance relative to max
        atol = 1e-12
        active = (conv >= max_c - atol)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
        else:
            w = np.ones_like(conv) / conv.size
        grad_a = 2.0 * _cross_correlation(w, a, corr_len_pow2)
        a = a * np.exp(-eta * grad_a)
        # renorm
        sum_a = a.sum()
        if sum_a <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
            sum_a = a.sum()
        a /= sum_a

        # update best perhaps
    return a

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.shape[0]-1]

def _cross_correlation(w, a, fft_len):
    # cross-correlation (w ★ a) length len(a)
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:

    x_old = np.linspace(0, 1, old_n)

    return np.interp(x_new, x_old, seq)

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    lengths = [16384, 8192, 4096, 2048]
    for n in lengths:
        if time.time() > deadline:
            break
        conv_len = 2*n - 1
        fft_len = 1 << ((conv_len - 1).bit_length())
        # Parameter t range:
        t_vals = np.linspace(0.5, 4.0, 350)   # enough resolution
        for t in t_vals:
            if time.time() > deadline:
                break
            gamma = t / n
            a = np.exp(-gamma * np.arange(n, dtype=np.float64))
            s = a.sum()
            if s < 1e-12:
                continue
            a /= s
            conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
            max_c = conv.max()
            val = 2.0 * n * max_c
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"new best n={n} t={t:.4f} val={val:.6f}")
    # After scanning gamma, optionally perform fine-tuning using Adam or EG starting from best_seq for short time.
    # Could run a few hundred iterations for improvement.
    if best_seq is not None and time.time() < deadline:
        # small Adam refine for e.g., 5000 iterations
        a = best_seq.copy()
        n = len(a)
        conv_len = 2*n-1
        fft_len = 1 << ((conv_len-1).bit_length())
        corr_len = 1 << ((conv_len + n -2).bit_length())
        logits = np.log(a + 1e-15)
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        lr = 0.05
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lam = 0.5
        lam_factor = 1.01
        max_iters = 5000
        for it in range(1, max_iters+1):
            if time.time() > deadline: break
            mx = logits.max()
            exp_x = np.exp(logits - mx)
            a = exp_x / exp_x.sum()
            conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
            max_c = conv.max()
            w = np.exp(lam * (conv - max_c))
            total = w.sum()
            if total == 0:
                w = np.full_like(conv, 1.0/conv.size)
            else:
                w /= total
            # gradient cross-correlation
            rev_a = a[::-1]
            fw = np.fft.rfft(w, corr_len)
            fr = np.fft.rfft(rev_a, corr_len)
            corr = np.fft.irfft(fw * fr, corr_len)
            grad_a = 2.0 * corr[n-1:n-1+n]
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)
            m = beta1*m + (1-beta1)*grad_log
            v = beta2*v + (1-beta2)*(grad_log**2)
            m_hat = m/(1-beta1**it)
            v_hat = v/(1-beta2**it)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
            lam = min(lam*lam_factor, 1e6)
            # evaluate after each...
            val = 2.0*n*max_c
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"refine rng iter={it} val={val:.6f}")
    # final fallback uniform
    if best_seq is None:
        n=2048
        best_seq = np.full(n, 1.0/n)
        best_val = 2.0 * n * (1.0/n)
    print(f"final best value {best_val}")
    return best_seq.tolist()

    v = np.zeros_like(logits)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lam=0.5
    lam_factor=1.02
    cnt=0
    best_local_val = float('inf')
    best_local_a = a.copy()
    while time.time() < deadline and cnt < max_iters:
        cnt += 1
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

import time
import numpy as np

def _next_pow2(x):
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)].
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective(a, n, fft_len):
    """Exact objective 2·n·max(a*a) for a already normalized (sum=1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=20000, lr=0.05):
    """Adam optimisation on a smooth‑max surrogate."""
    a = a0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = 0.5
    lam_factor = 1.02
    best_val = float('inf')
    best_a = a.copy()
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of the smooth‑max
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = min(lam * lam_factor, 1e6)
        # track best
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val

def _eg_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                 max_iters=5000, eta=0.55):
    """Exponentiated‑gradient optimisation on the exact max."""
    a = a0.copy()
    best_val = float('inf')
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (all indices attaining the max up to tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val

    The algorithm consists of two stages:
    1. A dense grid search over the one‑parameter family

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    best_seq = None
    best_val = float('inf')

    # try a few lengths – larger n gives a better asymptotic bound
    lengths = [16384, 8192, 4096, 2048]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(2 * n - 1 + n - 1)

        # Theory suggests  t≈2 .  Scan a generous interval.

        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


def _eg_optimize(a0: np.ndarray,
                 n: int,
                 conv_fft_len: int,
                 corr_fft_len: int,
                 deadline: float,
                 max_iters: int = 5000,
                 eta: float = 0.55) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient optimisation on the exact max‑convolution."""
    a = a0.copy()
    best_val = float("inf")
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # uniform weight on the active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Return a non‑negative sequence that (approximately) minimises
    F(a) = 2·n·max(a*a) / (sum(a)²).

    The search consists of a dense grid over the exponential family
    a_i ∝ exp(-γ·i) (asymmetric exponential) followed by a short
    Adam‑based refinement.  The routine respects the time budget and
    returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Larger n gives a better asymptotic bound; try a handful of powers of two.
    lengths = [16384, 8192, 4096, 2048]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(2 * n - 1 + n - 1)

        # 1️⃣ Grid search over t = γ·n (theoretical optimum around t≈2)
        t_vals = np.linspace(0.5, 4.0, 350)            # resolution ≈0.01
        for t_val in t_vals:
            if time.time() > deadline:
                break
            gamma = t_val / n
            a = np.exp(-gamma * np.arange(n, dtype=np.float64))
            s = a.sum()
            if s < 1e-12:
                continue
            a /= s
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[grid] n={n} t={t_val:.3f}  val={val:.6f}")

        # 2️⃣ Short Adam refinement on the current best of this length
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            a_ref, val_ref = _adam_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=5000,
                lr=0.04,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[adam‑refine] n={n}  val={val_ref:.6f}")

    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        n = 2048
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * (1.0 / n)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
