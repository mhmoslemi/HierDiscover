Propose a step‑function (non‑negative list) that tries to minimise

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “a*a” is the linear convolution (numpy.convolve).

The optimiser works in three stages for several lengths (2048, 4096,
8192, 16384):

1.   A large and diverse pool of *initial* candidates is built:
     * the best solution found so far (up‑sampled when the length grows)
     * random Dirichlet samples (concentrations 0.2,0.5,0.8,1.0)
     * monotone decreasing shapes (exponential, power‑law, linear drop)
     * two‑sided exponentials (the previous baseline)
     * sparse “k‑point” patterns (k≈√(2n)) with equal weights
     * historic solution if present.

2.   Each candidate is refined by an Adam optimiser on a *smooth‑max*
     surrogate of the convolution peak.  The temperature (λ) is raised
     slowly so that the optimisation first sees a smooth landscape and
     then focuses on the true maximum.

3.   The Adam‑output is further polished by a sub‑gradient ascent on the
     exact max (exponentiated‑gradient, i.e. “multiplicative” updates)
     and finally by a cheap simulated‑annealing / hill‑climbing pass
     that makes tiny random mass transfers while keeping the vector on
     the probability simplex.

All operations are performed with FFT‑based convolutions, which makes
the algorithm fast enough to explore many candidates within the 1000 s
budget.  The best sequence encountered (as a plain Python list) is
returned.
"""

def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max.  Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _obj_val(a: np.ndarray) -> float:
    """
    Exact objective value for a probability vector (sum of a = 1).
    """
    n = a.shape[0]
    conv = _conv_fft(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


def _make_candidate_shapes(n: int, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a heterogeneous list of initial candidates for a given length.
    """
    candidates: list[np.ndarray] = []

    # 1) up‑sample the current best (if any)
    if best_seq is not None:
        old_n = best_seq.shape[0]
        if old_n != n:
            # linear interpolation onto the new grid
            old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            cand = np.interp(new_x, old_x, best_seq)
        else:
            cand = best_seq.copy()
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        candidates.append(cand)

    # 2) pure Dirichlet draws
    candidates.append(np.random.dirichlet(np.ones(n)))                 # conc = 1
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 3) monotone decreasing exponential (one‑sided)
    for gamma_factor in (0.5, 1.0, 2.0, 4.0):
        gamma = gamma_factor / n
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i)
        shape /= shape.sum()
        candidates.append(shape)

    # 4) power‑law decreasing a_i ∝ (i+1)^(-p)
    for p in (0.5, 1.0, 1.5, 2.0):
        shape = np.power(np.arange(1, n + 1, dtype=np.float64), -p)
        shape /= shape.sum()
        candidates.append(shape)

    # 5) linear drop to zero after a cutoff
    for cut_frac in (0.25, 0.5, 0.75):
        cut = int(cut_frac * n)
        shape = np.concatenate([np.linspace(1.0, 0.0, cut, endpoint=False),
                                np.zeros(n - cut)])
        shape = np.clip(shape, 0.0, None)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 6) two‑sided exponential (the old baseline)
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 7) sparse “k‑point” patterns (equal mass on k positions)
    k_target = int(np.sqrt(2 * n))               # theoretical upper bound
    for mult in (0.8, 1.0, 1.2):
        k = max(2, int(mult * k_target))
        for _ in range(3):                       # a few random draws
            pos = np.random.choice(n, k, replace=False)
            shape = np.zeros(n, dtype=np.float64)
            shape[pos] = 1.0 / k
            candidates.append(shape)

    return candidates


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_len: int,
                   corr_len: int,
                   deadline: float,
                   init_best_val: float,
                   init_best_seq: np.ndarray | None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of the max‑convolution.
    Returns the best (a, value) found during the run.
    """
    # normalise to a probability vector
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.2          # start temperature
    lam_factor = 1.02

    best_a = a.copy()
    best_val = init_best_val
    last_improve = time.time()

    # iteration budget – a bit larger than the original code
    max_iters = int(35000 * 1024 / n) + 8000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (linear in a)
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature – slowly approach the true max
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum is 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new best {val:.6f}")

        # occasional random shake to escape plateaus
        if time.time() - last_improve > 9.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_len: int,
                      corr_len: int,
                      deadline: float,
                      init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Exact max‑convolution sub‑gradient phase using an exponentiated‑gradient
    (multiplicative) update.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = init_best_val
    last_improve = time.time()
    max_iters = 8000

    eta = 0.55

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        # uniform weight on the active indices
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # guard against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing / hill‑climbing pass.
    Small random mass transfers are attempted; only improvements are kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = init_best_val
    last_improve = time.time()

    # typical amplitude (as fraction of a_i) – decays with time
    sigma = 0.02
    it = 0
    while time.time() < deadline and it < 30000:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        delta = sigma * a[i] * (np.random.rand() - 0.5)  # can be positive or negative
        # transfer delta from i to j, staying non‑negative
        new_ai = a[i] - delta
        new_aj = a[j] + delta
        if new_ai < 0 or new_aj < 0:
            continue
        a[i], a[j] = new_ai, new_aj

        # renormalise (the total mass is unchanged up to round‑off)
        # but keep the small numerical drift in check
        a /= a.sum()

        val = _obj_val(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} hill‑climb it={it}] new best {val:.6f}")

        # occasionally shakedown sigma
        if it % 2000 == 0:
            sigma *= 0.9
            sigma = max(sigma, 0.001)

        if time.time() - last_improve > 12.0:
            break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # lengths to explore (ascending – we can up‑sample the current best)
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n)             # definitely enough for cross‑corr

        candidates = _make_candidate_shapes(n, best_seq)

        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam phase -------------------------------------------------
            a_adam, val_adam = _adam_optimize(cand,
                                               n,
                                               conv_len,
                                               corr_len,
                                               deadline,
                                               best_val,
                                               best_seq)

            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()

            # ---- Exact sub‑gradient phase -----------------------------------
            a_sub, val_sub = _subgrad_optimize(a_adam,
                                               n,
                                               conv_len,
                                               corr_len,
                                               deadline,
                                               best_val)

            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()

            # ---- Light hill‑climb on the current global best ----------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_hc, val_hc = _hill_climb_optimize(best_seq,
                                                    n,
                                                    deadline,
                                                    best_val)
                if val_hc < best_val:
                    best_val = val_hc
                    best_seq = a_hc.copy()

        # ------------------------------------------------------------------
        # Final fine‑tuning (Adam with a slower temperature schedule) for the
        # *global* best of this length.
        if best_seq is not None and best_seq.shape[0] == n:
            print(f"[n={n}] final fine‑tuning (val={best_val:.6f})")
            a_fine, val_fine = _adam_optimize(best_seq,
                                              n,
                                              conv_len,
                                              corr_len,
                                              deadline,
                                              best_val,
                                              best_seq)
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
