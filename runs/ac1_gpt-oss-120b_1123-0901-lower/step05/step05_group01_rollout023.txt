"""
Propose a low‑value step‑function sequence.

The algorithm follows three phases:

1️⃣  **Parametric exploration** – generate a diverse, symmetric set of
    candidates using classic window functions (Hann, Hamming, Blackman,
    Kaiser) and simple analytic families (power‑law edges, double‑exponential,
    inverse‑root).  All candidates are normalised to sum = 1 and evaluated
    with a fast FFT‑based convolution.  The best few are kept.

2️⃣  **Refinement** – each retained candidate is improved by a mirror‑symmetric
    exponentiated‑gradient (sub‑gradient) optimiser that works on the exact
    max‑convolution, followed by a cheap hill‑climber that swaps small amounts
    of mass between indices.  Symmetry (a[i] ≈ a[n‑1‑i]) is enforced after every
    update because the problem is invariant under reversal.

3️⃣  **Late‑stage annealing** – a lightweight simulated‑annealing walk
    (random Dirichlet redistributions on a small random subset of coordinates)
    is performed around the current global best until the time budget expires.
    This allows the search to escape tiny plateaus and possibly improve the
    bound further.

All heavy work is done with FFT‑based convolutions, making a single objective
evaluation cheap even for n ≈ 16 k.  The routine respects the 1000 s budget,
prints progress, and finally returns the best sequence as a plain list of
non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import i0  # Kaiser window


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
def _next_pow2(x: int) -> int:
    return 1 << (int(np.ceil(np.log2(x))))

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Symmetric candidate families (all return a normalised np.ndarray)

def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _shape_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return w / w.sum()


def _shape_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def _shape_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return w / w.sum()


def _shape_kaiser(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    w = i0(np.pi * beta * np.sqrt(1.0 - t * t))
    # Guard against beta=0 (i0(0)=1) – result is already uniform
    return w / w.sum()


def _shape_power_law(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    prod = i * (n - i + 1.0)
    # power with negative alpha blows up at edges – safe with where
    w = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
    if w.sum() == 0:
        w = np.ones_like(w)
    return w / w.sum()


def _shape_two_exp(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i ** 2) + np.exp(-gamma * (n - 1 - i) ** 2)
    return w / w.sum()


def _shape_inv_root(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    w = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
    return w / w.sum()


def _shape_three_gaussians(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
    return peaks / peaks.sum()


def _shape_edge_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-sigma * i) + np.exp(-sigma * (n - 1 - i))
    return w / w.sum()


# ----------------------------------------------------------------------
# Optimisation helpers --------------------------------------------------

def _subgrad_optimize_sym(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (sub‑gradient) optimisation on the exact max.
    The iterate is forced to stay symmetric: a = (a + rev(a))/2.
    """
    a = a0.copy()
    a /= a.sum()
    # enforce symmetry from the start
    a = (a + a[::-1]) * 0.5

    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        # enforce symmetry on the gradient as well
        grad_a = (grad_a + grad_a[::-1]) * 0.5

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise and re‑symmetrise
        a /= a.sum()
        a = (a + a[::-1]) * 0.5

        cur_val = _objective(a, n, fft_len)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[sym‑subgrad it={it}] new best {cur_val:.6f}")

        # stop if stagnating for a while
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small random fraction of mass between two
    coordinates and accept if the objective improves.  Symmetrisation after each
    accepted move keeps the sequence mirror‑symmetric.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # pick two distinct indices (randomly)
        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # transfer up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps

        # keep non‑negativity (tiny negative tolerated)
        if new_a[i1] < -1e-15:
            continue

        # renormalise and re‑symmetrise
        new_a /= new_a.sum()
        new_a = (new_a + new_a[::-1]) * 0.5

        new_val = _objective(new_a, n, fft_len)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            # occasional logging
            if it % 100 == 0:
                print(f"[hillclimb it={it}] new best {new_val:.6f}")
        else:
            # small probability to accept uphill move – helps escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

        # abort if no progress for a while
        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found (plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – the largest allowed length gives the best asymptotic bound.
    n = 16384
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    print(f"\n=== Starting search, length n={n}, FFT length={fft_len} ===")

    # ------------------------------------------------------------------
    # 1️⃣  Parametric candidate generation
    param_candidates = []

    # classic windows (all symmetric by construction)
    param_candidates.append(_shape_uniform(n))
    param_candidates.append(_shape_hann(n))
    param_candidates.append(_shape_hamming(n))
    param_candidates.append(_shape_blackman(n))

    # Kaiser window – sweep β from 0 (uniform) up to 12 (strong taper)
    for beta in np.linspace(0.0, 12.0, 13):
        param_candidates.append(_shape_kaiser(n, beta))

    # power‑law edge shapes – fine grid
    for alpha in np.linspace(-2.0, -0.2, 37):
        param_candidates.append(_shape_power_law(n, alpha))

    # double‑exponential edge shapes
    for gamma in np.linspace(0.05, 2.0, 30):
        param_candidates.append(_shape_two_exp(n, gamma))

    # inverse‑root edge shapes
    for alpha in (0.4, 0.6, 0.8, 1.0):
        param_candidates.append(_shape_inv_root(n, alpha))

    # three‑peak Gaussian (mid‑range widths)
    for sigma in (0.02, 0.05, 0.07, 0.10):
        param_candidates.append(_shape_three_gaussians(n, sigma))

    # edge‑biased narrow Gaussian
    for sigma in (0.02, 0.04, 0.06):
        param_candidates.append(_shape_edge_gaussian(n, sigma))

    # ------------------------------------------------------------------
    # Evaluate all parametric candidates, retain the best few
    evaluated = []
    best_val = float("inf")
    best_seq = None

    for idx, cand in enumerate(param_candidates):
        if time.time() > deadline:
            break
        val = _objective(cand, n, fft_len)
        evaluated.append((cand, val))
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if (idx + 1) % 10 == 0:
            print(f"[param] evaluated {idx + 1}/{len(param_candidates)} candidates")

    # keep the top K parametric solutions
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_candidates = [c for c, _ in evaluated[:top_k]]

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet start‑points (cheap, many)
    random_keep = 6
    random_evals = []

    max_random = 2500   # safety‑capped number of Dirichlet draws
    for i in range(max_random):
        if time.time() > deadline:
            break
        cand = np.random.dirichlet(np.ones(n))
        val = _objective(cand, n, fft_len)
        random_evals.append((cand, val))
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if (i + 1) % 500 == 0:
            print(f"[random] generated {i + 1}/{max_random} candidates")

    random_evals.sort(key=lambda x: x[1])
    top_candidates.extend([c for c, _ in random_evals[:random_keep]])

    # ------------------------------------------------------------------
    # 3️⃣  Refinement of the elite candidates
    print(f"\n=== Refinement phase – {len(top_candidates)} elite candidates ===")
    for cid, a_start in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"\n--- Refining candidate {cid} (initial value {_objective(a_start, n, fft_len):.6f}) ---")

        # sub‑gradient with symmetry
        a_sub, val_sub = _subgrad_optimize_sym(
            a_start, n, fft_len, deadline,
            init_val=_objective(a_start, n, fft_len),
            max_iters=5000,
        )

        # cheap hill‑climb
        a_hc, val_hc = _hill_climb_optimize(
            a_sub, n, fft_len, deadline,
            init_val=val_sub,
            max_iters=3000,
        )

        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"*** Global new best from elite {cid}: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Late‑stage simulated annealing around the current best
    if best_seq is not None and time.time() < deadline:
        print("\n=== Final annealing refinement ===")
        a = best_seq.copy()
        cur_val = best_val
        best_a = a.copy()
        best_val_local = cur_val

        # annealing parameters
        T0 = best_val * 0.12
        T = T0
        decay = 0.996

        while time.time() < deadline:
            # pick a small random subset and reshuffle its mass
            sub_idx = np.random.choice(n, size=7, replace=False)
            total = a[sub_idx].sum()
            # Dirichlet redistribution inside the subset
            new_weights = np.random.dirichlet(np.ones(len(sub_idx))) * total
            a_new = a.copy()
            a_new[sub_idx] = new_weights
            a_new = (a_new + a_new[::-1]) * 0.5          # keep symmetry
            a_new /= a_new.sum()

            new_val = _objective(a_new, n, fft_len)
            delta = new_val - cur_val

            if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
                a = a_new
                cur_val = new_val
                if cur_val < best_val_local:
                    best_val_local = cur_val
                    best_a = a.copy()
            # temperature schedule
            T *= decay
            if T < 1e-9:
                T = T0

        if best_val_local < best_val:
            best_val = best_val_local
            best_seq = best_a.copy()
            print(f"+++ Annealing improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (unlikely) return a uniform vector
    if best_seq is None:
        best_seq = _shape_uniform(n)

    print(f"\n=== Search completed – best value {best_val:.6f} ===")
    return best_seq.tolist()
