def project_onto_simplex(v, s):
    """Project onto the simplex {x >= 0, sum x = s}."""
    # v is 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_descent(a0, target_sum, max_iter, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = np.max(np.convolve(a, a))
    # step size schedule
    base_eta = target_sum * 0.1  # maybe
    for it in range(1, max_iter+1):
        if time.time() > deadline:
            break
        # compute conv
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        # gradient
        grad = np.zeros_like(a)
        start = max(0, max_idx - n + 1)
        end = min(max_idx, n-1)
        if start <= end:
            # subset length
            idx = np.arange(start, end+1)
            grad[idx] = 2.0 * a[max_idx - idx]
        # step size
        eta = base_eta / np.sqrt(it)
        new_a = a - eta * grad
        # projection onto simplex
        new_a = project_onto_simplex(new_a, target_sum)
        new_max = np.max(np.convolve(new_a, new_a))
        if new_max < best_val:
            best_val = new_max
            best_a = new_a.copy()
        # accept improvement else keep previous a
        if new_max < np.max(np.convolve(a, a)):
            a = new_a
        else:
            # occasional random jitter to escape local minima
            if it % 200 == 0 and np.random.rand() < 0.2:
                jitter = np.random.randn(n) * 1e-3
                a = project_onto_simplex(a + jitter, target_sum)
        # Optionally break if no improvement for many steps
    return best_a, best_val

def lp_refine(a, target_sum, n_steps, t, deadline):
    n = a.shape[0]
    L = 2*n - 1
    # Preconstruct A matrix once (coeff depends on current a? Actually rows have f entries which change each iteration,
    # So need update each step.

        M = np.max(np.convolve(a, a))
        # build A matrix from current a
        A = np.zeros((L, n), dtype=np.float64)
        # fill
        for j in range(n):
            A[j:j+n, j] = a
        # solve LP
        c = -np.ones(n)  # maximize sum(g) -> min -sum(g)
        bounds = [(0, None)] * n
        b = np.full(L, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # fallback: break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # scale to target sum
        g = g * (target_sum / sum_g)
        # convex combination
        a = (1 - t) * a + t * g
        # projection onto simplex not needed since sum remains target_sum (since sum of mixture is target_sum)
        # but numerical errors might accumulate: reproject occasionally
        if step_idx % 5 == 0:
            a = project_onto_simplex(a, target_sum)

"""Hybrid subgradient + linear‑programming refinement.

* Randomly sample a length n and a power‑law / exponential start.
* Scale the vector to the canonical sum √(2n).
* Run a simple sub‑gradient descent on the max‑convolution term.
* Perform a few Frank‑Wolfe‑type LP refinements that maximise the sum of a
  partner vector under the current max‑convolution bound.
* Keep the best sequence found and finish with a short Adam optimisation
  (soft‑max parameterisation) as a final polish.

The routine respects the time budget and returns a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# Helper functions -------------------------------------------------------

def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive random profile of length n."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:   # uniform
        shape = np.ones_like(idx)

    shape += rng.random(n) * 1e-6
    return np.maximum(shape, 1e-12)


def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    # Implementation of the algorithm by
    # Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a * a)[k] (full linear convolution)."""
    conv = np.convolve(a, a)
    return float(conv.max())


def subgradient_descent(a0: np.ndarray, target_sum: float,
                        max_iter: int, deadline: float,
                        rng: np.random.Generator):
    """Simple sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = max_convolution(a)

    base_eta = target_sum * 0.2   # tuning parameter
    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        conv = np.convolve(a, a)
        k = int(np.argmax(conv))

        # gradient of the selected convolution entry
        grad = np.zeros_like(a)
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        if start <= end:
            idx = np.arange(start, end + 1)
            grad[idx] = 2.0 * a[k - idx]

        # decreasing step size
        eta = base_eta / np.sqrt(it)

        a_candidate = a - eta * grad
        a_candidate = project_onto_simplex(a_candidate, target_sum)

        val = max_convolution(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()

        # accept only if we improve the current point, otherwise keep old a
        cur_val = max_convolution(a)
        if val < cur_val:
            a = a_candidate
        else:
            # occasional random jitter to avoid stagnation
            if it % 250 == 0 and rng.random() < 0.15:
                jitter = rng.normal(size=n) * 1e-3
                a = project_onto_simplex(a + jitter, target_sum)

    return best_a, best_val


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build matrix A with shape (2n‑1, n) such that A @ g = f * g (full conv).
    Entry A[k, j] = f[k‑j] if 0 ≤ k‑j < n else 0.
    """
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    # each column j contributes f shifted down by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refinement(a: np.ndarray, target_sum: float,
                  n_steps: int, step_size: float,
                  deadline: float):
    """
    Frank‑Wolfe‑type refinement:
    given current a and its max‑conv M, solve
        max   sum(g)
        s.t.  a * g ≤ M , g ≥ 0
    then update a ← (1‑t) a + t·ĝ, where ĝ is the scaled solution.
    """
    n = a.shape[0]
    L = 2 * n - 1

    for i in range(n_steps):
        if time.time() > deadline:
            break

        M = max_convolution(a)

        # build constraint matrix for the current a
        A = build_convolution_matrix(a)

        c = -np.ones(n)                              # maximise sum(g)
        b = np.full(L, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # give up the refinement if LP fails
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # scale to the canonical sum
        g = g * (target_sum / sum_g)

        # convex combination
        a = (1.0 - step_size) * a + step_size * g

        # numerical drift – re‑project every few iterations
        if (i + 1) % 3 == 0:
            a = project_onto_simplex(a, target_sum)

    return a

# Soft‑max Adam optimiser (used for a final polish)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real inputs)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max representation a = S·softmax(x)."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # map logits → simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] Adam n={n} local best = {max_conv:.8f}")

        # smooth‑max weights
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # gradient w.r.t logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        alpha *= 0.9997

        if iteration - last_improve > 2000:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

        shape = np.concatenate([left, right])
    else:   # uniform
        shape = np.ones_like(idx)

    shape += rng.random(n) * 1e-6
    return np.maximum(shape, 1e-12)


def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}.
    Implementation follows the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray) -> float:
    """Return max_k (a * a)[k] (full linear convolution)."""
    conv = np.convolve(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient descent on max(conv(a,a))
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray, target_sum: float,
                        max_iter: int, deadline: float,
                        rng: np.random.Generator):
    """Simple sub‑gradient descent that tries to lower max(a*a)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = max_convolution(a)

    base_eta = target_sum * 0.2   # step‑size scaling
    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        conv = np.convolve(a, a)
        k = int(np.argmax(conv))

        # gradient of the selected convolution entry
        grad = np.zeros_like(a)
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        if start <= end:
            idx = np.arange(start, end + 1)
            grad[idx] = 2.0 * a[k - idx]

        eta = base_eta / np.sqrt(it)
        a_candidate = a - eta * grad
        a_candidate = project_onto_simplex(a_candidate, target_sum)

        val = max_convolution(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()

        cur_val = max_convolution(a)
        if val < cur_val:
            a = a_candidate
        else:
            # occasional jitter to escape plateaus
            if it % 250 == 0 and rng.random() < 0.15:
                jitter = rng.normal(size=n) * 1e-3
                a = project_onto_simplex(a + jitter, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Build convolution matrix for the LP (A @ g = f * g)
# ----------------------------------------------------------------------
def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct A ∈ ℝ^{(2n‑1)×n} with A[k, j] = f[k‑j] if 0≤k‑j<n else 0.
    Then A @ g gives the full linear convolution f*g.
    """
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Frank‑Wolfe‑type LP refinement
# ----------------------------------------------------------------------
def lp_refinement(a: np.ndarray, target_sum: float,
                  n_steps: int, step_size: float,
                  deadline: float) -> np.ndarray:
    """
    Given a, solve repeatedly:
        maximise   Σ g_j
        subject to  a * g ≤ M   (M = max(a*a)),   g ≥ 0
    and update a ← (1‑t) a + t·ĝ where ĝ is the scaled solution.
    """
    n = a.shape[0]
    L = 2 * n - 1

    for i in range(n_steps):
        if time.time() > deadline:
            break

        M = max_convolution(a)
        A = build_convolution_matrix(a)

        c = -np.ones(n)                # maximise Σ g_j  →  minimise -Σ g_j
        b = np.full(L, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Scale g to have the canonical sum √(2 n)
        g = g * (target_sum / sum_g)

        a = (1.0 - step_size) * a + step_size * g

        if (i + 1) % 3 == 0:
            a = project_onto_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser (used for a final polish)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real inputs)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max representation a = S·softmax(x)."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # map logits → simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] Adam n={n} local best = {max_conv:.8f}")

        # smooth‑max weights (log‑sum‑exp surrogate)
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # gradient w.r.t. logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        alpha *= 0.9997

        if iteration - last_improve > 2000:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function that minimises
    ``2 * n * max(conv(a,a)) / (sum(a)**2)``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from any previously discovered construction.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start global = {sc:.8f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main stochastic search: many restarts → sub‑gradient → LP.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # pick a length (larger early on, smaller later for speed)
        if remaining > 300:
            n = rng.integers(1800, 3000 + 1)
        elif remaining > 120:
            n = rng.integers(1300, 2500 + 1)
        else:
            n = rng.integers(800, 1800 + 1)

        # --------------------------------------------------------------
        # Random initial shape, scaled to the canonical sum.
        # --------------------------------------------------------------
        init_shape = _initial_shape_random(n, rng)
        target_sum = np.sqrt(2.0 * n)
        init_shape = init_shape / init_shape.sum() * target_sum

        # --------------------------------------------------------------
        # Sub‑gradient phase (about half of the remaining slice).
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + remaining * 0.45)
        a_sub, _ = subgradient_descent(init_shape, target_sum,
                                       max_iter=5000,
                                       deadline=sub_deadline,
                                       rng=rng)

        # --------------------------------------------------------------
        # LP refinement phase (the rest of the slice).
        # --------------------------------------------------------------
        if remaining > 200:
            lp_steps, lp_t = 12, 0.20
        elif remaining > 80:
            lp_steps, lp_t = 8, 0.25
        else:
            lp_steps, lp_t = 5, 0.30

        a_refined = lp_refinement(a_sub, target_sum, lp_steps, lp_t, deadline)

        # --------------------------------------------------------------
        # Evaluate the candidate.
        # --------------------------------------------------------------
        try:
            score = evaluate_sequence(a_refined.tolist())
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = a_refined.copy()
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            if best_score < 1.503:      # target reached → stop early
                break

    # ------------------------------------------------------------------
    # Final polishing with Adam if time permits.
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 12.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = best_seq * (target_sum / best_seq.sum())
        x0 = np.log(a + 1e-12)
        seq_final, score_final = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            deadline,
            rng,
            init_tau_factor=0.12,
            min_tau_factor=1e-5,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.94,
            max_iter=None,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = seq_final.copy()
            print(f"[{time.time():.2f}] Adam refinement improved best to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Fallback: uniform construction (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the outer harness.
    return best_seq.tolist()
