def greedy_sidon_set(n):
    used = np.zeros(2*n, dtype=bool)
    indices = []
    for i in range(n):
        ok = True
        for j in indices:
            s = i + j
            if used[s]:
                ok = False
                break
        # also check self-sum
        if ok and used[i+i]:
            ok = False
        if ok:
            indices.append(i)
            for j in indices: # Now indices includes i, but we need add sums of i with all (including i)
                used[i+j] = True
    return np.array(indices, dtype=int)

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')

    lengths = [1024, 2048]   # maybe also 4096 if time allows

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} with LP Conditional Gradient ===")
        # initial distributions to try: uniform, warm start from height_sequence_1 (if size matches), random.
        init_strategies = []
        # Uniform
        init_strategies.append(np.full(n, 1.0/n))
        # Warm start if height_sequence_1 exists and length matches or longer
        if 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n-cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() > 0.01:
                cand = cand / cand.sum()
            else:
                cand = np.random.dirichlet(np.ones(n))
            init_strategies.append(cand)
        # Random Dirichlet
        for _ in range(2):
            init_strategies.append(np.random.dirichlet(np.ones(n)))

        for idx, f in enumerate(init_strategies):
            if time.time() > deadline:
                break
            f = f.astype(np.float64)
            # ensure sum 1
            f = f / f.sum()
            max_iters = 30
            step = 0.15  # convex combination step
            for it in range(max_iters):
                if time.time() > deadline:
                    break
                conv = np.convolve(f, f)
                M = conv.max()
                cur_val = 2.0 * n * M
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = f.copy()
                    print(f"[n={n} init={idx} iter={it}] new best {cur_val:.6f}")

                # Build constraint matrix A where A @ b <= M
                # Using Toeplitz shift of f
                rows = 2 * n - 1
                A = np.zeros((rows, n), dtype=np.float64)
                for i in range(n):
                    A[i:i+n, i] = f  # shift f down by i
                # Solve LP: maximize sum(b) s.t. b>=0, A @ b <= M
                # linprog minimizes c^T b, so use c = -1
                c = -np.ones(n, dtype=np.float64)

                ub = np.full(rows, M, dtype=np.float64)
                # bounds for each variable
                bounds = [(0, None)] * n
                # solve
                res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                              method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})

                    print(f"[n={n} init={idx} iter={it}] LP failed status {res.status}")

                f /= f.sum()

"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm is a conditional‑gradient (Frank–Wolfe) scheme where, at each
iteration, we solve a linear program that tells us how to redistribute the
mass without increasing the current worst convolution value.  The LP
maximises the total mass under the linear constraints

    (f * g)_k ≤ M  for every k,

where * is the ordinary linear convolution, f is the current vector,
g the unknown candidate and M = max(f*f).  The solution g is normalised
to have sum = 1 and the update

    f ← (1‑t)·f + t·g

is performed (t ≈ 0.15).  The process is repeated a modest number of times
and the best vector found (including a cheap random “mass‑shuffling”
refinement) is returned.

The routine works for several lengths (1024 and 2048 out of the box) and
automatically respects the global 1000 s wall‑time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k].
    This is a Toeplitz matrix: each column i contains f shifted by i rows.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f
    return A


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with a low value of the evaluation function.
    The search is time‑bounded: it stops when `budget_s` seconds have elapsed.
    Returns a plain Python list of non‑negative floats that sum to 1.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # a tiny safety margin

    best_seq = None      # best sequence (numpy array, sum = 1)
    best_val = float("inf")

    # Lengths to try – 1024 gives the best trade‑off between quality and speed.
    lengths = [1024, 2048]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} (LP conditional‑gradient) ===")

        # ------------------------------------------------------------------
        # 1) Build a list of initial candidates (uniform, warm‑start, random)
        # ------------------------------------------------------------------
        init_candidates = []

        # uniform start
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # warm start from a previously discovered sequence if available
        if "height_sequence_1" in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() >= 0.01:
                cand /= cand.sum()
            else:
                cand = np.random.dirichlet(np.ones(n))
            init_candidates.append(cand)

        # a few random Dirichlet starts
        for _ in range(2):
            init_candidates.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------
        # 2) Run the conditional‑gradient loop for each start
        # ------------------------------------------------------------------
        for start_id, f in enumerate(init_candidates):
            if time.time() > deadline:
                break

            f = np.maximum(f, 0.0)
            f /= f.sum()                      # enforce sum = 1

            max_iters = 30                    # number of CG steps
            step = 0.15                       # convex‑combination weight

            for it in range(max_iters):
                if time.time() > deadline:
                    break

                # current objective value (exact, no smoothing)
                conv = np.convolve(f, f)
                M = conv.max()
                cur_val = 2.0 * n * M          # sum(f) is 1, so denominator = 1

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = f.copy()
                    print(f"[n={n} start={start_id} it={it:02d}] new best {cur_val:.6f}")

                # Build linear constraints (f * g)_k ≤ M
                A = _build_convolution_matrix(f)

                # Linear program:
                #   maximise 1ᵀ g   (i.e. minimise –1ᵀ g)
                #   s.t.   A @ g ≤ M
                #          g ≥ 0
                c = -np.ones(n, dtype=np.float64)          # minimise –∑gᵢ
                ub = np.full(2 * n - 1, M, dtype=np.float64)
                bounds = [(0.0, None)] * n

                # Solve with SciPy's HiGHS backend (fast for moderate size)
                lp_res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                                 method="highs", options={"presolve": True})

                if lp_res.status != 0:           # non‑optimal – abort this start
                    print(f"[n={n} start={start_id} it={it}] LP failed (status {lp_res.status})")
                    break

                g_raw = lp_res.x
                sum_g = g_raw.sum()

                # If the LP cannot increase the total mass beyond the current 1.0,
                # we have reached a stationary point.
                if sum_g <= 1.0 + 1e-12:
                    break

                # Normalise to keep the sum equal to one
                g = g_raw / sum_g

                # Frank–Wolfe update
                f = (1.0 - step) * f + step * g
                # Numerical clean‑up
                f = np.maximum(f, 0.0)
                f /= f.sum()

            # ------------------------------------------------------------------
            # 3) Cheap greedy refinement – move a tiny amount of mass between
            #    two positions if it improves the objective.
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                for _ in range(3000):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a random fraction (20‑80 %) of a small portion of mass
                    eps_move = min(a_local[i], 0.008) * (0.2 + 0.6 * np.random.rand())
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = np.convolve(a_new, a_new)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")

    # ----------------------------------------------------------------------
    # Fallback: uniform vector if nothing was found (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        print("Fallback to uniform distribution.")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
