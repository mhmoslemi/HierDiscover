"""
Propose a non‑negative step‑function that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

Algorithm
---------
* Hierarchical refinement: start with a modest size (256) and successively
  double the length (512, 1024, 2048, 4096).  After each level the best
  sequence is interpolated to the next size and refined again.
* At a given length we optimise in the *log‑softmax* domain with Adam.
  The non‑smooth max is replaced by a temperature‑controlled log‑sum‑exp
  (smooth‑max) which yields a well‑behaved gradient.
* Temperature is annealed linearly from 1e‑1 down to 1e‑6.
* After the Adam phase a cheap greedy “mass‑shuffle’’ pass moves a small
  fraction of mass from coordinates that contribute most to the current
  maximal convolution coefficient to coordinates that contribute the
  least.  This often reduces the true max without many extra convolutions.
* The whole search respects the 1000 s wall‑time budget; the loop
  continuously checks the deadline.
* If a previous construction ``height_sequence_1`` is available it is used
  as the initial guess for the smallest problem.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑smoothed maximum of `conv` and its softmax weights.
    Returns (smooth_max, weights) where
        weights[i] = ∂smooth_max/∂conv[i].
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def evaluate_obj(a: np.ndarray) -> float:
    """Exact objective as used by `evaluate_sequence` (no safety checks)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def optimise_length(
    n: int,
    a_init: np.ndarray,
    deadline: float,
    *,
    max_iters: int = 80_000,
    shake_interval: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` starting from ``a_init``.
    Returns (best_sequence, best_value) observed before the deadline.
    """
    # Parameters for Adam on the logits (log‑softmax parametrisation)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_initial = 0.12
    lr = lr_initial
    decay_every = 10_000

    # Smooth‑max temperature schedule
    tau_initial = 0.12
    tau_final = 5e-7

    # FFT lengths (chosen once for the whole run)
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))

    # Initialise logits from the starting probability vector
    x = np.log(np.maximum(a_init, 1e-15))

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = np.inf
    last_improve = time.time()

    t = 0
    while time.time() < deadline and t < max_iters:
        # ----- soft‑max to obtain a probability vector (sum = 1) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()                     # a_i ≥ 0, Σ a_i = 1

        # ----- convolution and smooth‑max -----
        conv = full_convolution(a, fft_len=fft_len_conv)
        tau = max(tau_final, tau_initial * (1.0 - t / max_iters))
        smooth, weights = smooth_max_and_weights(conv, tau)

        # ----- exact objective (for bookkeeping) -----
        cur_val = 2.0 * n * conv.max()               # denominator = 1 because Σa = 1
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} it={t:06d}] new best {cur_val:.6f}")

        # ----- gradient of smooth‑max w.r.t. a -----
        # grad_a = ∂smooth/∂a = 2·(weights ★ a)
        grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

        # back‑propagation through the soft‑max
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # learning‑rate decay
        if t % decay_every == 0:
            lr *= 0.7

        # ----- stagnation handling (random shake) -----
        if time.time() - last_improve > shake_interval:
            # Small jitter in log‑space
            x += (np.random.rand(n) - 0.5) * 6e-3
            last_improve = time.time()
            print(f"[n={n} it={t}] shake (stagnation)")

    # ------------------------------------------------------------------
    # Cheap greedy refinement: move a tiny amount of mass from the most
    # “dangerous’’ coordinates (those that contribute most to the current
    # maximal convolution coefficient) to the least dangerous ones.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = a_init.copy()
        best_val = evaluate_obj(best_seq)

    a_local = best_seq.copy()
    conv_local = full_convolution(a_local, fft_len=fft_len_conv)
    for _ in range(3_000):
        if time.time() >= deadline:
            break
        # index of the current maximal convolution entry
        idx_max = int(np.argmax(conv_local))

        # build the contribution vector g where
        #   g[j] = a_local[idx_max - j]   if the partner index is inside [0,n)
        g = np.zeros_like(a_local)
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        partners = idx_max - np.arange(lo, hi + 1)
        g[lo : hi + 1] = a_local[partners]

        # source = coordinate with largest contribution, target = random other
        src = int(np.argmax(g))
        tgt = np.random.randint(0, n - 1)
        if tgt >= src:
            tgt += 1

        # move a small fraction from src to tgt
        eps_move = min(a_local[src] * 0.05, 0.02) * np.random.rand()
        if eps_move <= 0:
            continue
        a_new = a_local.copy()
        a_new[src] -= eps_move
        a_new[tgt] += eps_move

        conv_new = full_convolution(a_new, fft_len=fft_len_conv)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_seq = a_new.copy()
            a_local = a_new
            conv_local = conv_new
            print(f"[n={n} local] improved to {val_new:.6f}")

    return best_seq, best_val

            a0 = np.random.dirichlet(np.ones(size_list[0]))
        else:
            # truncate / pad to the smallest size
            if cand.shape[0] >= size_list[0]:
                a0 = cand[: size_list[0]]
            else:
                a0 = np.pad(cand, (0, size_list[0] - cand.shape[0]))
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-8:
                a0 = np.random.dirichlet(np.ones(size_list[0]))
            else:
                a0 = a0 / a0.sum()
    else:
        a0 = np.random.dirichlet(np.ones(size_list[0]))

    # ------------------------------------------------------------------
    # Hierarchical loop: optimise at a given length, then interpolate to
    # the next length and continue.
    # ------------------------------------------------------------------
    for n in size_list:
        if time.time() >= deadline:
            break

        # allocate a rough budget per level (remaining time split equally)
        remaining = deadline - time.time()
        # give more iterations to larger problems, but keep a floor
        if n <= 512:
            max_iters = 120_000
        elif n <= 1024:
            max_iters = 90_000
        elif n <= 2048:
            max_iters = 60_000
        else:
            max_iters = 40_000

        # make sure we do not over‑run the global deadline
        max_iters = int(min(max_iters, remaining / 0.0005))  # very rough bound

        print(f"\n=== Optimising length n={n} (max_iters={max_iters}) ===")
        best_n_seq, best_n_val = optimise_length(
            n, a_init=a0, deadline=deadline, max_iters=max_iters
        )
        if best_n_seq is not None and best_n_val < best_val - 1e-12:
            best_seq = best_n_seq.copy()
            best_val = best_n_val
            print(f"[global] improved best to {best_val:.6f} (n={n})")

        # prepare for the next level – linear interpolation + tiny jitter
        if n != size_list[-1]:
            n_next = size_list[size_list.index(n) + 1]
            # interpolate to the larger grid
            xs = np.linspace(0, 1, n, endpoint=False)
            xs_next = np.linspace(0, 1, n_next, endpoint=False)
            a0 = np.interp(xs_next, xs, best_n_seq)
            # keep it a proper probability vector
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0:
                a0 = np.random.dirichlet(np.ones(n_next))
            else:
                a0 = a0 / a0.sum()
            # add a very small random perturbation to avoid exact symmetry
            a0 += np.random.randn(n_next) * 1e-4
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
        best_val = evaluate_obj(best_seq)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
