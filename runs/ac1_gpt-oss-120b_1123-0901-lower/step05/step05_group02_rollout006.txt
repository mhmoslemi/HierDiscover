"""
Propose a step‑function (non‑negative list) that approximately minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “a*a” is the linear convolution (numpy.convolve).

The algorithm works in three stages:

1. **Coarse search on small sizes** – a few random Dirichlet/ exponential
   candidates are optimised with Adam on a smooth‑max surrogate.  The best
   sequence found for size *n* is upsampled (linear interpolation) to the
   next larger size.

2. **Fine optimisation on the current size** – the upsampled sequence is
   refined with many more Adam iterations, a slower temperature schedule and
   occasional random “shakes” to escape plateaus.

3. **Exponentiated‑gradient (sub‑gradient) polishing** – after the smooth‑max
   phase we switch to the exact max (no smoothing) and run a short
   exponentiated‑gradient loop, which is well‑suited for the simplex
   constraints.

All work is limited by the supplied ``budget_s`` (default 1000 s).  The
function returns the best sequence found (as a plain Python list) before the
deadline expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len`` (real FFT).
    Returns a vector of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    The result has the same length as ``a``.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int,
    *,
    lr: float = 0.06,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lam_start: float = 0.3,
    lam_factor: float = 1.015,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate of max(convolution).
    Returns the best (a, value) seen during the run.
    """
    # initialise soft‑max parameters (logits) from a0 (already normalised)
    logits = np.log(a0 + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = a0.copy()
    best_val = np.inf
    last_improve = time.time()
    lam = lam_start

    for t in range(1, max_iters + 1):
        # -------------------------------------------------
        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution of current a
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()

        # smooth‑max weights (softmax over the convolution values)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate (linear because conv is linear)
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # back‑prop through the outer soft‑max (mirror‑descent on the simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule – slowly turn the smooth‑max into a true max
        lam = min(lam * lam_factor, 1e6)

        # Exact objective for this a (sum == 1 → denominator = 1)
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # debugging output – will be seen when the function runs
            print(
                f"[n={n} Adam iter={t:05d}] new best {val:.6f}"
            )

        # occasional random shake if we are stuck for >8 s
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _exponentiated_gradient(
    a_start: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 5000,
    eta: float = 0.55,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient phase on the exact max (no smoothing) using
    exponentiated‑gradient updates (mirror descent w.r.t. KL).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = np.inf
    last_improve = time.time()

    for it in range(max_iters):
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        # active set of indices achieving the maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # exponentiated‑gradient step (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                         # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest list of starting points for a given length ``n``.
    ``previous`` may be a best‑so‑far vector of a smaller size – it will be
    interpolated to the current length.
    """
    cand = []

    # 1️⃣  Interpolated previous best (if any)
    if previous is not None:
        # linear interpolation to new size
        x_old = np.linspace(0, 1, previous.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n, endpoint=False)
        interp = np.interp(x_new, x_old, previous)
        interp = np.maximum(interp, 0.0)
        if interp.sum() < 0.01:
            interp = np.random.dirichlet(np.ones(n))
        else:
            interp = interp / interp.sum()
        cand.append(interp)

    # 2️⃣  Pure Dirichlet (concentration 1 – fairly uniform)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 3️⃣  Spiky Dirichlet (concentration < 1)
    for conc in (0.2, 0.5, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4️⃣  Two‑sided exponential shape
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # 5️⃣  Power‑law edge‑concentrated shape
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # 6️⃣  Slightly smoothed “tent’’ (linear rise‑then‑fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    tent = np.maximum(tent, 0)
    tent /= tent.sum()
    cand.append(tent)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The routine respects the given time budget and returns the best
    sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None          # numpy array
    best_val = float("inf")

    # Lengths that we will visit – start small, finish large.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths (next power of two) for speed.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _next_pow2(conv_len)
        # cross‑correlation needs size ≥ conv_len + n – 1 ≤ 3·n
        fft_len_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    # iterate over lengths, up‑sampling the previous best when possible
    prev_best_small = None
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        candidates = _initial_candidates(n, prev_best_small)

        # --------------------------------------------------------------
        # Optimise each candidate with Adam + EG
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a0 = cand / cand.sum()

            # Adam phase – allocate iterations roughly proportional to size
            # (more iterations for larger n, but keep total time reasonable)
            base_iters = 25000            # tuned for n≈2048
            max_iters = int(base_iters * 2048 / n) + 5000

            a_adam, val_adam = _adam_optimize(
                a0,
                n,
                conv_len,
                corr_len,
                deadline,
                max_iters,
                lr=0.06,
                lam_start=0.3,
                lam_factor=1.015,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam best now {best_val:.6f}")

            # Exponentiated‑gradient (sub‑gradient) polishing
            a_eg, val_eg = _exponentiated_gradient(
                a_adam,
                n,
                conv_len,
                corr_len,
                deadline,
                max_iters=5000,
                eta=0.55,
            )
            if val_eg < best_val:
                best_val = val_eg
                best_seq = a_eg.copy()
                print(f"[n={n} cand={cid}] EG best now {best_val:.6f}")

        # ------------------------------------------------------------------
        # Optional fine‑tuning of the *global* best for this length
        if best_seq is not None and best_seq.shape[0] == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a_fine, val_fine = _adam_optimize(
                best_seq,
                n,
                conv_len,
                corr_len,
                deadline,
                max_iters=int(40000 * 2048 / n) + 5000,
                lr=0.04,
                lam_start=0.5,
                lam_factor=1.01,
            )
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n} fine‑tune] new best {best_val:.6f}")

        # Store the best vector of this size to initialise the next (larger)
        # length – we simply keep it as‑is; the next iteration will interpolate.
        prev_best_small = best_seq.copy() if best_seq is not None else None

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        print("Fallback to uniform distribution.")
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as a plain Python list as required by the external evaluator
    return best_seq.tolist()
