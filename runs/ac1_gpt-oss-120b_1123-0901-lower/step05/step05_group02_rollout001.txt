"""
Propose a step‑function (non‑negative list) that approximately minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “a*a” is the linear convolution (numpy.convolve).

Algorithm
---------

* The search works on several lengths (2048 … 16384).
* For each length a low‑dimensional representation based on the first K
  DCT‑II basis functions is used:
        logits = B @ p          (B has shape (n, K))
        a      = softmax(logits)   → Σa = 1, a_i ≥ 0
  This reduces the optimisation problem from n variables to K≈√n.
* Adam optimisation is performed on the parameter vector `p` using a
  smooth‑max surrogate of the max‑convolution.  The temperature λ is
  annealed (slowly increased) so that the surrogate converges to the real
  max.
* Several initialisations for `p` are tried:
    – random Gaussian,
    – projection of simple analytic shapes (power‑law, two‑sided exponential)
      onto the DCT basis,
    – a warm start from the best full‑dimensional sequence discovered so
      far (if its length matches).
* After the low‑dimensional stage the best full sequence is further
  refined with a short Adam run on the original n‑dimensional logits
  (identical to the approach used in the reference code).
* The whole procedure respects a user‑supplied time budget (default 1000 s).

The function returns the best sequence found as a plain Python list of
floats (all entries non‑negative, automatically normalised).
"""

import time
import numpy as np


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _dct_basis(n: int, k: int) -> np.ndarray:
    """
    Return the first ``k`` DCT‑II basis vectors of length ``n`` as a matrix
    of shape (n, k).  Column ``j`` (0‑based) equals
        cos(π·(i+0.5)·j / n)   for i = 0..n‑1.
    No orthogonal normalisation is applied – the basis is only used as a
    flexible low‑dimensional parametrisation.
    """
    i = np.arange(n, dtype=np.float64)[:, None]          # (n,1)
    j = np.arange(k, dtype=np.float64)[None, :]          # (1,k)
    B = np.cos(np.pi * (i + 0.5) * j / n)                # (n,k)
    return B


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths for the search – larger n gives more expressive power,
    # but also larger FFT cost.
    lengths = [2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for convolution and cross‑correlation.
    fft_conv = {}
    fft_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # cross‑correlation needs length at least conv_len + n - 1 ≤ 3·n
        fft_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # Helper that evaluates the true objective (sum(a) is forced to 1
    # by the soft‑max, but we keep the denominator for safety).
    def _objective(a: np.ndarray, n: int) -> float:
        conv = _full_convolution(a, fft_conv[n])
        max_c = conv.max()
        s = a.sum()
        if s < 1e-12:
            return float("inf")
        return 2.0 * n * max_c / (s ** 2)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_conv[n]
        corr_len = fft_corr[n]

        # Number of low‑dimensional basis functions.
        k = min(200, max(32, int(np.sqrt(n))))
        print(f"Using {k} DCT basis functions for n={n}")

        # Build the basis matrix B (shape n×k) and its transpose.
        B = _dct_basis(n, k)            # (n, k)
        BT = B.T                         # (k, n)

        # ------------------------------------------------------------------
        # Build a list of initial vectors for the low‑dimensional parameters p.
        p_inits = []

        # 1) Random Gaussian start.
        p_inits.append(np.random.randn(k))

        # 2) Warm start from the current best full‑dimensional sequence
        #    (project its logits onto the DCT basis, if lengths match).
        if best_seq is not None and len(best_seq) == n:
            a_best = np.asarray(best_seq, dtype=np.float64)
            logits_best = np.log(a_best + 1e-15)
            # Least‑squares projection: B @ p ≈ logits_best
            p_proj, *_ = np.linalg.lstsq(B, logits_best, rcond=None)
            p_inits.append(p_proj)

        # 3) Analytic shapes projected onto the basis.
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)

        # Power‑law (edge‑concentrated) shapes.
        for alpha in np.linspace(-2.0, -0.1, 7):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() == 0:
                continue
            shape = shape / shape.sum()
            logits = np.log(shape + 1e-15)
            p_shape, *_ = np.linalg.lstsq(B, logits, rcond=None)
            p_inits.append(p_shape)

        # Two‑sided exponential shapes.
        for gamma in [0.0005, 0.001, 0.002, 0.005]:
            exp_shape = np.exp(-gamma * idx) + np.exp(-gamma * (n - 1 - idx))
            exp_shape /= exp_shape.sum()
            logits = np.log(exp_shape + 1e-15)
            p_exp, *_ = np.linalg.lstsq(B, logits, rcond=None)
            p_inits.append(p_exp)

        # ------------------------------------------------------------------
        # Adam optimiser on the low‑dimensional parameters.
        for cid, p0 in enumerate(p_inits):
            if time.time() > deadline:
                break

            p = p0.astype(np.float64).copy()
            m = np.zeros_like(p)
            v = np.zeros_like(p)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04          # learning rate for p
            lam = 0.1          # start temperature (smooth‑max)
            lam_factor = 1.02

            # iteration budget grows for larger n but stays modest.
            max_iters = int(30000 * 1024 / n) + 8000
            last_improve = time.time()

            for t in range(1, max_iters + 1):
                # ----- soft‑max over logits -----
                logits = B @ p                       # (n,)
                mx = logits.max()
                exp_logits = np.exp(logits - mx)
                a = exp_logits / exp_logits.sum()    # probability vector, Σa = 1

                # ----- convolution & smooth‑max weights -----
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                # ----- gradient wrt a (smooth‑max surrogate) -----
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                # ----- back‑prop through soft‑max -----
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)        # shape (n,)

                # ----- gradient wrt low‑dim parameters p -----
                grad_p = BT @ grad_log               # (k,)

                # ----- Adam update on p -----
                m = beta1 * m + (1.0 - beta1) * grad_p
                v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                p -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule -----
                lam = min(lam * lam_factor, 1e6)

                # ----- evaluate true objective -----
                val = 2.0 * n * max_c   # denominator = 1 because Σa = 1
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t:05d}] new best {val:.6f}")

                # ----- stagnation handling -----
                if time.time() - last_improve > 8.0:
                    # small random shake in parameter space
                    p += (np.random.rand(k) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ------------------------------------------------------------------
            # Optional short sub‑gradient (exponentiated‑gradient) phase on
            # the exact max (helps escape tiny flat regions).
            a = a / a.sum()
            eta = 0.55
            sub_last_improve = time.time()
            for sub_it in range(2000):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a /= s

                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 6.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Fine‑tune the *global* best for the current length (full‑dimensional).
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning best sequence (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.03
            lam = 0.4
            lam_factor = 1.01
            last_improve = time.time()
            fine_iters = int(20000 * 1024 / n) + 4000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    # Small random shake before convergence
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
