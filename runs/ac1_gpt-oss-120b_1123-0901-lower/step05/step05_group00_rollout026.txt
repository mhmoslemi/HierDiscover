"""
propose_candidate – gradient‑based optimiser for the max‑convolution bound.

The objective to minimise is

    F(a) = 2·n·max_k (a * a)[k] / (sum a)²

Because the expression is homogeneous we may restrict ourselves to the
probability simplex (sum a = 1, a_i ≥ 0).  The algorithm works on a cascade of
lengths, starts from a rich pool of analytically‑motivated shapes (uniform,
U‑shaped beta, edge‑concentrated power laws, double‑Gaussian, sine‑squared,
Dirichlet draws, historic best, etc.) and refines each start with two stages:

1. **Smooth‑max Adam** – the true max is replaced by a log‑sum‑exp
   surrogate with a temperature λ that is slowly increased.  The vector is
   parametrised by logits → soft‑max; a symmetric restriction
   (logits ↦ (logits+logits[::-1])/2) halves the effective dimensionality.
2. **Exact sub‑gradient** – after the surrogate has converged λ is large,
   the true max is used and an exponentiated‑gradient (mirror‑descent) step
   drives the iterate down the ℓ∞‑cone.

The schedule allocates many more iterations to the larger problems:
`max_iters = int(20000 * 1024 / n) + 5000`.  A tiny random shake is applied
if a candidate stagnates for several seconds.  The best sequence seen across
all lengths is returned as a plain Python list.

No external files are accessed; only numpy and scipy (for an optional LP
refinement on moderate sizes) are used.  The routine respects the supplied
wall‑clock budget (default 1000 s) and therefore never exceeds the
evaluation time limit.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _symmetrise_logits(logits: np.ndarray) -> np.ndarray:
    """Enforce exact symmetry of the underlying probability vector."""
    return (logits + logits[::-1]) / 2.0


def _symmetrise_prob(p: np.ndarray) -> np.ndarray:
    """Enforce symmetry on a probability vector."""
    return (p + p[::-1]) / 2.0


def _generate_candidates(n: int, best_seq, rng) -> list:
    """Return a diverse list of initial probability vectors of length n."""
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # warm‑start with exact same length, if available
    if best_seq is not None and len(best_seq) == n:
        candidates.append(np.asarray(best_seq, dtype=np.float64))

    # up‑sample a shorter best (linear interpolation)
    if best_seq is not None and len(best_seq) < n:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        up = up / up.sum()
        candidates.append(up)

    # historic global variable, if present
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        h = h / h.sum()
        candidates.append(h)

    # random Dirichlet
    candidates.append(rng.dirichlet(np.ones(n)))

    # symmetric Beta family (U‑shaped when alpha<1)
    for _ in range(2):
        alpha = rng.uniform(0.1, 0.8)          # shape parameter
        x = np.linspace(0.0, 1.0, n) + 1e-12
        shape = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
        shape = shape / shape.sum()
        shape = _symmetrise_prob(shape)
        candidates.append(shape)

    # edge‑concentrated power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in rng.uniform(-3.0, -0.5, size=3):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            shape = _symmetrise_prob(shape)
            candidates.append(shape)

    # two‑Gaussian edge shapes (centred at the ends)
    i = np.arange(n, dtype=np.float64)
    for sigma in rng.uniform(0.1, 1.5, size=3):
        shape = np.exp(-sigma * (i ** 2)) + np.exp(-sigma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        shape = _symmetrise_prob(shape)
        candidates.append(shape)

    # smooth sine‑squared bump (central hill)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    shape = shape / shape.sum()
    shape = _symmetrise_prob(shape)
    candidates.append(shape)

    # random “spike” patterns (few non‑zero entries)
    for _ in range(2):
        a = np.zeros(n, dtype=np.float64)
        k = rng.integers(2, 10)                     # number of spikes
        spots = rng.choice(n, size=k, replace=False)
        a[spots] = rng.random(k) + 0.1
        a += 1e-12                                   # avoid exact zeros in soft‑max
        a = a / a.sum()
        a = _symmetrise_prob(a)
        candidates.append(a)

    # ensure proper normalisation (numerical safety)
    normed = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        if c.sum() == 0:
            c = np.full(n, 1.0 / n)
        else:
            c = c / c.sum()
        normed.append(c)
    return normed


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_val: float,
    rng,
) -> tuple:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_a, best_val) seen for this start.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # temperature (smooth‑max sharpening)
    lam = 0.5
    lam_factor = 1.02               # a little faster sharpening than before

    last_improve = time.time()
    # iteration budget – larger n receives fewer raw steps, but we bump the base constant
    max_iters = int(20000 * 1024 / n) + 5000

    best_local_a = a.copy()
    best_local_val = best_val

    for it in range(1, max_iters + 1):
        # soft‑max → simplex (symmetric by construction)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a of the smooth‑max
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑prop through the soft‑max (mirror‑descent geometry)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # force exact symmetry of the underlying distribution
        logits = _symmetrise_logits(logits)

        # anneal temperature
        lam = min(lam * lam_factor, 1e7)

        # exact bound (normalised ⇒ denominator = 1)
        val = 2.0 * n * max_c

        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()
            # report improvement
            print(f"[n={n} Adam it={it:05d}] new best {val:.6f}")

        # stagnation handling – small random jitter on logits
        if time.time() - last_improve > 5.0:
            logits += (rng.random(n) - 0.5) * 1e-3
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_local_a, best_local_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple:
    """
    Exact sub‑gradient (exponentiated‑gradient) refinement on the true max.
    Returns (best_a, best_val) for this start.
    """
    a = a0.copy()
    eta = 0.6
    sub_last = time.time()
    best_local_a = a.copy()
    best_local_val = best_val

    for sub_it in range(4000):
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
        a = a * np.exp(-eta * grad_a)

        # numerical safety
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # re‑impose symmetry (helps stability)
        a = _symmetrise_prob(a)

        # evaluate exact objective
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            sub_last = time.time()
            print(f"[n={n} Subgrad it={sub_it:04d}] new best {val:.6f}")

        if time.time() - sub_last > 6.0:
            break
        if time.time() > deadline:
            break

    return best_local_a, best_local_val


def _lp_refine(f: np.ndarray, n: int, deadline: float, rng) -> np.ndarray:
    """
    One linear‑program improvement step (maximise sum(g) subject to (f*g) ≤ M).
    Returns a new candidate (or the original f if the LP fails or time runs out).
    """
    # time guard – building the dense matrix can be costly
    if time.time() > deadline - 2.0:
        return f

    conv = _full_convolution(f, fft_len=_next_pow2(2 * n - 1))
    M = conv.max()

    rows = 2 * n - 1
    # dense Toeplitz matrix A where (A @ g)[k] = (f * g)[k]
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # linprog minimises   cᵀx    →  minimise –1ᵀg  (i.e. maximise 1ᵀg)
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(rows, M, dtype=np.float64)

    # bounds: g_i ≥ 0
    bounds = [(0, None) for _ in range(n)]

    # solve – using the high‑performance "highs" method
    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, deadline - time.time() - 0.5)},
    )
    if not res.success:
        return f

    g = res.x
    # mix with the current f (step size 0.5) and renormalise
    new = 0.5 * f + 0.5 * g
    new = np.maximum(new, 0.0)
    if new.sum() == 0:
        new = np.full(n, 1.0 / n)
    else:
        new = new / new.sum()
    new = _symmetrise_prob(new)
    return new


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability distribution) that
    minimises the bound 2·n·max(a*a)/(sum a)².
    The routine respects the wall‑clock budget and returns the best sequence
    discovered as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # Process a cascade of lengths – small problems are cheap and provide
    # good warm‑starts for the larger instances.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # --------------------------------------------------------------
        # Build a diverse pool of starting points
        candidates = _generate_candidates(n, best_seq, rng)

        # --------------------------------------------------------------
        # Optimise each candidate independently
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam on smooth‑max surrogate ------------------------
            a_opt, val = _adam_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_val,
                rng,
            )
            if val < best_val:
                best_val, best_seq = val, a_opt.copy()

            # ---- Exact sub‑gradient refinement ----------------------
            a_opt, val = _subgradient_refine(
                a_opt,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_val,
            )
            if val < best_val:
                best_val, best_seq = val, a_opt.copy()

            # occasionally try a tiny LP‑based improvement on moderate sizes
            if n <= 4096 and time.time() < deadline - 1.0:
                a_lp = _lp_refine(best_seq, n, deadline, rng)
                # evaluate the LP‑refined candidate
                conv = _full_convolution(a_lp, fft_len=conv_fft_len)
                val_lp = 2.0 * n * conv.max()
                if val_lp < best_val:
                    best_val, best_seq = val_lp, a_lp.copy()
                    print(f"[n={n} LP refin.] new best {val_lp:.6f}")

        # --------------------------------------------------------------
        # Upsample the best sequence to the next length (if any) – this is
        # done implicitly at the start of the next length via interpolation.
        # No explicit action required here.

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
