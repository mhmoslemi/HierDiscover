def _project_to_simplex(v, target):
    # implementation from Duchi et al.
    # ensure v is 1D array.

    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u - (css - target) / (np.arange(len(v))+1) > 0)[0][-1]
    theta = (css[rho] - target) / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid random‑restart optimiser for the step‑function problem.

The search proceeds in three phases:

1.  Many short Adam runs on a soft‑max parametrisation of the
    sequence.  The Adam optimiser uses a smooth log‑sum‑exp surrogate
    for the max‑convolution and an aggressive temperature schedule.
    These runs explore the space quickly and provide a pool of good
    candidates.

2.  The current global best candidate is refined with an exact
    projected sub‑gradient method that minimises the *true* maximum
    of the convolution (no smoothing).  The gradient is the
    convolution of the current sequence with a uniform weight on the
    indices attaining the maximal convolution value.  After each
    gradient step the vector is projected onto the simplex
    {x≥0, Σx = √(2n)}.

3.  If time permits a final hill‑climbing stage makes tiny random
    mass transfers between positions and accepts improvements.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using a power‑of‑2 FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Random initial shapes (all are later scaled to the target sum)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive vector of length n with a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:                               # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Project a vector onto the simplex {x >= 0, Σx = target}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with given target sum."""
    if target < 0:
        raise ValueError("target sum must be non‑negative")
    # Based on Duchi et al., 2008
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho_arr = u - (css - target) / (np.arange(len(v)) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = (css[rho_max] - target) / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables (same idea as the
# previous state‑of‑the‑art but with a few tweaked hyper‑parameters)
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smoothed‑max surrogate.

    Returns the best sequence (as a NumPy array) and its official
    evaluation score (computed with ``evaluate_sequence``).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ---- 1. soft‑max mapping (enforces positivity and sum) ----------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stabilise
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # ---- 2. current max‑convolution ---------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  Adam local best = {max_conv:.9f}")

        # ---- 3. log‑sum‑exp (smooth‑max) weights -----------------------
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau               # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4. gradient w.r.t. a (derivative of Σ w_k·conv_k) ---------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # pick slice for grad
        grad_a = 2.0 * conv2[idx]

        # ---- 5. gradient w.r.t. logits x (chain rule) -------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping – prevents occasional huge steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # ---- 6. Adam update --------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to help escape flat plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9997

        # early stop if we have not improved for a long stretch
        if iteration - last_improve > 2000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # final evaluation with the official evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser that uses the *exact* max‑convolution.
# Works directly on the sequence (not logits) and projects onto the simplex.
# ----------------------------------------------------------------------
def _subgradient_optimize(
    a_init: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution.

    Returns (best_sequence, official_score).
    """
    a = a_init.copy()
    n = a.size

    # initial evaluation
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_M = max_conv
    best_a = a.copy()
    best_score = 2.0 * n * max_conv / (target_sum ** 2)

    # hyper‑parameters
    step_size = target_sum * 0.2          # raw step (will be normalised)
    min_step = 1e-8
    no_improve_cnt = 0
    max_no_improve = 800

    iteration = 0
    while time.time() < run_deadline:
        # ---- compute convolution and identify active indices ----------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        mask = conv >= max_conv - 1e-12          # tolerance for ties
        k = int(mask.sum())
        if k == 0:                               # safety
            mask = np.full_like(conv, False)
            mask[np.argmax(conv)] = True
            k = 1
        w = np.zeros_like(conv)
        w[mask] = 1.0 / k

        # ---- gradient of the max (sub‑gradient) ------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        # normalise gradient direction
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            direction = grad / gnorm
        else:
            direction = np.zeros_like(grad)

        # ---- gradient step + projection onto the simplex ---------------
        a_candidate = a - step_size * direction
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        # evaluate candidate
        conv_cand = _full_convolution_fft(a_candidate, a_candidate)
        max_cand = conv_cand.max()
        if max_cand < best_M - 1e-12:
            best_M = max_cand
            best_a = a_candidate.copy()
            best_score = 2.0 * n * max_cand / (target_sum ** 2)
            no_improve_cnt = 0
            print(f"[{time.time():.2f}] n={n}  SubGrad best = {best_M:.9f}")
        else:
            no_improve_cnt += 1

        a = a_candidate
        iteration += 1

        # adapt step size if stagnating
        if no_improve_cnt >= max_no_improve:
            step_size *= 0.5
            no_improve_cnt = 0
            if step_size < min_step:
                break

    final_score = evaluate_sequence(best_a.tolist())
    return best_a, final_score

# Main entry point required by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine respects a time budget of ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1          # small safety margin

    best_seq = None
    best_score = float("inf")
    top_candidates = []                        # keep a few elite candidates

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            top_candidates.append((sc, hs_arr.copy()))
            print(f"[{time.time():.2f}] Warm‑start global = {sc:.9f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main restart loop – each iteration receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose problem size (larger early on, smaller later)
        if remaining > 300:
            n = rng.integers(1500, 3500 + 1)
        elif remaining > 120:
            n = rng.integers(1100, 2600 + 1)
        else:
            n = rng.integers( 600, 1800 + 1)

        # ----------------------------------------------------------
        # Build an initial shape (occasionally reuse a good candidate)
        # ----------------------------------------------------------
        if top_candidates and rng.random() < 0.45:
            src_seq = top_candidates[rng.integers(len(top_candidates))][1]
            if src_seq.size >= n:
                init = src_seq[:n].copy()
            else:
                pad = rng.random(n - src_seq.size) * 1e-6
                init = np.concatenate([src_seq, pad])
        else:
            init = _initial_shape_random(n, rng)

        target_sum = np.sqrt(2.0 * n)

        # Normalise to the canonical simplex
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)      # logits for Adam

        # Allocate a time slice for this Adam run
        slice_time = min(12.0, remaining * 0.55)

        # ----------------------------------------------------------------
        # Run Adam optimiser (surrogate optimisation)
        # ----------------------------------------------------------------
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            rng,
            init_tau_factor=0.25,
            min_tau_factor=5e-4,
            alpha=0.07,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.96,
            max_iter=None,
        )

        # ----------------------------------------------------------------
        # Update global best and elite pool
        # ----------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            top_candidates.append((best_score, best_seq.copy()))
            top_candidates.sort(key=lambda t: t[0])
            top_candidates = top_candidates[:5]      # keep only a few elites
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.9f} (n={n})")
            # early exit if we already beat the target
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final refinement on the global best using the exact sub‑gradient.
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        # Re‑scale the incumbent to satisfy the exact sum constraint
        a0 = best_seq * (target_sum / best_seq.sum())
        run_deadline = deadline
        seq_refined, score_refined = _subgradient_optimize(
            a0,
            target_sum,
            run_deadline,
            rng,
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] SUB‑GRAD refinement improved best to {best_score:.9f}")

    # ------------------------------------------------------------------
    # Very short hill‑climbing tweak (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 2.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = best_seq.copy()
        # simple random mass‑transfer hill climb
        while time.time() < deadline:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] <= 0:
                continue
            # transfer a small fraction of a[i] to a[j]
            frac = rng.uniform(0.0, 0.02)   # up to 2 % of source mass
            delta = frac * a[i]
            a_candidate = a.copy()
            a_candidate[i] -= delta

            score = 2.0 * n * max_conv / (target_sum ** 2)
            if score < best_score:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
