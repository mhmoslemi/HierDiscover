def build_constraint_matrix(f):
    n = len(f)
    # number of constraints = 2*n - 1
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # overlapping indices i where 0 <= k-i < n => i in [max(0, k-(n-1)), min(n-1, k)]
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # all i from i_min to i_max inclusive
        for i in range(i_min, i_max+1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def _solve_lp_for_g(f, M, target_sum):
    """Solve LP maximize sum(g) s.t. f * g <= M, g >= 0.
    Returns g as np.ndarray of shape (n,) or None if LP fails.
    """
    import scipy.optimize as opt
    import scipy.sparse as sp
    n = len(f)
    # Build constraint matrix A (size (2n-1) x n)
    # Use diags to construct quickly.
    offsets = -np.arange(n)
    A = sp.diags([f]*n, offsets, shape=(2*n-1, n), format='csr')
    # Upper bounds vector
    b = np.full(2*n-1, M, dtype=float)
    # Objective: minimize -sum(g)
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    else:
        print("LP failed:", res.message)
        return None

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
* Load the best known construction (``height_sequence_1``) if available.
* For a series of lengths (≈2100, 1.5×, 2×) run a fast
  *soft‑max Adam* optimiser (log‑its parametrisation) that directly
  minimises the exact max‑convolution.  The optimiser obeys the simplex
  constraints Σa = √(2 n) and a_i ≥ 0.
* Starting from the best vector obtained, perform a few *LP‑mix* steps:
    - Given a current f and its worst convolution value M,
      solve the linear programme  

        maximise    Σ g_i
        subject to  (f * g)[k] ≤ M   for all k,
                    g_i ≥ 0.

      (the convolution constraint is linear in g, so scipy‑highs can
      handle it efficiently).
    - Scale g to the required normalisation and convex‑mix it into f
      (mixing factor slowly decays).
    - Keep the improved f if the max‑convolution decreased.
* Finish with a cheap stochastic pair‑wise mass‑transfer refinement
  (simulated‑annealing acceptance).
* Return the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range ignored) using FFTs.
    This is the gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # linear part
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small, diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a
    soft‑max approximation of the max‑convolution.
    Returns the best vector and its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current vector on the simplex
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # exact self‑convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # soft‑max weighting of the convolution entries
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # gradient (2·rev) w.r.t. a
        rev = _weighted_rev(a, w, Nfft)
        g_a = 2.0 * rev

        # chain rule to logits (mirror‑descent on the simplex)
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _solve_lp_for_g(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the linear programme:
        maximise    Σ g_i
        subject to  (f * g)[k] ≤ M   for all k,
                    g_i ≥ 0.
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.shape[0]
    # Build sparse constraint matrix A where A[k, i] = f_i if i contributes to shift k
    offsets = -np.arange(n)                     # rows are shifted downwards
    A = _sp.diags([f] * n, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M, dtype=float)     # upper‑bound vector

    c = -np.ones(n, dtype=float)                # minimise -Σ g_i  → maximise Σ g_i
    bounds = [(0, None)] * n

    # HiGHS (default method) is fast for these moderate‑size LPs
    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        print("[LP] failed:", res.message)
        return None


def _lp_mix_improve(
    f: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_steps: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Perform a series of LP‑mix steps starting from ``f``.
    Returns the improved vector and its max‑convolution value.
    """
    n = f.shape[0]
    target_sum = math.sqrt(2.0 * n)
    Nfft = 1 << ((2 * n - 1).bit_length())
    cur = f.copy()
    cur_M = _max_convolution(cur, Nfft)

    # mixing factor starts at 0.5 and decays geometrically
    base_mix = 0.5

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        g = _solve_lp_for_g(cur, cur_M, target_sum)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        g_scaled = g * (target_sum / sum_g)

        mix = base_mix * (0.9 ** step)           # slowly shrinking step size
        new = (1.0 - mix) * cur + mix * g_scaled
        new = np.maximum(new, 0.0)
        new = new / new.sum() * target_sum       # renormalise (numerical drift)

        new_M = _max_convolution(new, Nfft)

        if new_M < cur_M - 1e-12:
            cur = new
            cur_M = new_M
            print(f"[LP‑mix] length {n}, step {step}, new M = {cur_M:.6f}")

    return cur, cur_M


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length()))
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = np.maximum(new_seq, 0.0)

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score
