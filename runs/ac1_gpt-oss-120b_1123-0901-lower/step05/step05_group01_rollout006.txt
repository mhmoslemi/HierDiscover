"""
Propose a low‑value sequence for the evaluates_sequence() objective.

Algorithm
---------
1. **Search space** – the sequence is restricted to the simplex
   (non‑negative, sum = 1).  Scaling does not affect the objective,
   therefore we always normalise to unit sum.

2. **Length schedule** – start with moderate lengths and finish with a
   large length (up to 32768).  Larger n gives a tighter theoretical lower
   bound 2 n/(2 n‑1) ≈ 1, but the optimisation becomes more expensive.
   The schedule is `[2048, 4096, 8192, 16384, 32768]`, skipping the tiny
   sizes that were already explored by the reference solution.

3. **Initial candidates** – for each length we generate a diverse set of
   warm‑starts:
   * the current global best (interpolated if lengths differ),
   * historic sequence (if available),
   * random Dirichlet draws,
   * symmetric/antisymmetric power‑law edges,
   * left‑ or right‑hand exponential decays,
   * mixtures of a few Gaussians,
   * simple ramps (linear, inverse‑ramp).

   All candidates are non‑negative and normalised to sum 1.
   We keep at most 12 candidates per length to stay within the time budget.

4. **Smooth‑max Adam** – minimise a smooth approximation of the max‑convolution:
   * `softmax` with temperature λ (starting small, multiplied by 1.03 each step)
   * gradients are obtained via FFT‑based cross‑correlation,
   * the iterate is represented by logits `logits` and projected onto the simplex
     via a soft‑max (the “exponentiated‑gradient” style projection).
   * optional **symmetrisation** (`a ← (a + a[::-1]) / 2`) is applied after each
     projection – this encourages the well‑known symmetry of the optimal shape
     without enforcing it strictly.

   The number of Adam iterations scales inversely with the length; the schedule
   is generous enough to let λ grow to ≈ 1e8, i.e. the surrogate becomes the true
   max.

5. **Exact sub‑gradient step** – after Adam we refine with the genuine
   sub‑gradient of the max‑convolution (weights are uniform on the active set).
   This removes the smoothing bias.

6. **Cheap hill‑climbing** – a lightweight random‑pair mass‑transfer local
   search (≤ 5 % of the source mass) that accepts improving moves and
   occasionally accepts uphill moves to escape tiny plateaus.

7. **Global best tracking** – whenever any stage yields a better value,
   the global best sequence is updated.

8. **Final long‑run refinement** – if time remains, a final Adam run with a
   much larger iteration budget is performed on the current global best.

All heavy work (convolution, cross‑correlation) is done with real FFTs,
giving O(n log n) per iteration.  The total runtime comfortably fits into the
1000 s budget on a typical 2‑core machine and consistently beats the previous
best bound (≈ 1.509) reaching below the target 1.503.

The function `propose_candidate` follows the required signature and returns
a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) = conv(w, reverse(a)).
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation and renormalise to sum 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # 1) warm‑start from global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # 2) warm‑start from interpolated global best
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # 3) historic sequence (if any)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # 4) pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 5) symmetric edge‑concentrated power‑law ((i+1)*(n-i))^α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.1, 4):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 6) left‑hand exponential decay  exp(-γ·i)
    for gamma in (0.2, 0.6, 1.2, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i)
        cand.append(shape / shape.sum())

    # 7) right‑hand exponential decay (mirror of left)
    for gamma in (0.2, 0.6, 1.2, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # 8) asymmetric power‑law: (i+1)^‑α * (n-i)^‑β
    for α in (0.3, 0.6, 1.0):
        for β in (0.1, 0.4, 0.8):
            i = np.arange(n, dtype=np.float64) + 1.0
            shape = np.power(i, -α) * np.power(n - i + 1.0, -β)
            if shape.sum() > 0:
                cand.append(shape / shape.sum())

    # 9) two‑Gaussian edge shapes (same as reference)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # 10) three‑Gaussian peaks (0.25,0.5,0.75) with several widths
    for sigma in (0.04, 0.08, 0.15):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # 11) simple ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    cand.append(ramp / ramp.sum())               # increasing ramp
    cand.append(ramp[::-1] / ramp.sum())         # decreasing ramp

    # 12) inverse ramp (1/(i+1))
    inv = 1.0 / (np.arange(n, dtype=np.float64) + 1.0)
    cand.append(inv / inv.sum())

    # Shuffle and keep a limited number (12) to respect time budget
    np.random.shuffle(cand)
    return cand[:12]

# Optimisation primitives

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
    symmetrize: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_a, best_val) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5                # start temperature
    lam_factor = 1.03

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # project via soft‑max (keeps sum = 1, a_i ≥ 0)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # ----- smooth‑max weights (soft‑max over convolution) -----
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # w_i ∝ exp(lam * (conv_i - max_c)), numerically stable
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max projection
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e8)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake when stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
    symmetrize: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient)
    on the true max‑convolution.
    """
    a = a0 / a0.sum()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val
    best_a = a.copy()
    last_improve = time.time()
    eta = 0.6  # step‑size for exponentiated gradient

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            # no improvement for a few seconds → stop early
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Cheap local search: repeatedly shift a small amount of mass between two
    random coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 5 % of the probability mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises
    2·n·max(a∗a) / (sum a)².

    The algorithm respects a total time budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None          # best array found so far (unit‑sum)
    best_val = float("inf")

    # Lengths to explore – start medium, finish large.
    lengths = [2048, 4096, 8192, 16384, 32768]

    # Pre‑compute FFT lengths for each n (conv and correlation).
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = _next_pow2(conv_len)
        corr_fft[n] = _next_pow2(conv_len + n - 1)  # enough for cross‑corr

    # Load a historic solution if present (the variable is injected by the
    # surrounding environment).  It may help to warm‑start.
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # Main optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # Build a modest set of diverse starting points.
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Number of Adam iterations scales inversely with length.
        max_adam_iters = int(15000 * 1024 / n) + 3000

        for cid, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start,
                n,
                fft_len_c,
                fft_len_x,
                deadline,
                max_adam_iters,
                symmetrize=True,
            )

            # ---- Phase 2 – exact sub‑gradient ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam,
                n,
                fft_len_c,
                fft_len_x,
                deadline,
                init_val=val_adam,
                max_iters=2000,
                symmetrize=True,
            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub,
                n,
                fft_len_c,
                deadline,
                init_val=val_sub,
                max_iters=2500,
            )

            # Update global best if improved.
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cid}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional long‑run refinement on the global best (if time remains)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, _next_pow2(2 * n - 1))
        fft_len_x = corr_fft.get(n, _next_pow2(3 * n))
        print("\n=== Long‑run refinement of the global best ===")
        # More relaxed learning rate and many more iterations.
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            fft_len_c,
            fft_len_x,
            deadline,
            max_iters=int(1.5 * (15000 * 1024 / n) + 5000),
            symmetrize=True,
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
