"""
Optimiser for the step‑function bound.
---------------------------------------------------------------
We treat the coefficient vector as a probability distribution (its scale
does not matter because the objective  F(a)=2·n·max(a*a)/(sum(a)²)  is
homogeneous).  The aim is to flatten the self‑convolution as much as
possible, i.e. minimise its ℓ∞‑norm.

The optimiser works in three stages:

1. **Diverse initialisation** – for each length we build a pool of
   analytic families (uniform, Beta‑like, edge‑concentrated, Gaussian,
   triangular, sine‑squared, random Dirichlet, previous best, up‑sampled
   best, …).

2. **Smooth‑max Adam** – the vector is parametrised by logits → soft‑max.
   We minimise a log‑sum‑exp surrogate of the maximal convolution entry
   using Adam.  The temperature (λ) of the surrogate is annealed
   slowly; occasional jitter escapes plateaus.

3. **Exact sub‑gradient polishing** – after the surrogate phase we
   switch to the true max, applying an exponentiated‑gradient update on
   the active set for a few thousand iterations.

For small problems (n ≤ 2048) we also perform a **LP‑mixing step**:
solve a linear programme that maximises the total mass of a new vector
g under the current maximal convolution bound and mix it with the
current iterate.  This is the classic “dual” improvement used in the
state‑of‑the‑art method.

The routine respects a wall‑clock budget (default 1000 s) and returns
the best sequence found (as a plain Python list).  All numbers are
non‑negative, and the returned vector is normalised to sum = 1, which
does not affect the objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Helper to evaluate the true objective (uses the external function)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) because a is already normalised (sum=1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Linear programme used for the mixing step (only for modest n)
# ----------------------------------------------------------------------
def _lp_mix(a: np.ndarray, max_c: float):
    """
    Solve   max Σ b_j   s.t.  b ≥ 0,  (a * b)_k ≤ max_c for all k.
    Returns the optimiser b (may be None if the LP fails).
    """
    try:
        from scipy.optimize import linprog
    except Exception:            # SciPy not available – skip LP step
        return None

    n = a.shape[0]
    K = 2 * n - 1                     # number of convolution positions
    # Build matrix A_ub where each row k corresponds to Σ_i a_i b_{k-i}
    A = np.zeros((K, n), dtype=np.float64)
    for k in range(K):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # b_j where j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = a[i]

    b_ub = np.full(K, max_c, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σb → minimise -Σb
    bounds = [(0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the bound.
    Returns the best coefficient list discovered within the given time
    budget (in seconds).  The vector is normalised to unit sum.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the caller

    best_seq = None
    best_val = float("inf")

    # Lengths are processed from small to large – cheap runs first.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))   # (2n‑1)+(n‑1)

        # --------------------------------------------------------------
        # Build a diverse pool of initial candidates (all are normalised)
        # --------------------------------------------------------------
        candidates = []

        # 1. Uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2. Warm‑start from a previous solution of the same size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # 3. Upsample a shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum() if up.sum() > 0 else np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 4. Historic global variable, if any
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            h = h / h.sum() if h.sum() > 0 else np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # 5. Random Dirichlet draws (several)
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # 6. Edge‑concentrated power‑law families
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 7. Two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 8. Smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # 9. Triangular (ramp up then down)
        half = n // 2
        tri = np.concatenate([np.arange(1, half + 1),
                              np.arange(half, 0, -1)])[:n]
        tri = tri.astype(np.float64)
        tri = tri / tri.sum()
        candidates.append(tri)

        # 10. Cosine‑based bump (smooth, centred)
        x = np.linspace(-np.pi, np.pi, n)
        cosb = (1.0 + np.cos(x)) / 2.0
        cosb = cosb / cosb.sum()
        candidates.append(cosb)

        # 11. Exponential edge decay
        decay = np.exp(-0.5 * np.arange(n))
        decay = decay + decay[::-1]
        decay = decay / decay.sum()
        candidates.append(decay)

        # --------------------------------------------------------------
        # Optimise each candidate
        # --------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Initialise logits for soft‑max parametrisation
            eps = 1e-16
            a = np.maximum(a0, eps)
            a = a / a.sum()
            logits = np.log(a)

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            # Temperature for the smooth‑max surrogate
            lam = 0.3
            lam_factor = 1.01

            # Iteration budget (more for smaller n)
            max_iters = int(30000 * 1024 / n) + 2500
            last_improve = time.time()

            for it in range(1, max_iters + 1):
                # Soft‑max → simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # Convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w.fill(1.0 / w.size)
                else:
                    w /= w_sum

                # Gradient of the smooth‑max surrogate w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # Back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # Anneal temperature, cap to avoid overflow
                lam = min(lam * lam_factor, 1e7)

                # Exact objective (sum a = 1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # Stagnation handling – tiny random jitter to logits
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----------------------------------------------------------
            # Exact sub‑gradient refinement (exponentiated gradient)
            # ----------------------------------------------------------
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(4000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                # active set – indices where the max is attained (within tol)
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # numerical rescue
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluate exact objective
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

            # ----------------------------------------------------------
            # OPTIONAL LP‑mixing step (only for modest n)
            # ----------------------------------------------------------
            if n <= 2048 and time.time() < deadline:
                max_c = _full_convolution(a).max()
                b = _lp_mix(a, max_c)
                if b is not None and b.sum() > 0:
                    g = b / b.sum()
                    # convex combination – a small step towards the LP solution
                    t = 0.2
                    a = (1.0 - t) * a + t * g
                    a = a / a.sum()

                    # a short polishing pass
                    eta_lp = 0.5
                    for lp_it in range(1200):
                        conv = _full_convolution(a, fft_len=conv_fft_len)
                        max_c = conv.max()
                        active = np.isclose(conv, max_c, atol=1e-12)
                        w = np.zeros_like(conv)
                        if active.any():
                            w[active] = 1.0 / active.sum()
                        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                        a = a * np.exp(-eta_lp * grad_a)
                        a = a / a.sum()
                        val = 2.0 * n * max_c
                        if val < best_val:
                            best_val = val
                            best_seq = a.copy()
                            print(f"[n={n} LP‑mix it={lp_it}] new best {val:.6f}")
                        if time.time() > deadline:
                            break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (highly unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
