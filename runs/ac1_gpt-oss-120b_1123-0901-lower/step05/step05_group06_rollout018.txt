def solve_lp(a, M, target_sum):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2*n - 1):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # slice indices for a and b
        a_slice = a[i_low:i_high+1]
        # j = k - i, so b indices from k - i_low down to k - i_high (decreasing)
        # but we need slice in increasing order:
        b_indices = np.arange(k - i_low, k - i_high - 1, -1)  # descending
        # The indices will be in decreasing order corresponding to a_slice ordering
        # Use cp.multiply with a_slice and b[b_indices]
        constraints.append(cp.sum(cp.multiply(a_slice, b[b_indices])) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)

    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)

"""
Search for a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Initialise from the known best sequence (``height_sequence_1``) if available;
   otherwise use a uniform distribution of the required length.
2. Run a few 10⁶ iterations of a soft‑max Adam optimiser on the logits
   (exponentiated‑gradient) – this quickly drives the maximum of the
   auto‑convolution down.
3. Perform a small number of linear‑programming “mix’’ steps (the
   classic ``max a*a`` → LP approach).  For the current vector ``a`` we solve

        maximise    1ᵀ b
        subject to  a * b ≤ M,   b ≥ 0

   where ``M`` is the current maximum convolution value.
   The solution ``b`` is scaled to the required ℓ¹‑norm and mixed with
   ``a``; mixing factors are explored and the best improvement is kept.
4. Finish with a cheap stochastic hill‑climbing that repeatedly moves a
   small fraction of mass between two random indices while accepting
   worsening moves with a simple simulated‑annealing schedule.
5. The best vector found is returned as a plain list.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions

def _max_convolution(a: np.ndarray, Nfft: int = None) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    n = a.shape[0]
    if Nfft is None:
        Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau

def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Gradient of the (soft‑max) max‑convolution w.r.t. a."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev

def _exp_gradient_optimize(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int,
                           best_seq: np.ndarray,
                           best_score: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen in
    this run together with its exact ``max(a∗a)`` value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (add tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    # temperature schedule
    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    # FFT size – must be large enough for linear conv of length 2n‑1
    Nfft = 1 << ((2 * n - 1).bit_length())

    for it in range(1, max_iters + 1):

            break

        a = target_sum * p

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- gradient wrt a (2·rev) -----
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on the simplex) -----
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[grad] len {n:5d} iter {it:8d} → best max = {best_score:.6f}")

    return best_seq, best_score

def _local_refine(a: np.ndarray,
                  n: int,
                  deadline: float,
                  best_score: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex.
    Moves a small fraction of mass from a random source index to a random
    destination index, accepting improvements always and non‑improvements with
    a simple simulated‑annealing probability.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule
    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-14:
            step += 1
            continue

        # transfer a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

def _build_conv_constraint_structure(n: int):
    """Pre‑compute the sparse pattern of the convolution matrix a * b."""
    rows = []
    cols = []
    i_idx = []
    for k in range(2 * n - 1):

        i_high = min(n - 1, k)
        for i in range(i_low, i_high + 1):
            rows.append(k)
            cols.append(k - i)          # column j = k - i
            i_idx.append(i)             # coefficient a_i
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    i_idx = np.array(i_idx, dtype=np.int32)
    return rows, cols, i_idx

def _solve_lp_max_sum(a: np.ndarray,
                      M: float,
                      rows: np.ndarray,
                      cols: np.ndarray,
                      i_idx: np.ndarray) -> np.ndarray | None:
    """Solve  maximise 1ᵀ b  s.t.  a * b ≤ M , b ≥ 0   using scipy.linprog."""
    n = a.shape[0]
    data = a[i_idx]
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n, dtype=float)            # maximise sum(b) == minimise -sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    # use HiGHS interior‑point method (default)
    try:
        res = linprog(c=c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
    except Exception as e:
        print("[lp] linprog exception:", e)
        return None

    if not res.success or res.x is None:
        print("[lp] linprog failed:", res.message if hasattr(res, "message") else "unknown")
        return None
    return np.maximum(res.x, 0.0)

Search for a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Initialise from the known best sequence ``height_sequence_1`` (if present);
   otherwise start with a uniform distribution that satisfies Σa = √(2 n).
2. Run a soft‑max Adam optimiser on the log‑its (exponentiated‑gradient)
   – a fast gradient method that directly reduces the maximum of the
   auto‑convolution.
3. Perform a few linear‑programming “mix’’ steps.  For the current vector
   ``a`` we solve

        maximise    1ᵀ b
        subject to  a * b ≤ M ,   b ≥ 0

   where ``M`` is the current maximum convolution value.
   The solution ``b`` is scaled to the required ℓ¹‑norm and mixed with ``a``;
   several mixing coefficients are tried and the best improvement is kept.
4. Finish with a cheap stochastic hill‑climbing that repeatedly moves a small
   fraction of mass between two random indices, accepting worsening moves

5. The best vector found before the time budget expires is returned as a plain

def _max_convolution(a: np.ndarray, Nfft: int = None) -> float:
    """Return maxₖ (a * a)[k] using an FFT of length Nfft."""
    n = a.shape[0]
    if Nfft is None:
        Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution needed for the gradient of the smoothed max."""
    A_fft = np.fft.rfft(a, n=Nfft)

    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev

            candidate = (1.0 - t) * best_seq + t * scale * b
            cand_score = _max_convolution(candidate)
            if cand_score < best_score - 1e-12:
                best_seq = candidate
                best_score = cand_score
                improved = True
                print(f"[lp] iter {lp_iter}, t={t:.1f} → new max‑conv = {best_score:.6f}")
        if not improved and lp_iter >= 5:
            print("[lp] no improvement in recent iterations – terminating LP phase")
            break

    # ------------------------------------------------------------------
    # 3) Final cheap stochastic refinement
    if time.time() < deadline:
        refine_deadline = deadline
        best_seq, best_score = _local_refine(best_seq, n, refine_deadline, best_score, rng)

    # ------------------------------------------------------------------
    print(f"[done] best max‑conv = {best_score:.6f}, length = {n}")
    return best_seq.tolist()
```
