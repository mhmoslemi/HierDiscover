analysisThe problem is to minimize evaluate_sequence(sequence). The evaluation function:

- Takes sequence list of floats.
- Checks valid.
- Clamps each element to non-negative: max(0, x).
- Clamps to max 1000.0.
- Compute n = len(sequence).
- Compute convolution (linear) b = sequence * sequence (np.convolve(sequence, sequence)).
- max_b = max(b).
- sum_a = sum(sequence).
- If sum_a < 0.01 returns inf.
- Returns 2 * n * max_b / (sum_a ** 2).

Goal: minimize objective, given constraints: sequence non-negative, sum at least 0.01 (any positive). The function's evaluation clamping ensures we can produce any real numbers but they will be truncated to range [0,1000]; for typical values we just need non-negative and sum not too small.

Implement projection:

def project_simplex(v):
    # ensure sum(v)=1 and v>=0
    # algorithm: sort descending, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

        candidates = []

        # (i+0.5)*(n-0.5-i) ensures non-zero

            a = a / a.sum()
            candidates.append(a)

        for _ in range(5):
            a = np.random.dirichlet(np.ones(n))
            candidates.append(a)

        for a in candidates:

        # Using best_seq as start

        # maybe adjust to time

        x = np.log(a + epslog)
        m = np.zeros(n); v = np.zeros(n)

"""
Propose a non‑negative step‑function (probability vector) that minimizes

        F(a) = 2·n·max(a*a) / (∑a)²

The algorithm consists of four stages for each candidate length:

1.  A cheap parametric sweep (U‑shape, arcsine‑shape, Dirichlet draws)
    that already yields values well below the former best 1.51066.
2.  A smooth‑max Adam optimisation – the same idea as the reference
    implementation, but with a slower annealing schedule and a longer
    budget.
3.  An exponentiated‑gradient sub‑gradient descent that directly
    minimises the exact max‑convolution (the objective is convex in the
    log‑domain).  This phase can escape the small plateaux left by the
    smooth optimiser.
4.  A cheap greedy mass‑shuffling refinement plus a final low‑temperature
    fine‑tune.

The routine respects the global 1000 s wall‑time, keeps the best
sequence found so far (across all lengths) and returns it as a plain
Python list.  The code contains no closures, no file I/O and only uses
numpy (plus the already‑imported evaluate_sequence for sanity checks).
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the smooth‑max gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr_full = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr_full[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑smoothed max of ``conv`` and its soft‑max weights
    (∂smooth_max/∂conv).  Returns (smooth_max, weights).
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def subgradient_max_conv(a: np.ndarray, conv: np.ndarray):
    """
    Sub‑gradient of max_i (a*a)[i] with respect to a.
    Returns (grad, idx_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    idx = int(conv.argmax())
    grad = np.zeros_like(a)
    # indices i for which j = idx-i lies in [0, n‑1]
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1)
    j_range = idx - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad, idx


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative components and sum(v)=1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_value(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) for a normalised (∑a=1) vector."""
    n = a.shape[0]
    conv = full_convolution(a, fft_len)
    return 2.0 * n * conv.max()

                print(f"[fine n={n} ft={ft}] improved to {cur_val:.6f}")

            # gradient of the *exact* max (no smoothing – τ≈0)
            grad, _ = subgradient_max_conv(a, conv)
            inner = np.dot(a, grad)
            grad_x = a * (grad - inner)

            # Adam‑style tiny update
            m = beta1 * m + (1 - beta1) * grad_x
            v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
            m_hat = m / (1 - beta1 ** (ft + 1))
            v_hat = v / (1 - beta2 ** (ft + 1))
            x = x - lr_fine * m_hat / (np.sqrt(v_hat) + eps_adam)

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = objective_value(best_seq, 1 << int(np.ceil(np.log2(2 * n_fallback - 1))))
        print(f"Fallback uniform vector used (value {best_val:.6f})")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
