            a = np.maximum(a, 0.0)
            total = a.sum()
            if total == 0.0:
                a = np.full(n, target_sum / n)
            else:
                a *= target_sum / total
            # clip
            a = np.clip(a, 0.0, 1000.0)
            if it % 200 == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
        # final evaluation
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] after refinement best_score={final_score:.8f}")

        # fallback uniform
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum/n)

def project_simplex(v, z=1.0):
    # projects v onto the simplex {x>=0, sum x = z}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Search for a non‑negative height sequence that minimises

    does not change the objective) and evaluated immediately.

2.  **Adam on a soft‑max parametrisation** – the best seed so far is used
    as a warm‑start for a short Adam optimisation.  The objective is
    approximated by a log‑sum‑exp (“smooth‑max”) of the full linear
    convolution; gradients are obtained via two FFT‑based convolutions.
    The temperature is annealed aggressively so that the optimisation
    gradually behaves like a sub‑gradient step on the true max.

3.  **Time‑aware restart loop** – as long as the global‑time budget
    (default 1000 s) permits, the algorithm repeats the generate‑>optimise
    cycle with different lengths (≈ 1500–4500) and different seeds,
    keeping a handful of elite candidates.  A final refinement on the
    overall best is performed if time remains.

The routine returns the best sequence it has found as a plain Python
list – exactly the format expected by the outer harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (real‑to‑real)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via an FFT of power‑of‑2 size."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())               # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Central‑binomial coefficients  c_k = (2k choose k) / 4**k
# ----------------------------------------------------------------------
def _central_binomial_sequence(n: int) -> np.ndarray:
    """Generate the first n coefficients of (1‑z)^‑½ = Σ c_k z^k."""
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for k in range(1, n):
        # recurrence  c_k = c_{k‑1} * (2k‑1)*(2k) / (k*k*4)
        c[k] = c[k - 1] * (2 * k - 1) * (2 * k) / (k * k * 4.0)
    return c


# ----------------------------------------------------------------------
# Simple deterministic families
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=float)


def _make_power_law(n: int, exponent: float) -> np.ndarray:
    """a_i = (i+1)^{-exponent}   (exponent > 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-exponent)


def _make_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density proportional to 1/√(x·(1‑x))."""
    idx = np.arange(n) + 0.5
    x = idx / n
    return 1.0 / np.sqrt(x * (1.0 - x))


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.5,
    min_tau_factor: float = 1e-5,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.98,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed‑max surrogate.
    Returns the best sequence (as a NumPy array) and its exact score
    as computed by ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    it = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ----------------------------------------------------------
        # (1) map logits -> simplex (a = S·softmax(x))
        # ----------------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)            # numerically stable
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # ----------------------------------------------------------
        # (2) evaluate current max‑convolution (used only for early‑stop)
        # ----------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = it
            print(f"[{time.time():.2f}] n={n}  local best max‑conv = {max_conv:.8f}")

        # ----------------------------------------------------------
        # (3) smooth‑max weights (log‑sum‑exp surrogate)
        # ----------------------------------------------------------
        tau = max(init_tau_factor * max_conv * (tau_decay ** it),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau          # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # (4) gradient w.r.t. a   (derivative of Σ w_k·conv_k)
        # ----------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # desired slice
        grad_a = 2.0 * conv2[idx]

        # ----------------------------------------------------------
        # (5) gradient w.r.t. logits (chain rule through the soft‑max)
        # ----------------------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping to keep steps bounded
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 15.0:
            grad_x = grad_x * (15.0 / gnorm)

        # ----------------------------------------------------------
        # (6) Adam update
        # ----------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 400 == 0 and rng.random() < 0.12:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning‑rate
        alpha *= 0.9997

        # early stop if no improvement for a long while
        if it - last_improve > 3000:
            break
        if max_iter is not None and it >= max_iter:
            break

    # final evaluation with the official (exact) evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    given evaluation function.  The routine respects a total time
    budget (default 1000 s) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1      # safety margin

    best_seq = None
    best_score = float("inf")
    elite = []                             # list of (score, seq)

    # --------------------------------------------------------------
    # First pass: deterministic seeds (uniform, power‑law, arcsine,
    # central‑binomial).  A few lengths are tried quickly.
    # --------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500]
    power_exps = [0.3, 0.4, 0.5, 0.55, 0.6]   # exponent in a_i = i^{-p}
    for n in lengths:
        if time.time() > deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # uniform
        a = _make_uniform(n)
        a = a * (target_sum / a.sum())
        score = evaluate_sequence(a.tolist())
        print(f"[{time.time():.2f}] uniform n={n}  score={score:.8f}")
        if score < best_score:
            best_score, best_seq = score, a.copy()
            elite.append((score, a.copy()))

        # arcsine shape
        a = _make_arcsine(n)
        a = a * (target_sum / a.sum())
        score = evaluate_sequence(a.tolist())
        print(f"[{time.time():.2f}] arcsine  n={n}  score={score:.8f}")
        if score < best_score:
            best_score, best_seq = score, a.copy()
            elite.append((score, a.copy()))

        # central‑binomial (exact square‑root of constant sequence)
        a = _central_binomial_sequence(n)
        a = a * (target_sum / a.sum())
        score = evaluate_sequence(a.tolist())
        print(f"[{time.time():.2f}] binomial n={n}  score={score:.8f}")
        if score < best_score:
            best_score, best_seq = score, a.copy()
            elite.append((score, a.copy()))

        # power‑law family
        for p in power_exps:
            a = _make_power_law(n, p)
            a = a * (target_sum / a.sum())
            score = evaluate_sequence(a.tolist())
            print(f"[{time.time():.2f}] power p={p:.2f} n={n}  score={score:.8f}")
            if score < best_score:
                best_score, best_seq = score, a.copy()
                elite.append((score, a.copy()))

    # keep only the top few elite candidates
    elite.sort(key=lambda z: z[0])
    elite = elite[:5]

    # --------------------------------------------------------------
    # Main optimisation loop – many short Adam runs with random restarts
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # decide a problem size (bigger early on, smaller later)
        if remaining > 300:
            n = rng.integers(1500, 3500 + 1)
        elif remaining > 120:
            n = rng.integers(1200, 2600 + 1)
        else:
            n = rng.integers(800, 2000 + 1)

        target_sum = np.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # initialise either from an elite candidate or fresh random shape
        # ----------------------------------------------------------------
        if elite and rng.random() < 0.45:
            src = elite[rng.integers(len(elite))][1]
            if src.size >= n:
                init = src[:n].copy()
            else:
                # pad with a tiny random tail
                init = np.concatenate([src, rng.random(n - src.size) * 1e-6])
        else:
            # random power‑law with exponent in [0.1, 2.0] (same distribution
            # as the original script) – this gives a diverse set of starts.
            p = rng.uniform(0.1, 2.0)
            init = (np.arange(1, n + 1, dtype=float) ** (-p))
            init += rng.random(n) * 1e-6

        # log‑it / soft‑max representation
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)

        # allocate a slice of the remaining budget (max 12 s)
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam_softmax(
            n=n,
            x0=x0,
            target_sum=target_sum,
            run_deadline=run_deadline,
            rng=rng,
            init_tau_factor=0.5,
            min_tau_factor=1e-5,
            alpha=0.07,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.98,
            max_iter=None,
        )

        if score_local < best_score:
            best_score, best_seq = score_local, seq_local.copy()
            elite.append((best_score, best_seq.copy()))
            elite.sort(key=lambda z: z[0])
            elite = elite[:5]
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            if best_score < 1.30:               # strong early stop condition
                break

    # ----------------------------------------------------------------------
    # Final refinement on the incumbent best (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = best_seq * (target_sum / best_seq.sum())
        x0 = np.log(a + 1e-12)
        seq_ref, score_ref = _optimise_adam_softmax(
            n=n,
            x0=x0,
            target_sum=target_sum,
            run_deadline=deadline,
            rng=rng,
            init_tau_factor=0.25,
            min_tau_factor=1e-6,
            alpha=0.045,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.95,
            max_iter=None,
        )
        if score_ref < best_score:
            best_score, best_seq = score_ref, seq_ref.copy()
            print(f"[{time.time():.2f}] REFINEMENT improved best to {best_score:.8f}")

    # ----------------------------------------------------------------------
    # Fallback – should never happen but kept for safety
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The engine expects a plain Python list.
    return best_seq.tolist()
