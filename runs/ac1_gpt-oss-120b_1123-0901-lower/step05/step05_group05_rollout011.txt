def _directed_anneal(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a,a).max()
    # compute initial temperature
    T0 = best_val * 0.5
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1 - fraction))
        # compute conv and peak location
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        # compute gradient for this peak index (subgradient)
        grad = np.zeros(n)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end+1)
        grad[js] = 2.0 * a[k - js]
        # pick index to decrease (largest gradient entry)
        i = int(js[np.argmax(grad[js])])
        # pick random target index j
        j = np.random.randint(0, n)
        if i == j:
            continue
        if a[i] <= 0:
            continue
        # move fraction of mass from i to j
        delta = a[i] * np.random.rand() * 0.2   # up to 20% of source
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # project to simplex (to correct rounding errors)
        a_new = _project_onto_simplex(a_new, target_sum)
        new_val = _full_convolution_fft(a_new, a_new).max()
        # acceptance
        if new_val < best_val:
            a = a_new
            best_a = a_new.copy()
            best_val = new_val
        else:
            # metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                a = a_new
        # occasional random jitter
        if it % 500 == 0:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)
    return best_a, best_val

"""
Hybrid optimiser for the step‑function problem.

The objective (after normalising Σaᵢ = √(2 n)) simplifies to
    maxₖ (a * a)[k]            (linear convolution).
We therefore minimise the peak of the auto‑convolution.

The optimiser runs under the supplied time budget (default 1000 s) and
combines three complementary phases:

1. **Exploratory random sampling** – many short Dirichlet draws of
   varying length give a quick, cheap estimate of a good vector length
   and provide a warm‑start for the later stages.

2. **Adam optimisation on a smooth‑max surrogate** – the coefficients
   are represented by a soft‑max (ensuring non‑negativity and a fixed
   ℓ₁‑norm).  A log‑sum‑exp approximation smoothes the max‑operator;
   Adam with a slowly decaying temperature refines the vector to a
   low convolution peak.

3. **Directed simulated annealing** – starting from the best Adam
   result, tiny mass transfers are performed.  The index to lose mass
   is chosen from the current peak’s sub‑gradient (the most
   “responsible” coordinates), while the recipient index is chosen
   randomly.  A Metropolis acceptance rule with a linear cooling schedule
   allows the search to escape shallow local minima and converge to a
   very low final value.

All phases respect the overall budget and constantly keep the best
vector seen so far.  The returned object is a plain Python list ready
for the external ``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution (real FFT)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the simplex {w ≥ 0, Σw = s}
# (Euclidean projection – Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on the log‑sum‑exp surrogate for max(conv)
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parametrisation of a.
    Returns the best a (already normalised to target_sum) and its exact
    evaluation (max convolution value).
    """
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    # pre‑compute length for the FFT convolution used in the gradient
    size2 = 1 << ((3 * n - 2 - 1).bit_length())  # enough for a * w_rev

    while time.time() < deadline:
        # ----- soft‑max mapping (keeps a on the simplex) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stabilised exponentials
        p = exp_shift / np.sum(exp_shift)       # probabilities, sum 1
        a = target_sum * p                       # current coefficient vector

        # ----- exact max of convolution (used for logging) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp surrogate) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate w.r.t. a -----
        w_rev = w[::-1]                       # reversed weights
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)       # positions that correspond to Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- back‑propagation to logits (chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to get out of plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    # Final exact evaluation (identical to best_val, but we call the
    # supplied evaluator for safety)
    final_score = float(best_val)
    return best_a, final_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on max(conv(a,a)) with simplex projection.
    Returns the best vector and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n   # start with ~1 % of the mean entry

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        # Sub‑gradient of the active peak
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # Gradient step + projection
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02
        else:
            step_size *= 0.70

        # occasional jitter to keep the search alive
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}")

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Directed simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _directed_anneal(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that moves a small amount of mass from the
    most “responsible” index (according to the active peak’s sub‑gradient)
    to a random destination.  A simple linear cooling schedule is used.
    Returns the best sequence encountered and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # Temperature schedule – start relatively hot, finish near zero
    T0 = best_val * 0.4
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        # current peak index and its sub‑gradient
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        grad = np.zeros(n)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # index i that most contributes to the peak
        i = int(js[np.argmax(grad[js])])
        # random destination
        j = np.random.randint(0, n)
        if i == j or a[i] <= 0:
            continue

        # move a fraction of the source mass (up to 20 %)
        delta = a[i] * np.random.rand() * 0.20
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)

        new_val = _full_convolution_fft(a_new, a_new).max()

        if new_val < best_val:
            a = a_new
            best_a = a_new.copy()
            best_val = new_val
            print(f"[{time.time():.2f}] anneal it={it} new best={best_val:.6f}")
        else:
            # Metropolis acceptance rule
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                a = a_new

        # occasional global jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimiser that respects the given time budget and returns
    a list of non‑negative heights with a low evaluation score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15      # tiny safety margin

    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a known construction if it exists.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.asarray(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        val0 = float(_full_convolution_fft(hs_norm, hs_norm).max())
        best_seq = hs_norm.copy()
        best_val = val0
        print(f"[{time.time():.2f}] warm‑start from height_sequence_1 val={best_val:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 0 – cheap random exploration (≈ 5 % of the budget)
    # --------------------------------------------------------------
    while time.time() < deadline - 0.9 * (budget_s * 0.4):
        if deadline - time.time() < 2.0:
            break
        n = np.random.randint(1500, 3500)
        target_sum = np.sqrt(2.0 * n)
        a = np.random.dirichlet(np.ones(n)) * target_sum
        val = float(_full_convolution_fft(a, a).max())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] random sample n={n} val={best_val:.6f}")
        # also try a lightweight power‑law shape
        if np.random.rand() < 0.15:
            p = np.random.uniform(0.1, 0.9)
            idx = np.arange(1, n + 1, dtype=float)
            a2 = idx ** (-p)
            a2 = a2 / a2.sum() * target_sum
            val2 = float(_full_convolution_fft(a2, a2).max())
            if val2 < best_val:
                best_val = val2
                best_seq = a2.copy()
                print(f"[{time.time():.2f}] power‑law n={n} p={p:.3f} val={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on smooth max (large allocation)
    # --------------------------------------------------------------
    while time.time() < deadline - 50.0:
        remaining = deadline - time.time()
        # pick a length – longer when we have plenty of time
        if remaining > 600:
            n = np.random.randint(2000, 4000)
        elif remaining > 300:
            n = np.random.randint(1500, 3000)
        else:
            n = np.random.randint(1000, 2500)

        target_sum = np.sqrt(2.0 * n)

        # initialise logits – either from the global best (if size matches) or fresh
        if best_seq is not None and best_seq.shape[0] == n and np.random.rand() < 0.4:
            a0 = best_seq.copy()
        else:
            a0 = np.random.dirichlet(np.ones(n)) * target_sum
        logits0 = np.log(a0 + 1e-12)

        # allocate a time slice (up to 30 s or half of the remaining budget)
        slice_deadline = time.time() + min(30.0, remaining * 0.5)
        a_opt, val_opt = _adam_optimize(
            n,
            logits0,
            target_sum,
            slice_deadline,
            init_tau_factor=0.25,
            min_tau_factor=0.001,
            alpha0=0.02,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam new best val={best_val:.6f} (n={n})")
        if best_val < 1.503:
            break

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.5)
        a_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] sub‑gradient improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Directed simulated annealing for final polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        a_ann, val_ann = _directed_anneal(best_seq.copy(), target_sum, deadline)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] annealing final best {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_val = float(_full_convolution_fft(best_seq, best_seq).max())
        print(f"[{time.time():.2f}] fallback uniform val={best_val:.6f}")

    print(f"[{time.time():.2f}] final best value={best_val:.6f} (n={best_seq.shape[0]})")
    # external evaluator expects a plain Python list
    return best_seq.tolist()
