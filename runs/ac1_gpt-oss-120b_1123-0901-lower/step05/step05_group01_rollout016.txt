"""
Optimisation of a non‑negative step‑function to minimise

        F(a) = 2·n·max(a∗a) / (sum a)²

The implementation builds on the multi‑stage scheme that was used
previously, but adds several refinements:

* a larger and more diverse warm‑start pool,
* longer Adam runs with a slower temperature (lam) schedule,
* an exponentiated‑gradient (sub‑gradient) phase with a decaying step‑size,
* an optional projected‑gradient descent phase,
* an outer “refinement‑loop’’ that keeps improving the current best
  sequence until the time budget is exhausted.

All helper functions are defined at module level, no closures are used,
and the routine respects a strict deadline (default 1000 s).  The
returned value is a plain Python list of non‑negative floats ready to be
passed to ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective and helper utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    Implementation follows Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial‑candidate generation
# ----------------------------------------------------------------------
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise to sum‑1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new

    n: int,

    """

    # historic sequence (if any)

            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # three‑Gaussian peaks
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # shuffle and keep a limited number (do not exceed budget)
    np.random.shuffle(cand)
    return cand[:15]   # a slightly larger pool than before


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate
    (soft‑max over the convolution values).
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.3               # start a little smoother than before
    lam_factor = 1.015      # slower temperature increase

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_c)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max (soft‑max) weights
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (sub‑gradient) optimisation of the exact
    max‑convolution objective, with a decaying step size η_t = η₀/√t.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_c)
    else:
        best_val = init_val

    eta0 = 0.6
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len_x)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_c)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        # stop early if we have not improved for a while
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact objective.
    Uses the same sub‑gradient as the exponentiated‑gradient method
    but takes additive steps and projects back onto the simplex.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_c)
    else:
        best_val = init_val

    step = 0.3
    decay = 0.995
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len_x)

        # additive step opposite to the gradient
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        cur_val = _objective(a_new, n, fft_len_c)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            a = a_new
            last_improve = time.time()
            print(f"[n={n} PGD it={it}] new best {cur_val:.6f}")
        else:
            # if stuck, shrink the step size
            if time.time() - last_improve > 3.0:
                step *= 0.5
                last_improve = time.time()

        # gentle step‑size decay
        step *= decay

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a small fraction of mass
    between two random coordinates and keep the move if it improves the
    objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        eps = np.random.rand() * 0.05 * a[i1]          # up to 5 % of mass
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill jump to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore (start modest, end with a large candidate)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for convolution and correlation.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # Load a historic solution if available.
    # ------------------------------------------------------------------
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # Main optimisation loop over the chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # Build a diverse set of starting points.
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Adam iterations scale roughly with 1/n (more for small n)
        max_adam_iters = int(3000 * 1024 / n) + 3000
        max_subgrad_iters = int(2500 * 1024 / n) + 3000
        max_hc_iters = 2000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact sub‑gradient (exponentiated‑gradient) ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,
                init_val=val_adam, max_iters=max_subgrad_iters
            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline,
                init_val=val_sub, max_iters=max_hc_iters
            )

            # Keep global best.
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) F={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional final refinement on the best sequence (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n, 1 << int(np.ceil(np.log2(3 * n))))

        # Long‑run Adam refinement with a larger iteration budget.
        remaining = deadline - time.time()
        ref_iters = int(min(60000, remaining * 60))   # a generous upper bound
        print("\n=== Long‑run refinement of the global best ===")
        best_seq, best_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline, max_iters=ref_iters
        )

        # Follow‑up with the exact sub‑gradient method.
        if time.time() < deadline:
            best_seq, best_val = _subgrad_optimize(
                best_seq, n, fft_len_c, fft_len_x, deadline,
                init_val=best_val, max_iters=4000
            )

        # Finally a few projected‑gradient steps.
        if time.time() < deadline:
            best_seq, best_val = _pgd_optimize(
                best_seq, n, fft_len_c, fft_len_x, deadline,
                init_val=best_val, max_iters=3000
            )

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
