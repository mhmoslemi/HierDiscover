    size = 1 << ((2*n - 2).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*n - 1]
    # round negative small
    np.maximum(conv, 0, out=conv)
    return conv.max()

def _annealing_run(seq_init, target_sum, deadline, rng):
    seq = seq_init.copy()
    n = seq.size
    # ensure sum scaling
    cur_sum = seq.sum()
    if cur_sum != 0:
        seq = seq * (target_sum / cur_sum)
    else:
        seq = np.full(n, target_sum / n)
    current_max = _max_convolution(seq)
    best_seq = seq.copy()
    best_max = current_max

    # temperature param
    temperature = 0.5 * current_max
    iteration = 0
    last_best_iter = 0

    while time.time() < deadline:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        if seq[i] > 0:
            # fraction between 0 and some limit decreasing with temperature
            max_frac = 0.3
            delta = seq[i] * rng.uniform(0, max_frac)
            # Ensure we don't move more than seq[i] (delta <= seq[i])
        else:
            delta = 0.0
        # operate
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # compute
        new_max = _max_convolution(new_seq)
        # acceptance
        if new_max < current_max:
            accept = True
        else:
            # accept with probability exp(-(new_max - current_max)/temperature)
            if temperature > 0:
                prob = math.exp(-(new_max - current_max) / temperature)
                accept = rng.random() < prob
            else:
                accept = False
        if accept:
            seq = new_seq
            current_max = new_max
            # update best
            if new_max < best_max:
                best_max = new_max
                best_seq = new_seq.copy()
                last_best_iter = iteration
                print(f"[{time.time():.2f}] new global best max={best_max:.8f}")
        # cooling schedule
        iteration += 1
        temperature *= 0.9999  # slowly cool
        # occasional reheating if stuck
        if iteration - last_best_iter > 5000:
            temperature = max(temperature, 0.1 * current_max)
            last_best_iter = iteration

    return best_seq, best_max

"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a simple simulated‑annealing / hill‑climbing search.
The algorithm works with a “soft” L∞ objective: it repeatedly moves a
small amount of mass from one index to another while keeping the total
L1‑norm (≈ √(2 n)) fixed.  Every move is accepted if it improves the
current maximum of the self‑convolution, otherwise it is accepted with a
probability that depends on a temperature schedule.  A small fraction of
iterations perform a *targeted* move that directly attacks the index
producing the current maximum (by moving mass away from the largest
pairwise product that contributes to that peak).  Multiple restarts with
different lengths and random initialisations are performed until the
budget expires; the best sequence found is returned.

The code respects the 1000 s time budget, prints progress and works with
the already‑provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, max value and index)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a power‑of‑2 FFT (real input)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _max_convolution_and_index(seq: np.ndarray):
    """
    Return (max_value, argmax_index) of the full self‑convolution of *seq*.
    """
    conv = _full_convolution_fft(seq, seq)
    idx = int(conv.argmax())
    return float(conv.max()), idx


# ----------------------------------------------------------------------
# Random initial sequences (Dirichlet based)
# ----------------------------------------------------------------------
def _random_dirichlet_seq(n: int, target_sum: float, rng: np.random.Generator):
    """Generate a random non‑negative vector of length *n* with L1 = target_sum."""
    # concentration parameter < 1 gives a sparse‑ish Dirichlet
    alpha = rng.uniform(0.05, 0.5)
    seq = rng.dirichlet(np.full(n, alpha)) * target_sum
    return seq


# ----------------------------------------------------------------------
# Targeted move – reduces the contribution of the current peak offset
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, peak_idx: int, rng: np.random.Generator):
    """
    Move a fraction of the mass from the index participating most in the
    current peak (given by *peak_idx*) to a random other location.
    """
    n = seq.shape[0]
    # indices that have a partner for this peak
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    i_vals = np.arange(lo, hi + 1)
    j_vals = peak_idx - i_vals

    # product a_i * a_j for each valid pair
    prod = seq[i_vals] * seq[j_vals]
    if prod.size == 0:
        return seq  # nothing to do

    # index with the largest product (most responsible for the peak)
    i_max = i_vals[int(prod.argmax())]
    j_max = peak_idx - i_max

    if seq[i_max] <= 0.0:
        return seq

    # amount to relocate (10‑50 % of the selected entry)
    delta = seq[i_max] * rng.uniform(0.1, 0.5)

    # choose a destination that is not i_max or its partner
    forbid = {i_max, j_max}
    candidates = np.fromiter((k for k in range(n) if k not in forbid), dtype=int)
    if candidates.size == 0:
        return seq

    dest = rng.choice(candidates)

    new_seq = seq.copy()
    new_seq[i_max] = max(0.0, new_seq[i_max] - delta)
    new_seq[dest] += delta
    return new_seq


# ----------------------------------------------------------------------
# Simple annealing / hill climbing for a fixed length
# ----------------------------------------------------------------------
def _anneal_run(seq_init: np.ndarray,
                target_sum: float,
                deadline: float,
                rng: np.random.Generator):
    """
    Perform a short simulated‑annealing run starting from *seq_init*.
    Returns the best sequence found and its max‑convolution value.
    """
    seq = seq_init.copy()
    # enforce the exact L1‑norm (tiny drift may appear after copies)
    cur_sum = seq.sum()
    if cur_sum > 0:
        seq *= (target_sum / cur_sum)
    else:
        seq[:] = target_sum / seq.size

    n = seq.size
    cur_max, cur_idx = _max_convolution_and_index(seq)

    best_seq = seq.copy()
    best_max = cur_max

    # temperature (initially proportional to the current max)
    temperature = max(1e-3, 0.5 * cur_max)
    it = 0
    last_improve = 0

    while time.time() < deadline:
        # ---- propose a move -------------------------------------------------
        # With probability 0.3 we perform a targeted move, otherwise a random transfer
        if rng.random() < 0.3:
            new_seq = _targeted_move(seq, cur_idx, rng)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            if seq[i] > 0.0:
                # transfer a random fraction of the mass at i to j
                frac = rng.uniform(0.01, 0.25)   # step size adapts over time via cooling
                delta = seq[i] * frac
            else:
                delta = 0.0
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

        # ---- evaluate -------------------------------------------------------
        new_max, new_idx = _max_convolution_and_index(new_seq)

        # ---- acceptance -----------------------------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # Metropolis criterion
            if temperature > 0:
                prob = math.exp(-(new_max - cur_max) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            seq = new_seq
            cur_max = new_max
            cur_idx = new_idx
            if new_max < best_max:
                best_max = new_max
                best_seq = new_seq.copy()
                last_improve = it
                print(f"[{time.time():.2f}] n={n}  new local best max={best_max:.8f}")

        # ---- cooling ---------------------------------------------------------
        it += 1
        temperature *= 0.9999   # very slow exponential cooling

        # occasional re‑heat if we have been stagnant for a long time
        if it - last_improve > 5000:
            # reset temperature to a modest fraction of the current value
            temperature = max(temperature, 0.05 * cur_max)
            last_improve = it

        # early abort if temperature becomes vanishingly small
        if temperature < 1e-12:
            break

    return best_seq, best_max
