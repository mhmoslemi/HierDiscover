"""
propose_candidate – multi‑scale, multi‑family gradient search.

The optimiser builds a diverse pool of initial step‑functions (uniform,
Dirichlet draws, analytic windows such as Kaiser, Blackman, Tukey, DPSS,
edge‑concentrated power‑laws, two‑Gaussian edges, sine‑squared bumps, etc.).
Each candidate is refined by two stages:

1. Adam on a smooth‑max surrogate of max(a*a) (log‑sum‑exp with an
   annealed temperature).  The vector is parametrised by logits → soft‑max,
   guaranteeing non‑negativity and unit sum.

2. An exact sub‑gradient method on the true ℓ∞‑norm of the convolution,
   using exponentiated‑gradient updates.

The search proceeds over a cascade of lengths
[1024, 2048, 4096, 8192, 16384, 32768] – each longer problem is warm‑started
from the best solution at the previous size (linear interpolation).  The
iteration budget scales roughly with 1/n, so even the largest problem gets
several thousand Adam steps and a few thousand sub‑gradient steps within
the 1000 s wall‑clock limit.

The best sequence found (as a plain Python list) is returned.
"""

import time
import math
import numpy as np
from scipy import signal

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length a power of two.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]                      # reversed a
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]         # length n


# ----------------------------------------------------------------------
# Candidate factories
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


def _power_law_edge(n: int) -> list[np.ndarray]:
    """Edge‑concentrated power‑law shapes (negative exponents)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)          # i·(n‑i+1)
    out = []
    for alpha in np.linspace(-2.0, -0.5, 4):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            out.append(shape)
    return out


def _two_gaussian_edge(n: int) -> list[np.ndarray]:
    """Two symmetric Gaussians centred at the ends."""
    i = np.arange(n, dtype=np.float64)
    out = []
    for gamma in [0.2, 0.5, 1.0, 2.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        out.append(shape)
    return out


def _sine_squared_bump(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    return shape / shape.sum()


def _kaiser_windows(n: int) -> list[np.ndarray]:
    """Kaiser windows with several beta parameters."""
    betas = [2, 4, 6, 8, 10]
    out = []
    for beta in betas:
        w = signal.windows.kaiser(n, beta, sym=False)
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            out.append(w / w.sum())
    return out


def _blackman_window(n: int) -> np.ndarray:
    w = np.blackman(n, sym=False)
    w = np.maximum(w, 0.0)
    return w / w.sum()


def _tukey_windows(n: int) -> list[np.ndarray]:
    """Tukey windows for a few α values."""
    alphas = [0.1, 0.25, 0.5, 0.75, 0.9]
    out = []
    for alpha in alphas:
        w = signal.windows.tukey(n, alpha, sym=False)
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            out.append(w / w.sum())
    return out


def _exponential_decay(n: int) -> list[np.ndarray]:
    """Symmetric exponential decay centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    out = []
    for c in [0.0005, 0.001, 0.002, 0.005]:
        shape = np.exp(-c * np.abs(i - centre))
        shape = shape / shape.sum()
        out.append(shape)
    return out


def _dpss_windows(n: int) -> list[np.ndarray]:
    """Discrete prolate spheroidal sequences (DPSS). Use the first eigenvector."""
    out = []
    # Scipy's dpss returns shape (k, n); we take the first one.
    for nw in [2, 4, 6]:
        w = signal.windows.dpss(n, nw, Kmax=1, sym=False)[0]
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            out.append(w / w.sum())
    return out


def _cosine_window(n: int) -> np.ndarray:
    """Simple cosine (Hann) window."""
    w = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return w / w.sum()


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
        F(a) = 2·n·max(a*a) / (sum(a)²).

    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None            # numpy array (already normalised)
    best_val = float("inf")

    # Length cascade – larger problems inherit a warm start.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # --------------------------------------------------------------
        # Assemble a heterogeneous pool of starting points
        # --------------------------------------------------------------
        cand_pool = []

        # 1) uniform
        cand_pool.append(_uniform(n))

        # 2) exact previous solution of same size
        if best_seq is not None and best_seq.shape[0] == n:
            cand_pool.append(best_seq.copy())

        # 3) up‑sample a shorter best (linear interpolation)
        if best_seq is not None and best_seq.shape[0] < n:
            old = best_seq
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum() if up.sum() > 0 else np.random.dirichlet(np.ones(n))
            cand_pool.append(up)

        # 4) historic global variable, if it exists
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 0:
                h = h / h.sum()
                cand_pool.append(h)

        # 5) a few random Dirichlet draws
        for _ in range(3):
            cand_pool.append(_random_dirichlet(n))

        # 6) Analytic families
        cand_pool.extend(_power_law_edge(n))
        cand_pool.extend(_two_gaussian_edge(n))
        cand_pool.append(_sine_squared_bump(n))
        cand_pool.append(_blackman_window(n))
        cand_pool.extend(_kaiser_windows(n))
        cand_pool.extend(_tukey_windows(n))
        cand_pool.extend(_exponential_decay(n))
        cand_pool.extend(_dpss_windows(n))
        cand_pool.append(_cosine_window(n))

        # --------------------------------------------------------------
        # Optimise every candidate
        # --------------------------------------------------------------
        for cid, a0 in enumerate(cand_pool):
            if time.time() > deadline:
                break

            # Ensure strictly positive before logarithm
            eps = 1e-16
            a = np.maximum(a0, eps)
            a = a / a.sum()
            logits = np.log(a)

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            # Smooth‑max temperature
            lam = 0.5
            lam_factor = 1.01

            # Adaptive iteration budget – more for smaller n
            max_iters = int(30000 * 1024 / n) + 4000
            last_improve = time.time()

            for it in range(1, max_iters + 1):
                # Soft‑max logits → probability vector (stay on simplex)
                mx = logits.max()
                exp_log = np.exp(logits - mx)          # shift for stability
                a = exp_log / exp_log.sum()

                # Convolution a*a
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()

                # Smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # Gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # Back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # Anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # Exact objective (sum a = 1)
                val = 2.0 * n * max_c

                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam it={it:05d}] new best {val:.6f}")

                # Stagnation handling – small random shake
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # --------------------------------------------------------------
            # Sub‑gradient refinement on the true max (no smoothing)
            # --------------------------------------------------------------
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            max_sub_iters = 2500

            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)

                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cid} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

        # End of length loop – keep the best sequence for the next size.
        # (Optionally expose it as a global for reuse in later calls.)
        if best_seq is not None:
            globals()["height_sequence_1"] = best_seq.copy()

    # ----------------------------------------------------------------------
    # Fallback (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(2048)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
