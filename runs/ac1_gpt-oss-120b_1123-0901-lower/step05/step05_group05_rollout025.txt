def propose_candidate(seed=123, budget_s=1000.0):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_score = np.inf

    # grid search
    n_values = np.arange(800, 3501, 200)
    alphas = np.arange(0.45, 0.71, 0.02)   # inclusive
    for n in n_values:
        target_sum = np.sqrt(2.0 * n)
        idx = np.arange(1, n+1, dtype=float)
        for alpha in alphas:
            raw = idx ** (-alpha)
            a = raw * target_sum / raw.sum()
            val = evaluate_sequence(a.tolist())
            if val < best_score:
                best_score = float(val)
                best_seq = a.copy()
                print(f"[{time.time():.2f}] n={n}, α={alpha:.3f} new best {best_score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # local refinement: small hillclimb
    if best_seq is not None:
        # Run hillclimb for remaining time
        remaining = deadline - time.time()
        if remaining > 5.0:
            best_seq, _, = _simple_hillclimb(best_seq, remaining)
            best_score = evaluate_sequence(best_seq.tolist())

    return best_seq.tolist()
```

def _simple_hillclimb(a_initial, time_budget):
    a = a_initial.copy()
    n = len(a)
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    start = time.time()
    while time.time() - start < time_budget:
        i, j = np.random.randint(0, n, size=2)
        if a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.05   # up to 5%
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        val = evaluate_sequence(a_new.tolist())
        if val < best_val:
            best_val = val
            best_a = a_new
            a = a_new
        else:
            # possibly accept based on decreasing temperature
            # simple: reject
            pass
    return best_a, best_val

"""
Hybrid power‑law search for a step‑function with a small evaluation score.

The evaluator uses 2·n·max(conv(a,a))/(sum(a)**2).  When we fix the
normalisation sum(a)=sqrt(2·n) the score is exactly the maximal convolution
value.  For a monotone power‑law a_i ∝ (i+1)^(-α) the maximal value can be
computed analytically; the optimum α for a given length n is around 0.55–0.6.
We therefore perform a cheap exhaustive search over a grid of lengths and
exponents, keep the best construction and finish with a tiny stochastic
hill‑climb that moves a few percent of mass between two entries while keeping
the total sum exact.  The whole routine respects the 1000 s budget and
returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _generate_powerlaw_sequence(n: int, alpha: float, target_sum: float) -> np.ndarray:
    """
    Return a non‑negative sequence a_i = c·(i+1)^(-α) with Σa_i = target_sum.
    """
    idx = np.arange(1, n + 1, dtype=float)          # i = 1 … n
    raw = np.power(idx, -alpha)                    # unnormalised power‑law
    scale = target_sum / raw.sum()
    return raw * scale


def _hillclimb_refine(a: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Very small stochastic local search.
    Moves a tiny amount of mass from a random index i to a random index j.
    Accepts the move if the exact score improves.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = evaluate_sequence(best.tolist())

    # temperature only for occasional uphill moves – starts relatively hot
    T0 = best_val * 0.2
    T_end = 1e-5
    start = time.time()

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] <= 0.0:
            continue

        # move at most 8 % of the mass currently at i
        delta = best[i] * np.random.rand() * 0.08
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        val = evaluate_sequence(cand.tolist())
        improved = val < best_val
        if improved or np.random.rand() < np.exp(-(val - best_val) / (T + 1e-12)):
            best = cand
            if val < best_val:
                best_val = val
                # occasional progress report
                if int(time.time()) % 30 == 0:
                    print(f"[{time.time():.2f}] hill‑climb new best={best_val:.6f}")

        # early exit if we already beat the target
        if best_val < 1.503:
            break

    return best

def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function with the smallest possible evaluation score.
    Strategy:
      1. Warm‑start with a known construction (if available).
      2. Exhaustive grid search over length n and power‑law exponent α.
      3. Refine the best candidate with a tiny stochastic hill‑climb.
    The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # small safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start – use the provided global construction if it exists.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        sc = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(sc)
        print(f"[{time.time():.2f}] warm‑start score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – grid search over length and power‑law exponent.
    # --------------------------------------------------------------
    # lengths to try (roughly the range used in the previous solution)
    n_candidates = list(range(800, 3501, 200))          # 800,1000,…,3400
    # exponent grid – centred around the analytically promising 0.55‑0.65
    alpha_grid = np.arange(0.45, 0.71, 0.01)           # 0.45,0.46,…,0.70

    for n in n_candidates:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        for alpha in alpha_grid:
            # build the power‑law sequence
            a = _generate_powerlaw_sequence(n, alpha, target_sum)

            # direct call to the provided evaluator (handles all clipping)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = float(score)
                best_seq = a.copy()
                print(f"[{time.time():.2f}] n={n:4d} α={alpha:5.3f}  new best={best_score:.6f}")

            # stop if we already beat the target comfortably
            if best_score < 1.48:
                break
        # optional early stop if very good result already found
        if best_score < 1.48:
            break

    # --------------------------------------------------------------
    # Phase 2 – local stochastic refinement (tiny hill‑climb).
    # --------------------------------------------------------------
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:       # need a few seconds to be worthwhile
            n = best_seq.shape[0]
            target_sum = np.sqrt(2.0 * n)
            refined = _hillclimb_refine(best_seq.copy(), target_sum, time.time() + time_left)
            refined_score = evaluate_sequence(refined.tolist())
            if refined_score < best_score:

                best_seq = refined
                print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed).
    # --------------------------------------------------------------
    if best_seq is None:

        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
