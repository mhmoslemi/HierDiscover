def cross_corr(w, a, fft_len):
    # correlation (w ★ a)[i] = sum_k w[k] * a[i + k - (n-1)]
    # Equivalent to convolution of w with reversed a:
    rev_a = a[::-1]

    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]
```

def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) with a low
evaluation score 2·n·max(a*a)/(sum(a)²).  The method is a mixed
strategy:

* a cheap parametric sweep over edge‑heavy power‑law shapes
  (a_i ∝ (i·(n‑i+1))^α) and double‑exponential tails,
  keeping the best discrete candidate;
* a projected‑gradient descent on a smooth‑max surrogate of the
  convolution max.  The gradient is obtained by a weighted
  cross‑correlation (FFT based) and the update is projected onto the
  probability simplex (entropy‐mirror alternative to the exponentiated‑gradient
  in the previous solution);
* a final up‑sampling to a larger length (if time permits) and another
  short optimisation pass.

The routine respects the given time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater or equal to x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_convolution(a: np.ndarray, conv_fft_len: int,
                            lam: float) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Return (conv, max_c, w) where conv = a*a, max_c = max(conv) and w is the
    smooth‑max weighting (soft‑argmax) with temperature 1/lam.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # smooth‑max weights (log‑sum‑exp)
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return conv, max_c, w


def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  corr_fft_len: int,
                  deadline: float,
                  global_best_seq: np.ndarray | None,
                  global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth‑max surrogate.
    The algorithm updates the best sequence/value seen so far.
    """
    a = a0.copy()
    v_mom = np.zeros_like(a)          # momentum term
    beta = 0.9                        # momentum coefficient
    lr = 0.12                         # base learning rate
    lam = 0.5                         # starting smooth‑max temperature
    lam_factor = 1.5                  # geometric increase of lam
    last_improve = time.time()        # for stagnation handling
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        conv, max_c, w = _smooth_max_convolution(a, conv_fft_len, lam)
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={it:06d}] new best {val:.6f}")

        # gradient of the smooth‑max surrogate
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # momentum + projected‑gradient step
        v_mom = beta * v_mom + (1.0 - beta) * grad
        a_new = a - lr * v_mom
        a = _project_simplex(a_new)

        # temperature schedule – approach the true max
        lam = min(lam * lam_factor, 1e6)

        # learning‑rate decay (very gentle)
        lr *= 0.99997

        # occasional random shake when stalled
        if time.time() - last_improve > 4.5:
            a += (np.random.rand(n) - 0.5) * 2e-3
            a = _project_simplex(a)
            last_improve = time.time()

        it += 1

    return global_best_seq, global_best_val


def _generate_powerlaw_candidates(n: int) -> list[np.ndarray]:
    """Produce edge‑heavy shapes a_i ∝ (i·(n‑i+1))^α for a fine grid of α."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    alphas = np.arange(-2.5, -0.05, 0.05)   # fine sweep
    cand = []
    for alpha in alphas:
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        cand.append(shape)
    return cand


def _generate_double_exp_candidates(n: int) -> list[np.ndarray]:
    """Double‑exponential tails with a range of decay rates."""
    i = np.arange(n, dtype=np.float64)
    gammas = np.logspace(-2, 1.0, 12)   # from 0.01 to 10
    cand = []
    for gamma in gammas:
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)
    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point for the optimiser.
    Returns a list of non‑negative heights (float) that aims to minimise
    the evaluation function within the allotted time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Primary problem size (the earlier state‑of‑the‑art used 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Build a modest pool of start vectors
    candidates = []

    # Warm start (if a previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # resize by simple linear interpolation / truncation
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            candidates.append(warm.copy())
            # also a few cyclic shifts to diversify
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(warm, shift))

    # Random Dirichlet start (baseline)
    candidates.append(np.random.dirichlet(np.ones(n)))

    # Uniform vector (simple baseline)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Sinusoidal half‑wave (positive half‑sine)
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sine = np.maximum(sine, 0.0)
    sine /= sine.sum()
    candidates.append(sine)

    # Add parameterised families
    candidates.extend(_generate_powerlaw_candidates(n))
    candidates.extend(_generate_double_exp_candidates(n))

    # --------------------------------------------------------------
    # First pass – cheap evaluation of all candidates, keep the best few
    # (so that we do not waste the expensive PGD on clearly bad starts)
    elite = []            # (value, vector)
    for cand in candidates:
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        elite.append((val, cand.copy()))
    # keep at most 8 elite vectors for the deep optimisation
    elite.sort(key=lambda x: x[0])
    elite = [vec for _, vec in elite[:8]]

    # --------------------------------------------------------------
    # Deep optimisation on the elite set
    for idx, cand in enumerate(elite):
        if time.time() >= deadline:
            break
        a0 = cand.copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s
        best_seq, best_val = _pgd_optimize(
            a0, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val
        )
        print(f"[n={n} elite {idx}] best so far {best_val:.6f}")

    # --------------------------------------------------------------
    # If time remains, upscale to a larger grid (8192) and refine.
    if time.time() + 8 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Interpolate the best shape found so far to the larger grid.
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq if best_seq is not None else np.zeros(n))
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # A short optimisation run on the larger problem (fewer iterations).
        best_seq, best_val = _pgd_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val
        )
        print(f"[n={n_big}] final best {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – ensure we always return a valid vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
