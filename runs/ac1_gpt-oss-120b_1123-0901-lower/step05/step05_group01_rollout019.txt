"""
Propose a low‑value step‑function sequence.

The algorithm consists of three phases:

1. **Broad family search** – For a chosen length (default 16384) many
   parametric families (Beta‑shapes, edge‑power‑laws, edge‑Gaussians,
   mixtures with the uniform vector, random Dirichlet draws and
   small perturbations of the current best) are sampled.  The best
   few (≈5) based on the exact objective are kept.

2. **Fine‑grained optimisation** – Each kept candidate is refined
   by three successive optimisers:
        * Adam on a smooth‑max surrogate of the convolution peak.
        * Exact sub‑gradient (exponentiated‑gradient) on the true max.
        * Fast hill‑climbing with random mass transfers.
   All three respect the simplex constraint via a soft‑max
   normalisation.  Iteration limits are generous; each optimiser
   aborts automatically when the global deadline is reached.

3. **Late‑stage polishing** – The currently best sequence is fed
   again to a longer Adam run and finally to a simulated‑annealing
   local search that accepts occasional uphill moves.  This phase
   consumes any remaining time budget.

All arithmetic is carried out with NumPy and FFT‑based convolutions,
so a single iteration for a 16384‑vector costs only a few milliseconds.
The method stays well within the 1000 s budget and typically drives
the evaluation value well below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using an FFT of length fft_len.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------

def _objective(a: np.ndarray, fft_len: int) -> float:
    """
    Exact objective 2·n·max(conv) / (sum a)².
    a is assumed non‑negative; the function also works with a small
    numeric noise.
    """
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_c: int,
                   fft_len_x: int,
                   deadline: float,
                   max_iters: int = 10000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_a, best_value).
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04                # a bit smaller than the baseline
    lam = 0.5
    lam_factor = 1.012       # slower temperature rise

    best_a = a.copy()
    best_val = _objective(a, fft_len_c)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max projection onto the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (2 * cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, fft_len_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={t:06d}] new best {cur_val:.6f}")

        # small random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_c: int,
                      fft_len_x: int,
                      deadline: float,
                      max_iters: int = 4000,
                      init_val: float | None = None) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient) on the true max.
    """
    a = a0 / a0.sum()
    if init_val is None:
        cur_val = _objective(a, fft_len_c)
    else:
        cur_val = init_val

    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()
    eta = 0.55

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, fft_len_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it}] new best {cur_val:.6f}")

        # early stop if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_c: int,
                         deadline: float,
                         max_iters: int = 3000,
                         init_val: float | None = None) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small fraction of the probability mass
    between two coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    if init_val is None:
        cur_val = _objective(a, fft_len_c)
    else:
        cur_val = init_val

    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move at most 5 % of the source mass
        eps = np.random.rand() * 0.05 * a[i]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        if new_a[i] < -1e-15:   # numerical safety
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, fft_len_c)
        if new_val < best_val:
            best_val, best_a = new_val, new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 50 == 0:
                print(f"[Hill n={n} it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape plateaus
            if np.random.rand() < 0.001:
                a = new_a

        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val


def _simulated_annealing_optimize(a0: np.ndarray,
                                  n: int,
                                  fft_len_c: int,
                                  deadline: float,
                                  max_steps: int = 15000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement that can accept occasional uphill moves.
    """
    a = a0.copy()
    cur_val = _objective(a, fft_len_c)
    best_a, best_val = a.copy(), cur_val

    # exponential cooling schedule
    T_start = 0.1
    T_end = 1e-5
    t = 0

    while t < max_steps and time.time() < deadline:
        # geometric temperature
        T = T_start * (T_end / T_start) ** (t / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of mass from i to j
        max_move = 0.07 * a[i]          # keep steps modest
        delta = np.random.rand() * max_move
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < -1e-15:
            continue
        new_a /= new_a.sum()
        new_val = _objective(new_a, fft_len_c)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a, cur_val = new_a, new_val
            if new_val < best_val:
                best_val, best_a = new_val, new_a.copy()

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation ---------------------------------------------------

def _generate_candidates(n: int,
                         rng: np.random.Generator,
                         best_seq: np.ndarray | None,
                         hist_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a fairly large and diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # 0 – warm start from current global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())

    # 1 – warm start by interpolating the global best
    if best_seq is not None and best_seq.shape[0] != n:
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        seq = np.interp(x_new, x_old, best_seq)
        seq = np.maximum(seq, 0)
        seq_sum = seq.sum()
        if seq_sum < 1e-12:
            seq = rng.dirichlet(np.ones(n))
        else:
            seq /= seq_sum
        cand.append(seq)

    # 2 – historic sequence (if supplied)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s < 1e-12:
            tmp = rng.dirichlet(np.ones(n))
        else:
            tmp /= s
        cand.append(tmp)

    # 3 – pure random Dirichlet draws
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n)))

    # 4 – symmetric power‑law edge shapes: ((i+1)*(n-i))**α, α<0
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-3.0, -0.2, 8):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 5 – inverse‑root edge shapes: (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.3, 0.5, 0.8, 1.2, 1.6):
        i = np.arange(1, n + 1, dtype=np.float64)
        shape = np.power(i, -alpha) + np.power(n - i + 1, -alpha)
        cand.append(shape / shape.sum())

    # 6 – edge‑Gaussian shapes
    for gamma in (0.2, 0.5, 1.0, 2.0, 3.5):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # 7 – symmetric Beta shapes (α,β > 0)
    for _ in range(12):
        alpha = rng.uniform(0.1, 3.0)
        beta = rng.uniform(0.1, 3.0)
        i = np.arange(n, dtype=np.float64) + 0.5
        shape = np.power(i, alpha - 1.0) * np.power(n + 1.0 - i, beta - 1.0)
        cand.append(shape / shape.sum())

    # 8 – three‑peak Gaussian mixture (peaks at 0.25,0.5,0.75)
    for sigma in (0.04, 0.07, 0.12):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # 9 – mixture of edge‑shape with uniform (λ·edge + (1‑λ)·U)
    uniform = np.full(n, 1.0 / n)
    for _ in range(8):
        lam = rng.random()
        edge = cand[4] if len(cand) > 4 else rng.dirichlet(np.ones(n))
        mix = lam * edge + (1.0 - lam) * uniform
        cand.append(mix / mix.sum())

    # 10 – small perturbation of the current best (if we have one)
    if best_seq is not None:
        if best_seq.shape[0] != n:
            # interpolate
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            seq = np.interp(x_new, x_old, best_seq)
        else:
            seq = best_seq.copy()
        for _ in range(5):
            noisy = seq + rng.normal(scale=0.02, size=n) * seq
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s < 1e-12:
                noisy = rng.dirichlet(np.ones(n))
            else:
                noisy /= s
            cand.append(noisy)

    # Shuffle and keep a moderate pool (≈30) to keep the budget reasonable
    rng.shuffle(cand)
    return cand[:30]


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Historical data (if any) ------------------------------------------------
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to consider – start medium, end large.
    # The large length gives the smallest theoretical bound.
    lengths = [8192, 16384]

    # Pre‑compute FFT lengths for both convolution and cross‑correlation.
    fft_conv = {}
    fft_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_conv[n] = _next_pow2(conv_len)
        # correlation needs conv_len + n - 1 ≤ 3 n
        fft_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    # MAIN optimisation across the selected lengths
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = fft_conv[n]
        fft_len_x = fft_corr[n]

        # ------------------------------------------------------------------
        # 1. Broad family search – keep a few promising starters
        candidates = _generate_candidates(n, rng, best_seq, hist_seq)

        # Quick evaluate all candidates (exact objective, cheap)
        scores = []
        for seq in candidates:
            val = _objective(seq, fft_len_c)
            scores.append(val)

        # select the top few (≤5) for deep optimisation
        top_k = min(5, len(candidates))
        top_idx = np.argsort(scores)[:top_k]
        promising = [candidates[i] for i in top_idx]

        # ------------------------------------------------------------------
        # 2. Fine‑grained optimisation of each promising candidate
        for idx, cand in enumerate(promising):
            if time.time() > deadline:
                break
            print(f"[n={n}] deep‑optimising candidate {idx+1}/{top_k}")

            # --- Adam (smooth‑max) ---------------------------------------
            # aim for roughly 10 000 iterations for the largest n;
            # the optimiser stops early if the deadline is hit.
            max_adam_iters = int(12000 * 1024 / n) + 4000
            a_adam, val_adam = _adam_optimize(
                cand, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # --- Exact sub‑gradient --------------------------------------
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,
                max_iters=6000, init_val=val_adam
            )

            # --- Hill‑climb -----------------------------------------------
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline,
                max_iters=5000, init_val=val_sub
            )

            if val_hc < best_val:
                best_val, best_seq = val_hc, a_hc.copy()
                print(f"*** Global new best (n={n} cand={idx}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Long‑run polishing of the global best (if any time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = fft_conv.get(n, _next_pow2(2 * n - 1))
        fft_len_x = fft_corr.get(n, _next_pow2(3 * n))
        print("\n=== Final polishing of the overall best ===")
        # Longer Adam run
        a_refined, val_refined = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(20000 * 1024 / n) + 5000
        )
        if val_refined < best_val:
            best_val, best_seq = val_refined, a_refined.copy()
            print(f"*** Refined with Adam value={best_val:.6f}")

        # Simulated annealing for final tweaks
        a_final, val_final = _simulated_annealing_optimize(
            best_seq, n, fft_len_c, deadline, max_steps=20000
        )
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f"*** Refined with SA value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\n=== Finished. Best value = {best_val:.6f} with length {best_seq.shape[0]} ===")
    return best_seq.tolist()
