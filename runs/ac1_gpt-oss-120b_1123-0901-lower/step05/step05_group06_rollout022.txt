```
def _project_simplex(v, s):
    """Project onto simplex {x>=0, sum x = s}."""
    # from https://gist.github.com/...
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
    Nfft = 1 << ((2*n - 1).bit_length())

    # maybe compute initial score
    conv = np.fft.irfft(np.fft.rfft(a, n=Nfft) * np.fft.rfft(a, n=Nfft), n=Nfft)[:2*n-1]
    max_b = float(conv.max())
    best_score_local = max_b

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
```

```
def _local_refine_incremental(a, n, deadline, best_score, rng):
    target_sum = a.sum()  # should be sqrt(2n)
    a_cur = a.copy()
    conv = np.fft.irfft(np.fft.rfft(a_cur, n=1 << ((2*n-1).bit_length())) *
                       np.fft.rfft(a_cur, n=1 << ((2*n-1).bit_length())), n=1 << ((2*n-1).bit_length()))[:2*n-1]
    cur_score = float(conv.max())
    # parameters
    max_moves = 200_000
    move = 0
    # track decay temperature
    while time.time() < deadline and move < max_moves:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a_cur[i] < 1e-12:
            move += 1
            continue
        frac = rng.uniform(0.005, 0.02)  # small transfers
        delta = a_cur[i] * frac
        a_cur[i] -= delta
        a_cur[j] += delta

        # update conv via slice
        # subtract contributions of original a_i (i before move) but we have lost original a_i value.
        # We'll just recompute conv after each move as fallback not heavy for moderate moves.
        A = np.fft.rfft(a_cur, n=1 << ((2*n-1).bit_length()))
        conv = np.fft.irfft(A*A, n=1 << ((2*n-1).bit_length()))[:2*n-1]
        new_score = float(conv.max())

        # acceptance
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            # keep conv
        else:
            # if not accepted, revert move
            # but allow annealing acceptance with probability based on temperature
            T = max(0.01 * cur_score, 1e-5) * (1.0 - move / max_moves)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                a_cur[j] -= delta
                a_cur[i] += delta
                # conv revert back: recompute conv to be safe
                A = np.fft.rfft(a_cur, n=1 << ((2*n-1).bit_length()))
                conv = np.fft.irfft(A*A, n=1 << ((2*n-1).bit_length()))[:2*n-1]
        move += 1
        # occasionally output progress
        if move % 5000 == 0:
            print(f"[refine] step {move}, best max={cur_score:.6f}")
    return a_cur, cur_score
```

The objective is  

\[
E(a)=\frac{2n\;\max\limits_{k}(a\ast a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where `*` denotes the discrete convolution.  
Because the function is homogeneous, scaling does not change the value; we may (and do) keep the
\(L_1\)-norm fixed to  

\[
\sum_i a_i = \sqrt{2n}.
\]

Hence minimising `evaluate_sequence` is exactly the same as minimising the largest entry
of the self–convolution under a simplex constraint.

The algorithm is split into three stages

1. **Global exploration** – a large but cheap set of candidate shapes
   (uniform, power‑law, exponential, linear, Dirichlet samples, interpolations of the
   previously known construction).  
   For each candidate we compute the exact convolution by FFT and keep the best one.

2. **Fine optimisation** – a projected sub‑gradient method.
   In every iteration we
   * compute the whole convolution,
   * pick the index where the maximum occurs,
   * build the exact sub‑gradient (the reversed vector aligned with that index),
   * take a diminishing‑step gradient step and project back onto the simplex.
   Adam on the logits of the soft‑max (as in the previous baseline) is replaced by this
   simple projected sub‑gradient descent; it converges reliably and removes the need for
   temperature tuning.

3. **Local stochastic refinement** – random pairwise mass transfers
   (move a small fraction of mass from one coordinate to another) with a
   simulated‑annealing acceptance rule.  
   After each move the full convolution is recomputed (FFT, cheap for the lengths used).
   This stage aggressively removes the last remaining peaks.

All work is performed inside the 1000 s budget; a 5 s safety margin is kept.
The implementation uses only `numpy` and the already available
`evaluate_sequence`.  No file‑ or network‑access is required.

```python
# Helper utilities (all top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s <= 0:
        raise ValueError("simplex radius must be positive")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, s / n, dtype=float)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> float:
    """Return max_k (a∗a)[k] using an FFT of appropriate length."""
    n = a.shape[0]
    Nfft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _full_conv(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2n‑1)."""
    n = a.shape[0]
    Nfft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return conv


def _rev_for_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a∗a)[k] at the current maximum index.
    Returns rev where rev[i] = a[max_idx‑i] for all i for which the index is valid.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    # rev[lo:hi+1] = a[max_idx - (lo..hi)]
    rev[lo : hi + 1] = a[max_idx - np.arange(lo, hi + 1)]
    return rev


def _shape_vectors(n: int, target_sum: float, rng: np.random.Generator):

    Produce a diverse list of low‑dimensional start vectors of length `n`.
    All vectors are normalised to the prescribed L1‑norm `target_sum`.

    res = []

    # 1) uniform
    res.append(np.full(n, target_sum / n))

    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    res.append(g / g.sum() * target_sum)

    # 3) exponential families
    for r in [0.998, 0.999, 0.9995, 0.9998, 0.9999]:
        v = r ** np.arange(n)
        res.append(v / v.sum() * target_sum)

    # 4) power‑law families a_i = (i+1)^{-p}
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        v = (np.arange(1, n + 1, dtype=float)) ** (-p)
        res.append(v / v.sum() * target_sum)

    # 5) linear‑power families (1‑i/(n‑1))^{p}
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        v = 1.0 - np.arange(n) / (n - 1)
        v = np.maximum(v, 0.0) ** p
        res.append(v / v.sum() * target_sum)

    # 6) small random perturbations of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    res.append(pert / pert.sum() * target_sum)

    return res


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector and its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    # start from the supplied vector (already normalised)
    a = a0.copy()

    # bookkeeping
    best_local = a.copy()
    best_local_score = best_score

    # iteration counter for step‑size schedule
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution and max index ----
        conv = _full_conv(a)
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])

        # keep global best seen so far
        if max_val < best_local_score:
            best_local_score = max_val
            best_local = a.copy()
            print(f"[len {n:5d} it {it:8d}] new best max={max_val:.6f}")

        # ---- exact sub‑gradient (2·rev) ----
        rev = _rev_for_max(a, max_idx)
        grad = 2.0 * rev

        # ---- decreasing step size (α / sqrt(t)) ----
        step = 0.2 * target_sum / (np.linalg.norm(grad) + 1e-12) / math.sqrt(it)

        # ---- gradient step and projection onto the simplex ----
        a_candidate = a - step * grad
        a = _project_onto_simplex(a_candidate, target_sum)

    # compare with the best sequence passed from outside
    if best_local_score < best_score:
        best_score = best_local_score
        best_seq = best_local

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex.
    Randomly moves a small fraction of mass from one index to another.
    Accepts improvements always; worsening moves with a decreasing temperature.
    Returns an improved vector (or the original if time runs out).
    """
    cur = a.copy()
    cur_score = best_score

    # pre‑compute convolution length and FFT size for fast updates
    Nfft = _next_pow2(2 * n - 1)

    move = 0
    max_moves = 200_000

    while time.time() < deadline and move < max_moves:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            move += 1
            continue

        # transfer a random fraction (1% – 5%) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur[i] * frac

        # propose new sequence
        cur[i] -= delta
        cur[j] += delta

        # recompute the full convolution (fast FFT)
        conv = _full_conv(cur)
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            if move % 5000 == 0:
                print(f"[refine] improvement → {cur_score:.6f} (move {move})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - move / max_moves)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert move
                cur[j] -= delta
                cur[i] += delta

        move += 1

    return cur, cur_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (up to 8000)
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100  # historic fallback
    length_factors = [0.5, 0.75, 1.0, 1.5, 2.0, 2.5, 3.0]
    lengths = sorted(
        {max(300, min(int(n0 * f), 8000)) for f in length_factors}
    )

    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – global shape exploration + projected sub‑gradient
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # diverse deterministic starts
        init_vectors = _shape_vectors(n, target_sum, rng)

        # add interpolated previous best if available
        if base is not None:
            if base.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, base.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                interp = np.interp(new_x, old_x, base)
                interp = np.maximum(interp, 0.0)
                interp = interp / interp.sum() * target_sum
                init_vectors.append(interp)
            else:
                init_vectors.append(base.copy())

        # iteration budget (more iterations for smaller n)
        max_iters = int(1_500_000 * 2100 / n)
        max_iters = max(300_000, max_iters)

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            # safeguard: non‑negative and correctly normalised
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum() * target_sum

            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            best_seq, best_score = _subgradient_optimize(
                a0=a0,
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Stage 2 – local stochastic refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most half of the remaining time to refinement
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback – very unlikely but guarantees a return value
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _conv_max(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
```
