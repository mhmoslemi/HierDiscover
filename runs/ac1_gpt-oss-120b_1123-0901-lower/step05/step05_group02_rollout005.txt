"""
Step‑function optimizer for the objective

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm combines several ideas that were not all used together before:

* A rich pool of *analytic* initial shapes (edge‑power‑law, two‑sided
  exponential, arcsine‑law, sine‑law, spiky Dirichlet …) – more expressive
  than the plain Dirichlet draws of the reference code.
* An Adam optimiser on a *smooth‑max* surrogate of the convolution peak.
  The temperature (λ) is increased very slowly (factor 1.01) so that the
  surrogate stays smooth long enough for the optimiser to make progress.
* A projected *exponentiated‑gradient* phase that works directly on the
  true max‑convolution (the same sub‑gradient used in the reference).
* A final *simulated‑annealing* refinement that moves a tiny amount of
  mass between two random coordinates and accepts uphill moves with a
  decreasing temperature.  This cheap stochastic search can escape the
  basin of attraction of the deterministic phases.
* Time‑budget awareness: every major loop checks the global deadline and
  stops early if the remaining time is too small.

The code is deliberately written with only NumPy and the standard library
so that it runs on the execution sandbox.  All helper functions are
top‑level (no closures) and the returned object is a plain Python list as
required by the external `evaluate_sequence` routine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast, exact for the sizes we use)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len`` (real)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Both ``w`` and ``a`` are real, ``w`` has length 2·n‑1, ``a`` length n.
    Returns a vector of length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective_from_conv_max(n: int, conv_max: float) -> float:
    """Given n and max(conv) (with sum(a)=1) return the exact objective."""
    return 2.0 * n * conv_max


# ----------------------------------------------------------------------
# Analytic initial shapes (all give a non‑negative vector, sum=1)
# ----------------------------------------------------------------------
def _initial_shapes(n: int) -> list[np.ndarray]:
    """Return a list of diverse starting vectors of length n (already normalised)."""
    shapes = []

    # 1. Uniform (baseline)
    shapes.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. Random Dirichlet (concentration 1)
    shapes.append(np.random.dirichlet(np.ones(n)))

    # 3. Spiky Dirichlet (concentration < 1)
    for conc in (0.2, 0.5, 0.8):
        shapes.append(np.random.dirichlet(np.full(n, conc)))

    # 4. Edge‑concentrated power‑law ((i+1)*(n-i))**α, α negative
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            shapes.append(shape)

    # 5. Two‑sided exponential
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        shapes.append(shape)

    # 6. Symmetric “tent” (linear rise & fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    tent = np.maximum(tent, 0)
    tent /= tent.sum()
    shapes.append(tent)

    # 7. Arcsine law approximation (heavy at the borders)
    eps = 1e-6
    idx = np.arange(n, dtype=np.float64) + 0.5
    shape = 1.0 / np.sqrt((idx + eps) * (n + 0.5 - idx + eps))
    shape /= shape.sum()
    shapes.append(shape)

    # 8. Half‑sine (smooth bell)
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = np.sin(np.pi * i / n) ** 2
    shape /= shape.sum()
    shapes.append(shape)

    # 9. Mixture of arcsine & uniform (to increase interior mass)
    mix = 0.3
    shape = mix * (1.0 / n) + (1 - mix) * shape
    shape /= shape.sum()
    shapes.append(shape)

    # 10. Few random spikes (to give the optimiser something non‑smooth)
    spikes = np.zeros(n, dtype=np.float64)
    for _ in range(max(2, n // 2000)):
        pos = np.random.randint(0, n)
        spikes[pos] = np.random.rand() + 0.5
    spikes /= spikes.sum()
    shapes.append(spikes)

    # Ensure every entry is non‑negative and normalised
    for i, v in enumerate(shapes):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            v = np.random.dirichlet(np.ones(n))
        else:
            v /= s
        shapes[i] = v.astype(np.float64)

    return shapes


# ----------------------------------------------------------------------
# Deterministic optimiser (Adam → Exponentiated‑gradient)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, conv_len: int,
                corr_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate.
    Returns the best (a, value) discovered during this phase.
    """
    # initialise the soft‑max parameters (logits)
    logits = np.log(a0 + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04                # modest learning rate
    lam = 0.2                # start temperature for smooth‑max
    lam_factor = 1.01        # slow increase → smoother optimisation

    best_a = None
    best_val = float("inf")
    last_improve = time.time()

    # iteration budget – scale with length (larger n → fewer iterations)
    max_iters = int(30000 * 1024 / n) + 4000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → a lives on the probability simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate (linear in a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # back‑prop through the soft‑max (the usual “log‑soft‑max” trick)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a) == 1)
        val = _objective_from_conv_max(n, max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new best {val:.6f}")

        # occasional random shake if we are stuck
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 6e-3
            last_improve = time.time()

    # return the best found inside the Adam phase
    return best_a, best_val


def _expgrad_phase(a0: np.ndarray, n: int, conv_len: int,
                   corr_len: int, deadline: float,
                   current_best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (sub‑gradient) descent on the *exact* max.
    Returns the best (a, value) discovered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = current_best_val
    eta = 0.55                      # step‑size
    sub_last_improve = time.time()
    max_sub_iters = 6000

    for it in range(max_sub_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()

        # indices where the max is attained (active set)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # exponentiated‑gradient step (multiplicative update on simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:          # guard against numerical collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = _objective_from_conv_max(n, max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # simple stagnation cut‑off
        if time.time() - sub_last_improve > 8.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic refinement (simulated annealing)
# ----------------------------------------------------------------------
def _anneal_phase(a0: np.ndarray, n: int, conv_len: int,
                  deadline: float, start_val: float) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing that moves a tiny amount of mass between two
    random coordinates.  The temperature schedule is geometric.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    best_a = a.copy()
    best_val = start_val

    # temperature roughly covers the magnitude of the improvement we still
    # hope to achieve; we start a little above the current excess over the
    # theoretical lower bound (2·n/(2·n‑1) ≈ 1) and decay to near zero.
    lower_bound = 2.0 * n / (2 * n - 1)   # theoretical infimum for given n
    T0 = max((best_val - lower_bound) * 0.5, 1e-3)
    Tf = 1e-6
    total_steps = 4000

    # pre‑compute the FFT length for convolution (same as other phases)
    # (conv_len already is a power‑of‑two ≥ 2·n‑1)
    for step in range(total_steps):
        if time.time() > deadline:
            break

        T = T0 * (Tf / T0) ** (step / total_steps)

        # propose moving a small fraction of mass from i → j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # pick a delta that never empties i completely
        max_delta = min(a[i], 0.2)               # cap the jump size
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate the new candidate (exact max convolution)
        conv = _full_convolution(a_new, conv_len)
        val = _objective_from_conv_max(n, conv.max())

        # Metropolis acceptance
        if val < best_val or rng.random() < math.exp(-(val - best_val) / (T + 1e-12)):
            a = a_new
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                print(f"[n={n} anneal step={step}] new best {val:.6f}")

    return best_a, best_val

# Main entry point (the function that the scorer will call)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a wall‑clock budget and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore.  4096 gave the best result previously; we also
    # give 8192 a chance if time permits – longer sequences have a lower
    # theoretical lower bound.
    # ------------------------------------------------------------------
    lengths = [4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))     # power‑of‑two FFT size
        # For cross‑correlation we need a length at least 3·n‑2
        corr_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # ------------------------------------------------------------------
        # Build a diverse pool of initial candidates
        # ------------------------------------------------------------------
        candidates = _initial_shapes(n)

        # If we already have a good global best of the same size, warm‑start it
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # ------------------------------------------------------------------
        # Optimise every candidate independently
        # ------------------------------------------------------------------
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a0 = cand / cand.sum()

            # ----- Adam on smooth‑max -----
            a_adam, val_adam = _adam_phase(a0, n, conv_len, corr_len, deadline)

            # ----- Exponentiated‑gradient on exact max -----
            a_exp, val_exp = _expgrad_phase(a_adam, n, conv_len, corr_len,
                                            deadline, val_adam)

            # Keep the best of the deterministic phases
            if val_exp < best_val:
                best_val = val_exp
                best_seq = a_exp.copy()
                print(f"[n={n} cand={cid}] deterministic best {val_exp:.6f}")

            # ----- Simulated annealing refinement (only on the current
            #       global best for this length, not on every candidate) -----
            # we apply it after we have at least one feasible sequence
            # for the current length.
            if best_seq is not None and len(best_seq) == n:
                a_anneal, val_anneal = _anneal_phase(best_seq, n,
                                                    conv_len, deadline,
                                                    best_val)
                if val_anneal < best_val:
                    best_val = val_anneal
                    best_seq = a_anneal.copy()
                    print(f"[n={n}] post‑anneal improved to {val_anneal:.6f}")

        # ------------------------------------------------------------------
        # Optional fine‑tuning pass on the *global* best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] final fine‑tune (current best {best_val:.6f})")
            a_fine, val_fine = _adam_phase(best_seq, n, conv_len,
                                           corr_len, deadline)
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] fine‑tune improved to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Fallback: if nothing was discovered (should never happen) return a
    # uniform vector of a modest size.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
