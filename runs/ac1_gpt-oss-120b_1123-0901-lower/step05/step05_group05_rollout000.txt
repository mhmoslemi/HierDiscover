"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines three ideas:

1. **Warm start** – use the best known construction (``height_sequence_1``) if
   available, otherwise initialise a random Dirichlet vector.

2. **Linear‑programming refinement** – iterate the classical LP update:
   given a current shape ``f`` compute ``M = max(conv(f,f))`` and solve

       maximise   Σ b_j
       subject to  conv(f, b) ≤ M   (for every shift)
                   b_j ≥ 0

   The solution ``b`` is then scaled to the required simplex sum
   ( √(2 n) ) and mixed with ``f`` ( ``f ← (1‑t) f + t b`` ).  A few
   iterations of this update dramatically lower the convolution peak.

3. **Fine tuning** – a short projected sub‑gradient descent followed by a
   stochastic hill‑climb, both operating on the exact objective
   ``max(conv(a,a))``.

All steps respect the global time budget (default 1000 s) and keep the
best sequence found.  The returned object is a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)  # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Project onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build the (2n‑1) × n convolution matrix for a fixed left vector f:
#    conv(f, g) = A @ g
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
# Linear‑programming refinement (the iterative LP described in the
# problem statement).  Returns the refined sequence and its max‑conv
# value (the exact objective, not the scaled version).
# ----------------------------------------------------------------------
def _lp_refine(
    f0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    mix_factor: float = 0.20,
    max_iter: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Perform up to ``max_iter`` LP refinements of ``f0`` while time permits.
    Each iteration solves:
        maximise   Σ b_j
        subject to  conv(f, b) ≤ M
                    b ≥ 0
    where M = max(conv(f,f)).  The solution b is scaled to the required
    simplex sum and mixed back into f.
    """
    f = f0.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_val = _full_convolution_fft(f, f).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # current peak
        conv_ff = _full_convolution_fft(f, f)
        M = conv_ff.max()

        # build linear constraints conv(f, b) ≤ M
        A = _conv_matrix(f)                 # (2n‑1) × n
        b_ub = np.full(A.shape[0], M)       # RHS

        # linear programme: maximise Σ b_j  → minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)

        # solve
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "disp": False},
        )

        if not res.success:
            # If the LP fails (unlikely), abort the refinement.
            print(f"[{time.time():.2f}] LP iteration {it} failed, stopping LP phase.")
            break

        b = res.x
        b_sum = b.sum()
        if b_sum <= 0:
            # degenerate solution – stop
            print(f"[{time.time():.2f}] LP iteration {it} gave zero sum, aborting.")
            break

        # scale to the required simplex radius
        scale = target_sum / b_sum
        g = b * scale

        # mix the new vector into the current one
        f_new = (1.0 - mix_factor) * f + mix_factor * g

        # evaluate new max convolution
        new_val = _full_convolution_fft(f_new, f_new).max()

        # accept if we have improved (allow tiny tolerance)
        if new_val < best_val - 1e-12:
            f = f_new
            best_f = f_new.copy()
            best_val = new_val
            print(
                f"[{time.time():.2f}] LP it={it:02d} mix={mix_factor:.2f} "
                f"new max={best_val:.6f}"
            )
            # optional: slowly reduce the mixing factor to make later steps finer
            mix_factor = max(0.05, mix_factor * 0.95)
        else:
            # no improvement – shrink step size more aggressively
            mix_factor = max(0.02, mix_factor * 0.80)

        # early exit if already excellent
        if best_val < 1.503:
            break

    return best_f, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution.
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n  # initial step
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())  # index of the current maximum

        # construct sub‑gradient of the max entry
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # gradient step + projection
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate.copy()
            best_val = val_candidate
            step_size *= 1.02
        else:
            step_size *= 0.70

        # occasional jitter to keep exploration alive
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it:04d} step={step_size:.3e} best={best_val:.6f}"
            )

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the exact objective.
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Metropolis hill‑climber that moves a tiny amount of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    T0 = best_val * 0.30
    T_end = 1e-4
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        delta = a[i] * np.random.rand() * 0.10  # move up to 10% of source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it:06d} new best={best_val:.6f}")

        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a step‑function with a minimal evaluation score.
    The routine respects the given time budget, runs an LP‑based refinement,
    and finishes with sub‑gradient descent and stochastic hill‑climbing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Warm‑start from the globally supplied construction if available.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start score={best_score:.6f} (n={n0})")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # If no warm‑start, initialise a random Dirichlet vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = np.random.randint(1800, 2800)  # moderate size to keep LP cheap
        target_sum = np.sqrt(2.0 * n)
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum() * target_sum
        best_seq = a0.copy()
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] random init n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP refinement (core of the new approach)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        lp_deadline = time.time() + min(300.0, (deadline - time.time()) * 0.55)
        seq_lp, val_lp = _lp_refine(best_seq, target_sum, lp_deadline, mix_factor=0.20)
        if val_lp < best_score:
            best_score = val_lp
            best_seq = seq_lp.copy()
            print(f"[{time.time():.2f}] after LP best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.30)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic hill‑climb for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return as plain Python list
    return best_seq.tolist()
