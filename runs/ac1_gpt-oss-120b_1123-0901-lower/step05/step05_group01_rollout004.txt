def proj_simplex(v):
    """Return projection of v onto the probability simplex (non-negative sum=1)."""
    # Standard algorithm: sort descending, find threshold
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    ind = np.arange(len(v)) + 1
    cond = v_sorted - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers (FFT based)
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective (exact, no clipping)
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Simple interpolation used for warm‑starts
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


# ----------------------------------------------------------------------
# Candidate generation – we keep the broad palette from the reference code
def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
    max_cand: int = 15,
) -> list[np.ndarray]:
    """Return a diverse list (≤max_cand) of non‑negative vectors that sum to 1."""
    cand = []

    # 1) warm‑start from the current global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # 2) warm‑start from the current global best interpolated to this length
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # 3) historic sequence (if any)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # 4) pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 5) edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 6) two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # 7) inverse‑root edge shapes
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # 8) three‑Gaussian peaks
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # Shuffle and keep at most max_cand
    np.random.shuffle(cand)
    return cand[:max_cand]


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate of max(conv)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
    lr: float = 0.05,
    lam_start: float = 0.5,
    lam_factor: float = 1.03,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns (best_vector, best_value) encountered before the deadline.
    """
    # initialise on the simplex via soft‑max over logits
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lam = lam_start
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- simplex projection via soft‑max ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- smooth‑max weights ----
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (increase λ)
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # gentle random shake if stuck for a few seconds
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient optimisation (exponentiated‑gradient) with diminishing step size
def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
    eta_start: float = 0.6,
    eta_decay: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient (mirror‑descent / exponentiated‑gradient) optimisation.
    Step‑size decays geometrically.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # active set where convex is (almost) maximal
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        # decay step‑size
        eta *= eta_decay

        # stop early if no improvement for a few seconds
        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climbing local search (mass shift between two indices)
def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple local search: move a tiny amount of probability mass
    between two random coordinates and keep the move if it improves.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # transfer up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises
    the evaluation function.  Returns the best sequence found as a plain
    Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # numpy array, normalised to sum=1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Load historic solution if available
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # Lengths we will explore – larger lengths give more flexibility.
    # We start with medium sizes, finish with the biggest size.
    lengths = [4096, 8192, 16384]

    # Pre‑compute FFT lengths for convolution and correlation.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # correlation needs conv_len + n - 1 ≤ 3·n
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # MAIN optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # Build a diverse set of starting points (max 15)
        candidates = _generate_initial_candidates(n, best_seq, hist_seq, max_cand=15)

        # Number of Adam iterations – longer for smaller n, but still modest for the 1000 s budget.
        max_adam_iters = int(20000 * 1024 / n) + 3000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline,
                max_iters=max_adam_iters,
                lr=0.05, lam_start=0.5, lam_factor=1.03,
            )

            # ---- Phase 2 – exact sub‑gradient ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,
                init_val=val_adam,
                max_iters=2500,
                eta_start=0.6,
                eta_decay=0.9995,
            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=val_sub, max_iters=1500
            )

            # Update global best if improved.
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional long‑run refinement on the best sequence (if time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n, 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        # Longer Adam run with a slower learning rate for fine tuning
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            fft_len_c,
            fft_len_x,
            deadline,
            max_iters=int(2.5 * (20000 * 1024 / n) + 6000),
            lr=0.02,
            lam_start=0.1,
            lam_factor=1.05,
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best after Adam: {best_val:.6f}")

        # One more sub‑gradient pass with a smaller step‑size
        refined_seq2, refined_val2 = _subgrad_optimize(
            best_seq,
            n,
            fft_len_c,
            fft_len_x,
            deadline,
            init_val=best_val,
            max_iters=3000,
            eta_start=0.4,
            eta_decay=0.998,
        )
        if refined_val2 < best_val:
            best_val = refined_val2
            best_seq = refined_seq2.copy()
            print(f"*** Refined global best after sub‑gradient: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\nFinal best value = {best_val:.6f}")
    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
