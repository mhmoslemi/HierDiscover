def _cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)

    return corr[start:start + n]

def _adam_optimize(initial_a, n, fft_conv_len, fft_corr_len,
                   deadline, max_iters, lr=0.05, lam_start=0.01,
                   lam_factor=1.01, epsilon=1e-15):
    a = initial_a.copy()
    logits = np.log(a + epsilon)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_start
    best_a = a.copy()
    conv = _full_convolution(a, fft_conv_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c
    last_improve = time.time()
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute a from logits (softmax)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # convolution
        conv = _full_convolution(a, fft_conv_len)
        max_c = conv.max()
        # smooth max weighting
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_corr_len)
        # gradient wrt logits (softmax backprop)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # increase lam
        lam = min(lam * lam_factor, 1e6)
        # evaluate exact value
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # optional prints can be done by caller
        # Stagnation: occasionally add jitter
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
    # return best
    return best_a, best_val

def _expgrad_optimize(initial_a, n, fft_conv_len, fft_corr_len,
                     deadline, max_iters=5000, eta=0.5):
    a = initial_a.copy()
    best_a = a.copy()
    conv = _full_convolution(a, fft_conv_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c
    last_improve = time.time()
    eps = 1e-15
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, fft_conv_len)
        max_c = conv.max()
        # active set mask
        mask = np.isclose(conv, max_c, atol=1e-12)
        # if few active points, uniform weight
        if mask.any():
            w = mask.astype(float)
            w /= w.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)
        grad_a = 2.0 * _cross_correlation(w, a, fft_corr_len)
        # EG step
        a = a * np.exp(-eta * grad_a)
        # renorm
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        # evaluate
        conv = _full_convolution(a, fft_conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        if time.time() - last_improve > 8.0:
            break
    return best_a, best_val

def _generate_candidates(n, best_seq=None):
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))
    # if best_seq matches n
    if best_seq is not None and len(best_seq) == n:
        candidates.append(np.asarray(best_seq, dtype=np.float64))
    # Dirichlet random
    candidates.append(np.random.dirichlet(np.ones(n)))
    candidates.append(np.random.dirichlet(np.full(n, 0.3))) # spiky
    candidates.append(np.random.dirichlet(np.full(n, 0.5)))
    # power-law
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)
    # exponential
    for gamma in [0.0005, 0.001, 0.002, 0.005]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        candidates.append(shape)
    # mixture combos
    for _ in range(3):
        w1 = np.random.rand()
        w2 = 1 - w1
        shape = w1 * np.random.dirichlet(np.ones(n)) + w2 * np.random.dirichlet(np.full(n, 0.3))
        shape = shape / shape.sum()
        candidates.append(shape)
    return candidates

    break the symmetry of the continuous solution.

    deadline = time.time() + budget_s - 0.5

    lengths = [4096, 8192, 16384]

        fft_corr_len = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

   budget).  Because the autocorrelation depends on the order of the
   entries, a suitable permutation can lower the peak further.

The routine respects a user supplied time budget (default 1000 s) and
prints intermediate progress.  It returns a plain Python list of floats,
ready to be fed to the external ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _adam_optimize(initial_a: np.ndarray,
                   n: int,
                   fft_conv_len: int,
                   fft_corr_len: int,
                   deadline: float,
                   max_iters: int,
                   lr: float = 0.05,
                   lam_start: float = 0.01,
                   lam_factor: float = 1.01,
                   epsilon: float = 1e-15) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate of max(a*a).
    Parameters are kept deliberately simple – the function returns the
    best sequence found (as a probability vector, sum=1) and its exact
    objective value 2·n·max(conv).
    """
    a = initial_a.copy()
    logits = np.log(a + epsilon)                # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a.copy()
    conv = _full_convolution(a, fft_conv_len)
    best_val = 2.0 * n * conv.max()
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max (projects logits onto the simplex)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_corr_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum = 1 ⇒ denominator = 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional random shake to escape stagnation
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _expgrad_optimize(initial_a: np.ndarray,
                      n: int,
                      fft_conv_len: int,
                      fft_corr_len: int,
                      deadline: float,
                      max_iters: int = 5000,
                      eta: float = 0.55,
                      epsilon: float = 1e-15) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation that works directly
    on the true max‑convolution.  Returns the best vector found and its
    exact objective value.
    """
    a = initial_a.copy()
    best_a = a.copy()
    conv = _full_convolution(a, fft_conv_len)
    best_val = 2.0 * n * conv.max()
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_conv_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = active.astype(float)
            w /= w.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        grad_a = 2.0 * _cross_correlation(w, a, fft_corr_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # guard against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_conv_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val


def _permute_search(start_a: np.ndarray,
                    n: int,
                    fft_conv_len: int,
                    deadline: float,
                    cur_best_val: float,
                    max_trials: int = 5000) -> tuple[np.ndarray, float]:
    """
    Random permutation search: repeatedly permute the entries of ``start_a``
    and keep the permutation that yields the smallest exact objective.
    The number of trials is limited by ``max_trials`` and the remaining
    time budget.
    """
    best_a = start_a.copy()
    best_val = cur_best_val
    trials = 0

    while time.time() < deadline and trials < max_trials:
        perm = np.random.permutation(n)
        a = best_a[perm]                     # permute the *current* best
        conv = _full_convolution(a, fft_conv_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
        trials += 1

    return best_a, best_val


def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest set of diverse initial vectors of length ``n``.
    All vectors are normalised to have sum = 1.
    """
    candidates: list[np.ndarray] = []

    # 1. warm‑start if available
    if warm_start is not None and len(warm_start) == n:
        candidates.append(np.asarray(warm_start, dtype=np.float64))

    # 2. uniform vector (baseline)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3. random Dirichlet samples (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))                     # uniform concentration
    candidates.append(np.random.dirichlet(np.full(n, 0.3)))                # spiky
    candidates.append(np.random.dirichlet(np.full(n, 0.5)))                # medium

    # 4. power‑law shapes ((i+1)*(n-i))**α with α<0  (edge‑focused)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0.0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. two‑sided exponential shapes
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. random convex combinations of two simple families
    for _ in range(3):
        w = np.random.rand()
        a = w * np.random.dirichlet(np.ones(n))
        b = (1 - w) * np.random.dirichlet(np.full(n, 0.3))
        shape = a + b
        shape /= shape.sum()
        candidates.append(shape)

    return candidates


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  The function respects a time budget and returns the
    best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Lengths to explore – start with a medium size, then go larger if time permits
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        conv_len = 2 * n - 1
        fft_conv_len = _next_pow2(conv_len)
        fft_corr_len = _next_pow2(conv_len + n - 1)

        # Build a list of initial candidates
        candidates = _generate_candidates(n, best_seq)

        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure we work with a proper probability vector
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                continue
            a0 = a0 / a0.sum()

            # ----- Adam on smooth‑max -----
            max_iters = int(30000 * 1024 / n) + 8000
            a_opt, val_opt = _adam_optimize(
                a0, n, fft_conv_len, fft_corr_len,
                deadline, max_iters,
                lr=0.06, lam_start=0.02, lam_factor=1.02
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} cand={cand_id} Adam] new best {val_opt:.6f}")

            # ----- Exact max‑convolution sub‑gradient (exp‑grad) -----
            a_ref, val_ref = _expgrad_optimize(
                a_opt, n, fft_conv_len, fft_corr_len,
                deadline, max_iters=5000, eta=0.55
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[n={n} cand={cand_id} ExpGrad] new best {val_ref:.6f}")

        # ----- Permutation search on the current best for this length -----
        if best_seq is not None and len(best_seq) == n:
            remaining = deadline - time.time()
            if remaining > 6.0:
                # Roughly 10 permutation trials per remaining second
                perm_trials = int(remaining * 10)
                print(f"[n={n}] permutation search ({perm_trials} trials)")
                best_seq, best_val = _permute_search(
                    best_seq, n, fft_conv_len,
                    deadline, best_val,
                    max_trials=perm_trials
                )
                print(f"[n={n}] after permutation best {best_val:.6f}")

        # ----- Final fine‑tuning (short Adam run) -----
        if best_seq is not None and len(best_seq) == n:
            fine_iters = int(20000 * 1024 / n) + 2000
            a_fine, val_fine = _adam_optimize(
                best_seq, n, fft_conv_len, fft_corr_len,
                deadline, fine_iters,
                lr=0.04, lam_start=0.5, lam_factor=1.01
            )
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] fine‑tune new best {val_fine:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        print("Fallback: returning uniform vector of length 2048")
        best_seq = np.full(2048, 1.0 / 2048, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
