def _annealing_refine(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = eval_fast(a)
    start = time.time()
    # set initial temperature relative to value
    T_init = best_val * 0.5
    T_end = 1e-5
    it = 0
    while time.time() < deadline:
        it += 1
        # compute temperature linear cooling
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T_init * (1 - frac) + T_end * frac

        # choose i with a[i] > epsilon
        i = np.random.randint(0, n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # propose delta up to a[i]
        max_delta = a[i] * 0.15  # up to 15% of i's mass
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate quickly
        val_new = eval_fast(a_new)
        if val_new < best_val or np.random.rand() < np.exp(-(val_new - best_val) / (T + 1e-12)):
            a = a_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                # print progress occasionally
                if it % 5000 == 0:
                    print(f"[{time.time():.2f}] hill it={it} best={best_val:.6f}")
        # occasional reproject to correct drift
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)
        if best_val < 1.503:
            break
    return best_a, best_val

        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_conv_max:

            best_a = a.copy()

        # compute smooth weights
        tau = max(init_tau_factor * cur_max * (0.99 ** iteration), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)  # shape same as conv
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a
        w_rev = w[::-1]
        size_fft = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size_fft)

        m = beta1 * m + (1 - beta1) * grad_x

Hybrid optimisation for the step‑function evaluation

We minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

   maximum‑convolution.  The sub‑gradient is known analytically
   (a reversed slice of the current sequence).  An adaptive step size
   together with Euclidean projection onto the simplex (Duchi et al.)
   quickly reduces the peak.

3. **Annealed hill‑climbing** – a simulated‑annealing local search.
   Moves transfer a small amount of mass from indices that contribute
   strongly to the current peak to random other indices.  Temperature
   decays linearly; accepted moves are always improving or accepted
   probabilistically.

The best sequence found is returned as a plain Python list.  If no
candidate is discovered a uniform construction is used as a safe
fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors (full length)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # numerical noise may produce tiny negative values
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σw = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)            # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1               # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective (identical to evaluate_sequence for
# admissible inputs)
# ----------------------------------------------------------------------
def _eval_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
# Random initialisation (mixture of Dirichlet / power‑law / exponential)
# ----------------------------------------------------------------------
def _random_initialisation(n: int) -> np.ndarray:
    target = np.sqrt(2.0 * n)
    r = np.random.rand()
    if r < 0.55:
        # simple Dirichlet‑like
        a = np.random.rand(n)
    else:
        # power‑law with optional exponential tail
        p = np.random.uniform(0.1, 1.0)
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** (-p)
        if np.random.rand() < 0.45:
            tau = np.random.uniform(0.2 * n, 2.0 * n)
            a *= np.exp(-idx / tau)
    a = np.maximum(a, 0.0)
    a = a / a.sum() * target
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Run Adam on the log‑its for a limited time slice."""
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_max = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- soft‑max mapping ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)          # Σp = 1
        a = target_sum * p

        # ---- current exact max of the convolution ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} new local best max={cur_max:.6f}")

        # ---- smooth‑max (log‑sum‑exp) weights ----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient wrt a (2·a ⊛ w_rev) ----
        w_rev = w[::-1]
        size_fft = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size_fft)
        Fw = np.fft.rfft(w_rev, n=size_fft)
        conv2 = np.fft.irfft(Fa * Fw, n=size_fft)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ---- back‑propagation to logits ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # learning‑rate decay
        alpha *= 0.9995

    # final exact evaluation (required for the competition)
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with simplex projection."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _eval_objective(a)
    step = target_sum * 0.08 / n          # about 8 % of the average entry
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of the current peak

        # ----- sub‑gradient of the peak -----
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1, dtype=int)
        grad[idxs] = 2.0 * a[k - idxs]

        # ----- descent step + projection -----
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _eval_objective(a_candidate)

        if val_candidate < best_val - 1e-14:
            a = a_candidate
            best_a = a_candidate.copy()
            best_val = val_candidate
            step *= 1.02                # modestly increase after success
        else:
            step *= 0.70                # shrink aggressively after failure

        # occasional random jitter to keep diversity alive
        if it % 300 == 0 and np.random.rand() < 0.12:
            jitter = np.random.rand(n) * 1e-5
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] subgrad it={it:5d} step={step:.3e} best={best_val:.6f}")

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Annealed stochastic hill‑climbing (targeted moves)
# ----------------------------------------------------------------------
def _annealing_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing refinement that moves mass from high‑impact to random locations."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _eval_objective(a)

    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution and the position of the peak
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        # gradient w.r.t. a for this peak (positive numbers)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1, dtype=int)
        grads = 2.0 * a[k - idxs]
        if grads.sum() == 0.0:
            i = np.random.choice(idxs)
        else:
            probs = grads / grads.sum()
            i = np.random.choice(idxs, p=probs)

        # choose a destination different from i
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        # move a small fraction of the mass from i to j
        delta = a[i] * np.random.rand() * 0.18  # ≤ 18 % of the source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _eval_objective(a_new)

        # linear temperature schedule
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(1e-5, best_val * 0.5 * (1.0 - fraction))

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                if it % 5000 == 0:
                    print(f"[{time.time():.2f}] anneal it={it:6d} new best={best_val:.6f}")

        # re‑project occasionally to eliminate accumulated numerical drift
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function with a minimal evaluation.
    The routine respects the given time budget, combines Adam, sub‑gradient
    descent and annealed hill‑climbing, and finally returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12       # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optional warm‑start from a globally supplied construction.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.asarray(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many short Adam optimisations with diverse sizes.
    # --------------------------------------------------------------
    while True:
        now = time.time()
        remaining = deadline - now
        # reserve at least 300 s for the exact refinement + annealing
        if remaining < 320.0:
            break

        # allocate a modest slice (8‑12 s) for a single Adam run
        slice_time = min(12.0, remaining * 0.22)
        run_deadline = time.time() + slice_time

        # problem size – favour medium–large vectors early on
        if remaining > 600:
            n = np.random.randint(2100, 3800)
        elif remaining > 300:
            n = np.random.randint(1600, 3200)
        else:
            n = np.random.randint(1000, 2400)

        target_sum = np.sqrt(2.0 * n)

        # ---- initialise logits ----
        a_init = _random_initialisation(n)
        x0 = np.log(a_init + 1e-14)

        # sometimes inject a trimmed/padded version of the current best
        if best_seq is not None and np.random.rand() < 0.33:
            bs = best_seq
            if bs.shape[0] >= n:
                a_tmp = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.shape[0]) * 1e-6
                a_tmp = np.concatenate([bs, pad])
            a_tmp = a_tmp / a_tmp.sum() * target_sum
            x0 = np.log(a_tmp + 1e-14)

        # ---- run Adam optimiser ----
        seq_local, score_local = _adam_optimize(
            n, x0, target_sum, run_deadline,
            init_tau_factor=0.20,
            min_tau_factor=0.001,
            alpha0=0.025,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f} (n={n})")
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the exact objective.
    # --------------------------------------------------------------
    now = time.time()
    time_left = deadline - now
    if best_seq is not None and time_left > 90.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(250.0, time_left * 0.55)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – annealed stochastic hill‑climbing for fine tuning.
    # --------------------------------------------------------------
    now = time.time()
    time_left = deadline - now
    if best_seq is not None and time_left > 10.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _annealing_refine(best_seq.copy(), target_sum, hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
