Optimise a non‑negative step function for the objective

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

    sum(a) = sqrt(2*n)

   a = S·softmax(x).  A log‑sum‑exp surrogate of the max‑convolution is
   annealed (parameter τ) so that the optimisation finally behaves like a
   sub‑gradient step on the true max.  Many independent restarts of

   from one index to another while preserving the normalisation.
   Improvements are kept, otherwise the move is discarded.

# FFT based full convolution (real vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Random initial shapes (diverse families)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive vector of length n with a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Projection onto the simplex {x>=0, sum(x)=z}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum z (non‑negative)."""
    if np.all(v >= 0) and abs(v.sum() - z) <= 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will become zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.30,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.95,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed‑max surrogate.
    Returns the best sequence encountered (as a NumPy array) and its
    official evaluation score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # -------- 1. soft‑max mapping ----------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)            # stability
        p = exp_shift / exp_shift.sum()          # probabilities
        a = target_sum * p                       # current coefficients

        # -------- 2. convolution & current max ----------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  new local best = {max_conv:.8f}")

        # -------- 3. smooth‑max weights (log‑sum‑exp) ----------
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau          # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # -------- 4. gradient w.r.t. a ----------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)  # length = 3n‑2
        idx_slice = (2 * n - 2) - np.arange(n)    # pick needed part
        grad_a = 2.0 * conv2[idx_slice]

        # -------- 5. gradient w.r.t. logits x (chain rule) ----------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping to avoid huge steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 8.0:
            grad_x = grad_x * (8.0 / gnorm)

        # -------- 6. Adam update ----------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to help escape plateaux
        if iteration % 300 == 0 and rng.random() < 0.12:
            x += rng.normal(size=n) * 0.001

        # decay learning rate slowly
        alpha *= 0.9995

        # early stop if no improvement for a while
        if iteration - last_improve > 3000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # final official evaluation (preserves the exact behaviour of evaluate_sequence)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max(conv(a,a))
# ----------------------------------------------------------------------
def _subgradient_descent(
    a: np.ndarray,
    target_sum: float,
    time_deadline: float,
    rng: np.random.Generator,
    *,
    initial_step_factor: float = 0.5,
    step_decay: float = 0.995,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Deterministic sub‑gradient descent with simplex projection."""
    n = a.shape[0]
    best = a.copy()
    best_val = np.max(_full_convolution_fft(best, best))

    step = initial_step_factor * target_sum  # absolute step size

    it = 0
    while time.time() < time_deadline:
        conv = _full_convolution_fft(best, best)
        max_val = conv.max()

        # mask of active (maximal) positions
        mask = (conv >= max_val - 1e-12).astype(float)

        # sub‑gradient: 2 * (mask *conv a_rev)
        grad = 2.0 * np.convolve(mask, best[::-1], mode='full')[n - 1 : n - 1 + n]

        # gradient step
        step = max(step * step_decay, 1e-9)
        new_a = best - step * grad
        new_a = _simplex_projection(new_a, target_sum)
        new_a = np.clip(new_a, 0.0, 1000.0)

        new_val = np.max(_full_convolution_fft(new_a, new_a))

        if new_val < best_val - tol:
            best = new_a
            best_val = new_val
            # if we made progress, allow step to grow a little
            step = min(step * 1.02, target_sum)

        it += 1
        if it % 500 == 0:
            print(f"[{time.time():.2f}] sub‑grad iter={it}, max={best_val:.8f}")

        # stop if step has become negligible
        if step < 1e-7:
            break

    return best, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (mass transfer) local search
# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    target_sum: float,
    time_deadline: float,
    rng: np.random.Generator,
    *,
    max_trials: int = 20000,
    max_transfer_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from one index to another."""
    n = a.shape[0]
    best = a.copy()
    best_val = np.max(_full_convolution_fft(best, best))

    trials = 0
    while time.time() < time_deadline and trials < max_trials:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # transfer a random fraction of a[i]
        frac = rng.uniform(0.0, max_transfer_frac)
        delta = best[i] * frac
        if delta <= 0.0:
            continue

        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # sum is preserved automatically
        candidate = np.clip(candidate, 0.0, 1000.0)

        val = np.max(_full_convolution_fft(candidate, candidate))
        if val < best_val:
            best = candidate
            best_val = val
            # occasional report
            if trials % 500 == 0:
                print(f"[{time.time():.2f}] hill‑climb trial={trials}, max={best_val:.8f}")

        trials += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the prescribed
    evaluation function.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    best_seq = None
    best_score = float("inf")
    top_candidates = []          # list of (score, seq) – keep a handful

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction, if it exists.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            top_candidates.append((sc, hs_arr.copy()))
            print(f"[{time.time():.2f}] Warm‑start global = {sc:.8f}")
    except Exception:
        pass  # no pre‑existing seed

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam restarts with diverse initial shapes.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 30.0:  # leave time for refinement phases
            break

        # pick problem size – larger early on, smaller later for speed
        if remaining > 300.0:
            n = rng.integers(1500, 3500 + 1)
        elif remaining > 120.0:
            n = rng.integers(1000, 2500 + 1)
        else:
            n = rng.integers(800, 2000 + 1)

        # ----------------------------------------------------------------
        # Build an initial shape – sometimes from a good previous candidate.
        # ----------------------------------------------------------------
        if top_candidates and rng.random() < 0.45:
            src_seq = top_candidates[rng.integers(len(top_candidates))][1]
            if src_seq.size >= n:
                init = src_seq[:n].copy()
            else:
                pad = rng.random(n - src_seq.size) * 1e-6
                init = np.concatenate([src_seq, pad])
        else:
            init = _initial_shape_random(n, rng)

        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)

        # allocate a time slice for this run (≤ 12 s or a reasonable fraction)
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam_softmax(
            n=n,
            x0=x0,
            target_sum=target_sum,
            run_deadline=run_deadline,
            rng=rng,
            init_tau_factor=0.30,
            min_tau_factor=5e-4,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.95,
            max_iter=None,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            top_candidates.append((best_score, best_seq.copy()))
            top_candidates.sort(key=lambda z: z[0])
            top_candidates = top_candidates[:5]   # keep a few elites
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            if best_score < 1.503:   # early exit if target reached
                break

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient descent on the current best.
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 20.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(30.0, deadline - time.time())
        seq_sub, val_sub = _subgradient_descent(
            best_seq.copy(),
            target_sum,
            sub_deadline,
            rng,
            initial_step_factor=0.5,
            step_decay=0.995,
        )
        if val_sub < np.max(_full_convolution_fft(best_seq, best_seq)):
            best_seq = seq_sub
            best_score = evaluate_sequence(best_seq.tolist())
            print(f"[{time.time():.2f}] Sub‑gradient improved best to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – simple hill‑climbing local search.
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        hc_deadline = time.time() + min(12.0, deadline - time.time())
        seq_hc, val_hc = _hill_climb(
            best_seq.copy(),
            target_sum,
            hc_deadline,
            rng,
            max_trials=20000,
            max_transfer_frac=0.25,
        )
        if val_hc < np.max(_full_convolution_fft(best_seq, best_seq)):
            best_seq = seq_hc
            best_score = evaluate_sequence(best_seq.tolist())
            print(f"[{time.time():.2f}] Hill‑climb improved best to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that no candidate was built.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The outer harness expects a plain Python list.
    return best_seq.tolist()
