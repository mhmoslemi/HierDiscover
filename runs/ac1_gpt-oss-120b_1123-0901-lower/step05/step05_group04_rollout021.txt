           conv = np.convolve(a,a)
           M = conv.max()

               break

           sum_b = b.sum()

               break
           g = b * (target_sum / sum_b)

"""
Library that searches for a non‑negative step function minimizing the
max‑convolution objective.  It uses an iterative linear‑programming
procedure inspired by the classic “max‑convolution reduction” technique:
starting from a random shape f, we repeatedly solve

    max  Σ b_j
    s.t. (f * b)_k ≤ M   for all k,
         b_j ≥ 0,

where M = max(f*f).  The optimizer returns a new candidate g,
which is mixed with f (relaxation parameter t).  The bound M is
strictly decreasing and the loop stops when no further improvement is
seen.  Several independent restarts with random exponential shapes are
performed within the time budget; the best construction found is
returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def _build_conv_constraints(f):
    """Return the matrix A (shape (2n-1, n)) such that A @ b = f*b (full convolution)."""
    n = f.shape[0]
    # first column: f padded with zeros to length 2n-1
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row: f[0] followed by zeros
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return toeplitz(c, r)

def _lp_step(f, M, target_sum):
    """Solve the LP for the current f and bound M.
    Returns a new vector g (scaled to have sum = target_sum)."""
    n = f.shape[0]
    A = _build_conv_constraints(f)
    # objective: maximize sum(b) -> minimize -ones·b
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0.0, None)] * n
    # Solve with HiGHS
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None
    b = res.x
    s = b.sum()
    if s <= 0:
        return None, None
    g = b * (target_sum / s)
    # enforce tiny positivity to keep future LP bounded
    g = np.maximum(g, 1e-9)
    return g, res

def _mix(f, g, t):
    """Linear interpolation between f and g, staying on the simplex."""
    return (1 - t) * f + t * g

def _refine_via_lp(initial, target_sum, max_iters, t, time_deadline):
    """Iteratively apply the LP step to improve the bound."""
    f = initial.copy()
    best_f = f.copy()
    best_M = np.max(np.convolve(f, f))
    iteration = 0
    while iteration < max_iters and time.time() < time_deadline:
        M = np.max(np.convolve(f, f))
        # early termination if bound does not improve appreciably
        if M >= best_M - 1e-9:
            # try a smaller mixing factor or break
            t = max(t * 0.5, 1e-4)
            if t < 1e-4:
                break
        g, _ = _lp_step(f, M, target_sum)
        if g is None:
            break
        f_new = _mix(f, g, t)
        # re‑normalise to guard against numerical drift
        f_new = f_new * (target_sum / f_new.sum())
        new_M = np.max(np.convolve(f_new, f_new))
        if new_M < best_M:
            best_M = new_M
            best_f = f_new.copy()
            f = f_new
            # keep mixing factor moderate
            t = min(t * 1.1, 0.5)
        else:
            # no improvement – reduce step size
            t = max(t * 0.5, 1e-4)
        iteration += 1
    return best_f, best_M

def _local_hill_climb(seq, target_sum, time_deadline):
    """Simple hill‑climbing that preserves the sum and non‑negativity."""
    a = seq.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = np.max(np.convolve(a, a))
    # small fraction of total mass to move each step
    move_frac = 0.01
    while time.time() < time_deadline:

        if i == j:
            continue
        delta = a[i] * move_frac
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        # re‑normalize to target sum (should stay exact)

        if cur < best_val:
            best_val = cur
            best = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
    return best, best_val

        idx = np.arange(n, dtype=float)

"""
LP‑based step‑function optimizer

The goal is to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

for a non‑negative sequence a.  Because the objective is invariant to a
global scaling of a, we fix the L1‑norm to S = sqrt(2 n).  The algorithm
works as follows:

1. Initialise a random monotone exponential shape (all entries > 0) and
   scale it to have sum = S.

2. Let M = max(conv(a, a)).  Solve the linear programme

        maximise   Σ b_j
        subject to (a * b)_k ≤ M   for all k = 0…2n‑2
                   b_j ≥ 0

   The convolution constraints are linear in the unknown vector b.
   The LP is solved with SciPy’s HiGHS solver.

3. Rescale the LP solution b to the simplex (sum = S) obtaining a new
   candidate g and mix it with the current a:
        a ← (1‑t)·a + t·g   (t ≈ 0.3, reduced if no improvement).

   The mixing step keeps the L1‑norm fixed while strictly decreasing the
   bound M (unless the step size is too large).

4. Repeat steps 2–3 a few times (≤ 15) or until the time budget for the
   current restart runs out.

5. A cheap hill‑climbing phase moves a tiny fraction of mass between
   random positions while preserving the sum; any improvement is kept.

Multiple independent restarts with different lengths and random seeds are
performed within the overall 1000 s budget.  The best construction found
is returned as a plain Python list.  If a global variable
``height_sequence_1`` is present it is used as an initial warm‑start.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _conv_constraints_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that A @ b = np.convolve(f, b).
    """
    n = f.shape[0]
    # first column of the Toeplitz matrix: f padded with zeros
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row: f[0] followed by zeros
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return toeplitz(c, r)


def _lp_step(f: np.ndarray, bound: float, target_sum: float):
    """
    One LP step: maximise Σ b_j under (f * b)_k ≤ bound.
    Returns the scaled vector g (sum = target_sum) or (None, None) on failure.
    """
    n = f.shape[0]
    A = _conv_constraints_matrix(f)                # (2n‑1) × n matrix
    c = -np.ones(n)                                 # minimise –Σ b_j  ⇒ maximise Σ b_j
    b_ub = np.full(2 * n - 1, bound, dtype=float)   # upper‑bound for every convolution entry
    bounds = [(0.0, None)] * n                       # b_j ≥ 0

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None

    b = res.x
    total = b.sum()
    if total <= 0.0:
        return None, None

    # Scale to the required L1‑norm (the optimisation is scale‑invariant)
    g = b * (target_sum / total)
    # keep strict positivity to avoid unbounded LPs in the next iteration
    g = np.maximum(g, 1e-9)
    return g, res


def _mix_vectors(x: np.ndarray, y: np.ndarray, t: float) -> np.ndarray:
    """Linear interpolation (convex combination) of two vectors."""
    return (1.0 - t) * x + t * y


def _refine_via_lp(initial: np.ndarray, target_sum: float,
                   max_iters: int, mix_factor: float,
                   time_deadline: float):
    """
    Iteratively apply the LP step starting from ``initial``.
    Returns the best vector found and its max‑convolution value.
    """
    a = initial.copy()
    best_a = a.copy()
    best_M = np.max(np.convolve(a, a))
    t = mix_factor
    it = 0

    while it < max_iters and time.time() < time_deadline:
        M = np.max(np.convolve(a, a))

        # solve LP under the current bound M
        g, _ = _lp_step(a, M, target_sum)
        if g is None:
            break

        # mix and re‑normalise (tiny numerical drift may occur)
        a_new = _mix_vectors(a, g, t)
        a_new *= target_sum / a_new.sum()

        new_M = np.max(np.convolve(a_new, a_new))

        # accept if the bound improves
        if new_M < best_M - 1e-9:
            best_M = new_M
            best_a = a_new.copy()
            a = a_new
            # slowly increase the mixing factor (up to 0.5) when we are making progress
            t = min(t * 1.1, 0.5)
        else:
            # no improvement – become more conservative
            t = max(t * 0.5, 1e-4)

        it += 1

    return best_a, best_M


def _hill_climb(seq: np.ndarray, target_sum: float, time_deadline: float):
    """
    Very simple hill‑climbing that moves a small fraction of mass between two
    random positions while preserving the total sum.
    """
    a = seq.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = np.max(np.convolve(a, a))
    move_frac = 0.02  # percent of the source entry to move each move

    while time.time() < time_deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * move_frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur = np.max(np.convolve(a, a))
        if cur < best_val:
            best_val = cur
            best = a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

    return best, best_val


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the provided
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1  # small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Warm‑start from a globally supplied construction, if any.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed: {sc:.8f}")
    except Exception:
        pass  # no global seed available

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a slice of the remaining time.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # keep the problem size moderate so that the LP stays tractable
        n = rng.integers(1500, 2001)          # 1500 ≤ n ≤ 2000

        # ------------------------------------------------------------------
        # Random exponential shape (strictly positive).
        # ------------------------------------------------------------------
        tau = rng.uniform(0.2, 0.8) * n
        idx = np.arange(n, dtype=float)
        shape = np.exp(-idx / tau) + rng.random(n) * 1e-6
        shape = np.maximum(shape, 1e-9)

        target_sum = np.sqrt(2.0 * n)          # sum we keep fixed
        a0 = shape / shape.sum() * target_sum

        # Allocate a short time slice for this restart (max ≈ 15 s)
        slice_time = min(15.0, remaining * 0.6)
        run_deadline = time.time() + slice_time

        # ------------------------------------------------------------------
        # LP‑based refinement.
        # ------------------------------------------------------------------
        a_opt, _ = _refine_via_lp(
            initial=a0,
            target_sum=target_sum,
            max_iters=12,
            mix_factor=0.3,
            time_deadline=run_deadline
        )

        # ------------------------------------------------------------------
        # Optional cheap hill‑climb if time permits.
        # ------------------------------------------------------------------
        if time.time() < run_deadline:
            a_opt, _ = _hill_climb(a_opt, target_sum, run_deadline)

        # ------------------------------------------------------------------
        # Official evaluation and bookkeeping.
        # ------------------------------------------------------------------
        score = evaluate_sequence(a_opt.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] NEW GLOBAL BEST  n={n}  score={score:.8f}")

        # Early termination if we already beat the target.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case that nothing was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n, dtype=float)

    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
