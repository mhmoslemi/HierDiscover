returns it as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (no scaling – a may be any non‑negative vector)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Return a modest pool of diverse start vectors."""
    cand = []

    # warm start (previous best) and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # sinusoidal (positive half‑wave)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    cand.append(s / s.sum())

    # symmetric Beta shapes (α = β)
    x = np.linspace(0.0, 1.0, n)
    for α in (0.2, 0.5, 1.0, 2.0, 3.0, 5.0):
        b = (x ** (α - 1.0)) * ((1.0 - x) ** (α - 1.0))
        b += 1e-12                     # avoid exact zeros
        cand.append(b / b.sum())

    # asymmetric Beta shapes (α ≠ β)
    for α, β in ((0.5, 2.0), (2.0, 0.5), (0.3, 3.0), (3.0, 0.3)):
        b = (x ** (α - 1.0)) * ((1.0 - x) ** (β - 1.0))
        b += 1e-12
        cand.append(b / b.sum())

    # edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for p in (-2.0, -1.5, -1.0, -0.5):
        shape = np.where(prod > 0, prod ** p, 0.0)
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two narrow Gaussians glued to the edges
    i = np.arange(n, dtype=np.float64)
    for γ in (5.0, 8.0, 12.0):
        g = np.exp(-γ * (i ** 2)) + np.exp(-γ * ((n - 1 - i) ** 2))
        cand.append(g / g.sum())

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))
    return cand


# ----------------------------------------------------------------------
def _optimize_one_candidate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by an exponentiated‑gradient
    sub‑gradient phase.  The routine updates the global best sequence/value
    and returns the (possibly) improved best.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                      # softmax parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                      # a slightly larger step size
    lam = 0.4                                      # start temperature
    lam_factor = 1.02
    lam_max = 1e5

    last_improve = time.time()
    it = 0

    # ------------------- Adam on smooth‑max ---------------------------------
    while time.time() < deadline:
        # soft‑max → probability vector on the simplex
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # convolution & exact objective (for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        # conv ≤ max_c, so exponent is ≤ 0 → safe from overflow
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – slowly approach the true max
        lam = min(lam * lam_factor, lam_max)

        # occasional gentle smoothing to avoid spurious high‑frequency spikes
        if (it + 1) % 400 == 0:
            smooth = np.array([0.25, 0.5, 0.25])
            a_sm = np.convolve(a, smooth, mode="same")
            a_sm = np.maximum(a_sm, 0.0)
            a_sm /= a_sm.sum()
            # re‑initialise logits from the smoothed vector
            logits = np.log(a_sm + eps)

        # random shake if stagnating
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 150_000:                     # safety break
            break

    # ------------------- Exact‑max sub‑gradient phase --------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.55
    sub_last_improve = time.time()
    for sub_it in range(12_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad it={sub_it:05d}] new best {val:.6f}")

        # active set of maximal convolution entries
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise, protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # modest smoothing every few thousand steps
        if (sub_it + 1) % 3000 == 0:
            smooth = np.array([0.25, 0.5, 0.25])
            a = np.convolve(a, smooth, mode="same")
            a = np.maximum(a, 0.0)
            a /= a.sum()

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Respects the time budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Primary length – the current benchmark (4096).  Larger n can be tried later.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start from previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    candidates = _initial_candidates(n, warm)

    # --------------------------------------------------------------
    # Optimise each starting point (more iterations for the first few)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        print(f"--- Optimising candidate {cid+1}/{len(candidates)} (n={n})")
        best_seq, best_val = _optimize_one_candidate(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # If time permits, upscale to a larger grid (8192 then 16384) and
    # continue optimisation from the best found distribution.
    for n_big in (8192, 16384):
        if time.time() + 8.0 >= deadline:
            break

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation onto the finer grid
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print(f"--- Upscaling to n={n_big} and continuing optimisation")
        best_seq, best_val = _optimize_one_candidate(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        # prepare for possible next upscale
        n, conv_fft_len, corr_fft_len = n_big, conv_fft_len_big, corr_fft_len_big

    # --------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # expose the result for the next run (optional but convenient)
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"=== Finished. Best value = {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
