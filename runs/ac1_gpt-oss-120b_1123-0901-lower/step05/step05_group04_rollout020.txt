def build_conv_matrix(f):
    n = len(f)
    rows = 2*n -1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill:
    for i in range(n):
        # column i corresponds to g_i coefficient across rows
        # coefficient for row k = f_{k-i} if 0 <= k-i < n else 0.
        # So shift f by i positions:
        A[i:i+n, i] = f
    return A

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of two ideas:

1. **Linear‑programming based “mass transfer”**
   – With the current sequence f we compute M = max(conv(f,f)).
   – We solve a linear program that maximises the total mass of a new
     sequence g under the linear constraints  conv(f,g) ≤ M.
   – g is then rescaled to the canonical sum  √(2n) and mixed
     (f ← (1‑t)·f + t·g).  This always keeps the total sum fixed while
     never increasing the maximal convolution value, thus providing a
     guaranteed monotone descent.

2. **Optional Adam fine‑tuning**
   – After the LP stage a short Adam optimisation (soft‑max
     parameterisation) is run on the best sequence found so far.
   – The Adam optimiser uses a log‑sum‑exp surrogate of the max and an
     aggressive temperature annealing, exactly as in the reference
     implementation, but now starts from a much better point.

Both phases respect the 1000 s wall‑clock budget, keep a handful of the
best candidates and finally return the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast full linear convolution via FFT (used many times)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())                # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)                  # remove tiny negatives
    return conv


# ----------------------------------------------------------------------
# Build the dense convolution matrix A such that A @ g = conv(f, g)
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    For a fixed non‑negative vector ``f`` of length n,
    build a dense matrix A of shape (2n‑1, n) with

        (A @ g)[k] = (f * g)[k]   for all k.

    The matrix is very sparse‑structured (each column is a copy of f
    shifted by the column index) but building it explicitly is cheap for
    the problem sizes we use (n ≤ 2500).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Small random “shape” generators – used for restart initialisation
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a positive vector of length n with a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:               # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables – final polishing
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """Adam optimiser of a smoothed‑max surrogate (same code as reference)."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1. map logits to the simplex (a = S·softmax(x))
        # --------------------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stabilise
        p = exp_shift / exp_shift.sum()               # probabilities, sum = 1
        a = target_sum * p

        # --------------------------------------------------------------
        # 2. convolution and current (smooth) max
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            # pylint: disable=consider-using-f-string
            print("[{:.2f}] ADAM local best = {:.8f}".format(time.time(), max_conv))

        # --------------------------------------------------------------
        # 3. smooth‑max weights (log‑sum‑exp surrogate)
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau               # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 4. gradient w.r.t. a
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)               # slice needed
        grad_a = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 5. gradient w.r.t. logits x (chain rule through the soft‑max)
        # --------------------------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping – prevents occasional huge steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # --------------------------------------------------------------
        # 6. Adam update
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to help escape flat plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9997

        # early stop if we have not improved for a long stretch
        if iteration - last_improve > 2000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# One LP‑based improvement step (mass transfer)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                target_sum: float,
                t_mix: float,
                rng: np.random.Generator) -> tuple[np.ndarray, bool]:
    """
    Perform a single linear‑programming mass‑transfer step.

    Returns the new sequence (already renormalised to ``target_sum``) and a
    flag indicating whether the step strictly improved the evaluator.
    """
    n = a.shape[0]
    # 1. current maximal convolution value
    conv = _full_convolution_fft(a, a)
    M = conv.max()

    # 2. build convolution matrix (conv(f,·) ≤ M)
    A = _conv_matrix(a)

    # 3. linear program: maximise 1·g  subject to A·g ≤ M, g ≥ 0
    c = -np.ones(n, dtype=np.float64)          # linprog minimises c·g
    b = np.full(A.shape[0], M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # If the LP fails we simply abort this improvement.
        return a, False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return a, False

    # 4. rescale g to the canonical sum
    scale = target_sum / sum_g
    g_scaled = g * scale

    # 5. convex combination with the old sequence
    a_new = (1.0 - t_mix) * a + t_mix * g_scaled

    # 6. re‑normalise (tiny numerical drift)
    a_new = a_new / a_new.sum() * target_sum

    # 7. evaluate – we rely on the official evaluator to obtain the exact
    #    score (including clipping, rounding, etc.).
    old_score = evaluate_sequence(a.tolist())
    new_score = evaluate_sequence(a_new.tolist())
    improved = new_score < old_score - 1e-9
    return a_new, improved

# Main entry point required by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * many short restarts (random length, random initial shape)
    * each restart performs a few LP‑based mass‑transfer iterations,
      which strictly decrease the maximal convolution.
    * the globally best candidate is finally refined by a short Adam
      optimiser (the same optimiser used by the reference solution).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2           # keep a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is present, use it as a warm‑start.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] warm‑start (global) score = {sc:.8f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose problem size – we keep n ≤ 2500 to stay within memory limits.
        if remaining > 300:
            n = rng.integers(1500, 2500 + 1)
        elif remaining > 120:
            n = rng.integers(1200, 2200 + 1)
        else:
            n = rng.integers(800, 1500 + 1)

        # ------------------------------------------------------------------
        # Initial shape (occasionally reuse a good candidate)
        # ------------------------------------------------------------------
        if best_seq is not None and rng.random() < 0.35:
            src = np.array(best_seq)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = rng.random(n - src.size) * 1e-6
                init = np.concatenate([src, pad])
        else:
            init = _initial_shape_random(n, rng)

        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum

        # --------------------------------------------------------------
        # Early evaluation – keep the very best we see
        # --------------------------------------------------------------
        cur_score = evaluate_sequence(init.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = init.copy()
            print(f"[{time.time():.2f}] new global start score = {cur_score:.8f} (n={n})")
            if best_score < 1.503:   # reached the target – stop early
                break

        # --------------------------------------------------------------
        # LP‑based descent – a few iterations (usually 5‑12 are enough)
        # --------------------------------------------------------------
        a = init
        t_mix = 0.35                # start with a fairly aggressive mixing
        for lp_it in range(12):
            if time.time() >= deadline:
                break
            a, improved = _lp_improve(a, target_sum, t_mix, rng)
            if not improved:
                # reduce mixing strength and give up if nothing improves
                t_mix *= 0.6
                if t_mix < 0.02:
                    break
                continue

            # evaluate new candidate
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] LP improvement it={lp_it} score={cur_score:.8f} (n={n})")
                if best_score < 1.503:
                    break

        # ------------------------------------------------------------------
        # If time permits we keep a few elite candidates for later
        # ------------------------------------------------------------------
        # (In this simple implementation we only keep the single best.)

        if best_score < 1.503:
            break

    # ----------------------------------------------------------------------
    # Final refinement with Adam optimiser (if we still have time left)
    # ----------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        # re‑scale to the canonical simplex (precaution)
        a = best_seq * (target_sum / best_seq.sum())
        x0 = np.log(a + 1e-12)
        run_deadline = deadline
        seq_refined, score_refined = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            rng,
            init_tau_factor=0.12,
            min_tau_factor=1e-5,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.94,
            max_iter=None,
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] final Adam refinement score = {best_score:.8f}")

    # ----------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe default.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[{time.time():.2f}] DONE – best score = {best_score:.8f} (length {len(best_seq)})")
    return best_seq.tolist()
