def _project_sym_monotonic(a):
    n = len(a)
    # symmetrize: average pairs
    a_sym = a.copy()
    for i in range(n//2):
        avg = 0.5 * (a[i] + a[n-1-i])
        a_sym[i] = a_sym[n-1-i] = avg
    if n%2==1:
        a_sym[n//2] = a[n//2]  # unchanged (center)
    # monotonic increasing up to center
    half = n//2
    left = a_sym[:half]
    # enforce left non-decreasing (monotonic ascending towards centre)
    # Using isotonic regression: simple cumulative max
    left_proj = np.maximum.accumulate(left)
    a_sym[:half] = left_proj
    # reflect to right
    if n%2==0:
        a_sym[half:] = left_proj[::-1]
    else:
        a_sym[half+1:] = left_proj[::-1]
    # Normalize
    s = a_sym.sum()
    if s > 0:
        a_sym /= s
    else:
        a_sym = np.full_like(a_sym, 1.0/len(a_sym))
    return a_sym

"""
Propose a low‑score step‑function.

The algorithm works in three stages:
1.  A quick optimisation on a modest size (n≈1024) using Adam on a
    smooth‑max surrogate followed by an exact‑max sub‑gradient phase.
2.  The best distribution is interpolated to a larger size (4096, then
    optionally 8192) and re‑optimised.  The sub‑gradient phase now
    projects each iterate onto the convex set of *symmetric unimodal*
    vectors (average with its mirror and enforce non‑decreasing values
    from the left edge towards the centre).  This reduces the maximal
    auto‑correlation and yields a better bound.
3.  A short hill‑climbing stage performs random mass swaps while the
    time budget allows, accepting any improvement.

All updates keep the vector non‑negative and normalised to sum = 1.
The routine respects the supplied budget (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest yet diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    cand.append(np.random.dirichlet(np.ones(n)))                     # random Dirichlet

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))                                # uniform

    return cand


def _project_sym_monotonic(a: np.ndarray) -> np.ndarray:
    """
    Project onto the set of symmetric, unimodal vectors.
    * Symmetrise by averaging each pair (i, n‑1‑i).
    * Enforce non‑decreasing values from the left edge to the centre
      (a[0] ≤ a[1] ≤ … ≤ a[centre]).
    The result is renormalised to sum = 1.
    """
    n = a.shape[0]
    a_sym = a.copy()

    # 1. Symmetrise
    half = n // 2
    for i in range(half):
        avg = 0.5 * (a_sym[i] + a_sym[n - 1 - i])
        a_sym[i] = a_sym[n - 1 - i] = avg
    if n % 2 == 1:
        a_sym[half] = a[half]

    # 2. Enforce monotonic increase towards the centre on the left half.
    left = a_sym[:half]
    left_proj = np.maximum.accumulate(left)          # PAVA for increasing sequence
    a_sym[:half] = left_proj
    if n % 2 == 0:
        a_sym[half:] = left_proj[::-1]
    else:
        a_sym[half + 1:] = left_proj[::-1]

    # 3. Normalise
    s = a_sym.sum()
    if s > 0:
        a_sym /= s
    else:
        a_sym = np.full_like(a_sym, 1.0 / n)

    return a_sym


def _deep_optimize_v2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise a single start vector.
    - Adam on a smooth‑max surrogate (log‑sum‑exp).
    - Exact‑max sub‑gradient phase with exponentiated‑gradient updates.
      After each update the iterate is projected onto the symmetric‑unimodal
      set using `_project_sym_monotonic`.
    The routine updates the global best sequence/value and respects the deadline.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5                     # temperature for smooth‑max
    lam_factor = 1.02

    last_improve = time.time()
    it = 0

    # ------------------------------------------------------------------
    # Adam on smooth‑max surrogate
    while time.time() < deadline:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()            # a lives on simplex

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # exact objective (a sums to 1)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:05d}] new best {val:.6f}")

        # smooth‑max weighting (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (entropy mirror map)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 200_000:                       # safety cap
            break

    # ------------------------------------------------------------------
    # Exact‑max sub‑gradient phase (exponentiated‑gradient) with projection
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()
    max_sub_iter = 20_000

    for sub_it in range(max_sub_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active set of maximal entries
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)

        # projection onto symmetric‑unimodal set
        a = _project_sym_monotonic(a)

        # re‑normalise (projected version already normalised, but guard against drift)
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stagnation cut‑off
        if time.time() - sub_last_improve > 5.0:
            break

    return global_best_seq, global_best_val

    score within the given time budget (default 1000 s).  The routine
    proceeds through three resolution levels (≈1024 → 4096 → 8192)
    and finishes with a light hill‑climbing refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – small problem (n≈1024) – fast to explore many starts
    n1 = 1024
    conv_fft_len1 = _next_pow2(2 * n1 - 1)
    corr_fft_len1 = _next_pow2(3 * n1 - 2)

    warm = globals().get("height_sequence_1")
    if warm is not None:
        ws = np.asarray(warm, dtype=np.float64)
        if ws.shape[0] > n1:
            ws = ws[:n1]
        else:
            ws = np.pad(ws, (0, n1 - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() > 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    candidates1 = _initial_candidates(n1, ws)

    for cid, cand in enumerate(candidates1):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n1))
        else:
            a0 /= s
        best_seq, best_val = _deep_optimize_v2(
            a0, n1, conv_fft_len1, corr_fft_len1, deadline, best_seq, best_val
        )
        print(f"[Phase1] candidate {cid} best {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – medium size (n≈4096).  Upsample the best shape and re‑optimise.
    n2 = 4096
    if time.time() + 10 < deadline:
        conv_fft_len2 = _next_pow2(2 * n2 - 1)
        corr_fft_len2 = _next_pow2(3 * n2 - 2)

        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, n1, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
            a2 = np.interp(x_new, x_old, best_seq)
            a2 = np.maximum(a2, 0.0)
            if a2.sum() > 0:
                a2 /= a2.sum()
            else:
                a2 = np.random.dirichlet(np.ones(n2))
        else:
            a2 = np.random.dirichlet(np.ones(n2))

        best_seq, best_val = _deep_optimize_v2(
            a2, n2, conv_fft_len2, corr_fft_len2, deadline, best_seq, best_val
        )
        print(f"[Phase2] upsampled best {best_val:.6f}")

        # a few extra random restarts at this resolution
        for r in range(3):
            if time.time() >= deadline:
                break
            a0 = np.random.dirichlet(np.ones(n2))
            best_seq, best_val = _deep_optimize_v2(
                a0, n2, conv_fft_len2, corr_fft_len2, deadline, best_seq, best_val
            )
            print(f"[Phase2] random restart {r} best {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – larger size (n≈8192) if time permits.
    n3 = 8192
    if time.time() + 10 < deadline:
        conv_fft_len3 = _next_pow2(2 * n3 - 1)
        corr_fft_len3 = _next_pow2(3 * n3 - 2)

        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, n2, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n3, endpoint=False)
            a3 = np.interp(x_new, x_old, best_seq)
            a3 = np.maximum(a3, 0.0)
            if a3.sum() > 0:
                a3 /= a3.sum()
            else:
                a3 = np.random.dirichlet(np.ones(n3))
        else:
            a3 = np.random.dirichlet(np.ones(n3))

        best_seq, best_val = _deep_optimize_v2(
            a3, n3, conv_fft_len3, corr_fft_len3, deadline, best_seq, best_val
        )
        print(f"[Phase3] upsampled best {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final hill‑climbing refinement (mass swaps)
    if best_seq is not None:
        a = best_seq.copy()
        n = a.shape[0]
        while time.time() < deadline:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # transfer a small fraction of mass from i to j
            delta = (np.random.rand() - 0.5) * 0.02 * a[i]   # up to ±1 % of a[i]
            if a[i] - delta < 0:
                delta = a[i]                               # move everything if needed
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # re‑normalise
            a_new /= a_new.sum()
            val_new = evaluate_sequence(a_new.tolist())
            if val_new < best_val:
                best_val = val_new
                best_seq = a_new
                a = a_new
                print(f"[HillClimb] new best {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value ≈ {best_val:.6f} with length {best_seq.shape[0]}")
    return best_seq.tolist()
