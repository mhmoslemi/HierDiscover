"""
Hybrid optimiser for the max‑convolution objective.

* Phase 1 – many restarts of a softmax‑Adam optimiser.
  The objective is a smooth log‑sum‑exp approximation of
  `max(conv(a,a))`.  Temperature is aggressively annealed and the
  learning‑rate decays to obtain a near‑sub‑gradient descent.

* Phase 2 – a few linear‑programming “tightening’’ steps.
  For the current solution f we maximise the total weight of a
  companion vector g under the linear constraints  (f*g)[k] ≤ M,
  where M = max(conv(f,f)).  The LP pushes weight into positions
  where the convolution is small; after rescaling and convex‑
  combination the peak of the convolution is reduced.

Both phases respect non‑negativity, are scale‑invariant and run
within the 1000 s budget.  The best sequence found is returned as a
plain list of floats.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper – full linear convolution using a power‑of‑2 FFT (real → real)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT (zero‑padded)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Random initial shapes (later normalised to the target sum)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive vector of length ``n`` with a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:                               # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break symmetry, protect against zeros
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.2,
    min_tau_factor: float = 1e-5,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-9,
    tau_decay: float = 0.95,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence (as a NumPy array) and its *official*
    evaluation score (`evaluate_sequence`).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_conv = float("inf")          # best *max convolution* seen
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1. soft‑max mapping  a = target_sum * softmax(x)
        # --------------------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # stabilise
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # --------------------------------------------------------------
        # 2. current convolution & max value
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_conv:
            best_conv = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  new local best  max_conv={max_conv:.8f}")

        # --------------------------------------------------------------
        # 3. smooth‑max (log‑sum‑exp) weights
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau        # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 4. gradient w.r.t. a  (derivative of Σ w_k·conv_k)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)       # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # pick slice that aligns with a
        grad_a = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 5. gradient w.r.t. logits x (soft‑max chain rule)
        # --------------------------------------------------------------
        dotp = np.dot(p, grad_a)            # scalar
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping – keep steps bounded
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # --------------------------------------------------------------
        # 6. Adam update
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional random jitter to escape flat plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning‑rate
        alpha *= 0.9997

        # early stop if we have not improved for a while
        if iteration - last_improve > 2500:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # ------------------------------------------------------------------
    # Final official evaluation (scale‑invariant, therefore we keep the
    # best *raw* sequence we have seen)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n, target_sum / n, dtype=float)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (phase 2)
# ----------------------------------------------------------------------
def _lp_refine(
    seq: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Perform a handful of convex‑combination + LP steps starting from ``seq``.
    Stops when the wall‑clock exceeds ``time_limit``.
    Returns the refined sequence (still normalised to have the same sum as the input).
    """
    n = seq.shape[0]
    total_sum = seq.sum()
    if total_sum == 0:
        seq = np.full(n, 1.0 / n, dtype=float)
        total_sum = 1.0
    else:
        seq = seq / total_sum                     # work with unit‑sum vectors

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Pre‑build the convolution‑matrix template (dense, but modest size)
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = seq                       # column j = shifted copy of seq

    for step in range(10):
        now = time.time()
        if now >= time_limit - 2.0:
            break

        # 1. current max convolution
        conv = _full_convolution_fft(seq, seq)
        M = conv.max()
        # 2. build constraints  C g ≤ M
        b_ub = np.full(rows, M, dtype=np.float64)

        # 3. solve LP: maximise Σ g_i   ⇔ minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # if the LP fails we abort the refinement loop
            print(f"[{now:.2f}] LP step {step} failed – aborting refinement")
            break

        g = res.x
        if g.sum() == 0:
            break

        # 4. rescale g to the original total sum
        g = g * (total_sum / g.sum())

        # 5. convex combination (step size shrinks geometrically)
        t = 0.5 ** (step + 1)                      # 0.5, 0.25, 0.125, …
        new_seq = (1 - t) * seq + t * g
        new_seq = np.maximum(new_seq, 0.0)        # protect against tiny negatives
        new_seq = new_seq * (total_sum / new_seq.sum())   # renormalise sum

        # 6. evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[{time.time():.2f}] LP refinement step {step} improved score to {best_score:.8f}")

        # stop early if no improvement
        if score >= best_score - 1e-7:
            break

        # prepare for next iteration
        seq = new_seq.copy()
        # update matrix C for the new base vector
        for j in range(n):
            C[j:j + n, j] = seq

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine mixes many Adam‑optimised restarts (soft‑max
    parametrisation) with a short linear‑programming refinement
    phase.  It respects the 1000 s budget and returns the best
    sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1          # safety margin

    best_seq = None
    best_score = float("inf")
    top_candidates = []                        # keep a few elite sequences

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = evaluate_sequence(hs_arr.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            top_candidates.append((sc, hs_arr.copy()))
            print(f"[{time.time():.2f}] Warm‑start global = {sc:.8f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Phase 1 – many Adam restarts (short slices, diverse n)
    # --------------------------------------------------------------
    while time.time() < deadline - 60.0:        # keep ~1 min for LP phase
        remaining = deadline - time.time()

        # choose a problem size (larger early on)
        if remaining > 300:
            n = rng.integers(2000, 3500 + 1)
        elif remaining > 150:
            n = rng.integers(1500, 3000 + 1)
        else:
            n = rng.integers(1000, 2000 + 1)

        # ----------------------------------------------------------
        # Initialise – either from an elite candidate or from scratch
        # ----------------------------------------------------------
        if top_candidates and rng.random() < 0.45:
            src_seq = top_candidates[rng.integers(len(top_candidates))][1]
            if src_seq.size >= n:
                init = src_seq[:n].copy()
            else:
                pad = rng.random(n - src_seq.size) * 1e-6
                init = np.concatenate([src_seq, pad])
        else:
            init = _initial_shape_random(n, rng)

        # normalise to unit sum (scale does not affect the score)
        init = np.maximum(init, 1e-12)
        init = init / init.sum()

        # logits for soft‑max parametrisation
        x0 = np.log(init + 1e-12)

        # give each run a moderate time slice (up to 30 s)
        slice_time = min(30.0, remaining * 0.45)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam_softmax(
            n=n,
            x0=x0,
            target_sum=1.0,                # we work with unit sum
            run_deadline=run_deadline,
            rng=rng,
            init_tau_factor=0.2,
            min_tau_factor=1e-5,
            alpha=0.05,
            beta1=0.9,
            beta2=0.999,
            eps=1e-9,
            tau_decay=0.95,
            max_iter=None,
        )

        # update global best
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            top_candidates.append((best_score, best_seq.copy()))
            top_candidates.sort(key=lambda z: z[0])
            top_candidates = top_candidates[:5]   # keep only a few elites
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            if best_score < 1.503:               # stop early if target reached
                break

    # --------------------------------------------------------------
    # Phase 2 – LP refinement on the current global best
    # --------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 10.0:
        refine_deadline = deadline
        best_seq = _lp_refine(best_seq, refine_deadline, rng)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] AFTER REFINEMENT final score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Fallback in the (unlikely) case that nothing was generated
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, 1.0 / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
