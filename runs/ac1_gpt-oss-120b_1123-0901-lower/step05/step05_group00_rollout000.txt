"""
Propose a step‑function that minimises
    F(a) = 2·n·max(a*a) / (sum(a)²)

The optimisation is performed in three stages:

1. **Diverse warm‑starts** – several analytical families (uniform,
   sine‑squared, triangular, arcsine‑like, edge‑power‑law), the current
   best solution and a few random Dirichlet draws.

2. **Smooth‑max Adam** – the vector is represented by logits → soft‑max
   (so it always stays a probability distribution).  The objective
   “max(conv)” is replaced by a differentiable log‑sum‑exp surrogate,
   whose temperature λ is increased steadily to approach the true
   maximum.  Adam (β₁=0.9, β₂=0.999) is used with a modest learning‑rate
   schedule.

3. **Exact sub‑gradient & local‑search** – after the surrogate phase we
   switch to the true max, compute a sub‑gradient from the active set
   (all entries attaining the maximum) and perform an exponentiated‑
   gradient (mirror‑descent) update.  A lightweight simulated‑annealing
   hill‑climber then tries random perturbations to escape any remaining
   plateau.

The routine respects the supplied wall‑clock budget and returns the
best sequence discovered (as a plain Python list).  All numbers are
non‑negative and the sequence is normalised to sum = 1, which makes the
objective homogeneous and removes the need for an explicit denominator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_val_holder: dict,
) -> np.ndarray:
    """
    Adam optimisation on the smooth‑max surrogate of max(conv).
    Updates *best_val_holder* in‑place when a new record is found.
    Returns the final probability vector.
    """
    eps = 1e-20                     # avoid log(0)
    a = np.maximum(a0, eps)
    a /= a.sum()
    logits = np.log(a)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.08                       # a bit larger than the baseline

    # temperature (smooth‑max) schedule
    lam = 0.2
    lam_factor = 1.02               # faster annealing

    # iteration budget – more iterations for smaller n
    max_iters = int(30000 * 1024 / n) + 3000
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        # --- soft‑max to stay on simplex ---------------------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)    # stabilise
        a = exp_log / exp_log.sum()

        # --- convolution & smooth‑max weights ----------------------------
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))     # unnormalised weights
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # --- gradient of smooth‑max wrt a --------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # backprop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- temperature annealing ---------------------------------------
        lam = min(lam * lam_factor, 1e7)

        # --- record best -------------------------------------------------
        # Because a is normalised, denominator = 1, so objective = 2*n*max_c
        val = 2.0 * n * max_c
        if val < best_val_holder["val"]:
            best_val_holder["val"] = val
            best_val_holder["seq"] = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:06d}] new best {val:.6f}")

        # --- stagnation handling -----------------------------------------
        if time.time() - last_improve > 5.0:
            # tiny random shake of logits to jump out of flat plateaus
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        # --- time check ---------------------------------------------------
        if time.time() > deadline:
            break

    # final normalisation (already on simplex)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement with exponentiated gradient
# ----------------------------------------------------------------------


def _subgradient_refine(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_val_holder: dict,
) -> np.ndarray:
    """
    Refine a candidate using the true max of the convolution.
    Uses a mirror‑descent (exponentiated‑gradient) update on the simplex.
    """
    eta = 0.65                     # step size for exponentiated gradient
    last_improve = time.time()

    for it in range(4000):
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set (indices attaining the maximum, with tolerance)
        active = np.isclose(conv, max_c, atol=1e-13)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max(conv) w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated (mirror) update, keeping a on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluate exact objective (denominator = 1)
        val = 2.0 * n * max_c
        if val < best_val_holder["val"]:
            best_val_holder["val"] = val
            best_val_holder["seq"] = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it:04d}] new best {val:.6f}")

        # stop if no improvement for a while
        if time.time() - last_improve > 5.0:
            break
        if time.time() > deadline:
            break

    return a

# Light‑weight simulated‑annealing hill‑climber
# ----------------------------------------------------------------------


def _local_search(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_val_holder: dict,
) -> np.ndarray:
    """
    Randomly perturb the current vector and accept improvements (or,
    with a small probability, uphill moves).  Operates directly on the
    probability simplex.
    """
    temperature = 0.02                # initial acceptance temperature
    temp_decay = 0.9995               # geometric cooling
    last_improve = time.time()

    while time.time() < deadline:
        # propose a multiplicative perturbation of a few coordinates
        pert = np.random.randn(n) * 0.1      # relative Gaussian noise
        mask = np.random.rand(n) < 0.05      # ~5 % of entries are moved
        proposal = a.copy()
        proposal[mask] *= np.exp(pert[mask])  # multiplicative
        proposal = np.maximum(proposal, 0.0)
        sum_p = proposal.sum()
        if sum_p < 1e-12:
            proposal = np.random.dirichlet(np.ones(n))
            sum_p = proposal.sum()
        proposal /= sum_p

        # compute objective for proposal (fast via convolution)
        conv = _full_convolution(proposal, fft_len=_next_pow2(2 * n - 1))
        val = 2.0 * n * conv.max()

        cur_val = 2.0 * n * _full_convolution(a, fft_len=_next_pow2(2 * n - 1)).max()

        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / temperature):
            a = proposal
            if val < best_val_holder["val"]:
                best_val_holder["val"] = val
                best_val_holder["seq"] = a.copy()
                last_improve = time.time()
                print(f"[n={n} LocalSearch] new best {val:.6f}")

        # cooling
        temperature = max(temperature * temp_decay, 1e-6)

        # early exit if stagnating
        if time.time() - last_improve > 7.0:
            break

    return a

def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _shape_sine_squared(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / n) ** 2
    a = a / a.sum()
    return a


def _shape_triangular(n: int) -> np.ndarray:
    # Linear ramp up to centre then down – symmetric triangle
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(i - center) / center
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def _shape_arcsine_like(n: int) -> np.ndarray:
    # Discrete analogue of the arcsine density: 1/sqrt(i*(n-i))
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1.0)
    a = 1.0 / np.sqrt(prod)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def _shape_edge_powerlaw(n: int, alpha: float) -> np.ndarray:
    # a_i ∝ (i·(n-i+1))^alpha   (alpha < 0 concentrates mass at edges)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    if shape.sum() == 0:
        return np.random.dirichlet(np.ones(n))
    shape = shape / shape.sum()
    return shape


def _shape_two_gaussian_edges(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * (i / sigma) ** 2)
    g2 = np.exp(-0.5 * ((n - 1 - i) / sigma) ** 2)
    a = g1 + g2
    a = a / a.sum()
    return a

        if best_holder["seq"] is not None and len(best_holder["seq"]) < n:
            old = np.asarray(best_holder["seq"], dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            if up.sum() > 0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # historic global variable, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 0:
                h = h / h.sum()
            else:
                h = np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # a few random Dirichlet draws
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # ----------------------------------------------------------------
        # Optimise each candidate in turn
        # ----------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Stage 1: Adam on smooth‑max ---------------------------
            a_opt = _adam_optimize(
                a0=a0,
                n=n,
                conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len,
                deadline=deadline,
                best_val_holder=best_holder,
            )

            # ---- Stage 2: Exact sub‑gradient refinement -----------------
            a_opt = _subgradient_refine(
                a=a_opt,
                n=n,
                conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len,
                deadline=deadline,
                best_val_holder=best_holder,
            )

            # ---- Stage 3: Local stochastic search -----------------------
            a_opt = _local_search(
                a=a_opt,
                n=n,
                deadline=deadline,
                best_val_holder=best_holder,
            )

            # quick sanity check – keep the best in the outer scope
            if best_holder["seq"] is not None:
                best_seq = best_holder["seq"]
                best_val = best_holder["val"]

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = _shape_uniform(fallback_n)

    print(f"\nBest value found: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
