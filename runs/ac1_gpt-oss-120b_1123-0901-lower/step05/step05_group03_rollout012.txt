def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start+a.shape[0]]

    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)  # sum = 1 after projection
        if val < global_best_val:

def _adam_optimize(a0,n,conv_fft_len,corr_fft_len,deadline,best_seq,best_val):
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1,beta2,eps_adam = 0.9,0.999,1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.005  # slower increase for smoother
    max_iters = 500_000
    last_improve = time.time()
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it}] new best {val:.6f}")
        # smooth max weights
        # stable compute exp(lam*(conv-max_c))
        w = np.exp(lam * (conv - max_c))
        sum_w = w.sum()
        if sum_w == 0 or np.isnan(sum_w):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= sum_w
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # update lam dissipily
        lam = min(lam * lam_factor, 1e5)
        # shake if stagnated
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        it += 1
    return best_seq, best_val

def _proj_simplex(v):
    """Project onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # shift v by subtracting min, but not needed.
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all entries non-positive, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _generate_candidate_shapes(n):
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0 / n))
    # sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    candidates.append(shape / shape.sum())
    # Dirichlet random draws
    for _ in range(5):
        cand = np.random.dirichlet(np.ones(n))
        candidates.append(cand)
    # power law edge shapes
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1)
    eps = 1e-12
    for p in np.linspace(-4.0, -0.5, 12):
        shape = (prod + eps) ** p
        shape = np.maximum(shape, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)
    # exponential edge
    for gamma in np.linspace(0.1, 5.0, 15):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        candidates.append(shape / shape.sum())
    # combine edge and sinusoid
    for gamma in [0.5, 1.0, 2.0]:
        i = np.arange(n, dtype=np.float64)
        shape = (np.exp(-gamma*i) + np.exp(-gamma*(n-1-i))) * (np.sin(np.pi * (i + 0.5) / n) + 0.1)
        candidates.append(shape / shape.sum())
    return candidates

"""
Propose a step‑function (non‑negative vector) with a low value of

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm works as follows:

*   If a previous construction is available (`height_sequence_1`) it is
    interpolated to the working size (n = 4096) and used as a warm start.
    Otherwise a small pool of diverse shapes (uniform, sinusoid,
    edge‑exponential, power‑law, triangular, random Dirichlet) is built
    and the best one according to the official `evaluate_sequence`
    routine is selected.

*   The selected vector is refined by two optimisation phases that respect
    the time budget (default 1000 s):
    – **Adam on a smooth‑max surrogate** – the maximum convolution entry
      is approximated by a log‑sum‑exp with a temperature that slowly
      rises; the optimisation is performed on the simplex via a
      soft‑max (log‑its) parametrisation and the Adam optimiser.
    – **Projected‑gradient sub‑gradient descent** – after the surrogate
      has converged the exact max is used; the gradient of the max
      convolution is taken on the active set and a gradient‑step is
      projected back onto the probability simplex (mirror‑descent).

*   When enough time remains the best distribution is linearly interpolated
    to a larger grid (n = 8192) and a short refinement is performed; the
    better of the two lengths is finally returned.

All numbers are clipped to the range [0, 1000] before returning.
"""

import time
import numpy as np

# ------------------------------------------------------------
# FFT‑based convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a⋆a (length 2·len(a)−1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k−(n−1)]
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ------------------------------------------------------------
# Simplex projection (Duchi et al.)
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:          # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Candidate‑generation helpers
def _candidate_shapes(n: int) -> list[np.ndarray]:
    """Create a modest pool of diverse start vectors (all sum to 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # positive half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    cand.append(s / s.sum())

    # simple triangular (tent) shape
    t = np.arange(1, n + 1, dtype=np.float64)
    t = np.minimum(t, n - t + 1)
    cand.append(t / t.sum())

    # random Dirichlet draws (α = 1)
    for _ in range(5):
        cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1) + 1e-12
    for p in np.linspace(-4.0, -0.5, 12):
        shape = prod ** p
        shape = np.maximum(shape, 0.0)
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # exponential edge shapes
    for gamma in np.linspace(0.1, 5.0, 15):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # mixed edge‑exponential × sinusoid
    for gamma in (0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = (np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))) \
                * (np.sin(np.pi * (i + 0.5) / n) + 0.1)
        cand.append(shape / shape.sum())

    return cand


# ------------------------------------------------------------
# Optimisation phases
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)              # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                               # start temperature
    lam_factor = 1.005                    # slow schedule
    last_improve = time.time()
    it = 0
    max_iters = 500_000

    while time.time() < deadline and it < max_iters:
        # ------- simplex from logits -------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ------- smooth max -------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits (we move *against* the gradient)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, 1e5)

        # occasional random shake when stagnating
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _pgd_subgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient (sub‑gradient) descent on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)

    eta = 0.5               # initial step size
    eta_decay = 0.9999
    last_improve = time.time()
    it = 0
    max_iters = 60_000

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[PGD n={n} it={it:05d}] new best {val:.6f}")

        # active set of maximal convolution entries
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient step and projection onto the simplex
        a = _proj_simplex(a - eta * grad)

        eta *= eta_decay
        it += 1

        # stop if we have not improved for a while
        if time.time() - last_improve > 10.0:
            break

    return best_seq, best_val


def _optimize_candidate(
    a0: np.ndarray,
    n: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the full optimisation pipeline (Adam → PGD) for a single start.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq, best_val = _adam_optimize(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _pgd_subgrad_optimize(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    """

    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------

    return best_seq.tolist()
