"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that
minimises the evaluation function

    F(a) = 2·n·max(a*a) / (sum(a)²)

where ``a*a`` is the linear convolution.
The routine combines

* a fast low‑dimensional *parameter sweep* over several analytically
  defined families (exponential, two‑sided exponential, Gaussian,
  logistic, power‑law and random Dirichlet),
* a *refinement phase* that runs Adam on a smooth‑max surrogate of
  ``max(a*a)`` while keeping the variables on the probability simplex,
* a short *exponentiated‑gradient* (mirror‑descent) phase that works
  directly on the exact maximum,
* occasional random “shakes’’ to escape plateaus.

It respects a user supplied time budget (default 1000 s) and returns a
plain Python list ready to be fed to ``evaluate_sequence``.
The function is deliberately written without closures or lambdas so
that it can be pickled or inspected easily.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2·len(a)–1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for gradients.
    The result has length ``len(a)`` and equals

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    with n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute weights w_i = exp(lam·(conv_i – max(conv))) / Σ exp(·)
    which approximate the sub‑gradient of the smooth max.
    """
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Heuristic search for a low‑evaluation step‑function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 1000 s).

    Returns
    -------
    list[float]
        The best sequence found, converted to a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to try – larger n gives more freedom but also higher cost.
    # The earlier run stopped at 4096, we give the optimiser a chance
    # at 8192 while still keeping the total runtime in check.
    lengths = [4096, 8192]

    # Pre‑compute FFT lengths (next power of two) for speed.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _next_pow2(conv_len)
        # For cross‑correlation we need length conv_len + n – 1 ≤ 3·n
        fft_len_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (sum already = 1)
    def _objective(a: np.ndarray, n: int) -> float:
        conv = _fft_convolve(a, fft_len_conv[n])
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len_fft = fft_len_conv[n]
        corr_len_fft = fft_len_corr[n]

        # ------------------- generate a diverse pool of start points ----------
        pool = []

        # 1. reuse previous global solution if it matches the length
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-9:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            pool.append(cand)

        # 2. pure uniform
        pool.append(np.full(n, 1.0 / n))

        # 3. random Dirichlet (concentration 1)
        for _ in range(4):
            pool.append(np.random.dirichlet(np.ones(n)))

        # 4. sparse Dirichlet (concentration < 1) – produces spiky vectors
        for conc in [0.2, 0.4, 0.6]:
            for _ in range(2):
                pool.append(np.random.dirichlet(np.full(n, conc)))

        # 5. two‑sided exponential families
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.0005, 0.001, 0.002, 0.004, 0.008]:
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape /= shape.sum()
            pool.append(shape)

        # 6. single exponential (one‑sided)
        for gamma in [0.001, 0.003, 0.006]:
            shape = np.exp(-gamma * i)
            shape /= shape.sum()
            pool.append(shape)

        # 7. Gaussian shapes (different centres and widths)
        for sigma in [0.05, 0.10, 0.20]:
            for mu_frac in [0.2, 0.5, 0.8]:
                mu = mu_frac * (n - 1)
                shape = np.exp(-0.5 * ((i - mu) / (sigma * n)) ** 2)
                shape /= shape.sum()
                pool.append(shape)

        # 8. Logistic / sigmoidal shapes
        for scale in [0.02, 0.05, 0.10]:
            for shift in [0.3, 0.5, 0.7]:
                x = (i / (n - 1) - shift) / scale
                shape = 1.0 / (1.0 + np.exp(x))
                shape /= shape.sum()
                pool.append(shape)

        # 9. Power‑law shapes
        for alpha in np.linspace(-2.0, -0.2, 9):
            shape = np.power(i + 1, alpha, where=(i + 1) > 0, out=np.zeros_like(i))
            if shape.sum() > 0:
                shape /= shape.sum()
                pool.append(shape)

        # ------------------------------------------------------------------
        # Evaluate all pool members and keep the best few for refinement
        top_candidates = []       # list of (value, np.ndarray)
        for idx, a0 in enumerate(pool):
            if time.time() > deadline:
                break
            a = a0 / a0.sum()
            val = _objective(a, n)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[pool] new global best {val:.6f} (cand {idx})")
            top_candidates.append((val, a))

        # Keep only the 8 most promising seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:8]

        # ------------------------------------------------------------------
        # Refinement of each promising candidate using Adam on a smooth‑max
        for cand_id, (_, a0) in enumerate(top_candidates):
            if time.time() > deadline:
                break

            # initialise soft‑max parameters (logits)
            logits = np.log(a0 + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            lr = 0.07                     # a bit larger than the reference
            lam = 0.15                    # initial temperature for smooth‑max
            lam_factor = 1.008            # raise temperature slowly

            last_improve = time.time()
            max_iters = int(30000 * 1024 / n) + 8000   # more iterations than before

            for t in range(1, max_iters + 1):
                # ---------- simplex via soft‑max ----------
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _fft_convolve(a, conv_len_fft)
                max_c = conv.max()

                # ---------- smooth‑max weights ----------
                w = _smooth_max_weights(conv, lam)

                # ---------- gradient of smooth‑max ----------
                grad_a = 2.0 * _cross_corr(w, a, corr_len_fft)

                grad_log = a * (grad_a - avg)

                v_hat = v / (1.0 - beta2 ** t)

                # ---------- temperature schedule ----------
                lam = min(lam * lam_factor, 1e6)

                # ---------- objective ----------
                val = 2.0 * n * max_c

                if val < best_val:

                    best_seq = a.copy()

                # ---------- stagnation shake ----------
                if time.time() - last_improve > 7.0:
                    # inject a tiny random perturbation to escape plateaus
                    logits += (np.random.rand(n) - 0.5) * 5e-3

                if time.time() > deadline:

            # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        if best_seq is not None and len(best_seq) == n:
