Search algorithm for minimizing the max convolution objective.

The algorithm:
1. For each chosen length n (powers of two up to 16384) generate a
   diverse set of initial shapes (exponential, power‑law, random monotone,
   symmetric tent, etc.).
2. For each shape, run a simple projected exponential‑gradient descent
   on the exact max‑convolution objective.  The gradient is a sub‑gradient:
   for the active set of indices achieving the current maximum,
   grad = 2 * cross_correlation(w, a) where w is a uniform distribution
   over the active set.
3. The step size η decays gradually; after every few hundred iterations
   a small random perturbation is added to escape plateaus.
4. The best sequence found so far is kept globally; after processing all
   candidates a final fine‑tuning pass with Adam (smooth‑max) is applied
   to the current best for the same length, using many more iterations.
5. The routine respects a time budget (default 1000 s) and returns the
   best sequence discovered as a plain Python list.
"""

import time
import numpy as np

def _fft_len_pow2(x):
    # smallest power of two >= x
    return 1 << (x - 1).bit_length()

def _conv_fft_len(n):
    return _fft_len_pow2(2 * n - 1)

def _corr_fft_len(n):
    # length needed for cross-correlation (len(w)+len(a)-1)
    return _fft_len_pow2(3 * n - 2)

def _full_convolution(a, fft_len):
    """Linear convolution a*a using FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2*n-1 entries (linear conv length)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    """Cross‑correlation w ★ a (length len(a))."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    # slice out the central part of length n
    start = n - 1
    return corr[start:start + n]

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # implementation from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    """Helper that evaluates a sequence normalized to sum = 1."""

    # using the known formula: 2*n*max(conv) under sum=1
    n = a.shape[0]
    conv = _full_convolution(a, _conv_fft_len(n))
    return 2.0 * n * conv.max()

        conv_len = _conv_fft_len(n)

        init_pool = []

        # 3. one‑sided exponential families
        for gamma in np.logspace(-4, -1, 12):

            shape /= shape.sum()
            init_pool.append(shape)

            shape = idx ** alpha
            if shape.sum() > 0:

                init_pool.append(shape)

        for _ in range(6):
            rnd = np.random.rand(n)
            rnd.sort()
            shape = rnd[::-1]  # descending
            shape /= shape.sum()
            init_pool.append(shape)

        # 6. Dirichlet draws with various concentrations (including <1)
        for conc in [0.2, 0.5, 0.8, 1.0, 2.0]:
            shape = np.random.dirichlet(np.full(n, conc))
            shape /= shape.sum()
            init_pool.append(shape)

        # ------------------------------------------------------------------
        # Main optimisation loop over initial candidates
        for cand_id, a0 in enumerate(init_pool):
            if time.time() > deadline:
                break

            # Normalise to sum = 1
            a = a0 / a0.sum()
            # Initialise step size (exponential‑gradient)
            eta = 0.8
            # Use a simple schedule that shrinks η slowly
            eta_decay = 0.9995

            last_improve = time.time()

            # Decide on iteration budget based on remaining time and n
            # Roughly allocate a few hundred milliseconds per 100 iterations.
            # Let us limit total iterations per candidate to 20000 (or less).
            max_iters = 20000
            for it in range(max_iters):

                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} itr={it}] new best {val:.6f}")

                # convergence / time checks
                if time.time() > deadline:
                    break
                # Stagnation handling: every few seconds add noise
                if time.time() - last_improve > 6.0:
                    # small random perturbation of logits (log‑space)
                    log_a = np.log(a + 1e-15)
                    log_a += (np.random.rand(n) - 0.5) * 1e-2
                    a = np.exp(log_a)
                    a /= a.sum()
                    last_improve = time.time()
                    # optionally reset step size
                    eta = max(eta * 0.5, 0.01)

                # sub‑gradient of the max (uniformly over active set)
                eps = 1e-12
                active = np.isclose(conv, max_c, atol=eps)
                if not active.any():

                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()

                # gradient w.r.t. a
                grad = 2.0 * _cross_correlation(w, a, corr_len)

                # exponentiated‑gradient step (maintains positivity)
                a = a * np.exp(-eta * grad)
                a_sum = a.sum()
                if a_sum <= 0 or np.isnan(a_sum):
                    # re‑initialise if something went wrong
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a /= a_sum

                # decay step size
                eta *= eta_decay

            # End of candidate optimisation
        # ------------------------------------------------------------------
        # Optional fine‑tune phase for the global best at this length
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning best candidate (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            lam = 0.4
            lam_factor = 1.01
            # allocate a generous number of Adam iterations for this length
            fine_iters = int(50000 * 1024 / n) + 8000
            last_improve = time.time()
            for t in range(1, fine_iters + 1):
                if time.time() > deadline:
                    break
                # soft‑max to enforce simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w /= w_sum
                # gradient of smooth‑max
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
                # temperature schedule
                lam = min(lam * lam_factor, 1e6)
                # value
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t}] new best {val:.6f}")

                # occasional jitter to escape plateaus
                if time.time() - last_improve > 12.0:
                    logits += (np.random.rand(n) - 0.5) * 4e-3
                    last_improve = time.time()
        # ------------------------------------------------------------------
    # If nothing was found – fall back to uniform (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list as required
    return best_seq.tolist()

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _evaluate_normalised(a: np.ndarray) -> float:
    """
    Compute the exact objective for a normalised vector (sum(a)=1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _full_convolution(a, _conv_fft_len(n))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The search respects the given time budget and returns the
    best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin for clean exit

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – we start from the length that already gave a
    # decent bound (4096) and then try larger powers of two.
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = _conv_fft_len(n)
        corr_len = _corr_fft_len(n)

        # ------------------------------------------------------------------
        # Build a pool of initial shapes.
        # ------------------------------------------------------------------
        candidates = []

        # Warm‑start from the globally best sequence if it already has
        # the correct length.
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # Upscaled version of the best (if shorter).
        if best_seq is not None and len(best_seq) < n:
            src = np.arange(len(best_seq))
            dst = np.linspace(0, len(best_seq) - 1, n)
            up = np.interp(dst, src, best_seq)
            up = up / up.sum()
            candidates.append(up)
            candidates.append(up[::-1])    # also try the reversed version

        # Symmetric tent shape (already used in the reference code).
        left = np.linspace(0, 1, n // 2, endpoint=False)
        right = left[::-1]
        tent = np.concatenate([left, right])
        tent = np.maximum(tent, 0)
        tent /= tent.sum()
        candidates.append(tent)

        # One‑sided exponential families (decay towards the right).
        for gamma in np.logspace(-4, -1, 12):
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i)
            shape /= shape.sum()
            candidates.append(shape)

        # Power‑law decreasing families (i+1)^α.
        idx = np.arange(1, n + 1, dtype=np.float64)
        for alpha in np.linspace(-2.5, -0.1, 15):
            shape = idx ** alpha
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # Random monotone decreasing vectors (sorted random numbers).
        for _ in range(6):
            rnd = np.random.rand(n)
            rnd.sort()
            shape = rnd[::-1]          # descending order
            shape /= shape.sum()
            candidates.append(shape)

        # Dirichlet draws with several concentration parameters.
        for conc in [0.2, 0.5, 0.8, 1.0, 2.0]:
            shape = np.random.dirichlet(np.full(n, conc))
            shape /= shape.sum()
            candidates.append(shape)

        # ------------------------------------------------------------------
        # Optimize each candidate with exponentiated‑gradient sub‑gradient.
        # ------------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1).
            a = a0 / a0.sum()

            # Step‑size schedule for the exponentiated‑gradient phase.
            eta = 0.8
            eta_decay = 0.9995

            last_improve = time.time()

            # Choose an iteration budget that roughly respects the remaining
            # time.  We cap it to a safe maximum.
            max_iters = 20000
            # For very large n we lower the iteration count a bit.
            if n >= 8192:
                max_iters = 12000

            for it in range(max_iters):
                # --- evaluate -------------------------------------------------
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} it={it}] new best {val:.6f}")

                # --- time / stagnation checks --------------------------------
                if time.time() > deadline:
                    break
                if time.time() - last_improve > 6.0:
                    # Small random perturbation in log‑space to escape plateaus.
                    log_a = np.log(a + 1e-15)
                    log_a += (np.random.rand(n) - 0.5) * 1e-2
                    a = np.exp(log_a)
                    a /= a.sum()
                    last_improve = time.time()
                    eta = max(eta * 0.5, 0.01)

                # --- sub‑gradient of the max --------------------------------
                eps = 1e-12
                active = np.isclose(conv, max_c, atol=eps)
                if not active.any():
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()   # uniform over active set

                grad = 2.0 * _cross_correlation(w, a, corr_len)

                # --- exponentiated‑gradient step (keeps positivity) -------
                a = a * np.exp(-eta * grad)
                s = a.sum()
                if s <= 0 or np.isnan(s):
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a /= s

                # --- decay step size ----------------------------------------
                eta *= eta_decay

        # ------------------------------------------------------------------
        # Fine‑tune the current global best for this length using Adam on
        # a smooth‑max surrogate (as in the reference code).
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            lam = 0.4
            lam_factor = 1.01
            # Allocate a generous number of Adam iterations – it scales
            # inversely with n, so larger n gets proportionally more work.
            fine_iters = int(50000 * 1024 / n) + 8000
            last_improve = time.time()

            for t in range(1, fine_iters + 1):
                if time.time() > deadline:
                    break

                # Soft‑max to enforce the simplex.
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # Convolution and smooth‑max weights.
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w /= w_sum

                # Gradient of the smooth‑max.
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits.
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # Temperature schedule – makes the smooth‑max tighter.
                lam = min(lam * lam_factor, 1e6)

                # Current exact value.
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune it={t}] new best {val:.6f}")

                # Periodic jitter to escape tiny plateaus.
                if time.time() - last_improve > 12.0:
                    logits += (np.random.rand(n) - 0.5) * 4e-3
                    last_improve = time.time()

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed).
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
