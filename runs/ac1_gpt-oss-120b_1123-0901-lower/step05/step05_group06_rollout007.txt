"""
Propose a non‑negative step‑function (a vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* Choose a handful of lengths (up to 8000).  For each length we run a
  stochastic mirror‑descent on the simplex (exponentiated‑gradient /
  soft‑max Adam).  The objective is the *smooth* max of the linear
  convolution a∗a, i.e. a log‑sum‑exp of the convolution values.  The
  temperature is annealed exponentially.
* A diverse set of initial vectors (uniform, Dirichlet, exponentials,
  window functions, random perturbations and an interpolation of the
  previous best sequence) is used for each length.
* The best vector seen overall is kept.
* After the time budget for the gradient phase expires a cheap stochastic
  refinement (pairwise mass transfers with simulated‑annealing acceptance)
  is performed on the global best.
* Throughout the run we respect the time limit (default 1000 s) and stop
  cleanly before the deadline.

The code is written with only top‑level functions (no closures) and uses
NumPy FFTs for O(n log n) convolution / gradient computation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of an array.  Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract max for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k * a_{k-i}   (out‑of‑range indices ignored)
    using FFTs.  ``a`` has length n, ``w`` has length 2n‑1.
    The result is an array of length n.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)      # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # linear part of the convolution: length = n + (2n‑1) – 1 = 3n‑2
    conv_rev = conv_rev[: 3 * a.shape[0] - 2]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of the linear convolution a∗a."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Return a modestly diverse list of start vectors (already normalised)."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (γ=1)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.015)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak at centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sine‑shaped (half‑period)
    sine = np.sin(np.pi * np.arange(n) / (n - 1))
    init.append(sine / sine.sum() * target_sum)

    # 10) random Dirichlet with higher concentration (α=0.3) → spiky
    g = rng.gamma(shape=0.3, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the probability simplex.
    Uses a soft‑max (log‑sum‑exp) approximation of the max‑convolution.
    Returns the best vector found in this run together with its exact
    max‑convolution value.
    """
    # initialise logits (add a tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.10                       # a little larger than before

    # soft‑max temperature schedule
    tau_factor = 0.12                # smaller than the previous 0.30 → tighter approximation
    decay_rate = max_iters / 6.0     # decay a bit faster

    # FFT size – must accommodate linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # jitter to escape plateaux (more frequent than before)
    jitter_interval = 1500

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- probability simplex via soft‑max of logits ----
        logits = y - np.max(y)                # for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()     # lies on the simplex
        a = target_sum * p                     # Σa = target_sum, a_i ≥ 0

        # ---- exact convolution and its maximum ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- soft‑max weights for the max ----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-10:
            # when τ is tiny we revert to a uniform weight over all active maxima
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---- gradient of the smoothed max w.r.t. a ----
        rev = _weighted_rev(a, w, Nfft)          # shape (n,)
        g_a = 2.0 * rev                           # derivative of Σ a_i a_{k-i}

        # ---- chain rule: gradient w.r.t. logits (mirror‑descent) ----
        avg = np.dot(p, g_a)                     # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)       # ∂f/∂y_i

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)             # mild decay
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter on the logits
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.005, size=n)
            y += jitter

        # ---- keep the best exact solution seen so far ----
        if M < best_score - 1e-12:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex.
    Moves a random fraction of mass from i→j (pairwise transfer) and
    occasionally performs a 3‑way split/merge.  Improvements are always
    accepted; worsening moves follow a simple simulated‑annealing schedule.
    """
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for exact max‑convolution checks
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule parameters
    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        move_type = rng.integers(0, 3)   # 0: pairwise, 1: three‑way, 2: small random jitter

        if move_type == 0:
            # pairwise transfer
            i, j = rng.integers(0, n, size=2, endpoint=False)
            if i == j or cur_seq[i] < 1e-14:
                step += 1
                continue
            frac = rng.uniform(0.01, 0.07)
            delta = cur_seq[i] * frac
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif move_type == 1:
            # three‑way redistribution
            i, j, k = rng.integers(0, n, size=3, endpoint=False)
            if i == j or i == k or cur_seq[i] < 1e-14:
                step += 1
                continue
            frac1 = rng.uniform(0.01, 0.05)
            frac2 = rng.uniform(0.01, 0.05)
            delta_total = cur_seq[i] * (frac1 + frac2)
            new_seq = cur_seq.copy()
            new_seq[i] -= delta_total
            new_seq[j] += cur_seq[i] * frac1
            new_seq[k] += cur_seq[i] * frac2
        else:
            # small jitter on a random coordinate (helps breaking ties)
            idx = rng.integers(0, n)
            eps = rng.normal(scale=0.001 * cur_seq[idx])
            new_seq = cur_seq.copy()
            new_seq[idx] = max(new_seq[idx] + eps, 0.0)   # keep non‑negative
            # re‑normalise to keep the total sum unchanged
            diff = new_seq[idx] - cur_seq[idx]
            if diff != 0.0:
                # distribute the opposite amount uniformly over the other entries
                other = np.arange(n) != idx
                new_seq[other] -= diff / (n - 1)
                # clamp again to avoid tiny negatives
                new_seq = np.maximum(new_seq, 0.0)
                # renormalise to exact original sum (critical for evaluation)
                new_seq *= cur_seq.sum() / new_seq.sum()

        # evaluate the new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # acceptance
        if new_score < cur_score - 1e-12:
            cur_seq, cur_score = new_seq, new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing probability
            T = max(0.02 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq, cur_score = new_seq, new_score

        step += 1

    return cur_seq, cur_score

    length_factors = [0.5, 0.75, 1.0, 1.33, 2.0, 2.66, 3.33, 4.0]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in length_factors})

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None          # numpy array
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # collect diverse starts; also interpolate the old best when size changes
        init_vectors = _make_initial_vectors(n, target_sum, rng)

        if base is not None and base.shape[0] != n:
            # linear interpolation of the previous best to the new size
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # allocate a per‑length iteration budget (more for smaller n)
        # The total budget is capped by the global deadline.
        max_iters = int(2_000_000 * 2100 / n)
        max_iters = max(200_000, max_iters)

        print(f"[run] length={n:5d}  target_sum={target_sum:.3f}  max_iters≈{max_iters}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for n={n}")

            # run the exponentiated‑gradient optimiser
            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Local stochastic refinement (if any time left)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # spend at most half of the remaining time on refinement (capped at 30 s)
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe guard
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best max‑conv = {best_score:.6f}, length = {best_seq.shape[0]}")
    # The evaluation function treats the vector as a step‑function,
    # so we simply return the raw list.
    return best_seq.tolist()
