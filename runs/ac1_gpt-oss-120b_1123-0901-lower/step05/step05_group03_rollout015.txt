def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    start_time = time.time()

    # size
    n = 4096
    fft_len = _next_pow2(2*n-1)

    # warm start
    warm = globals().get('height_sequence_1')
    if warm is not None:
        ws = np.asarray(warm, dtype=np.float64)
        if ws.shape[0] != n:
            if ws.shape[0] > n:
                ws = ws[:n]
            else:
                ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() < 0.01:
            ws = None
        else:
            ws = ws / ws.sum()
    else:
        ws = None

    # initial candidates
    candidates = _initial_candidates(n, ws)

    best_seq = None
    best_val = float('inf')
    for cand in candidates:
        if time.time() >= deadline:
            break
        seq = np.asarray(cand, dtype=np.float64)
        if seq.sum() < 0.01:
            seq = np.random.dirichlet(np.ones(n))
        else:
            seq = seq / seq.sum()
        val = _objective(seq, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[init] new best {best_val:.6f}")

    # time split: allocate 40% for hill climbing, rest for deep optimization
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq.tolist() if best_seq is not None else np.full(n, 1.0/n).tolist()
    # allocate hill climbing time
    hc_frac = 0.4
    hc_deadline = time.time() + remaining * hc_frac

    # hill climbing
    best_seq, best_val = _hill_climb(best_seq, n, fft_len, hc_deadline, best_seq, best_val)

    # deep optimizer for the rest
    if time.time() < deadline:
        best_seq, best_val = _deep_optimize(best_seq, n, fft_len,

    # clip and return list
    best_seq = np.clip(best_seq, 0.0, 1000.0)

def _initial_candidates(n, warm):
    cand = []
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power-law edges as before (maybe keep)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # double Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        cand.append(shape)

    # sinusoidal half-wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

"""
Propose a low‑evaluation‑score step function.

The algorithm mixes diverse deterministic initial shapes, a log‑space
hill‑climbing phase (multiplicative random perturbations with a
temperature schedule) and a final Adam + exponentiated‑gradient refinement.
All work is done on the probability simplex (sum = 1); the evaluation
function is scale‑invariant, hence this does not affect the final score.
The routine respects the given time budget and returns the best vector
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, length 2·len(a)‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)^2 (sum a ≈ 1)."""
    conv = _full_convolution(seq, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (seq.sum() ** 2))


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a modest pool of diversified start vectors."""
    cand = []

    # warm start and rotated copies
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # classic windows (all non‑negative)
    # Hann
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    cand.append(shape)

    # Hamming
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    cand.append(shape)

    # triangular (linear ramp up then down)
    shape = 1 - np.abs(i - (n - 1) / 2) / ((n - 1) / 2)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # parabolic (concave quadratic)
    shape = 1 - ((i - (n - 1) / 2) / ((n - 1) / 2)) ** 2
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                fft_len: int,
                deadline: float,
                best_seq: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Log‑space multiplicative hill‑climbing.
    Random Gaussian perturbations on log‑probabilities with a cooling schedule.
    Accepts uphill moves with a temperature‑dependent probability.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq, n, fft_len)

    start_time = time.time()
    sigma_max, sigma_min = 0.5, 0.01      # noise amplitude (log‑space)
    T_max, T_min = 5e-3, 1e-5            # acceptance temperature

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0

        sigma = sigma_max * (1.0 - frac) + sigma_min * frac
        T = T_max * (1.0 - frac) + T_min * frac

        # multiplicative log‑space perturbation
        log_cur = np.log(cur_seq + 1e-12)
        log_new = log_cur + sigma * np.random.randn(n)
        seq_new = np.exp(log_new)
        seq_new_sum = seq_new.sum()
        if seq_new_sum == 0:
            continue
        seq_new /= seq_new_sum

        val_new = _objective(seq_new, n, fft_len)

        # accept if better or with a Metropolis probability
        if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / max(T, 1e-12)):
            cur_seq = seq_new
            cur_val = val_new

        if val_new < best_val:
            best_val = val_new
            best_seq = seq_new.copy()
            print(f"[hill] new best {best_val:.6f}  (t={time.time() - start_time:.1f}s)")

    return best_seq, best_val

def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by an exponentiated‑gradient
    sub‑gradient phase. Iteration caps are generous (up to the remaining time).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03          # slightly smaller than the original to aid stability
    lam = 0.5
    lam_factor = 1.015

    last_improve = time.time()
    it = 0
    # -------------------------------------------------- Adam phase
    while time.time() < deadline and it < 500_000:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – the smooth‑max becomes the true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter when stuck
        if time.time() - last_improve > 3.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    # -------------------------------------------------- Sub‑gradient phase
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.random.dirichlet(np.ones(n))

    eta = 0.6
    sub_last_improve = time.time()
    sub_it = 0
    while time.time() < deadline and sub_it < 20_000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={sub_it}] new best {val:.6f}")

        # active set (entries attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 5.0:
            break
        sub_it += 1

    return global_best_seq, global_best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin
    start_time = time.time()

    # ------------------------------------------------------------------
    # problem size (the current state‑of‑the‑art bound was achieved at 4096)
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # try to reuse a previously found construction
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() < 0.01:
            w = None
        else:
            w = w / w.sum()
    else:
        w = None

    # ------------------------------------------------------------------
    # build a diverse pool of starters and keep the best one
    candidates = _initial_candidates(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for cand in candidates:
        if time.time() >= deadline:
            break
        seq = np.asarray(cand, dtype=np.float64)
        if seq.sum() < 0.01:
            seq = np.random.dirichlet(np.ones(n))
        else:
            seq = seq / seq.sum()
        val = _objective(seq, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[init] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – log‑space hill climbing (≈40% of remaining budget)
    remaining = deadline - time.time()
    if remaining > 0 and best_seq is not None:
        hc_deadline = time.time() + remaining * 0.40
        best_seq, best_val = _hill_climb(best_seq, n, fft_len,
                                         hc_deadline, best_seq, best_val)

    # ------------------------------------------------------------------
    # Phase 2 – Adam + sub‑gradient refinement (rest of the time)
    if time.time() < deadline and best_seq is not None:
        best_seq, best_val = _deep_optimize(best_seq, n, fft_len,
                                            _next_pow2(3 * n - 2),
                                            deadline, best_seq, best_val)

    # ------------------------------------------------------------------
    # Phase 3 – a quick fine‑grained hill‑climb if there is a sliver left
    if time.time() < deadline and best_seq is not None:
        best_seq, best_val = _hill_climb(best_seq, n, fft_len,
                                         deadline, best_seq, best_val)

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range and return as a plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
