def project_onto_simplex(v):
    """Project onto the probability simplex (non-negative, sum=1)."""
    # v is 1D numpy array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm is a hybrid optimiser:

1. **Adam on a smooth approximation** – we replace the non‑smooth max by a
   temperature‑controlled log‑sum‑exp (smooth‑max).  Adam together with a
   temperature annealing quickly drives the sequence towards a good basin.

2. **Simulated‑annealing refinement** – after Adam we perform a large number of
   tiny “mass‑move” proposals (move a fraction of the mass from one index to
   another).  Moves are accepted with a Metropolis rule that slowly cools,
   allowing the search to escape local minima.

3. **Greedy local polishing** – a cheap hill‑climbing phase that repeatedly
   shifts a small amount of mass between two random coordinates and keeps the
   change when it improves the objective.

The whole search respects a global 1000 s wall‑time, uses only NumPy, and
starts from the best sequence found so far (`height_sequence_1`) when it is
available.  The returned list is a plain Python list of non‑negative numbers.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper functions (all top‑level, no closures)
# --------------------------------------------------------------------------- #

def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Compute a temperature‑smoothed max of ``conv`` and the corresponding
    softmax weights (∂smooth_max/∂conv).
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Time‑bounded optimiser that returns the best non‑negative sequence
    discovered within the allotted wall‑time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    best_seq = None          # best sequence (numpy array, sum ≈ 1)
    best_val = float("inf")  # best objective 2·n·max(conv)

    # Problem sizes – larger n gives a lower theoretical lower bound (≈1)
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (reused many times)
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # Number of restarts – more for the smaller problems
        if n <= 1024:
            restarts = 12
        elif n <= 2048:
            restarts = 8
        else:
            restarts = 5

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------------------------------------------------
            # Initialise the probability vector (logits)
            # -------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a0 = best_seq.copy()
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    a0 = cand[:n]
                else:
                    a0 = np.pad(cand, (0, n - cand.shape[0]))
                a0 = np.maximum(a0, 0.0)
                if a0.sum() < 0.01:
                    a0 = rng.dirichlet(np.ones(n))
                else:
                    a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a0 = rng.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random start")

            # -------------------------------------------------------------
            # Adam optimiser on the smooth‑max surrogate
            # -------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.1
            decay_every = 5000

            if n <= 1024:
                max_iters = 200_000
            elif n <= 2048:
                max_iters = 150_000
            else:
                max_iters = 120_000

            tau_initial = 0.15
            tau_final = 5e-7

            last_improve = time.time()
            best_val_local = best_val
            best_seq_local = None

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # probability vector from logits
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # convolution and current (true) objective
                conv = full_convolution(a, fft_len=fft_len_conv)
                cur_val = 2.0 * n * conv.max()

                if cur_val < best_val_local - 1e-12:
                    best_val_local = cur_val
                    best_seq_local = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best {cur_val:.6f}")

                # smooth‑max and its gradient
                tau = max(tau_final,
                          tau_initial * (1.0 - t / max_iters))
                _, weights = smooth_max_and_weights(conv, tau)

                # gradient of smooth‑max w.r.t. a  (∂smooth/∂a = 2·(weights ★ a))
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam step
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                if t % decay_every == 0:
                    lr *= 0.7

                # stagnation handling – gentle random jitter
                if time.time() - last_improve > 40.0:
                    jitter = (rng.random(n) - 0.5) * 6e-3
                    x += jitter
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # Record best from Adam phase
            if best_seq_local is not None and best_val_local < best_val - 1e-12:
                best_val = best_val_local
                best_seq = best_seq_local.copy()
                print(f"[n={n} restart={r}] post‑Adam best {best_val:.6f}")

            # -------------------------------------------------------------
            # Simulated‑annealing refinement (mass‑move proposals)
            # -------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                conv = full_convolution(a, fft_len=fft_len_conv)
                cur_val = 2.0 * n * conv.max()

                sa_iters = 25_000
                # initial temperature proportional to current objective
                T = max(cur_val * 0.05, 1e-3)
                decay_rate = 0.9994

                for it in range(sa_iters):
                    if time.time() > deadline:
                        break

                    i = rng.integers(0, n)
                    if a[i] < 1e-12:
                        continue
                    j = rng.integers(0, n)
                    if i == j:
                        continue

                    # move a random fraction (10‑50 %) of the mass at i towards j
                    eps_move = a[i] * (0.1 + 0.4 * rng.random())
                    a_new = a.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move

                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    delta = val_new - cur_val

                    accept = False
                    if delta < 0:
                        accept = True
                    else:
                        if rng.random() < math.exp(-delta / max(T, 1e-12)):
                            accept = True

                    if accept:
                        a = a_new
                        conv = conv_new
                        cur_val = val_new
                        if val_new < best_val - 1e-12:
                            best_val = val_new
                            best_seq = a.copy()
                            print(f"[SA n={n}] improved to {val_new:.6f}")

                    T *= decay_rate
                    if T < 1e-8:
                        T = 1e-8

            # -------------------------------------------------------------
            # Cheap greedy refinement (tiny mass shuffling)
            # -------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                conv_local = full_convolution(a_local, fft_len=fft_len_conv)
                cur_val_local = 2.0 * n * conv_local.max()

                for it in range(3_000):
                    if time.time() > deadline:
                        break
                    i, j = rng.choice(n, size=2, replace=False)
                    eps_move = min(a_local[i], 0.008) * (0.2 + 0.6 * rng.random())
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < cur_val_local - 1e-12:
                        a_local = a_new
                        conv_local = conv_new
                        cur_val_local = val_new
                        if val_new < best_val - 1e-12:
                            best_val = val_new
                            best_seq = a_local.copy()
                            print(f"[local n={n}] improved to {val_new:.6f}")

    # --------------------------------------------------------------------- #
    # Fallback – should never happen unless something went terribly wrong.
    # --------------------------------------------------------------------- #
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0
        print("Fallback to uniform vector.")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
