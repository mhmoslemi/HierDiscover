"""
search algorithm using multi-stage optimization, extended candidate families,
and long refinement, targeting evaluation score <= 1.503.
"""

import time, math, random
import numpy as np

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[:2*n-1]

def _cross_correlation(w, a, fft_len):
    # w length = 2n-1
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n-1
    return corr[start:start+n]

def _objective(a, n, fft_len):
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_c / (s*s)

def _project_simplex(v):
    # Projects onto the probability simplex (sum=1, v>=0)
    if v.ndim != 1:
        raise ValueError
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _interpolate_seq(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new

def _generate_initial_candidates(n, best_seq=None, hist_seq=None):
    cand = []

    # warm start from global best exact length
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # warm start by interpolating global best
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # historic sequence if provided
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # Edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 4):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # Two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # Inverse‑root edge shapes
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # Three‑Gaussian peaks
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # New families ---------------------------------------------------------
    # Cosine windows
    i = np.arange(n, dtype=np.float64)
    # Hann
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    cand.append(hann / hann.sum())
    # Hamming
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    cand.append(hamming / hamming.sum())
    # Blackman
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    cand.append(blackman / blackman.sum())
    # Triangular (tent)
    triangular = 1.0 - np.abs(2.0 * i / (n - 1) - 1.0)
    cand.append(triangular / triangular.sum())
    # Beta distributions (asymmetric)
    for a, b in [(0.2, 5.0), (5.0, 0.2), (0.5, 0.5), (2.0, 2.0), (0.8, 1.5), (1.5, 0.8)]:
        x = (i + 0.5) / n  # avoid 0 and 1 exactly
        beta_pdf = np.power(x, a - 1) * np.power(1 - x, b - 1)
        # handle near‑zero
        beta_pdf = np.where(beta_pdf > 0, beta_pdf, 0.0)
        if beta_pdf.sum() > 0:
            cand.append(beta_pdf / beta_pdf.sum())

    # Random sparse (few non‑zeros)
    for nz in (5, 10, 20):
        vec = np.zeros(n)
        idxs = np.random.choice(n, size=nz, replace=False)
        vals = np.random.rand(nz)
        vec[idxs] = vals
        total = vec.sum()
        if total > 0:
            cand.append(vec / total)

    # Shuffle and limit to a manageable size
    np.random.shuffle(cand)
    max_cand = 20 if n <= 4096 else 15
    return cand[:max_cand]

def _adam_optimize(a0, n, fft_len_c, fft_len_x, deadline, max_iters=2000):
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.3          # start with a softer max
    lam_factor = 1.04

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_c)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex via softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        # avoid overflow/underflow
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # Random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

    return best_a, best_val

def _subgrad_optimize(a0, n, fft_len_c, fft_len_x, deadline,
                      init_val=None, max_iters=2000):
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_c)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.65

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        # uniform weight on active indices
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # exponentiated gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_c)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 5.0:
            # stagnation -> break early
            break

    return best_a, best_val

def _hill_climb_optimize(a0, n, fft_len_c, deadline,
                         init_val=None, max_iters=3000):
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # pick two indices
        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue
        # move a fraction of mass from i1 to i2
        eps = np.random.rand() * 0.07 * a[i1]    # up to 7%
        if eps <= 0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-12:
            continue
        new_a /= new_a.sum()
        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 100 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasionally accept bad move
            if np.random.rand() < 0.001:
                a = new_a
    return best_a, best_val

def _perturb_and_refine(base_seq, n, fft_len_c, fft_len_x, deadline,
                        rounds=10):
    """Create noisy copies of base_seq and run a short sub‑gradient phase."""
    best_seq = base_seq.copy()
    best_val = _objective(best_seq, n, fft_len_c)
    for r in range(rounds):
        if time.time() > deadline:
            break
        # add zero‑mean Gaussian noise, keep non‑negative
        noise = np.random.randn(n) * 0.02
        cand = best_seq + noise
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
        else:
            cand = cand / cand_sum
        # short refinement
        a_sub, val_sub = _subgrad_optimize(cand, n, fft_len_c,
                                           fft_len_x, deadline,
                                           max_iters=800)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Perturb round {r}] new best {best_val:.6f}")
    return best_seq, best_val

"""
Multi‑stage optimiser for the step‑function problem.

Approach
--------
* Explore several lengths, starting with modest sizes and ending with a large one.
* For each length generate a rich set of initial guesses:
  - Dirichlet random, edge‑power‑laws, Gaussians, inverse‑roots,
    cosine windows (Hann, Hamming, Blackman), triangular,
    Beta‑distributed shapes and sparse random vectors.
* Optimise each guess with three phases:
  1. Adam on a smooth‑max surrogate of the convolution peak.
  2. Exact sub‑gradient (exponentiated‑gradient) on the true max.
  3. Cheap hill‑climbing local search.
* After the main passes run a short perturb‑and‑refine loop around the
  current global best.
* Finish with a long‑run Adam refinement on the best sequence found.

The algorithm respects a total time budget (default 1000 s) and returns a
plain Python list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse set of warm‑starts for length n (all sum to 1)."""
    cand = []

    # warm start from the current global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # warm start by interpolating the global best to this length
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # historic sequence if supplied
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 4):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # three Gaussian peaks (0.25,0.5,0.75)
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # ------------------------------------------------------------------
    # additional families to increase diversity
    i = np.arange(n, dtype=np.float64)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    cand.append(hann / hann.sum())
    # Hamming window
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    cand.append(hamming / hamming.sum())
    # Blackman window
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    cand.append(blackman / blackman.sum())
    # Triangular (tent) shape
    triangular = 1.0 - np.abs(2.0 * i / (n - 1) - 1.0)
    cand.append(triangular / triangular.sum())

    # Asymmetric Beta distributions (a,b > 0)
    for a, b in [(0.2, 5.0), (5.0, 0.2), (0.5, 0.5), (2.0, 2.0), (0.8, 1.5), (1.5, 0.8)]:
        x = (i + 0.5) / n            # avoid exact 0 / 1
        beta_pdf = np.power(x, a - 1) * np.power(1 - x, b - 1)
        beta_pdf = np.where(beta_pdf > 0, beta_pdf, 0.0)
        if beta_pdf.sum() > 0:
            cand.append(beta_pdf / beta_pdf.sum())

    # Random sparse vectors (few non‑zeros)
    for nz in (5, 10, 20):
        vec = np.zeros(n)
        idxs = np.random.choice(n, size=nz, replace=False)
        vec[idxs] = np.random.rand(nz)
        tot = vec.sum()
        if tot > 0:
            cand.append(vec / tot)

    np.random.shuffle(cand)
    max_cand = 20 if n <= 4096 else 15
    return cand[:max_cand]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) seen before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.3               # start with a very soft max
    lam_factor = 1.04

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_c)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex projection via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # raise temperature
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient) on the true max.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_c)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.65

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_c)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Simple cheap local search: repeatedly shift a small amount of mass between
    two random coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue
        eps = np.random.rand() * 0.07 * a[i1]   # move up to 7 % of a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-12:
            continue
        new_a /= new_a.sum()
        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 100 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a
    return best_a, best_val


def _perturb_and_refine(
    base_seq: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    rounds: int = 10,
) -> tuple[np.ndarray, float]:
    """Create noisy copies of a sequence and run a short sub‑gradient refinement."""
    best_seq = base_seq.copy()
    best_val = _objective(best_seq, n, fft_len_c)

    for r in range(rounds):
        if time.time() > deadline:
            break
        noise = np.random.randn(n) * 0.02
        cand = best_seq + noise
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
        else:
            cand = cand / s
        a_sub, val_sub = _subgrad_optimize(cand, n, fft_len_c,
                                           fft_len_x, deadline,
                                           max_iters=800)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Perturb round {r}] new best {best_val:.6f}")
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # lengths – start small, finish with the biggest feasible size
    lengths = [2048, 4096, 8192, 16384, 32768]

    # pre‑compute FFT sizes
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # historic solution, if any
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # MAIN optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # collect a modest but diverse set of start points
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # iteration budgets (scale roughly inversely with length)
        max_adam = int(20000 * 1024 / n) + 3000
        max_sub  = int(6000 * 1024 / n) + 2000
        max_hc   = int(8000 * 1024 / n) + 2000

        for cid, start_vec in enumerate(candidates):
            if time.time() > deadline:
                break

            # Phase 1 – smooth‑max Adam
            a_adam, v_adam = _adam_optimize(
                start_vec, n, fft_len_c, fft_len_x, deadline, max_iters=max_adam
            )

            # Phase 2 – exact sub‑gradient
            a_sub, v_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,
                init_val=v_adam, max_iters=max_sub
            )

            # Phase 3 – cheap hill‑climb
            a_hc, v_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=v_sub, max_iters=max_hc
            )

            if v_hc < best_val:
                best_val = v_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cid}) value={best_val:.6f}")

        # optional quick perturb‑search around the current best of this length
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline - 5.0:
            a_pert, v_pert = _perturb_and_refine(
                best_seq, n, fft_len_c, fft_len_x, deadline, rounds=12
            )
            if v_pert < best_val:
                best_val = v_pert
                best_seq = a_pert.copy()
                print(f"*** Global new best after perturb (n={n}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run refinement on the global best (any length)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n, 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(30000 * 1024 / n) + 5000
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    elapsed = time.time() - start_time
    print(f"\nFinished optimisation in {elapsed:.1f}s, best value={best_val:.6f}")
    return best_seq.tolist()
