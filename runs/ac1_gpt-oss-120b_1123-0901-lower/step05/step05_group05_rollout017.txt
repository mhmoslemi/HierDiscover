"""
Propose a non‑negative step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The score is exactly the maximal entry of the (full) convolution when the
sequence is normalised to the standard sum √(2 n).  The optimiser works in
four stages, all respecting a 1000 s wall‑clock budget.

1. **Rapid random sampling** – many cheap candidates are generated from a
   small library of parametric families (uniform, Dirichlet, exponential,
   power‑law, mixed).  Each candidate is normalised to the required sum and
   its peak convolution is evaluated.  The best sample seeds the later
   phases.

2. **Soft‑max Adam optimisation** – the best candidate is refined with a
   soft‑max parametrisation (logits → a = target·softmax(logits)).  A
   smooth log‑sum‑exp surrogate of the max is optimised with Adam; the
   temperature is driven down to a tiny value (1e‑5 · cur_max) so that the
   surrogate closely matches the true max.

3. **Projected sub‑gradient descent** – from the Adam result a classic
   sub‑gradient method is run.  The exact sub‑gradient of the active
   convolution entry is computed (a reversed slice of the sequence).  After
   each step the vector is projected onto the simplex {a≥0, Σa = √(2 n)}.

4. **Simulated‑annealing hill‑climb** – a final stochastic local search
   moves a random amount of mass between two indices (preserving the sum).
   Moves are accepted if they improve the peak, otherwise with a
   Metropolis probability that cools linearly to near zero.

The best sequence found is returned as a plain Python list; if nothing is
found a uniform construction is used as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear full convolution of a with itself using an FFT of the next power of two."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # Numerical noise can create slight negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex with L1‑radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple objective – maximal convolution entry (score = max_b because
#   we normalise the sum to sqrt(2*n) beforehand)
# ----------------------------------------------------------------------
def _objective_max_conv(a: np.ndarray) -> float:
    """Return max(conv(a, a)).  Assumes sum(a) ≈ sqrt(2*len(a))."""
    conv = _full_convolution_fft(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser on a smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 1e-5,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Return a refined sequence and its exact score (max conv)."""
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_score = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits → simplex using soft‑max -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # numeric stability
        p = exp_shift / np.sum(exp_shift)     # Σp = 1
        a = target_sum * p

        # ----- exact max conv (objective) -----
        cur_max = _objective_max_conv(a)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n}  new best max={cur_max:.6f}")

        # ----- smooth‑max surrogate (log‑sum‑exp) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        # if tau becomes extremely small the exponent can under‑flow;
        # clamp to avoid NaNs.
        w = np.exp((np.full_like(a, cur_max) - cur_max) / tau)   # placeholder
        # Re‑compute with correct values:
        conv = _full_convolution_fft(a)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a of the smooth max  -----
        w_rev = w[::-1]
        sz = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=sz)
        Fw = np.fft.rfft(w_rev, n=sz)
        conv2 = np.fft.irfft(Fa * Fw, n=sz)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                  # positions for Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits (chain rule through soft‑max) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    # final exact evaluation (identical to best_score but recomputed for safety)
    final_score = _objective_max_conv(best_a) if best_a is not None else np.inf
    return best_a, final_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑conv objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Perform projected sub‑gradient descent until time expires."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_max_conv(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a)
        k = int(conv.argmax())                     # index of current max

        # sub‑gradient of conv_k w.r.t. a (reversed slice)
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # Diminishing step size (∝ 1/√t)
        step = (target_sum * 0.5 / n) / np.sqrt(it)

        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        cand_score = _objective_max_conv(a_candidate)

        if cand_score < best_score - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_score = cand_score
            # modestly increase step after a success
            # (implemented implicitly by 1/√t schedule)
        else:
            # no improvement – keep current a but continue
            pass

        # occasional random jitter to keep the search alive
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(
                f"[{time.time():.2f}] sub‑grad it={it} best={best_score:.6f}"
            )
        # early exit if we already beat the target
        if best_score < 1.503:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Stochastic local search with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_max_conv(a)

    # temperature schedule – start hot, cool linearly to near‑zero
    T0 = best_score * 0.3
    T_end = 1e-5
    start = time.time()
    it = 0
    max_move_frac = 0.25  # up‑to 25 % of the donor mass

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        delta = a[i] * np.random.rand() * max_move_frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # no need to re‑project – the move preserves the sum exactly
        new_score = _objective_max_conv(a_new)

        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hill‑climb it={it} new best={best_score:.6f}")

        # periodic re‑normalisation (to correct possible tiny numerical drift)
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_score < 1.503:
            break

    return best_a, best_score

# Family of parametric shape generators for the fast random sampling phase
# ----------------------------------------------------------------------
def _generate_shape(n: int, mode: str) -> np.ndarray:
    idx = np.arange(n)
    if mode == "uniform":
        a = np.ones(n)
    elif mode == "dirichlet":
        # gamma variables with random concentration
        conc = np.random.uniform(0.05, 5.0)
        a = np.random.gamma(conc, 1.0, size=n)
    elif mode == "exp":
        tau = np.random.uniform(0.05 * n, 3.0 * n)
        a = np.exp(-idx / tau)
    elif mode == "power":
        p = np.random.uniform(0.0, 2.0)
        a = (idx + 1.0) ** (-p)
    elif mode == "mix":
        p = np.random.uniform(0.0, 1.5)
        tau = np.random.uniform(0.05 * n, 3.0 * n)
        a = (idx + 1.0) ** (-p) * np.exp(-idx / tau)
    else:  # random uniform positive values
        a = np.random.rand(n)
    return a.astype(float)

# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Hybrid optimiser returning a list of non‑negative heights."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if available)
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.asarray(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = _objective_max_conv(hs_norm)
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start n={n0} score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase A – fast random exploration (≈15 % of total budget)
    # --------------------------------------------------------------
    phase_a_end = time.time() + 0.15 * (deadline - time.time())
    shape_modes = ["uniform", "dirichlet", "exp", "power", "mix", "random"]
    while time.time() < phase_a_end:
        # pick a random length – more choices for earlier part of the run
        n = np.random.randint(800, 3500 + 1)
        target_sum = np.sqrt(2.0 * n)

        mode = np.random.choice(shape_modes)
        a = _generate_shape(n, mode)

        # optional random truncation (creates a leading block + trailing zeros)
        if np.random.rand() < 0.25:
            cut = np.random.randint(int(0.5 * n), n + 1)
            a = np.concatenate([a[:cut], np.zeros(n - cut)])

        # normalise to the required sum
        a = a / a.sum() * target_sum

        # evaluate – the objective is just the maximal convolution entry
        score = _objective_max_conv(a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] random phase new best n={n} score={best_score:.6f}")

        # early success?
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # Phase B – Adam soft‑max optimisation on the current best (≈30 %)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 150.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        # initialise logits from the current best
        logits = np.log(best_seq + 1e-12)

        adam_deadline = time.time() + min(120.0, time_left * 0.6)
        seq_adam, score_adam = _optimise_adam_softmax(
            n,
            logits,
            target_sum,
            adam_deadline,
            init_tau_factor=0.20,
            min_tau_factor=1e-5,
            alpha0=0.03,
        )
        if seq_adam is not None and score_adam < best_score:
            best_score = score_adam
            best_seq = seq_adam.copy()
            print(f"[{time.time():.2f}] after Adam new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected sub‑gradient descent (≈20 %)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(60.0, time_left * 0.5)
        seq_sub, score_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if score_sub < best_score:
            best_score = score_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑grad new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – simulated‑annealing hill‑climb (remaining time)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 30.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = time.time() + time_left
        seq_hill, score_hill = _hillclimb(best_seq.copy(), target_sum, hill_deadline)
        if score_hill < best_score:
            best_score = score_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _objective_max_conv(best_seq)
        print(f"[{time.time():.2f}] fallback uniform n={n} score={best_score:.6f}")

    # Final sanity check via the provided evaluator (guards against accidental clipping)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.2f}] final reported score={final_score:.6f}")

    return best_seq.tolist()
