from scipy.optimize import linprog
import numpy as np
import scipy.sparse as sp

def linprog_step(a, M, target_sum, t):
    n = len(a)
    # A_ub: each row k for k=0..2n-2
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # loop i from i_start to i_end
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)   # maximize sum(b)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return original a
        return a, M

    b0 = res.x
    # scale b to target_sum
    total = b0.sum()
    if total == 0:
        return a, M
    scale = target_sum / total
    b = b0 * scale
    # combine
    a_new = (1-t) * a + t * b
    # compute new max conv
    conv = _full_convolution_fft(a_new, a_new)
    M_new = conv.max()
    return a_new, M_new
```

def targeted_hillclimb(a, target_sum, deadline):
    n = len(a)
    best = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    # pre compute best_max_index also maybe
    while time.time() < deadline:
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        indices = np.arange(max(0, k - (n-1)), min(n-1, k)+1)
        # j = k - i
        # pick i from indices weighted by a[i] maybe
        i = np.random.choice(indices)
        j = k - i
        # choose destination idx not in positions that contribute to k
        # We'll pick random destination not equal to i and not in indices
        not_allowed = set(indices.tolist() + [i,j]) # actually i and j in indices
        # choose dest from remaining
        cand = np.random.randint(0, n)
        while cand in not_allowed:
            cand = np.random.randint(0, n)
        # move delta
        if a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.5 # up to 50% of mass at i
        a_new = a.copy()
        a_new[i] -= delta
        a_new[cand] += delta
        new_max = _full_convolution_fft(a_new, a_new).max()
        if new_max < best_val:
            a = a_new
            best_val = new_max
            best = a_new.copy()
        else:
            # accept with low probability maybe
            if np.random.rand() < 0.01:
                a = a_new
    return best, best_val
```

"""
Hybrid search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1. **Fast shape sweep** – systematic exploration of simple parametric
   families (uniform, power‑law, exponential, combined) for a variety of
   lengths.  The sequence is always scaled to the canonical sum
   √(2 n) so that the evaluation equals the maximal convolution value.

2. **Random descending restarts** – many short runs start from a random
   Dirichlet vector that is sorted in decreasing order (large entries at the
   beginning).  This ordering is known to minimise the central convolution
   (pairing large with small values).

3. **Targeted stochastic hill‑climb** – on every candidate we repeatedly
   move a small amount of mass from an index that participates in the
   current maximal convolution to a random index outside that interval.
   Moves are accepted if they lower the peak; otherwise a low‑temperature
   Metropolis jump is allowed.  The simplex projection guarantees the sum
   stays equal to √(2 n).

All three phases share a global best solution and respect the overall
time budget (≈ 1000 s).  The final sequence is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------
# Convolution utilities (FFT, full linear conv)
# ----------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using an FFT (full length)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # eliminate tiny negatives
    return conv


# ----------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {w ≥ 0, Σw = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------
# Targeted hill‑climbing – reduces the peak convolution directly
# ----------------------------------------------------
def _targeted_hillclimb(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Move mass from indices that belong to the current worst convolution
    shift to indices outside that shift.  Accept improvements always,
    otherwise accept with a low Metropolis probability.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                     # shift giving the max peak
        cur_max = conv.max()
        # indices that actually contribute to this peak
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idxs = np.arange(i_low, i_high + 1)

        # pick a source index among them (uniformly)
        i = np.random.choice(idxs)
        if a[i] <= 0.0:
            continue

        # choose a destination outside the contributing interval
        while True:
            j = np.random.randint(0, n)
            if j < i_low or j > i_high:
                break

        # move a small fraction of the mass from i to j
        delta = a[i] * np.random.rand() * 0.15   # ≤ 15 % of the source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _full_convolution_fft(a_new, a_new).max()

        # accept if we improve the peak
        if new_max < best_val - 1e-12:
            a = a_new
            best_a = a_new.copy()
            best_val = new_max
        else:
            # occasional Metropolis step (temperature cools to zero)
            elapsed = time.time() - start
            fraction = elapsed / (deadline - start)
            T = max(1e-4, cur_max * 0.2 * (1.0 - fraction))
            if np.random.rand() < np.exp(-(new_max - best_val) / (T + 1e-12)):
                a = a_new

        # occasional re‑projection to avoid numerical drift
        if it % 500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:          # early exit if we already beat the target
            break

    return best_a, best_val
