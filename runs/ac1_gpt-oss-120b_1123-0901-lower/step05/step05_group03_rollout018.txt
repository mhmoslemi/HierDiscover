def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Stage 1: small n optimization
    n_small = 1024
    seq_small, val_small = run_optimizations(n_small, deadline, warm=None)  # using various candidates

    # Stage 2: upscale if time left
    while time.time() + 5 < deadline:
        # pick next n bigger e.g., double
        n_big = min(16384, n_small*2)
        if n_big == n_small: break
        # upscale seq
        x_old = np.linspace(0, 1, n_small, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        seq_big = np.interp(x_new, x_old, seq_small)
        seq_big = np.maximum(seq_big, 0)
        seq_big /= seq_big.sum()
        # optimize at n_big
        seq_small, val_small = run_optimizations(n_big, deadline, warm=seq_big)
        n_small = n_big
    # final seq = seq_small
    best_seq = np.clip(seq_small, 0.0, 1000.0)
    return best_seq.tolist()

"""
Optimizing self‑convolution peak with multi‑scale Adam + sub‑gradient.
"""

import time, math, numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a, n):
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=200_000, lr=0.05, lam_start=0.5,
                   lam_factor=1.02, stagnate_sec=4.0, random_shake=1e-2):
    """Adam on the log‑sum‑exp surrogate. Returns best (seq, value) seen."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    best_seq = a0.copy()
    best_val = _objective(a0, n)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # optional print
            # print(f"[Adam n={n} it={it}] val={val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (through convolution)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        # random shake on stagnation
        if time.time() - last_improve > stagnate_sec:
            logits += (np.random.rand(n) - 0.5) * random_shake
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                          max_iters=20_000, eta=0.6, stagnate_sec=5.0):
    """Exponentiated‑gradient sub‑gradient on the exact max."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
        s = 1.0
    a /= s

    best_seq = a.copy()
    best_val = _objective(a, n)
    last_improve = time.time()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # print(f"[Subgrad n={n} it={it}] val={val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > stagnate_sec:
            break

    return best_seq, best_val

def _optimize_one_candidate(a0, n, deadline, global_best_seq, global_best_val):
    """Runs Adam + sub‑gradient on a single start vector."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Adam phase – adapt iteration budget to remaining time
    remaining = deadline - time.time()
    max_adam = int(150_000 if remaining > 30 else 50_000)
    seq_adam, val_adam = _adam_optimize(a0, n, conv_fft_len, corr_fft_len,
                                        deadline, max_iters=max_adam)

    if val_adam < global_best_val:
        global_best_val = val_adam
        global_best_seq = seq_adam.copy()

    # Sub‑gradient phase – takes the better of the two starts
    start = seq_adam if val_adam <= global_best_val else global_best_seq
    seq_sub, val_sub = _subgradient_optimize(start, n, conv_fft_len,
                                             corr_fft_len, deadline)

    if val_sub < global_best_val:
        global_best_val = val_sub
        global_best_seq = seq_sub.copy()

    return global_best_seq, global_best_val

"""
Multi‑scale optimisation of the self‑convolution peak.

The algorithm works in two phases:

  1️⃣  A quick search on a modest grid (n≈1024) explores many diverse
      initial shapes (warm‑start, edge‑biased Beta‑like, Gaussian spikes,
      sinusoid, uniform, random Dirichlet).  Each start is refined by
      Adam on a smooth‑max (log‑sum‑exp) surrogate and a final
      exponentiated‑gradient sub‑gradient pass on the exact max.

  2️⃣  The best distribution is linearly interpolated to a larger grid
      (doubling the length while time permits, up to 16384) and refined
      again with the same two‑stage optimiser.  The larger size gives
      the algorithm more degrees of freedom to flatten the convolution,
      pushing the evaluation value below the target 1.503.

All operations are carried out on the probability simplex
(`sum(a)=1`, `a_i≥0`) – the evaluation function is scale‑invariant, so
normalisation does not affect the score.  The implementation respects
the global time budget (default 1000 s) and returns the best sequence
found as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a∗a)/(sum(a)²) – sum(a)=1 ⇒ 2·n·max(conv)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.05,

                   lam_factor: float = 1.02,
                   stagnate_sec: float = 4.0,
                   shake_amp: float = 1e-2) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(conv). Returns the best (seq,
    value) observed during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_seq = a0.copy()
    best_val = _objective(a0, n)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and current value
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # print(f"[Adam n={n} it={it}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)

        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # after a handful of starts we keep most of the remaining budget
        # for refining the best found so far
        if idx >= 2 and (deadline - time.time()) < 30.0:
            break

    # --------------------------------------------------------------
    # Stage 2 – progressively enlarge the grid while time permits
    while time.time() + 5.0 < deadline:
        new_n = min(16384, n * 2)
        if new_n == n:
            break

        # interpolate the current champion onto the larger grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, new_n, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(new_n))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # run the optimiser again at the larger resolution
        best_seq, best_val = _optimize_one_candidate(a_big, new_n,
                                                     deadline,
                                                     best_seq, best_val)

        n = new_n   # continue upwards from the new size

    # --------------------------------------------------------------
    # Fallback – if nothing succeeded, fall back to a uniform vector.
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Clip to the evaluator’s accepted range and return a plain list.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
