Implementation of simplex projection:

def _project_simplex(v, S):
    """Project v onto the simplex {x >= 0, sum x = S}."""
    # We'll use the algorithm of Duchi et al. (2008).
    if S < 0:
        raise ValueError
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    # find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_refine(a, target_sum, deadline, max_iter=500000):
    n = len(a)
    Nfft = 1 << ((3 * n - 2).bit_length())
    step = target_sum * 0.5  # initial step size
    bestM = None
    best_seq = a.copy()
    # compute initial max
    conv = np.fft.irfft(np.fft.rfft(a, n=Nfft) * np.fft.rfft(a, n=Nfft), n=Nfft)[: 2 * n - 1]
    bestM = float(conv.max())
    # loop
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute conv, max, mask, weighted_rev to get grad
        conv = np.fft.irfft(np.fft.rfft(a, n=Nfft) * np.fft.rfft(a, n=Nfft), n=Nfft)[: 2 * n - 1]
        M = float(conv.max())
        mask = conv >= M - 1e-12
        w = mask.astype(float) / mask.sum()
        # weighted rev
        rev = _weighted_rev(a, w, Nfft)
        grad = 2.0 * rev
        # gradient descent step
        a_new_raw = a - step * grad
        a_new = _project_simplex(a_new_raw, target_sum)
        # evaluate new M
        conv_new = np.fft.irfft(np.fft.rfft(a_new, n=Nfft) * np.fft.rfft(a_new, n=Nfft), n=Nfft)[: 2 * n - 1]
        M_new = float(conv_new.max())
        # acceptance criterion
        if M_new < M - 1e-12:
            a = a_new
            step *= 1.05  # increase step slightly
            if M_new < bestM:
                bestM = M_new
                best_seq = a.copy()

        # optional random jitter occasionally
        if it % 1000 == 0:
            # add small jitter to avoid stagnation
            jitter = np.random.default_rng().normal(scale=step*0.1, size=n)
            a = _project_simplex(a + jitter, target_sum)
    return best_seq, bestM

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
* Load the best known construction (``height_sequence_1``) if present.
* Interpolate it to several lengths (≈2100, 1.5×, 2× … up to 8000).
* For each length run a fast *projected sub‑gradient descent* on the exact
  max‑convolution objective:
      – compute the convolution by FFT,
      – identify the lag(s) attaining the current maximum,
      – obtain the sub‑gradient  g = 2·rev where rev_i = Σ_k w_k·a_{k‑i}
        (w are uniform weights on the active lags, implemented by an FFT‑based
        weighted reverse convolution),
      – take a step a ← a – α·g and project back onto the simplex
        {a_i≥0, Σ a_i = √(2 n)}.
  The step size α is adapted (increase on improvement, halve on failure)
  and occasional Gaussian jitters are added to escape plateaus.
* Keep the best sequence seen over all lengths and restarts.
* If time remains, perform a final refinement on the global best.
* Return the sequence as a plain Python list.

This routine replaces the previous soft‑max/Adam phase by a direct
sub‑gradient optimisation that works with the exact max‑convolution,
therefore providing a tighter descent direction and often achieving a
lower bound within the same time budget.
"""

import time
import math
import numpy as np

# Helper functions – must be top‑level (no closures)

def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of vector ``v`` onto the simplex
    {x | x_i ≥ 0, Σ_i x_i = S}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    # Find the largest ρ satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = (np.sum(v) - S) / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical cleanup: enforce exact sum
    diff = w.sum() - S
    if abs(diff) > 1e-12:
        # distribute the small error uniformly over positive entries
        pos = w > 0
        if pos.any():
            w[pos] -= diff / pos.sum()
            w = np.maximum(w, 0.0)
    return w


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k‑i} (out‑of‑range indices ignored)
    using FFTs.  ``w`` must have length ``2*n‑1`` where ``n = len(a)``.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # length = 3 n‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _subgradient_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 200_000,
    rng: np.random.Generator = None,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on ``max(a*a)`` with simplex constraint.
    Returns the best sequence found and its exact max‑convolution value.
    """
    if rng is None:
        rng = np.random.default_rng()

    n = a0.shape[0]
    a = a0.copy()
    Nfft = 1 << ((3 * n - 2).bit_length())          # power‑of‑two for linear conv
    step = max(0.01 * target_sum, 1e-3)               # initial step size
    best_a = a.copy()
    # initial objective
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    best_M = float(conv.max())

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ---- compute current max‑convolution and its active lags ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # active set (all lags within a tiny tolerance)
        mask = conv >= M - 1e-12
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            # numerical safeguard – treat a single max lag
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ---- sub‑gradient of the max‑convolution ----
        rev = _weighted_rev(a, w, Nfft)          # shape (n,)
        grad = 2.0 * rev

        # ---- take a gradient step and project onto the simplex ----
        a_candidate_raw = a - step * grad
        a_candidate = _project_simplex(a_candidate_raw, target_sum)

        # ---- evaluate candidate ----
        A_fft_cand = np.fft.rfft(a_candidate, n=Nfft)
        conv_cand = np.fft.irfft(A_fft_cand * A_fft_cand, n=Nfft)[: 2 * n - 1]
        M_cand = float(conv_cand.max())

        # ---- accept / reject / adapt step size ----
        if M_cand < M - 1e-12:
            # improvement
            a = a_candidate
            step *= 1.05                       # be a bit more aggressive
            if M_cand < best_M:
                best_M = M_cand
                best_a = a_candidate.copy()
                print(f"[subgrad] iter {it:6d}  new best M = {best_M:.6f}")
        else:
            # no improvement – shrink step
            step *= 0.5
            # occasionally add a small random jitter to escape flat regions
            if it % 1000 == 0:
                jitter = rng.normal(scale=step * 0.2, size=n)
                a = _project_simplex(a + jitter, target_sum)

        # early stop if step becomes tiny
        if step < 1e-12:
            break

    return best_a, best_M


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """
    Produce a modest collection of diverse starting vectors.
    Used only for occasional random restarts.
    """
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init
