import time, numpy as np, math, sys
from scipy.optimize import linprog

def full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return conv

def build_conv_matrix(f):
    n = f.shape[0]
    m = 2*n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    return A

def gradient_descent_step(seq, max_iters, deadline):
    # similar to previous code but simpler
    # uses adam with smooth max
    n = seq.shape[0]
    # init logits from seq
    x = np.log(seq + 1e-15)
    # adam params
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    decay_every = 5000
    tau0, tau_f = 0.1, 1e-6
    best = seq.copy()
    best_val = 2.0 * n * full_convolution(seq).max()
    t = 0
    while t < max_iters and time.time() < deadline:
        # softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        conv = full_convolution(a)
        # smooth max
        tau = tau_f + (tau0 - tau_f) * (1 - t / max_iters)
        cmax = conv.max()
        shifted = (conv - cmax) / tau
        exp_shifted = np.exp(shifted - np.max(shifted)) # stable
        weights = exp_shifted / exp_shifted.sum()
        # gradient of smooth max
        # conv = a*a, gradient = 2* convolution of weights with a reversed?
        # compute cross correlation
        # cross-correlation of weights with a = correlate(weights, a)
        # use FFT cross-correlation
        n2 = len(weights)
        fft_len = 1 << int(np.ceil(np.log2(2*n-1 + n-1)))
        # use method from earlier:
        rev_a = a[::-1]
        fw = np.fft.rfft(weights, fft_len)
        fr = np.fft.rfft(rev_a, fft_len)

        grad_a = 2.0 * corr[n-1:n-1+n]  # length n
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)
        # adam update
        t += 1
        m = beta1 * m + (1-beta1) * grad_x

        if t % decay_every == 0:

        # evaluate
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:

            best = a.copy()
    return best, best_val

def lp_step(f, M):
    n = f.shape[0]
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    # constraints: A @ g <= M
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    sumg = g.sum()
    if sumg < 1e-15:
        # fallback uniform
        g = np.full(n, 1.0/n)
    else:
        g = g / sumg
    return g

"""
Propose a non‑negative step‑function (probability vector) that minimizes

               F(a) = 2·n·max(a*a) / (sum(a)^2)

The implementation follows two complementary ideas:

1. **Warm‑up with smooth‑max gradient descent** (Adam on a soft‑max
   parametrisation).  This quickly reaches a decent local optimum.

2. **Iterative LP‑mixing** – the classic “alternating linear programme”
   described in the statement.  For the current vector `f` we solve

        maximise   Σ g_i
        subject to A_f g ≤ M ,   g ≥ 0

   where `A_f` is the convolution matrix of `f` (so that
   `A_f @ g = f * g`) and `M = max(f*f)`.  The optimal `g` puts as much
   mass as possible on entries that cause a small convolution with `f`.
   Mixing `f ← (1‑t)·f + t·g` (with a decreasing step size `t`) provably
   never increases `M`.  Repeating this step drives the peak of the
   auto‑convolution down.

3. **Cheap local refinement** – random mass transfers between two
   coordinates are accepted only if they improve the exact objective.
   This cleans up small remaining irregularities.

The procedure runs for a total wall‑time ≤ `budget_s` seconds (default
1000 s) and returns the best sequence found.  It works for any `n`,
preferring lengths 256 → 512 → 1024 → 2048.  If a global variable
`height_sequence_1` is present it is used as a warm start.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of length len(a).
    Returns a vector c where
        c[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i = 0 … n‑1 .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """Smooth max (log‑sum‑exp) of `conv` and its soft‑max weights."""
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    # for numerical stability we shift again before exponentiation
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def gradient_descent_warm_start(a0: np.ndarray,
                                max_iters: int,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parametrisation.
    Returns the best vector (sum = 1) and its objective value.
    """
    n = a0.shape[0]
    # initialise logits from a0 (avoid log(0))
    x = np.log(a0 + 1e-15)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    decay_every = 5000

    # temperature schedule
    tau_initial, tau_final = 0.12, 5e-7

    best = a0.copy()
    best_val = 2.0 * n * full_convolution(a0).max()
    t = 0

    while t < max_iters and time.time() < deadline:
        # ----- soft‑max -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()                     # a ≥ 0, Σa = 1

        # ----- convolution & smooth max -----
        conv = full_convolution(a)
        tau = max(tau_final,
                  tau_initial * (1.0 - t / max_iters))
        smooth, weights = smooth_max_and_weights(conv, tau)

        # ----- exact objective (for bookkeeping) -----
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best = a.copy()
            print(f"[gd] new best {cur_val:.6f} (iter {t})")

        # ----- gradient of smooth max -----
        # grad_a = 2·(weights ★ a)
        fft_len = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1
        grad_a = 2.0 * cross_correlation(weights, a, fft_len)

        # back‑prop through soft‑max
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % decay_every == 0:
            lr *= 0.7

    return best, best_val


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that A @ g = f * g (linear convolution).
    Shape of A is (2·n‑1, n).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def lp_step(f: np.ndarray, M: float):
    """
    Solve the linear programme:
        maximise   Σ g_i
        subject to   A_f @ g ≤ M ,   g ≥ 0
    where A_f is the convolution matrix of `f`.
    Returns a normalised vector g (Σ g = 1) or None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # maximise Σ g_i  → minimise -Σ g_i

    # Use the HiGHS interior point solver – fast for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if res.status != 0 or not res.success:
        return None

    g = res.x
    s = g.sum()
    if s < 1e-12:               # degenerate – fall back to uniform
        g = np.full(n, 1.0 / n)
    else:
        g = g / s
    return g

        if best_seq is not None and best_seq.shape[0] == n:
            a0 = best_seq.copy()
            print("[init] warm‑start from previous best")
        elif external is not None:
            if external.shape[0] >= n:
                a0 = external[:n].copy()
            else:
                a0 = np.pad(external, (0, n - external.shape[0]), mode='constant')
            print("[init] start from height_sequence_1")
        else:
            a0 = np.random.dirichlet(np.ones(n))
            print("[init] random start")

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-8:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ---------- warm‑up gradient descent ----------
        gd_iters = 15000 if n <= 512 else 12000
        a, val = gradient_descent_warm_start(a0, gd_iters, deadline)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[gd] best value after warm‑up: {val:.6f}")

        # ---------- LP‑mixing loop ----------
        t_mix = 0.35               # initial mixing coefficient
        max_lp_iters = 45          # a few dozen LP steps are cheap
        for it in range(max_lp_iters):
            if time.time() >= deadline:
                break

            conv = full_convolution(a)
            M = conv.max()
            g = lp_step(a, M)
            if g is None:
                print("[lp] linprog failed – aborting LP stage")
                break

            # convex combination
            a = (1.0 - t_mix) * a + t_mix * g
            a = a / a.sum()

            cur_val = 2.0 * n * full_convolution(a).max()
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[lp] iter {it:02d} improved to {cur_val:.6f}")

            # slowly decay the mixing weight
            t_mix = max(0.02, t_mix * 0.92)

        # ---------- cheap random local refinement ----------
        local_iters = 6000
        for it in range(local_iters):
            if time.time() >= deadline:
                break
            i, j = np.random.choice(n, 2, replace=False)
            # move a small fraction from i to j
            move = min(a[i], 0.009) * (0.25 + 0.5 * np.random.rand())
            if move <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= move
            a_new[j] += move
            val_new = 2.0 * n * full_convolution(a_new).max()
            if val_new < best_val - 1e-12:
                best_val = val_new
                a = a_new
                best_seq = a_new.copy()
                print(f"[local] iter {it:04d} improved to {val_new:.6f}")

        # keep the best for the next (larger) length by up‑sampling
        if best_seq is not None and best_seq.shape[0] == n:
            # simple repeat‑each‑entry up‑sampling when the next n is a multiple
            # (this keeps the shape but adds more degrees of freedom)
            pass  # the next loop will either use the same vector or re‑initialise

    # ------------------------------------------------------------------------
    # Fallback in the improbable case that nothing was found.
    # ------------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * full_convolution(best_seq).max()
        print("[fallback] using uniform vector")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
