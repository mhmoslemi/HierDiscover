def local_refine(seq, deadline):
    a = np.array(seq, dtype=np.float64)
    n = len(a)
    best_val = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    last_improve = time.time()
    while time.time() < deadline:
        # propose move
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # fraction up to 20% of a[i]
        frac = np.random.rand() * 0.2
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        val = evaluate_sequence(a_new.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a_new.copy()
            a = a_new   # accept and continue from new point
            last_improve = time.time()
        else:
            # maybe accept with small prob (annealing)
            # simple "shake" if stagnated
            if time.time() - last_improve > 30.0:
                # random jitter
                i2 = np.random.randint(n)
                j2 = np.random.randint(n)
                if a[i2] > 0:
                    delta2 = a[i2] * (np.random.rand() * 0.4 + 0.1)
                    a[i2] -= delta2
                    a[j2] += delta2
                last_improve = time.time()
    return best_seq.tolist(), best_val

"""
Propose a step‑function (non‑negative vector) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The implementation consists of three phases:

1. **Fast parametric sweep** – generate families of simple analytic
   shapes (power law, exponential, symmetric “edge‑heavy” shapes) for a
   few modest lengths (256, 512, 1024, 2048).  Every candidate is
   evaluated with the supplied ``evaluate_sequence`` and the best
   candidate is remembered.

2. **Fine‑grained local search** – starting from the best candidate,
   perform a hill‑climbing / simulated‑annealing style random mass
   transfer.  A move transfers a small fraction of the mass from one
   index to another while preserving the total sum.  Improvements are
   kept, and occasional “shakes’’ avoid stagnation.

3. **Return the best sequence** – the final best vector (as a plain
   ``list``) is returned.  The whole routine respects the 1000 s global
   wall‑time budget, prints progress and never produces negative
   numbers.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper functions to generate families of candidate sequences
# ----------------------------------------------------------------------


def _normalise(a: np.ndarray) -> np.ndarray:
    """Scale a to have sum 1 (if the sum is zero fallback to uniform)."""
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


def power_sequence(n: int, p: float, reverse: bool = False) -> np.ndarray:
    """
    a_i ∝ (i+1)^p   (or reversed if ``reverse=True``).
    The sequence is normalised to sum 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, p)
    if reverse:
        a = a[::-1]
    return _normalise(a)


def exponential_sequence(n: int, alpha: float, reverse: bool = False) -> np.ndarray:
    """
    a_i ∝ exp(alpha * i).  If ``reverse`` is True the sign of alpha is flipped.
    """
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        alpha = -alpha
    a = np.exp(alpha * idx)
    return _normalise(a)


def edge_sequence(n: int, p: float) -> np.ndarray:
    """
    Edge‑heavy shape: a_i ∝ 1 / (dist_to_edge(i)+1)^p,
    where dist_to_edge(i)=min(i, n‑1‑i).
    """
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx) + 1.0  # distance to the nearest border, ≥1
    a = 1.0 / np.power(dist, p)
    return _normalise(a)


def uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution (baseline)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Local random‑mass‑transfer refinement
# ----------------------------------------------------------------------


def local_refine(start_seq: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Simple hill climbing: repeatedly move a random fraction of mass from one
    coordinate to another and keep the move if it improves the objective.
    Occasionally a random “shake’’ is performed when no improvement has been
    seen for a while.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_val = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    last_improve = time.time()

    # parameters controlling the magnitude of moves
    max_frac = 0.25          # at most this fraction of the source mass is moved
    min_frac = 0.02          # at least this fraction is moved (if source allows)
    shake_interval = 30.0    # seconds of stagnation before a shake

    while time.time() < deadline:
        # pick a source index with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to transfer
        frac = np.random.rand() * (max_frac - min_frac) + min_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = evaluate_sequence(a_new.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a_new.copy()
            a = a_new                      # continue from the improved point
            last_improve = time.time()
            # occasional logging
            print(f"[refine] improved to {best_val:.6f} (n={n})")
        else:
            # occasional shake when we are stuck
            if time.time() - last_improve > shake_interval:
                # random jitter on a few coordinates
                for _ in range(3):
                    ii = np.random.randint(n)
                    if a[ii] <= 1e-12:
                        continue
                    jj = np.random.randint(n)
                    if ii == jj:
                        continue
                    d = a[ii] * (np.random.rand() * 0.5 + 0.1)
                    a[ii] -= d
                    a[jj] += d
                last_improve = time.time()
    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Parameter sweep over simple analytic families
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]   # a few problem sizes to try

    edge_grid = np.linspace(0.5, 3.0, 18)          # p for edge‑heavy shape

    for n in lengths:

        # ---- start from previously known good sequence if available ----

        print(f"[refinement] improved to {best_val:.6f}")

    print(f"\nFinal best value: {best_val:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
