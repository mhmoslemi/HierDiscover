"""
Propose a non‑negative step function that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation uses a three‑phase hybrid:

1. **Exploratory Adam on a soft‑max parametrisation** – many short
   Adam runs with different lengths (n ≈ 1500…8000) and diverse
   initialisations (Dirichlet, power‑law, warm‑start from the current
   best).  The loss uses a smooth log‑sum‑exp approximation of the
   max‑convolution, which provides a usable gradient.

2. **Targeted stochastic hill‑climbing** – starting from the best Adam
   candidate we repeatedly move a small amount of mass from the index
   that contributes most to the current convolution peak to a low‑mass
   index.  The move is accepted if it reduces the peak (or with a
   tiny Metropolis probability).  This directly attacks the true
   objective and typically improves the Adam result.

3. **Final refinement** – a short projected sub‑gradient descent and a
   longer hill‑climb finish the optimisation.  All phases respect the
   global time‑budget (default 1000 s) and keep the global best answer.

The code is fully self‑contained, uses only NumPy and the provided
`evaluate_sequence` function, and returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # clip negative rounding noise
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {w ≥ 0, Σ w = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (logits) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate of max(conv).
    Returns the best coefficient vector encountered (already scaled to
    target_sum) and its *exact* evaluation (via evaluate_sequence).
    """
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_max = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits -> probabilities -> non‑negative vector with Σ = target_sum
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # stable soft‑max
        p = exp_shift / np.sum(exp_shift)      # Σ p = 1
        a = target_sum * p

        # current exact max of the convolution
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} new Adam best max={cur_max:.6f}")

        # smooth‑max weights (log‑sum‑exp surrogate)
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a  (2 · a ⊛ w_rev)
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # positions for Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    # final exact evaluation via the provided evaluator (identical to max)
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.1 * target_sum / n   # initial step (≈1 % of the mean)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()
        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate.copy()
            best_val = val_candidate
            step *= 1.02
        else:
            step *= 0.70

        if it % 300 == 0:
            # tiny random jitter to keep the dynamics alive
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] subgrad it={it} step={step:.3e} best={best_val:.6f}")

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _targeted_hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of mass from the index that contributes most
    to the current convolution peak to a low‑mass index.  Accept if
    the peak decreases (or with a vanishing Metropolis probability).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        k = int(conv.argmax())

        # indices participating in the maximal convolution entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        # donor – where the contribution is highest
        donor_local = np.argmax(contrib)
        donor = i_range[donor_local]
        partner = j_range[donor_local]

        # recipient – a low‑mass index (median‑based filter)
        low_mask = a < np.median(a)
        low_candidates = np.where(low_mask)[0]
        # exclude donor and its partner
        low_candidates = low_candidates[(low_candidates != donor) & (low_candidates != partner)]
        if low_candidates.size == 0:
            recipient = np.random.randint(0, n)
            if recipient == donor:
                continue
        else:
            recipient = np.random.choice(low_candidates)

        if a[donor] <= 0:
            continue

        # move at most 5 % of the donor’s mass
        delta = a[donor] * np.random.rand() * 0.05
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta

        # projection onto simplex guards against accumulated numerical drift
        a_new = _project_onto_simplex(a_new, target_sum)

        new_max = _full_convolution_fft(a_new, a_new).max()

        # deterministic improvement step
        if new_max < best_val:
            a = a_new
            best_a = a_new.copy()
            best_val = new_max
            if it % 100 == 0:
                print(f"[{time.time():.2f}] targeted hillclimb it={it} new best={best_val:.6f}")
        else:
            # occasional Metropolis acceptance (very low temperature)
            elapsed = time.time() - start
            frac = elapsed / (deadline - start + 1e-12)
            T = max(1e-6, cur_max * 0.3 * (1.0 - frac))
            if np.random.rand() < np.exp(-(new_max - best_val) / (T + 1e-12)):
                a = a_new

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            a = _project_onto_simplex(a + np.random.rand(n) * 1e-5, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimiser that mixes soft‑max Adam, a targeted hill‑climber
    and a short sub‑gradient refinement.  Returns the best sequence
    (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many short Adam runs with diverse lengths / initialisations
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 20.0:
            break

        # allocate a modest slice (≈ 8‑12 s) for a single Adam optimisation
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = time.time() + slice_time

        # pick a problem size – gradually bias towards larger n while time remains
        if remaining > 600:
            n = np.random.randint(3000, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 3500 + 1)

        target_sum = np.sqrt(2.0 * n)

        # ---------- initialise logits ----------
        if np.random.rand() < 0.6:
            a0 = np.random.rand(n)
        else:
            # power‑law / exponential blend
            p_exp = np.random.uniform(0.2, 0.8)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p_exp)
            if np.random.rand() < 0.5:
                tau = np.random.uniform(0.5, 2.0) * n
                a0 *= np.exp(-idx / tau)

        a0 = a0 / a0.sum() * target_sum
        logits0 = np.log(a0 + 1e-12)

        # sometimes inject the current global best (trim / pad)
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                a0 = np.concatenate([best_seq, pad])
            a0 = a0 / a0.sum() * target_sum
            logits0 = np.log(a0 + 1e-12)

        # ---------- run Adam ----------
        seq_adam, val_adam = _optimise_adam_softmax(
            n=n,
            logits=logits0,
            target_sum=target_sum,
            deadline=run_deadline,
            init_tau_factor=0.20,
            min_tau_factor=0.001,
            alpha0=0.03,
        )
        if val_adam < best_score:
            best_score = val_adam
            best_seq = seq_adam.copy()
            print(f"[{time.time():.2f}] GLOBAL new best (Adam) = {best_score:.6f} (n={n})")
            if best_score < 1.503:
                break

        # ---------- brief targeted hill‑climb on the Adam result ----------
        if time.time() < deadline:
            hc_deadline = time.time() + min(6.0, remaining * 0.12)
            seq_hc, val_hc = _targeted_hillclimb(seq_adam.copy(), target_sum, hc_deadline)
            if val_hc < best_score:
                best_score = val_hc
                best_seq = seq_hc.copy()
                print(f"[{time.time():.2f}] GLOBAL new best (Hill) = {best_score:.6f} (n={n})")
                if best_score < 1.503:
                    break

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max) on the best so far
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 12.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(30.0, time_left * 0.5)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final long hill‑climb for fine‑tuning
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        seq_final, val_final = _targeted_hillclimb(best_seq.copy(), target_sum, time.time() + time_left)
        if val_final < best_score:
            best_score = val_final
            best_seq = seq_final.copy()
            print(f"[{time.time():.2f}] after final hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (unlikely to be used)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
