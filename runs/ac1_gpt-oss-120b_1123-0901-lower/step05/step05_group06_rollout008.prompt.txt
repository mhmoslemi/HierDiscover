def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max).

    repeatedly move a small fraction of mass from a random index i to j.

    simulated‑annealing probability.  Returns an improved vector (or the
    original one if no improvement is found before the deadline).

            step += 1
            continue

            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")

            if rng.random() < prob:
