"""
Hybrid optimisation for the “step‑function” challenge.

The objective is
    2·n·max(conv(a,a)) / (sum(a)^2)      (scale‑invariant)

Strategy
---------

0. Warm‑start from the global variable ``height_sequence_1`` if it exists.

0′. Very cheap parametric sweep (power‑law, exponential, linear, triangular,
    a few random Dirichlets) for several lengths.  The best of these gives a
    solid deterministic seed.

1. Adam on a soft‑max parametrisation (logits → simplex).  The smooth‑max
   surrogate (log‑sum‑exp) provides a gradient; we keep the exact best
   objective seen during the run.

2. Differential Evolution (small population, mutation+binomial crossover,
   projection onto the simplex).  This explores the neighbourhood of the
   Adam solution more globally and is cheap because each fitness call uses a
   fast FFT‑based convolution.

3. Final stochastic hill‑climb (tiny mass transfers) with a simple
   temperature schedule for fine‑tuning.

All stages respect the global time budget (default 1000 s) and the best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)            # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (no security checks, used internally)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is never zero for our candidates
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed‑max surrogate."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- map logits → simplex (soft‑max) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)    # stability
        p = exp_shift / np.sum(exp_shift)   # Σp = 1
        a = target_sum * p

        # ---- exact max of convolution (for tracking) ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new local best max={cur_max:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp surrogate) ----
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a  (2·a⊛w_rev) ----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ---- back‑prop to logits (chain rule) ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    # final exact evaluation via the provided evaluator (identical to max)
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (exact objective)
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Local search that moves a tiny amount of mass between two positions."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule – start hot, cool to near zero
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move at most 10 % of the mass at i
        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # occasional re‑normalisation to avoid drift
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that minimises the
    provided evaluation function.  The routine combines a deterministic
    parametric sweep, Adam optimisation, differential evolution and a final
    hill‑climb, always respecting the global time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    best_seq = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Warm‑start from the globally provided construction, if present.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        sc = _eval_fast(hs_norm)
        best_seq = hs_norm.copy()
        best_score = sc
        best_n = n0
        print(f"[{time.time():.2f}] warm‑start score={best_score:.6f} (n={best_n})")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric families (power‑law, exponential,…)
    # ------------------------------------------------------------------
    param_n_vals = [1500, 2000, 2500, 3000]
    power_ps = [0.30, 0.55, 0.80, 1.05, 1.30]
    exps = [0.001, 0.002, 0.005, 0.010, 0.020]

    def _update_best(arr):
        nonlocal best_seq, best_score, best_n
        sc = _eval_fast(arr)
        if sc < best_score:
            best_seq = arr.copy()
            best_score = sc
            best_n = len(arr)
            print(f"[{time.time():.2f}] parametric new best {best_score:.6f} (n={best_n})")

    for n in param_n_vals:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)
        idx = np.arange(1, n + 1, dtype=float)

        # power‑law families
        for p in power_ps:
            a = idx ** (-p)
            a = a * (target / a.sum())
            _update_best(a)

        # exponential families
        for aexp in exps:
            a = np.exp(-aexp * (idx - 1))
            a = a * (target / a.sum())
            _update_best(a)

        # linear increasing
        a = idx
        a = a * (target / a.sum())
        _update_best(a)

        # symmetric triangular (mountain)
        a = np.minimum(idx, n + 1 - idx)
        a = a * (target / a.sum())
        _update_best(a)

        # few random Dirichlet seeds
        for _ in range(2):
            a = np.random.dirichlet(np.ones(n)) * target
            _update_best(a)

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (soft‑max) with several restarts.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        # allocate at most 25 % of the remaining budget to Adam
        adam_budget = min(remaining * 0.25, 150.0)
        adam_deadline = time.time() + adam_budget
        while time.time() < adam_deadline:
            # choose a length – often stay close to the current best
            if best_seq is not None and np.random.rand() < 0.6:
                n = best_n
                init_seq = best_seq.copy()
            else:
                n = np.random.randint(1200, 3500)
                target = np.sqrt(2.0 * n)
                init_seq = np.random.dirichlet(np.ones(n)) * target

            # initialise logits for the soft‑max parametrisation
            x0 = np.log(init_seq + 1e-12)

            seq_local, _ = _optimise_adam_softmax(
                n,
                x0,
                np.sqrt(2.0 * n),
                adam_deadline,
                init_tau_factor=0.25,
                min_tau_factor=0.001,
                alpha0=0.02,
            )
            if seq_local is not None:
                _update_best(seq_local)

            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Phase 2 – Differential Evolution (population based global search)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0 and best_seq is not None:
        n = best_n
        target = np.sqrt(2.0 * n)
        popsize = 12
        # initialise population: keep the current best and fill with random Dirichlet vectors
        population = [best_seq.copy()]
        for _ in range(popsize - 1):
            pop = np.random.dirichlet(np.ones(n)) * target
            population.append(pop)
        population = np.stack(population)                     # shape (popsize, n)

        scores = np.array([_eval_fast(ind) for ind in population])
        best_idx = int(np.argmin(scores))
        best_seq = population[best_idx].copy()
        best_score = scores[best_idx]
        print(f"[{time.time():.2f}] DE init best {best_score:.6f}")

        F = 0.8          # differential weight
        CR = 0.9         # crossover probability
        gen = 0
        while time.time() < deadline:
            gen += 1
            for i in range(popsize):
                # mutation: pick three distinct individuals ≠ i
                idxs = list(range(popsize))
                idxs.remove(i)
                r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
                mutant = population[r1] + F * (population[r2] - population[r3])
                mutant = np.maximum(mutant, 0.0)
                mutant = _project_onto_simplex(mutant, target)

                # binomial crossover
                cross_mask = np.random.rand(n) < CR
                if not np.any(cross_mask):
                    cross_mask[np.random.randint(0, n)] = True
                trial = np.where(cross_mask, mutant, population[i])

                trial_score = _eval_fast(trial)
                if trial_score < scores[i]:
                    population[i] = trial
                    scores[i] = trial_score
                    if trial_score < best_score:
                        best_seq = trial.copy()
                        best_score = trial_score
                        print(f"[{time.time():.2f}] DE gen {gen} new best {best_score:.6f}")

                if best_score < 1.503:
                    break
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing fine tune
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        seq_hill, val_hill = _hillclimb(best_seq.copy(),
                                        np.sqrt(2.0 * best_n),
                                        time.time() + remaining)
        if val_hill < best_score:
            best_seq = seq_hill.copy()
            best_score = val_hill
            print(f"[{time.time():.2f}] final hill‑climb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _eval_fast(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    print(f"[{time.time():.2f}] returning best sequence (n={len(best_seq)}) score={best_score:.6f}")
    return best_seq.tolist()
