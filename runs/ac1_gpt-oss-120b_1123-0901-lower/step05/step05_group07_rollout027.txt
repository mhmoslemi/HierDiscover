                        break
                    max_x = np.max(x)
                    exp_x = np.exp(x - max_x)
                    a = exp_x / exp_x.sum()
                    conv = full_convolution(a, fft_len=fft_len_conv)
                    cur_val = 2.0 * n * conv.max()
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_seq = a.copy()
                        best_len = n
                        print(f"[fine] improved to {cur_val:.6f}")

                    # exact max → hard indicator vector
                    hard = np.zeros_like(conv)
                    hard[conv.argmax()] = 1.0
                    grad_a = 2.0 * cross_correlation(hard, a, fft_len=fft_len_corr)
                    inner = np.dot(a, grad_a)
                    grad_x = a * (grad_a - inner)

                    # Adam step
                    m = beta1 * m + (1.0 - beta1) * grad_x
                    v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                    m_hat = m / (1.0 - beta1 ** (ft + 1))
                    v_hat = v / (1.0 - beta2 ** (ft + 1))
                    x = x - lr_fine * m_hat / (np.sqrt(v_hat) + eps)

            # ------------------- cheap greedy refinement -------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 5000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small fraction from i to j
                    eps_move = min(a_local[i], 0.015) * (np.random.rand() * 0.6 + 0.2)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")

    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # return plain Python list as required by the evaluation harness
    return best_seq.tolist()
