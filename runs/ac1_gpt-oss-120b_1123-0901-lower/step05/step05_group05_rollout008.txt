    shape_type = np.random.choice(['exp','power','logistic','double_exp','uniform','dirichlet'])
    if shape_type == 'exp':
        alpha = np.random.uniform(0.5, 3.0) / n  # scaling factor
        a = np.exp(-alpha * np.arange(n))
    elif shape_type == 'power':
        p = np.random.uniform(0.2, 2.5)
        a = (np.arange(1, n+1) ** (-p))
    elif shape_type == 'logistic':
        beta = np.random.uniform(0.005, 0.1)
        a = 1.0 / (1.0 + np.exp(beta * (np.arange(n) - (n-1)/2)))

        alpha = np.random.uniform(0.5, 3.0) / n
        a = np.exp(-alpha * np.arange(n)) + np.exp(-alpha * (np.arange(n)[::-1]))

"""
Hybrid search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation equals max(conv(a, a)) when the sequence is normalised
to Σ a = √(2 n).  The algorithm consists of four stages that respect a
1000 s wall‑clock budget:

1. **Exploratory shape sweep** – generate many deterministic
   families (exponential, power‑law, logistic, double‑exponential,
   random Dirichlet) for a random length n, normalise and keep the best.

2. **Soft‑max Adam refinement** – initialise the Adam optimiser
   (log‑soft‑max parametrisation) from the current global best.
   The surrogate uses a temperature‑scaled log‑sum‑exp of the
   convolution, allowing smooth optimisation of the non‑smooth max.

3. **Exact projected sub‑gradient descent** – starting from the best
   Adam result, perform sub‑gradient steps on the true max‑convolution
   with Euclidean projection onto the simplex (Σ a = √(2 n), a ≥ 0).

4. **Stochastic hill‑climbing** – a final Metropolis‑type local search
   that transfers tiny mass between random indices; a linear cooling
   schedule gradually freezes the system.

All stages share a global best solution and respect the time budget.
The code is deliberately self‑contained (no closures) and uses only
NumPy and the provided ``evaluate_sequence`` function.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 size."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex { w ≥ 0, Σ w = s }
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Normalise a vector to the required ℓ₁‑norm √(2 n)
# ----------------------------------------------------------------------
def _normalise_to_target(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    tgt = np.sqrt(2.0 * n)
    s = a.sum()
    if s == 0:
        # avoid division by zero – fall back to uniform
        return np.full_like(a, tgt / n)
    return a * (tgt / s)


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser on the log‑it parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (already normalised) and its
    exact evaluation as reported by ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits to the simplex (a = target_sum * softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stable soft‑max
        p = exp_shift / np.sum(exp_shift)       # probabilities, Σp = 1
        a = target_sum * p

        # ----- exact max convolution (for logging & temperature) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best max={cur_max:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp surrogate) -----
        tau = max(init_tau_factor * cur_max * (0.98 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a  (2 · a ⊛ w_rev) -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # positions for Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits (chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 300 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of the base learning rate
        alpha *= 0.9996

    # final exact evaluation of the best encountered vector
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(conv(a, a)) with Euclidean projection onto the simplex.
    Returns the best vector encountered and its exact evaluation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n      # initial step ~ 1 % of the average entry

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index attaining the current max

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step + projection
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02               # gently increase step on success
        else:
            step_size *= 0.70               # shrink aggressively on failure

        # occasional jitter to keep the search alive
        if it % 250 == 0 and np.random.rand() < 0.12:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 400 == 0:
            print(f"[{time.time():.2f}] sub‑grad it={it} step={step_size:.3e} best={best_val:.6f}")

        if best_val < 1.45:                 # theoretical curiosity – stop if we get very good
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (Metropolis) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Random local search that moves a tiny amount of mass between two indices.
    Accepts improving moves and, with a temperature schedule, occasional
    worsening moves (Metropolis criterion).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    T0 = best_val * 0.35
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move a fraction of the mass at i towards j
        delta = a[i] * np.random.rand() * 0.08   # up to 8 %
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # re‑project every now and then to counteract accumulated drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.45:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic shape families – used in the exploratory Stage A
# ----------------------------------------------------------------------
def _make_exponential(n: int) -> np.ndarray:
    """Monotone exponential decay a[i] = exp(-α·i)."""
    alpha = np.random.uniform(0.5, 3.0) / n   # make α dimensionless
    a = np.exp(-alpha * np.arange(n))
    return a


def _make_power_law(n: int) -> np.ndarray:
    """Power law a[i] = (i+1)^(-p)."""
    p = np.random.uniform(0.2, 2.5)
    a = (np.arange(1, n + 1, dtype=float)) ** (-p)
    return a


def _make_logistic(n: int) -> np.ndarray:
    """Logistic decay a[i] = 1 / (1 + exp(β·(i‑mid)))."""
    beta = np.random.uniform(0.004, 0.12)          # slope
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(beta * (np.arange(n) - mid)))
    return a


def _make_double_exponential(n: int) -> np.ndarray:
    """Symmetric sum of two exponentials glued at the centre."""
    alpha = np.random.uniform(0.5, 3.0) / n
    idx = np.arange(n)
    a = np.exp(-alpha * idx) + np.exp(-alpha * (n - 1 - idx))
    return a


def _make_random_dirichlet(n: int) -> np.ndarray:
    """Flat Dirichlet (uniform‑like) random start."""
    a = np.random.rand(n)
    return a


def _random_shape(n: int) -> np.ndarray:
    """Pick a random family and return a normalised shape."""
    families = [_make_exponential,
                _make_power_law,
                _make_logistic,
                _make_double_exponential,
                _make_random_dirichlet]
    chosen = np.random.choice(families)
    a = chosen(n).astype(float)

    # tiny additive noise to avoid exact ties
    a += np.random.rand(n) * 1e-6
    a = np.maximum(a, 0.0)
    a = _normalise_to_target(a)
    return a


# ----------------------------------------------------------------------
# Main entry point – the function called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimisation that first sweeps deterministic shapes,
    then refines the best using Adam, followed by exact sub‑gradient
    descent and a final hill‑climbing stage.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin to avoid overrun
    deadline = start_time + budget_s - 0.12

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Warm‑start from a globally provided construction, if present
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start length={n0}  score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage A – fast sweep over a collection of analytic shapes
    # ------------------------------------------------------------------
    stage_a_end = start_time + (deadline - start_time) * 0.25
    while time.time() < stage_a_end:
        # pick a random length – larger lengths increase resolution but cost more time
        n = np.random.randint(1200, 3400 + 1)
        a = _random_shape(n)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Stage A new best={best_score:.6f} (n={n})")
        # early exit if we have already sunk below the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage B – Adam optimiser on the soft‑max surrogate (several restarts)
    # ------------------------------------------------------------------
    stage_b_end = start_time + (deadline - start_time) * 0.70
    while time.time() < stage_b_end:
        remaining = stage_b_end - time.time()
        # allocate a few seconds for each Adam run (shorter as time runs out)
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = time.time() + slice_time

        # Deciding the problem size – favour a medium range but allow variability
        if remaining > 300:
            n = np.random.randint(1500, 3200 + 1)
        else:
            n = np.random.randint( 900, 2500 + 1)

        target_sum = np.sqrt(2.0 * n)

        # initialise either from the global best (cropped/padded) or from scratch
        if best_seq is not None and np.random.rand() < 0.4:
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                a0 = np.concatenate([best_seq, pad])
        else:
            a0 = _random_shape(n)

        # convert to logits for the soft‑max parametrisation
        x0 = np.log(a0 + 1e-12)

        # run Adam
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.30,
            min_tau_factor=0.002,
            alpha0=0.025,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] Stage B GLOBAL new best={best_score:.6f} (n={n})")
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Stage C – projected sub‑gradient descent (exact max objective)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 8.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_end = time.time() + min(30.0, time_left * 0.4)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_end)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage D – final stochastic hill‑climbing for fine tuning
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_end = time.time() + time_left
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_end)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be required)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the outer harness
    return best_seq.tolist()
