def optimize_candidate(a0, n, conv_fft_len, corr_fft_len):
    # a0 = initial vector (numpy) already sum to 1.
    # Use logits = log(a0) (adding epsilon)
    eps = 1e-16
    a = np.maximum(a0, eps)
    a /= a.sum()
    logits = np.log(a)

    # Adam hyperparameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.01
    last_improve = time.time()

    # iteration budget based on n: (maybe 2000 + 20000 * 1024 / n)
    max_iters = int(20000 * 1024 / n) + 5000

    best_local_val = float('inf')
    best_local_seq = None

    for it in range(1, max_iters+1):
        if time.time() > deadline: break

        # softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # smooth-max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # grad_a = 2 * cross_corr(w, a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal lam
        lam = min(lam * lam_factor, 1e7)

        val = 2.0 * n * max_c

        if val < best_local_val:
            best_local_val = val
            best_local_seq = a.copy()
            last_improve = time.time()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} Adam it={it}] new global best {val:.6f}")

        # stagnation
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    # subgradient refinement
    if best_local_seq is not None:
        a = best_local_seq.copy()
    else:
        a = np.maximum(a0, eps)
        a /= a.sum()
    a = a / a.sum()
    eta = 0.6
    sub_last = time.time()
    for sub_it in range(2000):
        if time.time() > deadline: break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * np.max(conv)
        if val < best_local_val:
            best_local_val = val
            best_local_seq = a.copy()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} Subgrad it={sub_it}] new global best {val:.6f}")

        if time.time() - sub_last > 5.0:
            break
        sub_last = time.time()

"""
Propose a step‑function that (empirically) improves the upper bound on
evaluate_sequence().  The optimiser works on several lengths, builds a rich
pool of start vectors (uniform, up‑sampled previous best, edge‑concentrated,
exponential, reciprocal, multi‑spike, random Dirichlet …) and refines each
candidate with two phases:

1.  Adam on a smooth‑max surrogate of max(conv(a,a)).  The surrogate is the
    log‑sum‑exp with an annealed temperature λ.
2.  An exact sub‑gradient step on the true max (exponentiated‑gradient
    update that keeps a on the probability simplex).

After the length sweep a cheap hill‑climbing phase moves a little mass
between random coordinates to escape any remaining plateaus.  All work is
bounded by the wall‑clock budget (default 1000 s).  The best vector found
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – placed at module level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine respects the given wall‑clock budget and returns the best
    sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the outer runner

    best_seq = None
    best_val = float("inf")
    target_val = 1.503  # early stopping threshold

    # Lengths – we start small (cheap) and finish with a large problem.
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Build an extensive, diverse pool of initial candidates
        # ------------------------------------------------------------------
        candidates = []

        # 0. uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 1. warm‑start from a previous solution of the same size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # 2. up‑sample a shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum() if up.sum() > 0 else np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 3. historic global variable, if it exists
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            h = h / h.sum() if h.sum() > 0 else np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # 4. random Dirichlet draws
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # 5. power‑law edge‑concentrated families (already used in the reference)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 6. two‑Gaussian edge shapes (reference)
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 7. smooth sine‑squared bump (reference)
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # ------------------------------------------------------------------
        # New families that were missing in the reference
        # ------------------------------------------------------------------
        # 8. exponential edge‑decay
        for alpha in [0.001, 0.003, 0.008, 0.02, 0.05]:
            shape = np.exp(-alpha * idx) + np.exp(-alpha * (n - idx + 1.0))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 9. reciprocal (harmonic) edge‑shape
        shape = 1.0 / idx + 1.0 / (n - idx + 1.0)
        shape = shape / shape.sum()
        candidates.append(shape)

        # 10. multi‑spike random shapes
        for _ in range(4):
            k = np.random.randint(2, min(8, n // 2))
            spikes = np.random.choice(n, size=k, replace=False)
            weights = np.random.rand(k)
            shape = np.zeros(n, dtype=np.float64)
            shape[spikes] = weights
            shape = shape / shape.sum()
            candidates.append(shape)

        # ------------------------------------------------------------------
        # Optimise every candidate
        # ------------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # initialise on simplex (logits for soft‑max)
            eps = 1e-16
            a = np.maximum(a0, eps)
            a = a / a.sum()
            logits = np.log(a)

            # -------------------- Adam on smooth‑max --------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            lam = 0.5                     # temperature
            lam_factor = 1.01
            last_improve = time.time()

            # iteration budget grows when n is small, shrinks for large n
            max_iters = int(20000 * 1024 / n) + 5000

            for it in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # soft‑max → probability simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution and smooth‑max weighting
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (sum of a is 1 → denominator = 1)
                val = 2.0 * n * max_c

                # store global best
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # staggering handling – small random shake if stuck
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                # early exit if we already beat the target
                if best_val < target_val:
                    print("Target achieved – exiting early.")
                    return best_seq.tolist()

            # -------------------- Sub‑gradient refinement --------------------
            # start from the best version of this candidate (the last a)
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(2000):
                if time.time() > deadline:
                    break

                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()

                # active set of entries that achieve the max (sub‑gradient)
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # exponentiated‑gradient update (keeps a ≥ 0)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # re‑initialise if we collapsed
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * np.max(conv)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it:04d}] new best {val:.6f}")

                # stagnation – give up after 5 s without improvement
                if time.time() - sub_last > 5.0:
                    break
                sub_last = time.time()

                if best_val < target_val:
                    print("Target achieved – exiting early.")
                    return best_seq.tolist()

    # ----------------------------------------------------------------------
    # Final cheap hill‑climbing on the current global best (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n=== Local hill‑climbing on the global best ===")
        a = np.array(best_seq, copy=True)
        n = a.shape[0]
        # number of moves roughly proportional to remaining time
        max_moves = int(5e4 * (deadline - time.time()))
        for mv in range(max_moves):
            if time.time() > deadline:
                break
            # propose moving a small fraction of mass from i to j
            i, j = np.random.randint(0, n, size=2)
            if a[i] <= 1e-12:
                continue
            delta = a[i] * (np.random.rand() * 0.2)   # up to 20 % of a[i]
            a[i] -= delta
            a[j] += delta
            # evaluate with the true (non‑smooth) objective
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Hill‑climb move {mv}] new best {val:.6f}")
            else:
                # revert the move
                a[i] += delta
                a[j] -= delta
            if best_val < target_val:
                print("Target achieved during hill‑climbing – exiting.")
                break

    # ----------------------------------------------------------------------
    # Fallback – if nothing was discovered (unlikely) return a uniform vector
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # expose the best sequence for the next run (optional)
    globals()["height_sequence_1"] = best_seq.tolist()

    # return plain Python list as required by the external evaluator
    return best_seq.tolist()
