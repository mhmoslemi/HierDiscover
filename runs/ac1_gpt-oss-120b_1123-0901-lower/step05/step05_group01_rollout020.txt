def construct_sequence(params, n):
    K = len(params)//3
    a = np.zeros(n)
    xs = np.linspace(0, 1, n)
    for i in range(K):
        w = max(params[3*i], 0.0)
        p = np.clip(params[3*i+1], 0.0, 1.0)
        sigma = np.clip(params[3*i+2], 1e-4, 0.5)  # ensure not zero
        # gaussian bump
        a += w * np.exp(-0.5 * ((xs - p) / sigma) ** 2)
    # ensure non-negative
    a = np.maximum(a, 0.0)
    # Normalize sum to 1
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s
    return a
```

Propose a low‑value step‑function sequence.

The search works in three stages:

1.  For each considered length n a cheap global search is performed with

3.  If time permits a final long Adam run refines the global best.

All operations respect the overall 1000 s budget and return a plain

"""

import math
import numpy as np

# --------------------------------------------------------------
# Helper FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# --------------------------------------------------------------
# Sequence construction from a low‑dimensional parameter vector
def _gauss_mix_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a sequence of length n from K Gaussian bumps.
    params length = 3*K -> (weight, centre, sigma) repeated.
    The resulting vector is clipped to >=0 and renormalised to sum 1.
    """
    K = len(params) // 3
    xs = np.linspace(0.0, 1.0, n)
    a = np.zeros(n, dtype=np.float64)

    for i in range(K):
        w = max(params[3 * i], 0.0)
        p = np.clip(params[3 * i + 1], 0.0, 1.0)
        sigma = np.clip(params[3 * i + 2], 0.01, 0.5)
        # Gaussian bump
        a += w * np.exp(-0.5 * ((xs - p) / sigma) ** 2)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= total
    return a

# Initial candidate generator (same ideas as the reference code)
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # three‑Gaussian peaks
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    np.random.shuffle(cand)
    return cand[:12]   # keep a modest number

# Adam optimisation (smooth‑max surrogate)
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max projection to the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights over the convolution vector
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate (see paper)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule – make the smooth‑max sharper
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val

# Exact sub‑gradient optimisation (exponentiated‑gradient)
def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient steps on the true max‑convolution objective.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (maintains non‑negativity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


# --------------------------------------------------------------
# Simple hill‑climbing local search
def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Random pairwise mass transfers; keep a move if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        eps = np.random.rand() * 0.05 * a[i1]          # move up to 5 % of mass
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:                        # numeric safety
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            if np.random.rand() < 0.001:              # occasional uphill jump
                a = new_a

    return best_a, best_val

# Differential‑Evolution global search (low‑dimensional Gaussian mixture)
def _de_optimize_for_length(n: int,
                            deadline: float,
                            seed: int = 0,
                            max_de_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Run a brief DE optimisation on a K‑Gaussian mixture to obtain a good
    starting sequence for length n.  Returns (seq, value).  If the time
    budget is already exhausted the function returns (None, inf).
    """
    try:
        from scipy.optimize import differential_evolution
    except Exception:                     # SciPy not available – skip DE
        return None, float("inf")

    K = 5                                 # number of Gaussian components
    dim = 3 * K
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 1.0))                # weight
        bounds.append((0.0, 1.0))                # centre
        bounds.append((0.01, 0.5))               # sigma

    conv_fft_len = _next_pow2(2 * n - 1)

    def de_obj(p):
        a = _gauss_mix_sequence(p, n)
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        return 2.0 * n * M                      # sum(a) is 1 after construction

    # Run DE with a small number of generations – fast enough for our budget
    result = differential_evolution(de_obj,
                                    bounds,
                                    maxiter=max_de_iters,
                                    polish=False,
                                    seed=seed,
                                    updating='deferred',
                                    disp=False)
    seq = _gauss_mix_sequence(result.x, n)
    val = de_obj(result.x)
    return seq, val


# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional historic solution (if any)
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # Lengths to explore – small to medium are cheap enough for many tries
    lengths = [1024, 2048, 4096, 8192]

    # pre‑compute FFT lengths for each n
    fft_conv = {}
    fft_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_conv[n] = _next_pow2(conv_len)
        fft_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = fft_conv[n]
        corr_fft_len = fft_corr[n]

        # ------- Global DE candidate ---------------------------------
        de_seq, de_val = _de_optimize_for_length(
            n, deadline, seed=seed + n, max_de_iters=30
        )
        # If DE failed (e.g. missing SciPy) we just ignore it.
        extra_candidates = []
        if de_seq is not None:
            extra_candidates.append(de_seq)
            print(f"[DE] length {n} initial value {de_val:.6f}")

        # ------- Other diverse warm‑starts ---------------------------
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)
        # include the DE candidate (if any)
        candidates = extra_candidates + candidates

        # Reduce the number of candidates a bit to keep runtime in check
        candidates = candidates[:8]

        # ----- Determine iteration budget for this length -------------
        # Rough scaling: larger n → fewer iterations.
        max_adam_iters = int(15000 * 1024 / n) + 2000

        for idx, start in enumerate(candidates):
            if time.time() > deadline:
                break
            print(f"[n={n}] candidate {idx} (Adam iters={max_adam_iters})")

            # Phase 1 – Adam on the smooth‑max surrogate
            a_adam, val_adam = _adam_optimize(
                start, n, conv_fft_len, corr_fft_len, deadline, max_adam_iters
            )

            # Phase 2 – exact sub‑gradient optimisation
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, conv_fft_len, corr_fft_len, deadline, init_val=val_adam
            )

            # Phase 3 – cheap hill‑climbing local search
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, conv_fft_len, deadline, init_val=val_sub
            )

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={idx}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run refinement on the best sequence (if any time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        conv_fft_len = fft_conv.get(n, _next_pow2(2 * n - 1))
        corr_fft_len = fft_corr.get(n, _next_pow2(3 * n))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=int(1.5 * (15000 * 1024 / n) + 3000),
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    print(f"\n=== Finished. Best objective = {best_val:.6f} (length {best_seq.shape[0]}) ===")
    return best_seq.tolist()
