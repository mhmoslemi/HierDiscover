"""
A multi‑scale, multi‑restart optimizer for the step‑function problem.

The algorithm works in three stages:

1. **Coarse search (n≈2 k).**  
   A diverse pool of initial shapes (warm‑start, Dirichlet,
   power‑law edges, exponentials, sigmoids, sinusoid, uniform) is
   generated.  Each candidate is quickly refined with a few
   Adam + sub‑gradient iterations; the three best are kept.

2. **Focused refinement.**  
   The selected candidates are optimized longer (up to 200 k Adam
   steps and 10 k sub‑gradient steps) on the same size, keeping the
   overall best sequence.

3. **Upscaling.**  
   The best distribution is linearly interpolated to larger
   lengths (8192 and, if time permits, 16384) and a short
   optimisation is performed again.  This exploits the extra
   degrees of freedom of longer vectors while keeping the runtime
   modest.

All updates stay on the probability simplex (non‑negative, sum = 1),
use an entropy‑mirror (soft‑max) parametrisation and Adam for the
smooth‑max surrogate, followed by an exponentiated‑gradient
sub‑gradient phase.  Random “shakes’’ are added when progress stalls.
The function respects the supplied time budget and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective value (sum(a) assumed ≈ 1)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    *,
    max_adam_iters: int = 200_000,
    max_subgrad_iters: int = 8_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on a smooth‑max surrogate followed by a sub‑gradient phase.
    The routine updates the global best sequence/value and respects
    the overall time budget.
    """
    # ---- initialise soft‑max parametrisation ----
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.1               # start temperature (smooth‑max)
    lam_factor = 1.02

    last_improve = time.time()
    it = 0

    # ------------------- Adam phase -------------------
    while time.time() < deadline and it < max_adam_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:05d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weighting
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # random shake when stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    # ------------------- Sub‑gradient phase -------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()

    for sub_it in range(max_subgrad_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active set of maximal convolution entries
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid complete collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 5.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a modest pool of diverse start vectors for the optimiser."""
    cand = []

    # warm‑start (previous best) and simple rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge concentrated shapes (dense α sweep)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.3, 10):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # exponential decays from left/right
    for delta in (0.001, 0.003, 0.01):
        r = 1.0 - delta
        shape = r ** np.arange(n)
        shape /= shape.sum()
        cand.append(shape)                     # left → right
        shape = r ** np.arange(n)[::-1]
        shape /= shape.sum()
        cand.append(shape)                     # right → left

    # logistic (sigmoid) shapes
    for beta in (5, 10, 15):
        x = np.arange(n)
        shape = 1.0 / (1.0 + np.exp(-beta * (x - n / 2) / n))
        shape /= shape.sum()
        cand.append(shape)

    # simple half‑wave sinusoid (positive)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and returns the best
    sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ---------- base size (fast coarse optimisation) ----------
    base_n = 2048
    conv_fft_len = _next_pow2(2 * base_n - 1)
    corr_fft_len = _next_pow2(3 * base_n - 2)

    # warm‑start from previous global variable if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > base_n:
            # down‑sample by simple averaging over blocks
            factor = warm.shape[0] // base_n
            warm = warm[: factor * base_n].reshape(base_n, factor).mean(axis=1)
        elif warm.shape[0] < base_n:
            warm = np.pad(warm, (0, base_n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    candidates = _initial_candidates(base_n, warm)

    # ------------------------------------------------------------------
    # 1) Very short optimisation for all candidates – rank them
    short_adam = 50_000
    short_sub = 2_000
    ranked = []

    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(base_n))
        else:
            a0 /= a0.sum()

        seq, val = _deep_optimize(
            a0,
            base_n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            None,
            float("inf"),
            max_adam_iters=short_adam,
            max_subgrad_iters=short_sub,
        )
        if seq is not None:
            ranked.append((val, seq))
            if val < best_val:
                best_val, best_seq = val, seq
                print(f"[rank] new global best {val:.6f}")

    # keep the three most promising
    ranked.sort(key=lambda x: x[0])
    top_seqs = [seq for _, seq in ranked[:3]]

    # ------------------------------------------------------------------
    # 2) Longer refinement of the top candidates
    long_adam = 200_000
    long_sub = 10_000

    for seq0 in top_seqs:
        if time.time() >= deadline:
            break
        seq, val = _deep_optimize(
            seq0,
            base_n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            max_adam_iters=long_adam,
            max_subgrad_iters=long_sub,
        )
        if seq is not None and val < best_val:
            best_val, best_seq = val, seq
            print(f"[refine] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # 3) Upscaling to larger lengths (more degrees of freedom)
    def upscale_and_optimize(prev_seq, target_n, adam_iters, sub_iters):
        if prev_seq is None:
            return None, float("inf")
        x_old = np.linspace(0, 1, len(prev_seq), endpoint=False)
        x_new = np.linspace(0, 1, target_n, endpoint=False)
        a_big = np.interp(x_new, x_old, prev_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        conv_len = _next_pow2(2 * target_n - 1)
        corr_len = _next_pow2(3 * target_n - 2)
        seq, val = _deep_optimize(
            a_big,
            target_n,
            conv_len,
            corr_len,
            deadline,
            best_seq,
            best_val,
            max_adam_iters=adam_iters,
            max_subgrad_iters=sub_iters,
        )
        return seq, val

    # first upscale to 8192
    if time.time() + 5 < deadline:
        seq, val = upscale_and_optimize(best_seq, 8192, 150_000, 5_000)
        if seq is not None and val < best_val:
            best_val, best_seq = val, seq
            print(f"[upscale 8192] new global best {val:.6f}")

    # then, if budget permits, upscale to 16384
    if time.time() + 5 < deadline:
        seq, val = upscale_and_optimize(best_seq, 16384, 80_000, 3_000)
        if seq is not None and val < best_val:
            best_val, best_seq = val, seq
            print(f"[upscale 16384] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Tiny random local polishing (optional, cheap)
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n - 2)
        # minor moves – move a tiny amount between two random positions
        step = 0
        while time.time() < deadline and step < 5_000:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps_move = 1e-4 * a[i]   # move at most 0.01 % of the source mass
            if eps_move <= 0:
                continue
            a_new = a.copy()
            a_new[i] = max(a_new[i] - eps_move, 0.0)
            a_new[j] = a_new[j] + eps_move
            a_new /= a_new.sum()
            val = _objective(a_new, n)
            if val < best_val:
                best_val, best_seq, a = val, a_new, a_new
                # keep the improvement
            step += 1

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return
    if best_seq is None:
        best_seq = np.full(base_n, 1.0 / base_n, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
