def proj_simplex(v):
    # return projection of v onto the probability simplex (nonnegative, sum=1)
    if v.sum() == 1 and np.all(v>=0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a hybrid of (i) stochastic Adam on a soft‑max
parameterisation with a temperature‑controlled smooth‑max, (ii) a
sub‑gradient phase that uses a hard one‑hot max, and (iii) a cheap
pair‑wise mass‑shuffling refinement.  It searches several lengths,
starting from the best known construction (global variable
`height_sequence_1`) and gradually increases the problem size up to
8192 coordinates.  The whole optimisation respects the 1000 s wall‑time
budget and always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Compute a temperature‑smoothed max of ``conv`` and the
    corresponding softmax weights (∂smooth_max/∂conv).
    """
    c_max = conv.max()
    # shift for numerical stability
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Time‑bounded optimisation that returns the best non‑negative
    sequence found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # NumPy array, sum = 1
    best_val = float("inf")  # current best value of 2·n·max(conv)

    # Problem sizes – we go up to 8192 (still feasible within the budget)
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths reused many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # number of restarts – a few for the large problems
        if n <= 1024:
            restarts = 6
        elif n <= 4096:
            restarts = 4
        else:
            restarts = 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------------------------------------------------------
            # initialise logits (soft‑max parameters)
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a0 = best_seq.copy()
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] warm‑start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] > n:
                    cand = cand[:n]
                elif cand.shape[0] < n:
                    # repeat / interpolate to the required length
                    reps = int(np.ceil(n / cand.shape[0]))
                    cand = np.tile(cand, reps)[:n]
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand = cand / cand.sum()
                x = np.log(cand + 1e-15)
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random start")

            # Adam state
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.12                     # initial learning rate
            decay_every = 6000
            decay_factor = 0.7

            # iteration budget (adjusted to length)
            if n <= 1024:
                max_iters = 150_000
            elif n <= 4096:
                max_iters = 100_000
            else:
                max_iters = 70_000

            # temperature schedule for smooth‑max
            tau_initial = 0.15
            tau_final = 5e-7

            last_improve = time.time()
            t = 0
            while time.time() < deadline and t < max_iters:
                # --------------------------------------------------------------
                # obtain probability vector a from logits
                # --------------------------------------------------------------
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()               # a_i ≥ 0, Σa = 1

                # --------------------------------------------------------------
                # convolution and (smooth) max
                # --------------------------------------------------------------
                conv = full_convolution(a, fft_len=fft_len_conv)

                # first 70 % of the iterations use a smoothed max,
                # the last 30 % switch to a hard one‑hot max
                if t < int(0.7 * max_iters):
                    tau = max(tau_final,
                              tau_initial * (1.0 - t / max_iters))
                    _, weights = smooth_max_and_weights(conv, tau)
                else:
                    # one‑hot weight on the current maximum entry
                    idx = np.argmax(conv)
                    weights = np.zeros_like(conv)
                    weights[idx] = 1.0

                # true objective value (for bookkeeping)
                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best {cur_val:.6f}")

                # --------------------------------------------------------------
                # gradient of (smooth) max w.r.t. a
                # --------------------------------------------------------------
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # --------------------------------------------------------------
                # Adam update on logits
                # --------------------------------------------------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay
                if t % decay_every == 0:
                    lr *= decay_factor

                # ----------------------------------------------------------------
                # occasional “shake” when stuck
                # ----------------------------------------------------------------
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 8e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # ------------------------------------------------------------------
            # Phase 2 – projected‑gradient refinement on the simplex (hard max)
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                lr2 = 0.03
                # run a few thousand sub‑gradient steps with explicit projection
                proj_iters = 5000
                for it in range(proj_iters):
                    if time.time() > deadline:
                        break
                    conv = full_convolution(a, fft_len=fft_len_conv)
                    t_max = np.argmax(conv)

                    # gradient of the hard max (one‑hot weight)
                    grad_a = 2.0 * cross_correlation(
                        np.eye(1, 2 * n - 1, t_max, dtype=np.float64)[0],
                        a,
                        fft_len=fft_len_corr,
                    )
                    # simple gradient step
                    a_new = a - lr2 * grad_a
                    # projection back onto the simplex
                    a_new = project_simplex(a_new)

                    # evaluate
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a = a_new
                        print(f"[proj‑refine n={n}] improved to {val_new:.6f}")
                    else:
                        a = a_new

            # ------------------------------------------------------------------
            # cheap mass‑shuffling local refinement (same as original code)
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    eps_move = min(a_local[i], 0.008) * (np.random.rand() * 0.6 + 0.2)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine n={n}] improved to {val_new:.6f}")

    # -------------------------------------------------------------------------
    # fallback – uniform vector (objective = 2)
    # -------------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
