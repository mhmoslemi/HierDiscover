def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = np.inf

    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline: break

        # precompute FFT lengths
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))

        # deterministic seeds
        seeds = []

        # uniform
        seeds.append(np.full(n, 1.0 / n))

        # linear inc
        linear_inc = np.arange(1, n+1, dtype=float)
        linear_inc = linear_inc / linear_inc.sum()
        seeds.append(linear_inc)

        # linear dec
        linear_dec = linear_inc[::-1]
        seeds.append(linear_dec)

        # exponential dec with r values
        for r in [0.999, 0.995, 0.99, 0.97]:
            exp_seq = np.power(r, np.arange(n))
            exp_seq = exp_seq / exp_seq.sum()
            seeds.append(exp_seq)

        # power law p values
        for p in [0.5, 1.0, 1.5, 2.0]:
            idx = np.arange(1, n+1)
            pl = 1.0 / np.power(idx, p)
            pl = pl / pl.sum()
            seeds.append(pl)

        # random Dirichlet seeds
        for _ in range(3):
            seeds.append(np.random.dirichlet(np.ones(n)))

        # For each seed, run optimizer
        for seed_seq_idx, a0 in enumerate(seeds):
            if time.time() > deadline: break

            # convert to logit
            eps = 1e-12
            x = np.log(a0 + eps)

            # Adam states
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
            lr = 0.15
            decay_every = 5000

            max_iters = 100000 if n <= 2048 else 80000

            # temperature schedule
            tau_initial = 0.2
            tau_final = 1e-7

            # track improvement
            last_improve = time.time()

            for t in range(1, max_iters+1):
                if time.time() > deadline: break

                # softmax to get a
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # convolution
                conv = full_convolution(a, fft_len=fft_len_conv)

                # compute true objective
                cur_val = 2.0 * n * conv.max() / (np.sum(a)**2)

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    # optionally print improvement

                # decide whether to use smooth-max or exact
                tau = max(tau_final, tau_initial * (1.0 - t / max_iters))

                if tau > 1e-4:
                    smooth, weights = smooth_max_and_weights(conv, tau)
                    grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)
                else:
                    # exact max
                    max_idx = np.argmax(conv)
                    w = np.zeros_like(conv)
                    w[max_idx] = 1.0
                    grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # backprop through softmax
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                if t % decay_every == 0:
                    lr *= 0.7

                # stagnation shake
                if time.time() - last_improve > 30.0:
                    # add jitter (small Gaussian)
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    # optionally print

            # after optimization, do local refinement moves
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                for _ in range(2000):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    delta = min(a_local[i], 0.005) * (np.random.rand() * 0.8 + 0.1)
                    if delta <= 0: continue
                    a_new = a_local.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max() / (np.sum(a_new)**2)
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new

    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0/n_fallback)

    # final scaling: ensure sum not too small (should be >0.01). if sum small, adjust
    sum_a = np.sum(best_seq)
    if sum_a < 0.01:
        # rescale by adding small uniform mass
        best_seq = np.maximum(best_seq, 1e-8)
        best_seq /= best_seq.sum()
        best_seq *= np.sqrt(2 * len(best_seq))

    print(f"Final best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Propose a non‑negative step‑function (vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The implementation combines several ideas:

* A soft‑max (log‑sum‑exp) smoothing of the “max” is used while the
  temperature is still large – this yields a smooth gradient and helps
  the optimiser escape poor basins.
* When the temperature becomes small we switch to the exact sub‑gradient
  of the true max (the convolution entry that attains the maximum).  This
  allows a precise final fine‑tuning.
* Adam with a decaying learning rate is used on the logits of a soft‑max
  parametrisation (so the simplex constraints are always satisfied).
* A diverse set of deterministic initialisations (uniform, linear,
  exponential, power‑law, and the previously found ``height_sequence_1``)
  is tried, together with a few random Dirichlet restarts.
* After each optimisation run a cheap greedy “mass‑shuffling’’ refinement
  moves a small amount of probability mass between two positions whenever
  it improves the objective.
* The whole routine is time‑bounded (default 1000 s) and respects the
  budget by checking the deadline frequently.

The best sequence found (its list of floats) is returned.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑smoothed max of `conv` and its soft‑max weights.
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(a)^2)"""
    n = a.shape[0]
    conv = full_convolution(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Time‑bounded optimisation that returns the best non‑negative step‑function
    (list of floats).  The strategy is a multi‑restart Adam on the simplex
    with a temperature‑annealed smooth max followed by an exact max stage and
    a final greedy local refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None   # best sequence (numpy array, sum = 1)
    best_val = float("inf")

    # lengths to try – larger n gives a lower theoretical bound
    lengths = [1024, 2048, 4096]

    # ----------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (reused many times)
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # ---------- build a list of initial candidates --------------------
        init_candidates = []

        # uniform
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # linear increasing and decreasing
        lin = np.arange(1, n + 1, dtype=np.float64)
        init_candidates.append(lin / lin.sum())
        init_candidates.append(lin[::-1] / lin.sum())

        # exponential families
        for r in (0.999, 0.995, 0.99, 0.97):
            exp_seq = np.power(r, np.arange(n, dtype=np.float64))
            exp_seq = exp_seq / exp_seq.sum()
            init_candidates.append(exp_seq)

        # power‑law families
        for p in (0.5, 1.0, 1.5, 2.0):
            idx = np.arange(1, n + 1, dtype=np.float64)
            pl = 1.0 / np.power(idx, p)
            pl = pl / pl.sum()
            init_candidates.append(pl)

        # use the previously discovered sequence if available
        if "height_sequence_1" in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() >= 0.01:
                init_candidates.append(cand / cand.sum())
            else:
                # fall back to Dirichlet if the pasted vector is degenerate
                init_candidates.append(np.random.dirichlet(np.ones(n)))

        # a few random Dirichlet starts
        for _ in range(3):
            init_candidates.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------
        # For each candidate run an Adam optimisation
        for start_id, a0 in enumerate(init_candidates):
            if time.time() > deadline:
                break

            # initialise logits (soft‑max parameters)
            eps = 1e-15
            x = np.log(a0 + eps)

            # Adam state
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
            lr = 0.15                       # initial learning rate
            decay_every = 4000              # learning‑rate decay interval

            # iteration budget – depends on n
            if n <= 1024:
                max_iters = 80000
            elif n <= 2048:
                max_iters = 60000
            else:
                max_iters = 40000

            # temperature schedule for smooth‑max
            tau_initial = 0.25
            tau_final = 5e-7

            last_improve = time.time()

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # ---- obtain probability vector from logits -----------------
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()                # Σ a = 1, a ≥ 0

                # ---- convolution and (smooth) max -------------------------
                conv = full_convolution(a, fft_len=fft_len_conv)

                # exact objective (used for bookkeeping)
                cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} start={start_id} iter={t:06d}] new best {cur_val:.6f}")

                # ---- choose gradient mode ---------------------------------
                tau = max(tau_final, tau_initial * (1.0 - t / max_iters))

                if tau > 1e-4:
                    # smooth max – smooth gradient
                    _, weights = smooth_max_and_weights(conv, tau)
                    grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)
                else:
                    # exact max – sub‑gradient of the true max
                    max_idx = int(conv.argmax())
                    w = np.zeros_like(conv)
                    w[max_idx] = 1.0
                    grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # ---- back‑prop through soft‑max ---------------------------
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ---- Adam update -----------------------------------------
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # learning‑rate decay
                if t % decay_every == 0:
                    lr *= 0.7

                # ---- stagnation handling ---------------------------------
                if time.time() - last_improve > 30.0:
                    # small random jitter in log‑space
                    x += (np.random.rand(n) - 0.5) * 6e-3
                    last_improve = time.time()
                    print(f"[n={n} start={start_id} iter={t}] shake (stagnation)")

            # ---- cheap greedy mass‑shuffling refinement --------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                for _ in range(2500):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small fraction from i to j
                    eps_move = min(a_local[i], 0.006) * (np.random.rand() * 0.7 + 0.2)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[n={n} refine] improved to {val_new:.6f}")

    # ----------------------------------------------------------------------
    # fallback – uniform vector if nothing was found
    if best_seq is None:
        n_fb = 1024
        best_seq = np.full(n_fb, 1.0 / n_fb, dtype=np.float64)

    # final sanity: avoid a zero‑sum vector (evaluate_sequence would reject it)
    if best_seq.sum() < 0.01:
        best_seq = np.maximum(best_seq, 1e-12)
        best_seq /= best_seq.sum()

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
