def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[:2*n-1]

def _cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n-1
    return corr[start:start+n]

"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1.  Multi‑scale optimisation: start with a short sequence (n≈256) and
    progressively double the length up to 32768 (or the largest size that
    fits in the remaining time).  After each scale the best shape is
    interpolated and used as a warm‑start for the next scale.
2.  For every length a modest set of diverse seeds is built
    (warm‑start, random Dirichlet, edge‑power‑laws, edge‑Gaussians …).
3.  Each seed is refined by a sequence of L‑BFGS runs on a *smooth‑max*
    surrogate of the maximum autocorrelation.  The temperature β is
    increased geometrically (1 → 5 → 25 → …) so that the optimisation
    starts very smooth and ends almost exactly at the true max.
    Gradients are obtained analytically via FFT‑based cross‑correlation.
4.  After the L‑BFGS chain a cheap hill‑climb (mass‑swap) is applied to
    polish the result.
5.  The overall best sequence is finally refined with a long Adam run
    (the Adam routine from the reference implementation) for maximal
    accuracy.
6.  The entire procedure respects the 1000 s budget – it checks the
    deadline before every heavy block and aborts gracefully.

The returned object is a plain Python list of non‑negative floats,
ready to be passed directly to ``evaluate_sequence``.

import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new

        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    np.random.shuffle(cand)
    return cand[:12]   # keep a modest number


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *smooth‑max* surrogate.
    Returns (best_sequence, best_value) before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex projection via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # random shake after stagnation
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small amount of mass between two random
    coordinates and keep the move if it improves the true objective.
    """
    a = a0.copy()
    cur_val = (
        _objective(a, n, fft_len_conv) if init_val is None else init_val
    )
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


def _objective(sequence: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(sequence, fft_len)
    max_c = conv.max()
    s = sequence.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a minimal value of
    ``evaluate_sequence``.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: float
        Total time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found, as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # initialise with any historic solution that might exist
    best_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            arr = np.asarray(tmp, dtype=np.float64)
            if arr.ndim == 1:
                best_seq = np.maximum(arr, 0.0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s

    best_val = float("inf")

    # ------------------------------------------------------------------
    # multiscale lengths (start small, finish large)
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]

    # pre‑compute FFT lengths for each n
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_fft[n] = _next_pow2(2 * n - 1)
        corr_fft[n] = _next_pow2(3 * n - 2)   # 2n‑1 + n‑1 ≤ 3n‑2

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_c = conv_fft[n]
        fft_x = corr_fft[n]

        # generate a diverse pool of seeds
        candidates = _generate_initial_candidates(n, best_seq, None)
        # keep the pool small for the largest lengths
        if n >= 8192:
            candidates = candidates[:6]
        else:
            candidates = candidates[:10]

        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            a0 = np.maximum(cand, 0.0)
            s = a0.sum()
            if s < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            else:
                a0 = a0 / s

            logits = np.log(a0 + 1e-15)

            # geometric schedule of smooth‑max temperatures
            beta_schedule = [1.0, 5.0, 25.0, 125.0, 625.0, 2500.0]

            for beta in beta_schedule:
                if time.time() > deadline:
                    break

                # ----- L‑BFGS on smooth‑max -----
                def fun_logit(x, n=n, fft_len_conv=fft_c,
                              fft_len_corr=fft_x, beta=beta):
                    # soft‑max parametrisation
                    m = np.max(x)
                    exp_x = np.exp(x - m)
                    a = exp_x / exp_x.sum()

                    # convolution
                    conv = _full_convolution(a, fft_len_conv)
                    max_c = conv.max()
                    shifted = beta * (conv - max_c)
                    exp_shifted = np.exp(shifted)
                    sum_exp = exp_shifted.sum()
                    smooth_max = max_c + (1.0 / beta) * np.log(sum_exp)

                    loss = 2.0 * n * smooth_max

                    # gradient components
                    w = exp_shifted / sum_exp
                    grad_a = 4.0 * n * _cross_correlation(w, a, fft_len_corr)
                    avg = np.dot(a, grad_a)
                    grad_log = a * (grad_a - avg)

                    return loss, grad_log

                res = minimize(fun_logit,
                               logits,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 30,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                logits = res.x

                # evaluate the true (non‑smoothed) objective
                a_opt = np.exp(logits - np.max(logits))
                a_opt /= a_opt.sum()
                val = evaluate_sequence(a_opt.tolist())

                if val < best_val:
                    best_val = val
                    best_seq = a_opt.copy()
                    print(f"[n={n} cand={cid} β={beta}] NEW GLOBAL best {val:.6f}")

            # ----- cheap hill‑climb polish (few iterations) -----
            if time.time() > deadline:
                break
            a_hc, val_hc = _hill_climb_optimize(
                best_seq,
                n,
                fft_c,
                deadline,
                init_val=best_val,
                max_iters=500,
            )
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"[n={n} cand={cid} HillClimb] NEW GLOBAL best {val_hc:.6f}")

    # ------------------------------------------------------------------
    # final long‑run Adam refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_c = conv_fft.get(n, _next_pow2(2 * n - 1))
        fft_x = corr_fft.get(n, _next_pow2(3 * n - 2))
        print("\n=== Long‑run Adam refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            fft_c,
            fft_x,
            deadline,
            max_iters=int(1.5 * (15000 * 1024 / n) + 3000),
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # fallback (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print("Fallback uniform sequence used.")

    print(f"\nFinal best value = {best_val:.6f}")
    return best_seq.tolist()
