def _lp_improve(a, target_sum, deadline, t=0.2, max_iter=10):
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_val = evaluate_sequence(best_a.tolist())
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv and M
        conv = _full_convolution_fft(a_curr, a_curr)
        M = conv.max()
        n = a_curr.shape[0]
        # build matrix A
        A = np.zeros((2*n-1, n), dtype=float)
        for j in range(n):
            A[j:j+n, j] = a_curr  # column fill
        # LP
        c = np.zeros(n)  # zero objective
        A_ub = A
        b_ub = np.full(2*n-1, M)
        A_eq = np.ones((1, n))
        b_eq = np.array([target_sum])
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'disp': False})
        except Exception as e:
            print(f"LP failed: {e}")
            break
        if not res.success:
            print(f"LP not feasible at iteration {it}")
            break
        g = res.x
        # Mix
        a_new = (1 - t) * a_curr + t * g
        # Evaluate
        val_new = evaluate_sequence(a_new.tolist())
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP improve iter={it} new best={best_val:.6f}")
            a_curr = a_new
        else:
            # reduce t and possibly break
            t *= 0.5
            if t < 1e-4:
                break
    return best_a, best_val
```

"""
Hybrid optimisation for the “self‑convolution peak” problem.

The evaluation of a step‑function `a` is  

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is invariant under scaling of `a`.  The goal is to minimise the
maximum coefficient of the polynomial `P(x)=∑ a_i x^i` squared.  The
procedure consists of four stages:

1. **Warm‑start** – use the global construction if available.
2. **Soft‑max Adam** – many short runs on random lengths (800‑3000) and
   random initialisations (Dirichlet, power‑law, etc.).  The logits are
   optimised with Adam on a smooth log‑sum‑exp surrogate of the max.
3. **Linear‑programming refinement** – for the current best sequence `f`,
   solve a feasibility LP (via `scipy.optimize.linprog`) that finds a
   sequence `g` with the same ℓ₁‑norm and satisfying
   `conv(f, g) ≤ max(conv(f, f))`.  Mixing `f←(1‑t)f + t g` (t≈0.2) strictly
   reduces the peak while keeping the simplex constraint.
4. **Projected sub‑gradient descent** and **stochastic hill‑climb** – final
   local polishing.

All stages respect the overall time budget (default 1000 s).  The
function `propose_candidate` returns the best list found.
"""

import time
import numpy as np

# optional: scipy for the LP‑refinement phase
try:
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False


# ----------------------------------------------------------------------
# Fast full convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {w ≥ 0, Σ w = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """Adam optimiser for the smooth‑max surrogate."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- soft‑max mapping ------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                # stable
        p = exp_shift / np.sum(exp_shift)            # Σp = 1
        a = target_sum * p

        # ---- exact max of the convolution (for logging) --------------------
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}   new local best max={cur_max:.6f}")

        # ---- smooth‑max surrogate (log‑sum‑exp) -----------------------------
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient wrt a (2·a⊛w_rev) ------------------------------------
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # slice for grad
        grad_a = 2.0 * conv2[idx]

        # ---- back‑prop to logits --------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9995

    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max(conv(a, a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the active peak
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate.copy()
            best_val = val_candidate
            step_size *= 1.02
        else:
            step_size *= 0.70

        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}"
            )
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# LP refinement – solve for g with conv(f,g) ≤ M and Σg = target_sum
# ----------------------------------------------------------------------
def _lp_refine(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    mixing: float = 0.20,
    max_iter: int = 5,
) -> tuple[np.ndarray, float]:
    """Alternating linear‑programming refinement."""
    if not _HAS_SCIPY:
        return f, evaluate_sequence(f.tolist())

    n = f.shape[0]
    best = f.copy()
    best_val = evaluate_sequence(best.tolist())
    t = mixing

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution_fft(f, f)
        M = conv.max()

        # Build convolution matrix A (size (2n‑1) × n)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f  # column j is f shifted by j

        c = np.zeros(n)                     # zero objective – just find feasibility
        A_ub = A
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        A_eq = np.ones((1, n), dtype=np.float64)
        b_eq = np.array([target_sum], dtype=np.float64)
        bounds = [(0, None)] * n

        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(5.0, deadline - time.time())},
            )
        except Exception as e:               # pragma: no cover
            print(f"[{time.time():.2f}] LP failed (it={it}): {e}")
            break

        if not res.success:
            print(f"[{time.time():.2f}] LP infeasible (it={it})")
            break

        g = res.x
        # convex blend
        f_new = (1.0 - t) * f + t * g
        val_new = evaluate_sequence(f_new.tolist())

        if val_new < best_val:
            best_val = val_new
            best = f_new.copy()
            f = f_new
            print(f"[{time.time():.2f}] LP refine it={it} new best={best_val:.6f}")
        else:
            # if no improvement, shrink the mixing factor
            t *= 0.5
            if t < 1e-4:
                break

    return best, best_val
