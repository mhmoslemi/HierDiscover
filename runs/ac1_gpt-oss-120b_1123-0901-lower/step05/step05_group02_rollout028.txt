All helper functions are defined at module level (no closures) and the

# Linear convolution a*a using an FFT of length ``fft_len`` (full convolution)
def linear_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a*a (full linear convolution) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the 2*len(a)-1 valid samples
    return conv[: 2 * a.shape[0] - 1]


# Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
def linear_cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]  for i=0..n‑1,
    where n = len(a).  Uses an FFT of length ``fft_len``.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# Exact objective (sum is assumed to be 1, i.e. a lies on the simplex)
def exact_objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """2·n·max(conv) for a vector with unit sum."""
    conv = linear_convolution(a, conv_len)
    return float(2.0 * n * conv.max())

def _refine_candidate(
    a0: np.ndarray,

    # ------- Adam on smooth‑max ------------------------------------------------
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.03                      # moderate learning rate
    lam = 0.1                      # start temperature (very smooth)
    lam_factor = 1.005             # slow increase
    lam_max = 200.0                # never grow too large
    t = 0
    last_improve = time.time()

    while time.time() < deadline - 2.0:
        t += 1
        # soft‑max → simplex
        exp_x = np.exp(logits - logits.max())
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = linear_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient of the smooth surrogate (linear in the convolution)
        grad_a = 2.0 * linear_cross_correlation(w, a, corr_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, lam_max)

        # stagnation shake
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    # ------- Sub‑gradient on the exact max ------------------------------------
    a = a / a.sum()                     # ensure exact simplex
    eta = 0.55                          # step size for exponentiated gradient
    sub_last = time.time()
    for sub_it in range(20000):
        if time.time() > deadline - 1.0:
            break
        conv = linear_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last = time.time()
            print(f"[n={n} SubGrad] new best {val:.6f}")

        # uniform distribution over the active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * linear_cross_correlation(w, a, corr_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                         # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        if time.time() - sub_last > 8.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin for the outer call

    best_seq = None
    best_val = float("inf")

    # Lengths to try – larger length gives more expressive power but costs more
    # time per iteration.  We start with the length that yielded the previous
    # best (4096) and then move to a larger one if time permits.
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (powers of two → fast FFT)
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        # correlation length must accommodate conv_len + n - 1 ≤ 3·n
        corr_len = 1 << int(np.ceil(np.log2(3 * n)))

        # ------------------------------------------------------------------
        # 1. Random / parametric initial pool
        init_pool = _initial_candidates(n)

        # evaluate all candidates (cheap) and keep the best few
        scored = [(exact_objective(c, n, conv_len), c) for c in init_pool]
        scored.sort(key=lambda t: t[0])
        top_candidates = [c for _, c in scored[:5]]
        # also keep the absolute best seen so far
        if scored and scored[0][0] < best_val:
            best_val = scored[0][0]
            best_seq = scored[0][1].copy()
            print(f"  Initial best value {best_val:.6f}")

        # ------------------------------------------------------------------
        # 2. Refine each top candidate (the heavy part).  Time is split evenly
        #    among the candidates.
        remaining = max(0.0, deadline - time.time())
        if remaining < 1.0:
            break
        per_candidate_budget = remaining / max(1, len(top_candidates))
        for cand in top_candidates:
            cand_deadline = time.time() + per_candidate_budget
            best_seq, best_val = _refine_candidate(
                cand,
                n,
                conv_len,
                corr_len,
                cand_deadline,
                best_val,
                best_seq,
            )
            if time.time() > deadline:
                break

        # ------------------------------------------------------------------
        # 3. Optional fine‑tuning of the global best for this length
        if best_seq is not None and len(best_seq) == n:
            print(f"  Fine‑tuning global best (value={best_val:.6f})")
            # a single extra Adam run with a slower schedule
            best_seq, best_val = _refine_candidate(
                best_seq,
                n,
                conv_len,
                corr_len,
                deadline,
                best_val,
                best_seq,
            )
        # end of length loop

    # fallback – uniform vector (should never be needed)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
