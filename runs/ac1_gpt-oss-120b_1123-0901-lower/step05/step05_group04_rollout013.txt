    n = len(f)
    # Build constraint matrix A of shape (2n-1, n)
    rows = []
    for k in range(2*n-1):
        # for each i where 0<=i<n and 0 <= k-i < n => i in [max(0, k-(n-1)), min(n-1, k)]
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # g indices j = k - i, for i range.
        # But we need coefficients f_i for each g_j.
        # Actually conv_k = Σ_i f_i * g_{k-i}
        # So for each i in range, coefficient for g_{k-i} is f_i.
        # So we can for each i compute column index j = k - i.
        # We can put in row vector of length n.
    # Build using efficient algorithm:
    A = np.zeros((2*n-1, n), dtype=float)
    for i in range(n):
        # f_i contributes to conv_k for k from i to i+n-1 (when g index = k-i)
        # set A[i:i+n, i] = f_i
        A[i:i+n, i] = f[i]   # broadcast
    # For each row k, conv_k = A[k,:] @ g.
    # Constraint: A @ g <= M (elementwise)
    # Note: also implicit nonneg g.

    # Solve LP: max sum(g) subject to A @ g <= M and g >=0.
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    # Solve with ECOS automatically
    prob.solve(solver=cp.ECOS, warm_start=True)
    if prob.status not in ("optimal", "optimal_inaccurate"):
        # fallback: return zeros.
        return np.zeros_like(f)
    g_opt = np.array(g.value).flatten()
    return g_opt

def _solve_lp_max_sum_under_conv(f, M, time_limit):
    """
    Solve LP: maximize sum(g) s.t. conv(f,g) <= M, g>=0.
    Returns g as np.ndarray, or None if time exceeded.
    """
    n = f.shape[0]
    # Build sparse constraint matrix
    from scipy.sparse import csr_matrix
    row = []
    col = []
    data = []
    for i in range(n):
        start = i
        end = i + n   # exclusive
        row.extend(range(start, end))
        col.extend([i] * n)
        data.extend([f[i]] * n)
    A = csr_matrix((data, (row, col)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)

    # Use time limit for linprog by setting max_iter maybe? Not direct.
    # We'll check overall deadline prior to call.
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    except Exception as e:
        print("LP solve failed:", e)
        return None
    if res.success:
        return res.x
    else:
        # return None
        return None

"""
Propose a step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation is scale‑invariant, therefore we keep all candidates on the
simplex sum(a)=sqrt(2*n).  The search consists of two complementary phases:

1.  A fast Adam optimiser on the soft‑max parametrisation (exactly the
    code that produced the 1.5095 bound).  Many independent restarts with
    diverse initial shapes are performed; the best few candidates are kept.

2.  A polishing phase based on a *linear‑programming* step described in the
    problem statement: for a current vector ``f`` we solve

        maximise   sum(g)
        subject to   f * g  ≤  M   (componentwise),   g ≥ 0

    where ``M = max(f*f)``.  The LP is solved with SciPy’s HiGHS
    implementation.  The obtained ``g`` is rescaled to the canonical sum and
    mixed with ``f`` (convex combination).  Because the mixing never raises the
    max‑convolution, this phase can only improve the bound.

Both phases respect the global time budget (default 1000 s).  The routine
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a power‑of‑2 FFT.
    Small negative values caused by round‑off are clamped to zero.
    """
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  This is exactly the value returned by
    ``evaluate_sequence`` when sum(a)=sqrt(2*n)."""
    return _full_convolution_fft(a, a).max()


# ----------------------------------------------------------------------
# Random initial shapes
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive vector of length n with a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:                               # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max surrogate.
    Returns the best sequence found (as a NumPy array) and its official
    evaluation score (the exact max‑convolution).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # 1. soft‑max → probabilities with the required total mass
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # avoid overflow
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # 2. current max‑convolution
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  local best = {best_val:.12f}")

        # 3. smooth‑max weights (log‑sum‑exp)
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau               # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4. gradient w.r.t. a
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # slice we need
        grad_a = 2.0 * conv2[idx]

        # 5. gradient w.r.t. logits x (chain rule through soft‑max)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # 6. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9997

        # early stop if no improvement for a while
        if iteration - last_improve > 2000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # official evaluation (keeps the same rounding behaviour as the judge)
    final_score = best_val if best_seq is not None else float("inf")
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear‑programming polishing step
# ----------------------------------------------------------------------
def _solve_lp_max_sum_under_conv(f: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve the LP

        maximise   sum(g)
        subject to   conv(f, g) ≤ M   (componentwise)
                    g ≥ 0

    using SciPy's HiGHS interior‑point solver.  Returns the optimal g
    as a dense NumPy array, or ``None`` if the solver fails.
    """
    n = f.shape[0]
    # Build dense constraint matrix A where A[k,i] = f[i] if 0 ≤ k-i < n else 0
    A = np.zeros((2 * n - 1, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f[i]          # contribution of column i

    b = np.full(2 * n - 1, M, dtype=float)

    # maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=float)

    # run the LP; the solver itself respects its own internal time limit,
    # we only guard against an overall timeout
    start = time.time()
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP failure: {e}")
        return None
    if time.time() - start > time_limit:
        # solved too late – discard the result to keep the outer budget safe
        return None
    if not res.success:
        # no feasible solution (should not happen) – fall back to zeros
        return None
    return np.asarray(res.x, dtype=float)
