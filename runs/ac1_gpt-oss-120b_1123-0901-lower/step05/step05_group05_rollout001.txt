        grad = np.zeros_like(cur_a)

        grad[i_range] = 2*cur_a[k_max - i_range]

        # temperature schedule
        elapsed = time.time() - start
        # linear cooling

Propose a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is an **adaptive stochastic local search** tailored to the
objective’s structure:

* The sequence length `n` is chosen in the range 1500–2500 (the region where
  previous runs found good values).  For each length we perform an
  independent “run”.

* A run starts from a random Dirichlet vector (or a slight perturbation of
  the current global best) that is scaled to the required simplex
  `∑a = √(2n)`.

* The main optimisation loop is a **targeted annealing**:
  - At each step we compute the full convolution `c = a * a` (FFT based)
    and locate the index `k` where the maximum occurs.
  - The sub‑gradient of `c_k` w.r.t. `a` is known analytically:
        ∂c_k/∂a_i = 2·a_{k-i}   (if the index `k‑i` is inside the array,
                                 otherwise 0).
    This gives a cheap vector `g` that tells how much each coefficient
    contributes to the current worst peak.
  - We move a small amount of mass from a coefficient with **large**
    gradient (i.e. a large contributor) to one with **small** gradient.
    The move keeps the total sum unchanged, so the simplex constraint is
    satisfied automatically.
  - Moves are accepted if they improve the maximal convolution, otherwise
    they are accepted with a Metropolis probability using a linearly
    cooling temperature (`T` drops from `0.3·current_max` to `1e‑6`).
  - Occasionally a random jitter is injected to avoid stagnation.

* After each run we keep the best sequence seen.  When the global time
  budget is almost exhausted we run a short **projected sub‑gradient**
  descent (the same routine used in the previous baseline) to polish the
  final answer.

The whole search respects the 1000 s budget, prints progress, and finally
returns the best sequence found as a plain Python list.

# Fast full linear convolution using real FFT (same as the baseline)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)  # remove tiny negatives from rounding
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex { w ≥ 0, Σ w = s }."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1            # zero‑based
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Targeted annealing step (core of the new search)
# ----------------------------------------------------------------------
def _targeted_anneal(
    a0: np.ndarray,
    deadline: float,
    *,
    temp_factor: float = 0.30,
    min_temp: float = 1e-6,
    move_fraction: float = 0.20,
    jitter_prob: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that always moves mass away from indices that
    contribute most to the current maximal convolution peak.
    Returns the best sequence encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # exact objective = max convolution (because sum = sqrt(2n))
    conv = _full_convolution_fft(a, a)
    cur_val = conv.max()
    best_val = cur_val

    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1

        # recompute convolution and the index of the current worst peak
        conv = _full_convolution_fft(a, a)
        cur_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of the peak at position k
        grad = np.zeros_like(a)
        # i runs over indices that actually appear in the sum for conv_k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # choose a donor index (large gradient, positive mass)
        positive = np.where(a > 1e-12)[0]
        if positive.size == 0:
            break
        donor = positive[np.argmax(grad[positive])]

        # choose a receiver index (small gradient – ideally the least harmful)
        receiver = np.argmin(grad)
        if receiver == donor:
            # fall back to a random different index
            receiver = np.random.choice(np.delete(np.arange(n), donor))

        # amount of mass to transfer – limited to a fraction of the donor
        delta = min(a[donor] * move_fraction, 1e-3 * np.sqrt(2.0 * n))

        # propose new sequence
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # evaluate new max convolution
        new_val = _full_convolution_fft(a_new, a_new).max()

        # temperature schedule (linear cooling)
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(min_temp, cur_val * temp_factor * (1.0 - frac))

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(
                    f"[{time.time():.2f}] anneal it={it} new best={best_val:.6f}"
                )
                # early exit if we already beat the target
                if best_val < 1.503:
                    break

        # occasional random jitter to shake the system
        if np.random.rand() < jitter_prob:
            i = np.random.randint(n)
            if a[i] > 1e-12:
                eps = a[i] * np.random.rand() * 0.05
                j = np.random.randint(n)
                a[i] -= eps
                a[j] += eps
                # keep sum exactly constant – no projection needed

        # safety net: if we have stagnated for a while, re‑normalize to the simplex
        if it % 2000 == 0:
            a = _project_onto_simplex(a, np.sqrt(2.0 * n))

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max, for final polishing)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on max(conv(a, a)) with simplex projection.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    step = 0.1 * target_sum / n
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # projected gradient step
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        cand_val = _full_convolution_fft(a_candidate, a_candidate).max()

        if cand_val < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate.copy()
            best_val = cand_val
            step *= 1.02
        else:
            step *= 0.70

        if it % 500 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it} step={step:.3e} best={best_val:.6f}"
            )
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid random‑restart + targeted annealing search.
    Returns a list of non‑negative heights (the step‑function) that
    minimizes the evaluation function within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Warm‑start from the historic construction if available.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_val = float(score0)
        print(f"[{time.time():.2f}] warm‑start score={best_val:.6f} (n={n0})")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many independent annealing runs with random lengths.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate ~8‑12 s per individual run (or less when we are close to the end)
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = time.time() + slice_time

        # pick a problem size (more runs with medium lengths)
        if remaining > 400:
            n = np.random.randint(1600, 2500 + 1)
        elif remaining > 180:
            n = np.random.randint(1300, 2300 + 1)
        else:
            n = np.random.randint(1000, 1900 + 1)

        target_sum = np.sqrt(2.0 * n)

        # ----------------------------------------------------------
        # Initialise a – either random Dirichlet or a perturbed best.
        # ----------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.35:
            # reuse current global best (trim / pad)
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                a0 = np.concatenate([best_seq, pad])
        else:
            # random Dirichlet (uniform on the simplex)
            a0 = np.random.rand(n)
        a0 = a0 / a0.sum() * target_sum

        # ----------------------------------------------------------
        # Run the targeted annealing optimizer.
        # ----------------------------------------------------------
        seq_local, val_local = _targeted_anneal(a0, run_deadline)

        # evaluate with the official evaluator (identical to max conv)
        exact_val = evaluate_sequence(seq_local.tolist())

        if exact_val < best_val:
            best_val = float(exact_val)
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best={best_val:.6f} (n={n})"
            )
            if best_val < 1.503:          # early exit if we already beat the target
                break

    # --------------------------------------------------------------
    # Phase 2 – polished sub‑gradient descent on the best sequence.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 10.0:
        n = best_seq.shape[0]
        sub_end = time.time() + min(30.0, time_left * 0.4)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), sub_end)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final tiny hill‑climb (fallback if we still have time).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        # Re‑use the previous hillclimb code (it is lightweight)
        # but with a very short horizon.
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)

        def _tiny_hillclimb(a0, sub_dead):
            a = a0.copy()
            best_a = a.copy()
            best_v = _full_convolution_fft(a, a).max()
            start = time.time()
            it = 0
            while time.time() < sub_dead:
                it += 1
                # pick two indices and transfer a tiny fraction
                i, j = np.random.randint(0, n, size=2)
                if i == j or a[i] <= 0:
                    continue
                delta = a[i] * np.random.rand() * 0.05
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                v_new = _full_convolution_fft(a_new, a_new).max()
                if v_new < best_v:
                    best_v = v_new
                    best_a = a_new.copy()
                    a = a_new
                elif np.random.rand() < 0.001:   # very low‑probability uphill moves
                    a = a_new
                if it % 2000 == 0:
                    a = _project_onto_simplex(a, target_sum)
                if best_v < 1.503:
                    break
            return best_a, best_v

        hill_end = time.time() + time_left
        seq_hill, val_hill = _tiny_hillclimb(best_seq.copy(), hill_end)
        if val_hill < best_val:
            best_val = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be used).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1800
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_val:.6f}")

    # Return as plain Python list (required by the outer framework)
    print(f"[{time.time():.2f}] final best value = {best_val:.6f}")
    return best_seq.tolist()
