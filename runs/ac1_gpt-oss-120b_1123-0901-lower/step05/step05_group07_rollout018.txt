"""
Propose a non‑negative step function (probability vector) that minimises

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of three layers:

1. **Low‑dimensional families** –  Quick scans of simple parametric shapes
   (uniform, power‑law, exponential and a 2‑exponential mixture).  These
   provide good warm‑starts that are often much better than a random start.

2. **High‑dimensional sub‑gradient optimisation** –  The exact sub‑gradient of
   the (non‑smooth) max is used.  After the soft‑max re‑parameterisation the
   method performs an Adam‑like mirror‑descent on the simplex (exponentiated
   gradient).  No temperature‑annealing is needed because the true max is
   used directly.

3. **Local greedy refinement** –  Repeated random pairwise mass transfers
   (tiny ℇ‑moves) accept only improvements.  This cheap “mass‑shuffling”
   often squeezes a little more out of a converged solution.

The whole routine respects a global 1000 s wall‑time limit, stops early

far.  The implementation uses only NumPy and FFT‑based convolutions for

import time

# ----------------------------------------------------------------------
# Helper functions (FFT based convolution, cross‑correlation)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear (full) convolution a*a using FFT.  Length = 2·len(a)‑1."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

                # ---- Sub‑gradient of max(conv) --------------------
                # indicator of active (binding) shifts
                active = np.where(np.abs(conv - cur_max) <= 1e-12)[0]
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

                # gradient wrt a (exact, no smoothing)
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # back‑prop through soft‑max (mirror‑descent)
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ---- Adam update ---------------------------------
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # Simple learning‑rate decay
                if t % 5000 == 0:
                    lr *= 0.7

                # Stagnation handling – small random jitter
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 4e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t}] shake (stagnation)")

            # ----------------------------------------------------
            # Fine‑tune with a very low learning‑rate (true max still active)
            # ----------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
                a = best_seq.copy()
                x = np.log(a + 1e-15)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.015
                fine_iters = 12000
                for ft in range(1, fine_iters + 1):
                    if time.time() > deadline:
                        break
                    mx = np.max(x)
                    exp_x = np.exp(x - mx)
                    a = exp_x / exp_x.sum()
                    conv = full_convolution(a, fft_len=fft_len_conv)
                    cur_max = conv.max()
                    cur_val = 2.0 * n * cur_max
                    if cur_val < best_val - 1e-12:
                        best_val, best_seq = cur_val, a.copy()
                        print(f"[fine tune] improved to {cur_val:.6f}")

                    active = np.where(np.abs(conv - cur_max) <= 1e-12)[0]
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.size
                    grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)
                    inner = np.dot(a, grad_a)
                    grad_x = a * (grad_a - inner)

                    m = beta1 * m + (1 - beta1) * grad_x
                    v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
                    m_hat = m / (1 - beta1 ** ft)
                    v_hat = v / (1 - beta2 ** ft)
                    x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----------------------------------------------------
            # 3. Local greedy refinement (tiny mass shuffles)
            # ----------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small random fraction of a[i] to a[j]
                    eps_move = min(a_local[i], 0.008) * (np.random.rand() * 0.6 + 0.2)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val, best_seq = val_new, a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (objective = 2) – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
