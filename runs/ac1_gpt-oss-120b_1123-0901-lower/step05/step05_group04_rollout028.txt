```
def project_simplex(v):
    # ensure non-negative after projection onto simplex sum=1
    # v can be any real array
    # algorithm: shift so that sum(v) = 1 and all entries >=0.
    # reference: Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def local_search(a, n, deadline, rng):
    best = a.copy()
    best_score = compute_score(a)
    cur = a.copy()
    cur_score = best_score
    step_factor = 0.2 # up to 20% of source mass moved each time; can adapt
    while time.time() < deadline:
        # choose donor i with cur[i] > 0
        i = rng.choice(np.where(cur > 1e-12)[0])
        if cur[i] <= 0:
            continue
        # choose recipient j diff
        j = rng.integers(0, n-1)
        if i == j: continue
        delta = cur[i] * rng.uniform(0, step_factor)
        # ensure delta small
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # compute score
        new_score = compute_score(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score
```

"""
Proposes a step‑function by random restarts combined with a simple
projected‑gradient method that directly minimises the true max‑convolution.
The algorithm works on probability vectors (sum = 1).  Starting from
various random shapes (power‑law, Dirichlet, exponential) a short
gradient descent on the current max‑convolution index is performed,
projected back onto the simplex after each step.  The best candidate
found within the time budget is returned.
"""

import time
import numpy as np

def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # remove tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    # Duchi et al. (2008) algorithm
    # works for any real vector
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # v is all zeros → return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray) -> float:
    """Objective value 2 * n * max(conv(a,a)). a is assumed to sum to 1."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

def _gradient_step(a: np.ndarray, n: int, step: float) -> np.ndarray:
    """One projected‑gradient step on the current maximal‑convolution index."""
    conv = _full_convolution_fft(a, a)
    k = int(conv.argmax())
    # gradient of conv[k] w.r.t. a is g_i = 2 * a_{k-i} (if index in range)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k - idx]
    # take a descent step and re‑project onto the simplex
    a_next = a - step * g
    a_next = np.maximum(a_next, 0.0)
    a_next = _project_simplex(a_next)
    return a_next

def _optimise_pg(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple projected‑gradient optimisation starting from a0.
    Returns (best_vector, best_score) found before ``deadline``.
    """
    n = a0.shape[0]
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    it = 0
    # initial step size – a small fraction of the simplex diameter
    step = 0.5
    start = time.time()
    while time.time() < deadline:
        # linearly decay the step size
        elapsed = time.time() - start
        total = deadline - start
        step = max(0.02, step * (1.0 - elapsed / total))
        a = _gradient_step(a, n, step)
        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] pg iter={it} new best={best_val:.8f}")
        # occasional random jitter to escape plateaux
        if it % 400 == 0 and rng.random() < 0.1:
            jitter = rng.uniform(0.0, 1e-3, size=n)
            a = _project_simplex(a + jitter)
        it += 1
    return best, best_val

def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random probability vector of length n.
    Different families (power‑law, exponential, mixed, uniform) are mixed.
    """
    typ = rng.choice(['power', 'exp', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)
    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        x = idx ** (-p)
    elif typ == 'exp':
        tau = rng.uniform(0.2, 0.9) * n
        x = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, max(2, n - 1))
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        x = np.concatenate([left, right])
    else:   # uniform
        x = np.ones_like(idx)
    # add a tiny jitter and normalise to sum = 1
    x += rng.random(n) * 1e-6
    x = np.maximum(x, 1e-12)
    x /= x.sum()
    return x

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises

    Strategy

    * many short projected‑gradient runs from diverse random starts,
      each run works directly on the true max‑convolution (no surrogate).
    * the best candidate is kept globally; occasional jitter improves
      exploration.
    * when time permits a final refinement on the incumbent is executed.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # pick a problem size – larger early on, smaller later
        if remaining > 300:
            n = int(rng.integers(1500, 3500 + 1))
        elif remaining > 120:
            n = int(rng.integers(1000, 2500 + 1))
        else:
            n = int(rng.integers(600, 1800 + 1))

        # initialise a random probability vector
        a0 = _random_initial(n, rng)

        # allocate a time slice for this optimisation run
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = time.time() + slice_time

        # run the projected‑gradient optimiser
        a_opt, score_opt = _optimise_pg(a0, run_deadline, rng)

        # keep the global best
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            if best_score < 1.45:          # early exit if we already beat the target comfortably
                break

    # ------------------------------------------------------------------

    if best_seq is not None and (deadline - time.time()) > 6.0:

    # Return a plain Python list (the harness will call evaluate_sequence)
    return best_seq.tolist()

"""
Proposes a step‑function that minimises the objective

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search works on probability vectors (sum = 1) – the scaling of the
vector does not affect the objective because the numerator and the
denominator both scale quadratically.

Strategy

  mixed, uniform).  The initial vector is turned into a probability
  vector by normalising its sum.
* each run performs a very simple projected‑gradient descent that
  directly minimises the *true* max‑convolution:
      – compute the convolution,
      – locate the index `k` attaining the maximum,
      – the gradient of `conv[k]` w.r.t. the vector is `g_i = 2*a_{k‑i}`,
      – take a step `a ← a – η g` and project back onto the simplex.
  The step size decays linearly, a small random jitter is injected
  periodically to escape plateaux.
* the best candidate seen so far is kept globally; when time permits a
  final refinement on the incumbent is carried out.
* the routine respects the given time budget (default 1000 s) and returns

import numpy as np


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT (full length)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho_arr = np.where(u - cssv / ind > 0)[0]
    if rho_arr.size == 0:                     # pathological – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Return the objective value 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)

def _gradient_step(a: np.ndarray, n: int, step: float) -> np.ndarray:

    One projected‑gradient step on the current maximal‑convolution index.
    Returns the updated probability vector.

    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)

    g[idx] = 2.0 * a[k - idx]          # ∂conv[k] / ∂a_i = 2·a_{k‑i}
    a_next = a - step * g
    a_next = np.maximum(a_next, 0.0)
    return _project_simplex(a_next)
