def _simplex_project(v):
    # Project onto {x >=0, sum x = 1}
    # Return projected vector.
    # v is numpy array.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

- `_alternating_extremes(vals)` sorts descending and fills positions alternating ends.

```
def _alternating_extremes(vals):
    sorted_vals = np.sort(vals)[::-1]
    n = sorted_vals.size
    out = np.empty(n, dtype=sorted_vals.dtype)
    left = 0
    right = n - 1
    for i, val in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out
```

"""
Propose a low‑value step function.

Strategy
--------
1.  Build a large portfolio of diverse seeds:
    * uniform, Dirichlet, power‑law, Gaussian‑edge, triangular.
    * Each seed is turned into three orderings:
        – original,
        – anti‑pair (large values left, small values right),
        – alternating extremes (largest → leftmost, second largest → rightmost,
          third → next‑left, …).
    * Warm‑start from the global variable ``height_sequence_1`` if present.

2.  Keep the best few seeds (by the exact objective) and optimise them.
    * First a smooth‑max Adam phase (log‑sum‑exp surrogate).
    * Then a long‐run exact‑max projected sub‑gradient phase.
      The sub‑gradient step uses Euclidean projection onto the simplex
      (mirrored‑descent style) which works well for high‑dimensional
      simplex problems.

3.  A cheap stochastic hill‑climber (swap / mass‑transfer moves) is run
    on the current best solution to exploit any remaining time.

All heavy work is done with FFT‑based linear convolution, so a single
iteration costs ~O(n log n).  The routine adapts the time budget
dynamically and always returns a valid non‑negative list before the
deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, without safety checks) -----------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple deterministic / random generators -----------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    v = i ** (-p)
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return v


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    v = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return v


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _binary_two_level_candidate(n: int, frac: float = 0.2) -> np.ndarray:
    """
    Simple two‑level vector: a fraction `frac` of entries receive a large
    mass, the rest a small mass.  The positions are drawn uniformly at
    random.
    """
    k = max(1, int(frac * n))
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = 1.0
    a += 1e-6  # avoid zeros which can stall the optimiser
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Ordering helpers ------------------------------------------------------
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair) ordering."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _alternating_extremes(vals: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternating from the left and right ends:
    largest → position 0, second largest → position n‑1,
    third largest → position 1, fourth → n‑2, …
    """
    sorted_vals = np.sort(vals)[::-1]          # descending
    n = sorted_vals.size
    out = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex ----------
def _simplex_project(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1} (Euclidean)."""
    # based on the algorithm by Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are ≤0 after projection – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex (explicit simplex via softmax)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:07d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp surrogate)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # move to the log‑domain (mirrored‑descent style)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_proj_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.6,
    eval_every: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with Euclidean projection onto the simplex.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a_sum

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # periodic exact evaluation
        if it % eval_every == 0:
            val = 2.0 * n * max_c / (a.sum() ** 2)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Subproj it={it:07d}] new best {best_val:.6f}")

        # active set for the current max
        active = np.isclose(conv, max_c, atol=1e-12)
        nd = active.sum()
        w = np.zeros_like(conv)
        if nd > 0:
            w[active] = 1.0 / nd

        # sub‑gradient of the max‑conv w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule (diminishing)
        eta = eta0 / np.sqrt(it + 1)

        # take a step and project back onto the simplex
        a = _simplex_project(a - eta * grad_a)

        # safeguard against numerical issues
        if np.isnan(a).any() or a.sum() == 0:
            a = np.random.dirichlet(np.ones(n))

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: swap two entries or transfer a fraction of mass.
    Accept improvements always; otherwise accept with a linearly decreasing
    temperature.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap two indices
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                               # transfer a fraction of mass
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:07d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Warm‑start helper -----------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²).

    The routine builds a large pool of seeds (different analytic shapes,
    random Dirichlet draws, two‑level binary vectors) and a few
    permutations of each (original, anti‑pair, alternating extremes).
    The best seeds are refined with a smooth‑max Adam phase followed by
    a long projected sub‑gradient phase, and finally a cheap stochastic
    hill‑climber is applied if time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem dimensions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a diverse candidate pool
    candidates = []

    # Warm start (if present) – three orderings
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        candidates.append(_anti_pair_arrangement(warm.copy()))
        candidates.append(_alternating_extremes(warm.copy()))
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Baseline uniform
    candidates.append(_uniform_candidate(n))

    # Random Dirichlet seeds with many concentration parameters
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        d = _random_dirichlet_candidate(n, alpha)
        candidates.append(d)
        candidates.append(_anti_pair_arrangement(d))
        candidates.append(_alternating_extremes(d))

    # Power‑law families
    for p in (0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0):
        pl = _powerlaw_candidate(n, p)
        candidates.append(pl)
        candidates.append(_anti_pair_arrangement(pl))
        candidates.append(_alternating_extremes(pl))

    # Gaussian‑edge families
    for gamma in (0.1, 0.2, 0.5, 0.8, 1.2):
        ge = _gaussian_edges_candidate(n, gamma)
        candidates.append(ge)
        candidates.append(_anti_pair_arrangement(ge))
        candidates.append(_alternating_extremes(ge))

    # Triangle (tent) shape
    tri = _triangle_candidate(n)
    candidates.append(tri)
    candidates.append(_anti_pair_arrangement(tri))
    candidates.append(_alternating_extremes(tri))

    # Two‑level binary patterns (different densities)
    for frac in (0.05, 0.1, 0.15, 0.20, 0.30):
        bl = _binary_two_level_candidate(n, frac)
        candidates.append(bl)
        candidates.append(_anti_pair_arrangement(bl))
        candidates.append(_alternating_extremes(bl))

    # --------------------------------------------------------------
    # 2) Quick evaluation; retain the best few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 20 == 0:
            print(f"[Init cand {idx:04d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Continuous optimisation on the top candidates
    # Allocate roughly half the remaining budget to the optimisation stages
    remaining = deadline - time.time()
    per_candidate_budget = remaining * 0.5 / top_k

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        # ---------- Adam phase (smooth‑max) ----------
        adam_dead = time.time() + per_candidate_budget * 0.45
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=500_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid} Adam] improved to {val_adam:.6f}")

        # ---------- Sub‑gradient (projected) phase ----------
        sub_dead = time.time() + per_candidate_budget * 0.55
        a_sub, val_sub = _subgradient_proj_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=1_200_000,
            eta0=0.6,
            eval_every=3000,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {cid} Subproj] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Local stochastic hill‑climbing (if time left)
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 5) Final safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # Return as plain Python list as required by the evaluator
    return best_seq.tolist()
