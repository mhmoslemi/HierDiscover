def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n=4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)
    
    # Build candidate pool
    candidates = []
    # add warm start if any
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
    # uniform
    candidates.append(_uniform_candidate(n))
    # random Dirichlet many samples
    for alpha in (0.1,0.3,0.5,0.7,1.,2.,5.):
        for _ in range(3):
            candidates.append(_random_dirichlet_candidate(n,alpha))
    # anti-pair candidate
    for alpha in (0.1,0.3,0.5,0.7,1.,2.):
        candidates.append(_anti_pair_candidate(n,alpha))
    # powerlaw
    for p in (0.2, 0.5, 0.8, 1.,1.5,2.):
        candidates.append(_powerlaw_candidate(n,p))
    # ushape
    for p in (0.2,0.5,1.,2.):
        candidates.append(_ushape_candidate(n,p))
    # gaussian edges
    for gamma in (0.2,0.5,1.,2.):
        candidates.append(_gaussian_edges_candidate(n,gamma))
    # triangle
    candidates.append(_triangle_candidate(n))
    # sine shape
    x = np.arange(n)/ (n-1)
    sine = (np.sin(np.pi * x) + 0.1)   # add small offset
    sine = np.maximum(sine,0)
    sine = sine / sine.sum()
    candidates.append(sine)
    # random permutations
    base = _anti_pair_candidate(n, alpha=0.7)
    for _ in range(5):
        perm = np.random.permutation(n)
        candidates.append(base[perm])
    
    # Quick evaluation
    evals = []
    for idx,cand in enumerate(candidates):
        cand = np.maximum(cand,0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evals.append((val,cand))
        if idx % 20 ==0:
            print(f"[Init {idx:03d}] val {val:.6f}")
    evals.sort(key=lambda x:x[0])
    top_k = 5
    best_seq = evals[0][1].copy()
    best_val = evals[0][0]
    print(f"Initial best {best_val:.6f}")

    # Continue refinement
    remaining = deadline - time.time()
    # allocate half to Adam+subgrad across top_k
    per = remaining * 0.5 / top_k
    for cid in range(top_k):
        cand = evals[cid][1]
        # Adam
        adam_dead = time.time() + per*0.5
        a_adam,val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=800_000,
            lr=0.05,
            lam_start=0.2,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[cand {cid} Adam] new best {best_val:.6f}")
        # subgradient
        sub_dead = time.time() + per*0.5
        a_sub,val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=800_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[cand {cid} Subgrad] new best {best_val:.6f}")
        if time.time() > deadline - 30:
            break

    # Local search
    if time.time() < deadline - 5:
        local_dead = deadline - 2
        a_loc,val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val,best_seq = val_loc, a_loc.copy()
            print(f"[Local] new best {best_val:.6f}")

    # Optional upsampling
    if time.time() < deadline - 30:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2*n_big-1)
        corr_fft_len_big = _next_pow2(3*n_big-2)
        # interpolate
        x_old = np.linspace(0,1,n,endpoint=False)
        x_new = np.linspace(0,1,n_big,endpoint=False)
        a_big = np.interp(x_new,x_old,best_seq)
        a_big = np.maximum(a_big,0)
        a_big = a_big / a_big.sum()
        big_dead = min(deadline - 5, time.time() + 180)
        a_big_adam,val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_dead,
            max_iters=400_000,
            lr=0.05,
            lam_start=0.2,
            lam_factor=1.07,
        )
        a_big_sub,val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=400_000,
            eta0=0.7,
        )
        if val_big_sub < best_val:
            best_val,best_seq = val_big_sub, a_big_sub.copy()
            print(f"[Big] new best {best_val:.6f}")

    # Final safeguard
    best_seq = np.clip(best_seq,0,1000)
    print(f"Final best value {best_val:.6f}, length {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a step function that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

using a multi‑restart, long‑horizon gradient phase followed by an
aggressive stochastic hill‑climb.  The pipeline is:

1. Build a rich pool of warm‑starts (uniform, Dirichlet samples,
   anti‑pair arrangements, power‑law, Gaussian‑edge, sine‑shaped…).
   Keep the best candidate (by the exact objective).

2. Run two independent optimisation “runs”.  Each run gets a fixed
   fraction of the remaining time‑budget and consists of:
   * Adam on a smooth‑max (log‑sum‑exp) surrogate.
   * Exact‑max sub‑gradient descent (exponentiated‑gradient).
   The two runs start from different points (the best pool element
   and a fresh anti‑pair sample) to increase diversity.

3. With the remaining time perform a large‑scale local random search
   (swap or transfer a fraction of mass) that always accepts improvements
   and otherwise uses a tiny simulated‑annealing temperature.

4. If time permits, linearly up‑sample the best sequence to 8192 points
   and repeat a short Adam+sub‑gradient refinement.

All steps respect the security checks of ``evaluate_sequence``.  The
function returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety checks) ------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start helper -----------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm‑start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Generators ------------------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair) ordering."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _sine_candidate(n: int) -> np.ndarray:
    """Non‑negative sine‑shaped bump."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    shape = np.sin(np.pi * x) + 0.1   # offset to avoid zeros
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lr: float = 0.05,
    lam_start: float = 0.2,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best exact objective seen and the corresponding vector.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices attaining the max (active set)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (stay on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
      – swap two entries OR
      – transfer a fraction of mass.
    Accept only improving moves (or very rare uphill moves when temperature > 0).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                                          # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            # up to 60 % of the mass of i moves to j
            frac = np.random.rand() * 0.6
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # tiny annealing to escape flat plateaus
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of warm‑starts
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())

    candidates.append(_uniform_candidate(n))

    # Dirichlet samples with many concentration parameters
    for alpha in (0.1, 0.3, 0.5, 0.7, 1.0, 2.0, 5.0):
        for _ in range(2):
            candidates.append(_random_dirichlet_candidate(n, alpha))

    # Anti‑pair Dirichlet
    for alpha in (0.1, 0.3, 0.5, 0.7, 1.0, 2.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # Deterministic analytical shapes
    for p in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0):
        candidates.append(_powerlaw_candidate(n, p))
        candidates.append(_ushape_candidate(n, p))
    for gamma in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))
    candidates.append(_sine_candidate(n))

    # Random permutations of a strong anti‑pair starter
    base_ap = _anti_pair_candidate(n, alpha=0.7)
    for _ in range(5):
        candidates.append(base_ap[np.random.permutation(n)])

    # ------------------------------------------------------------------
    # 2) Quick eval – keep the overall best
    best_seq = None
    best_val = np.inf

    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
        if idx % 20 == 0:
            print(f"[Init cand {idx:03d}] value {val:.6f}")

    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Two independent long‑horizon optimisation runs
    #    – they share the overall time‑budget
    runs = [
        ("run_best", best_seq.copy()),                         # start from pool best
        ("run_fresh", _anti_pair_candidate(n, alpha=0.5)),     # fresh anti‑pair start
    ]

    for run_name, start_vec in runs:
        if time.time() > deadline - 30.0:
            break
        # allocate a slice of the remaining budget
        now = time.time()
        remaining = deadline - now
        # 40 % for Adam, 40 % for sub‑gradient, 20 % reserved for later steps
        adam_dead = now + 0.4 * remaining
        sub_dead = adam_dead + 0.4 * remaining

        # Adam phase
        a_adam, val_adam = _adam_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=1_000_000,
            lr=0.05,
            lam_start=0.2,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[{run_name} Adam] improved to {best_val:.6f}")

        # Sub‑gradient phase
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=1_000_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[{run_name} Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Aggressive local stochastic hill‑climbing
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=400_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (8192) if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        big_adam_dead = min(deadline - 5.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_dead,
            max_iters=500_000,
            lr=0.05,
            lam_start=0.2,
            lam_factor=1.07,
        )

        # short sub‑gradient refinement on the large grid
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=500_000,
            eta0=0.7,
        )
        if val_big_sub < best_val:
            best_val, best_seq = val_big_sub, a_big_sub.copy()
            print(f"[Big] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
