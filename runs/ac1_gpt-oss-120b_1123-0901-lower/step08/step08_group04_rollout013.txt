def _targeted_sa(a0, time_limit, start_temp=0.02, max_frac=0.2):
    a = a0.copy()
    n = len(a)
    conv_len = 2*n -1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)
    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val
    start = time.time()
    while time.time() < time_limit:
        max_c = conv.max()
        # compute active mask
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        # gradient
        grad = 2.0 * _cross_correlation(w, a, _next_pow2(3*n-2))
        # source indices: top fraction of grad (excluding zeros)
        # choose random from top_k
        topk = int(max(1, n * 0.01))
        sources = np.argpartition(-grad, topk-1)[:topk]
        dests = np.argpartition(grad, topk-1)[:topk]
        i = np.random.choice(sources)
        if a[i] < 1e-12:
            continue
        j = np.random.choice(dests)
        # move small fraction
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)
        new_val = conv.max()
        # linear temperature schedule
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))
        diff = new_val - cur_val
        if diff <= 0 or (temp > 0 and np.random.rand() < np.exp(-diff / temp)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)
    return best_a, best_val
```

def _enhanced_sa(a0, deadline, start_temp=0.1, start_frac=0.5, end_frac=0.01):
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2*n-1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    # compute total time of SA
    total = deadline - time.time()
    while time.time() < deadline:
        remaining = deadline - time.time()
        frac = end_frac + (start_frac - end_frac) * (remaining / total)
        # temperature schedule sqrt
        temp = start_temp * np.sqrt(remaining / total)

        # choose source index weighted by contribution to max
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        # pick random active shift
        shifts = np.where(active)[0]
        # if many active shifts choose random
        p = np.random.choice(shifts)

        # compute contributions for indices i where a[i] > 0 and p-i in range
        i_low = max(0, p - (n-1))
        i_high = min(p, n-1)
        idx_range = np.arange(i_low, i_high+1)
        partners = p - idx_range
        # contributions = a[idx_range] * a[partners]
        contributions = a[idx_range] * a[partners]
        # get source idx weighted by contributions (higher)
        if contributions.sum() > 0:
            src = np.random.choice(idx_range, p=contributions/contributions.sum())
        else:
            src = np.random.randint(n)
        # ensure a[src] > 0
        if a[src] < 1e-12:
            # choose random non-zero
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src = np.random.choice(nonzero)

        # choose destination index random, but prefer low contribution:
        # compute destination contributions (for same active shift)
        # we can compute dest_contrib = a[dest] * a[p-dest] if valid else 0
        # choose dest with probability inversely proportional.
        # compute probs:
        dest_contrib = np.zeros(n)
        # only compute for valid dest indices where partner in range
        # partner = p - dest
        dest_valid = (p - np.arange(n) >= 0) & (p - np.arange(n) < n)
        dest_idx = np.arange(n)[dest_valid]
        dest_contrib[dest_idx] = a[dest_idx] * a[p - dest_idx]
        # invert contributions
        inv = 1.0 / (dest_contrib + 1e-12)
        dest = np.random.choice(n, p=inv / inv.sum())

        # transfer small fraction of mass from src to dest
        frac_move = np.random.rand() * frac
        delta = a[src] * frac_move
        if delta <= 0:
            continue

        _apply_delta(a, conv, src, -delta)
        _apply_delta(a, conv, dest, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        accept = diff <= 0 or (temp > 0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Enhanced SA] new best {best_val:.6f}")
        else:
            # revert
            _apply_delta(a, conv, dest, -delta)
            _apply_delta(a, conv, src, delta)

    return best_a, best_val
```

"""
Propose a step‑function with a low value of
    F(a)=2·n·max(a∗a)/(sum(a)²)

The method builds a warm start (the previous best sequence if available),
normalises it to a probability vector and then performs a long, targeted
simulated‑annealing search.  The move generator is biased towards indices
that actually contribute to the current maximum of the convolution, which
helps to push the peak down faster than pure random moves.  The temperature
and the maximum transferable fraction are faded out smoothly with the
remaining time budget.

All helpers are defined at the top level; the only public entry point is
`propose_candidate`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Incremental convolution update (used by SA) ---------------------------
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    # update all entries that involve a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


# ----------------------------------------------------------------------
# Exact objective (used only for logging) -------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start handling ----------------------------------------------------
def _warm_start(n: int):
    """If a previous solution exists, reshape it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


# ----------------------------------------------------------------------
# Simple candidate generators --------------------------------------------
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange a sorted vector so that large values are opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Enhanced simulated‑annealing -------------------------------------------
def _enhanced_sa(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing that focuses moves on indices
    that actually participate in the current maximal convolution value.
    The temperature and the maximal transferred fraction both decay
    smoothly with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()                     # this is max_c, not the final objective
    cur_val = best_val

    start = time.time()
    total = deadline - start
    # annealing schedule parameters
    start_temp = 0.20
    max_frac_start = 0.40
    max_frac_end = 0.02

    it = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        fraction = remaining / total

        # temperature (sqrt schedule) and decreasing maximal move size
        temp = start_temp * (fraction ** 0.5)
        max_frac = max_frac_end + (max_frac_start - max_frac_end) * fraction

        # locate a (random) shift that attains the current maximum
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        active_idxs = np.where(active)[0]
        p = np.random.choice(active_idxs)    # shift index with maximal value

        # source index is taken from the interval that actually influences
        # conv[p]; we bias the choice by the contribution a[i]*a[p-i].
        low = max(0, p - (n - 1))
        high = min(p, n - 1)
        idx_range = np.arange(low, high + 1, dtype=np.int64)
        partner = p - idx_range
        w = a[idx_range] * a[partner]        # contribution of each i to conv[p]

        if w.sum() > 0:
            # sample proportionally to the contribution
            cum = np.cumsum(w)
            r = np.random.rand() * cum[-1]
            src = idx_range[np.searchsorted(cum, r)]
        else:
            # fallback: any index with non‑zero mass in the interval
            nz = idx_range[a[idx_range] > 1e-12]
            if nz.size == 0:
                src = np.random.randint(low, high + 1)
            else:
                src = np.random.choice(nz)

        if a[src] < 1e-14:
            # should not happen, but skip this move
            continue

        # pick a destination index uniformly (different from src)
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # amount transferred – a random fraction of the source mass
        delta = a[src] * (np.random.rand() * max_frac)
        if delta <= 0.0:
            continue

        # apply the move (two incremental updates)
        _apply_delta(a, conv, src, -delta)
        _apply_delta(a, conv, dst, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # acceptance rule – Metropolis
        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Enhanced SA it={it:07d}] new best max_c {best_val:.8f}")
        else:
            # revert the move
            _apply_delta(a, conv, dst, -delta)
            _apply_delta(a, conv, src, delta)

        it += 1
        if it % 50000 == 0:
            print(f"[Enhanced SA it={it:07d}] best max_c so far {best_val:.8f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    The routine builds a warm start (re‑using the previous best if
    available) and then spends the bulk of the time in an enhanced
    simulated‑annealing stage that focuses moves on the indices that
    contribute to the current convolution peak.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096

    # ------------------------------------------------------------------
    # 1) initialise from the previous best (if any) or a random anti‑pair
    warm = _warm_start(n)
    if warm is not None:
        seq = warm
        print("[Init] using warm start from `height_sequence_1`")
    else:
        seq = _anti_pair_candidate(n, alpha=0.7)
        print("[Init] generated fresh anti‑pair start")

    # normalise to a probability vector (sum = 1)
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total < 0.01:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= total

    # ------------------------------------------------------------------
    # 2) long enhanced simulated‑annealing
    sa_deadline = deadline - 5.0
    if time.time() < sa_deadline:
        print("[SA] starting enhanced simulated‑annealing")
        seq, max_c = _enhanced_sa(seq, sa_deadline)
        final_obj = 2.0 * n * max_c          # exact objective (since sum=1)
        print(f"[SA] finished – final max_c = {max_c:.8f}, objective = {final_obj:.6f}")
    else:
        final_obj = _objective(seq)
        print("[SA] not enough time left – skipping SA")

    # ------------------------------------------------------------------
    # 3) final safety clipping and normalisation
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= total

    print(f"Final reported objective {final_obj:.6f} (length {len(seq)})")
    return seq.tolist()
