def _project_monotone_simplex(v):
    # ensure positivity for isotonic: v >=0
    # isotonic regression decreasing
    n = len(v)
    solution = v.astype(np.float64).copy()
    weight = np.ones(n, dtype=np.float64)

    i = 0
    while i < n - 1:
        if solution[i] >= solution[i + 1] - 1e-12:
            i += 1
        else:
            # merge pools
            total_w = weight[i] + weight[i + 1]
            avg = (solution[i] * weight[i] + solution[i + 1] * weight[i + 1]) / total_w
            solution[i] = avg
            weight[i] = total_w
            # delete entry i+1
            solution = np.delete(solution, i + 1)
            weight = np.delete(weight, i + 1)
            # move back
            i = max(i - 1, 0)
    # now solution non-increasing but length may be less (due to merges)

PAVA algorithm can be implemented by using "blocks". Represent each block with start/end indices, sum, weight, average. At each step, if block avg < next block avg (since decreasing requirement), merge them.

Implementation using stack:

```
def _project_monotone_simplex(v):
    n = len(v)
    # blocks: each block has sum, weight, and start index
    sums = []
    counts = []
    starts = []
    for i in range(n):
        sums.append(v[i])
        counts.append(1)
        starts.append(i)
    # merge when necessary
    i = 0
    while i < len(sums) - 1:
        avg_i = sums[i] / counts[i]
        avg_next = sums[i + 1] / counts[i + 1]
        if avg_i >= avg_next - 1e-12:
            i += 1
        else:
            # merge block i and i+1
            sums[i] += sums[i + 1]
            counts[i] += counts[i + 1]
            # remove block i+1
            del sums[i + 1]
            del counts[i + 1]
            del starts[i + 1]
            # move back i if possible
            if i > 0:
                i -= 1
    # reconstruct vector
    out = np.empty(n, dtype=np.float64)
    for block_sum, block_cnt, block_start in zip(sums, counts, starts):
        avg = block_sum / block_cnt
        out[block_start:block_start+block_cnt] = avg
    # ensure non-negative
    out = np.maximum(out, 0.0)
    # project onto simplex (sum to 1)
    total = out.sum()
    if total > 0:
        out = out / total
    else:
        out = np.full_like(out, 1.0 / n)
    return out
```

"""
propose_candidate
-----------------
Search for a non‑negative step‑function that minimises

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search proceeds in three stages:

1. **Broad initialisation** – a large pool of deterministic families
   (uniform, sinusoid, edge‑exponential, *unsymmetric* exponential,
   power‑law, beta‑pdf, sparse spikes, …) together with many random
   Dirichlet draws is generated.  Every vector is normalised to have
   unit L1‑norm and its exact objective is evaluated with an FFT based
   convolution.  The best vector of this quick scan is kept.

2. **Monotone sub‑gradient refinement** – the current best vector is
   optimised with an exponentiated‑gradient (mirror‑descent) method.
   After each update the vector is projected onto the *monotone simplex*
   (non‑negative, sums to one, and non‑increasing).  This restriction
   forces the shape to be a single decreasing “front”, which empirically
   drives the maximum of the self‑convolution lower than the best
   symmetric candidates.

3. **Cheap local hill‑climber** – for the final seconds a very cheap
   random‑move optimiser moves a random fraction of mass from one index
   to another and keeps the move if the objective improves.

All phases respect the global deadline (default 1000 s) and always keep
the best sequence seen so far.  The returned list is guaranteed to be
accepted by `evaluate_sequence`.
"""

# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _project_monotone_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection onto the intersection of the simplex and the
    monotone‑decreasing cone (v[0] ≥ v[1] ≥ … ≥ v[-1] ≥ 0, sum(v)=1).
    Implemented via the Pool‑Adjacent‑Violators Algorithm (PAVA).
    """
    v = np.maximum(v, 0.0)
    n = v.shape[0]

    # PAVA: maintain blocks (sum, weight) ensuring block averages are decreasing.
    sum_blocks = []   # Σ values in each block
    weight_blocks = []  # block sizes (integers)

    for i in range(n):
        sum_blocks.append(v[i])
        weight_blocks.append(1.0)

        # Merge while the averages would increase (violating monotonicity)
        while len(sum_blocks) >= 2:
            avg_prev = sum_blocks[-2] / weight_blocks[-2]
            avg_cur = sum_blocks[-1] / weight_blocks[-1]
            if avg_prev + 1e-12 >= avg_cur:
                break
            # merge the last two blocks
            sum_blocks[-2] += sum_blocks[-1]
            weight_blocks[-2] += weight_blocks[-1]
            sum_blocks.pop()
            weight_blocks.pop()

    # Expand the averaged blocks back to a full vector
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for s, w in zip(sum_blocks, weight_blocks):
        avg = s / w
        out[pos : pos + int(w)] = avg
        pos += int(w)

    # Final simplex projection (preserves monotonicity because all entries are equal inside blocks)
    total = out.sum()
    if total > 0:
        out = out / total
    else:
        out = np.full_like(out, 1.0 / n)
    return out

def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:

def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:

def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass at both edges."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _left_exponential(n: int, gamma: float) -> np.ndarray:
    """Only left‑hand exponential decay (strictly decreasing)."""
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law shape."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _left_power_law(n: int, alpha: float) -> np.ndarray:
    """Strictly decreasing power‑law (a_i ∝ (i+1)^(-α))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = idx ** (-alpha)
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * (i / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k equal spikes placed at random positions."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a

def _subgradient_monotone_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation with an additional
    projection onto the monotone simplex (non‑increasing, non‑negative,
    unit‑sum).  Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c               # sum(a) == 1 → denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Mono‑subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (mirror descent on the simplex)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # monotone‑simplex projection
        a = _project_monotone_simplex(a)

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    Returns the best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: transfer a random fraction of mass from
    one index to another and keep the move if it improves the objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)          # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a safety margin

    # ------------------------------------------------------------------
    # problem size (larger n gives a tighter asymptotic bound)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # optional warm start from a previous solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build a diverse pool of initial candidates
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – quick scan of many candidates (≈ 25 % of the time)
    phase1_deadline = min(deadline - 10.0, time.time() + 0.25 * (deadline - time.time()))
    per_candidate_budget = (phase1_deadline - time.time()) / max(1, len(pool))

    for cid, cand in enumerate(pool):
        if time.time() >= phase1_deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = rng.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(phase1_deadline, time.time() + per_candidate_budget)

        # evaluate directly – no inner optimisation
        conv = _full_convolution(a0, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                # sum(a0)=1
        if val < best_val:
            best_val = val
            best_seq = a0.copy()
            print(f"[Init cand {cid}] val {val:.6f}")

        # occasional very short refinement (helps to discard obviously bad candidates)
        if time.time() < phase1_deadline and np.random.rand() < 0.03:
            a_sub, val_sub = _subgradient_monotone_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=5_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Quick mono‑subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deep monotone sub‑gradient refinement of the current best
    if best_seq is not None and time.time() < deadline - 15:
        print("[Phase 2] deep monotone sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_monotone_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Monotone sub‑grad] new best {val_sub:.6f}")

        # a short projected‑gradient refinement (different dynamics)
        if time.time() < deadline - 5:
            print("[Phase 2] projected‑gradient refinement")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=200_000,
                eta0=0.4,
            )
            if val_pgd < best_val:
                best_val = val_pgd
                best_seq = a_pgd.copy()
                print(f"[PGD] new best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap random local moves
    if best_seq is not None and time.time() < deadline - 3:
        print("[Phase 3] final local random search")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
