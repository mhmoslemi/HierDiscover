def _pgd_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=500000, eta0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_val = _eval_objective(a, n, conv_fft_len)
    it=0
    while it<max_iters and time.time()<deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0*n*max_c/(a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it%5000==0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")
        # find active indices
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            # random move if max not stable
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)   # gradient of max conv w.r.t a
        # gradient of objective = 2n * grad (since sum=1)
        grad_F = 2.0 * n * grad
        # step size
        eta = eta0 / np.sqrt(it+1)
        a_new = a - eta * grad_F
        # projection onto simplex
        a = _project_simplex(a_new)
        it += 1
    return best_a, best_val

def _project_simplex(v):
    # ensure non-negative; we want projection onto simplex {x >=0, sum=1}
    # Implementation as per Duchi et al.
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _lp_refine(f, n, conv_fft_len, deadline, max_iter=20, t=0.2, tol=1e-12):
    # f: current distribution (numpy array, sum=1)
    best_f = f.copy()
    best_val = _eval_objective(f, n, conv_fft_len)
    for it in range(max_iter):
        # compute current conv and M
        conv_f = _full_convolution(best_f, conv_fft_len)
        M = conv_f.max()
        # active constraints indices where conv_f close to M
        active = np.where(conv_f >= M - 1e-12)[0]  # indices in 0..2n-2
        # Build sparse constraint matrix A (len(active) rows, n cols)
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active):
            # For each b_j, coefficient = f_{k-j} if 0 <= k-j < n
            # compute j range
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            # indices j = j_min..j_max
            # corresponding f index idx = k - j
            idxs = np.arange(j_min, j_max+1)
            f_vals = best_f[k - idxs]
            rows.extend([row_idx] * f_vals.size)
            cols.extend(idxs.tolist())
            data.extend(f_vals.tolist())
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        # objective: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n)
        # bounds: b >= 0
        bounds = [(0, None)] * n
        # Solve LP
        res = sp_opt.linprog(c, A_ub=A_ub, b_ub=np.full(len(active), M), bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            break
        b = res.x
        # normalize b to sum to 1 (if sum>0)
        s_b = b.sum()
        if s_b < 1e-12:
            # if all zero, break
            break
        b = b / s_b
        # mix
        best_f = (1 - t) * best_f + t * b
        # project onto simplex (should already sum to 1)
        # compute objective
        val = _eval_objective(best_f, n, conv_fft_len)
        if val < best_val:
            best_val = val
            # continue
        else:
            # if no improvement, maybe reduce t
            t *= 0.5
        # check time
        if time.time() >= deadline:
            break
    return best_f, best_val

Hybrid search for a step‑function that minimises  

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of four stages

1. **Very broad random sampling** (Dirichlet, exponentials,
   sinusoid, beta‐pdf, piecewise constant, Gaussian mixtures, sparse spikes).
   The best few candidates are kept.

2. **Adam optimisation** of a smooth‑max (log‑sum‑exp) surrogate.

   For the current distribution `f` we solve  

        maximise   Σ b_j
        subject to  f ★ b ≤ M          (M = max(f★f))
                    b ≥ 0

   The LP solution `b` is mixed with `f` (step‑size `t`) and the process
   repeats a few times – this is the “LP‑refinement” described in the

4. **Cheap random hill‑climb** (move a random fraction of mass between
   two coordinates) for a final polishing.

All heavy work uses FFT‑based convolution, the whole routine respects the

# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.sum(v) == 0:
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against tiny round‑off errors
    s = w.sum()
    if s == 0:
        return np.full_like(v, 1.0 / n)
    return w / s


# ----------------------------------------------------------------------
# Exact objective (used only for bookkeeping) -------------------------
def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Adam optimiser (smooth max surrogate) --------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    lr: float = 0.07,
    lam_start: float = 0.08,
    lam_factor: float = 1.015,
    noise_sigma: float = 0.025,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(np.clip(a0, eps, None))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective -------------------------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp) -------------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate ------------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # Adam update on logits ----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule -----------------------------------------
        lam = min(lam * lam_factor, 1e6)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        if it % 4_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Active‑set LP refinement -----------------------------------------------
def _lp_refine(
    f: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 12,
    mix_coeff: float = 0.20,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Perform the linear‑programming refinement described in the prompt.
    At each iteration we
        * compute M = max(f★f)
        * collect the indices where this bound is active,
        * solve   maximise Σ b_j   s.t.   f★b ≤ M   (only active rows)
        * normalise b and mix it with f.
    """
    best_f = f.copy()
    best_val = _eval_objective(best_f, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_f = _full_convolution(best_f, conv_fft_len)
        M = conv_f.max()
        active = np.where(conv_f >= M - tol)[0]

        # Build a sparse matrix for the active constraints
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            # indices j where 0 ≤ j < n and 0 ≤ k-j < n
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            j_range = np.arange(j_min, j_max + 1, dtype=np.int64)
            coeffs = best_f[k - j_range]
            rows.extend([row_idx] * len(j_range))
            cols.extend(j_range.tolist())
            data.extend(coeffs.tolist())
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M)

        # maximise Σ b_j  → minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        res = sp_opt.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
        if not res.success:
            # If LP fails we abort the refinement.
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        s = b.sum()
        if s < 1e-12:
            break
        b = b / s  # bring b back onto the simplex

        # Mix the current distribution with the LP direction
        best_f = (1.0 - mix_coeff) * best_f + mix_coeff * b

        val = _eval_objective(best_f, n, conv_fft_len)
        if val < best_val:
            best_val = val
            print(f"[LP it={it:02d}] improved best {val:.6f}")

    return best_f, best_val


# ----------------------------------------------------------------------
# Cheap local random hill‑climb ------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Random‑candidate generators ---------------------------------------------
def _candidate_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n))


def _candidate_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _candidate_sinusoid(n: int) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_beta(n: int) -> np.ndarray:
    a, b = np.random.rand() * 5 + 0.1, np.random.rand() * 5 + 0.1
    xs = np.linspace(0, 1, n, dtype=np.float64)
    vals = xs ** (a - 1) * (1 - xs) ** (b - 1)
    if vals.sum() == 0 or np.any(np.isnan(vals)):
        vals = np.ones(n)
    vals /= vals.sum()
    return vals


def _candidate_piecewise_constant(n: int) -> np.ndarray:
    blocks = np.random.randint(2, 9)
    cuts = np.sort(np.random.choice(n - 1, size=blocks - 1, replace=False))
    points = np.concatenate(([0], cuts, [n]))
    a = np.empty(n, dtype=np.float64)
    for i in range(blocks):
        lo, hi = points[i], points[i + 1]
        height = np.random.rand()
        a[lo:hi] = height
    a /= a.sum()
    return a


def _candidate_gaussian_mix(n: int, max_comp: int = 5) -> np.ndarray:
    xs = np.arange(n, dtype=np.float64) / n
    k = np.random.randint(2, max_comp + 1)
    weights = np.random.dirichlet(np.ones(k))
    a = np.zeros(n, dtype=np.float64)
    for w in weights:
        mu = np.random.rand()
        sigma = np.random.rand() * 0.12 + 0.02
        a += w * np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def _candidate_spike_set(n: int, k: int) -> np.ndarray:
    idx = np.random.choice(n, size=k, replace=False)
    w = np.random.dirichlet(np.ones(k))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # optional warm start ------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    # ------------------------------------------------------------------
    # cheap random sampling ------------------------------------------------
    def _try(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # baseline candidates
    if warm is not None:
        _try(warm)
    else:
        _try(np.full(n, 1.0 / n))
    _try(_candidate_sinusoid(n))

    # Dirichlet family with varied concentration
    for _ in range(30):
        alpha = np.random.rand() * 4.0 + 0.2
        _try(_candidate_dirichlet(n, alpha))

    # Exponential left‑right families
    for gamma in (0.2, 0.5, 1.0, 2.0, 4.0):
        _try(_candidate_exponential(n, gamma))

    # Beta‑pdf shapes
    for _ in range(12):
        _try(_candidate_beta(n))

    # Piecewise constant (few blocks)
    for _ in range(12):
        _try(_candidate_piecewise_constant(n))

    # Gaussian mixtures (2‑6 components)
    for _ in range(20):
        _try(_candidate_gaussian_mix(n, max_comp=6))

    # Sparse spike sets with increasing support sizes
    for k in (32, 64, 128, 256, 512):
        for _ in range(6):
            _try(_candidate_spike_set(n, k))

    # ------------------------------------------------------------------
    # Adam optimisation (smooth‑max surrogate) -------------------------
    remaining = deadline - time.time()
    if remaining > 30 and best_seq is not None:
        adam_deadline = time.time() + 0.55 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=250_000,
            lr=0.07,
            lam_start=0.08,
            lam_factor=1.015,
            noise_sigma=0.025,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam main] improved best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Active‑set Linear‑Programming refinement ---------------------------
    if time.time() < deadline - 15 and best_seq is not None:
        lp_deadline = deadline - 10
        a_lp, val_lp = _lp_refine(
            best_seq,
            n,
            conv_fft_len,
            lp_deadline,
            max_iters=12,
            mix_coeff=0.20,
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] improved best {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Final cheap hill‑climb --------------------------------------------
    if time.time() < deadline - 5 and best_seq is not None:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # safety fallback ----------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
