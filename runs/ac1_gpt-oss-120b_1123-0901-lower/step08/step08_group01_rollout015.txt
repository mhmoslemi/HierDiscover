"""
Optimise a non‑negative step function to minimise
F(a)=2·n·max(a∗a)/(sum(a)^2).

The implementation is a time‑aware three‑phase search:

1. **Diverse warm‑starts** – a large pool of deterministic families
   (exponential / power‑law / beta / sinusoid / uniform) and many
   random Dirichlet draws with a wide range of concentration parameters.
   For each candidate a very short Adam run is performed; the best
   few candidates are retained.

2. **Deep refinement** – the current best candidate is refined by a long
   Adam run on the smooth‑max surrogate, followed by an exponentiated‑gradient
   (sub‑gradient) phase and a projected‑gradient phase.

3. **Fine‑grained exploration** – a cheap random‑move hill‑climber
   and a simulated‑annealing style sampler are executed until the global
   deadline is reached.  Throughout the run the globally best vector
   (as witnessed by the exact `evaluate_sequence`) is kept.

All FFT‑based convolutions are performed with lengths that are powers of two
to maximise speed.  The code respects the 1000 s budget and always returns a
valid list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# --- families of deterministic start vectors --------------------------

def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _tent(n: int) -> np.ndarray:
    """Symmetric tent shape, max at the centre, zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = 1.0 - np.abs(i - centre) / centre
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _mix_two(a: np.ndarray, b: np.ndarray, t: float) -> np.ndarray:
    """Convex combination t*a + (1‑t)*b, then normalise to sum 1."""
    mix = t * a + (1.0 - t) * b
    s = mix.sum()
    if s <= 0:
        return np.full_like(mix, 1.0 / mix.shape[0])
    return mix / s


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    pool = []

    # Dirichlet draws with a series of concentration parameters
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential family (wide range of decays)
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.4, 0.7, 1.0, 2.0, 4.0, 8.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law family (negative exponents give edge‑heavy shapes)
    for alpha in (-5.0, -4.0, -3.0, -2.5, -2.0, -1.5, -1.0, -0.7, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf shapes (include very peaked edge cases)
    for a, b in [(0.1, 0.1), (0.2, 0.2), (0.5, 0.5), (0.5, 2.0),
                 (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0), (0.3, 0.7)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian families (different spreads)
    for sigma in (0.02, 0.05, 0.08, 0.12, 0.2, 0.35):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid, tent and uniform
    pool.append(_sinusoid(n))
    pool.append(_tent(n))
    pool.append(_uniform(n))

    # Sparse random spikes with many sparsity levels
    for k in (16, 32, 64, 128, 256, 512, 1024, 2048):
        pool.append(_sparse_spikes(n, k, rng))

    # Mix a few randomly chosen families to enrich diversity
    for _ in range(20):
        a = pool[rng.integers(len(pool))]
        b = pool[rng.integers(len(pool))]
        t = rng.random()
        pool.append(_mix_two(a, b, t))

    return pool


# ----------------------------------------------------------------------
# --- optimisation primitives -----------------------------------------

def _softmax(logits: np.ndarray) -> np.ndarray:
    mx = logits.max()
    exp_logits = np.exp(logits - mx)
    return exp_logits / exp_logits.sum()


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.04,
    lam_start: float = 0.02,
    lam_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp).
    Returns best vector and its exact evaluation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                       # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # --- simplex via soft‑max ---
        a = _softmax(logits)

        # --- exact objective ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # --- smooth‑max (log‑sum‑exp) weights ---
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- mirror‑descent (soft‑max) chain rule ---
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # --- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # --- temperature schedule ----------
        lam = min(lam * lam_factor, 1e7)

        # --- occasional tiny random perturbation near deadline ---
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # --- learning‑rate decay ----------
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on simplex) using the exact sub‑gradient.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)  # up to 30 % of source

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_temp: float = 1e-3,
    decay: float = 0.9995,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simple annealer that occasionally accepts uphill moves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val
    T = init_temp

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.25)   # 0‑25 % of source
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        accept = False
        if val2 < cur_val:
            accept = True
        else:
            # Metropolis criterion, scaled by temperature
            if np.random.rand() < np.exp(-(val2 - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a = a2
            conv = conv2
            cur_val = val2
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Anneal {it:06d}] new best {best_val:.6f}")

        T *= decay
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Respects the global time budget and always returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- optional warm start – use previous best if present ----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a diverse pool of start vectors
    pool = [_mix_two(_uniform(n), _edge_exponential(n, 0.2), 0.5)]  # a quick mixed seed
    if warm is not None:
        pool.append(warm)
    pool.extend(_initial_pool(n, rng))

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – quick scan of many candidates (≈20 % of total time)
    phase1_deadline = min(deadline - 10.0, time.time() + 0.2 * (deadline - time.time()))
    per_candidate_time = (phase1_deadline - time.time()) / max(1, len(pool))

    for cid, cand in enumerate(pool):
        if time.time() >= phase1_deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(phase1_deadline, time.time() + per_candidate_time)

        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=120_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase1 cand {cid}] Adam best {val_adam:.6f}")

        # short sub‑gradient refinement (optional, cheap)
        if time.time() < phase1_deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=12_000,
                eta0=0.85,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase1 cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deep refinement of the current best (≈55 % of time)
    if best_seq is not None and time.time() < deadline - 30:
        # Long Adam run
        a_adam2, val_adam2 = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            lr_start=0.03,
        )
        if val_adam2 < best_val:
            best_val = val_adam2
            best_seq = a_adam2.copy()
            print(f"[Phase2 Adam] new best {val_adam2:.6f}")

        # Extended exponentiated‑gradient sub‑gradient phase
        a_sub_long, val_sub_long = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            eta0=0.8,
        )
        if val_sub_long < best_val:
            best_val = val_sub_long
            best_seq = a_sub_long.copy()
            print(f"[Phase2 Subgrad] new best {val_sub_long:.6f}")

        # Projected‑gradient descent for a different dynamics
        if time.time() < deadline - 15:
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=250_000,
                eta0=0.42,
            )
            if val_pgd < best_val:
                best_val = val_pgd
                best_seq = a_pgd.copy()
                print(f"[Phase2 PGD] new best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random local moves (≈10 % of time)
    if best_seq is not None and time.time() < deadline - 10:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=100_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase3 Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing to escape possible plateaus (remainder)
    if best_seq is not None and time.time() < deadline - 3:
        a_anneal, val_anneal = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            init_temp=5e-4,
            decay=0.9997,
            max_moves=150_000,
        )
        if val_anneal < best_val:
            best_val = val_anneal
            best_seq = a_anneal.copy()
            print(f"[Phase4 Anneal] improved best {val_anneal:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was ever found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to allowed range for the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Optimise a non‑negative step function to minimise
    F(a) = 2·n·max(a∗a) / (sum(a)^2).

The algorithm runs in four time‑aware phases:

1. **Diverse warm‑starts** – many deterministic families (exponential,
   power‑law, beta, double‑Gaussian, sinusoid, tent, uniform) together
   with a large set of random Dirichlet and sparse‑spike vectors.
   Each candidate is given a very short Adam run; the best few are kept.

2. **Deep refinement** – the current best candidate is refined by a long
   Adam optimisation on a log‑sum‑exp surrogate, followed by an
   exponentiated‑gradient sub‑gradient phase and a projected‑gradient phase.

3. **Cheap hill‑climbing** – random mass transfers improve the objective
   quickly while staying within the time budget.

4. **Simulated annealing** – a final stochastic search that accepts
   occasional uphill moves, helping to escape any remaining local minima.

All convolutions are performed by FFT with power‑of‑two lengths for speed.
The routine always returns a valid list of non‑negative floats and respects
the global 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# family generators ----------------------------------------------------
def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _tent(n: int) -> np.ndarray:
    """Symmetric tent shape – max at centre, zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = 1.0 - np.abs(i - centre) / centre
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _mix_two(a: np.ndarray, b: np.ndarray, t: float) -> np.ndarray:
    """Convex combination t·a + (1‑t)·b, then renormalise."""
    mix = t * a + (1.0 - t) * b
    s = mix.sum()
    if s <= 0:
        return np.full_like(mix, 1.0 / mix.shape[0])
    return mix / s


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a large, diverse pool of start vectors."""
    pool = []

    # Dirichlet draws with a wide range of concentration parameters
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential (varied decay)
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.4, 0.7, 1.0, 2.0, 4.0, 8.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law (negative exponents – edge‑heavy)
    for alpha in (-5.0, -4.0, -3.0, -2.5, -2.0, -1.5, -1.0, -0.7, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf shapes (including very sharp edge spikes)
    for a, b in [(0.1, 0.1), (0.2, 0.2), (0.5, 0.5), (0.5, 2.0),
                 (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0), (0.3, 0.7)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian with several spreads
    for sigma in (0.02, 0.05, 0.08, 0.12, 0.2, 0.35):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid, tent and uniform
    pool.append(_sinusoid(n))
    pool.append(_tent(n))
    pool.append(_uniform(n))

    # Sparse spikes of many sparsities
    for k in (16, 32, 64, 128, 256, 512, 1024, 2048):
        pool.append(_sparse_spikes(n, k, rng))

    # Random convex mixes of two already‑generated vectors
    for _ in range(20):
        a = pool[rng.integers(len(pool))]
        b = pool[rng.integers(len(pool))]
        t = rng.random()
        pool.append(_mix_two(a, b, t))

    return pool

def _softmax(logits: np.ndarray) -> np.ndarray:
    mx = logits.max()
    exp_logits = np.exp(logits - mx)
    return exp_logits / exp_logits.sum()


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.04,
    lam_start: float = 0.02,
    lam_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp).
    Returns best vector (exact objective) and its value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)      # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        a = _softmax(logits)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # tiny random shake in the final seconds
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on the simplex) using the exact
    sub‑gradient of the max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_temp: float = 5e-4,
    decay: float = 0.9997,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simple annealer: occasionally accepts uphill moves to escape plateaus.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val
    T = init_temp

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.25)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        accept = False
        if val2 < cur_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(val2 - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a = a2
            conv = conv2
            cur_val = val2
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Anneal {it:06d}] new best {best_val:.6f}")

        T *= decay
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best sequence
    found (as a Python list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build a large, diverse pool of start vectors
    pool = [_mix_two(_uniform(n), _edge_exponential(n, 0.2), 0.5)]
    if warm is not None:
        pool.append(warm)
    pool.extend(_initial_pool(n, rng))

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – quick scan of many starts (≈20 % of time)
    phase1_deadline = min(deadline - 10.0,
                         time.time() + 0.2 * (deadline - time.time()))
    per_candidate_time = (phase1_deadline - time.time()) / max(1, len(pool))

    for cid, cand in enumerate(pool):
        if time.time() >= phase1_deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # give each candidate a small personal budget
        local_deadline = min(phase1_deadline,
                             time.time() + per_candidate_time)

        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=120_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase1 cand {cid}] Adam best {val_adam:.6f}")

        # short sub‑gradient refinement (use remaining per‑candidate budget)
        if time.time() < phase1_deadline:
            sub_deadline = min(phase1_deadline,
                               time.time() + per_candidate_time * 0.6)
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=12_000,
                eta0=0.85,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase1 cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deep refinement of the current best (≈55 % of time)
    if best_seq is not None and time.time() < deadline - 30:
        # long Adam run
        a_adam2, val_adam2 = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            lr_start=0.03,
        )
        if val_adam2 < best_val:
            best_val = val_adam2
            best_seq = a_adam2.copy()
            print(f"[Phase2 Adam] new best {val_adam2:.6f}")

        # extended exponentiated‑gradient sub‑gradient phase
        a_sub_long, val_sub_long = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            eta0=0.8,
        )
        if val_sub_long < best_val:
            best_val = val_sub_long
            best_seq = a_sub_long.copy()
            print(f"[Phase2 Subgrad] new best {val_sub_long:.6f}")

        # projected‑gradient phase for a different dynamics
        if time.time() < deadline - 15:
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=250_000,
                eta0=0.42,
            )
            if val_pgd < best_val:
                best_val = val_pgd
                best_seq = a_pgd.copy()
                print(f"[Phase2 PGD] new best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap hill‑climbing (≈10 % of time)
    if best_seq is not None and time.time() < deadline - 10:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=100_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase3 Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (remaining time)
    if best_seq is not None and time.time() < deadline - 3:
        a_anneal, val_anneal = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            init_temp=5e-4,
            decay=0.9997,
            max_moves=150_000,
        )
        if val_anneal < best_val:
            best_val = val_anneal
            best_seq = a_anneal.copy()
            print(f"[Phase4 Anneal] improved best {val_anneal:.6f}")

    # ------------------------------------------------------------------
    # fallback if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ensure the output adheres to evaluator’s domain
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
