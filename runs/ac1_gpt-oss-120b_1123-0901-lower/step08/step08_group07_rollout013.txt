"""
Propose a low‑value step function.

Strategy
--------
1. **Broad candidate generation** – many analytical shapes (uniform,
   Dirichlet, anti‑pair, power‑law, exponential, Gaussian edges,
   triangle, random piece‑wise linear).  All are normalised to sum 1.
2. **Quick screening** – evaluate all candidates with a fast FFT‑based
   convolution and keep the best few.
3. **Intensive refinement** – the top candidate receives a long
   Adam‑smooth‑max phase followed by an exact‑max sub‑gradient phase.
   The other top candidates get short refinement passes.
4. **Stochastic hill‑climbing** – from the current best a long
   simulated‑annealing style local search explores swaps and mass
   transfers.
5. **Safety net** – guarantee a non‑negative, bounded list is returned.

The routine respects the 1000 s budget, prints intermediate progress,
and returns the best sequence found (as a plain Python list).  The
implementation uses only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a). Returns a vector of length len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (without the security wrappers)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a≥0."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)

# Candidate generators -------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑to‑smallest on the left, smallest‑to‑largest on the right."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    half = n // 2
    a = np.empty(n, dtype=np.float64)
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = _random_dirichlet_candidate(n, alpha)
    return _anti_pair_arrangement(vals)


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _exponential_candidate(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i).  rate>0."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-rate * i)
    vals /= vals.sum()
    return vals


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _piecewise_linear_candidate(n: int, knots: int = 10) -> np.ndarray:
    """Random piece‑wise linear shape with `knots` control points."""
    # knot positions are uniformly spaced
    pos = np.linspace(0, 1, knots)
    heights = np.random.rand(knots)
    # enforce positivity
    heights = np.maximum(heights, 1e-6)
    # linear interpolation
    xs = np.linspace(0, 1, n)
    vals = np.interp(xs, pos, heights)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _warm_start(n: int) -> np.ndarray | None:
    """If a previous best sequence exists, resize and normalise it."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    lr: float = 0.08,
    lam_start: float = 0.2,
    lam_factor: float = 1.08,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector seen and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max -> simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact value for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # centred gradient in log‑space
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → tighten towards true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (multiplicative updates on the simplex).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # re‑project onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - temperature decays linearly from 0.02 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:
            # transfer a fraction of mass from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.5)   # up to 50 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 10_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the evaluate_sequence checks.

    The routine respects the time budget and returns the best sequence
    (as a Python list) found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a large, diverse pool of candidates
    candidates = []

    # Warm start (if any)
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few shifted copies – shifting changes the anti‑pair structure
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Deterministic shapes
    candidates.append(_uniform_candidate(n))
    candidates.append(_triangle_candidate(n))
    for gamma in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    for p in (0.2, 0.5, 1.0, 2.0, 3.0):
        candidates.append(_powerlaw_candidate(n, p))
    for rate in (0.0005, 0.001, 0.002, 0.005, 0.01):
        candidates.append(_exponential_candidate(n, rate))
    # Random piece‑wise linear
    for _ in range(6):
        candidates.append(_piecewise_linear_candidate(n, knots=12))

    # Random Dirichlet samples (various concentrations)
    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0, 5.0):
        for _ in range(6):
            candidates.append(_random_dirichlet_candidate(n, alpha))

    # Anti‑pair Dirichlet samples
    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0):
        for _ in range(4):
            candidates.append(_anti_pair_candidate(n, alpha))

    # ------------------------------------------------------------------
    # 2) Quick evaluation, keep top few
    print("[Stage 1] evaluating initial pool …")
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 30 == 0:
            print(f"  candidate {idx:04d}: value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 7
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"[Init] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Intensive refinement
    remaining = deadline - time.time()
    # allocate ~60 % of remaining time to the absolute best candidate,
    # the remainder split among the other top candidates.
    best_budget = remaining * 0.60
    others_budget = (remaining - best_budget) / (top_k - 1)

    # ---- Refine the absolute best candidate (long run) ----
    print("[Stage 2] long refinement of global best …")
    adam_dead = time.time() + best_budget * 0.55
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_dead,
        lr=0.09,
        lam_start=0.15,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val, best_seq = val_adam, a_adam.copy()
        print(f"  after Adam → {best_val:.6f}")

    sub_dead = time.time() + best_budget * 0.45
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        sub_dead,
        eta0=0.75,
    )
    if val_sub < best_val:
        best_val, best_seq = val_sub, a_sub.copy()
        print(f"  after Sub‑grad → {best_val:.6f}")

    # ---- Short refinement for the remaining top candidates ----
    for rank in range(1, top_k):
        cand = evaluated[rank][1].copy()
        print(f"[Stage 2] short refinement of rank {rank+1} candidate …")
        # short Adam
        sub_deadline = time.time() + others_budget * 0.45
        a_tmp, v_tmp = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            lr=0.07,
            lam_start=0.2,
            lam_factor=1.06,
        )
        # short sub‑gradient
        sub_deadline2 = time.time() + others_budget * 0.55
        a_tmp2, v_tmp2 = _subgradient_optimize(
            a_tmp,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline2,
            eta0=0.6,
        )
        # keep the better of the two refinements
        if v_tmp2 < best_val:
            best_val, best_seq = v_tmp2, a_tmp2.copy()
            print(f"  improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic hill‑climbing on the current best
    if time.time() < deadline - 5.0:
        print("[Stage 3] local random search …")
        loc_dead = deadline - 2.0
        a_loc, v_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            loc_dead,
            max_moves=250_000,
        )
        if v_loc < best_val:
            best_val, best_seq = v_loc, a_loc.copy()
            print(f"  after local search → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final safeguard – clip and return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
