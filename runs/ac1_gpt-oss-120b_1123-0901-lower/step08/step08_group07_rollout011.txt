"""
Propose a step‑function with a very low value of

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The routine follows a “diversify‑then‑refine” workflow:

1️⃣  **Diverse warm‑starts** – a pool of candidates is created from many
    analytical shapes (uniform, power‑law, U‑shape, Gaussian edges,
    triangle), a few random Dirichlet draws with a wide range of
    concentrations, and a number of sparse 0/‑mass vectors (random Sidon‑like
    supports).  The current best sequence (if present) is also added and
    shifted/​reversed.

2️⃣  **Quick filter** – each candidate is normalised to sum = 1 and evaluated
    with an exact, fast objective (no security checks).  Only the best few
    (default 5) are retained.

3️⃣  **Continuous optimisation** – on every retained candidate we run
    *Adam* on the smooth‑max (log‑sum‑exp) surrogate and then an exact‑max
    *exponentiated‑gradient* sub‑gradient phase.  Both phases respect the
    simplex and keep the best vector seen.

4️⃣  **Aggressive stochastic hill‑climbing** – the current global best is
    further improved by a long simulated‑annealing style local search that
    swaps entries or transfers a fraction (up to 60 %) of mass between
    positions.

5️⃣  **Optional up‑sampling** – if time permits the solution is linearly
    interpolated to 8192 points and refined once more (short Adam + sub‑grad).

All steps constantly check the remaining time (budget ≈ 1000 s) and stop
gracefully.  The returned list complies with the safety checks in
``evaluate_sequence`` (non‑negative, sum > 0.01, each entry ≤ 1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no security checks) ----------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) / (sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start utilities --------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm‑start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest ordering."""
    n = len(vals)
    s = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]                    # biggest half on the left
    a[-half:] = s[-half:][::-1]            # smallest half on the right
    if n % 2 == 1:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float) -> np.ndarray:
    vals = _random_dirichlet_candidate(n, alpha)
    return _anti_pair_arrangement(vals)


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    return vals / vals.sum()


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    return vals / vals.sum()


def _gaussian_edges_candidate(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    return shape / shape.sum()


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    return shape / shape.sum()


def _sparse_random_candidate(n: int, m: int) -> np.ndarray:
    """m distinct positions, each equal weight 1/m."""
    if m > n:
        m = n
    idx = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.05,
    lam_start: float = 0.2,
    lam_factor: float = 1.08,
    prune_thresh: float = 1e-7,
    prune_every: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector seen and its exact value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- simplex from logits ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- evaluate exact objective (for tracking) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam  it={it:06d}] new best {best_val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a (smooth surrogate) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule (approach true max) ----
        lam = min(lam * lam_factor, 1e6)

        # ---- occasional jitter to escape plateaus ----
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        # ---- optional pruning for sparsity ----
        if prune_thresh > 0 and (it + 1) % prune_every == 0:
            a_pruned = np.where(a < prune_thresh, 0.0, a)
            if a_pruned.sum() > 0:
                a_pruned /= a_pruned.sum()
                logits = np.log(a_pruned + eps)

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (exponentiated‑gradient) on the simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where conv == max_c)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size decreasing ~ 1/sqrt(t)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (multiplicative)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing.
    - swap two entries   OR
    - transfer a fraction (up to 60 %) of mass from i to j.
    Accept improvements or probabilistically accept worsenings.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.015                     # a bit hotter than before
    it = 0
    while time.time() < deadline and it < max_moves:
        # --- decide move type ------------------------------------------------
        if np.random.rand() < 0.5:         # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                              # transfer
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.6)   # up to 60 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        # --- evaluate ---
        val2 = _objective(a2)

        # --- acceptance ---
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

        candidates.append(_ushape_candidate(n, p))

        candidates.append(_gaussian_edges_candidate(n, gamma))

        candidates.append(_anti_pair_candidate(n, a))

    # sparse random supports (Sidon‑like)
    for m in (80, 100, 120):
        for _ in range(3):
            candidates.append(_sparse_random_candidate(n, m))

    # ------------------------------------------------------------------

    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 20 == 0:
            print(f"[Init  cand {idx:03d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Continuous optimisation on the top candidates
    remaining = deadline - time.time()
    per_candidate_budget = (remaining * 0.6) / top_k

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        # Adam phase
        adam_dead = time.time() + per_candidate_budget * 0.55
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=350_000,
            lr=0.05,
            lam_start=0.2,
            lam_factor=1.08,
            prune_thresh=1e-7,
            prune_every=4000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid} Adam] improved to {val_adam:.6f}")

        # Sub‑gradient phase
        sub_dead = time.time() + per_candidate_budget * 0.45
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=400_000,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {cid} Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Aggressive local stochastic hill‑climbing
    if time.time() < deadline - 8.0:
        local_dead = deadline - 4.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=400_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid (8192) if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of current best
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on the larger grid
        big_adam_dead = min(deadline - 5.0, time.time() + 120.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_dead,
            max_iters=250_000,
            lr=0.05,
            lam_start=0.2,
            lam_factor=1.08,
            prune_thresh=1e-7,
            prune_every=5000,
        )
        # short sub‑gradient refinement
        if time.time() < deadline - 1.0:
            a_big_sub, val_big_sub = _subgradient_optimize(
                a_big_adam,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=250_000,
                eta0=0.8,
            )
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big_sub.copy()
                print(f"[Big] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 7) Safeguard – guarantee a valid return
    if best_seq is None or not isinstance(best_seq, np.ndarray):
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
