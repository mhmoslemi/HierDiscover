"""
propose_candidate()
------------------

A multi‑phase heuristic for the “minimise max‑convolution” problem.

The algorithm proceeds as follows

1️⃣  Build a diverse pool of warm‑starts (uniform, Dirichlet, anti‑pair,
    triangular, Gaussian‑edge, sine‑shaped, and a possible previous
    solution).  The best of them is kept as the initial point.

2️⃣  Refine the starter with a short Adam optimiser on a smooth‑max
    surrogate (log‑sum‑exp) followed by an exact‑max exponentiated‑gradient
    sub‑gradient step.  Both phases respect the remaining time budget.

3️⃣  Perform a long incremental simulated‑annealing (SA) local search.
    The SA move transfers a random fraction of mass from one index to
    another and updates the full linear convolution in O(n) by
    slice‑addition.  The schedule is split into three stages:
        – coarse (large fractions, higher temperature),
        – medium (moderate fractions),
        – fine (tiny fractions, low temperature).
    The best sequence seen during SA is retained.

4️⃣  If time permits, several short random restarts are launched, each
    running a reduced SA to try to escape the current basin.

All vectors are kept non‑negative, normalised to sum≈1 (the objective
is scale‑invariant).  The routine returns a Python list that can be fed
directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2*len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for gradient (weighted sum of shifts)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(a*a) / (sum(a)**2) .  Assumes a≥0.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start handling --------------------------------------------------
def _warm_start(n: int) -> np.ndarray | None:
    """If a previous global solution exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


# ----------------------------------------------------------------------
# Candidate generators -------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Large values on the left, small values on the right (anti‑pair)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]                     # biggest half left
    a[-half:] = sorted_vals[-half:][::-1]             # smallest half right
    if n % 2:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Hat shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _sine_candidate(n: int, power: float = 2.0) -> np.ndarray:
    """Smooth sine‑squared shape, always non‑negative."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / n) ** power
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape

# Optimisation primitives -----------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
    lam_start: float = 0.5,
    lam_factor: float = 1.03,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max -> simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current convolution & exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a of the smoothed objective
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # convert to gradient on logits (soft‑max manifold)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape flat regions
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient (exponentiated‑gradient) phase."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indicator of active constraints (where conv reaches max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental convolution update (for the long SA) --------------------
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float) -> None:
    """
    Incrementally update full convolution a*a when a[idx] += delta (delta may be negative).
    Operates in O(n) via slice addition.
    """
    # contribution from pairs (idx, j) and (j, idx), j=0..n-1
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    # self‑pair term (idx, idx)
    conv[2 * idx] += delta * delta
    a[idx] += delta


def _incremental_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.04,
    max_frac: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing with incremental convolution updates.
    Parameters:
        a0          – starting probability vector (sum≈1)
        time_limit  – absolute deadline (seconds since epoch)
        start_temp  – initial temperature (linear decay)
        max_frac    – maximal fraction of mass moved in a single step
    Returns the best sequence found and its convolution maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        # pick source index with non‑negligible mass
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        # pick a different destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply the move (incremental updates)
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear temperature schedule
        remaining = max(0.0, time_limit - time.time())
        total = max(1e-9, time_limit - start)
        temp = start_temp * (remaining / total)

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert the move
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 100_000 == 0:
            print(f"[SA it={it:07d}] best max={best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good compromise for quality vs speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build diverse initial candidates
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few rotations – sometimes help to break symmetry
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    for alpha in (0.2, 0.5, 0.8, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))
    for alpha in (0.3, 0.7, 1.5):
        candidates.append(_anti_pair_candidate(n, alpha))
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))
    candidates.append(_sine_candidate(n, power=2.0))

    # ------------------------------------------------------------------
    # 2) Pick the best starter based on cheap exact evaluation
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) Continuous optimisation (Adam + sub‑gradient) – modest budget
    # Reserve ~30 s for the final SA, split the rest between Adam and sub‑grad.

            lam_factor=1.03,

        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # ----- Sub‑gradient ------------------------------------------------------
        sub_deadline = min(deadline - 5.0, time.time() + 250.0)
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=250_000,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Long incremental simulated‑annealing (three‑stage schedule)
    # Stage parameters (time fractions are approximate)
    remaining = deadline - time.time()
    if remaining > 10.0:
        # Stage 1 – coarse moves
        stage1_time = min(remaining * 0.45, 250.0)
        print("[SA] Stage 1 (coarse) start")
        seq, max_c = _incremental_sa(
            best_seq,
            time_limit=time.time() + stage1_time,
            start_temp=0.06,
            max_frac=0.4,
        )

            best_seq = seq.copy()
            print(f"[SA Stage1] new best max={max_c:.6f}")

                best_seq,
                time_limit=time.time() + stage2_time,
                start_temp=0.03,
                max_frac=0.15,
            )

                best_seq = seq.copy()
                print(f"[SA Stage2] new best max={max_c:.6f}")

        # Stage 3 – fine moves
        remaining = deadline - time.time()
        if remaining > 3.0:
            stage3_time = remaining - 0.5
            print("[SA] Stage 3 (fine) start")
            seq, max_c = _incremental_sa(
                best_seq,
                time_limit=time.time() + stage3_time,
                start_temp=0.015,
                max_frac=0.04,
            )
            if max_c * 2 * n < best_val:
                best_val = max_c * 2 * n
                best_seq = seq.copy()
                print(f"[SA Stage3] new best max={max_c:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional short random restarts (if time still permits)
    while time.time() < deadline - 5.0:
        # generate a fresh random start
        cand = _random_dirichlet_candidate(n, alpha=0.5)
        cand /= cand.sum()
        # quick SA for ~15 s
        sa_time = min(15.0, deadline - time.time() - 1.0)
        print("[Restart] random SA start")
        seq, max_c = _incremental_sa(
            cand,
            time_limit=time.time() + sa_time,
            start_temp=0.04,
            max_frac=0.2,
        )
        if max_c * 2 * n < best_val:
            best_val = max_c * 2 * n
            best_seq = seq.copy()
            print(f"[Restart] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Final safety checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq /= total

    final_obj = _objective(best_seq)
    print(f"Final best objective {final_obj:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
