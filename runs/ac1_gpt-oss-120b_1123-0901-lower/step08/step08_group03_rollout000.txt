"""
propose_candidate
=================

A time‑bounded, multi‑phase optimizer for the objective

    F(a) = 2·n·max(a*a) / (sum(a)^2)   (a ≥ 0, a not all zero)

The strategy is:

1. Choose a fairly large problem size (default n = 8192).  
2. Warm‑start from the global variable ``height_sequence_1`` (if present)
   by linear up‑sampling.
3. **Random sampling** – a large, diverse pool of candidates
   (Dirichlet draws with random concentrations, edge‑heavy power laws,
   Beta‑shaped families, mixtures of a few Gaussians, random sparse spikes,
   sinusoidal windows, etc.).
4. **Adam optimisation** on the smooth‑max surrogate (log‑sum‑exp) with a
   slowly increasing temperature (λ).  The parameters are chosen to give
   a long, steady run; several short noisy restarts are performed as well.
5. **Exponentiated‑gradient sub‑gradient** descent (mirror descent on the
   simplex) that directly follows the true max‑convolution.
6. **Very cheap random hill‑climb** that moves a random fraction of mass
   between two coordinates.
7. Return the best vector found (always a valid non‑negative list).

All phases are guarded by a global deadline (default 1000 s).  The code
contains no closures, uses only NumPy, and prints progress so that the
running process can be monitored.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (ignores the security wrapper)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.07,
    lam_start: float = 0.05,
    lam_factor: float = 1.02,
    noise_sigma: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp) with temperature λ.
    Returns the best vector (already normalised to sum=1) and its exact value.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)               # avoid log(0)
    logits = np.log(a0)                         # start on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # ----- projection onto the simplex (soft‑max) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (for bookkeeping) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights (temperature = 1/lam) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional stochastic perturbation
        if noise_sigma > 0.0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # occasional gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        # shake near the deadline to escape possible plateaus
        if it % 3_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _expgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.85,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent on the simplex) with a simple
    momentum term on the gradient.  Works directly with the true max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    g_mom = np.zeros_like(a)

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # ----- active set (indices attaining the current max) -----
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- momentum on the gradient -----
        g_mom = momentum * g_mom + (1.0 - momentum) * grad

        # ----- step size schedule -----
        eta = eta0 / np.sqrt(it + 1)

        # ----- exponentiated‑gradient update -----
        a = a * np.exp(-eta * g_mom)

        # ----- renormalise -----
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny hill‑climber: move a random fraction of mass from i to j,
    accept only if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.30)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local it={it:07d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine respects the time budget and always
    returns the best sequence seen.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (larger n gives a richer search space)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start from the global variable `height_sequence_1` if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # up‑sample / down‑sample to the chosen n
        if warm.shape[0] != n:
            old_n = warm.shape[0]
            old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            warm = np.interp(new_x, old_x, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # initialise best sequence (uniform fallback)
    if warm is not None:
        best_seq = warm.copy()
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_val = _eval_objective(best_seq, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # ---------- Phase 1 : diverse random sampling --------------------
    def try_candidate(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # 1) Dirichlet draws with random concentration parameters
    for _ in range(12):
        # random concentration in [0.2, 5.0] for each coordinate
        conc = np.random.uniform(0.2, 5.0, size=n)
        a = np.random.gamma(conc, 1.0)
        a /= a.sum()
        try_candidate(a)

    # 2) Edge‑heavy power‑law families
    for _ in range(8):
        p = np.random.uniform(0.3, 1.8)   # exponent
        left = (np.arange(n, dtype=np.float64) + 1) ** (-p)
        right = left[::-1]
        a = left + right
        a /= a.sum()
        try_candidate(a)

    # 3) Symmetric Beta‑shaped densities
    for _ in range(10):
        alpha = np.random.uniform(0.2, 4.0)
        beta = np.random.uniform(0.2, 4.0)
        x = np.linspace(0.0, 1.0, n, endpoint=False)
        a = (x ** (alpha - 1.0)) * ((1 - x) ** (beta - 1.0))
        a += 1e-12   # avoid exact zeros
        a /= a.sum()
        try_candidate(a)

    # 4) Mixtures of a few Gaussian bumps
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(12):
        m = np.random.randint(2, 5)   # number of components
        a = np.zeros_like(xs)
        for _ in range(m):
            mu = np.random.rand()
            sigma = np.random.rand() * 0.12 + 0.02
            amp = np.random.rand()
            a += amp * np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
        a += 1e-12
        a /= a.sum()
        try_candidate(a)

    # 5) Random sparse spikes (supports of various cardinalities)
    for k in (32, 64, 128, 256, 512, 1024):
        for _ in range(6):
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            try_candidate(a)

    # 6) Sine‑window families (different frequencies and powers)
    freqs = np.random.choice([1, 2, 3, 4, 5, 6, 8, 10, 12], size=4, replace=False)
    for freq in freqs:
        phase = np.random.rand() * 2 * np.pi
        power = np.random.uniform(0.5, 3.0)
        a = np.sin(np.pi * freq * (np.arange(n) + 0.5) / n + phase) ** 2
        a = np.maximum(a, 0.0) ** power
        a += 1e-12
        a /= a.sum()
        try_candidate(a)

    # ------------------------------------------------------------------
    # ---------- Phase 2 : long Adam optimisation --------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + 0.60 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=350_000,
            lr=0.07,
            lam_start=0.05,
            lam_factor=1.018,
            noise_sigma=0.03,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam main] improved best {val_adam:.6f}")

    # ---------- Phase 3 : noisy‑restart Adam (short bursts) ----------
    for restart in range(5):
        if time.time() >= deadline - 25:
            break
        # perturb current best in log‑space
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.35, size=n)   # fairly large perturbation
        a0 = np.exp(logits - np.max(logits))
        a0 /= a0.sum()

        r_deadline = min(deadline - 5, time.time() + 0.12 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=200_000,
            lr=0.06,
            lam_start=0.05,
            lam_factor=1.02,
            noise_sigma=0.025,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # ---------- Phase 4 : exponentiated‑gradient sub‑gradient ----------
    if time.time() < deadline - 20:
        sg_deadline = deadline - 15
        a_sub, val_sub = _expgrad_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=300_000,
            eta0=0.85,
            momentum=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[ExpGrad] improved best {val_sub:.6f}")

    # ---------- Phase 5 : cheap random hill‑climb ----------
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # ensure a valid return (non‑negative, finite, sum>0)
    if best_seq is None or best_seq.sum() < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
