"""
Propose a low‑value step‑function for the benchmark

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search consists of three stages:

1️⃣  **Fast stochastic exploration** – many parametric families (Dirichlet,
    edge‑exponential, edge‑power‑law, Beta‑pdf, double‑Gaussian, sinusoid,
    uniform and sparse spikes) are instantiated with random parameters.
    Each candidate is normalised to sum = 1 and its exact objective (computed
    with an FFT‑based convolution) is recorded.  The best vector found in this
    phase becomes the warm‑start for the next stage.

2️⃣  **Intensive continuous optimisation** – the current best vector is
    refined by three complementary gradient‑based methods that respect the
    simplex constraint:
       * Adam on a soft‑max (log‑sum‑exp) surrogate of the max‑convolution.
       * Exponentiated‑gradient sub‑gradient descent (exact max‑convolution).
       * Euclidean projected‑gradient descent (exact max‑convolution).
    Each optimiser runs until the global deadline is reached.

3️⃣  **Final cheap hill‑climber** – a few‑thousand random “mass‑move’’ steps
    are tried; a move is kept only if it improves the true objective.
    This squeezes out the last bits of gain.

All phases are time‑aware, never exceed the supplied budget, and keep the
best sequence observed so far.  The returned list is guaranteed to be
non‑negative, finite and accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of parametric candidates (all return a vector that sums to 1)
def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass concentrated at the two edges."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (i+1)^α + (n‑i)^α."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k spikes with equal weight placed at random positions."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Objective (exact – no security checks, faster for internal use)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return the exact value of the benchmark objective."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # sum(a) is 1 for all candidates we generate, but we keep the formula generic
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Optimisers (all respect the simplex and stop at a given deadline)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr_start: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02            # temperature (inverse of smoothness)
    lam_factor = 1.015    # very slow growth → smoother gradients early on

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- soft‑max to enforce simplex ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective (for bookkeeping) ----
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (soft‑max over convolution) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through the soft‑max (mirror‑descent form) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e7)

        # ---- learning‑rate decay ----
        if it % 80000 == 0 and it > 0:
            lr *= 0.94

        # ---- tiny random jitter near the end of the budget ----
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size – classic 1/√t schedule
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """Euclidean projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # -------------------- problem size --------------------
    # A larger n gives a tighter theoretical lower bound (≈1).  8192 is a
    # comfortable compromise between quality and runtime.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # -------------------- warm start (if supplied) --------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] > n:
            warm_arr = warm_arr[:n]
        elif warm_arr.shape[0] < n:
            warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() >= 0.01:
            warm_arr = warm_arr / warm_arr.sum()
            best_seq = warm_arr.copy()
            best_val = _objective(best_seq, n, conv_fft_len)
            print(f"[Warm start] value {best_val:.6f}")
        else:
            best_seq = None
            best_val = float("inf")
    else:
        best_seq = None
        best_val = float("inf")

    # -------------------- Phase 1 – fast random exploration --------------------
    phase1_end = start_time + 0.35 * budget_s   # 35 % of the total time
    families = [
        "dirichlet",
        "edge_exp",
        "edge_pow",
        "beta",
        "double_gauss",
        "sinusoid",
        "uniform",
        "sparse",
    ]

    while time.time() < phase1_end:
        fam = rng.choice(families)

        if fam == "dirichlet":
            alpha = rng.choice([0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 2.0, 5.0])
            cand = rng.dirichlet(np.full(n, alpha))

        elif fam == "edge_exp":
            # log‑uniform gamma ∈ [0.01, 20]
            gamma = 10 ** rng.uniform(-2, 1.3)
            cand = _edge_exponential(n, gamma)

        elif fam == "edge_pow":
            # power‑law exponent ∈ [‑5, ‑0.1]
            alpha = rng.uniform(-5.0, -0.1)
            cand = _edge_power_law(n, alpha)

        elif fam == "beta":
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            cand = _beta_shape(n, a, b)

        elif fam == "double_gauss":
            sigma = rng.uniform(0.01, 0.35)
            cand = _double_gaussian(n, sigma)

        elif fam == "sinusoid":
            cand = _sinusoid(n)

        elif fam == "uniform":
            cand = _uniform(n)

        elif fam == "sparse":
            max_k = min(1024, n // 4)
            k = int(rng.integers(2, max_k + 1))
            cand = _sparse_spikes(n, k, rng)

        # occasional symmetrisation (helps flatten the autocorrelation)
        if rng.random() < 0.2:
            cand = (cand + cand[::-1]) * 0.5

        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random {fam}] new best {val:.6f}")

    # -------------------- Phase 2 – deep continuous refinement --------------------
    if best_seq is not None and time.time() < deadline - 12:
        # 1) Adam (smooth‑max surrogate)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            lr_start=0.08,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # 2) Exponentiated‑gradient sub‑gradient
        if time.time() < deadline - 8:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=500_000,
                eta0=0.7,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Subgrad] improved to {val_sub:.6f}")

        # 3) Projected‑gradient descent
        if time.time() < deadline - 5:
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=300_000,
                eta0=0.45,
            )
            if val_pgd < best_val:
                best_val = val_pgd
                best_seq = a_pgd.copy()
                print(f"[PGD] improved to {val_pgd:.6f}")

    # -------------------- Phase 3 – final cheap hill‑climber --------------------
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # -------------------- Fallback (should never be needed) --------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)

    # Clip to the domain accepted by evaluate_sequence and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
