def propose_candidate(seed=0, budget_s=1000):

    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    
    # Phase 1: parameter sweep over families
    # create lists of parameter values for each family
    # for each param value evaluate; also try random mixtures.
    # allocate about 0.3 of time for this.
    phase1_end = min(deadline-0.3, time.time() + 0.3*(deadline - time.time()))
    
    # Evaluate families
    for gamma in np.logspace(-2, 2, num=100):
        img = _edge_exponential(n, gamma)
        val = evaluate_sequence(img.tolist())
        if val < best_val:
            best_val = val; best_seq = img
    # similarly many loops
    # ensure we stop if time.
    
    # Phase 2: refine best_seq with Adam+subgrad+PGD
    # allocate ~0.5 * remaining time
    if best_seq is not None:
        t2_deadline = deadline - 0.1
        best_seq, best_val = refine(best_seq, n, fft_len, t2_deadline)

    if best_seq is not None:

"""
Search for a non‑negative step‑function that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm works in three stages:

1. **Exploratory sampling** – systematic grids over several analytic families
   (edge‑exponential, edge‑power‑law, beta‑pdf, double‑Gaussian, sinusoid,
   uniform) and a large set of random mixtures of these families.
   Each candidate is evaluated with a fast FFT‑based convolution; the best
   K candidates are kept.

2. **Refinement** – the current best candidate is improved by three
   gradient‑based phases:
   * Adam on a smooth‑max (log‑sum‑exp) surrogate,
   * Exponentiated‑gradient sub‑gradient phase,
   * Euclidean projected‑gradient descent.
   The number of iterations is limited only by the remaining time budget.

3. **Final local hill‑climbing** – cheap random mass‑transfer moves for the
   last seconds.

All phases respect the global deadline (default 1000 s) and the routine
always returns the best sequence found.  The returned list is clipped to the
range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT. Length of result = 2·len(a)-1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) of equal‑length vectors."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (all return a normalized non‑negative vector)
def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int, phase: float = 0.0) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5 + phase) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak at centre)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence but without safety checks)
def _fast_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2) ; assumes a≥0 and sum>0."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Random mixture generator (used in Phase 1)
def _sample_mixed_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Build a random convex combination of 1‑3 basic families.
    """
    families = [
        "edge_exp",
        "edge_power",
        "beta",
        "dbl_gauss",
        "sinusoid",
        "uniform",
        "triangular",
    ]
    k = rng.integers(1, 4)               # number of components (1‑3)
    weights = rng.dirichlet(np.ones(k))
    components = []

    for _ in range(k):
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2.0, 2.0)   # [0.01, 100]
            comp = _edge_exponential(n, gamma)
        elif fam == "edge_power":
            alpha = rng.uniform(-4.0, -0.1)        # negative exponent
            comp = _edge_power_law(n, alpha)
        elif fam == "beta":
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            comp = _beta_shape(n, a, b)
        elif fam == "dbl_gauss":
            sigma = rng.uniform(0.01, 0.5)
            comp = _double_gaussian(n, sigma)
        elif fam == "sinusoid":
            phase = rng.uniform(0, 2.0)            # phase shift in index units
            comp = _sinusoid(n, phase)
        elif fam == "triangular":
            comp = _triangular(n)
        else:  # uniform
            comp = _uniform(n)
        components.append(comp)

    shape = np.zeros(n, dtype=np.float64)
    for w, comp in zip(weights, components):
        shape += w * comp
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = _uniform(n)
    return shape

# Optimisers (reuse/extend the ones from the reference code)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.03
    best_a = a0.copy()
    best_val = np.inf
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max projection onto simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100000 == 0 and it > 0:
            lr *= 0.95

        # tiny random jitter near deadline
        if it % 5000 == 0 and time.time() > deadline - 20:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = np.inf
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = np.inf
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[PGD it={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (rng.random() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 20000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 1 – exploratory sampling (≈30 % of budget)
    phase1_deadline = min(deadline - 15.0,
                          start_time + 0.3 * (deadline - start_time))
    top_k = 30                     # keep the 30 best candidates seen
    pool = []                      # list of (fast_val, ndarray) sorted asc.

    def _insert_candidate(val_fast, vec):
        # insert maintaining ascending order, keep size ≤ top_k
        i = 0
        while i < len(pool) and val_fast > pool[i][0]:
            i += 1
        pool.insert(i, (val_fast, vec.copy()))
        if len(pool) > top_k:
            pool.pop()   # remove worst entry

    # ----- systematic grids -----
    # edge exponential
    for gamma in np.logspace(-2, 2, num=120):
        if time.time() >= phase1_deadline:
            break
        vec = _edge_exponential(n, gamma)
        val = _fast_objective(vec, n, conv_fft_len)
        _insert_candidate(val, vec)

    # edge power‑law
    for alpha in np.linspace(-4.0, -0.1, num=120):
        if time.time() >= phase1_deadline:
            break
        vec = _edge_power_law(n, alpha)
        val = _fast_objective(vec, n, conv_fft_len)
        _insert_candidate(val, vec)

    # beta‑pdf (coarse grid)
    beta_vals = [0.3, 0.7, 1.0, 2.0, 5.0]
    for a in beta_vals:
        for b in beta_vals:
            if time.time() >= phase1_deadline:
                break
            vec = _beta_shape(n, a, b)
            val = _fast_objective(vec, n, conv_fft_len)
            _insert_candidate(val, vec)

    # double Gaussian
    for sigma in np.linspace(0.02, 0.5, num=80):
        if time.time() >= phase1_deadline:
            break
        vec = _double_gaussian(n, sigma)
        val = _fast_objective(vec, n, conv_fft_len)
        _insert_candidate(val, vec)

    # sinusoid (no extra param) and uniform
    if time.time() < phase1_deadline:
        vec = _sinusoid(n)
        _insert_candidate(_fast_objective(vec, n, conv_fft_len), vec)
    if time.time() < phase1_deadline:
        vec = _uniform(n)
        _insert_candidate(_fast_objective(vec, n, conv_fft_len), vec)

    # ----- random mixtures -----
    while time.time() < phase1_deadline:
        vec = _sample_mixed_shape(rng, n)
        val = _fast_objective(vec, n, conv_fft_len)
        _insert_candidate(val, vec)

    # Re‑evaluate the K best candidates with the official evaluator
    best_seq = None
    best_val = np.inf
    for fast_val, cand in pool:
        real_val = evaluate_sequence(cand.tolist())
        if real_val < best_val:
            best_val = real_val
            best_seq = cand.copy()
            print(f"[Phase1] candidate evaluated to {real_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deep refinement of the current best (≈55 % of remaining time)
    if best_seq is not None and time.time() < deadline - 8.0:
        # allocate roughly half of the remaining budget to Adam,
        # quarter to subgradient, quarter to PGD.
        remain = deadline - time.time()
        sub_deadline_adam = time.time() + 0.5 * remain
        a_adam, v_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline_adam,
            max_iters=400_000,
            lr_start=0.04,
        )
        if v_adam < best_val:
            best_val, best_seq = v_adam, a_adam.copy()
            print(f"[Refine‑Adam] new best {v_adam:.6f}")

        # sub‑gradient
        if time.time() < deadline - 5.0:
            sub_deadline_sub = time.time() + 0.25 * (deadline - time.time())
            a_sub, v_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline_sub,
                max_iters=300_000,
                eta0=0.8,
            )
            if v_sub < best_val:
                best_val, best_seq = v_sub, a_sub.copy()
                print(f"[Refine‑Sub] new best {v_sub:.6f}")

        # projected‑gradient descent
        if time.time() < deadline - 2.0:
            sub_deadline_pgd = time.time() + 0.2 * (deadline - time.time())
            a_pgd, v_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline_pgd,
                max_iters=200_000,
                eta0=0.4,
            )
            if v_pgd < best_val:
                best_val, best_seq = v_pgd, a_pgd.copy()
                print(f"[Refine‑PGD] new best {v_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap local moves (last few seconds)
    if best_seq is not None and time.time() < deadline - 0.5:
        a_loc, v_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if v_loc < best_val:
            best_val, best_seq = v_loc, a_loc.copy()
            print(f"[Local‑final] improved best to {v_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback to uniform if everything failed
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to the evaluator's domain (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
