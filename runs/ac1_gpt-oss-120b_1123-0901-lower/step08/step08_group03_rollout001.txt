def _adam_basis_optimize(p0, n, conv_fft_len, corr_fft_len, deadline,
                         K=15, max_iters=500_000, lr=0.04, lam_start=0.1, lam_factor=1.02,
                         noise_sigma=0.02):
    # Precompute basis matrix of shape (n, d)
    d = 2*K + 1
    i = np.arange(n, dtype=np.float64)
    theta = 2*np.pi * i / n
    B = np.empty((n,d), dtype=np.float64)
    B[:,0] = 1.0
    col = 1
    for k in range(1, K+1):
        B[:,col] = np.cos(k*theta)
        col+=1
        B[:,col] = np.sin(k*theta)
        col+=1
    # Adam state for p
    m = np.zeros_like(p0)
    v = np.zeros_like(p0)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    p = p0.copy()
    best_a = None
    best_val = float('inf')
    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        logits = B @ p
        # softmax for a
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Basis Adam it={it}] new best {val:.6f}")
        # smooth-max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        
        # grad_a as before
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient w.r.t logits: a*(grad_a - avg)
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)
        # gradient w.r.t parameters p
        grad_p = B.T @ grad_logits   # shape (d,)
        # add noise
        if noise_sigma > 0:
            grad_p += np.random.normal(scale=noise_sigma, size=grad_p.shape)
        # Adam update on p
        m = beta1 * m + (1-beta1) * grad_p
        v = beta2 * v + (1-beta2) * (grad_p**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p -= step
        # temperature schedule
        lam = min(lam*lam_factor, 1e6)
        # learning rate decay (optional)
        if it % 100000 == 0 and it > 0:
            lr *= 0.8
        it += 1
    if best_a is None:
        # fallback: compute a from p at the end
        logits = B @ p
        exp_logits = np.exp(logits - logits.max())
        a = exp_logits/exp_logits.sum()
        best_a = a
        best_val = _eval_objective(a, n, conv_fft_len)
    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) with a low
value of  F(a) = 2·n·max(a∗a) / (∑a)² .

Algorithm
---------

1.  Choose a fixed length `n` (4096) and pre‑compute FFT lengths.
2.  Evaluate a few classical windows (Hann, Hamming, Blackman, Kaiser,
    Chebyshev, Tukey, Gaussian) – they are cheap strong baselines.
3.  Represent a candidate by a low‑dimensional Fourier series

        logits[i] = p₀ + Σ_{k=1}^{K} (p_{2k‑1}·cos(2πk i/n) +
                                    p_{2k  }·sin(2πk i/n))

   The actual step heights are a soft‑max of the logits
   (guarantees non‑negativity and unit sum).

   Optimise the parameter vector `p` with Adam on the **smooth‑max**
   surrogate (log‑sum‑exp) of the convolution peak.  Because the
   dimension of `p` (≈2·K+1, K=15) is tiny, many optimisation
   iterations fit into the time budget and the optimiser explores the
   global shape rather than getting stuck in a high‑dimensional local
   minimum.

4.  Run a few random restarts of the low‑dimensional Adam optimiser,
    keep the best sequence.
5.  Finish with a cheap hill‑climbing (move a random fraction of mass
    between two coordinates) to polish the result.
6.  Return the best sequence found (as a plain Python list).  The
    sequence is already normalised to sum = 1, all entries are ≥ 0,
    and values are clipped to the range accepted by `evaluate_sequence`.

The whole routine respects a 1000 s budget; the best candidate seen so
far is always reported.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear (full) convolution a * a using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑n vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_basis_optimize(
    p0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    K: int = 15,
    max_iters: int = 400_000,
    lr: float = 0.04,
    lam_start: float = 0.08,
    lam_factor: float = 1.02,
    noise_sigma: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in a low‑dimensional Fourier basis.
    Parameters:
        p0 – initial parameter vector (shape 2*K+1)
        K  – number of Fourier frequencies (controls smoothness)
    Returns:
        (best_a, best_val) – the best sequence encountered and its exact value.
    """
    # Pre‑compute the basis matrix B (n × d)
    d = 2 * K + 1
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * i / n
    B = np.empty((n, d), dtype=np.float64)
    B[:, 0] = 1.0
    col = 1
    for k in range(1, K + 1):
        B[:, col] = np.cos(k * theta)
        col += 1
        B[:, col] = np.sin(k * theta)
        col += 1

    p = p0.astype(np.float64).copy()
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_a = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # ----- logits → probabilities (soft‑max) -----
        logits = B @ p
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()    # guaranteed sum=1

        # ----- exact objective (for bookkeeping) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() is 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Basis Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t a (exact‑max surrogate) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- back‑prop through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # ----- chain rule to parameters p -----
        grad_p = B.T @ grad_logits

        # optional Gaussian noise on the gradient
        if noise_sigma > 0:
            grad_p += np.random.normal(scale=noise_sigma, size=grad_p.shape)

        # ----- Adam update on p -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        p -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.85

        # shake the parameters close to the deadline
        if it % 4_000 == 0 and time.time() > deadline - 60:
            p += np.random.normal(scale=0.02, size=p.shape)

        it += 1

    # If nothing was recorded (should not happen) fall back to final p
    if best_a is None:
        logits = B @ p
        exp_logits = np.exp(logits - logits.max())
        best_a = exp_logits / exp_logits.sum()
        best_val = _eval_objective(best_a, n, conv_fft_len)

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass
    from one coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

        # classic symmetric windows
        try_window(win.hann(n))
        try_window(win.hamming(n))
        try_window(win.blackman(n))

        # Kaiser – sweep a few betas
        for beta in (0, 2, 4, 6, 8, 10, 12, 14):
            try_window(win.kaiser(n, beta))

        # Chebyshev – sweep attenuation levels
        for att in (30, 50, 70, 90, 110):
            try_window(win.chebwin(n, at=att))

        # Tukey – sweep the cosine‑taper parameter
        for alpha in np.linspace(0, 1, 11):
            try_window(win.tukey(n, alpha))

        # Gaussian – sweep standard deviation (as fraction of length)
        for frac in np.linspace(0.02, 0.20, 10):
            std = frac * n
            try_window(win.gaussian(n, std))

    # ------------------------------------------------------------
    # allocate the bulk of the budget to the low‑dimensional Adam optimiser
    remaining = deadline - time.time()
    if remaining > 30:   # leave some time for polishing
        # number of restarts – adjust according to remaining time
        restarts = 4
        for r in range(restarts):
            if time.time() >= deadline - 5:
                break
            K = 15                     # Fourier frequency budget
            d = 2 * K + 1
            p0 = np.random.normal(scale=0.01, size=d)  # small random init
            a_opt, val_opt = _adam_basis_optimize(
                p0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                K=K,
                max_iters=300_000,
                lr=0.04,
                lam_start=0.08,
                lam_factor=1.02,
                noise_sigma=0.02,
            )
            update_best(a_opt)

    # ------------------------------------------------------------
    # final cheap hill‑climb to polish the best candidate
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        update_best(a_loc)

    # ------------------------------------------------------------
    # fallback – ensure a non‑empty valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
