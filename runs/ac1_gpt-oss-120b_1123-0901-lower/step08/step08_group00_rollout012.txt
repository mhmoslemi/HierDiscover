"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Parameterised random search (≈20 % of time).**  
   Generate candidates from several analytic families
   (symmetric exponentials, centred exponentials,
   cosine‐powers, beta‑shapes and mixtures).  Each candidate is
   normalised to a probability simplex and evaluated with a fast
   FFT‑based objective.  The best survivor becomes the warm start.

2. **Smooth‑max Adam (≈30 %).**  
   Minimise a log‑sum‑exp surrogate for the convolution maximum.
   The temperature (λ) is increased slowly, logits are updated with
   Adam and the best exact objective encountered is recorded.

3. **Exact‑max sub‑gradient on the simplex (≈30 %).**  
   A Euclidean projected‑sub‑gradient method (mirror descent) is run
   with step‑size ηₜ=η₀/√t.  The gradient of the true max‑convolution
   is obtained from the active set of indices attaining the maximal
   value.

4. **Simulated‑annealing refinement (≈15 %).**  
   Random mass transfers between two positions are accepted with
   Metropolis probability exp(−Δ/T); the temperature decays linearly.

5. **Optional up‑sampling (if time permits).**  
   Interpolate the current best vector to length 8192 and repeat a
   short Adam + sub‑gradient phase.

All phases respect the global deadline and keep the best feasible
vector found.  The routine always returns a list of non‑negative
floats; it never raises an exception, guaranteeing a finite score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradient.
    Result length = len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of F(a) = 2 n max(a*a)/ (sum a)^2."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the *smooth* max (log‑sum‑exp) surrogate.
    Returns the best exact solution seen.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)  # start on simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                     # a bit more conservative
    lam = 0.05                    # start temperature
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter near the end
        if it % 5000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # decay learning rate gently
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_proj(a0: np.ndarray,
                               n: int,
                               conv_fft_len: int,
                               corr_fft_len: int,
                               deadline: float,
                               max_iters: int = 500_000,
                               eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient with Euclidean projection onto the simplex.
    Returns the best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Proj‑SG it={it:07d}] new best {val:.6f}")

        # active set of the maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # safeguard against numerical collapse
        if a.sum() < 1e-12 or np.isnan(a).any():
            a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient with exponentiated‑gradient (mirror descent).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Exp‑SG it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = float(2.0 * n * conv.max() / (a.sum() ** 2))
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = float(2.0 * n * conv2.max() / (a2.sum() ** 2))

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _candidate_from_params(n: int) -> np.ndarray:
    """
    Randomly draw a candidate from a collection of analytic families.
    Each family respects a≥0 and is normalised to sum‑1.
    """
    family = np.random.choice(5)
    idx = np.arange(n, dtype=np.float64)

    if family == 0:  # symmetric edge‑exponential
        alpha = 10 ** np.random.uniform(-2, 1)      # 0.01 … 10
        beta = np.random.uniform(0.2, 4.5)
        d = np.minimum(idx, n - 1 - idx) / (n / 2)
        a = np.exp(-alpha * (d ** beta))
    elif family == 1:  # centre exponential
        alpha = 10 ** np.random.uniform(-2, 1)
        beta = np.random.uniform(0.2, 4.5)
        d = np.abs(idx - (n - 1) / 2) / (n / 2)
        a = np.exp(-alpha * (d ** beta))
    elif family == 2:  # cosine‑power (non‑negative half‑wave)
        gamma = np.random.uniform(0.5, 12.0)
        a = np.cos(np.pi * (idx + 0.5) / n)
        a = np.maximum(a, 0.0) ** gamma
    elif family == 3:  # beta‑distribution shape
        p = np.random.uniform(0.0, 6.0)
        q = np.random.uniform(0.0, 6.0)
        x = idx / (n - 1)
        a = (x ** p) * ((1.0 - x) ** q)
    else:  # mixture of edge‑ and centre‑exponential
        alpha = 10 ** np.random.uniform(-2, 1)
        beta = np.random.uniform(0.2, 4.5)
        d_edge = np.minimum(idx, n - 1 - idx) / (n / 2)
        edge = np.exp(-alpha * (d_edge ** beta))
        d_center = np.abs(idx - (n - 1) / 2) / (n / 2)
        cen = np.exp(-alpha * (d_center ** beta))
        w = np.random.rand()
        a = w * edge + (1.0 - w) * cen

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        # fallback to uniform if numerical disaster
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 0 – random parametric exploration (~20 % of budget)
    phase0_dead = time.time() + (deadline - time.time()) * 0.20
    best_seq = warm.copy() if warm is not None else None
    best_val = float("inf") if best_seq is None else _objective(best_seq, n, conv_fft_len)

    while time.time() < phase0_dead:
        cand = _candidate_from_params(n)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase0] new best {val:.6f}")

        # occasionally throw in a sparse spike pattern (helps escape)
        if np.random.rand() < 0.03:
            k = np.random.choice([32, 64, 128, 256, 512])
            a_sp = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            a_sp[pos] = np.random.dirichlet(np.ones(k))
            a_sp /= a_sp.sum()
            val_sp = _objective(a_sp, n, conv_fft_len)
            if val_sp < best_val:
                best_val = val_sp
                best_seq = a_sp.copy()
                print(f"[Phase0‑spike] new best {val_sp:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam refinement (~30 %)
    if best_seq is not None and time.time() < deadline - 40:
        phase1_dead = time.time() + (deadline - time.time()) * 0.30
        print("[Phase1] Adam refinement")
        a_adam, val_adam = _adam_optimize(best_seq, n,
                                          conv_fft_len, corr_fft_len,
                                          phase1_dead)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase1] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max sub‑gradient (projected) (~30 %)
    if best_seq is not None and time.time() < deadline - 30:
        phase2_dead = time.time() + (deadline - time.time()) * 0.30
        print("[Phase2] Projected sub‑gradient")
        a_proj, val_proj = _subgradient_optimize_proj(best_seq, n,
                                                      conv_fft_len,
                                                      corr_fft_len,
                                                      phase2_dead)
        if val_proj < best_val:
            best_val = val_proj
            best_seq = a_proj.copy()
            print(f"[Phase2] Proj‑SG improved to {val_proj:.6f}")

        # a secondary exp‑gradient run (often refines further)
        if time.time() < deadline - 20:
            phase2b_dead = time.time() + (deadline - time.time()) * 0.20
            print("[Phase2b] Exponentiated‑gradient sub‑gradient")
            a_exp, val_exp = _subgradient_optimize_expgrad(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          phase2b_dead)
            if val_exp < best_val:
                best_val = val_exp
                best_seq = a_exp.copy()
                print(f"[Phase2b] Exp‑SG improved to {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing refinement (~15 %)
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase3] Simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(best_seq, n,
                                          conv_fft_len,
                                          deadline)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase3] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid (if time permits)
    if best_seq is not None and time.time() + 30 < deadline:
        print("[Upsample] Trying finer grid (8192 points)")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on big grid
        sub_dead_big = min(deadline, time.time() + 8.0)
        a_big_adam, _ = _adam_optimize(a_big, n_big,
                                       conv_fft_len_big,
                                       corr_fft_len_big,
                                       sub_dead_big,
                                       max_iters=200_000)

        # longer sub‑gradient on big grid
        if time.time() < deadline - 5:
            a_big_ref, val_big = _subgradient_optimize_proj(a_big_adam, n_big,
                                                            conv_fft_len_big,
                                                            corr_fft_len_big,
                                                            deadline)
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big_ref.copy()
                print(f"[Upsample] refined best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
