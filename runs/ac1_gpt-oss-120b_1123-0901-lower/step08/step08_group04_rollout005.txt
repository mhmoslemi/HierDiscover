"""
Propose a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²)

The algorithm builds a large diverse pool of warm‑starts,
picks the best one, refines it with a longer Adam phase on a
log‑sum‑exp surrogate, follows with many sub‑gradient updates,
and finally performs an aggressive simulated‑annealing with a
linearly decreasing temperature and step‑size.  A short
fine‑tuning pass of tiny mass transfers is executed at the
very end.  All phases respect a user supplied time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety checks) ------------------------------------

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm start ------------------------------------------------------------

def _warm_start(n: int):
    """If a previous solution exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange a sorted vector so that large values are opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _edge_power_candidate(n: int, p: float = 1.0) -> np.ndarray:
    """Heavy at the edges, power‑law decay towards the centre."""
    half = n // 2
    idx = np.arange(half, dtype=np.float64) + 1.0
    left = idx ** (-p)
    if n % 2 == 0:
        full = np.concatenate([left, left[::-1]])
    else:
        full = np.concatenate([left, [0.0], left[::-1]])
    full = np.maximum(full, 0.0)
    full /= full.sum()
    return full

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 500_000,
                   lr: float = 0.07,
                   lam_start: float = 1.0,
                   lam_factor: float = 1.07) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # soft‑max → simplex

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c               # denominator is 1 (sum=1)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            # occasional progress report
            if it % 5_000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Adam on logits (implicit simplex via soft‑max)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # jitter occasionally
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # weights for the exact max (active set)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # re‑normalize onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    # update all entries that involve a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term (appears once)
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(a0: np.ndarray, time_limit: float,
                    start_temp: float = 0.03,
                    init_max_frac: float = 0.4,
                    final_max_frac: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing using the incremental convolution
    update.  The temperature and the maximal transferred fraction decay
    linearly to zero.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    while time.time() < time_limit:
        # linear schedules
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        total = time_limit - start
        fraction = remaining / total if total > 0 else 0.0
        temp = start_temp * fraction
        max_frac = final_max_frac + (init_max_frac - final_max_frac) * fraction

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        # occasional progress report
        if int(elapsed) % 30 == 0 and elapsed > 0:
            print(f"[SA] t={elapsed:.0f}s best_conv_max={best_val:.6f}")

    return best_a, best_val


def _final_fine_tune(a0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Very small mass‑transfer moves (≤0.1 % of source) that are only
    accepted if they improve the exact maximum.  Intended for the last
    few seconds of the budget.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))

    best_a = a.copy()
    best_val = conv.max()

    start = time.time()
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * 0.001 * np.random.rand()   # ≤0.1 %
        if delta <= 0.0:
            continue

        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        # light progress output
        if (time.time() - start) % 30 < 0.1:
            print(f"[Fine‑tune] best_conv_max={best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 offers a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a large pool of initial candidates
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few rotations of the warm start (sometimes useful)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))

    # Dirichlet draws with a wide range of α
    alphas = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 2.0]
    for alpha in alphas:
        for _ in range(12):                     # 12 draws each → ~120 candidates
            candidates.append(_random_dirichlet_candidate(n, alpha))

    # anti‑pair candidates (same alphas)
    for alpha in alphas:
        for _ in range(8):
            candidates.append(_anti_pair_candidate(n, alpha))

    # edge‑focused shapes
    for gamma in (0.1, 0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # power‑law edge shapes
    for p in (0.5, 1.0, 1.5, 2.0):
        candidates.append(_edge_power_candidate(n, p))

    candidates.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Evaluate the pool and keep the best starter
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:                     # safety net
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:03d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) Continuous optimisation (Adam + Sub‑gradient)
    # allocate a few seconds for the final SA phases
    sa_deadline = deadline - 30.0
    if time.time() < sa_deadline:
        # ---- Adam phase (smooth‑max)
        adam_deadline = min(sa_deadline - 20.0,
                            time.time() + 180.0)   # cap at ≈180 s
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # ---- Sub‑gradient refinement
        sub_deadline = min(sa_deadline - 5.0,
                           time.time() + 300.0)    # cap at ≈300 s
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=600_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Long incremental simulated‑annealing local search
    if time.time() < deadline - 10.0:
        sa_seq, sa_conv_max = _incremental_sa(
            best_seq,
            time_limit=deadline - 10.0,
            start_temp=0.03,
            init_max_frac=0.4,
            final_max_frac=0.01,
        )
        sa_val = 2.0 * n * sa_conv_max
        if sa_val < best_val:
            best_val = sa_val
            best_seq = sa_seq.copy()
            print(f"[SA] improved to {sa_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final fine‑tuning (tiny moves) if any time left
    if time.time() < deadline - 2.0:
        ft_seq, ft_conv_max = _final_fine_tune(
            best_seq,
            time_limit=deadline - 2.0,
        )
        ft_val = 2.0 * n * ft_conv_max
        if ft_val < best_val:
            best_val = ft_val
            best_seq = ft_seq.copy()
            print(f"[Fine‑tune] improved to {ft_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Final safety checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq /= total                     # enforce sum ≈ 1

    # report the exact bound obtained from evaluate_sequence
    try:
        final_bound = evaluate_sequence(best_seq.tolist())
        print(f"Final reported bound (evaluate_sequence) = {final_bound:.6f}")
    except Exception as e:
        print(f"Unable to call evaluate_sequence: {e}")

    print(f"Final best internal value {best_val:.6f} (length {n})")
    return best_seq.tolist()
