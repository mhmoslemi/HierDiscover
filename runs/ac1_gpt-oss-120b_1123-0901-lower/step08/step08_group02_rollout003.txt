        it += 1

"""
Gradient‑Balancing optimizer for the step‑function problem.

The core idea:
* The objective is 2·n·max(a∗a) / (sum a)².  After normalising
  sum(a)=1 the objective equals max(a∗a).
* At a current a the active offsets (where the convolution attains its
  maximum) define a sub‑gradient g = 2·(w★a), where w is uniform over the
  active set.
* Moving mass from entries with large g to entries with small g reduces
  the active convolution value.  We take a whole‑vector step in the
  direction –(g−mean(g)) and project back onto the simplex.
* The procedure is run many times from diverse warm‑starts (including the
  previous best sequence).  A short diversification phase explores many
  starts; a final long refinement focuses on the current best.
* All work respects the 1000 s budget; the best feasible sequence seen so
  far is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of length len(a).
    Result i = Σ_k w[k]·a[i+k-(n‑1)],  i=0..n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _gradient_optimize(a0: np.ndarray,
                       n: int,
                       conv_fft_len: int,
                       corr_fft_len: int,
                       deadline: float,
                       max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple projected‑gradient descent on the exact max‑convolution.
    Returns the best vector (sum≈1) encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    # objective = max(a∗a) because sum(a)=1
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            if it % 2_000 == 0:
                print(f"[Grad it={it:06d}] new best {best_val:.6f}")

        # build the active‑set weight vector w
        active = np.isclose(conv, max_c, atol=1e-12)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # direction that moves mass from large‑grad entries to small‑grad entries
        direction = -(grad - np.mean(grad))

        # back‑tracking line search (max 5 reductions)
        step = 0.5
        improved = False
        for _ in range(5):
            a_candidate = a + step * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s == 0.0:
                step *= 0.5
                continue
            a_candidate /= s
            conv_cand = _full_convolution(a_candidate, conv_fft_len)
            max_cand = conv_cand.max()
            if max_cand < max_c - 1e-15:       # genuine improvement
                a = a_candidate
                improved = True
                break
            step *= 0.5

        # if line‑search failed, perform a tiny random jitter to escape flat regions
        if not improved and np.random.rand() < 0.01:
            logits = np.log(a + 1e-12)
            logits += np.random.randn(n) * 0.03
            a = np.exp(logits - np.max(logits))
            a /= a.sum()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size (fixed to 4096 – the size used by the reference code)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # --------------------------------------------------------------
    # build a modest but diverse pool of initial vectors
    pool = []

    if w is not None:
        pool.append(w.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(w, shift))

    # random Dirichlet draws with random concentration
    for _ in range(8):
        alpha = np.random.rand() * 4.0 + 0.05   # α ∈ [0.05,4.05]
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    # exponential and uniform primitives
    pool.append(np.random.exponential(scale=1.0, size=n))
    pool.append(np.full(n, 1.0 / n))

    # sinusoid‑shaped candidate (always non‑negative)
    sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_shape = np.maximum(sin_shape, 0.0)
    sin_shape /= sin_shape.sum()
    pool.append(sin_shape)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – quick exploration of many starts (≈30 % of the budget)
    phase1_budget = (deadline - time.time()) * 0.3
    per_start = max(0.5, phase1_budget / max(1, len(pool)))

    for idx, cand in enumerate(pool):
        if time.time() >= deadline - 5.0:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 5.0, time.time() + per_start)
        a_opt, val_opt = _gradient_optimize(a0, n,
                                            conv_fft_len, corr_fft_len,
                                            local_deadline, max_iters=120_000)

        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Phase 1 cand {idx}] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deep refinement of the current best (≈65 % of budget)
    while time.time() < deadline - 5.0:
        if best_seq is None:
            # fallback uniform if nothing succeeded yet
            best_seq = np.full(n, 1.0 / n)

        a_ref, val_ref = _gradient_optimize(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            deadline, max_iters=300_000)

        if val_ref < best_val:
            best_val, best_seq = val_ref, a_ref.copy()
            print(f"[Phase 2] new best {val_ref:.6f}")

        # occasional random shake to help escape possible plateaus
        if np.random.rand() < 0.15:
            logits = np.log(best_seq + 1e-12)
            logits += np.random.randn(n) * 0.04
            a_shake = np.exp(logits - np.max(logits))
            a_shake = a_shake / a_shake.sum()
            a_shake_opt, val_shake = _gradient_optimize(a_shake, n,
                                                        conv_fft_len, corr_fft_len,
                                                        deadline, max_iters=80_000)
            if val_shake < best_val:
                best_val, best_seq = val_shake, a_shake_opt.copy()
                print(f"[Shake] improved to {val_shake:.6f}")

    # --------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
