            delta = best_seq[donor_idx] * max_frac * np.random.rand()
            if delta <= 1e-12:
                continue
            # apply move
            _apply_delta(best_seq, conv, donor_idx, -delta)
            _apply_delta(best_seq, conv, j, delta)
            new_max = conv.max()
            diff = new_max - cur_max
            # temperature schedule
            elapsed = time.time() - start
            remaining = deadline - time.time()
            t_total = deadline - start
            temp = start_temp * max(0.0, remaining / t_total)
            accept = diff <= 0 or (temp > 0 and np.random.rand() < np.exp(-diff / temp))
            if accept:
                cur_max = new_max
                if new_max < best_local_val:
                    best_local_val = new_max
                    best_seq_local = best_seq.copy()
            else:
                # revert
                _apply_delta(best_seq, conv, j, -delta)
                _apply_delta(best_seq, conv, donor_idx, delta)
            # occasional random move to escape
            if np.random.rand() < 0.001:
                # random donor i not donor_idx; random move
                i = np.random.randint(n)
                if best_seq[i] < 1e-12:
                    continue
                j_rand = np.random.randint(n - 1)
                if j_rand >= i:
                    j_rand += 1
                delta2 = best_seq[i] * max_frac * np.random.rand()
                _apply_delta(best_seq, conv, i, -delta2)
                _apply_delta(best_seq, conv, j_rand, delta2)
                new_max2 = conv.max()
                diff2 = new_max2 - cur_max
                accept2 = diff2 <= 0 or (temp > 0 and np.random.rand() < np.exp(-diff2 / temp))
                if accept2:

                    if new_max2 < best_local_val:

                        best_seq_local = best_seq.copy()
                else:
                    _apply_delta(best_seq, conv, j_rand, -delta2)
                    _apply_delta(best_seq, conv, i, delta2)
    # final
    final_seq = best_seq_local
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    total_sum = final_seq.sum()
    if total_sum < 1e-12:
        final_seq = np.full(n, 1.0 / n)
    else:
        final_seq /= total_sum
    # logging
    print(f"Final conv.max {best_local_val:.8f}, objective approx {2.0*n*best_local_val:.6f}")
    return final_seq.tolist()

"""
Propose a step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm consists of three phases:

1. **Diverse warm‑starts** – a small pool of candidates
   (uniform, Dirichlet samples, anti‑paired Dirichlet, triangular,
   Gaussian‑edges, optional warm start from a previous solution) is built,
   the best one (lowest exact objective) is kept.

2. **Fast gradient descent** – a short Adam phase on the smooth‑max
   (log‑sum‑exp) surrogate followed by an exact‑max sub‑gradient phase.
   Both use FFT‑based convolutions and stay on the probability simplex.

3. **Targeted simulated‑annealing** – the best vector is refined by a large
   number of inexpensive incremental moves.  Each move transfers a random
   fraction of mass from an index that currently contributes most to the
   maximal convolution entry to a random low‑mass index.  Moves are accepted
   if they improve the objective or with a temperature‑dependent probability.
   The incremental update of the convolution is O(n) via slice operations,
   allowing tens of millions of evaluations within the remaining budget.

All vectors are kept non‑negative and normalised to sum≈1, therefore the
returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant) ------------------------------------

def _objective(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _warm_start(n: int):
    """If a previous solution exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange a sorted vector so that large values are opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = _random_dirichlet_candidate(n, alpha)
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.07,
                   lam_start: float = 1.0,
                   lam_factor: float = 1.07) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c                     # sum(a) == 1
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental convolution update (used by the long SA)

def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _targeted_sa(a0: np.ndarray,
                 conv0: np.ndarray,
                 deadline: float,
                 start_temp: float = 0.03,
                 max_frac: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style descent that focusses the mass transfer on
    indices that currently contribute most to the maximal convolution entry.
    """
    a = a0.copy()
    conv = conv0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        # 1. locate the current worst convolution index
        k = int(conv.argmax())

        # 2. pick the donor – the index that yields the largest product a[i]·a[k‑i]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_range = np.arange(i_min, i_max + 1)
        prod = a[i_range] * a[k - i_range]
        donor = i_range[int(prod.argmax())]

        # 3. choose a recipient from a low‑mass region (simple uniform draw)
        rec = np.random.randint(n - 1)
        if rec >= donor:
            rec += 1

        # 4. transfer a random fraction of the donor's mass
        frac = np.random.rand() * max_frac
        delta = a[donor] * frac
        if delta <= 0.0:
            continue

        # apply the move
        _apply_delta(a, conv, donor, -delta)
        _apply_delta(a, conv, rec, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear temperature schedule
        remaining = max(0.0, deadline - time.time())
        total = deadline - start
        temp = start_temp * (remaining / total) if total > 0 else 0.0

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, rec, -delta)
            _apply_delta(a, conv, donor, delta)

        it += 1
        if it % 100_000 == 0:
            print(f"[Targeted SA it={it:06d}] best conv.max={best_val:.8f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    The routine builds a diverse pool of starts, runs a short Adam +
    sub‑gradient phase, and finishes with a long targeted simulated‑annealing
    that focuses on the current maximal convolution entry.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a pool of diverse initial candidates
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few rotated copies – sometimes useful
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))

    # Dirichlet draws with several concentration parameters
    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        for _ in range(12):               # 12 draws each → 72 samples
            d = _random_dirichlet_candidate(n, alpha)
            candidates.append(d)
            candidates.append(_anti_pair_arrangement(d))

    candidates.append(_triangle_candidate(n))
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # --------------------------------------------------------------
    # 2) Pick the best starter (cheap exact evaluation)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)                 # exact objective (2·n·max/∑²)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 3) Continuous optimisation (Adam + sub‑gradient)
    # Give the first phases a modest share of the budget.
    sa_deadline = deadline - 30.0                # reserve ~30 s for final SA
    if time.time() < sa_deadline:
        # ---- Adam phase (smooth‑max surrogate)
        adam_deadline = min(sa_deadline - 20.0,
                            time.time() + 150.0)    # ≤150 s
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # ---- Sub‑gradient refinement
        sub_deadline = min(sa_deadline - 5.0,
                           time.time() + 250.0)      # ≤250 s
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=300_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Long targeted simulated‑annealing local search
    if time.time() < deadline - 5.0:
        # initialise convolution for the current best sequence
        conv0 = _full_convolution(best_seq, conv_fft_len)
        sa_seq, sa_val = _targeted_sa(
            best_seq,
            conv0,
            time_limit=deadline - 5.0,
            start_temp=0.04,
            max_frac=0.4,
        )
        if sa_val < best_val:
            best_val = sa_val
            best_seq = sa_seq.copy()
            print(f"[Targeted SA] improved to conv.max={sa_val:.8f}")

    # --------------------------------------------------------------
    # 5) Final safety checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq /= total                     # ensure sum ≈ 1 (scale‑invariant)

    # reporting for the log
    print(f"Final conv.max {best_val:.8f} → objective ≈ {2.0 * n * best_val:.6f}")
    print(f"Sequence length = {len(best_seq)}")
    return best_seq.tolist()
