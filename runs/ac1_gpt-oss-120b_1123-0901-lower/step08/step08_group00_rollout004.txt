def _flat_spectrum_candidate(a0: np.ndarray, n: int, rng: np.random.Generator, proj_iters: int = 30) -> np.ndarray:
    '''
    Starting from a non‑negative vector a0 (sum=1), perform alternating projections
    onto the set of vectors with flat magnitude spectrum (constant |FFT|) and the
    non‑negative simplex. Returns the resulting vector.
    '''
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    for _ in range(proj_iters):
        # zero‑pad to length L
        padded = np.zeros(L, dtype=np.float64)
        padded[:n] = a
        # forward FFT (real)
        A = np.fft.rfft(padded)
        # preserve DC component to keep sum (will be renorm later)
        dc = A[0]
        # compute target magnitude for non‑DC frequencies
        mag_target = np.mean(np.abs(A[1:]))
        # replace magnitude
        phase = np.angle(A[1:])
        A[1:] = mag_target * np.exp(1j * phase)
        # Inverse transform
        padded = np.fft.irfft(A, n=L)
        a = padded[:n]
        # project onto non‑negative simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    return a

"""
Step‑function optimizer – enhanced candidate generation and focused refinement.

The optimizer works in three stages:

1. **Diverse start pool** – builds a large pool of seeds consisting of
   Dirichlet draws, symmetric Beta‑shaped families, sparse spike patterns
   and, importantly, *flat‑spectrum* candidates obtained via an alternating
   projection that forces the magnitude of the (zero‑padded) FFT to be (almost)
   constant.  All seeds are projected onto the probability simplex.

2. **Pre‑selection** – the pool is evaluated quickly (FFT‑based convolution)
   and the best K ≈ 3 candidates are kept for deep optimisation.

3. **Deep refinement** – each of the selected seeds is refined with a long
   exponentiated‑gradient sub‑gradient run (the exact max‑convolution sub‑gradient)
   followed, if time permits, by a short simulated‑annealing local search.
   The best vector found overall is returned.

The whole routine respects the given time budget (default 1000 s) and
always returns a non‑empty, non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    eps = 1e-15
    logits = np.log(a0 + eps)                     # start on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style local search."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Additional candidate generators
def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    # random‑position sparse spikes
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _flat_spectrum_candidate(a0: np.ndarray, n: int, rng: np.random.Generator,
                             proj_iters: int = 30) -> np.ndarray:
    """
    Alternating projection: enforce non‑negative simplex and (approximately) flat
    magnitude of the zero‑padded FFT.  The result tends to have a self‑convolution
    that is close to uniform.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)

    for _ in range(proj_iters):
        # zero‑pad
        padded = np.zeros(L, dtype=np.float64)
        padded[:n] = a

        # FFT (real‑to‑complex)
        A = np.fft.rfft(padded)

        # Preserve the DC component (sum) but flatten all other magnitudes
        mag_target = np.mean(np.abs(A[1:]))  # average non‑DC magnitude
        phase = np.angle(A[1:])
        A[1:] = mag_target * np.exp(1j * phase)

        # Inverse transform
        padded = np.fft.irfft(A, n=L)

        # Extract first n entries and project onto the simplex
        a = padded[:n]
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

    return a


def _beta_shape_candidate(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta‑like shape: i^{α‑1}·(n‑i)^{β‑1}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = np.power(idx, alpha - 1.0) * np.power(n + 1 - idx, beta - 1.0)
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        return np.full(n, 1.0 / n)
    return shape / shape.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point used by the evaluation harness.
    Builds a large, diverse pool of seed vectors (including a novel flat‑spectrum
    family), keeps the few most promising, and refines them with sub‑gradient
    and simulated‑annealing phases.  The routine always returns a valid,
    non‑negative list before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size and FFT lengths
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)      # for convolution a*a
    corr_fft_len = _next_pow2(3 * n - 2)      # for cross‑correlation gradients

    # ------------------------------------------------------------------
    # Assemble a large pool of initial candidates
    pool: list[np.ndarray] = []

    # Warm start, if available
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # Dirichlet draws with a wide range of concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Symmetric Beta families (α,β) – includes arcsine‑like shapes
    for a_param in np.linspace(0.2, 2.0, 9):
        for b_param in np.linspace(0.2, 2.0, 9):
            pool.append(_beta_shape_candidate(n, a_param, b_param))

    # Flat‑spectrum candidates via alternating projections
    for _ in range(8):
        a0 = np.random.dirichlet(np.ones(n))
        pool.append(_flat_spectrum_candidate(a0, n, np.random.default_rng(), proj_iters=30))

    # Add the sparse spike patterns from the original code
    pool.extend(_spike_candidates(n))

    # ------------------------------------------------------------------
    # Quick evaluation of the pool; keep the best few
    top_k = 3
    top: list[tuple[float, np.ndarray]] = []   # (value, vector)

    for a in pool:
        # sanity – enforce non‑negativity and unit sum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if len(top) < top_k:
            top.append((val, a.copy()))
            top.sort(key=lambda x: x[0])
        elif val < top[-1][0]:
            top[-1] = (val, a.copy())
            top.sort(key=lambda x: x[0])

    # If the pool was somehow empty, fall back to uniform
    if not top:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        top = [(2.0, uniform)]  # objective of uniform is exactly 2

    best_seq = top[0][1].copy()
    best_val = top[0][0]

    # ------------------------------------------------------------------
    # Deep refinement of the selected top candidates
    remaining_time = deadline - time.time()
    if remaining_time > 30.0:
        # split remaining time roughly equally among the top candidates
        per_candidate_deadline = time.time() + remaining_time / len(top)

        for init_val, init_a in top:
            # Sub‑gradient (exponentiated‑gradient) phase
            sub_dead = min(per_candidate_deadline, time.time() + 12.0)  # at most 12 s per candidate
            a_ref, val_ref = _subgradient_optimize_expgrad(
                init_a,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_dead,
                max_iters=250_000,
                eta0=0.85,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Refine] Sub‑gradient improved to {val_ref:.6f}")

            # Short annealing phase if time permits
            if time.time() < deadline - 5.0:
                ann_dead = min(deadline, time.time() + 6.0)
                a_ann, val_ann = _anneal_optimize(
                    a_ref,
                    n,
                    conv_fft_len,
                    ann_dead,
                    max_moves=30_000,
                )
                if val_ann < best_val:
                    best_val = val_ann
                    best_seq = a_ann.copy()
                    print(f"[Refine] Annealing improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final safety check & clipping
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
