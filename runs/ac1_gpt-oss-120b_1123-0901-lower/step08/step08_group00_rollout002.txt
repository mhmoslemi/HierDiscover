"""
Propose a low‑value step‑function.

The algorithm works in three stages (within the allotted 1000 s):

1. **Fast sub‑gradient descent** on the exact max‑convolution objective.
   Starting from a warm‑start (if available) or a random Dirichlet draw it
   performs a long exponentiated‑gradient run on the probability simplex.
   This gives a decent baseline (≈1.51).

2. **Linear‑programming refinement**.
   Given the current distribution `f`, we solve the LP

        minimise   t
        subject to conv(f, g) ≤ t          (all lags)
                    sum(g)   = 1
                    g ≥ 0

   The solution `g` is the distribution that, for the fixed `f`,
   yields the smallest possible maximum cross‑correlation.  We replace
   `f` by a convex combination of `f` and `g` (typically 50 % each)
   and repeat while time permits.  This step is very effective at
   shaving the remaining few‑hundredths off the score.

3. **Light simulated‑annealing tidy‑up** to escape any tiny plateau
   that the deterministic phases may leave behind.

All vectors are kept non‑negative and normalised to sum = 1 (the
objective is scale‑invariant).  The implementation uses `numpy` for
FFT‑based convolutions and `cvxpy` (ECOS) for the linear programs.
The routine always returns the best sequence found before the deadline.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation on the simplex.
    Returns the best vector encountered and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)          # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, n: int, conv_fft_len: int,
               deadline: float) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    Solve  minimise t
           s.t. conv(f, g) ≤ t,  sum(g)=1, g≥0
    Returns (g, t) if successful, otherwise (None, None).
    """
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    f_param = cp.Parameter(n, value=f, constant=True)

    # Linear convolution (f * g) ; length 2*n-1
    conv_expr = cp.conv(f_param, g)

    constraints = [
        conv_expr <= t,
        cp.sum(g) == 1,
        t >= 0,
    ]

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, warm_start=True, verbose=False)
    except Exception as e:
        print("[LP] Solver exception:", e)
        return None, None

    if g.value is None or t.value is None:
        return None, None

    g_arr = np.array(g.value).flatten()
    t_val = float(t.value)
    return g_arr, t_val


# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing style local search.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_a = a.copy()
    best_val = val

    T_start = 0.08
    T_end = 1e-4
    it = 0

    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    safety = 0.5                     # leave a small safety margin
    end_time = time.time() + budget_s - safety
    total_time = budget_s - safety

    # problem size – a moderate size keeps the LP tractable
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to the required size
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Stage 1 – long exponentiated‑gradient sub‑gradient run
    stage1_end = time.time() + 0.55 * total_time
    if warm is None:
        a_start = np.random.dirichlet(np.ones(n))
    else:
        a_start = warm.copy()

    print("[Stage 1] Sub‑gradient optimisation")
    a_best, val_best = _subgradient_optimize_expgrad(
        a_start,
        n,
        conv_fft_len,
        corr_fft_len,
        stage1_end,
        max_iters=800_000,
        eta0=1.0,
    )
    best_seq = a_best.copy()
    best_val = val_best
    print(f"[Stage 1] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Linear‑programming refinement
    stage2_end = time.time() + 0.30 * total_time
    lp_iter = 0
    while time.time() < stage2_end:
        lp_iter += 1
        g, t_val = _lp_refine(best_seq, n, conv_fft_len, stage2_end)
        if g is None:
            print("[LP] failed, terminating LP phase")
            break

        # mix old and new (helps keep the max‑auto‑correlation down)
        alpha = 0.5
        candidate = (1.0 - alpha) * best_seq + alpha * g
        candidate = np.maximum(candidate, 0.0)
        candidate /= candidate.sum()

        # evaluate both the mixed candidate and the raw LP solution
        val_cand = evaluate_sequence(candidate.tolist())
        if val_cand < best_val:
            best_val = val_cand
            best_seq = candidate.copy()
            print(f"[LP {lp_iter}] mixed improvement to {best_val:.6f}")

        # also try the pure LP solution (occasionally beneficial)
        pure = g / g.sum()
        val_pure = evaluate_sequence(pure.tolist())
        if val_pure < best_val:
            best_val = val_pure
            best_seq = pure.copy()
            print(f"[LP {lp_iter}] pure improvement to {best_val:.6f}")

        # stop early if we have already beat the target
        if best_val < 1.503:
            print("[LP] target reached, exiting LP phase")
            break

    # ------------------------------------------------------------------
    # Stage 3 – final light annealing (if time permits)
    if time.time() < end_time - 5:
        print("[Stage 3] Simulated‑annealing tidy‑up")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            end_time,
            max_moves=80_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid if generous time remains
    if time.time() + 30 < end_time:
        try:
            n_big = 4096
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            print("[Upsample] Interpolating to 4096 points")
            x_small = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            big = np.interp(x_big, x_small, best_seq)
            big = np.maximum(big, 0.0)
            big /= big.sum()

            # a quick few‑hundred‑iteration sub‑gradient pass on the larger grid
            stage_big_end = end_time
            big_opt, val_big = _subgradient_optimize_expgrad(
                big,
                n_big,
                conv_fft_len_big,
                _next_pow2(3 * n_big - 2),
                stage_big_end,
                max_iters=200_000,
                eta0=0.9,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = big_opt.copy()
                print(f"[Upsample] refined best to {best_val:.6f}")
        except Exception as e:
            print("[Upsample] failed:", e)

    # ------------------------------------------------------------------
    # Ensure return format complies with evaluator requirements
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
