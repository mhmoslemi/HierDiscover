def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]

    return w

"""
Propose a step‑function with low  F(a)=2·n·max(a∗a)/(sum(a)²).

The algorithm is a hybrid global‑local search:

* Build a diverse pool of candidates (Dirichlet draws, edge‑exponential,
  edge‑polynomial, double‑Gaussian, sinusoid, spikes, random mixtures of
  Gaussians and Beta‑shapes).  If a previous best ``height_sequence_1`` is
  available it is added to the pool.

* For each candidate run a short Adam optimisation on a smooth‑max
  (log‑sum‑exp) surrogate of the objective.  Afterwards perform an
  **additive** projected sub‑gradient descent (instead of the multiplicative
  exponentiated‑gradient used before).  Additive updates tend to fine‑tune
  the solution more aggressively.

* Keep the best sequence seen so far and spend the rest of the time
  repeatedly refining it:
    – long additive sub‑gradient phases,
    – occasional Adam fine‑tuning,
    – random perturbations of the logits,
    – a cheap random‑move hill‑climber.

All steps respect the global time budget (default 1000 s) and the best
feasible sequence is always returned.  The implementation works only with
NumPy and standard library modules.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for a gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start and cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet with several concentration values
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # edge‑exponential families (mass at the left/right)
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.5, 7):   # extended range
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.1, 0.2, 0.4):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))
    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05                                 # a little more conservative
    lam = 0.02                                 # start temperature (soft)
    lam_factor = 1.07                         # annealing factor

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake as deadline approaches
        if it % 5_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _additive_subgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Additive projected sub‑gradient descent on the exact max objective.
    Uses the simple projection ``a←max(a,0); a←a/∑a`` after each step.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[AddSubgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size η_t = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: move a random fraction of the mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 20 % of the source mass
        delta = a[i] * (np.random.rand() * 0.20)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²).

    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build a large, diverse pool of starts
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # extra random mixtures of Gaussians
    i_grid = np.arange(n, dtype=np.float64)
    for _ in range(5):
        k = np.random.randint(2, 5)          # 2‑4 components
        comp = np.zeros(n)
        for __ in range(k):
            mu = np.random.rand() * (n - 1)
            sigma = np.random.rand() * 0.12 * n + 0.01 * n
            weight = np.random.rand()
            comp += weight * np.exp(-0.5 * ((i_grid - mu) / sigma) ** 2)
        comp = np.maximum(comp, 0.0)
        if comp.sum() > 0:
            comp /= comp.sum()
            pool.append(comp)

    # extra random Beta‑shapes
    for _ in range(5):
        alpha = np.random.uniform(0.2, 5.0)
        beta = np.random.uniform(0.2, 5.0)
        x = (np.arange(n) + 0.5) / n
        shape = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – diversified short runs (≈20 % of budget)
    per_candidate_budget = (deadline - time.time()) * 0.20 / max(1, len(pool))
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 5.0, time.time() + per_candidate_budget)

        # Adam optimisation (smooth‑max surrogate)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[cand {cid}] Adam improves to {val_adam:.6f}")

        # Additive sub‑gradient refinement
        a_sg, val_sg = _additive_subgrad(
            best_seq, n, conv_fft_len, corr_fft_len, local_deadline,
            max_iters=80_000, eta0=0.8
        )
        if val_sg < best_val:
            best_val, best_seq = val_sg, a_sg.copy()
            print(f"[cand {cid}] AddSubgrad improves to {val_sg:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive refinement of the current best (≈70 % of budget)
    while time.time() < deadline - 5.0:
        # long additive sub‑gradient phase
        a_sg, val_sg = _additive_subgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000, eta0=0.65
        )
        if val_sg < best_val:
            best_val, best_seq = val_sg, a_sg.copy()
            print(f"[Refine] AddSubgrad new best {val_sg:.6f}")

        # occasional Adam fine‑tuning
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=80_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Refine] Adam new best {val_adam:.6f}")

        # random perturbation of the logits + short additive refinement
        logits = np.log(best_seq + 1e-12)
        logits += np.random.randn(n) * 0.06
        a_pert = np.exp(logits - np.max(logits))
        a_pert = a_pert / a_pert.sum()
        a_pert_sg, val_pert_sg = _additive_subgrad(
            a_pert, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=70_000, eta0=0.55
        )
        if val_pert_sg < best_val:
            best_val, best_seq = val_pert_sg, a_pert_sg.copy()
            print(f"[Perturb] AddSubgrad new best {val_pert_sg:.6f}")

        # cheap random local search
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=60_000
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved best to {val_loc:.6f}")

        # break early if we are out of time
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Fallback – ensure a valid, non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
