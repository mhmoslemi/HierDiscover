def _alternating_candidate(n, alpha=1.0):
    vals = np.random.dirichlet(alpha * np.ones(n))
    sorted_vals = np.sort(vals)[::-1]
    left = sorted_vals[: (n+1)//2]
    right = sorted_vals[(n+1)//2:][::-1]
    # interleave
    result = np.empty(n, dtype=np.float64)
    result[0::2] = left
    result[1::2] = right[:len(result[1::2])]
    return result
```

Works for odd n: left length ceil(n/2), right length floor(n/2). If n odd, right length = floor => positions fill.

def _targeted_sa(a0: np.ndarray, time_limit: float, start_temp: float = 0.1, max_frac: float = 0.3) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    while time.time() < time_limit:
        # Find shift where conv is maximized
        k = int(conv.argmax())
        # Determine contributing index i with max product a_i * a_{k-i}
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        products = a[i_range] * a[j_range]
        # choose i with max product
        idx = i_range[np.argmax(products)]
        # Choose destination j randomly different from idx
        j = np.random.randint(n-1)
        if j >= idx:
            j += 1

        # Transfer fraction of mass from idx to j
        frac = np.random.rand() * max_frac
        delta = a[idx] * frac
        if delta <= 0:
            continue

        _apply_delta(a, conv, idx, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # compute temperature linear decay
        elapsed = time.time() - start
        total = time_limit - start
        temp = start_temp * (1 - elapsed/total)

        # accept move if improves or with probability
        if diff <= 0 or (temp > 0 and np.random.rand() < np.exp(-diff / temp)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, idx, delta)

    return best_a, best_val

"""
Hybrid random‑search + gradient + targeted simulated annealing for the
step‑function minimisation problem.

* generate a diverse pool of warm‑starts (uniform, Dirichlet, edge‑focused,
  anti‑pair and alternating arrangements);
* pick the best of them (exact objective via FFT);
* refine it with a short Adam phase on a smooth‑max surrogate,
  followed by an exact‑max exponentiated‑gradient sub‑gradient phase;
* finally run a targeted simulated‑annealing that always moves mass from the
  index contributing most to the current maximal convolution coefficient.
  The schedule is a linear cooling from a relatively high temperature.

All updates of the linear convolution a*a are performed in O(n) with a
slice‑based incremental routine, so millions of moves can be evaluated
within the 1000 s budget.  The resulting vector is projected onto the
simplex (non‑negative, sum ≈ 1) before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence for sum = 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Incremental convolution update (used by SA)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float) -> None:
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    conv[2 * idx] += delta * delta
    a[idx] = old + delta

# Candidate generators -------------------------------------------------
def _warm_start(n: int):
    """If a previous solution exists, adapt it to length n."""
    # `height_sequence_1` may be defined globally by a previous run.
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Large values on the left, small on the right (no interleaving)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _alternating_arrangement(vals: np.ndarray) -> np.ndarray:
    """Interleave large and small values (alternating high/low)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    # split into two halves
    left_len = (n + 1) // 2                    # ceil
    left = sorted_vals[:left_len]              # big values
    right = sorted_vals[left_len:][::-1]       # small values (reversed)
    res = np.empty(n, dtype=np.float64)
    res[0::2] = left
    res[1::2] = right[: len(res[1::2])]
    return res


def _alternating_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _alternating_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _random_perm_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Random permutation of a Dirichlet draw."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    np.random.shuffle(vals)
    return vals

# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # soft‑max → simplex

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c               # denominator is 1 (sum=1)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _targeted_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.1,
    max_frac: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that always moves mass from the index that
    contributes most to the current maximal convolution coefficient.
    Linear temperature decay.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        # shift (index) where convolution attains its maximum
        k = int(conv.argmax())
        # indices that participate in that shift
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        # product contributions a[i]*a[j]
        products = a[i_range] * a[j_range]
        # index with the largest contribution
        idx = i_range[np.argmax(products)]

        # choose a destination index different from idx
        j = np.random.randint(n - 1)
        if j >= idx:
            j += 1

        # transfer a random fraction of the mass at idx to j
        frac = np.random.rand() * max_frac
        delta = a[idx] * frac
        if delta <= 0.0:
            continue

        # apply move (incremental update)
        _apply_delta(a, conv, idx, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear cooling
        elapsed = time.time() - start
        total = time_limit - start
        temp = start_temp * max(0.0, 1.0 - elapsed / total)

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, idx, delta)

        it += 1
        if it % 100_000 == 0:
            print(f"[Targeted SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given budget.
    The pipeline mixes random warm‑starts, Adam, sub‑gradient, and a
    targeted simulated‑annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # 1) try several problem sizes (trading off quality vs. speed)
    size_options = [2048, 3072, 4096, 6144, 8192]
    best_seq = None
    best_val = float("inf")
    best_n = None

    for n in size_options:
        if time.time() > deadline:
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        candidates = []

        # warm‑start (if available)
        warm = _warm_start(n)
        if warm is not None:
            candidates.append(warm.copy())
            # a few rotations of the warm start
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(warm, shift))

        # deterministic / simple baselines
        candidates.append(_uniform_candidate(n))
        candidates.append(_triangle_candidate(n))

        # Dirichlet samples with a few concentration parameters
        for alpha in (0.3, 0.7, 1.0, 2.0):
            candidates.append(_random_dirichlet_candidate(n, alpha))

        # anti‑pair and alternating arrangements
        for alpha in (0.3, 0.7, 1.0):
            candidates.append(_anti_pair_candidate(n, alpha))
            candidates.append(_alternating_candidate(n, alpha))

        # edge‑focused shapes
        for gamma in (0.2, 0.5, 0.8, 1.0):
            candidates.append(_gaussian_edges_candidate(n, gamma))

        # random permutation of a Dirichlet draw
        candidates.append(_random_perm_candidate(n, alpha=1.0))

        # ----------------------------------------------------------
        # cheap exact evaluation of the pool
        for idx, cand in enumerate(candidates):
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            cand /= cand.sum()
            val = _objective(cand)
            print(f"[Init n={n} cand {idx:02d}] value {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                best_n = n

    if best_seq is None:
        # fallback – uniform vector of length 4096
        best_n = 4096
        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)

    # --------------------------------------------------------------
    # Pre‑compute FFT lengths for the selected size
    n = best_n
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Adam phase (smooth‑max)
    now = time.time()
    remaining = deadline - now
    if remaining > 20.0:
        adam_deadline = now + min(200.0, remaining * 0.35)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr=0.06,
            lam_start=0.8,
            lam_factor=1.06,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Sub‑gradient refinement
    now = time.time()
    remaining = deadline - now
    if remaining > 15.0:
        sub_deadline = now + min(200.0, remaining * 0.30)
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=400_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Targeted simulated‑annealing (fine‑grained)
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0:
        sa_seq, sa_val = _targeted_sa(
            best_seq,
            time_limit=now + remaining,
            start_temp=0.12,
            max_frac=0.25,
        )
        if sa_val < best_val:
            best_val = sa_val
            best_seq = sa_seq.copy()
            print(f"[Targeted SA] improved to {sa_val:.6f}")

    # --------------------------------------------------------------
    # 5) Final safety checks and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])
    else:
        best_seq = best_seq / total

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
