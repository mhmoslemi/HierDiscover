"""
Propose a step‑function that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm is a multi‑stage search that differs from the previous
implementation in three important ways:

1. **Richer warm‑starts** – we generate many Dirichlet, log‑normal and
   handcrafted candidates and try several deterministic permutations:
   simple descending, “anti‑pair” (large block opposite small block),
   “alternating extremes” (high‑low‑high‑low…) and a few random cyclic
   shifts.  This explores the space of permutations far more thoroughly.

2. **Permutation‑focused local search** – after a smooth‑max L‑BFGS
   refinement we run a long simulated‑annealing hill‑climber that only
   swaps two positions at a time (no mass transfer).  Swapping directly
   manipulates the pattern that determines the convolution peak and is
   very effective for shaving off the last few hundredths of the objective.

3. **Adaptive time budgeting** – each phase consumes a portion of the
   1000 s budget, falling back gracefully if the deadline is reached.
   If time permits we up‑sample the best 4096‑vector to 8192 points and
   run a quick L‑BFGS+sub‑gradient fine‑tuning on the finer grid.

All heavy work respects the global deadline; the final sequence is
clipped to the limits imposed by ``evaluate_sequence``.
"""
import time

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact) -----------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by the optimiser:
        2·n·max(a*a) / (sum(a)²)
    a must be non‑negative.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# Simplex projection (Euclidean) ----------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _warm_start(n: int):
    """Return a previously stored sequence (`height_sequence_1`) trimmed/padded to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    # normalise to unit sum – the objective is scale‑invariant
    warm = warm / warm.sum()
    return warm

def _uniform_candidate(n: int) -> np.ndarray:

def _triangle_candidate(n: int) -> np.ndarray:
    """Symmetric triangle (1,2,…,⌈n/2⌉,…,2,1) normalised to sum = 1."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape = shape.astype(np.float64)
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two mirrored Gaussian bells at the ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape.astype(np.float64)
    shape /= shape.sum()
    return shape

    """Log‑normal distribution, normalised to sum = 1."""
    vals = np.random.lognormal(mean=0.0, sigma=sigma, size=n)

    vals /= vals.sum()
    return vals


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet with concentration `alpha` (equal for all coordinates)."""
    a = np.random.dirichlet(alpha * np.ones(n))

# Permutation helpers ----------------------------------------------------
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    “Anti‑pair’’ – put the biggest half in the front,
    the smallest half (reversed) at the back.
    """
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _alternating_extremes(vals: np.ndarray) -> np.ndarray:
    """
    Interleave largest, smallest, second‑largest, second‑smallest, …
    This spreads large values throughout the sequence.
    """
    n = len(vals)
    s = np.sort(vals)[::-1]            # descending
    out = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    pos = 0
    while left <= right:
        out[pos] = s[left]
        pos += 1
        left += 1
        if left <= right:
            out[pos] = s[right]
            pos += 1
            right -= 1
    return out


def _random_shift(arr: np.ndarray) -> np.ndarray:
    """Cyclic shift by a random amount."""
    shift = np.random.randint(arr.shape[0])
    return np.roll(arr, shift)

# Smooth‑max L‑BFGS -------------------------------------------------------
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """
    Smooth‑max surrogate (log‑sum‑exp) and its gradient w.r.t. logits.
    """
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp of (lam·(conv-max_c))
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                       # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)       # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0),
                    max_iter_stage: int = 250) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate, growing λ progressively."""
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 10)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] improved to {val:.6f}")
    return best_a, best_val

def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,

                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑conv objective."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:                         # occasional re‑init
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:06d}] improved to {val:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation‑only simulated annealing -----------------------------------
def _perm_swap_hill_climb(a0: np.ndarray, n: int,
                          conv_fft_len: int,
                          deadline: float,
                          start_temp: float = 0.025,
                          max_moves: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing that only swaps two positions.
    The mass distribution stays unchanged – we are searching the permutation space.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_moves:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        a2 = a.copy()
        a2[i], a2[j] = a2[j], a2[i]

        val2 = _objective(a2)
        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2.copy()
        else:
            # annealing acceptance
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0 and it > 0:
            print(f"[PermSwap it={it:06d}] best {best_val:.6f}")
        it += 1
    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a)=2·n·max(a∗a)/(sum(a)²)

    The routine respects a global time budget (`budget_s` seconds) and
    returns the best sequence it found as a plain ``list`` of floats.
    """
    np.random.seed(seed)
    # safety margin – leave a tiny buffer for the harness
    deadline_total = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size (start with 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0) warm start from previous run, if any
    warm = _warm_start(n)
    if warm is not None:
        cand_val = _objective(warm)
        best_seq, best_val = warm.copy(), cand_val
        print(f"[Warm start] value {cand_val:.6f}")

    # ------------------------------------------------------------------
    # 1) cheap handcrafted starters
    starters = [
        _uniform_candidate,
        _triangle_candidate,
        lambda nn: _gaussian_edges_candidate(nn, 0.3),
        lambda nn: _gaussian_edges_candidate(nn, 0.7),
        lambda nn: _gaussian_edges_candidate(nn, 1.2),
    ]
    for fn in starters:
        cand = fn(n)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Starter] value {val:.6f}")

    # ------------------------------------------------------------------
    # 2) massive random warm‑starts with several permutations
    #    (stop early if we run out of time)
    perm_funcs = [
        lambda x: x,                     # identity (desc already)
        _anti_pair_arrangement,
        _alternating_extremes,
    ]
    max_rand_samples = 2500
    sample_cnt = 0
    while (time.time() < deadline_total - 80) and sample_cnt < max_rand_samples:
        # draw a Dirichlet with a random concentration α
        log_alpha = np.random.uniform(-2.5, 1.5)   # α ∈ [~0.003, ~31]
        alpha = 10.0 ** log_alpha
        base = _random_dirichlet_candidate(n, alpha)

        # evaluate each permutation (including a random cyclic shift)
        for perm in perm_funcs:
            cand = perm(base)
            # try a few random shifts – they change the linear convolution peak
            for _ in range(2):
                cand_shift = _random_shift(cand)
                val = _objective(cand_shift)
                if val < best_val:
                    best_val, best_seq = val, cand_shift.copy()
                    print(f"[Random pool] new best {val:.6f}")
        sample_cnt += 1

    # ------------------------------------------------------------------
    # 3) L‑BFGS smooth‑max refinement (use up to ~15 % of remaining budget)
    time_left = deadline_total - time.time()
    if time_left > 30.0:
        lbfgs_deadline = time.time() + min(180.0, time_left * 0.45)
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n, conv_fft_len,
                                         corr_fft_len, lbfgs_deadline,
                                         lam_schedule=(5, 15, 40, 120, 300),
                                         max_iter_stage=300)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact‑max sub‑gradient (multiplicative) – short burst
    time_left = deadline_total - time.time()
    if time_left > 30.0:
        sub_deadline = time.time() + min(200.0, time_left * 0.35)
        seq_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                 conv_fft_len, corr_fft_len,
                                                 sub_deadline,
                                                 max_iters=800_000,
                                                 eta0=0.6)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[MulSubgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Permutation‑only hill‑climbing (heavy part – aim for the last 0.003)
    time_left = deadline_total - time.time()
    if time_left > 30.0:
        perm_deadline = deadline_total - 5.0
        seq_perm, val_perm = _perm_swap_hill_climb(best_seq, n,
                                                   conv_fft_len,
                                                   perm_deadline,
                                                   start_temp=0.03,
                                                   max_moves=2_000_000)
        if val_perm < best_val:
            best_val, best_seq = val_perm, seq_perm.copy()
            print(f"[PermSwap] improved to {val_perm:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid (8192) if time permits
    time_left = deadline_total - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the best 4096‑vector
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # quick L‑BFGS on the big grid
        big_deadline = time.time() + min(120.0, time_left * 0.5)
        seq_big_lb, val_big_lb = _lbfgs_optimize(a_big, n_big,
                                                 conv_fft_len_big,
                                                 corr_fft_len_big,
                                                 big_deadline,
                                                 lam_schedule=(5, 20, 80, 200, 500),
                                                 max_iter_stage=150)

        # short sub‑gradient fine‑tuning
        final_deadline = deadline_total - 2.0
        seq_big_sub, val_big_sub = _subgradient_optimize(seq_big_lb, n_big,
                                                         conv_fft_len_big,
                                                         corr_fft_len_big,
                                                         final_deadline,
                                                         max_iters=400_000,
                                                         eta0=0.55)
        if val_big_sub < best_val:
            best_val, best_seq = val_big_sub, seq_big_sub.copy()
            print(f"[Big grid] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 7) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
