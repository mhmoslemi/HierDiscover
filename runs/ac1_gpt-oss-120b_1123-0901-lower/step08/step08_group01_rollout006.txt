    # Predefine basis families
    def generate_candidates():
         cand= []

         # Sinusoid

         # Edge exponential
         for gamma in np.logspace(-2, 1.5, 30):  # 0.01 to ~30
             cand.append(_edge_exponential(n, gamma))
         # Edge power law
         for alpha in np.linspace(-5, 2, 30):

         # Beta
         a_vals = np.logspace(-1, 1, 15)  # 0.1 to 10

"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2) .

The search consists of three stages:

1. **Broad generation** – a large, diverse pool of candidate vectors is
   created from many deterministic families (uniform, sinusoid,
   edge‑exponential, edge‑power‑law, beta‑pdf, double‑Gaussian) together
   with a few Dirichlet draws and cyclic shifts of an optional warm‑start.

   Each candidate is normalised to have unit sum and its exact objective
   is evaluated (single O(n log n) convolution).  The best `top_k`
   candidates are kept for refinement.

2. **Refinement** – a simple projected gradient descent with an exact
   back‑tracking line‑search is run on each of the retained candidates.
   The method works on the *true* max‑convolution (no smooth surrogate) and
   therefore drives the real objective down efficiently.  Time is split
   among the candidates, giving the most promising one the longest
   refinement budget.

3. **Final polishing** – a cheap random‑move hill‑climber performs a
   few hundred thousand local mass‑transfer steps on the current best
   vector, keeping any improvement.

All phases respect the global 1000 s wall‑clock budget and the best
sequence found is returned as a plain Python list (compatible with the
provided `evaluate_sequence` function).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Deterministic families (all return vectors with sum == 1)
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Add a few cyclic shifts of a warm start."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse seed pool (deterministic families + random Dirichlet)."""
    pool = []

    # Dirichlet with several concentration levels
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(3):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential (coarse grid – finer grid added later)
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Power‑law
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf (symmetric and skewed)
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool


def _grid_candidates(n: int) -> list[np.ndarray]:
    """
    A finer sweep over the main parametric families – many more gamma/alpha
    values than in the coarse pool.  The vectors are already normalised.
    """
    out = []

    # Edge‑exponential – log‑spaced
    for gamma in np.logspace(-2, 1.5, 30):      # 0.01 … ≈31.6
        out.append(_edge_exponential(n, float(gamma)))

    # Edge‑power‑law – linear sweep
    for alpha in np.linspace(-4.0, 2.0, 30):
        out.append(_edge_power_law(n, float(alpha)))

    # Beta – grid in log‑space
    a_vals = np.logspace(-1, 1, 15)   # 0.1 … 10
    b_vals = np.logspace(-1, 1, 15)
    for a in a_vals:
        for b in b_vals:
            out.append(_beta_shape(n, float(a), float(b)))

    # Double‑Gaussian – linear sweep
    for sigma in np.linspace(0.02, 0.5, 30):
        out.append(_double_gaussian(n, float(sigma)))

    return out

# Exact projected‑gradient descent with back‑tracking line‑search
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact objective.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    # a modest initial step – will be reduced by back‑tracking if needed
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set – indices attaining the current maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # ----- back‑tracking line‑search (projected) -----
        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: transfer a random fraction of the mass from one
    position to another and keep the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
