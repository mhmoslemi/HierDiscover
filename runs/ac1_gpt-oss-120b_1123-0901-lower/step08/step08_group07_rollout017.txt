"""
search for a non‑negative step function minimizing

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The strategy differs from the original script in three ways:

1. **Two‑stage optimisation** – first a fast search on a reduced
   size (n_small = 1024) and then a refinement after linear
   interpolation to the target length (n_target = 4096).  The cheap
   stage allows many more gradient steps within the overall 1000 s
   budget.

2. **Longer sub‑gradient phase** – the exact‑max (exponentiated‑gradient)
   update is run for as many iterations as time permits (instead of a
   fixed count).  This drives the max‑convolution down aggressively.

3. **Hybrid loop** – after each sub‑gradient phase a short simulated‑
   annealing hill‑climb is performed, and a few Adam‑smooth‑max
   refinements are interleaved.  Random perturbations keep the search
   from stagnating.

All helper routines are top‑level; the function `propose_candidate`
conforms to the required signature and returns a plain Python list of
floats that respects the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (without the safety wrapper) -------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)^2).  Assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest values on the left, smallest on the right (anti‑pair)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate) ---------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   lr: float = 0.07,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.05) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth‑max of the convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current exact objective (for book‑keeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:06d}  new best {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to keep the dynamics alive
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad] it={it:06d}  new best {val:.6f}")

        # indices attaining the max (active constraints)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (exponentiated‑gradient)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing ---------------------------------------
def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search:
    - swap two entries or transfer a fraction of mass,
    - accept if improvement or with a temperature‑dependent probability.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:                # swap two indices
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                      # move a fraction of mass
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local] move={it:06d}  best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Warm‑start helper ------------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that minimises
        F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine runs a two‑stage optimisation (small → target size)
    using Adam, exact sub‑gradient updates and a short stochastic
    hill‑climb.  The best sequence found before the time budget expires
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Stage 1 : cheap optimisation on a reduced grid
    n_small = 1024
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # ----- build a pool of diverse initial vectors (size n_small) -----
    candidates_small = []

    warm = _warm_start(n_small)
    if warm is not None:
        candidates_small.append(warm.copy())

    candidates_small.append(_uniform_candidate(n_small))

    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0):
        candidates_small.append(_random_dirichlet_candidate(n_small, alpha))

    for alpha in (0.1, 0.3, 0.7, 1.0):
        candidates_small.append(_anti_pair_candidate(n_small, alpha))

    for p in (0.2, 0.5, 1.0, 2.0):
        candidates_small.append(_powerlaw_candidate(n_small, p))
        candidates_small.append(_ushape_candidate(n_small, p))

    for gamma in (0.2, 0.5, 1.0):
        candidates_small.append(_gaussian_edges_candidate(n_small, gamma))

    candidates_small.append(_triangle_candidate(n_small))

    # ----- quick evaluation & keep the most promising -----------------
    evaluated_small = []
    for idx, cand in enumerate(candidates_small):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n_small))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated_small.append((val, cand))
        if idx % 5 == 0:
            print(f"[Init‑small {idx:02d}] value {val:.6f}")

    evaluated_small.sort(key=lambda x: x[0])
    best_seq_small = evaluated_small[0][1].copy()
    best_val_small = evaluated_small[0][0]
    print(f"[Small] initial best {best_val_small:.6f}")

    # ----- extensive sub‑gradient descent on the small grid ----------
    # Run until we have roughly 300 s left for the final stage
    sub_small_dead = max(deadline - 300.0, time.time() + 10.0)
    best_seq_small, best_val_small = _subgradient_optimize(
        best_seq_small,
        n_small,
        conv_fft_len_small,
        corr_fft_len_small,
        sub_small_dead,
        max_iters=2_000_000,
        eta0=0.8,
    )
    print(f"[Small] after sub‑grad best {best_val_small:.6f}")

    # A short Adam fine‑tuning (helps to escape flat plateaus)
    adam_small_dead = time.time() + min(60.0, deadline - time.time() - 200.0)
    if adam_small_dead > time.time():
        best_seq_small, val_adam_small = _adam_optimize(
            best_seq_small,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            adam_small_dead,
            max_iters=300_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_adam_small < best_val_small:
            best_val_small, best_seq_small = val_adam_small, best_seq_small.copy()
            print(f"[Small] after Adam best {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 : lift to the target size (n_target = 4096) and final refine
    n_target = 4096
    conv_fft_len_target = _next_pow2(2 * n_target - 1)
    corr_fft_len_target = _next_pow2(3 * n_target - 2)

    # linear interpolation to the larger grid
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_target = np.linspace(0.0, 1.0, n_target, endpoint=False)
    best_seq = np.interp(x_target, x_small, best_seq_small)

    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() < 1e-12:
        best_seq = _uniform_candidate(n_target)
    else:
        best_seq = best_seq / best_seq.sum()

    best_val = _objective(best_seq)
    print(f"[Target] after up‑sample best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Iterative refinement loop on the target size
    # We alternate: sub‑gradient → Adam → local random search.
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        # 1) sub‑gradient phase (takes ~40 % of remaining time)
        sub_dead = time.time() + 0.4 * time_left
        seq_sub, val_sub = _subgradient_optimize(
            best_seq,
            n_target,
            conv_fft_len_target,
            corr_fft_len_target,
            sub_dead,
            max_iters=2_000_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[Refine] sub‑grad improved to {best_val:.6f}")

        # 2) short Adam phase (≈20 % of remaining time)
        time_left = deadline - time.time()
        if time_left < 3.0:
            break
        adam_dead = time.time() + 0.2 * time_left
        seq_adam, val_adam = _adam_optimize(
            best_seq,
            n_target,
            conv_fft_len_target,
            corr_fft_len_target,
            adam_dead,
            max_iters=300_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Refine] Adam improved to {best_val:.6f}")

        # 3) local stochastic hill‑climb (≈20 % of remaining time)
        time_left = deadline - time.time()
        if time_left < 3.0:
            break
        local_dead = time.time() + 0.2 * time_left
        seq_loc, val_loc = _local_random_search(
            best_seq,
            n_target,
            conv_fft_len_target,
            local_dead,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Refine] local improved to {best_val:.6f}")

        # occasional random perturbation to avoid stagnation
        if np.random.rand() < 0.15:
            pert = np.log(best_seq + 1e-12) + np.random.normal(scale=0.05, size=n_target)
            pert = np.exp(pert - np.max(pert))
            pert = pert / pert.sum()
            val_pert = _objective(pert)
            if val_pert < best_val:
                best_val, best_seq = val_pert, pert.copy()
                print(f"[Refine] random perturb improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – guarantee a valid return
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.full(n_target, 1.0 / n_target, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # expose the best sequence for a possible warm‑start in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
