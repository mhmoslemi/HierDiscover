"""
Propose a step‑function (non‑negative list) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)²)

The search proceeds in several fast phases, always respecting the
global (≈1000 s) time budget:

1. Warm‑start (if a previous best vector is available) and a cheap
   collection of handcrafted candidates (uniform, triangular, Gaussian,
   anti‑pair, bit‑reverse, geometric).

2. Massive random anti‑pair Dirichlet pool – the best shape is kept.

3. Smooth‑max L‑BFGS refinement – the exact max is approximated by a
   log‑sum‑exp surrogate with progressively increasing temperature
   (λ).  The optimisation is performed in the soft‑max (log‑its) space
   so the simplex constraint is enforced automatically.

4. Additive projected sub‑gradient descent (mirror‑descent on the
   simplex) – works directly on the true max‑objective.

5. Local stochastic hill‑climbing (swap / mass‑transfer / segment‑reverse
   / rotation moves) with a simple annealing schedule, using as many
   moves as the remaining time permits.

6. Optional up‑sampling to a finer grid (8192 points) followed by a
   short L‑BFGS + sub‑gradient fine‑tuning.

All phases keep the current best vector and its objective value.
The final sequence is clipped to the evaluator’s limits (≥0, ≤1000)
and returned as a plain list.

The implementation is fully self‑contained, uses only NumPy,
SciPy (if available) and the provided ``evaluate_sequence`` for
validation.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (the value returned by evaluate_sequence)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair) ordering."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _bit_reverse_permutation(n: int) -> np.ndarray:
    """Return the bit‑reversed permutation for n = 2**k."""
    if n & (n - 1):
        raise ValueError("n must be a power of two for bit‑reverse")
    bits = n.bit_length() - 1
    perm = np.empty(n, dtype=int)
    for i in range(n):
        rev = 0
        x = i
        for _ in range(bits):
            rev = (rev << 1) | (x & 1)
            x >>= 1
        perm[i] = rev
    return perm


def _bit_reverse_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Place descending Dirichlet masses according to bit‑reversed indices."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    sorted_vals = np.sort(vals)[::-1]        # descending
    perm = _bit_reverse_permutation(n)
    a = np.empty(n, dtype=np.float64)
    a[perm] = sorted_vals
    return a / a.sum()


def _geometric_candidate(n: int, r: float = 0.99) -> np.ndarray:
    """Exponentially decreasing shape (one‑sided)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.power(r, i)
    shape /= shape.sum()
    return shape


def _geometric_symmetric_candidate(n: int, r: float = 0.95) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.power(r, i) + np.power(r, n - 1 - i)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """Return a previously stored sequence (global ``height_sequence_1``) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    # normalise only to avoid numerical issues – scaling does not change objective
    return warm / warm.sum()

# L‑BFGS optimisation of a smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) surrogate and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # stable log‑sum‑exp with temperature λ
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                     # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)    # chain rule through soft‑max
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0, 300.0, 600.0),
                    max_iter_stage: int = 500) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    try:
        import scipy.optimize as opt
    except Exception as e:
        print("SciPy not available – skipping L‑BFGS stage.")
        return a0, _objective(a0)

    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        rem = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(rem * 12)))  # generous per‑second budget
        res = opt.minimize(
            _lbfgs_obj, logits,
            args=(n, conv_fft_len, corr_fft_len, lam),
            method='L-BFGS-B', jac=True,
            options={'maxiter': iters, 'disp': False}
        )
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.0f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Additive projected sub‑gradient optimisation (mirror‑descent on simplex)
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 600_000,
                                  eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)  # sub‑gradient of max
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (annealed random moves)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.03
    it = 0
    while time.time() < deadline and it < max_moves:
        move_type = np.random.rand()
        if move_type < 0.33:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        elif move_type < 0.66:                   # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.85)   # up to 85 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
        elif move_type < 0.83:                   # reverse a random segment
            i = np.random.randint(n - 1)
            j = np.random.randint(i + 1, n)
            a2 = a.copy()
            a2[i:j] = a2[i:j][::-1]
        else:                                    # rotate (circular shift)
            shift = np.random.randint(1, n)
            a2 = np.roll(a, shift)

        val2 = _objective(a2)

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a = a2
        else:
            # simple annealing acceptance for occasional uphill moves
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")
        it += 1
    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size (power of two for cheap FFTs)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0) Warm start (if supplied by the harness)
    warm = _warm_start(n)
    if warm is not None:
        v = _objective(warm)
        best_seq, best_val = warm.copy(), v
        print(f"[Warm start] value {v:.6f}")

    # --------------------------------------------------------------
    # 1) Hand‑crafted starter vectors
    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
        _geometric_candidate(n, 0.97),
        _geometric_symmetric_candidate(n, 0.94),
        _bit_reverse_candidate(n, 1.0),
    ]

    for c in starters:
        v = _objective(c)
        if v < best_val:
            best_val, best_seq = v, c.copy()
            print(f"[Starter] improved to {v:.6f}")

    # --------------------------------------------------------------
    # 2) Massive random anti‑pair Dirichlet pool
    time_left = deadline - time.time()
    if time_left > 30.0:
        pool_deadline = time.time() + min(30.0, time_left * 0.4)
        n_samples = 5000
        for _ in range(n_samples):
            if time.time() >= pool_deadline:
                break
            # α sampled log‑uniformly over a wide range
            log_alpha = np.random.uniform(-3.0, 2.0)   # α ∈ [1e‑3, 1e2]
            alpha = 10.0 ** log_alpha
            cand = _anti_pair_candidate(n, alpha)
            v = _objective(cand)
            if v < best_val:
                best_val, best_seq = v, cand.copy()
                print(f"[Random pool] new best {v:.6f}")

    # --------------------------------------------------------------
    # 3) L‑BFGS smooth‑max refinement (if time permits)
    time_left = deadline - time.time()
    if time_left > 120.0:
        lbfgs_deadline = min(deadline - 70.0, time.time() + 110.0)
        seq_lb, val_lb = _lbfgs_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            lbfgs_deadline,
            lam_schedule=(5, 15, 40, 120, 300, 600, 1200),
            max_iter_stage=800
        )
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # 4) Additive projected sub‑gradient optimisation
    time_left = deadline - time.time()
    if time_left > 100.0:
        sub_deadline = min(deadline - 40.0, time.time() + 90.0)
        seq_add, val_add = _additive_subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            sub_deadline,
            max_iters=600_000,
            eta0=0.7
        )
        if val_add < best_val:
            best_val, best_seq = val_add, seq_add.copy()
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # --------------------------------------------------------------
    # 5) Local stochastic annealing (as many moves as time allows)
    time_left = deadline - time.time()
    if time_left > 30.0:
        local_deadline = deadline - 5.0
        # a generous move budget proportional to remaining seconds
        max_moves = int(min(2_000_000, time_left * 2000))
        seq_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, local_deadline, max_moves=max_moves
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid (8192) and fine‑tuning
    time_left = deadline - time.time()
    if time_left > 50.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best vector
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # brief L‑BFGS on the larger grid
        big_deadline = min(deadline - 20.0, time.time() + 60.0)
        seq_big_lb, val_big_lb = _lbfgs_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            big_deadline,
            lam_schedule=(5, 20, 80, 200, 600, 1500),
            max_iter_stage=500
        )

        # final sub‑gradient fine‑tuning
        time_left2 = deadline - time.time()
        if time_left2 > 15.0:
            final_deadline = time.time() + min(15.0, time_left2)
            seq_big_sub, val_big_sub = _additive_subgradient_optimize(
                seq_big_lb, n_big, conv_fft_len_big, corr_fft_len_big,
                final_deadline,
                max_iters=300_000,
                eta0=0.6
            )
            if val_big_sub < best_val:
                best_val, best_seq = val_big_sub, seq_big_sub.copy()
                print(f"[Big grid] improved to {val_big_sub:.6f}")

    # --------------------------------------------------------------
    # 7) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
