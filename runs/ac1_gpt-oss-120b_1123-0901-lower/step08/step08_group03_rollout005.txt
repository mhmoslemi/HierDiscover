"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm overview
------------------
1.  Warm‑start from the previously found solution (global ``height_sequence_1``
    if it exists) otherwise use a uniform vector.
2.  Generate a large pool of cheap parametric candidates
    (Dirichlet, exponentials, beta‑pdf, Gaussian mixtures, sinusoid, power‑law
    etc.) and keep the best.
3.  Refine the best candidate with a *smooth‑max* Adam optimiser
    (soft‑max parametrisation, logarithmic temperature schedule, small
    injected Gaussian noise).
4.  Run a few short Adam restarts from perturbed logits to escape possible
    local minima.
5.  Perform an exact‑max exponentiated‑gradient (sub‑gradient) phase.
6.  Apply a cheap random hill‑climb that moves a random fraction of mass
    between two coordinates.
7.  Finish with a simulated‑annealing stage that perturbs the logits,
    accepts worsening moves with a temperature schedule and keeps the best
    ever observed vector.
8.  Clip the final vector to the evaluator’s allowed range and return it.

All steps respect the overall time budget (default 1000 s) and the best
vector found so far is always returned, guaranteeing a valid answer.
"""

import time
import numpy as np
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.1,
    lam_factor: float = 1.02,
    noise_sigma: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    a0 = np.clip(a0, eps, None)
    logits = np.log(a0)                     # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # soft‑max projection to the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp with temperature 1/lam)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (wrt a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # stochastic perturbation
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (slowly tighten the smooth‑max)
        lam = min(lam * lam_factor, 1e6)

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        # occasional shake close to deadline
        if it % 4_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set – indices attaining the max (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size η_t = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against under/overflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing: move a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)    # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    init_sigma: float = 0.4,
    final_sigma: float = 0.02,
    init_temp: float = 0.1,
    final_temp: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the logits (log‑softmax) parametrisation.
    Accepts worsening moves with probability exp(-Δ/T).
    """
    eps = 1e-15
    a = np.clip(a0, eps, None)
    a = a / a.sum()
    logits = np.log(a)

    best_a = a.copy()
    best_val = _eval_objective(best_a, n, conv_fft_len)

    cur_val = best_val
    cur_logits = logits.copy()

    it = 0
    while it < max_iters and time.time() < deadline:
        # linear schedules for sigma and temperature
        frac = it / max_iters
        sigma = init_sigma * (1 - frac) + final_sigma * frac
        temp = init_temp * (1 - frac) + final_temp * frac

        # propose: Gaussian perturbation of logits
        prop_logits = cur_logits + np.random.normal(scale=sigma, size=n)
        # soft‑max to obtain a proper probability vector
        mx = prop_logits.max()
        exp_log = np.exp(prop_logits - mx)
        prop_a = exp_log / exp_log.sum()

        prop_val = _eval_objective(prop_a, n, conv_fft_len)

        delta = prop_val - cur_val
        if delta < 0 or np.random.rand() < math.exp(-delta / max(temp, 1e-12)):
            cur_logits = prop_logits
            cur_val = prop_val
            if prop_val < best_val:
                best_val = prop_val
                best_a = prop_a.copy()
                print(f"[SA it={it:06d}] new best {prop_val:.6f}")

        it += 1

    return best_a, best_val

def _candidate_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


def _candidate_exponential(n: int, gamma: float, symmetric: bool = False) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    if symmetric:
        right = left[::-1]
        a = left + right
    else:
        a = left
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_half_sinusoid(n: int) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_power_law(n: int, p: float) -> np.ndarray:
    # a_i = (i+1)^p  (p negative -> decreasing)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # using the Beta pdf formula (unnormalised, then normalise)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_two_gaussians(n: int) -> np.ndarray:
    xs = np.arange(n, dtype=np.float64) / n
    mu1, mu2 = np.random.rand(2)
    sigma1, sigma2 = np.random.rand(2) * 0.15 + 0.02
    g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
    g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
    a = g1 + g2
    a /= a.sum()
    return a


def _candidate_spikes(n: int, k: int) -> np.ndarray:
    idx = np.random.choice(n, size=k, replace=False)
    weights = np.random.dirichlet(np.ones(k))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start ---------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – cheap random + parametric candidates --------------------
    best_seq = None
    best_val = float("inf")

    def _consider(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # baseline
    if warm is not None:
        _consider(warm)
    else:
        _consider(np.full(n, 1.0 / n))

    # Dirichlet draws
    for _ in range(16):
        _consider(_candidate_dirichlet(n))

    # Exponential families (different decay rates, symmetric/asymmetric)
    for gamma in (0.2, 0.4, 0.7, 1.0, 1.5, 2.0):
        _consider(_candidate_exponential(n, gamma, symmetric=False))
        _consider(_candidate_exponential(n, gamma, symmetric=True))

    # Half‑wave sinusoid
    _consider(_candidate_half_sinusoid(n))

    # Power‑law shapes
    for p in (-4.0, -3.0, -2.0, -1.0, -0.5):
        _consider(_candidate_power_law(n, p))

    # Beta pdfs (various alphas/betas)
    for alpha, beta in ((0.5, 0.5), (2.0, 2.0), (2.0, 5.0), (5.0, 2.0), (0.7, 1.3)):
        _consider(_candidate_beta_pdf(n, alpha, beta))

    # Two‑Gaussian mixtures
    for _ in range(12):
        _consider(_candidate_two_gaussians(n))

    # Spike candidates – various supports
    for k in (32, 64, 128, 256, 512):
        for _ in range(8):
            _consider(_candidate_spikes(n, k))

    # ------------------------------------------------------------------
    # Phase 2 – main Adam optimisation (large budget)
    remaining = deadline - time.time()
    if remaining <= 0:
        # fallback
        final_seq = best_seq if best_seq is not None else np.full(n, 1.0 / n)
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        return final_seq.tolist()

    adam_deadline = time.time() + 0.55 * remaining   # ~55 % of what’s left
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=400_000,
        lr=0.07,
        lam_start=0.09,
        lam_factor=1.015,
        noise_sigma=0.025,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam main] improved best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – a few noisy‑restart Adam runs
    for restart in range(6):
        if time.time() >= deadline - 35:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.35, size=n)   # stronger perturbation
        a0 = np.exp(logits - np.max(logits))
        a0 /= a0.sum()

        r_deadline = min(deadline - 5, time.time() + 0.10 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=200_000,
            lr=0.06,
            lam_start=0.08,
            lam_factor=1.018,
            noise_sigma=0.022,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient refinement (medium budget)
    if time.time() < deadline - 20:
        sg_deadline = deadline - 15
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=300_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – cheap random hill‑climb (short budget)
    if time.time() < deadline - 8:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=100_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – simulated annealing for fine‑grained polishing
    if time.time() < deadline - 2:
        sa_deadline = deadline - 0.5
        max_sa_iters = int(1e5)   # enough iterations for the remaining time
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            sa_deadline,
            max_iters=max_sa_iters,
            init_sigma=0.3,
            final_sigma=0.01,
            init_temp=0.15,
            final_temp=1e-5,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
