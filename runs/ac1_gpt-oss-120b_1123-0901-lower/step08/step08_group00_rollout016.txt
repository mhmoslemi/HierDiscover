from scipy.optimize import linprog
import numpy as np
import scipy.sparse as sp

def solve_lp_for_g(f, M):
    n = len(f)
    rows = 2*n-1
    # Build sparse matrix A where A @ g <= M * 1 (i.e., each row <= M)
    # For each j, create a column vector which is f placed at rows j..j+n-1
    data = []
    row_idx = []
    col_idx = []
    for j in range(n):
        # rows affected are from j to j + n - 1 inclusive (total n rows)
        r_start = j
        r_end = j + n # exclusive
        # For each r in [j, j+n-1], coefficient is f[r - j] = f[i] where i = r - j
        # So for i from 0..n-1, row = j + i, col = j, value = f[i]
        # Append to lists
        row_idx.extend(range(r_start, r_end))
        col_idx.extend([j] * n)
        data.extend(f.tolist())
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
    # upper bound vector b = M * np.ones(rows)
    b = np.full(rows, M)
    # objective minimize -sum(g)
    c = -np.ones(n)
    # bounds for each variable: (0, None)
    bounds = [(0, None)] * n
    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback maybe return current f
        return None
```

def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    n = f.shape[0]
    rows = 2 * n - 1
    # Build sparse matrix A
    diags = []
    offsets = []
    for d in range(n):
        val = f[d]
        if val == 0.0:
            continue
        diag_len = rows - d
        diags.append(np.full(diag_len, val, dtype=np.float64))
        offsets.append(-d)
    A = sp.diags(diags, offsets, shape=(rows, n), format='csr')
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g) -> minimize -sum(g)
    bounds = [(0, None)] * n
    # Solve using high-level linear programming
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        # Rescale to sum 1 (simplex)
        s = g.sum()
        if s > 0:
            g = g / s
        return g
    else:
        return None
```

"""
Hybrid multi‑phase optimizer for the step‑function objective

    F(a) = 2·n·max(a∗a) / (∑a)²

The routine builds a diverse set of start vectors (including a new
arcsine‑beta shape that is known to flatten the self‑convolution), refines
them with three classical phases (Adam on a smooth‑max surrogate,
exponentiated‑gradient sub‑gradient descent, and simulated‑annealing moves)
and finally applies a tiny linear‑programming “boost’’ that exploits the
feasibility of any vector g with (a∗g)≤max(a∗a).  The LP maximises the total
mass of g; after a simple rescaling this extra mass can be blended into the
current best and provably reduces the peak of the convolution.  The whole
pipeline respects the 1000 s budget and always returns the best sequence
found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value 2·n·max(conv)/sum² (sum is 1 for simplex vectors)."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (a.sum() ** 2)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_fac = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # simplex point
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_fac, 1e7)

        # occasional tiny shake near the end
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set for the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # mirror‑descent update on the simplex
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing with random mass transfers."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _arcsine_candidate(n: int) -> np.ndarray:
    """Beta(½,½) – discretised arcsine density, heavily weighted at the edges."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n          # mid‑point rule
    eps = 1e-12
    shape = 1.0 / (np.sqrt(x * (1.0 - x)) + eps)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Diverse pool of start vectors (adds arcsine shape)."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Arcsine (new!)
    cand.append(_arcsine_candidate(n))

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse spike patterns for extra exploration."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.  (f * g)_k ≤ M  ∀k ,  g ≥ 0.
    Returns the optimal g (scaled to Σg = 1) or None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build sparse constraint matrix A (shape rows × n)
    # A_{k,j} = f_{k-j}  if 0 ≤ k-j < n else 0.
    diags = []
    offsets = []
    nz = np.where(f > 0)[0]
    for d in nz:
        val = f[d]
        diag_len = rows - d
        diags.append(np.full(diag_len, val, dtype=np.float64))
        offsets.append(-d)               # negative → sub‑diagonal
    A = sp.diags(diags, offsets, shape=(rows, n), format='csr')
    b = np.full(rows, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)   # maximise Σg → minimise -Σg
    bounds = [(0, None)] * n

    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True}
        )
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s > 0:
        g = g / s
    return g

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # Build diverse start pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – quick exploration of many candidates (≈30 % of budget)
    explore_deadline = time.time() + (deadline - time.time()) * 0.30
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Short Adam optimisation
        sub_dead = min(explore_deadline, time.time() + 5.0)

        # Brief sub‑gradient refinement

    # ------------------------------------------------------------------

        a_big /= a_big.sum()

            a_big_adam,

    # ------------------------------------------------------------------
    # Fallback – ensure a valid, non‑empty return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
