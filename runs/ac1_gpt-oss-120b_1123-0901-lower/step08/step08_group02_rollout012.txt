def beta_shape(alpha, n):
    x = (np.arange(n) + 0.5) / n
    # Avoid zero values for power - epsilon
    eps = 1e-12
    shape = np.power(np.maximum(x, eps), alpha - 1) * np.power(np.maximum(1 - x, eps), alpha -1)
    return shape

def exp_shape(gamma, n):
    idx = np.arange(n)
    shape = np.exp(-gamma * idx) + np.exp(-gamma * (n-1-idx))
    return shape

def cosine_power_shape(p, n):
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)  # positive
    # raise to power p: if p negative, compute sin^p = sin**p = 1/(sin**(-p))
    # To avoid division by zero when sin low (close to zero at endpoints) and p negative,
    # add epsilon in denominator.
    eps = 1e-12
    if p >= 0:
        shape = sin_vals ** p
    else:
        shape = sin_vals ** p  # works as sin_val ** negative yields large numbers near edges - may overflow.
        # use np.where to prevent overflow?
        # Use safe: shape = np.exp(p * np.log(np.maximum(sin_vals, eps)))
        shape = np.exp(p * np.log(np.maximum(sin_vals, eps)))
    return shape

def poly_power_shape(alpha, n):
    # shape = (x*(1-x) + eps) ** alpha, where x in [0,1]
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    base = np.maximum(x * (1 - x), eps)
    shape = np.exp(alpha * np.log(base))
    return shape

def double_gaussian_shape(sigma, n):
    i = np.arange(n, dtype=np.float64)
    # sigma expressed as fraction of n (relative width)
    shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + np.exp(-0.5 * ((i - (n-1)) / (sigma * n)) ** 2)
    return shape

def _subgradient_optimize(a0, n, fft_len_conv, fft_len_corr, deadline, max_iters=500_000, eta0=0.9):
    a = np.array(a0, copy=True)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s

    best_a = a.copy()
    best_val = compute_objective(a, n, fft_len_conv)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # compute objective
        val = 2.0 * n * max_c / (a.sum() ** 2)  # sum should be 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-14)
        # w distribution over conv entries (length 2n-1)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # compute gradient
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # step size
        eta = eta0 / np.sqrt(it + 1)
        # multiplicative update
        a = a * np.exp(-eta * grad)  # elementwise
        # renormalize
        sum_a = a.sum()
        if sum_a <= 0 or np.isnan(sum_a):
            a = np.random.dirichlet(np.ones(n))
            sum_a = a.sum()
        a /= sum_a

        it += 1

    return best_a, best_val

def _local_random_search(a0, n, fft_len_conv, deadline, max_moves=200_000):
    a = np.array(a0, copy=True)
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # fraction up to 20%
        delta = a[i] * np.random.rand() * 0.2
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, fft_len_conv)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            # optional print
            if it % 20000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

"""
Propose a step‑function sequence that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a))² .

The algorithm works in three stages:

1.  **Parametric exploration** – a fast random / grid search over several
    families of analytically‑parameterised non‑negative vectors
    (Beta, edge‑exponential, cosine‑power, polynomial‑power,
    double‑Gaussian).  Each candidate is normalised to sum = 1 and
    evaluated with a single FFT‑based convolution.

2.  **Refinement** – the best candidate from stage 1 is improved by a
    sub‑gradient method on the exact max‑convolution (mirror‑descent on
    the simplex).  The step‑size decays as ηₜ = η₀/√(t+1).

3.  **Fine local search** – a cheap hill‑climber moves a random
    fraction of mass between two indices and keeps any improving move.

All stages respect the global time budget (default 1000 s) and the best
feasible sequence found so far is always returned.  A warm‑start from a
previous solution `height_sequence_1` is used if available.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for a sub‑gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact value of 2·n·max(a∗a) / (sum(a))²."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Shape factories --------------------------------------------------------
def _beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete grid."""
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    a = np.power(np.maximum(x, eps), alpha - 1.0)
    b = np.power(np.maximum(1.0 - x, eps), alpha - 1.0)
    return a * b


def _exp_edge_shape(gamma: float, n: int) -> np.ndarray:
    """Mass concentrated near the ends: exp(-γ·i) + exp(-γ·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))


def _cosine_power_shape(p: float, n: int) -> np.ndarray:
    """(sin(π·(i+0.5)/n))^p  – works for positive and negative p."""
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    eps = 1e-12
    # use log‑exp to stay safe for negative powers
    return np.exp(p * np.log(np.maximum(sin_vals, eps)))


def _poly_power_shape(alpha: float, n: int) -> np.ndarray:
    """((i/n)*(1‑i/n))^α  – a symmetric polynomial family."""
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    base = np.maximum(x * (1.0 - x), eps)
    return np.exp(alpha * np.log(base))


def _double_gauss_shape(sigma: float, n: int) -> np.ndarray:
    """Two equal Gaussians centred at the ends."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + np.exp(
        -0.5 * ((i - (n - 1)) / (sigma * n)) ** 2
    )


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    Uses an exponentiated‑gradient (mirror descent on the simplex).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # step size ηₜ = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)          # multiplicative update
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of the mass from one
    index to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 20 % of the source mass
        delta = a[i] * (np.random.rand() * 0.2)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 20_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # problem size
    n = 4096

    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from a previous solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # initialise best with a simple uniform candidate
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_val = _objective(best_seq, n, conv_fft_len)

    if warm is not None:
        val_warm = _objective(warm, n, conv_fft_len)
        if val_warm < best_val:
            best_seq, best_val = warm.copy(), val_warm
            print(f"[Init] warm start improves to {val_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – fast random / grid exploration of parametric families
    param_deadline = start + 0.30 * (budget_s - 0.5)  # ~30 % of total budget

        if family == "beta":

        shape = np.maximum(shape, 0.0)
        if shape.sum() < 1e-12:

        shape /= shape.sum()

        val = _objective(shape, n, conv_fft_len)

            print(f"[Param] new best {val:.6f} (family={family})")

    # systematic grid search (uses remaining time if we have >200 s left)
    remaining = deadline - time.time()
    if remaining > 200.0:

        # Beta grid
        alphas = np.logspace(np.log10(0.1), np.log10(10.0), 30)
        for alpha in alphas:
            shape = _beta_shape(alpha, n)
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            val = _objective(shape, n, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, shape.copy()
                print(f"[Grid β] new best {val:.6f}")

        # Exponential edge grid

            val = _objective(shape, n, conv_fft_len)

        # Polynomial‑power grid

            val = _objective(shape, n, conv_fft_len)

    # Safety fallback (should never be needed)
    if best_seq is None:

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
