def lp_update(f, active_eps=1e-6, step=0.15):
    conv = conv_full(f)
    M = conv.max()
    active = np.where(conv >= M - active_eps)[0]
    if len(active) == 0:
        return f
    # build A
    n = len(f)
    A = np.zeros((len(active), n))
    for row_i, i in enumerate(active):
        idx = np.arange(n)
        diff = i - idx
        mask = (diff >= 0) & (diff < n)
        A[row_i, mask] = f[diff[mask]]
    # solve LP: maximize sum(b) subject to A b <= M, b >=0
    # Use scipy.optimize.linprog, which minimizes c^T b. So set c = -np.ones(n)
    res = scipy.optimize.linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(active), M), bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return f unchanged
        return f
    b = res.x
    if b.sum() < 1e-12:
        # fallback: random
        b = np.random.dirichlet(np.ones(n))
    g = b / b.sum()
    f_new = (1 - step) * f + step * g
    # renormalize
    f_new = f_new / f_new.sum()
    return f_new
```

from scipy.optimize import linprog

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _objective(a):
    # assumes a non‑negative, sum(a) > 0
    conv = _full_convolution(a, _next_pow2(2*a.shape[0]-1))
    max_c = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_c / (a.sum()**2))

def _lp_refine_step(f, active_eps=1e-8, max_active=80, step=0.3):
    n = f.shape[0]
    # convolution of f with itself
    fft_len = _next_pow2(2*n-1)
    conv = _full_convolution(f, fft_len)
    M = conv.max()
    # identify near‑max entries
    eps = max(active_eps, M*1e-8)
    active = np.where(conv >= M - eps)[0]
    if active.size > max_active:
        # keep the largest ones only
        active = np.argsort(-conv)[:max_active]
    # build linear constraint matrix A b <= M, each row corresponds to an active index
    A = np.zeros((active.size, n))
    idx = np.arange(n)
    for r, i in enumerate(active):
        diff = i - idx
        mask = (diff >= 0) & (diff < n)
        A[r, mask] = f[diff[mask]]
    # solve linear program: maximize sum(b) s.t. A b <= M, b >= 0
    # scipy.linprog minimizes, so we minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(active.size, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: keep f unchanged
        return f, _objective(f)
    b = res.x
    if b.sum() < 1e-12:
        # fallback to a small random perturbation
        b = np.random.dirichlet(np.ones(n))
    g = b / b.sum()
    # line search on mixing step to guarantee non‑increase
    cur_val = _objective(f)
    t = step

        val_new = _objective(f_new)
        if val_new < cur_val:
            return f_new, val_new
        t *= 0.5
    # if no improvement, return original
    return f, cur_val

"""
Fast LP‑based active‑set refinement for the step‑function problem.

The algorithm works in three stages:

1. **Warm‑start** – reuse the previous best sequence (if available) or start
   from uniform weights.

2. **Iterative LP refinement** – repeatedly solve a tiny linear program that
   maximises the ℓ₁‑norm of a candidate while keeping all (near‑maximal)
   convolution entries below the current maximum M.  The solution is mixed
   with the current iterate using a simple back‑tracking line‑search so the
   objective never increases.  Only the most critical convolution indices
   (≤ 80) are kept as constraints, which makes each LP tiny and cheap.

3. **Short stochastic hill‑climb** – a simulated‑annealing style local search
   (swaps and fractional mass transfers) cleans up any remaining small
   peaks.

The whole routine respects the supplied time budget (default 1000 s) and
returns a non‑negative list whose sum is 1 (any scaling is harmless for the
evaluation function).  The implementation relies only on NumPy and SciPy.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear‑time O(n log n))
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Exact objective (sum of entries is kept = 1)
def _objective(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a)   (because sum(a)=1).
    """
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# One LP‑driven refinement step
def _lp_refine_step(
    f: np.ndarray,
    active_eps: float = 1e-8,
    max_active: int = 80,
    step_size: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Given the current distribution `f`, solve a tiny linear program that
    maximises ∑b while keeping all (near‑maximal) convolution entries
    ≤ M = max(f*f).  The result `g` is mixed with `f` using a back‑tracking
    line search to guarantee a strict decrease of the exact objective.
    Returns the new distribution and its objective value.
    """
    n = f.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(f, fft_len)
    M = conv.max()

    # --- active set ----------------------------------------------------
    eps = max(active_eps, M * 1e-8)
    active = np.where(conv >= M - eps)[0]
    if active.size > max_active:
        # keep the `max_active` largest entries only
        active = np.argsort(-conv)[:max_active]

    # --- build linear constraints A b ≤ M -------------------------------
    # Each row i corresponds to (f * b)[i] ≤ M.
    A = np.zeros((active.size, n), dtype=np.float64)
    idx = np.arange(n)
    for r, i in enumerate(active):
        diff = i - idx                # i - column index
        mask = (diff >= 0) & (diff < n)
        A[r, mask] = f[diff[mask]]

    # --- solve LP (maximize sum(b) ≡ minimise -sum(b)) -----------------
    c = -np.ones(n, dtype=np.float64)   # minimise -∑b
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(active.size, M), bounds=bounds,
                  method="highs", options={"presolve": True})

    if not res.success:
        # fall back to returning the unchanged iterate
        return f, _objective(f)

    b = res.x
    if b.sum() < 1e-12:                     # pathological case – randomise
        b = np.random.dirichlet(np.ones(n))

    g = b / b.sum()                         # re‑normalise to sum = 1

    # --- line‑search on mixing factor to ensure monotonic decrease -------
    cur_val = _objective(f)
    t = step_size
    while t > 1e-4:
        f_new = (1.0 - t) * f + t * g
        f_new /= f_new.sum()
        val_new = _objective(f_new)
        if val_new < cur_val:
            return f_new, val_new
        t *= 0.5

    # no improvement found – keep the old iterate
    return f, cur_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (swap / fractional transfer)
def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:

    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline:
        # choose move type
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:
            # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:

            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.3)      # ≤30 % transfer
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        if val2 < best_val:

        if it % 10000 == 0 and it > 0:
