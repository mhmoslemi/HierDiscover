"""
Propose a step‑function (non‑negative list) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)**2)

Algorithm
---------
1. Warm‑start from any previously found sequence.
2. Generate many random Dirichlet vectors, apply the classic anti‑pair
   ordering and also an “extreme‑interleave’’ ordering.  Keep the best.
3. Sweep a few cheap parametric families (logistic shape, Gaussian edges)
   and keep any improvements.
4. Run several L‑BFGS refinements on the best few candidates using an
   increasingly sharp log‑sum‑exp surrogate of the max‑convolution.
5. Perform long multiplicative and additive sub‑gradient phases.
6. Finish with a stochastic hill‑climbing (simulated annealing) local search.
7. If time permits, up‑sample the best sequence to 8192 points and repeat
   steps 4‑5.

All phases respect the global 1000 s budget and the returned list is
clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """Return a previously stored sequence (`height_sequence_1`) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Basic candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _extreme_interleave_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Interleave the sorted values: largest, smallest, 2nd‑largest,
    2nd‑smallest, …
    """
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    left, right, i = 0, n - 1, 0
    while left <= right:
        a[left] = s[i]
        i += 1
        left += 1
        if left <= right:
            a[right] = s[i]
            i += 1
            right -= 1
    return a


def _extreme_interleave_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _extreme_interleave_arrangement(vals)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _logistic_candidate(n: int, steep: float = 1.0, shift: float = 0.0) -> np.ndarray:
    """
    Monotone logistic profile.  The values are normalised to sum‑one.
    """
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0 + shift
    shape = 1.0 / (1.0 + np.exp(-steep * i))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # stable log‑sum‑exp of lam·(conv‑max_c)
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                      # soft max weights (sum to 1)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)      # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5, 15, 40, 120, 300, 800),
                    max_iter_stage: int = 250) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    import scipy.optimize as opt

    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 10)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Multiplicative (exponentiated‑gradient) sub‑gradient optimisation
def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 800_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:                         # re‑initialise occasionally
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Additive projected sub‑gradient optimisation
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 600_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (simulated annealing)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.025
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                    # transfer a fraction of mass
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.85)   # up to 85 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

        it += 1
    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0) Warm‑start / simple starters
    best_seq = None
    best_val = float("inf")

    warm = _warm_start(n)
    if warm is not None:
        cand_val = _objective(warm)
        best_seq, best_val = warm.copy(), cand_val
        print(f"[Warm start] value {cand_val:.6f}")

    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]
    for c in starters:
        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Starter] value {val:.6f}")

    # --------------------------------------------------------------
    # 1) Massive random warm‑starts (anti‑pair + interleave)
    n_samples = 15000
    print(f"Generating up to {n_samples} random anti‑pair candidates …")
    for _ in range(n_samples):
        if time.time() >= deadline:
            break
        log_alpha = np.random.uniform(-3.5, 2.0)   # α ∈ [10⁻³·⁵, ≈100]
        alpha = 10.0 ** log_alpha
        cand = _anti_pair_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random anti‑pair] new best {val:.6f}")

    # a few extreme‑interleave samples
    print("Generating extreme‑interleave candidates …")
    for _ in range(3000):
        if time.time() >= deadline:
            break
        log_alpha = np.random.uniform(-3.5, 2.0)
        alpha = 10.0 ** log_alpha
        cand = _extreme_interleave_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Interleave] new best {val:.6f}")

    # --------------------------------------------------------------
    # 2) Cheap parametric families
    # Logistic profiles
    for steep in (0.6, 1.0, 2.0, 3.5):
        if time.time() >= deadline:
            break
        cand = _logistic_candidate(n, steep=steep)
        # apply anti‑pair ordering to force large‑small pairing
        cand = _anti_pair_arrangement(cand)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Logistic steep={steep}] new best {val:.6f}")

    # Gaussian‑edge profiles (wider sweep)
    for gamma in (0.1, 0.3, 0.6, 1.0, 1.5):
        if time.time() >= deadline:
            break
        cand = _gaussian_edges_candidate(n, gamma=gamma)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Gauss γ={gamma}] new best {val:.6f}")

    # --------------------------------------------------------------
    # 3) L‑BFGS smooth‑max refinement (multi‑restart)
    # keep the single best candidate – it is already the best we have
    lbfgs_deadline = min(deadline - 150.0, time.time() + 250.0)
    if time.time() < lbfgs_deadline:
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n, conv_fft_len,
                                         corr_fft_len, lbfgs_deadline,
                                         lam_schedule=(5, 15, 40, 120,
                                                       300, 800),
                                         max_iter_stage=400)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # 4) Exact‑max sub‑gradient phases
    # (a) multiplicative (exponentiated‑gradient)
    sub_deadline = min(deadline - 60.0, time.time() + 300.0)
    if time.time() < sub_deadline:
        seq_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                 conv_fft_len, corr_fft_len,
                                                 sub_deadline,
                                                 max_iters=2_000_000,
                                                 eta0=0.5)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[MulSubgrad] improved to {val_sub:.6f}")

    # (b) additive projected sub‑gradient
    add_deadline = min(deadline - 30.0, time.time() + 200.0)
    if time.time() < add_deadline:
        seq_add, val_add = _additive_subgradient_optimize(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          add_deadline,
                                                          max_iters=1_200_000,
                                                          eta0=0.45)
        if val_add < best_val:
            best_val, best_seq = val_add, seq_add.copy()
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # --------------------------------------------------------------
    # 5) Local stochastic hill‑climbing
    if best_seq is not None and time.time() < deadline - 20.0:
        local_deadline = deadline - 5.0
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                conv_fft_len, local_deadline,
                                                max_moves=800_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid (8192)
    if time.time() < deadline - 10.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best sequence
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        big_deadline = min(deadline - 2.0, time.time() + 200.0)
        seq_big_lb, val_big_lb = _lbfgs_optimize(a_big, n_big,
                                                 conv_fft_len_big,
                                                 corr_fft_len_big,
                                                 big_deadline,
                                                 lam_schedule=(5, 20, 80, 200, 600),
                                                 max_iter_stage=250)
        if val_big_lb < best_val:
            best_val, best_seq = val_big_lb, seq_big_lb.copy()
            print(f"[Big LBFGS] improved to {val_big_lb:.6f}")

        # final fine‑tuning on the big grid
        final_deadline = min(deadline, time.time() + 120.0)
        if time.time() < final_deadline:
            seq_big_sub, val_big_sub = _subgradient_optimize(best_seq, n_big,
                                                             conv_fft_len_big,
                                                             corr_fft_len_big,
                                                             final_deadline,
                                                             max_iters=600_000,
                                                             eta0=0.5)
            if val_big_sub < best_val:
                best_val, best_seq = val_big_sub, seq_big_sub.copy()
                print(f"[Big Subgrad] improved to {val_big_sub:.6f}")

    # --------------------------------------------------------------
    # 7) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
